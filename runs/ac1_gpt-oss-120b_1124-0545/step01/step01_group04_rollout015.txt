def solve_linear_subproblem(a, M):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    # Build constraints matrix row-wise
    # Rather than building full dense matrix, we can generate constraints using a loop,
    # each constraint is linear in b: sum_i a[i] * b[k-i] <= M.
    constraints = []
    for k in range(2*n-1):
        # Determine overlapping indices
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # positions for b: j = k - i
        # Build expression: sum_{i=i_start}^{i_end} a[i] * b[k-i]
        expr = 0
        # vectorized: we can compute a_slice * b_slice reversed
        # However cp does not support variable indexing with slice of variable? It does: b[i_slice] works.
        # So we can write cp.sum(cp.multiply(a_slice, b_slice)) 
        a_slice = a[i_start:i_end+1]
        # reversed indices for b: j_start = k - i_end, j_end = k - i_start
        j_start = k - i_end
        j_end = k - i_start
        # b slice (must be contiguous)
        # Since slice from j_start to j_end inclusive.
        b_slice = b[j_start:j_end+1]
        expr = cp.sum(cp.multiply(a_slice, b_slice))
        constraints.append(expr <= M)
    # No constraint on sum(b) in LP.
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    if b.value is None:
        # fallback: return a
        return a.copy()
    return np.array(b.value, dtype=np.float64)

import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz

def compute_catalan_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    a /= a.sum()
    return a

def max_convolution_fft(a):
    n = len(a)
    fft_len = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    conv = conv[:2*n-1]
    return float(conv.max())

def solve_max_sum_lp(a, M):
    n = len(a)
    # Build Toeplitz matrix of a: rows 0..2n-2, columns 0..n-1
    col = np.r_[a, np.zeros(n-1)]
    row = np.r_[a[0], np.zeros(n-1)]
    A = toeplitz(col, row)[:2*n-1, :]
    # CVXPY variable
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M * np.ones(2*n-1)]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except Exception as e:
        print("LP solve exception:", e)
        return None
    if b.value is None:
        return None
    b_val = np.array(b.value, dtype=np.float64)
    # Clip small negatives that sometimes appear due to numerical errors
    b_val = np.maximum(b_val, 0.0)
    return b_val

def random_perturb(a, eps):
    n = len(a)
    i = np.random.randint(n)
    j = np.random.randint(n)
    if i == j:
        return a
    delta = eps * a[i]
    if delta <= 0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return a_new / a_new.sum()

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Frank‑Wolfe‑style optimisation for the L∞ self‑convolution problem.
    Starts from the (1‑x)^−½ coefficients, repeatedly solves a linear
    program that maximises the L1‑mass while respecting the current
    convolution bound, rescales the solution and mixes it with the current
    iterate.  Occasional random perturbations help to escape plateaus.
    The best sequence seen before the time budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose length of the step function
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore
    except Exception:
        n = 2000  # reasonable default; fits in memory

    # ------------------------------------------------------------------
    # 0. Initial point: Catalan (1‑x)^‑½  normalised to L1=1
    # ------------------------------------------------------------------
    a = compute_catalan_sequence(n)
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Main loop – alternating LP step and stochastic perturbations
    # ------------------------------------------------------------------
    iter_count = 0
    eps = 0.05  # magnitude of random moves
    t_mix = 0.5  # mixing factor for LP candidate
    while time.time() < deadline:
        iter_count += 1

        # ---- LP direction -------------------------------------------------
        M = max_convolution_fft(a)
        b = solve_max_sum_lp(a, M)
        if b is not None:
            sum_b = b.sum()
            if sum_b > 1e-8:
                # Scale b to unit L1 norm (same as a)
                b_scaled = b / sum_b
                # Mix current point with this direction
                cand = (1 - t_mix) * a + t_mix * b_scaled
                cand = np.maximum(cand, 0.0)
                cand /= cand.sum()
                cand_score = evaluate_sequence(cand.tolist())
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    a = cand
                    print(f"[lp] iter {iter_count:5d} – improved score {best_score:.6f}")
                    # after a successful LP step, we keep the larger mixing weight
                    t_mix = min(0.9, t_mix * 1.05)
                else:
                    # if LP step not improving, slowly reduce mixing strength
                    t_mix = max(0.1, t_mix * 0.95)

        # ---- occasional random perturbation --------------------------------
        if np.random.rand() < 0.2:
            cand = random_perturb(a, eps)
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                print(f"[rand] iter {iter_count:5d} – improved score {best_score:.6f}")

        # decay random‑move magnitude
        eps = max(eps * 0.999, 1e-4)

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    return final_seq.tolist()

from scipy.optimize import linprog
def solve_max_sum_lp_linprog(a, M):
    n = len(a)
    col = np.r_[a, np.zeros(n-1)]
    row = np.r_[a[0], np.zeros(n-1)]
    A = toeplitz(col, row)[:2*n-1, :]
    # linprog expects A_ub x <= b_ub
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    else:
        return None
```

def build_conv_matrix_sparse(a):
    n = len(a)

        # This diagonal has length n (for column range) but occupies rows i..i+n-1.

"""
Frank‑Wolfe style optimisation for the self‑convolution bound.

The algorithm starts from the (1−x)^‑½ (Catalan‑type) coefficients,
normalised to L1 = 1.  For a vector ``a`` we compute

    M = max_k (a * a)[k]

and then solve the linear programme

    max   sum(b_j)
    s.t.  (a * b)[k] ≤ M  for all k
          b_j ≥ 0

The LP finds a direction that can increase the total mass while keeping
the convolution bound unchanged.  Scaling the LP solution back to
L1‑norm 1 reduces the maximal convolution value by the factor
``sum(b)`` and therefore improves the evaluation metric
``2 * n * max(conv) / (sum^2)``.
We repeat this step until the time budget expires (or the LP can no
longer increase the mass).  Random perturbations are used as a fallback
to escape possible plateaus.

The returned sequence is the best one seen during the run.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def compute_catalan_sequence(n: int) -> np.ndarray:
    """
    Coefficients a_i = (2i choose i) / 4**i  (i=0…n‑1), normalised
    so that they sum to 1.  This is the power series of (1‑x)^‑½.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def max_convolution_fft(a: np.ndarray) -> float:
    """
    Return max_k (a*a)[k] using an FFT (linear convolution).
    """
    n = len(a)
    # size ≥ 2·n‑1, next power of two for speed
    fft_len = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    conv = conv[: 2 * n - 1]
    return float(conv.max())


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Construct the matrix A such that (A @ b) = a * b (full linear convolution).
    Shape of A is (2n‑1) × n.
    """
    n = len(a)
    col = np.r_[a, np.zeros(n - 1)]
    row = np.r_[a[0], np.zeros(n - 1)]
    A = toeplitz(col, row)[: 2 * n - 1, :]
    return A


def solve_max_sum_lp(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max sum(b)   s.t.   a * b ≤ M  (component‑wise) , b ≥ 0.
    Returns the optimal b as a dense ndarray, or None if the solve fails.
    """
    n = len(a)
    A = build_conv_matrix(a)
    # linear programme in the form min c^T x  with  A_ub x ≤ b_ub
    c = -np.ones(n)                     # minimise –sum(b)  ⇔ maximise sum(b)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n            # b_j ≥ 0

    # HiGHS (default for method='highs') is fast and robust for large LPs
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success or res.x is None:
        return None
    b = np.maximum(res.x, 0.0)          # clean tiny negatives
    return b


def random_perturb(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Transfer a fraction ``eps`` of the mass from a random index i to another
    random index j (i ≠ j).  The vector stays normalised.
    """
    n = len(a)
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    delta = eps * a[i]
    if delta <= 0.0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------

    """
    Return a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method uses the linear‑programming
    refinement described above, with a small random‑perturbation fallback.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ---- pick a length -------------------------------------------------
    try:
        n = len(height_sequence_1)        # use previous best length if available
    except Exception:

    # ---- initialise with the Catalan (1‑x)^‑½ sequence ------------------

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # ---- main optimisation loop ----------------------------------------
    eps = 0.05               # magnitude for occasional random moves
    it = 0
    while time.time() < deadline:
        it += 1

        # 1) current maximal self‑convolution
        M = max_convolution_fft(a)

        # 2) LP step – try to increase total mass while keeping the bound M
        b = solve_max_sum_lp(a, M)
        if b is not None:
            sum_b = b.sum()
            # a feasible improvement requires sum_b > 1 (current mass)
            if sum_b > 1.0 + 1e-12:
                # scale back to L1 = 1  (this shrinks the max‑convolution bound)
                a_candidate = b / sum_b
                cand_score = evaluate_sequence(a_candidate.tolist())
                if cand_score < best_score:

                    best_seq = a_candidate.copy()
                    a = a_candidate

            cand_score = evaluate_sequence(a_candidate.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                a = a_candidate
                print(f"[rand {it:5d}] improved score = {best_score:.6f}")

        # decay the random‑move magnitude
        eps = max(eps * 0.999, 1e-4)

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    final_seq = np.clip(best_seq, 0.0, 1000.0)   # safety clip
    return final_seq.tolist()
