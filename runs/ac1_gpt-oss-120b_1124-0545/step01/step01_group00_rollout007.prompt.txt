def compute_score(a: np.ndarray) -> float:

def gradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    Very cheap gradient‑like update based on the peak of the convolution.
    The direction d = -g + mean(g) keeps the sum unchanged (∑d = 0).
