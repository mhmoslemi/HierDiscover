"""
Propose a candidate step‑function that lowers the evaluation metric.

The algorithm combines three ideas:

1. **Good initialisations** – Catalan (almost flat convolution),
   the uniform distribution, the previous best sequence (if available) and
   a few random Dirichlet draws.
2. **Smoothed max‑convolution minimisation** – we minimise a
   soft‑max approximation of  max(a∗a) using an Adam optimiser.
   The gradient of the smoothed objective is
       ∇ fβ(a) = 2·(a * w_rev) ,
   where w is a soft‑max over the convolution values (β controls the
   smoothness).  After each step we project onto the probability simplex
   (non‑negative entries summing to 1) – this respects the scale‑invariance
   of the original metric.
3. **Final stochastic polishing** – a cheap random‑perturbation/SA loop
   that explores the neighbourhood of the best vector found.

The whole search runs for up to `budget_s` seconds (leaving a small margin)
and returns the best sequence discovered.
"""

# ----------------------------------------------------------------------
# Imports & tiny utilities
# ----------------------------------------------------------------------
import numpy as np
import time

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def compute_catalan_sequence(n: int) -> np.ndarray:
    """Catalan‑type coefficients (coefficients of (1‑x)^‑½) normalised to sum 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution of two real vectors using FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1).bit_length())        # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute a soft‑max weighting of the convolution values.
    For β→∞ this approaches the indicator of the maximal entry.
    """
    scaled = beta * conv
    m = np.max(scaled)
    exp_vals = np.exp(scaled - m)                # stabilise
    s = np.sum(exp_vals)
    if s == 0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s


def gradient_softmax_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(a∗a) w.r.t. a:
        ∇ = 2·(a * w_rev)   (restricted to the first |a| entries)
    """
    n = a.shape[0]
    L = w.shape[0]                               # = 2n‑1
    w_rev = w[::-1]
    # convolution length = n + L – 1  ( = 3n‑2 )
    conv_full = convolve_fft(a, w_rev)
    start = L - 1                                # position where a aligns with w
    grad = 2.0 * conv_full[start:start + n]
    return grad


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("project_simplex expects a 1‑D array")
    n = v.shape[0]
    # Work on a copy to avoid side‑effects
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerically unstable case – fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def random_perturb(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Transfer a fraction *eps* of the mass from a random index i to a
    distinct random index j (conserves the sum, stays non‑negative).
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = eps * a[i]
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # re‑normalise to guard against tiny round‑off errors
    a_new /= a_new.sum()
    return a_new

    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best list of heights found.
    """
    np.random.seed(seed)
    # we keep a tiny safety margin so the harness can receive the answer
    deadline = time.time() + budget_s - 0.5

    # --------------------------------------------------------------
    # 1. Determine the length of the vector we will work with
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # use the previous best if available
    except NameError:
        n = 2000                            # fallback
    if n < 500:
        n = 500                             # enforce a modest size for speed

    # --------------------------------------------------------------
    # 2. Build a collection of diverse starting points
    # --------------------------------------------------------------
    inits = []

    # Catalan (almost flat self‑convolution)
    inits.append(compute_catalan_sequence(n))

    # Uniform distribution (very flat, but not optimal)
    inits.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Previous best, if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.shape[0] != n:
            # truncate or pad with zeros to match the working length
            if prev.shape[0] > n:
                prev = prev[:n]
            else:
                prev = np.pad(prev, (0, n - prev.shape[0]))
        inits.append(prev / prev.sum())
    except NameError:
        pass

    # Random Dirichlet draws – low concentration (spiky) and moderate
    for alpha in (0.3, 0.7):
        rnd = np.random.dirichlet(np.full(n, alpha))
        inits.append(rnd)

    # Simple triangular shape (peaked in the centre)
    tri = np.arange(1, n // 2 + 1, dtype=np.float64)
    if n % 2 == 1:
        tri = np.concatenate((tri, tri[-2::-1]))
    else:
        tri = np.concatenate((tri, tri[::-1]))
    tri = tri[:n]
    tri /= tri.sum()
    inits.append(tri)

    # --------------------------------------------------------------
    # 3. Evaluate the raw initial vectors and keep the best one
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for idx, a0 in enumerate(inits):
        a0 = np.asarray(a0, dtype=np.float64)
        if a0.sum() == 0:
            a0 = np.full_like(a0, 1.0 / n)
        else:
            a0 /= a0.sum()
        cur = evaluate_sequence(a0.tolist())
        if cur < best_score:
            best_score = cur
            best_seq = a0.copy()
        print(f"[init {idx+1:02d}] score = {cur:.6f}")

    # --------------------------------------------------------------
    # 4. Smoothed max‑convolution minimisation (Adam + projection)
    # --------------------------------------------------------------
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.12                    # base learning rate (before Adam scaling)

    # Soft‑max temperature schedule
    beta_start = 1.0
    beta_max = 5000.0
    beta_factor = 1.015               # geometric growth per iteration

    # We iterate over the initial vectors one after another.
    # Each run gets as many iterations as the remaining time allows.
    for run_id, a0 in enumerate(inits, start=1):
        if time.time() >= deadline:
            break

        a = a0.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta_val = beta_start
        it = 0

        # Fixed maximum number of inner iterations; we also bail out when time is low.
        while time.time() < deadline:
            # --------------------------------------------------
            # (a) compute self‑convolution and its soft‑max weights
            # --------------------------------------------------
            conv_self = convolve_fft(a, a)                     # length 2n‑1
            w = softmax_weights(conv_self, beta_val)          # same length

            # --------------------------------------------------
            # (b) gradient of the smooth objective
            # --------------------------------------------------
            grad = gradient_softmax_max(a, w)

            # --------------------------------------------------
            # (c) Adam update + projection onto the simplex
            # --------------------------------------------------
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            a_new = a - step
            a = project_simplex(a_new)                         # keep non‑negative, sum = 1

            # --------------------------------------------------
            # (d) slowly increase the soft‑max temperature
            # --------------------------------------------------
            if beta_val < beta_max:
                beta_val = min(beta_val * beta_factor, beta_max)

            # --------------------------------------------------
            # (e) occasional evaluation
            # --------------------------------------------------
            if it % 80 == 0:
                cur_score = evaluate_sequence(a.tolist())
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[run {run_id:02d} it {it:05d}] new best = {best_score:.6f}")
                # tiny early‑stop if we are already extremely good
                if best_score < 1.50:
                    print("[early exit] target reached")
                    break

            # --------------------------------------------------
            # (f) crude convergence test – stop if step is tiny
            # --------------------------------------------------
            if np.linalg.norm(step) < 1e-12:
                break

        # End of one restart
        if best_score < 1.50:
            break

    # --------------------------------------------------------------
    # 5. Final polishing: cheap stochastic local search (SA)
    # --------------------------------------------------------------
    a = best_seq.copy()
    eps = 0.07               # initial magnitude of a perturbation
    temperature = 0.07
    while time.time() < deadline:
        cand = random_perturb(a, eps)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand
            print(f"[SA] improved score = {best_score:.6f}")
        else:
            # uphill acceptance (simulated annealing)
            prob = np.exp(-(cand_score - best_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                a = cand

        # decay move size & temperature
        eps = max(eps * 0.9997, 1e-5)
        temperature *= 0.9995

    # --------------------------------------------------------------
    # 6. Return the best sequence (as plain Python list)
    # --------------------------------------------------------------
    final_seq = np.clip(best_seq, 0.0, 1000.0)   # final safety clamp
    print(f"Final best score = {best_score:.6f}")
    return final_seq.tolist()
