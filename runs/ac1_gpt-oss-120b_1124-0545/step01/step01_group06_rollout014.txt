"""
Propose a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

where ``conv`` is the linear convolution.  The implementation works in three
stages:

1.  **Broad parametric sweep** – a few hundred candidates from simple
    families (geometric, power–law, their products and symmetric versions).
    The sweep also explores a wide range of decay factors (r down to 0.05)
    and power exponents (p up to 2.5).  Every candidate is normalised to
    ``sum(a)=1`` and evaluated with an FFT‑based convolution.

2.  **Adam‑based multiplicative optimisation** – starting from the best
    candidate we optimise a smooth approximation of the ``max`` using the
    log‑sum‑exp trick.  The parameters are kept in the *log‑softmax* domain,
    which automatically enforces non‑negativity and the simplex constraint.
    An Adam optimiser (first‑order momentum) together with a geometric schedule
    for the smoothing temperature yields a faster and more stable descent
    than the plain exponentiated‑gradient used previously.

3.  **Final simulated‑annealing fine‑tuning** – a cheap Metropolis step
    perturbs the current best sequence by moving a small amount of mass
    between two random positions.  The move is accepted if it improves the
    objective, otherwise it is accepted with probability ``exp(-Δ/T)``.
    The temperature is slowly cooled to allow the search to escape shallow
    local minima.

The whole routine respects the supplied time budget (default 1000 s) and
returns the best sequence found (plain ``list[float]``).  It can be called
repeatedly – the function is self‑contained and does not rely on any global
state except the optional ``height_sequence_1`` variable that may already
contain a good construction.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (FFT convolution, projection, simple families)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _evaluate_norm1(a: np.ndarray) -> float:
    """Objective for a vector already normalised to sum == 1:
    2 * n * max(conv(a, a)).
    """
    conv = _conv_fft(a, a)
    return float(2.0 * len(a) * np.max(conv))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric progression a_i ∝ r**i."""
    if r <= 0.0:
        return np.ones(n) / n
    a = r ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric: a_i ∝ r**i + r**(n-1-i)."""
    if r <= 0.0:
        return np.ones(n) / n
    a = r ** np.arange(n, dtype=np.float64) + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    a /= a.sum()
    return a


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a /= a.sum()
    return a


def _sym_power(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^(-p) + (n-i)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p) + (n - idx + 1) ** (-p)
    a /= a.sum()
    return a


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power‑law a_i ∝ r**i / (i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    a /= a.sum()
    return a


def _sym_geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Symmetric combination of geometric and power‑law."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    left = (r ** (idx - 1)) / (idx ** p)
    right = (r ** (n - idx)) / ((n - idx + 1) ** p)
    a = left + right
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Piecewise linear “tent” shape – rises then falls."""
    if n % 2 == 0:
        half = n // 2
        inc = np.arange(1, half + 1, dtype=np.float64)
        dec = inc[::-1]
        a = np.concatenate([inc, dec])
    else:
        half = n // 2
        inc = np.arange(1, half + 2, dtype=np.float64)
        dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation (Adam on the softmax parameters)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, budget: float) -> np.ndarray:
    """
    Adam optimiser in the log‑softmax space.
    Returns the best vector (still normalised to sum 1) found within the
    supplied time budget (seconds).
    """
    n = len(a0)
    # initialise logits = log(a)  (add eps to avoid log(0))
    eps = 1e-20
    logits = np.log(np.maximum(a0, eps))

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    lr = 0.05                        # base learning rate
    eps_adam = 1e-8

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    # temperature schedule for the smooth‑max (τ = 1/β)
    tau0 = 1.0
    tau_mul = 0.995  # slowly cool down
    tau = tau0

    best_a = a0.copy()
    best_val = _evaluate_norm1(best_a)

    deadline = time.time() + budget
    t = 0
    while time.time() < deadline:
        t += 1
        # current probabilities
        a = np.exp(logits - np.max(logits))   # stabilise
        a /= a.sum()

        # convolution and smooth max weights
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            # occasional progress report
            print(f"[adam] iter {t:5d} new best {cur_val:.6f}")

        # smoothed max (log‑sum‑exp) with temperature tau
        shift = cur_max                     # improve numerical stability
        exp_arg = (conv - shift) / max(tau, 1e-12)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            # fallback to hard max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient wrt a (ordered‑pair derivative)
        grad_full = _conv_fft(a[::-1], w)   # length = 2n‑1
        grad_a = 2.0 * grad_full[n - 1: n - 1 + n]   # slice of length n

        # chain rule through the softmax
        dot = np.dot(a, grad_a)
        grad_log = a * (grad_a - dot)

        # Adam update of logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # cool the temperature
        tau *= tau_mul
        if tau < 1e-4:
            tau = 1e-4

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(a0: np.ndarray, budget: float) -> np.ndarray:
    """
    Simple Metropolis search that moves a tiny amount of mass between two
    random positions.  Used as a polishing step after gradient descent.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_val = _evaluate_norm1(best_a)

    deadline = time.time() + budget
    rng = np.random.default_rng()
    temperature = 0.01
    decay = 0.9995

    it = 0
    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount to transfer (up to a fraction of the source)
        delta = rng.random() * min(a[i], 0.1 / n)
        if delta == 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        # evaluate
        val = _evaluate_norm1(a)
        accept = False
        if val < best_val:
            best_val, best_a = val, a.copy()
            accept = True
            print(f"[anneal] iter {it:6d} new best {val:.6f}")
        else:
            # Metropolis criterion
            if rng.random() < np.exp(-(val - best_val) / temperature):
                accept = True

        if not accept:
            # revert move
            a[i] += delta
            a[j] -= delta

        # annealing schedule
        temperature *= decay
        if temperature < 1e-6:
            temperature = 1e-6

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    best_seq = None

    lengths = [2000, 3000, 4000, 5000]          # moderate sizes

    # Return plain list as required by the evaluator
