def propose_candidate(seed=42, budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.1
    rng = np.random.default_rng(seed)
    best_seq = None
    best_score = float('inf')
    # define update_best(seq)
    def update_best(seq):
        nonlocal best_seq, best_score
        # Need to ensure list items are normalised (but evaluate will clip anyway)
        # Convert to list, as evaluate expects.
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time() - start:.1f}s] New global best: {best_score:.6f}")
    # Warm-start from existing global
    hs1 = globals().get('height_sequence_1')
    if hs1 is not None:
        seq0 = np.array(hs1, dtype=float)
        seq0 = _normalise_sum(seq0)  # sum=1
        update_best(seq0.tolist())
    # Phase 1: param sweep
    lengths = [256, 512, 1024, 2048]
    # Keep top seeds list
    top_seeds = []
    top_k = 5
    for n in lengths:
        # generate candidates and evaluate
        candidates = []
        # constant shape
        candidates.append(np.ones(n))
        # linear decreasing
        candidates.append(np.arange(n, 0, -1))
        # linear increasing
        candidates.append(np.arange(1, n+1))
        # exponential random
        for _ in range(20):
            r = rng.uniform(0.6, 0.99)
            candidates.append(r ** np.arange(n))
            candidates.append((1-r) ** np.arange(n))  # also decaying
        # power law random
        for _ in range(20):
            p = rng.uniform(0.2, 3.0)
            i = np.arange(1, n+1)
            candidates.append(1.0 / i**p)
        # logistic random
        for _ in range(10):
            mid = rng.uniform(0, n)
            scale = rng.uniform(0.1, n/2)
            x = np.arange(n)
            cand = 1.0 / (1.0 + np.exp((x - mid) / scale))
            candidates.append(cand)
        # double exponential mixture
        for _ in range(10):
            r1 = rng.uniform(0.6, 0.98)
            r2 = rng.uniform(0.6, 0.98)
            w = rng.uniform(0, 1)
            cand = w * (r1 ** np.arange(n)) + (1-w) * (r2 ** np.arange(n))
            candidates.append(cand)
        # triangular
        t = np.minimum(np.arange(1, n+1), n - np.arange(n) )
        candidates.append(t)
        # valley
        centre = (n-1)/2
        cand = 1.0 / (1.0 + (np.arange(n) - centre)**2)
        candidates.append(cand)
        # evaluate each candidate after normalisation
        for cand in candidates:
            if time.time() >= deadline:
                break
            seq = _normalise_sum(cand)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"[{time.time() - start:.1f}s] New best (len={n}) score={score:.6f}")
            # store top seeds
            if len(top_seeds) < top_k:
                top_seeds.append((score, seq.copy()))
                top_seeds.sort(key=lambda x: x[0])
            else:
                # keep if better than worst in top_k
                if score < top_seeds[-1][0]:
                    top_seeds[-1] = (score, seq.copy())
                    top_seeds.sort(key=lambda x: x[0])

    # If still no best_seq, fallback random
    if best_seq is None:
        seq = rng.random(256)
        seq = _normalise_sum(seq)
        best_seq = seq.tolist()
        best_score = evaluate_sequence(best_seq)
        top_seeds.append((best_score, np.array(best_seq)))

    # Phase 2: refinement with guided annealing on each top seed
    # Determine target length: we choose 8192 as max if time permits.
    target_len = 8192
    # Upcale each seed to target length if smaller
    refined_seeds = []
    for _, seed_seq in top_seeds:
        if len(seed_seq) != target_len:
            # linear interpolation
            x_old = np.linspace(0, 1, len(seed_seq))
            x_new = np.linspace(0, 1, target_len)
            up = np.interp(x_new, x_old, seed_seq)
        else:
            up = seed_seq.copy()
        up = _normalise_sum(up)
        refined_seeds.append(up)

    # Now run guided annealing on each refined seed till deadline
    for start_seq in refined_seeds:

            best_seq = cur_best_seq.tolist()
            print(f"[{time.time() - start:.1f}s] Global improved after anneal: {best_score:.6f}")

    # Final output
    print(f"Finished search. Best score {best_score:.6f} length {len(best_seq)}")
    return best_seq

def _normalise_sum(arr, target_sum=1.0):
    a = np.array(arr, dtype=float)
    total = a.sum()
    if total <= 0:
        a = np.full_like(a, target_sum / a.size)
    else:
        a = a * (target_sum / total)
    # Clip max to 0.9
    max_val = a.max()
    if max_val > 0.9:
        a = a * (0.9 / max_val)
    # ensure sum still >= minimal; renormalise
    s = a.sum()
    if s < 0.01:
        a = a + (0.01 - s) / a.size
    return a
```

def _conv_max(a):
    n = len(a)
    # compute convolution using FFT of size power of two >= 2n
    size = 1 << ((2*n-1).bit_length())

    max_b = conv.max()
    k_max = conv.argmax()

            t = rng.choice(n, p=w / w.sum())
            if t == i:
                t = (i + 1) % n
            new = cur.copy()
            new[i] = max(0.0, new[i] - eps)
            new[t] = new[t] + eps
        else:
            start_idx = rng.integers(0, n)
            block_len = rng.integers(1, min(200, n - start_idx) + 1)
            new = cur.copy()
            new[start_idx:start_idx+block_len] = new[start_idx:start_idx+block_len][::-1]
        # fast score
        new_score = _fast_score(new)
        # temperature schedule
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = 1.0 * (1 - elapsed / total) + 0.001 * (elapsed / total)  # linear decay
        if new_score < best_score_fast - 1e-12:
            # improvement
            cur = new
            cur_score = new_score
            best_score_fast = new_score
            # evaluate true score
            true_score = evaluate_sequence(new.tolist())
            if true_score < best_score_true:
                best_score_true = true_score
                best_seq = new.copy()
                # also update global if needed
                if true_score < global_best_score:
                    print(f"[{time.time() - start_time:.1f}s] *** NEW GLOBAL BEST {true_score:.6f}")
                    global_best_score = true_score
            # continue
        else:
            # accept with probability exp(-(new_score - cur_score) / T)
            # using cur_score as reference (fast score)
            delta = new_score - cur_score
            if delta < 0:
                prob = 1.0
            else:
                prob = np.exp(-delta / max(T, 1e-12))
            if rng.random() < prob:
                cur = new
                cur_score = new_score
        # optionally early exit if best_score_true <= 1.503 (target)
        if best_score_true <= 1.503:
            break
    return best_seq, best_score_true

"""
Propose a step‑function sequence that minimises `evaluate_sequence`.

Algorithm
---------
1. **Parameter sweep** – generate many analytic shapes (exponential,
   power‑law, logistic, mixtures, triangular/valley etc.) for several
   lengths (256 … 2048).  Each shape is normalised to a unit sum and
   evaluated with the given `evaluate_sequence`.  The best few candidates
   are kept for further refinement.

2. **Upscaling** – the selected seeds are interpolated to the maximal
   length (8192) because longer sequences give a larger “attack surface”.

3. **Guided simulated annealing** – on each up‑scaled seed a fast
   optimisation loop is run.  In every iteration the convolution
   `b = a * a` (computed by FFT) is inspected; the shift `k_max` where the
   maximum coefficient occurs is identified and the contribution
   `c_i = a_i * a_{k_max-i}` of each entry to that maximum is computed.
   Moves are bias‑sampled according to `c_i`:
   * move a small amount of mass from a high‑contribution index to a
     low‑contribution one,
   * optionally reverse a short block,
   * or perform a small multiplicative gradient step that reduces the
     dominant contributions.
   The acceptance follows a linear temperature schedule; only the fast
   FFT‑based score is used to decide, while the true score is evaluated
   (via `evaluate_sequence`) only when an improvement is found.

4. **Global bookkeeping** – the best sequence seen over the whole run is
   stored and returned.  The routine respects the provided time budget
   (≈1000 s) and never returns an empty list.

All helper functions are defined at module level so that the evaluator can
import `propose_candidate` directly.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _normalise_sum(arr: np.ndarray, target_sum: float = 1.0) -> np.ndarray:
    """
    Scale ``arr`` so that its sum equals ``target_sum``.
    The maximum entry is capped at 0.9 (well below the 1000 clip limit)
    and a minimal total of 0.01 is enforced.
    """
    a = np.array(arr, dtype=float, copy=True)
    s = a.sum()
    if s > 0:
        a *= target_sum / s
    else:
        a[:] = target_sum / a.size
    # keep values comfortably below the 1000‐clip
    m = a.max()
    if m > 0.9:
        a *= 0.9 / m
    # guard against the “sum too close to zero” case
    s = a.sum()
    if s < 0.01:
        a += (0.01 - s) / a.size
    return a


def _conv_max(a: np.ndarray):
    """
    Return ``max_b``, the index ``k_max`` where the maximum of the
    self‑convolution occurs and the whole convolution array.
    FFT is used for speed (O(n log n)).
    """
    n = a.shape[0]
    # next power of two for a*a (length 2n-1)
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    # numerical noise may produce tiny negative numbers – clamp to 0
    conv = np.maximum(conv, 0.0)
    k_max = int(conv.argmax())
    max_b = float(conv[k_max])
    return max_b, k_max, conv


def _fast_score(a: np.ndarray) -> float:
    """
    Compute the evaluation function using the fast FFT based convolution.
    This mirrors ``evaluate_sequence`` for arrays whose entries are <1000.
    """
    max_b, _, _ = _conv_max(a)
    s = a.sum()
    return 2.0 * a.shape[0] * max_b / (s * s)


# ----------------------------------------------------------------------
# Guided annealing on a single seed
# ----------------------------------------------------------------------


def _guided_anneal(
    start_seq: np.ndarray,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
    global_best_score: float,
):
    """
    Perform a temperature‑controlled local search starting from ``start_seq``.
    The function stops when the current time exceeds ``deadline``.
    Returns the best sequence found (as ``np.ndarray``) and its *true*
    ``evaluate_sequence`` score.
    """
    cur = start_seq.copy()
    cur_score_fast = _fast_score(cur)
    best_seq = cur.copy()
    best_score_fast = cur_score_fast
    best_score_true = float("inf")  # will be filled on first improvement

    # initial true evaluation (costly) – only if we haven't done it yet
    try:
        # ``evaluate_sequence`` may not be imported in the sandbox,
        # but the surrounding harness guarantees its presence.
        best_score_true = evaluate_sequence(best_seq.tolist())
        if best_score_true < global_best_score:
            print(
                f"[{time.time() - start_time:.1f}s] *** NEW GLOBAL BEST (init) {best_score_true:.6f}"
            )
            global_best_score = best_score_true
    except Exception:
        pass  # safety – ignore, the loop will improve later

    it = 0
    n = cur.shape[0]

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1. analyse the dominant convolution coefficient
        # --------------------------------------------------------------
        max_b, k_max, _ = _conv_max(cur)

        # contributions of each entry to the max coefficient
        contrib = np.zeros_like(cur)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n, k_max + 1)
        idx = np.arange(i_start, i_end)
        contrib[idx] = cur[idx] * cur[k_max - idx]

        # --------------------------------------------------------------
        # 2. propose a move (biased by contributions)
        # --------------------------------------------------------------
        move_type = rng.random()
        if move_type < 0.60:
            # --- weighted mass transfer ---------------------------------
            total = contrib.sum()
            if total > 0:
                probs = contrib / total
                i = rng.choice(n, p=probs)
            else:
                i = rng.integers(0, n)

            # epsilon is a small fraction of the source entry
            eps = rng.random() * cur[i] * 0.15
            # target index – favour low contribution locations
            inv = 1.0 / (contrib + 1e-12)
            t = rng.choice(n, p=inv / inv.sum())
            if t == i:
                t = (i + 1) % n
            cand = cur.copy()
            cand[i] = max(0.0, cand[i] - eps)
            cand[t] += eps

        elif move_type < 0.80:
            # --- block reversal ----------------------------------------
            start_idx = rng.integers(0, n)
            block_len = rng.integers(1, min(200, n - start_idx) + 1)
            cand = cur.copy()
            cand[start_idx : start_idx + block_len] = cand[
                start_idx : start_idx + block_len
            ][::-1]

        else:
            # --- tiny gradient descent step -----------------------------
            # gradient of log( max_b ) - 2*log(sum) w.r.t. a_i (ignoring
            # the shift change) is  a_{k_max-i}/max_b - 2/sum.
            rev = np.zeros_like(cur)
            rev[i_start:i_end] = cur[k_max - idx]
            grad = rev / max_b - 2.0 / cur.sum()
            # multiplicative update, then re‑normalise
            lr = 0.04
            cand = cur * np.exp(-lr * grad)
            # re‑normalise to keep the sum constant
            cand *= cur.sum() / cand.sum()

        # --------------------------------------------------------------
        # 3. evaluate the candidate (fast) and decide acceptance
        # --------------------------------------------------------------
        cand_score_fast = _fast_score(cand)

        # temperature linear from 1.0 → 0.001
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = 1.0 * (1.0 - elapsed / total) + 0.001 * (elapsed / total)

        improved = cand_score_fast < best_score_fast - 1e-12
        if improved:
            # accept unconditionally and evaluate the true score
            cur, cur_score_fast = cand, cand_score_fast
            best_seq, best_score_fast = cand.copy(), cand_score_fast
            true_score = evaluate_sequence(cand.tolist())
            if true_score < best_score_true:
                best_score_true = true_score
                # report a new global best if applicable
                if true_score < global_best_score:
                    print(
                        f"[{time.time() - start_time:.1f}s] *** NEW GLOBAL BEST {true_score:.6f}"
                    )
                    global_best_score = true_score
                if true_score <= 1.503:
                    # target reached – early exit
                    break
        else:
            # simulated‑annealing acceptance of worse moves
            delta = cand_score_fast - cur_score_fast
            prob = np.exp(-delta / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_score_fast = cand, cand_score_fast

        # occasional pure gradient boost (to avoid stagnation)
        if it % 200 == 0:
            # recompute with current k_max (may have changed)
            max_b, k_max, _ = _conv_max(cur)
            i_start = max(0, k_max - (n - 1))
            i_end = min(n, k_max + 1)
            idx = np.arange(i_start, i_end)
            rev = np.zeros_like(cur)
            rev[i_start:i_end] = cur[k_max - idx]
            grad = rev / max_b - 2.0 / cur.sum()
            lr = 0.03
            cur = cur * np.exp(-lr * grad)
            cur *= cur.sum() / cur.sum()  # re‑normalise (sum unchanged)

    return best_seq, best_score_true

# Main entry point

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The method consists of:

    * a quick systematic scan of many analytic families,
    * up‑scaling the most promising shapes to length 8192,
    * a guided simulated annealing that preferentially modifies entries
      contributing to the dominant convolution coefficient.

    The function respects the given time budget and returns the best
    sequence found before the deadline.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to update the global best using the official evaluator
    # ------------------------------------------------------------------
    def _update_global(seq_arr):
        nonlocal best_seq, best_score
        score = evaluate_sequence(seq_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq_arr.tolist()
            print(
                f"[{time.time() - start_time:.1f}s] New global best score={best_score:.6f}"
            )
        return score

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        arr = _normalise_sum(np.asarray(hs1, dtype=float))
        _update_global(arr)

    # ------------------------------------------------------------------
    # Phase 1 – parameter sweep over analytic families
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]
    top_seeds = []          # store (score, ndarray)
    TOP_K = 5               # keep the best K seeds for refinement

    for n in lengths:
        if time.time() >= deadline:
            break

        # a compact list of candidate generators
        candidates = []

        # 1. constant
        candidates.append(np.ones(n))

        # 2. linear decreasing / increasing
        candidates.append(np.arange(n, 0, -1, dtype=float))
        candidates.append(np.arange(1, n + 1, dtype=float))

        # 3. exponential families (random bases)
        for _ in range(20):
            r = rng.uniform(0.6, 0.99)
            candidates.append(r ** np.arange(n, dtype=float))
            candidates.append((1.0 - r) ** np.arange(n, dtype=float))

        # 4. power‑law families
        for _ in range(20):
            p = rng.uniform(0.2, 3.0)
            i = np.arange(1, n + 1, dtype=float)
            candidates.append(1.0 / (i ** p))

        # 5. logistic / sigmoid shapes
        for _ in range(10):
            mid = rng.uniform(0, n)
            scale = rng.uniform(0.1, n / 2.0)
            x = np.arange(n, dtype=float)
            cand = 1.0 / (1.0 + np.exp((x - mid) / scale))
            candidates.append(cand)

        # 6. mixture of two exponentials
        for _ in range(10):
            r1 = rng.uniform(0.6, 0.98)
            r2 = rng.uniform(0.6, 0.98)
            w = rng.random()
            cand = w * (r1 ** np.arange(n, dtype=float)) + (1.0 - w) * (
                r2 ** np.arange(n, dtype=float)
            )
            candidates.append(cand)

        # 7. triangular “mountain”
        t = np.minimum(np.arange(1, n + 1, dtype=float), n - np.arange(n, dtype=float))
        candidates.append(t)

        # 8. valley (large at ends, small centre)
        centre = (n - 1) / 2.0
        cand = 1.0 / (1.0 + (np.arange(n, dtype=float) - centre) ** 2)
        candidates.append(cand)

        # ------------------------------------------------------------------
        # evaluate all generated candidates for the current length
        # ------------------------------------------------------------------
        for cand in candidates:
            if time.time() >= deadline:
                break
            seq = _normalise_sum(cand)
            score = _fast_score(seq)  # cheap fast score for filtering
            # only call the expensive official evaluator when we think it's promising
            if score < best_score:
                _update_global(seq)
                # store for later refinement
                if len(top_seeds) < TOP_K:
                    top_seeds.append((score, seq.copy()))
                    top_seeds.sort(key=lambda x: x[0])
                elif score < top_seeds[-1][0]:
                    top_seeds[-1] = (score, seq.copy())
                    top_seeds.sort(key=lambda x: x[0])
            # early exit if we already hit the target
            if best_score <= 1.503:
                break

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Ensure we have at least one seed (fallback to random)
    # ------------------------------------------------------------------
    if not top_seeds:
        rnd = _normalise_sum(rng.random(256))
        top_seeds.append((_fast_score(rnd), rnd))

    # ------------------------------------------------------------------
    # Phase 2 – upscale to the maximal length and run guided annealing
    # ------------------------------------------------------------------
    TARGET_LEN = 8192
    for _, seed_seq in top_seeds:
        if time.time() >= deadline:
            break

        # upscale by linear interpolation (preserves shape)
        if seed_seq.shape[0] != TARGET_LEN:
            x_old = np.linspace(0.0, 1.0, seed_seq.shape[0])
            x_new = np.linspace(0.0, 1.0, TARGET_LEN)
            up_seq = np.interp(x_new, x_old, seed_seq)
        else:
            up_seq = seed_seq.copy()

        up_seq = _normalise_sum(up_seq)

        # run the guided annealing on this seed
        seq_opt, true_score = _guided_anneal(
            up_seq, deadline, start_time, rng, best_score
        )
        # update the global best if this annealed seed improved
        if true_score < best_score:
            best_score = true_score
            best_seq = seq_opt.tolist()
            print(
                f"[{time.time() - start_time:.1f}s] Global best after anneal: {best_score:.6f}"
            )
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Final safety: if nothing was found (should never happen), return a
    # random feasible sequence.
    # ------------------------------------------------------------------
    if best_seq is None:
        seq = _normalise_sum(rng.random(256))
        best_seq = seq.tolist()
        best_score = evaluate_sequence(best_seq)

    print(
        f"Search finished – best score {best_score:.6f} (length {len(best_seq)})"
    )
    return best_seq
