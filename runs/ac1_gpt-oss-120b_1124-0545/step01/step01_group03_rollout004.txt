def build_convolution_matrix(a):
    n = a.shape[0]

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0
    n = 2000
    a = binomial_sequence(n)
    # Scale sum to sqrt(2*n) for consistency (optional)
    target_sum = np.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())
    best_seq = a.copy()
    best_val = fast_score(best_seq)
    print(f"[init] n={n}, score={best_val:.6f}")

    iteration = 0
    max_iter = 15
    while time.time() < deadline and iteration < max_iter:
        iteration += 1

        M = conv_max(a)
        # Build LP
        A = build_convolution_matrix(a)
        b = np.full(2 * n - 1, M, dtype=np.float64)

            break
        g = res.x
        sum_g = g.sum()
        sum_a = a.sum()
        if sum_g <= sum_a * (1.0 + 1e-12):
            print(f"[LP] iteration {iteration}: no sum improvement ({sum_g:.6f})")
            break

        scale = sum_a / sum_g
        g_scaled = g * scale
        # try blend factors
        best_local_seq = a
        best_local_score = best_val
        for t in [0.1, 0.3, 0.5, 0.7]:
            new_seq = (1.0 - t) * a + t * g_scaled
            # ensure nonnegative
            if np.any(new_seq < 0):
                continue
            new_score = fast_score(new_seq)
            if new_score < best_local_score:
                best_local_score = new_score
                best_local_seq = new_seq.copy()
                print(f"[mix] iter {iteration}, t={t:.2f}, new_score={new_score:.6f}")
        # update a if improved
        if best_local_score < best_val - 1e-12:
            a = best_local_seq
            best_seq = a.copy()
            best_val = best_local_score
            # continue, maybe more improvement
        else:
            # if no blending improved, break
            print(f"[mix] iteration {iteration}: no improvement")
            break
    # final hill climb (optional)
    attempts = 0
    max_attempts = 200_000
    while time.time() < deadline and attempts < max_attempts:
        attempts += 1
        i = np.random.randint(n)
        if a[i] <= 1e-12:

        if i==j:

        delta = 0.02 * a[i]   # 2% move

        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < -1e-12:

"""
Propose a candidate step‑function sequence.

The base construction is the truncated central‑binomial (arcsine) sequence,
which gives the known bound π/2 ≈ 1.571.  To push the bound lower we
iteratively solve a linear programme that maximises the total mass of a new
sequence `g` while keeping the cross‑convolution `a*g` bounded by the
current maximal auto‑convolution `M = max(a*a)`.  The optimum `g` shifts mass
from the high‑overlap region (the beginning of the sequence) towards the
tail.  By rescaling `g` to the original ℓ₁‑norm and convex‑mixing it with
`a` we strictly reduce `M` while preserving the sum, thus improving the
objective

    2·n·max(a*a) / (sum(a)^2).

The LP is solved with SciPy’s HiGHS interior‑point method using a sparse
Toeplitz‐matrix representation of the convolution constraints.  After a few
LP updates we finish with a light stochastic hill‑climb that moves tiny
fractions of mass between indices while the objective improves.

The whole routine respects the 1000 s time budget, prints progress and
returns the best sequence as a plain Python list of non‑negative floats.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper 1: truncated central‑binomial sequence a_i = C(2i,i) / 4**i
# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """Return an array a of size `length` with
       a[i] = binom(2*i,i) / 4**i .
    """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Helper 2: fast convolution maximum (uses FFT)
# ----------------------------------------------------------------------
def conv_max(seq: np.ndarray) -> float:
    """Maximum value of the discrete self‑convolution seq*seq."""
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Helper 3: fast objective (identical to evaluate_sequence but without
# security checks).  Returns the value that the competition scores.
# ----------------------------------------------------------------------
def fast_score(seq: np.ndarray) -> float:
    """2·n·max(conv) / (sum(seq)**2) – the exact objective."""
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    max_b = float(conv.max())
    sum_a = float(seq.sum())
    if sum_a < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Helper 4: build the sparse Toeplitz‑matrix that encodes the constraints
#            (a * g)[k] <= M  for all k.
# ----------------------------------------------------------------------
def build_convolution_matrix(a: np.ndarray) -> sp.csr_matrix:
    """Return a sparse matrix A such that A @ g = a * g (full convolution)."""
    n = a.shape[0]
    offsets = np.arange(n, dtype=np.int32)             # 0 … n‑1
    # each diagonal receives the whole vector a (length n)
    A = sp.diags([a] * n, offsets, shape=(2 * n - 1, n), format="csr")
    return A


# ----------------------------------------------------------------------
# Main entry point required by the framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) beats the
       π/2 bound.  The algorithm consists of:

       1. Initialise a scaled binomial sequence.
       2. Repeatedly:
          – solve a linear programme that maximises ℓ₁‑norm under the
            current convolution bound,
          – rescale the LP solution to the original ℓ₁‑norm,
          – convex‑mix it with the current sequence, keeping the bound
            strictly decreasing.
       3. A final cheap stochastic hill‑climb.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1) Initialise – length 2000 gives a good trade‑off between speed and
    #    quality.  We normalise the ℓ₁‑norm to sqrt(2·n) (the normalisation
    #    used in the reference paper).  Scaling does not affect the final
    #    ratio, but it keeps numbers in a comfortable range.
    # ------------------------------------------------------------------
    n = 2000
    a = binomial_sequence(n)

    target_sum = np.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())

    best_seq = a.copy()
    best_val = fast_score(best_seq)
    print(f"[init] n={n}, initial score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) LP‑based improvement loop.
    # ------------------------------------------------------------------
    iteration = 0
    max_lp_iters = 12                     # enough to see a trend, cheap
    while time.time() < deadline and iteration < max_lp_iters:
        iteration += 1

        M = conv_max(a)                    # current L∞‑convolution bound
        # Build sparse constraint matrix A such that (A @ g)[k] = (a*g)[k]
        A = build_convolution_matrix(a)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # Linear programme: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # Solve – HiGHS handles sparse matrices efficiently.
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(deadline - time.time(), 1.0)},
        )

        if not res.success:
            print(f"[LP] iteration {iteration} failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        sum_a = a.sum()

        # If the LP cannot increase the ℓ₁‑norm we are stuck.
        if sum_g <= sum_a * (1.0 + 1e-12):
            print(f"[LP] iteration {iteration}: no ℓ₁‑gain (sum_g={sum_g:.6f})")
            break

        # Rescale g to keep the ℓ₁‑norm unchanged (important for the
        # objective; the bound M will still drop after mixing).
        scale = sum_a / sum_g
        g_scaled = g * scale

        # Try a handful of mixing ratios and keep the best improvement.
        improved = False
        best_local_seq = a
        best_local_score = best_val

        for t in (0.1, 0.25, 0.4, 0.55, 0.7):
            new_seq = (1.0 - t) * a + t * g_scaled
            # Numerical safety: tiny negative values can appear because of
            # rounding – clip them to zero.
            np.clip(new_seq, 0.0, None, out=new_seq)

            new_score = fast_score(new_seq)

            if new_score < best_local_score - 1e-10:
                best_local_score = new_score
                best_local_seq = new_seq.copy()
                improved = True
                print(
                    f"[mix] iter {iteration:2d}, t={t:.2f}, "
                    f"new score={new_score:.6f}"
                )

        if improved:
            a = best_local_seq
            best_seq = a.copy()
            best_val = best_local_score
        else:
            print(f"[mix] iteration {iteration}: no improvement")
            break

    # ------------------------------------------------------------------
    # 3) Light stochastic hill‑climb – uses the cheap FFT scorer.
    # ------------------------------------------------------------------
    attempts = 0
    max_attempts = 150_000
    while time.time() < deadline and attempts < max_attempts:
        attempts += 1
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move a modest fraction (2 %) from i to j
        delta = 0.02 * a[i]
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        if new_a[i] < -1e-12:          # sanity check
            continue

        new_score = fast_score(new_a)
        if new_score < best_val:
            a = new_a
            best_seq = new_a.copy()
            best_val = new_score
            print(f"[hill] iter={attempts:6d}, score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Final official evaluation.
    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    try:
        official = evaluate_sequence(final_list)   # provided by the platform
        print(f"[final] official evaluation = {official:.6f}")
    except Exception as exc:  # defensive – should never happen
        print("[final] evaluation failed:", exc)
        official = float("inf")

    return final_list
