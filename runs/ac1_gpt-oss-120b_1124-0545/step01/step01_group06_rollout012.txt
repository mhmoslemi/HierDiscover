"""
Propose a non‑negative step‑function (list of heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

where ``conv`` is the linear convolution (np.convolve).  

Algorithm
---------

1. **Broad parametric sweep** – generate many candidate sequences from a
   handful of analytic families (geometric, power‑law, geometric·power,
   symmetric geometric, pure exponential, Gaussian‑like and shifted
   power‑laws).  All candidates are normalised to sum = 1, evaluated by a
   fast FFT‑based convolution and the best few (by objective value) are
   stored.

2. **Refinement by exponentiated‑gradient descent** – for each stored
   candidate we run a mirror‑descent style optimisation that respects the
   simplex constraint:

   * a is kept on the probability simplex by multiplicative updates
     ``a ← a·exp(-η·g)`` followed by Euclidean projection onto the simplex.
   * The gradient ``g`` is an (sub)gradient of the *maximum* convolution.
     For the first half of the iterations we use a smooth approximation
     (soft‑max) of the max to obtain a dense gradient; for the second half
     we switch to the exact sub‑gradient at the true maximal index.
   * Momentum (exponential moving average) and a decaying step size are
     employed.  Random jitter is injected when progress stalls.

3. The best sequence seen during the entire budget (including sweep and
   refinements) is returned as a plain Python list.

The whole routine respects the provided time budget (default 1000 s) and
contains no file or network I/O.
"""

import time
import numpy as np

# ---------- utilities -------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w ≥ 0, Σw = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    mask = u - cssv / ind > 0
    if not np.any(mask):
        theta = cssv[-1] / n
    else:
        rho = np.where(mask)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _evaluate_normed(a: np.ndarray) -> float:
    """
    Objective for a sequence already normalised to sum = 1:
        2·n·max(conv(a,a))
    """
    conv = _conv_fft(a, a)
    return float(2.0 * len(a) * conv.max())


# ---------- parametric families ---------------------------------------
def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    return a / a.sum()


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64) + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _exp_decay(n: int, alpha: float) -> np.ndarray:
    """a_i = exp(-alpha * i) (normalised)."""
    a = np.exp(-alpha * np.arange(n, dtype=np.float64))
    return a / a.sum()


def _gauss(n: int, sigma: float) -> np.ndarray:
    """a_i = exp(-0.5 * (i/σ)^2) (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * (i / sigma) ** 2)
    return a / a.sum()


def _power_offset(n: int, p: float, offset: float) -> np.ndarray:
    """a_i = 1/(i+offset)^p (normalised)."""
    i = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / (i ** p)
    return a / a.sum()


# ---------- refinement -------------------------------------------------
def _refine_sequence(a0: np.ndarray, deadline: float,
                    max_iters: int = 6000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent refinement with momentum.
    Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _evaluate_normed(a)

    # momentum buffers
    grad_mom = np.zeros_like(a)
    v_mom = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta_start = 1.0
    beta_mul = 1.02

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()
            # progress report
            print(f"[refine] iter {it:4d} new best val={cur_val:.6f}")

        # -------------------------------------------------------------
        # gradient of the (smooth) max‑convolution
        if it < max_iters // 2:
            beta = beta_start * (beta_mul ** it)
            shift = cur_max
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0.0:
                continue
            w /= w_sum
            # gradient = 2 * (rev_a * w) on the relevant slice
            grad_full = _conv_fft(a[::-1], w)
            grad = 2.0 * grad_full[n - 1:n - 1 + n]
        else:
            # exact sub‑gradient at the true maximal index
            k_max = int(conv.argmax())
            idx = k_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad = np.zeros_like(a)
            grad[mask] = 2.0 * a[idx[mask]]

        # -------------------------------------------------------------
        # Adam‑style moments on the gradient
        grad_mom = beta1 * grad_mom + (1.0 - beta1) * grad
        v_mom = beta2 * v_mom + (1.0 - beta2) * (grad ** 2)

        m_hat = grad_mom / (1.0 - beta1 ** (it + 1))
        v_hat = v_mom / (1.0 - beta2 ** (it + 1))

        # step size – decays like 1/√t
        eta = 0.5 / np.sqrt(it + 1.0)

        # multiplicative (exponentiated) update
        a *= np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))

        # re‑normalise onto the simplex
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = np.ones_like(a) / n
        else:
            a /= a_sum

        # occasional random jitter to escape flat regions
        if it % 750 == 0 and it > 0:
            jitter = np.random.normal(scale=0.001, size=n)
            a = a * np.exp(jitter)
            a /= a.sum()

    return best_seq, best_val


# ---------- main entry point -----------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine runs for approximately ``budget_s``
    seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    best_seq = None
    best_val = float("inf")
    candidates = []                     # (value, array) tuples

    # -----------------------------------------------------------------
    # 1.  Broad parametric sweep
    # -----------------------------------------------------------------
    lengths = [800, 1200, 2000, 4000]    # a few representative sizes
    c_vals = np.linspace(0.3, 5.0, 25)   # controls geometric ratio r = 1 - c/n
    p_vals = np.linspace(0.0, 2.5, 22)   # power‑law exponent
    sigma_vals = np.linspace(0.08, 0.45, 12)   # for Gaussian (relative to n)
    offset_vals = np.linspace(0.0, 4.0, 12)    # offset for shifted power law

    for n in lengths:
        # ---- geometric ------------------------------------------------
        for c in c_vals:
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            a = _geom(n, r)
            val = _evaluate_normed(a)
            candidates.append((val, a))
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[sweep] geom   n={n:4d} c={c:5.2f} val={val:.6f}")

        # ---- power law ------------------------------------------------
        for p in p_vals:
            a = _power(n, p)
            val = _evaluate_normed(a)
            candidates.append((val, a))
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[sweep] power  n={n:4d} p={p:5.2f} val={val:.6f}")

        # ---- geometric × power ----------------------------------------
        for c in c_vals:
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in p_vals:
                a = _geom_power(n, r, p)
                val = _evaluate_normed(a)
                candidates.append((val, a))
                if val < best_val:
                    best_val, best_seq = val, a.copy()
                    print(f"[sweep] gp     n={n:4d} c={c:5.2f} p={p:5.2f} val={val:.6f}")

        # ---- symmetric geometric ---------------------------------------
        for c in c_vals:
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            a = _sym_geom(n, r)
            val = _evaluate_normed(a)
            candidates.append((val, a))
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[sweep] sym‑g  n={n:4d} c={c:5.2f} val={val:.6f}")

        # ---- pure exponential decay ------------------------------------
        for c in c_vals:
            alpha = c / n
            a = _exp_decay(n, alpha)
            val = _evaluate_normed(a)
            candidates.append((val, a))
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[sweep] exp    n={n:4d} α={alpha:.6f} val={val:.6f}")

        # ---- Gaussian-like ---------------------------------------------
        for s in sigma_vals:
            sigma = s * n
            a = _gauss(n, sigma)
            val = _evaluate_normed(a)
            candidates.append((val, a))
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[sweep] gauss  n={n:4d} σ={sigma:.1f} val={val:.6f}")

        # ---- shifted power-law -----------------------------------------
        for p in p_vals:
            for off in offset_vals:
                a = _power_offset(n, p, off)
                val = _evaluate_normed(a)
                candidates.append((val, a))
                if val < best_val:
                    best_val, best_seq = val, a.copy()
                    print(f"[sweep] p‑off  n={n:4d} p={p:5.2f} off={off:5.2f} val={val:.6f}")

        if time.time() >= deadline:
            break

    # -----------------------------------------------------------------
    # 2.  Keep the top K candidates and refine them
    # -----------------------------------------------------------------
    K = 6
    candidates.sort(key=lambda x: x[0])
    top_candidates = [c for _, c in candidates[:K]]

    for idx, cand in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[refine] starting refinement #{idx+1} (len={len(cand)})")
        refined_seq, refined_val = _refine_sequence(cand, deadline)
        if refined_val < best_val:
            best_val, best_seq = refined_val, refined_seq.copy()
            print(f"[refine] improvement from sweep: new best val={best_val:.6f}")

    # -----------------------------------------------------------------
    # 3.  Fallback – uniform (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(1000, dtype=np.float64)
        best_seq /= best_seq.sum()
        best_val = _evaluate_normed(best_seq)

    print(f"[final] best objective value found = {best_val:.6f}")
    return best_seq.tolist()
