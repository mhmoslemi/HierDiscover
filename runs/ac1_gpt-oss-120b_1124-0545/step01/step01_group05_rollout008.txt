"""
Propose a step‑function (list of non‑negative heights) that minimizes
`evaluate_sequence`.  

The search consists of two phases:

1. **Global random sampling** – repeatedly draw candidates from a
   collection of parametrised families:
   * exponential decay
   * power‑law decay
   * two‑block (high / low mass)
   * Dirichlet draws (random sparse vectors)
   * mixture of two exponentials
   * beta‑decay (a_i ∝ (1‑i/(n‑1))**β )
   * logistic decay (a_i ∝ 1/(1+exp((i‑μ)/σ)))
   * small random perturbations of the current best.

   Candidates are evaluated with a fast surrogate
   `2·n·max(conv(a,a))` (the scaling `sum(a)=1` makes the denominator equal to 1).
   Whenever a surrogate improvement is found the exact `evaluate_sequence`
   is called, and the best vector is updated.

2. **Targeted local refinement** – after the time‑budget‐driven random phase
   we take the best vector and perform a hill‑climbing that reduces the
   largest convolution peak.  For the current peak index `k` we compute the
   contribution of each coefficient `a[i]·a[k‑i]`.  Mass is moved from the
   coefficient with the largest contribution to a coefficient with the
   smallest contribution.  The step size is slowly decreased; occasional
   random nudges avoid getting stuck in a local minimum.

Both phases respect the 1000 s limit (a 5 s safety margin is kept).  The
function returns the best list of floats found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based – fast for lengths up to a few 10⁴)
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())        # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def _conv_max(a: np.ndarray) -> float:
    """Maximum value of the linear self‑convolution."""
    return float(_conv_full(a).max())


def _conv_max_and_idx(a: np.ndarray):
    """Return (max value, index of the max) of the linear self‑convolution."""
    conv = _conv_full(a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Fast surrogate (sum(a)=1 ⇒ denominator = 1)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) – identical to `evaluate_sequence` for sum(a)=1."""
    n = a.shape[0]
    return 2.0 * n * _conv_max(a)


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------
def _make_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def _make_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = i ** p
    a /= a.sum()
    return a


def _make_two_block(n: int, m: int, high_frac: float) -> np.ndarray:
    """`m` high entries, the rest low; total mass = 1."""
    high = high_frac / m
    low = (1.0 - high_frac) / (n - m)
    a = np.empty(n, dtype=float)
    a[:m] = high
    a[m:] = low
    return a


def _make_mix_exponential(n: int, w: float, lam1: float, lam2: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = w * np.exp(-lam1 * i) + (1.0 - w) * np.exp(-lam2 * i)
    a /= a.sum()
    return a


def _make_beta_decay(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (1 - i/(n‑1))**beta  (beta≥0 gives decreasing mass)."""
    i = np.arange(n, dtype=float)
    a = (1.0 - i / (n - 1)) ** beta
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=float)
    else:
        a /= s
    return a


def _make_logistic(n: int, mu: float, scale: float) -> np.ndarray:
    """Logistic decay: a_i ∝ 1/(1+exp((i‑μ)/scale))."""
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / scale))
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=float)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Local refinement – reduces the highest convolution peak
# ----------------------------------------------------------------------
def _local_refine(a: np.ndarray, best_score: float, deadline: float):
    """Hill‑climbing that moves a tiny fraction of mass from high‑contribution
    indices to low‑contribution indices.  Returns (best_a, best_score)."""
    n = a.shape[0]
    step = 0.05               # fraction of the source coefficient to move
    no_improve = 0

    while time.time() < deadline:
        # current peak
        cur_max, peak_idx = _conv_max_and_idx(a)

        # indices that actually contribute to this peak
        lo = max(0, peak_idx - (n - 1))
        hi = min(peak_idx, n - 1)
        idxs = np.arange(lo, hi + 1, dtype=int)
        paired = peak_idx - idxs
        # contribution of each term a[i] * a[peak_idx-i]
        contrib = a[idxs] * a[paired]

        # source: index with largest contribution (and non‑zero mass)
        src = int(idxs[np.argmax(contrib)])
        if a[src] <= 1e-12:
            # nothing to move, break out
            break

        # destination: index with smallest coefficient that is not the source
        dest_candidates = np.arange(n)
        dest_candidates = dest_candidates[dest_candidates != src]
        dest = int(dest_candidates[np.argmin(a[dest_candidates])])

        delta = step * a[src]
        delta = min(delta, a[src])                # keep non‑negative
        a[src] -= delta
        a[dest] += delta                           # total mass unchanged

        # recompute the fast surrogate score
        cur_score = _fast_score(a)

        if cur_score < best_score:
            # exact evaluation (expensive → only on improvements)
            try:
                exact = evaluate_sequence(a.tolist())
                if exact < best_score:
                    best_score = exact
                    best_a = a.copy()
                    print(f"[{time.time():.1f}s] Refine improved to {best_score:.6f}")
                    a = best_a                      # continue from the exact best
                    no_improve = 0
                    continue
            except Exception:
                pass

        # slowly shrink the step size
        step *= 0.9999
        no_improve += 1
        if no_improve > 5000 or step < 1e-6:
            # stop if stagnating
            break

    return a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights with a low evaluation
    score.  Runs for at most `budget_s` seconds and returns the best
    sequence (as a plain Python list) found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # keep a few seconds safety margin

    # ------------------------------------------------------------------
    # Length and starting point
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        init = np.array(height_sequence_1, dtype=float)
        init = np.clip(init, 0.0, None)
        if init.sum() > 0.0:
            init /= init.sum()
        else:
            init = np.full(n, 1.0 / n, dtype=float)
        best_a = init.copy()
        try:
            best_score = evaluate_sequence(best_a.tolist())
        except Exception:
            best_score = _fast_score(best_a)
    else:
        n = 2000
        best_a = np.full(n, 1.0 / n, dtype=float)
        best_score = _fast_score(best_a)

    # (store best as a copy to avoid accidental modifications)
    best_a = best_a.copy()

    # ------------------------------------------------------------------
    # Randomised global search over parametrised families
    # ------------------------------------------------------------------
    it = 0
    method_choices = [
        "exp", "power", "two_block", "dirichlet",
        "mix_exp", "beta", "logistic", "perturb"
    ]
    method_weights = [0.12, 0.12, 0.14, 0.22, 0.10, 0.10, 0.08, 0.12]

    while time.time() < deadline:
        it += 1
        method = np.random.choice(method_choices, p=method_weights)

        if method == "exp":
            lam = np.random.uniform(0.0, 8.0 / n)          # allow a slightly larger range
            a = _make_exponential(n, lam)

        elif method == "power":
            p = np.random.uniform(-6.0, 6.0)
            a = _make_power_law(n, p)

        elif method == "two_block":
            m = np.random.randint(int(0.05 * n), int(0.5 * n) + 1)
            high_frac = np.random.uniform(0.3, 0.9)
            a = _make_two_block(n, m, high_frac)

        elif method == "dirichlet":
            alpha = np.random.uniform(0.05, 5.0)
            a = np.random.dirichlet([alpha] * n)

        elif method == "mix_exp":
            w = np.random.uniform(0.0, 1.0)
            lam1 = np.random.uniform(0.0, 8.0 / n)
            lam2 = np.random.uniform(0.0, 8.0 / n)
            a = _make_mix_exponential(n, w, lam1, lam2)

        elif method == "beta":
            beta = np.random.uniform(0.0, 12.0)
            a = _make_beta_decay(n, beta)

        elif method == "logistic":
            mu = np.random.uniform(0.3 * n, 0.7 * n)
            scale = np.random.uniform(0.02 * n, 0.4 * n)
            a = _make_logistic(n, mu, scale)

        else:  # "perturb"
            a = best_a.copy()
            i, j = np.random.choice(n, 2, replace=False)
            delta = a[i] * np.random.uniform(-0.3, 0.3)
            delta = np.clip(delta, -a[i] + 1e-12, a[i])   # keep a[i] ≥ 0
            a[i] -= delta
            a[j] += delta
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                continue
            a /= s

        # safety – non‑negative & normalised
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s

        # fast surrogate score
        score = _fast_score(a)

        if score < best_score:
            # exact check only on improvements (saves time)
            try:
                exact = evaluate_sequence(a.tolist())
                if exact < best_score:
                    best_score = exact
                    best_a = a.copy()
                    elapsed = time.time() - start
                    print(f"[{elapsed:6.1f}s] New best: {best_score:.6f} (iter {it}, method={method})")
            except Exception:
                pass

        # occasional progress report
        if it % 3000 == 0:
            elapsed = time.time() - start
            remain = max(0.0, deadline - time.time())
            print(f"[{elapsed:6.1f}s] iter={it} best={best_score:.6f} time left≈{int(remain)}s")

    # ------------------------------------------------------------------
    # Targeted local refinement (peak‑reduction hill climbing)
    # ------------------------------------------------------------------
    # give the remaining time (if any) to the refinement phase
    if time.time() < deadline:
        best_a, best_score = _local_refine(best_a.copy(), best_score, deadline)

    # return as plain Python list
    return best_a.tolist()
