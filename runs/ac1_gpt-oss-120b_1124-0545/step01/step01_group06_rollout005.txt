def lp_solve(a, M):
    n = len(a)
    L = 2*n - 1
    # Build A matrix
    A = np.zeros((L, n))
    for k in range(L):
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # rows of A
        indices = np.arange(i_start, i_end+1)
        j = k - indices
        A[k, j] = a[indices]
    # bounded constraints
    bounds = [(0, None)] * n
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(L, M),
                  bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return None
    b_opt = res.x
    return b_opt

"""
Propose a step‑function (list of non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

by a three‑phase search:

1. **Broad parametric sweep** – families such as geometric, power‑law,
   geometric·power, symmetric geometric, logistic, piecewise‑geometric and
   random Dirichlet draws are sampled for several lengths (up to 4096).
   Each candidate is normalised to sum 1 and evaluated with an FFT‑based
   convolution (fast surrogate of ``evaluate_sequence``).

2. **Linear‑program improvement** – for the current best sequence *a* we
   solve the LP  

        maximise   Σ b_j
        subject to b ≥ 0,
                   (a * b)_k ≤ M      for all k,

   where *M* = max(a * a).  The LP pushes mass to positions that keep the
   autocorrelation with the reverse of *a* low.  The solution *b* is mixed
   into *a* (convex combination) and the process is repeated a few times.

3. **Smoothed‑max gradient descent (Adam‑style)** – the smoothed max
   ``log‑sum‑exp`` of the convolution is used as a differentiable proxy.
   An Adam update is applied to the log‑variables (multiplicative update)
   and the vector is projected onto the probability simplex after each
   step.

The whole routine respects the supplied ``budget_s`` time limit (default
1000 s).  The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    mask = u - cssv / ind > 0
    if not np.any(mask):
        theta = cssv[-1] / n
    else:
        rho = np.where(mask)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families of sequences (all return a numpy array that sums to 1)
# ----------------------------------------------------------------------
def _geom(n: int, r: float) -> np.ndarray:
    """Geometric a_i = r**i."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i = 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Combined geometric·power."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    if s == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (mirrored)."""
    a = r ** np.arange(n, dtype=np.float64) + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    a /= a.sum()
    return a


def _logistic(n: int, k: float) -> np.ndarray:
    """Logistic decreasing shape."""
    x = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(k * (x - n / 2.0)))
    s = a.sum()
    if s == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _piecewise_geom(n: int, r: float, cutoff: int) -> np.ndarray:
    """Geometric up to `cutoff`, zero afterwards."""
    idx = np.arange(n, dtype=np.float64)
    a = np.where(idx < cutoff, r ** idx, 0.0)
    s = a.sum()
    if s == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Random Dirichlet draw with concentration `alpha`."""
    a = np.random.dirichlet(np.full(n, alpha))
    return a


# ----------------------------------------------------------------------
# Fast surrogate of evaluate_sequence (scale‑invariant)
# ----------------------------------------------------------------------
def _fast_obj(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a, a)) – the objective when sum(a)=1."""
    conv = _conv_fft(a, a)
    return float(2.0 * len(a) * np.max(conv))


# ----------------------------------------------------------------------
# Linear‑program improvement (solve max‑sum(b) s.t. conv(a,b) ≤ M)
# ----------------------------------------------------------------------
def _solve_lp(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP for a given a and bound M.
    Returns the optimal b (normalised to sum 1) or None on failure.
    """
    n = a.shape[0]
    L = 2 * n - 1

    # Build the full convolution matrix A where (A @ b)_k = (a * b)_k.
    A = np.zeros((L, n), dtype=np.float64)
    for k in range(L):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int32)
        j_range = k - i_range
        A[k, j_range] = a[i_range]

    # Linear program: maximise Σ b  → minimise -(Σ b)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(L, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Use HiGHS through scipy.optimize.linprog (fast for moderate sizes)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    b_opt = np.maximum(res.x, 0.0)
    s = b_opt.sum()
    if s == 0:
        return None
    b_opt /= s
    return b_opt


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the target
    evaluation function.  Returns the best sequence found (list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    best_seq = None          # numpy array, sums to 1
    best_val = float('inf')  # fast surrogate value

    # ------------------------------------------------------------------
    # Phase 1 – parametric sweep (broad exploration)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]          # modest set of lengths
    r_vals = np.linspace(0.5, 0.99, 12)       # geometric ratio
    p_vals = np.linspace(0.1, 2.0, 12)        # power‑law exponent
    k_vals = np.linspace(0.01, 0.5, 12)       # logistic steepness
    cutoff_fracs = [0.1, 0.3, 0.5, 0.7]       # for piecewise‑geom
    alpha_vals = [0.05, 0.1, 0.5, 1.0, 5.0]   # Dirichlet concentration

    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_val
        if time.time() > deadline:
            return
        val = _fast_obj(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[sweep] new best n={len(a)} val={val:.6f}")

    for n in lengths:
        # geometric
        for r in r_vals:
            a = _geom(n, r)
            _maybe_update(a)

        # power‑law
        for p in p_vals:
            a = _power(n, p)
            _maybe_update(a)

        # combined geo·power (coarse grid)
        for r in r_vals[::3]:
            for p in p_vals[::3]:
                a = _geom_power(n, r, p)
                _maybe_update(a)

        # symmetric geometric
        for r in r_vals:
            a = _sym_geom(n, r)
            _maybe_update(a)

        # logistic
        for k in k_vals:
            a = _logistic(n, k)
            _maybe_update(a)

        # piecewise geometric
        for frac in cutoff_fracs:
            cutoff = int(frac * n)
            if cutoff == 0:
                continue
            for r in r_vals:
                a = _piecewise_geom(n, r, cutoff)
                _maybe_update(a)

        # random Dirichlet draws
        for alpha in alpha_vals:
            for _ in range(4):
                a = _random_dirichlet(n, alpha)
                _maybe_update(a)

        if time.time() > deadline:
            break

    # Fallback guard
    if best_seq is None:
        n = 1024
        best_seq = np.ones(n, dtype=np.float64) / n
        best_val = _fast_obj(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – linear‑program improvement (Frank‑Wolfe style)
    # ------------------------------------------------------------------
    a = best_seq.copy()                     # current iterate (sum = 1)
    lp_iters = 30
    t = 0.4                                  # convex‑combination weight

    for it in range(lp_iters):
        if time.time() > deadline:
            break
        M = np.max(_conv_fft(a, a))
        b_opt = _solve_lp(a, M)
        if b_opt is None:
            break
        # convex update and projection
        a = (1 - t) * a + t * b_opt
        a = _project_simplex(a)

        cur_val = _fast_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[lp] iter {it:02d} new best val={cur_val:.6f}")

        t = max(t * 0.92, 0.02)   # slowly shrink step size

    # ------------------------------------------------------------------
    # Phase 3 – smoothed‑max gradient descent (Adam‑style)
    # ------------------------------------------------------------------
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    max_refine = 3000

    for it in range(max_refine):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_conv = np.max(conv)

        # smoothed max weights (log‑sum‑exp)
        beta = 5.0 * (1.5 ** it)
        beta = min(beta, 200.0)                 # cap to keep numerics stable
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            continue
        w /= w_sum

        # gradient of the smoothed max (2 * a_rev * w)
        grad_full = _conv_fft(a[::-1], w)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update on log‑variables (multiplicative change)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        eta = 0.5 / np.sqrt(it + 1.0)          # decreasing step size
        a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))
        a = _project_simplex(a)                # stay on the simplex

        # periodic reporting
        if (it % 100) == 0:
            cur_val = _fast_obj(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[refine] iter {it:04d} new best val={cur_val:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python list)
    # ------------------------------------------------------------------
    return best_seq.tolist()
