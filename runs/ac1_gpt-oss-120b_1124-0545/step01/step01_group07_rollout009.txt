"""
propose_candidate
-----------------
Search for a non‑negative step‑function (a list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the full linear convolution.  The objective is homogeneous,
so only the *shape* of the vector matters (its absolute scale is irrelevant
as long as the entries stay within [0, 1000]).

The algorithm consists of three stages:

1. **Parametric sweep** – generate a rich family of analytically defined
   shapes (inverse‑power‑law, shifted power‑law and a few elementary forms)
   for several lengths.  Each candidate is normalised (max < 0.9, sum ≥ 0.01)
   and evaluated; the best one is kept.

2. **Targeted mass–transfer hill‑climbing** – starting from the current
   best shape we repeatedly identify the convolution index *k* where the
   maximum occurs, locate the pair of entries that contribute most to this
   peak and move a small fraction of mass from the “heavy’’ entry to a
   “light’’ entry.  After each move we recompute the score; moves that
   improve the score are always accepted, otherwise they are accepted
   with a temperature‑controlled probability (simulated annealing).

3. **Fine‑grained random perturbations** – occasional Gaussian perturbations
   of a random subset of entries are performed to escape possible local
   minima.

All operations are performed with NumPy; the convolution required for the
score is computed only once per iteration.  The routine respects the
time budget (default 1000 s) and returns the best sequence found so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _normalise_seq(a: np.ndarray) -> np.ndarray:
    """Scale a non‑negative array so that
       * max entry < 0.9  (far below the 1000‑clip limit)
       * total sum >= 0.01 (otherwise evaluate_sequence returns ∞)
       The scaling does not affect the final score because the
       evaluation function is homogeneous.
    """
    a = np.asarray(a, dtype=float)
    a = np.clip(a, 0.0, 999.0)                # stay safely below the 1000 clamp
    max_v = a.max()
    if max_v > 0.0:
        a = a * (0.9 / max_v)                # keep the peak well under 1
    s = a.sum()
    if s < 0.01:
        # increase uniformly so that the sum guard is satisfied
        a = a + (0.01 - s) / a.size
    return a


def _score(seq: np.ndarray) -> float:
    """Fast replica of evaluate_sequence (without the security checks)."""
    n = seq.size
    conv = np.convolve(seq, seq)              # full linear convolution
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:                          # same guard as in the original
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _power_law_shape(n: int, p: float, offset: float = 0.0,
                    shift: int = 0) -> np.ndarray:
    """
    Return a decreasing power‑law shape:
        a_i = 1 / ( (i + 1 + offset) ** p )
    Optionally prepend `shift` leading zeros.
    """
    i = np.arange(1, n - shift + 1, dtype=float) + offset
    shape = 1.0 / np.power(i, p)
    if shift > 0:
        shape = np.concatenate((np.zeros(shift, dtype=float), shape))
    return shape


def _candidate_shapes_extended(n: int):
    """Generate a diverse collection of parametric shapes for length `n`."""
    cand = []

    # 1. elementary families (constant, linear, etc.) – kept for completeness
    cand.append(np.ones(n, dtype=float))
    cand.append(np.arange(1, n + 1, dtype=float))          # increasing line
    cand.append(np.arange(n, 0, -1, dtype=float))          # decreasing line

    # 2. power‑law families – exponent in a neighbourhood of 0.5
    exponents = np.linspace(0.40, 0.60, 9)                 # 0.40 … 0.60
    offsets   = [0.0, 0.5, 1.0, 2.0]                       # tiny shift of the denominator
    shifts    = [0, max(1, n // 200), max(1, n // 100)]   # allow a few leading zeros

    for p in exponents:
        for off in offsets:
            for sh in shifts:
                cand.append(_power_law_shape(n, p, offset=off, shift=sh))

    # 3. “valley’’ – small values in the centre, larger at the edges
    centre = (n - 1) / 2.0
    valley = 1.0 / (1.0 + ((np.arange(n, dtype=float) - centre) ** 2))
    cand.append(valley)

    # 4. sinusoidal squaring (already in the previous version – kept)
    cand.append(np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) / (n + 1)) ** 2)

    return cand


def _transfer_mass(a: np.ndarray, high_idx: int, low_idx: int,
                  frac: float = 0.05) -> np.ndarray:
    """
    Move a fraction `frac` of the mass from `high_idx` to `low_idx`.
    The total sum is preserved and non‑negativity is guaranteed.
    """
    delta = a[high_idx] * frac
    if delta <= 0.0:
        return a
    new = a.copy()
    new[high_idx] -= delta
    new[low_idx] += delta
    return new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The function respects the given time budget and
    prints progress information.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm start – use a previously discovered construction if available
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        seq = _normalise_seq(np.asarray(hs1, dtype=float))
        score = _score(seq)
        best_seq = seq.copy()
        best_score = score
        print(f"[{time.time() - start_time:.1f}s] Warm start from height_sequence_1 "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – systematic exploration of parametric families
    # ------------------------------------------------------------------
    # try a few different lengths (the final answer may use any length)
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    for n in lengths:
        if time.time() >= deadline:
            break
        for raw in _candidate_shapes_extended(n):
            if time.time() >= deadline:
                break
            cand = _normalise_seq(raw)
            score = _score(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.1f}s] New best (n={n}) "
                      f"score={best_score:.6f}")
            # early stop if we already beat the (probably unattainable) target
            if best_score <= 1.503:
                print("[Info] Target reached during Phase 1 – exiting early.")
                return best_seq.tolist()

    # fallback in the unlikely case that nothing was accepted
    if best_seq is None:
        best_seq = rng.random(512)
        best_seq = _normalise_seq(best_seq)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – targeted mass‑transfer hill climbing (simulated annealing)
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_score

    # linear temperature schedule for simulated annealing
    start_temp, end_temp = 1.0, 0.001

    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = start_temp * (1 - elapsed / total) + end_temp * (elapsed / total)

        # compute convolution once per iteration
        conv = np.convolve(cur, cur)
        max_k = int(conv.argmax())
        max_val = float(conv[max_k])

        # ----- locate the entry contributing most to the peak -----------------
        # indices that actually appear in the sum for this k
        lo = max(0, max_k - (cur.size - 1))
        hi = min(cur.size - 1, max_k)
        idxs = np.arange(lo, hi + 1, dtype=int)
        contrib = cur[idxs] * cur[max_k - idxs]       # a_i * a_{k-i}
        i_high = idxs[int(contrib.argmax())]

        # ----- pick a low‑contributing index as a sink for the transferred mass
        # we aim at indices where the contribution to the current peak is small
        median = np.median(contrib)
        low_candidates = idxs[contrib <= median]
        if low_candidates.size == 0:
            i_low = rng.integers(0, cur.size)
        else:
            i_low = int(rng.choice(low_candidates))

        # ----- perform a small mass transfer
        new = _transfer_mass(cur, i_high, i_low, frac=0.05)

        # Occasionally perform a random Gaussian perturbation to explore
        if rng.random() < 0.15:
            m = max(1, int(0.01 * cur.size))
            idx = rng.choice(cur.size, size=m, replace=False)
            delta = rng.standard_normal(m) * cur.mean() * 0.08
            new[idx] += delta
            new = np.clip(new, 0.0, 999.0)

        # keep the total sum unchanged (the transfer already does that)
        # but guard against accidental drift caused by the random perturbation
        sum_new = new.sum()
        if sum_new < 0.01:
            new += (0.01 - sum_new) / new.size
        else:
            # renormalise to the original sum to stay on the simplex
            new = new * (cur.sum() / sum_new)

        # ----- evaluate the candidate -----------------------------------------
        cand_score = _score(new)

        # ----- acceptance ------------------------------------------------------
        if cand_score < cur_score:
            cur = new
            cur_score = cand_score
        else:
            # simulated‑annealing acceptance of a worse move
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur = new
                cur_score = cand_score

        # ----- update the global best -----------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best_seq = new.copy()
            print(f"[{time.time() - start_time:.1f}s] Improved best score={best_score:.6f} "
                  f"(it={it})")
            if best_score <= 1.503:
                print("[Info] Target reached – terminating search.")
                break

    # return the best sequence found (as plain Python list)
    return best_seq.tolist()
