def project_simplex(v):
    # ensure non-negative and sum =1
    # using algorithm: sort v descending, find rho etc.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def conv_fft(a,b):
    n = len(a)
    m = len(b)
    size = 1 << ((n+m-1).bit_length())
    Fa = np.fft.rfft(a, n=size)

def gradient_descent(a, steps=200, eta0=0.5, eta1=0.05, t0=5.0, t1=2000.0):
    a_cur = a.copy()
    n = len(a_cur)
    for k in range(steps):
        t = t0 * (t1/t0)**(k/steps)
        eta = eta0 * (eta1/eta0)**(k/steps)
        c = conv_fft(a_cur, a_cur)
        max_c = np.max(c)
        # compute softmax weights w

  points.
* A smooth‑max gradient descent on the surrogate objective

      M_t(a) = (1/t) * log( sum_k exp(t·(a*a)_k) )

  which converges to the true max as the temperature `t` grows.
  The gradient is computed analytically using convolutions and the
  iterate is projected onto the probability simplex after each step.
* A final annealing / refinement phase that repeatedly perturbs the
  best sequence and runs a short gradient descent.

All heavy work is done with FFT‑based convolutions, so the method scales
to a few thousand coefficients within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the linear convolution of two real vectors using FFT."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    Fa = np.fft.rfft(a, n=size)
    Fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(Fa * Fb, n=size)[: n + m - 1]
    return conv


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the (linear) self‑convolution of a."""
    return float(_conv_fft(a, a).max())


def _fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for `evaluate_sequence` when sum(a) == 1 :
    2·n·max(conv(a,a)).
    """
    return 2.0 * a.shape[0] * _conv_max(a)


# ----------------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i >= 0, sum x_i = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]                # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                 # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise to exactly 1
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Families for random initialisation
# ----------------------------------------------------------------------


def _make_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def _make_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = i ** p
    a /= a.sum()
    return a


def _make_two_block(n: int, m: int, high_frac: float) -> np.ndarray:
    if not (0 < m < n):
        raise ValueError("invalid block size")
    high = high_frac / m
    low = (1.0 - high_frac) / (n - m)
    a = np.empty(n, dtype=float)
    a[:m] = high
    a[m:] = low
    return a


def _make_mix_exponential(n: int, w: float, lam1: float, lam2: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = w * np.exp(-lam1 * i) + (1.0 - w) * np.exp(-lam2 * i)
    a /= a.sum()
    return a


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet([alpha] * n)


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (temperature schedule + step‑size schedule)
# ----------------------------------------------------------------------


def _smooth_max_grad(a: np.ndarray, t: float):
    """
    Compute smooth‑max value M_t(a) and its gradient w.r.t. a.
    Returns (value, gradient) where gradient has the same shape as a.
    """
    n = a.shape[0]
    # self‑convolution (full length 2n‑1)
    c = _conv_fft(a, a)
    # softmax weights using the temperature t
    c_max = c.max()
    # subtract max for numerical stability
    w_raw = np.exp(t * (c - c_max))
    w = w_raw / w_raw.sum()          # shape (2n‑1,)

    # gradient: 2 * conv_rev(a_rev, w) restricted to indices n‑1 … 2n‑2
    rev = _conv_fft(a[::-1], w)      # length 3n‑2
    grad = 2.0 * rev[n - 1 : 2 * n - 1]

    # smooth‑max value (for diagnostics)
    smooth_val = (c_max + np.log(w_raw.sum()) / t)

    return smooth_val, grad


def _gradient_descent(
    a0: np.ndarray,
    steps: int = 300,
    eta_start: float = 0.5,
    eta_end: float = 0.02,
    t_start: float = 5.0,
    t_end: float = 2000.0,
) -> np.ndarray:
    """
    Run a temperature‑annealed gradient descent starting from a0.
    The iterate is projected onto the simplex after each step.
    """
    a = a0.copy()
    n = a.shape[0]

    for k in range(steps):
        # geometric schedules for temperature and step size
        frac = k / steps
        t = t_start * (t_end / t_start) ** frac
        eta = eta_start * (eta_end / eta_start) ** frac

        # smooth‑max and gradient
        _, grad = _smooth_max_grad(a, t)

        # update and project
        a = _project_simplex(a - eta * grad)

        # optional early stop if gradient becomes tiny (not essential)
        if np.linalg.norm(grad) < 1e-12:
            break

    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative sequence that minimises the evaluation metric.
    The function runs for at most `budget_s` seconds and returns the best
    sequence (as a plain Python list) found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Determine the length of the search.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        base_seq = np.array(height_sequence_1, dtype=float)
        base_seq = np.clip(base_seq, 0.0, None)
        if base_seq.sum() > 0:
            base_seq = base_seq / base_seq.sum()
        else:
            base_seq = np.full(n, 1.0 / n, dtype=float)
    else:
        # fallback length; a few thousand gives a good balance
        n = 3000
        base_seq = np.full(n, 1.0 / n, dtype=float)

    # ------------------------------------------------------------------
    # Initialise the best solution with the provided construction (if any)
    # ------------------------------------------------------------------
    best_a = base_seq.copy()
    best_score = _fast_score(best_a)
    # get the exact score (expensive) only once for the baseline
    try:
        exact = evaluate_sequence(best_a.tolist())
        best_score = exact
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate quickly and (occasionally) exactly
    # ------------------------------------------------------------------
    def try_update(candidate: np.ndarray):
        """Evaluate `candidate` and update globals if it improves."""
        nonlocal best_a, best_score
        # quick surrogate
        surrogate = _fast_score(candidate)
        if surrogate >= best_score:
            return
        # exact evaluation
        try:
            exact_val = evaluate_sequence(candidate.tolist())
        except Exception:
            return
        if exact_val < best_score:
            best_score = exact_val
            best_a = candidate.copy()
            elapsed = time.time() - start
            print(
                f"[{elapsed:6.1f}s] NEW BEST: {best_score:.6f} (surrogate={surrogate:.6f})"
            )

    # ------------------------------------------------------------------
    # Main stochastic search: random families -> gradient descent
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ------- sample a random family ---------------------------------
        method = np.random.choice(
            ["exp", "power", "two_block", "dirichlet", "mix_exp", "perturb"],
            p=[0.15, 0.15, 0.20, 0.25, 0.15, 0.10],
        )
        if method == "exp":
            lam = np.random.uniform(0.0, 5.0 / n)
            cand = _make_exponential(n, lam)

        elif method == "power":
            p = np.random.uniform(-5.0, 5.0)
            cand = _make_power_law(n, p)

        elif method == "two_block":
            m = np.random.randint(int(0.05 * n), int(0.95 * n) + 1)
            hi_frac = np.random.uniform(0.0, 1.0)
            cand = _make_two_block(n, m, hi_frac)

        elif method == "dirichlet":
            alpha = np.random.uniform(0.05, 5.0)
            cand = _make_dirichlet(n, alpha)

        elif method == "mix_exp":
            w = np.random.uniform(0.0, 1.0)
            lam1 = np.random.uniform(0.0, 5.0 / n)
            lam2 = np.random.uniform(0.0, 5.0 / n)
            cand = _make_mix_exponential(n, w, lam1, lam2)

        else:  # perturb the current best
            cand = best_a.copy()
            i, j = np.random.choice(n, 2, replace=False)
            delta = cand[i] * np.random.uniform(-0.4, 0.4)
            delta = np.clip(delta, -cand[i] + 1e-12, 1e12)
            cand[i] -= delta
            cand[j] += delta
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s

        # ------- run a short gradient descent from this start ----------
        # The number of steps is deliberately modest so we can explore many starts.
        cand = _gradient_descent(cand, steps=150, eta_start=0.5, eta_end=0.02,
                                 t_start=10.0, t_end=1500.0)

        # ------- check improvement --------------------------------------
        try_update(cand)

        # occasional progress report
        if iteration % 2000 == 0:
            elapsed = time.time() - start
            remaining = max(0.0, deadline - time.time())
            print(
                f"[{elapsed:6.1f}s] iter={iteration} best={best_score:.6f} "
                f"time left≈{int(remaining)}s"
            )

    # ------------------------------------------------------------------
    # Final refinement: longer descent + occasional random perturbations
    # ------------------------------------------------------------------
    print(f"Starting final refinement (elapsed {time.time() - start:.1f}s)")
    for _ in range(8):
        # perturb the current best slightly
        pert = best_a.copy()
        i, j = np.random.choice(n, 2, replace=False)
        delta = pert[i] * np.random.uniform(-0.05, 0.05)
        delta = np.clip(delta, -pert[i] + 1e-12, 1e12)
        pert[i] -= delta
        pert[j] += delta
        pert = _project_simplex(pert)

        # deeper descent
        pert = _gradient_descent(
            pert,
            steps=600,
            eta_start=0.3,
            eta_end=0.01,
            t_start=20.0,
            t_end=3000.0,
        )
        try_update(pert)

    # ------------------------------------------------------------------
    # Return the best sequence as a plain list
    # ------------------------------------------------------------------
    print(f"Finished search (total time {time.time() - start:.1f}s), best score: {best_score:.6f}")
    return best_a.tolist()
