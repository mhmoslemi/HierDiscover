Pseudo:
def simplex_projection(v):
    # ensures sum(v)=1, v>=0.
    # Returns w.
    v = np.asarray(v)
    if v.sum() == 1 and np.all(v >= 0): return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w.

def convolve_self(a):
    n = len(a)
    size = 2*n - 1
    # Compute next power of 2 for FFT (maybe for speed)
    nfft = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, n=nfft)
    B = A * A
    conv = np.fft.irfft(B, n=nfft)[:size]
    return conv

"""
Projected subgradient descent + short simulated‑annealing refinement.

Phase 1: deterministic projected subgradient descent on the value
        f(a)=max_k (a⋆a)[k] (the L∞ norm of the self‑convolution).  The
        gradient of the active shift is g_i = 2·a_{k_max−i} (zero if out of
        range).  After a gradient step we project onto the probability
        simplex (a_i≥0, Σa_i=1).  The step size is slowly reduced and, when
        progress stalls, a small random perturbation is added.

Phase 2: a lightweight simulated‑annealing / targeted‑move search
        started from the best point of phase 1.  Moves transfer a tiny
        amount of mass from an index that contributes most to the current
        maximal convolution entry to a random destination.  Classical
        Metropolis acceptance with a linear temperature schedule is used.

Both phases keep the total mass normalized, so the objective
        2·n·max(conv(a,a))/(Σa)² = 2·n·max(conv(a,a)) is invariant under scaling.
The final best sequence (list of floats) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=float)
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the rho (largest i satisfying the condition)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_self_fft(a: np.ndarray) -> np.ndarray:
    """Return the self‑convolution a∗a using a real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    size = 2 * n - 1
    # next power of two for efficient FFT
    nfft = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, nfft)
    B = A * A
    conv = np.fft.irfft(B, nfft)[:size]
    # Numerical noise may produce tiny negative numbers
    return np.clip(conv, 0.0, None)


def compute_score_fft(a: np.ndarray) -> float:
    """Exact objective value for a (assumes a ≥ 0, Σa>0)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    conv = convolve_self_fft(a)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def random_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from i to j (simple exploration)."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return simplex_projection(a_new)


def targeted_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Move mass away from an index that contributes most to the current
    maximal convolution entry.
    """
    n = a.shape[0]
    conv = convolve_self_fft(a)
    k_max = int(conv.argmax())

    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1)

    w = np.zeros(n, dtype=float)
    w[idx] = a[idx] * a[k_max - idx]

    if w.sum() == 0.0:
        i = np.random.randint(n)
    else:
        probs = w / w.sum()
        i = np.random.choice(n, p=probs)

    # pick a distinct destination
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1

    max_transfer = a[i]
    if max_transfer == 0.0:
        return a.copy()
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return simplex_projection(a_new)


def projected_gradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One sub‑gradient descent step on max(conv(a,a)) and projection onto the simplex.
    """
    n = a.shape[0]
    conv = convolve_self_fft(a)
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)

    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]      # ∂b_k/∂a_i = 2·a_{k-i}
    a_new = a - step * grad
    return simplex_projection(a_new)

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value height sequence within ``budget_s`` seconds.
    Returns the best list of non‑negative floats found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # Initial sequence (try to reuse a known good construction if present)
    # ------------------------------------------------------------------
    init_seq = None
    try:
        # ``height_sequence_1`` is provided by the harness if available
        init_seq = np.array(height_sequence_1, dtype=float)
        if init_seq.shape[0] != n:
            # Length mismatch – fall back to random start
            init_seq = None
    except NameError:
        init_seq = None

    if init_seq is not None:
        a = np.clip(init_seq, 0.0, None)
        if a.sum() == 0.0:
            a = np.random.rand(n)
        a = a[:n]
    else:
        a = np.random.rand(n)

    a = simplex_projection(a)          # enforce Σa = 1, a_i ≥ 0
    best_seq = a.copy()
    best_score = compute_score_fft(best_seq)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic projected sub‑gradient descent
    # ------------------------------------------------------------------
    phase1_frac = 0.75
    phase1_end = start + phase1_frac * (deadline - start)

    # step schedule (large → small)
    step_start = 0.30
    step_end = 0.001

    last_improve = start
    iter_phase1 = 0

    while time.time() < phase1_end:

        prog = (elapsed) / (phase1_end - start)            # 0 → 1 over phase 1

            best_score = cand_score

            print(f"[phase1 {iter_phase1:06d}] new best {best_score:.6f}  time={elapsed:.1f}s")
        else:

            if np.random.rand() < 0.02:
                cand2 = random_move(a, step * 0.5)
                cand2_score = compute_score_fft(cand2)
                if cand2_score < best_score:
                    a = cand2
                    best_seq = cand2.copy()
                    best_score = cand2_score
                    last_improve = time.time()
                    print(f"[phase1 {iter_phase1:06d}] new best via random {best_score:.6f}")

            a = best_seq + np.random.normal(scale=step * 0.2, size=n)
            a = simplex_projection(a)
            last_improve = time.time()
            print(f"[phase1 {iter_phase1:06d}] restart (perturbation)")

        iter_phase1 += 1

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score

    # annealing schedule
    init_temp = 1.0
    final_temp = 0.001
    step_start2 = 0.10
    step_end2 = 0.001

    last_improve = time.time()
    iter_phase2 = 0

        elapsed = time.time() - start
        # progress over the *whole* remaining budget (phase 2)
        prog_total = (elapsed) / (deadline - start)
        temp = init_temp * (1.0 - prog_total) + final_temp * prog_total
        step_scale = step_start2 * (1.0 - prog_total) + step_end2 * prog_total

        # pick a move type

        if r < 0.55:
            cand = targeted_move(cur_seq, step_scale)
        else:
            cand = random_move(cur_seq, step_scale)

        cand_score = compute_score_fft(cand)

        # Metropolis acceptance (lower is better)
        if cand_score < cur_score or np.random.rand() < math.exp(
            -(cand_score - cur_score) / max(temp, 1e-12)
        ):
            cur_seq = cand
            cur_score = cand_score

            if cand_score < best_score:

                best_score = cand_score

            print(f"[phase2 {iter_phase2:06d}] restart due to stagnation")

    print(f"[done] total iterations={iter_phase1 + iter_phase2}, best score={best_score:.6f}, time={total_time:.1f}s")
    return best_seq.tolist()
