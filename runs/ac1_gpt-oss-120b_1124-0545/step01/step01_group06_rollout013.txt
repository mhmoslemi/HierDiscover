"""
Propose a non‑negative step function (list of heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

by a multi‑restart Adam‑style optimiser with a smoothed‑max loss.
The optimiser works on the probability simplex (∑a_i = 1, a_i ≥ 0) so
the objective reduces to 2·n·max(conv(a,a)).  Several length scales and
a few parametric seeds (geometric, power‑law, combined) are tried,
each run is limited by the overall time budget.  The best sequence
found (as a plain Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT convolution, simplex projection, parametric families)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    L = x.size + y.size - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    mask = u - cssv / ind > 0
    if not np.any(mask):
        theta = cssv[-1] / n
    else:
        rho = np.where(mask)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----- families of parametric sequences --------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric a_i ∝ r**i, normalised to sum 1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ 1/(i+1)**p, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    """a_i ∝ r**i / (i+1)**p, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Optimisation core (Adam on smoothed‑max loss)
# ----------------------------------------------------------------------


def _adam_optimize(
    a_init: np.ndarray,
    max_iters: int,
    deadline: float,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Run Adam optimiser on the smoothed‑max objective.

    Returns the best vector seen and its (approximate) objective value.
    """
    a = a_init.copy()
    n = a.size
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule for the soft‑max temperature
    beta_start = 5.0
    beta_inc = 0.001

    best_local = a.copy()
    best_val_local = float("inf")

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # ----- evaluate current objective (sum(a)==1) -----
        conv = _conv_fft(a, a)
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val_local:
            best_val_local = cur_val
            best_local = a.copy()
            print(f"[adam] n={n:5d} it={it:4d} val={cur_val:.6f}")

        # ----- smoothed max (log‑sum‑exp) -----
        beta = beta_start + beta_inc * it
        shift = cur_max                     # numerical stabiliser
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of the smoothed max w.r.t. a -----
        # grad = 2 * (a_rev * w)  (restricted to length n)
        grad_full = _conv_fft(a[::-1], w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- Adam update (additive) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = 0.05 * m_hat / (np.sqrt(v_hat) + eps)   # learning‑rate ≈0.05
        a = a - step
        a = _project_simplex(a)

    return best_local, best_val_local


# ----------------------------------------------------------------------
# Main entry point – propose_candidate
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The function respects the given time budget
    (default 1000 s) and returns the best sequence found as a Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to possibly use a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            h = np.asarray(height_sequence_1, dtype=np.float64)
            if h.size > 0:
                h = np.maximum(h, 0.0)
                h = h / h.sum()
                best_seq, best_val = h, 2.0 * len(h) * _conv_fft(h, h).max()
                print("[seed] using height_sequence_1 as initial best")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Lengths we will explore – small ones first for speed
    # ------------------------------------------------------------------
    lengths = [1500, 3000, 5000, 8000]  # can be truncated if time runs out

    for n in lengths:
        if time.time() > deadline:
            break

        # ---- a few deterministic seeds ---------------------------------
        seeds = []

        # uniform
        seeds.append(np.ones(n, dtype=np.float64) / n)

        # geometric families (c = r‑offset * n)
        for c in (1.0, 1.5, 2.0, 3.0):
            r = 1.0 - c / n
            if r > 0.0:
                seeds.append(_geom(n, r))

        # power‑law families
        for p in (0.4, 0.8, 1.2, 1.6):
            seeds.append(_power(n, p))

        # combined geometric × power
        for c in (1.0, 2.0):
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in (0.5, 1.0, 1.5):
                seeds.append(_geom_power(n, r, p))

        # a few random Dirichlet draws
        for _ in range(3):
            rnd = np.random.rand(n)
            rnd /= rnd.sum()
            seeds.append(rnd)

        # ----------------------------------------------------------------
        # Run a short Adam optimisation from each seed
        # ----------------------------------------------------------------
        for idx, a0 in enumerate(seeds):
            if time.time() > deadline:
                break
            # modest iteration count, longer for larger n
            max_iters = 1800 if n <= 3000 else 1000
            a_opt, val_opt = _adam_optimize(
                a0, max_iters=max_iters, deadline=deadline
            )
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[global] new best from seed {idx} (n={n}) val={val_opt:.6f}")

    # ------------------------------------------------------------------
    # Final polishing of the current best (if any)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # a few more Adam steps with a higher temperature schedule
        extra_iters = 3000
        best_seq, best_val = _adam_optimize(
            best_seq, max_iters=extra_iters, deadline=deadline
        )
        print(f"[final] polished best value = {best_val:.6f}")

    # Fallback – should never happen, but keep a safe default
    if best_seq is None:
        n = 2000
        best_seq = np.ones(n, dtype=np.float64) / n
        print("[fallback] using uniform sequence")

    return best_seq.tolist()
