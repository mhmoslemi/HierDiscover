"""
Propose a sequence that minimizes the given evaluation function.

The algorithm consists of two phases:

1. **Global improvement via a linear‑programming (LP) mixing step**  
   Starting from the classical  a_i ∝ 1/√(i+1)  sequence, we repeatedly
   solve the LP

       maximise   Σ b_i
       subject to  (f * b)[k] ≤ M   for all k,
                   b_i ≥ 0

   where `f` is the current sequence, `M = max(f*f)`.  The solution `b`
   is scaled to have unit sum (the same normalisation as `f`) giving a
   candidate `g`.  The new sequence is a convex combination
   `f←(1‑t)·f + t·g`.  The mixing weight `t` is chosen by a simple
   back‑tracking line search so that the new maximal convolution value
   strictly decreases.  Repeating this step drives the value of
   `max(f*f)` down.

2. **Local fine‑tuning**  
   After the LP phase we run a projected‑gradient descent that directly
   decreases the active (largest) convolution entry.  This removes the
   remaining tiny irregularities.

Both phases respect the non‑negativity and the sum‑to‑one normalisation,
which makes the score `2·n·max(f*f)` invariant to scaling.
The routine stops after the allotted time (default 1000 s) and returns
the best sequence encountered.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _initial_root_sequence(n: int) -> np.ndarray:
    """Return a normalised sequence a_i ∝ 1/√(i+1) of length n."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def _projected_gradient_step(seq: np.ndarray, step_size: float) -> np.ndarray:
    """
    One projected gradient step that reduces the current maximal
    convolution entry.

    Parameters
    ----------
    seq : np.ndarray
        Current non‑negative sequence (already normalised to sum 1).
    step_size : float
        Step size for the gradient descent.

    Returns
    -------
    np.ndarray
        Updated sequence, still non‑negative and summed to 1.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    k_max = int(np.argmax(conv))

    # Gradient of the active convolution entry with respect to seq.
    grad = np.zeros_like(seq)
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad[idx] = seq[k_max - idx]

    # Make the step tangent to the hyperplane sum(seq)=1.
    grad -= grad.mean()

    # Gradient descent step followed by projection onto the non‑negative
    # orthant and renormalisation.
    new_seq = seq - step_size * grad
    new_seq = np.maximum(new_seq, 0.0)
    total = new_seq.sum()
    if total > 0.0:
        new_seq /= total
    else:               # extremely unlikely, but keep a valid fallback
        new_seq = _initial_root_sequence(n)

    return new_seq


def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the (2n‑1) × n matrix A such that A @ b = f * b,
    i.e. the discrete convolution of the fixed vector `f` with a variable
    vector `b`.  Each column i of A contains a shifted copy of `f`.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # Column i contributes f[0] ... f[n‑1] to rows i … i+n‑1.
    for i in range(n):
        A[i:i + n, i] = f
    return A


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a high‑quality non‑negative sequence within the given time
    budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total time budget in seconds.
    kwargs : dict
        Ignored – present for compatibility with the harness.

    Returns
    -------
    list[float]
        The best sequence discovered (non‑negative, sum>0).
    """
    np.random.seed(seed)
    start = time.time()
    # Keep a small safety margin for the harness.
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Initialisation – a fairly long root‑sequence.
    # ------------------------------------------------------------------
    n = 1500                         # moderate size: good balance speed/quality
    seq = _initial_root_sequence(n)  # already summed to 1

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, score={best_score:.6f}")

    # Pre‑compute the current maximal self‑convolution value.
    conv_seq = np.convolve(seq, seq)
    M = conv_seq.max()

    # ------------------------------------------------------------------
    # 2. LP‑based mixing phase.
    # ------------------------------------------------------------------
    lp_iter = 0
    # Values for the mixing weight `t`.  They are reduced as the optimisation
    # proceeds, but a back‑tracking line search guarantees a decrease of M.
    t_initial = 0.4
    while time.time() < deadline:
        lp_iter += 1
        # Build the convolution matrix for the current `seq`.
        A = _build_convolution_matrix(seq)

        # Linear program: maximise Σ b_i  <=> minimise -Σ b_i.
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)  # (f * b)[k] ≤ M
        bounds = [(0.0, None)] * n

        # Solve the LP.
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            # In rare cases the LP might fail (numerics).  Switch to a pure
            # gradient descent fallback.
            print(f"[LP] iteration {lp_iter} failed ({res.message}); aborting LP phase.")
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0.0:
            print("[LP] Trivial solution encountered, stopping LP phase.")
            break

        # Scale to unit sum to obtain a feasible candidate `g`.
        g = b_opt / sum_b

        # --------------------------------------------------------------
        # Mixing: f ← (1‑t)·f + t·g  with a line search on `t`.
        # --------------------------------------------------------------
        t = t_initial * (0.9 ** (lp_iter // 10))   # slowly decay the step size
        while True:
            new_seq = (1.0 - t) * seq + t * g
            new_M = np.convolve(new_seq, new_seq).max()
            if new_M < M or t < 1e-4:
                break
            t *= 0.5                               # back‑track

        # Accept the new point.
        seq = new_seq
        M = new_M

        # Evaluate the current sequence.
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[improve LP] iter={lp_iter}, t={t:.4f}, score={best_score:.6f}")

        # If the LP phase becomes too slow (too many iterations) we break.
        if lp_iter >= 40:     # about 40 LP solves is plenty for the time budget
            break

        # Safety check for the time budget.
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3. Local projected‑gradient refinement (optional, cheap).
    # ------------------------------------------------------------------
    grad_iter = 0
    step = 0.05            # start step size for gradient descent
    decay = 0.9999
    while time.time() < deadline:
        grad_iter += 1
        seq = _projected_gradient_step(seq, step)
        step = max(step * decay, 1e-7)

        # Periodically evaluate – cheap because the sequence length is fixed.
        if grad_iter % 10 == 0 or time.time() + 0.5 >= deadline:
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[improve G] iter={grad_iter}, step={step:.6f}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence found.
    # ------------------------------------------------------------------
    total_time = time.time() - start
    print(f"[final] total_time={total_time:.1f}s, best_score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
