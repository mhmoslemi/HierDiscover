```
def softmax_gradient_step(a, step, beta, fft_len):
    conv = np.fft.irfft(np.fft.rfft(a, n=fft_len) * np.fft.rfft(a, n=fft_len), n=fft_len)
    conv = conv[: 2 * len(a) - 1]

    # Compute softmax weights
    maxc = np.max(conv)
    exp_weights = np.exp(beta * (conv - maxc))
    Z = np.sum(exp_weights)
    w = exp_weights / Z   # normalized

    a_rev = a[::-1]

    # correlation: convolve w with a_rev
    # Use fft
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    corr = np.fft.irfft(fw * fa_rev, n=fft_len)
    corr = corr[: len(w) + len(a_rev) - 1]   # length 3n-2

    n = len(a)
    # gradient = 2 * correlation[n-1 - np.arange(n)]
    idx = n-1 - np.arange(n)
    grad = 2.0 * corr[idx]   # idx reversed slice

    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    sum_a = a_new.sum()
    if sum_a == 0:
        a_new = np.full_like(a, 1.0 / len(a))
    else:
        a_new /= sum_a
    return a_new
```

```
    beta_max = 30.0
    beta_inc = 0.02   # incremental
    step = 0.5
    step_decay = 0.9995

    # Determine FFT lengths
    conv_len = 2*n - 1
    fft_len_conv = 1 << ((conv_len).bit_length())
    corr_len = 3*n - 2
    fft_len_corr = 1 << ((corr_len).bit_length())

    iteration = 0
    while time.time() < deadline and step > 1e-6:
        # compute convolution
        r_fft = np.fft.rfft(a, n=fft_len_conv)
        conv_full = np.fft.irfft(r_fft * r_fft, n=fft_len_conv)[:conv_len]   # real

        # compute softmax weights
        # increasing beta gradually
        if beta < beta_max:
            beta = min(beta_max, beta + beta_inc)

        max_conv = np.max(conv_full)
        # subtract max for numerical stability
        exp_weights = np.exp(beta * (conv_full - max_conv))
        Z = np.sum(exp_weights)
        if Z == 0:
            # fallback uniform
            w = np.full_like(conv_full, 1.0 / conv_len)
        else:
            w = exp_weights / Z

        # correlation (w * a_rev)
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=fft_len_corr)
        fa_rev = np.fft.rfft(a_rev, n=fft_len_corr)
        corr_full = np.fft.irfft(fw * fa_rev, n=fft_len_corr)[:corr_len]

        idx = n - 1 - np.arange(n)
        grad = 2.0 * corr_full[idx]

        a_new = a - step * grad
        # projection
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0:
            a_new = np.full_like(a, 1.0/n)
        else:
            a_new /= s

        a = a_new

        if iteration % 10 == 0:
            cur_score = evaluate_sequence(a.tolist())
```

Let's write final answer code as block.assistantfinal**Solution Overview**

The evaluation function is  

\[
F(a)=\frac{2n\;\max_k (a*a)[k]}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

If we keep the vector normalised to  

\[
\sum_i a_i = \sqrt{2n},
\]

the denominator equals the numerator’s factor `2 n` and the score reduces to  

\[
F(a)=\max_k (a*a)[k] .
\]

Hence the whole problem is **“make the self‑convolution as flat as possible”**.

The classic Catalan sequence (coefficients of \((1-x)^{-1/2}\)) already gives a very flat convolution (≈ π⁄2).  
To improve it we use a *smooth* optimisation of the quantity  
\(\max_k (a*a)[k]\).

* **Soft‑max surrogate** – instead of the non‑smooth max we minimise  

\[
L_\beta(a)=\frac{1}{\beta}\log\!\Bigl(\sum_k\exp(\beta\,(a*a)[k])\Bigr) .
\]

For large β the surrogate coincides with the true maximum, but its gradient is well defined:

\[
\frac{\partial L_\beta}{\partial a_i}= 
\frac{2}{Z}\sum_{k} \exp(\beta\,(a*a)[k])\; a_{k-i},
\qquad 
Z=\sum_{k}\exp(\beta\,(a*a)[k]).
\]

The gradient can be evaluated in \(O(n\log n)\) with FFTs (convolution + correlation).

* **Projected gradient descent** – after each step we
  1. clip negative entries,
  2. re‑normalise to the fixed sum \(\sqrt{2n}\).

* **Progressive β** – start with a small β (smooth landscape) and increase it slowly up to a moderate value (≈ 30).  

* **Final local search** – a cheap stochastic “move a tiny fraction of mass from one coordinate to another’’ phase (simulated‑annealing style) to escape tiny plateaus.

The whole optimisation runs until the 1000 s budget expires and always keeps the best vector seen so far.

```python
# -*- coding: utf-8 -*-
"""
Propose a step‑function that minimises the supplied ``evaluate_sequence`` metric.

The algorithm
--------------
1.  Initialise with the Catalan (almost flat) coefficients and scale them
    so that   sum(a) = sqrt(2·n).  With this scaling the score equals
    max(convolution).

2.  Perform projected gradient descent on the smooth surrogate
    L_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k]).
    * The convolution (a*a) and the correlation needed for the gradient
      are computed by FFT (O(n log n)).
    * β is increased slowly from 1 to a preset maximum (≈30).
    * After each step a is projected onto the non‑negative orthant and
      renormalised to the fixed sum.

3.  When the gradient phase finishes, a short stochastic local‑search
    (mass‑transfer moves with a tiny temperature) refines the solution.

All intermediate results are evaluated with the exact ``evaluate_sequence``
only sporadically; the bulk of the optimisation uses the cheap closed‑form
score ``max(convolution)``.

The function returns the best sequence found before the time budget runs
out.
"""

import numpy as np
import time
from typing import List


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def compute_catalan_sequence(n: int) -> np.ndarray:
    """
    First n Catalan‑type coefficients
        c_i = (2i choose i) / 4**i,   i = 0 … n‑1,
    normalised so that their sum is 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real). Result length = 2·len(a)‑1."""
    ra = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(ra * ra, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def correlation_fft(w: np.ndarray, a_rev: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Correlation (w * a_rev) using FFT.
    Length of the full linear correlation = len(w) + len(a_rev) - 1.
    """
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    corr = np.fft.irfft(fw * fa, n=fft_len)
    return corr[: w.shape[0] + a_rev.shape[0] - 1]   # linear part


def softmax_gradient_step(
    a: np.ndarray,
    step: float,
    beta: float,
    fft_len_conv: int,
    fft_len_corr: int,
    target_sum: float,
) -> np.ndarray:
    """
    One projected gradient step for the soft‑max surrogate L_β.
    Returns the new a (already projected and renormalised to target_sum).
    """
    n = a.shape[0]

    # ---- convolution a*a (needed for soft‑max weights) ----
    conv = full_convolution_fft(a, fft_len_conv)          # length 2n‑1

    # ---- soft‑max weights ----
    max_c = np.max(conv)
    # for numerical stability shift by max_c
    w_unnorm = np.exp(beta * (conv - max_c))
    Z = np.sum(w_unnorm)
    if Z == 0.0:                         # should not happen, but guard
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = w_unnorm / Z                 # normalised weights, sum to 1

    # ---- correlation (w * a_rev) -> gradient ----
    a_rev = a[::-1]
    corr = correlation_fft(w, a_rev, fft_len_corr)      # length 3n‑2

    # gradient_i = 2 * corr[n-1-i]   (derived in the analysis)
    idx = n - 1 - np.arange(n)
    grad = 2.0 * corr[idx]

    # ---- gradient step + projection ----
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)       # project onto non‑negative orthant

    s = a_new.sum()
    if s == 0.0:                         # degenerate case – fallback to uniform
        a_new = np.full_like(a, target_sum / n)
    else:
        a_new = a_new * (target_sum / s)   # renormalise to the fixed sum

    return a_new


def random_perturb(a: np.ndarray, eps: float, target_sum: float) -> np.ndarray:
    """
    Transfer a fraction *eps* of the mass of a random index i to a different
    random index j (i ≠ j).  The vector stays normalised to *target_sum*.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    delta = eps * a[i]
    if delta <= 0.0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new *= target_sum / a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields a score
    below the current best known bound.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5            # safety margin

    # ------------------------------------------------------------------
    # Length of the vector – reuse previous best if available
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except NameError:
        n = 2500                             # reasonable default

    # Fixed sum  = sqrt(2·n)  → the score equals the maximal convolution entry
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1. Start from the Catalan (almost‑flat) distribution
    # ------------------------------------------------------------------
    a = compute_catalan_sequence(n)
    a = a * (target_sum / a.sum())          # scale to the required sum

    best_seq = a.copy()
    # exact score is simply max(convolution) because of the normalisation
    conv = full_convolution_fft(a, next_pow2(2 * n - 1))
    best_score = np.max(conv)               # lower is better
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute FFT lengths for the two transforms we need repeatedly
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)          # for a*a
    fft_len_corr = next_pow2(3 * n - 2)          # for (w * a_rev)

    # ------------------------------------------------------------------
    # 2. Projected gradient descent on the soft‑max surrogate
    # ------------------------------------------------------------------
    beta = 1.0
    beta_max = 30.0
    beta_inc = 0.02            # how fast we tighten the surrogate

    step = 0.5                 # initial step size
    step_decay = 0.9995

    it = 0
    while time.time() < deadline and step > 1e-7:
        # ----- gradient step -----
        a = softmax_gradient_step(
            a,
            step,
            beta,
            fft_len_conv,
            fft_len_corr,
            target_sum,
        )

        # ---- update surrogate smoothness ----
        if beta < beta_max:
            beta = min(beta_max, beta + beta_inc)

        # ---- occasional exact evaluation & book‑keeping ----
        if it % 20 == 0:
            conv = full_convolution_fft(a, fft_len_conv)
            cur_score = np.max(conv)          # true objective value
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[grad] iter {it:5d} – improved score {best_score:.6f}")

        # ---- step size schedule ----
        step *= step_decay
        it += 1

    # ------------------------------------------------------------------
    # 3. Stochastic local‑search phase (tiny mass transfers)
    # ------------------------------------------------------------------
    eps = 0.08               # start with moving up to 8 % of a source entry
    temperature = 0.08
    it_local = 0
    while time.time() < deadline:
        cand = random_perturb(a, eps, target_sum)
        conv_cand = full_convolution_fft(cand, fft_len_conv)
        cand_score = np.max(conv_cand)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand
            print(f"[pert] iter {it_local:5d} – improved score {best_score:.6f}")
        else:
            # uphill acceptance (simulated annealing)
            prob = np.exp(-(cand_score - best_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                a = cand

        # decay move size and temperature
        eps = max(eps * 0.9999, 1e-4)
        temperature *= 0.9997
        it_local += 1

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python list)
    # ------------------------------------------------------------------
    final_seq = np.clip(best_seq, 0.0, 1000.0)   # final safety clamp
    # final sanity check – optional (very cheap)
    # print("final score =", np.max(full_convolution_fft(final_seq, fft_len_conv)))
    return final_seq.tolist()
```
