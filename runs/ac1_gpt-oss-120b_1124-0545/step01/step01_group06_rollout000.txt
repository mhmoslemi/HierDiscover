def propose_candidate(budget_s=1000, seed=0):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5

    best_seq = None
    best_val = float('inf')
    lengths = [2000, 4000, 8000]
    # Parameter for beta scheduling
    beta_final = 200.0
    beta_init = 1.0

    for n in lengths:
        # Use initial distributions
        init_seqs = []

        # Include uniform
        init_seqs.append(np.ones(n)/n)

        # Include geometric family (we can compute r ~ 0.99)
        r = 0.99
        init_seqs.append(_geom(n, r))

        # Include previous best if length matches
        if best_seq is not None and len(best_seq) == n:
            init_seqs.append(best_seq.copy())
        elif 'height_sequence_1' in globals():
            hs = np.array(globals()['height_sequence_1'])
            if len(hs) != n:
                # Resample via interpolation to n
                hs_interp = np.interp(np.linspace(0, len(hs)-1, n), np.arange(len(hs)), hs)
                hs_interp /= hs_interp.sum()
                init_seqs.append(hs_interp)
            else:
                init_seqs.append(hs.copy())

        # Add a few random Dirichlet
        for _ in range(3):
            a = np.random.dirichlet(np.ones(n) * 0.5)  # sparser
            init_seqs.append(a)

        for a0 in init_seqs:
            if time.time() > deadline:
                break
            # run gradient descent from a0
            a = a0.copy()
            # convert to softmax variables z: take log
            # To avoid zeros, add epsilon.
            eps = 1e-12
            z = np.log(a + eps)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            beta1 = 0.9
            beta2 = 0.999
            eps_adam = 1e-8
            # schedule of beta (smoothness)
            # linear increase over iterations upto beta_final
            max_iters = 100000  # but break early on time
            for it in range(max_iters):
                # Time check each 1000 iter
                if it % 500 == 0 and time.time() > deadline:
                    break
                # compute a as softmax of z
                # using stable method
                z_shift = z - np.max(z)
                expz = np.exp(z_shift)
                a = expz / expz.sum()
                # Compute conv (full)
                conv = _conv_fft(a, a)
                cur_max = np.max(conv)
                cur_val = 2.0 * n * cur_max
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[best] n={n} iter={it} val={cur_val:.6f}")
                # Compute beta
                beta = beta_init + (beta_final - beta_init) * it / max_iters
                # Smoothed softmax weights for conv
                shift = cur_max
                # compute exp of beta*(conv - shift)
                # To avoid overflow for large beta, use np.exp with shift.
                w = np.exp(beta * (conv - shift))
                w_sum = w.sum()
                if w_sum == 0:
                    # fallback to uniform weights if overflow
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w = w / w_sum

                # Gradient of smoothed objective

Propose a step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Coarse parametric sweep** – a few families (geometric, power‑law,
   combined geometric×power, symmetric geometric) are evaluated for a few
   candidate lengths.  The best candidate of each length is kept.

2. **Mirror‑descent refinement** – each candidate is transformed to
   soft‑max (log‑probability) variables so that the simplex constraint
   `a_i ≥ 0, Σ a_i = 1` is automatically satisfied.  We minimise a
   smooth approximation of the `max` using the log‑sum‑exp surrogate.
   The gradient of the surrogate is obtained analytically via FFT‑based
   convolutions.  An Adam optimiser (with a slowly decreasing learning
   rate) updates the log‑variables while a temperature parameter `β`
   (which controls the smoothness) is increased from a small value to a
   large one, thereby tightening the approximation to the true max.

3. **Random restarts & length schedule** – the refinement is run on a
   handful of initial points (the sweep best, a few Dirichlet draws and,
   if available, the historic `height_sequence_1`).  Several lengths
   (2000, 4000, 8000) are tried; the best overall sequence is returned.

The whole search respects the supplied time budget (default 1000 s).
All arithmetic is performed with NumPy; convolutions use an FFT of the
next power‑of‑two for speed.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (top‑level – no closures or lambdas)
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution via FFT (real input → real output).
    Returns the full vector of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decreasing sequence, normalised to sum 1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ 1 / i^p  (i starts at 1), normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Combined geometric × power‑law, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    if s == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (mirror) sequence, normalised."""
    a = r ** np.arange(n, dtype=np.float64) + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    a /= a.sum()
    return a


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    shift = np.max(z)
    expz = np.exp(z - shift)
    return expz / expz.sum()


def _evaluate_fast(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective when `a` already sums to 1.
    Returns 2 * n * max(conv(a, a)).
    """
    conv = _conv_fft(a, a)
    return float(2.0 * len(a) * np.max(conv))

            v = beta2 * v + (1 - beta2) * (grad_z ** 2)
            m_hat = m / (1 - beta1 ** (it + 1))
            v_hat = v / (1 - beta2 ** (it + 1))

            # learning‑rate schedule (slow decay)
            lr = 0.7 / np.sqrt(it + 1.0)

            # update logits
            z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # optional early stop if gradient tiny
            if np.linalg.norm(grad_z) < 1e-12:
                break

    # --------------------------------------------------------
    # 4. Fallback (should never happen)
    # --------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.ones(n, dtype=np.float64) / n
        best_val = _evaluate_fast(best_seq)

    print(f"Final best value (approx, sum‑1): {best_val:.6f} on length {len(best_seq)}")
    return best_seq.tolist()
