"""
Propose a step‑function height vector that minimizes the evaluation metric.

The algorithm is a hybrid of:
1. A fast simulated‑annealing / sub‑gradient walk on the ℓ₁‑simplex
   (the same idea as the previous baseline).
2. Periodic linear‑programming refinements.  Given the current candidate
   `f`, we solve a LP that maximises Σ g_j under the linear constraints
   (f * g)[k] ≤ M where M = max(f*f).  The solution g is then scaled to
   the required ℓ₁‑norm and mixed back into `f`.  This LP step finds
   extreme points that push mass into positions that are “inactive’’ for
   the convolution, which tends to flatten the self‑convolution and
   lowers the maximum entry.

The search runs until the allotted time budget expires, always keeping
the best vector seen.  All vectors are projected onto the simplex
{ x ≥ 0, Σ x = √(2·n) } after every update to respect the problem’s
scale‑invariance.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex { w ≥ 0 , Σ w = s }."""
    # Wang & Carreira‑Perpiñán (2013) algorithm
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform point
        return np.full_like(v, s / n, dtype=float)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _lp_refine_g(f: np.ndarray, M: float, target_sum: float):
    """
    Solve the LP:
        maximise   Σ g_j
        subject to (f * g)[k] ≤ M   for all k,
                   g_j ≥ 0,
                   g_j ≤ target_sum   (to avoid unboundedness).

    Returns the optimal g (may have Σ g_j > target_sum) or None on failure.
    """
    n = f.size
    try:
        from scipy.linalg import toeplitz
        from scipy.optimize import linprog
    except Exception:   # SciPy not available
        return None

    # Build the full convolution matrix A where A @ g = np.convolve(f, g)
    # A has shape (2*n-1, n)
    first_col = np.concatenate([f, np.zeros(n - 1, dtype=float)])
    first_row = np.concatenate([np.array([f[0]], dtype=float), np.zeros(n - 1, dtype=float)])
    A = toeplitz(first_col, first_row)          # dense (2n‑1)×n matrix

    # Objective: maximise Σ g  →  minimise  -Σ g
    c = -np.ones(n, dtype=float)

    # Inequality constraints: A @ g ≤ M  (one per convolution entry)
    b_ub = np.full(2 * n - 1, M, dtype=float)

    # Upper bound on each variable to keep the LP bounded
    var_bounds = [(0.0, target_sum) for _ in range(n)]

    # Solve
    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=var_bounds,
        method='highs',
        options={'presolve': True, 'time_limit': 5.0}  # tiny per‑LP limit
    )
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function height vector that minimises the given
    evaluation function within the supplied time budget.

    Returns the best sequence (as a plain Python list) discovered.
    """
    np.random.seed(seed)
    random.seed(seed)

    # --------------------------------------------------------------
    # Problem size – reuse the known construction if present.
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        init = np.array(height_sequence_1, dtype=float)
    else:
        n = 1500                     # a reasonable default
        init = np.full(n, 1.0)

    target_sum = math.sqrt(2.0 * n)       # the normalising ℓ₁‑norm

    # Normalise the initial vector onto the simplex
    init = init * (target_sum / init.sum())
    init = _simplex_projection(init, target_sum)

    best_seq = init.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # Occasionally try a completely random start (helps escape bad basins)
    rand_seq = np.random.exponential(scale=1.0, size=n)
    rand_seq = rand_seq * (target_sum / rand_seq.sum())
    rand_score = evaluate_sequence(rand_seq.tolist())
    if rand_score < best_score:
        best_seq = rand_seq.copy()
        best_score = rand_score

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin
    iteration = 0

    # Simulated‑annealing schedule
    T_start = 1.0
    T_end = 1e-4

    # LP refinement schedule (every `lp_interval` iterations)
    lp_interval = 3000

    while time.time() < deadline:
        # ----- temperature -----
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        T = T_start * ((T_end / T_start) ** frac)   # exponential cooling

        # ----------------------------------------------------------
        # 1) Small random transfer on the simplex (i → j)
        proposal = best_seq.copy()
        i, j = np.random.randint(n, size=2)
        if i != j:
            # magnitude proportional to the current temperature
            delta = np.random.normal(loc=0.0, scale=target_sum * 0.02 * T)
            delta = max(-proposal[i], min(delta, proposal[i]))   # keep non‑negative
            proposal[i] -= delta
            proposal[j] += delta

        # Re‑project onto the simplex (keeps the ℓ₁‑norm exact)
        proposal = _simplex_projection(proposal, target_sum)

        # ----------------------------------------------------------
        # 2) Sub‑gradient step aimed at the currently largest convolution entry
        conv = np.convolve(proposal, proposal)
        k_max = int(np.argmax(conv))

        # Gradient of conv[k_max] w.r.t. proposal: g_i = proposal[k_max - i] (if in range)
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        gradient = np.zeros_like(proposal)
        gradient[mask] = proposal[partner[mask]]

        # Diminishing step size, aligned with the temperature schedule
        eta = target_sum * 0.1 * T
        proposal -= eta * gradient
        proposal = _simplex_projection(proposal, target_sum)

        # ----------------------------------------------------------
        # 3) Periodic LP refinement
        if (iteration % lp_interval == 0) and (iteration > 0):
            # Compute the current bound M = max(conv(f,f))
            cur_conv = np.convolve(best_seq, best_seq)
            M_cur = float(cur_conv.max())
            g0 = _lp_refine_g(best_seq, M_cur, target_sum)
            if g0 is not None:
                sum_g0 = float(g0.sum())
                if sum_g0 > 1e-12:
                    # Scale to the target ℓ₁‑norm
                    g = g0 * (target_sum / sum_g0)

                    # 3a) Try the pure LP candidate
                    score_g = evaluate_sequence(g.tolist())
                    if score_g < best_score:
                        best_seq = g.copy()
                        best_score = score_g
                        print(f"[{iteration:6d}] LP candidate improved score to {best_score:.6f}")

                    else:
                        # 3b) Mix the LP candidate with the current best
                        tau = 0.5   # fixed mixing weight
                        mixed = (1.0 - tau) * best_seq + tau * g
                        mixed = _simplex_projection(mixed, target_sum)
                        score_mix = evaluate_sequence(mixed.tolist())
                        if score_mix < best_score:
                            best_seq = mixed.copy()
                            best_score = score_mix
                            print(f"[{iteration:6d}] LP‑mix improved score to {best_score:.6f}")

        # ----------------------------------------------------------
        # 4) Evaluate the (pure) proposal from the SA step
        try:
            score = evaluate_sequence(proposal.tolist())
        except Exception:
            # Any numerical oddities – just skip this candidate
            continue

        # Accept improvements outright; otherwise accept uphill moves with SA probability
        if score < best_score or np.random.rand() < math.exp(-(score - best_score) / (T + 1e-12)):
            if score < best_score:
                best_score = score
                best_seq = proposal.copy()
                print(f"[{iteration:6d}] New best from SA: {best_score:.6f} (time {elapsed:.1f}s)")
            else:
                # occasional uphill move to diversify the search
                best_seq = proposal.copy()

        # ----------------------------------------------------------
        # 5) Occasionally perform a purely random mass‑transfer move
        if iteration % 73 == 0:   # a cheap way to add stochasticity
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = np.random.rand() * best_seq[i]
                cand = best_seq.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _simplex_projection(cand, target_sum)
                sc = evaluate_sequence(cand.tolist())
                if sc < best_score:
                    best_seq = cand
                    best_score = sc
                    print(f"[{iteration:6d}] Random‑move improved score to {best_score:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    return best_seq.tolist()
