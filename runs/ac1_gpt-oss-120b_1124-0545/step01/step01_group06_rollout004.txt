    x = np.log(a0 + 1e-12)  # initially logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    beta = beta0
    for t in range(1, n_iters+1):
        a = np.exp(x)
        a_sum = a.sum()
        a = a / a_sum
        # conv a*a
        conv = _conv_fft(a, a)
        cur_max = conv.max()
        # test actual eval maybe
        # compute weighted w
        w = np.exp(beta * (conv - cur_max))
        w_sum = w.sum()
        if w_sum == 0:
            continue
        w = w / w_sum
        # gradient wrt a
        grad_a = 2.0 * _conv_fft(a[::-1], w)[n-1:n-1+n]
        # gradient wrt x (softmax):
        # dL/dx_i = a_i * (grad_a_i - np.dot(grad_a, a))
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1-beta1) * grad_x
        v = beta2 * v + (1-beta2) * (grad_x**2)

        v_hat = v / (1 - beta2**t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        # maybe increase beta gradually
        beta *= beta_inc
        # maybe monitor evaluate_sequence

def _adam_optimize(a0, deadline, max_iters=5000):
    a = a0.copy()
    n = len(a)
    # Precompute FFT sizes
    N1 = _next_pow2(2*n-1)   # for autocorrelation
    N2 = _next_pow2(3*n-2)   # for gradient conv
    beta = 5.0
    beta_factor = 1.015
    max_beta = 200.0
    lr = 0.2
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_val_local = float('inf')
    best_a_local = a.copy()
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break

        # compute auto convolution
        fft_a = np.fft.rfft(a, n=N2)  # use N2 to have enough size
        conv = np.fft.irfft(fft_a * fft_a, n=N2)[:2*n-1]
        cur_max = conv.max()
        val = 2.0 * n * cur_max
        if val < best_val_local:
            best_val_local = val
            best_a_local = a.copy()
        # compute w (softmax)
        shift = cur_max
        w_raw = np.exp(beta * (conv - shift))
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            # fallback try smaller beta?
            beta = max(beta/2, 1.0)
            continue
        w = w_raw / w_sum

        # gradient wrt a
        fft_rev = np.fft.rfft(a[::-1], n=N2)
        fft_w = np.fft.rfft(w, n=N2)
        grad_full = np.fft.irfft(fft_rev * fft_w, n=N2)[:3*n-2]
        grad_a = 2.0 * grad_full[n-1:n-1+n]

        # gradient wrt logits (softmax)
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        x = np.log(a + 1e-20) - lr * m_hat / (np.sqrt(v_hat) + eps)
        # reconstitute a via softmax
        x -= np.max(x)  # for numeric stability
        a = np.exp(x)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.ones_like(a) / n
        else:
            a = a / a_sum

        # schedule beta
        beta = min(max_beta, beta * beta_factor)

    # after loop, return best local a and its evaluation
    return best_a_local, best_val_local

    """Assumes sum(a) == 1, returns 2 * n * max(conv(a,a))."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fft_a * fft_a, n=N)[:2 * n - 1]
    return float(2.0 * n * np.max(conv))

def _geom(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    total = a.sum()
    if total == 0:
        a = np.ones(n) / n
    else:
        a /= total
    return a

def _power(n, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    total = a.sum()
    if total == 0:
        a = np.ones(n) / n
    else:
        a /= total
    return a

def _geom_power(n, r, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    total = a.sum()
    if total == 0:
        a = np.ones(n) / n
    else:
        a /= total
    return a

def _sym_geom(n, r):
    a = r ** np.arange(n, dtype=np.float64) + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    a /= a.sum()
    return a

def _beta_shape(n, alpha, beta):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** alpha) * ((n - i + 1) ** beta)
    total = a.sum()
    if total == 0:
        a = np.ones(n) / n
    else:
        a /= total
    return a

def _adam_optimize(a0, deadline, max_iters=5000, beta0=5.0, beta_factor=1.015, max_beta=200.0,
                   lr=0.2, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam on log‑softmax parametrisation, minimizing a smooth max of the autocorrelation."""
    a = a0.copy()
    n = len(a)
    N2 = _next_pow2(3 * n - 2)   # size sufficient for gradient convolution
    beta = beta0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_val = float('inf')
    best_a = a.copy()
    # Pre‑compute indices for slicing
    slice_start = n - 1
    slice_end = slice_start + n
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- autocorrelation -------------------------------------------------
        fft_a = np.fft.rfft(a, n=N2)
        conv = np.fft.irfft(fft_a * fft_a, n=N2)[:2 * n - 1]
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # occasional progress output
            if it % 100 == 0:
                print(f"[adam] it={it:4d} val={cur_val:.6f}")

        # ----- smoothed‑max weights -------------------------------------------
        # shift = cur_max ensures exponent ≤ 0
        w = np.exp(beta * (conv - cur_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            # degenerate case, lower beta to regain diversity
            beta = max(beta / 2.0, 1.0)
            continue
        w /= w_sum

        # ----- gradient wrt a --------------------------------------------------
        fft_rev = np.fft.rfft(a[::-1], n=N2)
        fft_w = np.fft.rfft(w, n=N2)
        grad_full = np.fft.irfft(fft_rev * fft_w, n=N2)[:3 * n - 2]
        grad_a = 2.0 * grad_full[slice_start:slice_end]

        # ----- gradient wrt logits (softmax) -----------------------------------
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        # work in log‑space
        log_a = np.log(a + 1e-30)
        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # softmax re‑normalisation
        log_a -= np.max(log_a)            # numerical stability
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.ones_like(a) / n
        else:
            a /= a_sum

        # ----- beta schedule ---------------------------------------------------
        beta = min(max_beta, beta * beta_factor)

    return best_a, best_val

        # beta‑shape (symmetric polynomial)

    else:

        total = best_seq.sum()
        if total == 0.0:
            best_seq = np.ones_like(best_seq) / len(best_seq)
        else:
            best_seq = best_seq / total

    # Return plain Python list for the evaluation harness
    return best_seq.tolist()

        a_opt, val_opt = _adam_optimize(a0, deadline, max_iters=1200)
        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            print(f"[rand] length={n} new best val={val_opt:.6f}")

    # ------------------------------------------------------------------- #
    # final safety polishing
    # ------------------------------------------------------------------- #
    if best_seq is None:
        # fallback – uniform distribution
        n = 1024
        best_seq = np.ones(n, dtype=np.float64) / n
    else:
        best_seq = np.maximum(best_seq, 0.0)
        s = best_seq.sum()
        best_seq = best_seq / s if s > 0 else np.ones_like(best_seq) / len(best_seq)

    return best_seq.tolist()
