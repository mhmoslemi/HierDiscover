def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    n = 15000
    seq = _initial_root_sequence(n)  # sum=1

    best_seq = seq.copy()
    best_score = 2*n*np.max(np.convolve(seq, seq)) # compute

    # momentum
    v = np.zeros_like(seq)
    mu = 0.9
    step = 0.1
    decay = 0.99997  # step decay
    iter = 0

    while time.time() < deadline:
        iter += 1
        # compute convolution
        conv = np.convolve(seq, seq)
        max_b = conv.max()
        score = 2.0 * n * max_b

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[improve] iter={iter}, score={best_score:.6f}")

        # tolerance for active set: relative tolerance
        tol = max_b * 1e-5
        active = (conv >= max_b - tol).astype(float)

        # gradient via correlation
        grad = np.convolve(active, seq[::-1])  # length 3n-2
        grad = grad[n-1 : n-1 + n]  # slice length n
        grad = grad * 2.0  # factor 2 from quadratic derivative
        # project to hyperplane sum=1: remove mean
        grad -= grad.mean()

        # momentum update
        v = mu * v - step * grad
        seq = seq + v

        # projection: nonnegative
        seq = np.maximum(seq, 0.0)
        # renormalize sum to 1
        total = seq.sum()
        if total > 0:
            seq = seq / total
        else:
            seq = _initial_root_sequence(n)

        # step size decay
        step = max(step * decay, 1e-8)

        # random perturbation every so often to escape plateau
        if iter % 100 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = step * seq[i]  # small mass to move
            eps = min(eps, seq[i])
            seq[i] -= eps
            seq[j] += eps
            seq = np.maximum(seq, 0.0)
            seq = seq / seq.sum()

        # maybe restart from best if stagnation
        if iter % 5000 == 0:
            # small random perturb around best
            seq = best_seq.copy()
            # add some random jitter
            seq += step * np.random.randn(n) * seq
            seq = np.maximum(seq, 0.0)
            seq = seq / seq.sum()
            v = np.zeros_like(seq)

    print(f"[final] iterations={iter}, best_score={best_score:.6f}")
    # Convert best_seq to list and return
    return best_seq.tolist()

"""
Simple parameter search + momentum gradient descent for minimizing the evaluation function.
The algorithm:
1. Search over power-law exponents p and shift β for a root‑like sequence a_i∝(i+β)^{-p}.
2. Initialise a long sequence with the best parameters.
3. Perform a projected gradient descent using the full active set of the
   largest convolution entries, with momentum and occasional random mass
   transfers (simulated‑annealing style).  The sequence is always projected
   onto the simplex (non‑negative and sum=1).
4. Return the best sequence observed.
"""
import time
import numpy as np

def _param_sequence(p: float, beta: float, n: int) -> np.ndarray:
    """Return a normalised sequence a_i = (i+1+beta)^{-p}."""
    idx = np.arange(1, n + 1, dtype=float) + beta
    a = idx ** (-p)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / a_sum
    return a

def _fast_score(seq: np.ndarray) -> float:
    """Return the evaluation score (2*n*max(convolution)) assuming sum(seq)=1."""
    n = seq.size
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    return 2.0 * n * max_b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # small safety margin

    # -----------------------------------------------------------------------
    # 1. Coarse parameter sweep on a modest size (n_grid) to locate a good
    #    exponent p ≈ 0.5 and a small shift beta.
    # -----------------------------------------------------------------------
    n_grid = 8000
    p_vals = np.linspace(0.45, 0.55, 21)      # 0.45 … 0.55
    beta_vals = np.linspace(0.0, 5.0, 21)     # 0 … 5
    best_grid_seq = None
    best_grid_score = np.inf
    print("[grid] start parameter sweep")
    for p in p_vals:
        for beta in beta_vals:
            seq = _param_sequence(p, beta, n_grid)
            score = _fast_score(seq)
            if score < best_grid_score:
                best_grid_score = score
                best_grid_seq = seq.copy()
                best_p, best_beta = p, beta
                print(f"[grid] new best p={p:.4f} beta={beta:.3f} score={score:.6f}")

    print(f"[grid] done, best p={best_p:.4f} beta={best_beta:.3f} score={best_grid_score:.6f}")

    # -----------------------------------------------------------------------
    # 2. Build a larger sequence (n_big) using the best parameters.
    # -----------------------------------------------------------------------
    n_big = 15000
    seq = _param_sequence(best_p, best_beta, n_big)

    # Ensure the exact simplex constraint (sum = 1)
    seq = seq / seq.sum()

    best_seq = seq.copy()
    best_score = _fast_score(best_seq)
    print(f"[init] n={n_big}, score={best_score:.6f}")

    # -----------------------------------------------------------------------
    # 3. Projected gradient descent with momentum.
    # -----------------------------------------------------------------------
    v = np.zeros_like(seq)            # momentum term
    mu = 0.9                          # momentum coefficient
    step = 0.12                       # initial step size
    decay = 0.99997                   # step decay per iteration
    iteration = 0

    # hyper‑parameters for the random “mass‑swap’’ perturbation
    swap_every = 70
    swap_frac = 0.02                  # fraction of the source entry moved

    while time.time() < deadline:
        iteration += 1

        # 3.1 convolution and current score
        conv = np.convolve(seq, seq)
        max_b = conv.max()
        cur_score = 2.0 * n_big * max_b

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # 3.2 active set – a small number of the largest convolution entries
        # Use a relative tolerance to catch a handful of indices.
        tol = max_b * 1e-6
        act_idx = np.where(conv >= max_b - tol)[0]
        if act_idx.size > 10:
            # keep only the most extreme ones to keep the sub‑gradient informative
            act_idx = np.sort(act_idx)[-10:]

        # create an active mask (0/1) of length 2n-1
        active = np.zeros_like(conv, dtype=float)
        active[act_idx] = 1.0

        # 3.3 gradient of the (sub‑)max – correlation of active with seq
        # grad_i = Σ_k∈active a_{k-i}  (the factor 2 comes from the quadratic form)
        grad = np.convolve(active, seq[::-1])
        grad = grad[n_big-1 : n_big-1 + n_big]   # slice of length n
        grad = grad * 2.0
        # project to the hyperplane Σ a_i = 1 (by removing the mean)
        grad -= grad.mean()

        # 3.4 momentum update and projection onto the simplex
        v = mu * v - step * grad
        seq = seq + v

        # enforce non‑negativity
        seq = np.maximum(seq, 0.0)

        # renormalise to keep Σ a_i = 1 (the score is scale‑invariant)
        total = seq.sum()
        if total > 0:
            seq = seq / total
        else:
            # fallback – re‑initialise from the best known point
            seq = best_seq.copy()

        # decay the step size slowly
        step = max(step * decay, 1e-8)

        # occasional random mass exchange to break possible plateaus
        if iteration % swap_every == 0:
            i, j = np.random.choice(n_big, 2, replace=False)
            eps = swap_frac * seq[i]
            eps = min(eps, seq[i])
            seq[i] -= eps
            seq[j] += eps
            seq = np.maximum(seq, 0.0)
            seq = seq / seq.sum()

        # optional restart from best if we have stagnated for long
        if iteration % 5000 == 0:
            seq = best_seq.copy()
            # add a tiny jitter so that the optimiser can explore a neighbourhood
            jitter = np.random.randn(n_big) * step * 0.5
            seq += jitter
            seq = np.maximum(seq, 0.0)
            seq = seq / seq.sum()
            v = np.zeros_like(seq)

    print(f"[final] iterations={iteration}, best_score={best_score:.6f}")

    # Return the best sequence found (as a plain Python list)
    return best_seq.tolist()

"""
Simple parameter sweep + momentum projected gradient descent.

The algorithm works in three stages.

1.  A cheap grid search over power‑law exponent `p` and shift `β`
    for a root‑like family a_i ∝ (i+β)^{-p}.  The best (p,β) on a
    moderate size (n≈8000) gives a very good starting shape.

2.  The best parameters are used to build a longer sequence (n≈15000)
    which is then refined by a projected gradient descent on the
    max‑convolution objective.  The sub‑gradient is taken as the
    (scaled) correlation of the active set (the indices where the
    convolution is close to its maximum).  Momentum and a slowly
    decaying stepsize accelerate convergence.

3.  Occasional random “mass‑swap’’ moves and a small jittered restart
    keep the optimisation from getting stuck in plateaus.

All constraints required by `evaluate_sequence` (non‑negative,
finite numbers, sum>0) are enforced at every iteration.  The function
returns the best sequence discovered before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _param_sequence(p: float, beta: float, n: int) -> np.ndarray:
    """
    Produce a normalised sequence of length ``n`` with elements

        a_i = (i + 1 + beta)^{-p}   (i = 0,…,n‑1)

    The vector is scaled so that its L1‑norm equals 1.
    """
    idx = np.arange(1, n + 1, dtype=float) + beta          # i+1+β
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / s
    return a


def _fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective when ``seq`` is already normalised
    (sum(seq) == 1).  The score equals 2·n·max(conv(seq,seq)).
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑scoring non‑negative step‑function.
    The routine respects the 1000 s time budget and returns a list of
    floats (the best sequence found).
    """
    np.random.seed(seed)
    start_time = time.time()
    # a few seconds are kept as safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1) Coarse parameter sweep on a modest size
    # ------------------------------------------------------------------
    n_grid = 8000
    p_vals = np.linspace(0.45, 0.55, 21)          # 0.45 … 0.55
    beta_vals = np.linspace(0.0, 5.0, 21)         # 0 … 5
    best_grid_seq = None
    best_grid_score = np.inf

    print("[grid] start sweep (n=%d)" % n_grid)
    for p in p_vals:
        for beta in beta_vals:
            seq = _param_sequence(p, beta, n_grid)
            sc = _fast_score(seq)
            if sc < best_grid_score:
                best_grid_score = sc
                best_grid_seq = seq.copy()
                best_p, best_beta = p, beta
                print(f"[grid] new best  p={p:.4f}  β={beta:.3f}  score={sc:.6f}")

    print(f"[grid] completed – best p={best_p:.4f} β={best_beta:.3f} "
          f"score={best_grid_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Initialise a larger sequence with the best parameters
    # ------------------------------------------------------------------
    n_big = 15000
    seq = _param_sequence(best_p, best_beta, n_big)
    seq /= seq.sum()                     # enforce exact simplex

    best_seq = seq.copy()
    best_score = _fast_score(best_seq)
    print(f"[init] n={n_big}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Projected gradient descent with momentum
    # ------------------------------------------------------------------
    v = np.zeros_like(seq)                  # momentum buffer
    mu = 0.9                                 # momentum coefficient
    step = 0.12                              # initial step size
    decay = 0.99997                          # geometric decay of the step
    iteration = 0

    # parameters for occasional random perturbations
    swap_every = 70
    swap_frac = 0.02                         # fraction of source mass moved

    while time.time() < deadline:
        iteration += 1

        # ----- 3.1 current convolution and score --------------------
        conv = np.convolve(seq, seq)          # linear convolution, length 2n‑1
        max_b = conv.max()
        cur_score = 2.0 * n_big * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] iter={iteration} score={best_score:.6f}")

        # ----- 3.2 active set (indices contributing to the max) -----
        # Keep only a handful of the largest entries – this gives a
        # meaningful sub‑gradient while avoiding a completely flat direction.
        tol = max_b * 1e-6
        act_idx = np.where(conv >= max_b - tol)[0]
        if act_idx.size > 10:                # limit to at most 10 indices
            act_idx = np.sort(act_idx)[-10:]

        active = np.zeros_like(conv, dtype=float)
        active[act_idx] = 1.0

        # ----- 3.3 sub‑gradient of the max‑convolution -------------
        # grad_i = Σ_{k∈active} a_{k-i}  (factor 2 from the quadratic form)
        grad = np.convolve(active, seq[::-1])                     # length 3n‑2
        grad = grad[n_big - 1: n_big - 1 + n_big]                # slice length n
        grad *= 2.0
        grad -= grad.mean()                                      # keep Σa constant

        # ----- 3.4 momentum step + projection onto the simplex -------
        v = mu * v - step * grad
        seq = seq + v
        seq = np.maximum(seq, 0.0)                               # non‑negativity
        s = seq.sum()
        if s > 0.0:
            seq /= s
        else:
            # very unlikely – fall back to the best known point
            seq = best_seq.copy()

        # ----- 3.5 step‑size decay ----------------------------------
        step = max(step * decay, 1e-8)

        # ----- 3.6 occasional random “mass‑swap’’ move --------------
        if iteration % swap_every == 0:
            i, j = np.random.choice(n_big, 2, replace=False)
            eps = swap_frac * seq[i]
            eps = min(eps, seq[i])
            seq[i] -= eps
            seq[j] += eps
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()

        # ----- 3.7 restart with a tiny jitter if we have been idle ---
        if iteration % 5000 == 0:
            seq = best_seq.copy()
            jitter = np.random.randn(n_big) * step * 0.5
            seq += jitter
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            v = np.zeros_like(seq)

    print(f"[final] iterations={iteration}, best_score={best_score:.6f}")
    # The harness will call ``evaluate_sequence`` on the returned list,
    # but we already keep the sequence on the simplex, so the score is
    # directly comparable.
    return best_seq.tolist()
