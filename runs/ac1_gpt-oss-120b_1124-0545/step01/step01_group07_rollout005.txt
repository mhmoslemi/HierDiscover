    best_score = evaluate_sequence(best_seq.tolist())

    # Gradient descent loop
    while time.time() < deadline:
        # compute convolution b (full)
        b = np.convolve(cur, cur)
        max_b = b.max()
        # if max_b too close to best_score? We'll compute score = 2*n*max_b / (sum=1)^2 = 2*n*max_b
        # find index of max
        k0 = b.argmax()
        # compute gradient g_i = 2*cur[k0-i] for valid i
        n = len(cur)
        g = np.zeros_like(cur)
        for i in range(n):
            j = k0 - i
            if 0 <= j < n:
                g[i] = 2 * cur[j]
        # step size: we want to reduce max_b => move in direction -g.
        # Step size parameter adapt: start with alpha = 0.01 maybe relative to cur.max?
        # We'll try exponential backtracking: for factor in [1.0, 0.5, 0.25, 0.125] decreasing until improvement.
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            eta = factor * 0.01 / (np.linalg.norm(g) + 1e-12)  # small normalized step
            new = cur - eta * g
            new[new < 0] = 0.0
            # renorm to sum 1
            s = new.sum()
            if s < 1e-12:
                continue
            new = new / s
            new_score = evaluate_sequence(new.tolist())
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()
                cur = new
                improved = True
                print(f"[{time.time() - start_time:.1f}s] GD step improved score={best_score:.6f}")
                break
        if not improved:
            # if gradient step not improving, apply random perturbation (simulated annealing) to escape
            # small random adjustments
            idx = np.random.randint(len(cur))
            delta = np.random.normal(scale=cur.mean()*0.05)
            new = cur.copy()
            new[idx] = max(0.0, new[idx] + delta)
            new = new / new.sum()
            new_score = evaluate_sequence(new.tolist())
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()
                cur = new

    # bell-shaped (Lorentzian)

"""
Improved search for a step‑function minimizing

    2 * n * max(convolution(a,a)) / (sum(a)**2)

The scoring function is homogeneous, therefore we keep the L1 norm
fixed to 1 (sum == 1).  The algorithm consists of three phases:

1️⃣  Systematic exploration of many analytically‑defined shape families
    (constant, linear, quadratic, triangular, Lorentzian, sine‑based,
    exponential, power‑law, Gaussian, semicircle, sqrt‑product and a few
    custom parametrised families).  Each shape is normalised to a unit
    sum and evaluated via a fast FFT‑based convolution.  The best sequence
    discovered so far is kept.

2️⃣  Gradient‑descent / “peak‑flattening” optimisation.
    The current maximum of the autocorrelation is identified; its
    gradient (∂/∂a_i max (a*a) ) is simply a reversed copy of the
    original sequence around the peak.  A back‑tracking line‑search
    (several step‑size candidates) proposes a new non‑negative,
    renormalised candidate.  Improvements are always kept; otherwise a
    simulated‑annealing acceptance rule (temperature decays linearly)
    allows occasional worsening steps.

3️⃣  Randomised refinement (Dirichlet mixing, block reversal) to escape
    possible plateaus.  All steps respect the 1000 s time budget.

The code returns the best sequence found before the deadline; if the
target score 1.503 is reached the search stops early.

import time

def _normalise_to_one(a: np.ndarray) -> np.ndarray:
    """Scale non‑negative vector to have L1 norm equal to 1.
    Very small entries are left untouched; zeros stay zero."""
    a = np.asarray(a, dtype=float)
    total = a.sum()
    if total <= 0.0:
        # fallback to a uniform tiny vector to avoid division by zero
        a = np.full_like(a, 1.0 / a.size)
        return a
    return a / total


def _score_fft(a: np.ndarray) -> float:
    """Fast computation of the evaluation score for a sequence with sum(a)=1.
    Uses FFT‑based convolution (real input)."""
    n = a.shape[0]
    # length for a power‑of‑two FFT (covers 2*n-1 points)
    nfft = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b


def _autocorr_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[: 2 * n - 1]
    return conv


def _gradient_from_peaks(a: np.ndarray, b: np.ndarray, rel_thr: float = 0.9995) -> np.ndarray:
    """
    Gradient of the max‑peak of the autocorrelation.
    All indices whose value exceeds rel_thr * max(b) contribute.
    For a single peak at index k the gradient is:
        g_i = 2 * a_{k-i}   (where the index is in range)
    """
    n = a.shape[0]
    max_b = b.max()
    if max_b <= 0:
        return np.zeros_like(a)
    thresh = max_b * rel_thr
    peak_idxs = np.where(b >= thresh)[0]

    g = np.zeros_like(a)
    for k in peak_idxs:
        if k < n:
            # i runs 0 .. k
            length = k + 1
            g[:length] += 2.0 * a[:length][::-1]
        else:
            # i runs k-(n-1) .. n-1
            start = k - (n - 1)
            length = 2 * n - 1 - k
            g[start : start + length] += 2.0 * a[:length][::-1]
    return g


# ----------------------------------------------------------------------
# Shape generators (families with a few parameters)
# ----------------------------------------------------------------------


def _generate_shapes(n: int):
    """Yield a collection of candidate shape vectors of length n."""
    # 1 – constant
    yield np.ones(n)

    # 2 – linear (decreasing / increasing)
    yield np.arange(n, 0, -1)
    yield np.arange(1, n + 1)

    # 3 – quadratic (decreasing / increasing)
    yield np.square(np.arange(n, 0, -1))
    yield np.square(np.arange(1, n + 1))

    # 4 – triangular “mountain”
    t = np.arange(1, n + 1, dtype=float)
    yield np.minimum(t, n - t + 1)

    # 5 – Lorentzian‑like bump (large centre, small edges)
    centre = (n - 1) / 2.0
    yield 1.0 / (1.0 + ((np.arange(n, dtype=float) - centre) ** 2))

    # 6 – sine‑squared (smooth bell)
    yield np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) / (n + 1)) ** 2

    # 7 – exponential decays
    for r in (0.95, 0.9, 0.8, 0.7, 0.6, 0.5):
        yield r ** np.arange(n, dtype=float)

    # 8 – power‑law decays
    for p in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n + 1, dtype=float)
        yield 1.0 / (i ** p)

    # 9 – Gaussian families
    mu = (n - 1) / 2.0
    for sigma_factor in (0.05, 0.07, 0.10, 0.15, 0.20, 0.30):
        sigma = sigma_factor * n
        g = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
        yield g

    # 10 – semicircle (Arch) shape
    radius = (n - 1) / 2.0
    semic = np.sqrt(np.clip(1.0 - ((np.arange(n, dtype=float) - mu) / radius) ** 2, 0.0, None))
    yield semic

    # 11 – sqrt‑product (discrete approximation of a half‑ellipse)
    i = np.arange(1, n + 1, dtype=float)
    sqrtprod = np.sqrt(i * (n - i + 1))
    yield sqrtprod

    # 12 – sin^p families (p controls concentration)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0):
        s = np.sin(np.pi * (np.arange(1, n + 1, dtype=float) / (n + 1))) ** p
        yield s

    # 13 – logarithmic growth (slowly rising)
    logshape = np.log(np.arange(2, n + 2, dtype=float))
    yield logshape

    # 14 – reversed logarithmic (slowly descending)
    revlog = np.log(np.arange(2, n + 2, dtype=float)[::-1])
    yield revlog


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative coefficient list that minimizes ``evaluate_sequence``.
    The function respects the supplied time budget and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1          # safety margin

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously saved construction if present
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        cur = _normalise_to_one(np.asarray(hs1, dtype=float))
        best_seq = cur.copy()
        best_score = _score_fft(cur)
        print(f"[{time.time() - start_time:.1f}s] Warm‑start from height_sequence_1 "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – systematic parametrised search
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    for n in lengths:
        if time.time() >= deadline:
            break
        for shape in _generate_shapes(n):
            if time.time() >= deadline:
                break
            seq = _normalise_to_one(shape)
            # quick score without calling evaluate_sequence
            score = _score_fft(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.1f}s] New best ({n} terms) "
                      f"score={best_score:.6f}")
            if best_score <= 1.503:
                print("Target reached in Phase 1 – stopping early.")
                break
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – if nothing was accepted (should not happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = rng.random(1024)
        best_seq = _normalise_to_one(best_seq)
        best_score = _score_fft(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – gradient‑descent “peak flattening”
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    n = cur.shape[0]

    # temperature schedule for simulated‑annealing acceptance
    start_temp, end_temp = 1.0, 0.001

    while time.time() < deadline:
        elapsed = time.time() - start_time
        T = start_temp * (1 - elapsed / (deadline - start_time)) + end_temp * (elapsed / (deadline - start_time))

        # compute autocorrelation and its peak(s)
        b = _autocorr_fft(cur)
        max_b = b.max()
        score_cur = 2.0 * n * max_b  # exact score because sum(cur)=1
        # gradient from (almost) maximal indices
        grad = _gradient_from_peaks(cur, b, rel_thr=0.9995)

        # If the gradient is zero (unlikely) break
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0.0:
            break

        # Back‑tracking line search – try a few step sizes
        improvement = False
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            # step magnitude proportional to inverse gradient norm
            eta = factor * 0.2 / (grad_norm + 1e-12)
            cand = cur - eta * grad
            cand = np.clip(cand, 0.0, None)
            cand_sum = cand.sum()
            if cand_sum < 1e-12:
                continue
            cand = cand / cand_sum
            cand_score = _score_fft(cand)

            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                cur = cand
                improvement = True
                print(f"[{time.time() - start_time:.1f}s] GD improvement score={best_score:.6f}")
                break
        if improvement:
            if best_score <= 1.503:
                print("Target reached – stopping search.")
                break
            continue

        # ------------------------------------------------------------------
        # If no gradient step helped, try a stochastic move (SA style)
        # ------------------------------------------------------------------
        if rng.random() < 0.6:
            # Dirichlet‑mixing move: blend current vector with a fresh Dirichlet sample
            alpha = 0.3
            dir_sample = rng.dirichlet(np.full(n, alpha))
            t = rng.random() * 0.2  # blending factor up to 0.2
            cand = (1 - t) * cur + t * dir_sample
        else:
            # Block reversal (reordering) – simple but sometimes useful
            start_idx = rng.integers(0, n)
            block_len = rng.integers(1, min(200, n - start_idx) + 1)
            cand = cur.copy()
            cand[start_idx:start_idx + block_len] = cand[start_idx:start_idx + block_len][::-1]

        # re‑normalise (sum is already 1, but rounding may drift)
        cand = _normalise_to_one(cand)
        cand_score = _score_fft(cand)

        # simulated‑annealing acceptance
        if cand_score < best_score or rng.random() < np.exp(-(cand_score - best_score) / max(T, 1e-12)):
            cur = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.1f}s] SA improvement score={best_score:.6f}")

        if best_score <= 1.503:
            print("Target reached – finalising.")
            break

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list (evaluate_sequence will
    # perform its own clipping / sum‑checks)
    # ------------------------------------------------------------------
    return best_seq.tolist()
