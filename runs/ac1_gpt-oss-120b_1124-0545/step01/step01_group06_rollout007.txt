"""
Propose a step‑function (non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the linear convolution.  The algorithm works in three stages
within the given time budget (default 1000 s).

1. **Fast parametric sweep** – a variety of analytic windows (geometric,
   power‑law, combined, symmetric, Gaussian, Kaiser, Hann, Hamming,
   Blackman, half‑sine, triangular) are generated for several lengths.
   Each candidate is normalised to a unit sum and evaluated with an
   FFT‑based convolution.  The best few candidates are kept.

2. **Adam‑based smoothed‑max descent** – starting from each retained
   candidate we minimise the (smooth) maximum of the convolution using an
   exponentiated‑gradient/Adam update:
        w ∝ exp(β·(conv−max(conv)))                (softmax over lags)
        grad = 2·(rev(a) * w)                     (FFT‑based)
   The step‑size decays like 1/√t and the smoothing factor β is increased
   geometrically.  After each update the vector is projected onto the
   probability simplex (Euclidean projection).

3. **Resolution lifting** – if time permits, the best sequence is
   up‑sampled (linear interpolation) to a longer length and refined again.
   Longer sequences have a lower theoretical lower bound (≈ 1), giving
   extra room for improvement.

All helper functions are pure NumPy; the routine respects the 1000 s
budget, prints intermediate progress, and finally returns the best
sequence as a plain Python list of floats.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {w ≥ 0, sum w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    mask = u - cssv / ind > 0
    if not np.any(mask):
        theta = cssv[-1] / n
    else:
        rho = np.where(mask)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (all return a normalized (sum = 1) non‑negative array)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    """Monotonically decreasing geometric series."""
    a = r ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    a /= a.sum()
    return a


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Combined geometric × power‑law."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    a /= a.sum()
    return a


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (mirrored at both ends)."""
    a = r ** np.arange(n, dtype=np.float64) + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    a /= a.sum()
    return a


def _gaussian(n: int, sigma: float) -> np.ndarray:
    """Gaussian window centred on the array."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a /= a.sum()
    return a


def _kaiser(n: int, beta: float) -> np.ndarray:
    """Kaiser window (beta ≥ 0)."""
    a = np.kaiser(n, beta)
    a /= a.sum()
    return a


def _hann(n: int) -> np.ndarray:
    """Hann (raised‑cosine) window."""
    a = np.hanning(n)               # also called Hahn/Hann
    a /= a.sum()
    return a


def _hamming(n: int) -> np.ndarray:
    """Hamming window."""
    a = np.hamming(n)
    a /= a.sum()
    return a


def _blackman(n: int) -> np.ndarray:
    """Blackman window."""
    a = np.blackman(n)
    a /= a.sum()
    return a


def _half_sine(n: int) -> np.ndarray:
    """Positive half‑sine shape."""
    a = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular window."""
    a = 1.0 - np.abs((np.arange(n) - (n - 1) / 2.0)) / ((n - 1) / 2.0)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function used in the challenge.  Returns the best sequence found
    (list of Python floats) before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0      # small safety margin

    best_seq = None
    best_val = float("inf")
    best_n = None

    # --------------------------------------------------------------
    # 1) Fast parametric sweep
    # --------------------------------------------------------------
    # lengths to try (larger length = better theoretical bound)
    lengths = [1024, 2048, 4096, 8192]

    # collection of top candidates (value, array)
    candidate_pool = []

    def _fast_obj(a: np.ndarray) -> float:
        """Objective assuming a already sums to 1 (no clipping needed)."""
        n = len(a)
        conv = _conv_fft(a, a)
        return float(2.0 * n * conv.max())

    for n in lengths:
        if time.time() > deadline:
            break

        # ----- uniform (baseline) -----
        a = np.ones(n, dtype=np.float64) / n
        val = _fast_obj(a)
        candidate_pool.append((val, a.copy()))
        if val < best_val:
            best_val, best_seq, best_n = val, a.copy(), n
            print(f"[sweep] uniform n={n:5d} val={val:.6f}")

        # ----- geometric families -----
        for r in np.linspace(0.90, 0.9999, 12):
            a = _geom(n, r)
            val = _fast_obj(a)
            candidate_pool.append((val, a.copy()))
            if val < best_val:
                best_val, best_seq, best_n = val, a.copy(), n
                print(f"[sweep] geom    n={n:5d} r={r:.5f} val={val:.6f}")

            a_rev = a[::-1]          # increasing geometric
            val = _fast_obj(a_rev)
            candidate_pool.append((val, a_rev.copy()))
            if val < best_val:
                best_val, best_seq, best_n = val, a_rev.copy(), n
                print(f"[sweep] revgeom n={n:5d} r={r:.5f} val={val:.6f}")

        # ----- power‑law -----
        for p in np.linspace(0.0, 2.0, 15):
            a = _power(n, p)
            val = _fast_obj(a)
            candidate_pool.append((val, a.copy()))
            if val < best_val:
                best_val, best_seq, best_n = val, a.copy(), n
                print(f"[sweep] power   n={n:5d} p={p:.3f} val={val:.6f}")

        # ----- geometric × power‑law -----
        for r in np.linspace(0.90, 0.9995, 8):
            for p in np.linspace(0.0, 2.0, 10):
                a = _geom_power(n, r, p)
                val = _fast_obj(a)
                candidate_pool.append((val, a.copy()))
                if val < best_val:
                    best_val, best_seq, best_n = val, a.copy(), n
                    print(f"[sweep] g×p     n={n:5d} r={r:.5f} p={p:.3f} val={val:.6f}")

        # ----- symmetric geometric -----
        for r in np.linspace(0.90, 0.9995, 10):
            a = _sym_geom(n, r)
            val = _fast_obj(a)
            candidate_pool.append((val, a.copy()))
            if val < best_val:
                best_val, best_seq, best_n = val, a.copy(), n
                print(f"[sweep] sym‑g   n={n:5d} r={r:.5f} val={val:.6f}")

        # ----- Gaussian windows (different sigmas) -----
        for sigma_factor in np.linspace(0.15, 0.45, 8):
            sigma = sigma_factor * n
            a = _gaussian(n, sigma)
            val = _fast_obj(a)
            candidate_pool.append((val, a.copy()))
            if val < best_val:
                best_val, best_seq, best_n = val, a.copy(), n
                print(f"[sweep] gauss   n={n:5d} σ={sigma_factor:.3f} val={val:.6f}")

        # ----- Kaiser windows (different betas) -----
        for beta in np.linspace(0.0, 10.0, 11):
            a = _kaiser(n, beta)
            val = _fast_obj(a)
            candidate_pool.append((val, a.copy()))
            if val < best_val:
                best_val, best_seq, best_n = val, a.copy(), n
                print(f"[sweep] kaiser  n={n:5d} β={beta:.2f} val={val:.6f}")

        # ----- classic windows -----
        for win_func, name in [(_hann, "hann"), (_hamming, "hamm"), (_blackman, "blck"), (_half_sine, "hsine"), (_triangular, "tri")]:
            a = win_func(n)
            val = _fast_obj(a)
            candidate_pool.append((val, a.copy()))
            if val < best_val:
                best_val, best_seq, best_n = val, a.copy(), n
                print(f"[sweep] {name:6s} n={n:5d} val={val:.6f}")

        # ----- few random Dirichlet draws (to escape local minima) -----
        for _ in range(8):
            a = np.random.dirichlet(np.ones(n))
            val = _fast_obj(a)
            candidate_pool.append((val, a.copy()))
            if val < best_val:
                best_val, best_seq, best_n = val, a.copy(), n
                print(f"[sweep] random  n={n:5d} val={val:.6f}")

    # keep the top‑k candidates for the refinement stage
    candidate_pool.sort(key=lambda x: x[0])
    top_k = min(6, len(candidate_pool))
    refinement_starts = [c[1] for c in candidate_pool[:top_k]]

    # --------------------------------------------------------------
    # 2) Smoothed‑max Adam descent (exponentiated‑gradient style)
    # --------------------------------------------------------------
    def _refine(a_init: np.ndarray, max_iters: int = 25000):
        """Return a refined version of a_init (list) and its objective."""
        a = a_init.copy()
        n = len(a)
        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr0 = 0.4                     # base learning rate (exponential update)
        beta = 5.0                    # smoothing for log‑sum‑exp
        beta_mul = 1.012

        best_local_seq = a.copy()
        best_local_val = _fast_obj(a)

        it = 0
        while it < max_iters and time.time() < deadline:
            # forward pass
            conv = _conv_fft(a, a)
            cur_max = conv.max()
            cur_val = 2.0 * n * cur_max / (a.sum() ** 2)   # a.sum()≈1 after projection
            if cur_val < best_local_val:
                best_local_val = cur_val
                best_local_seq = a.copy()
                print(f"[refine] n={n:5d} it={it:5d} new best={cur_val:.6f}")

            # smoothed‑max weights (log‑sum‑exp)
            shift = cur_max
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0.0:
                it += 1
                beta *= beta_mul
                continue
            w /= w_sum

            # gradient of the smoothed max (exponentiated‑gradient)
            grad_full = _conv_fft(a[::-1], w)
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # Adam update on the *additive* gradient
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # exponential (multiplicative) update – guarantees positivity
            eta = lr0 / np.sqrt(it)
            a = a * np.exp(-eta * (m_hat / (np.sqrt(v_hat) + eps)))
            a = _project_simplex(a)    # keep sum=1 and a_i≥0

            # increase smoothing slowly
            beta *= beta_mul

        return best_local_seq, best_local_val

    # apply refinement to each of the selected seeds
    for start in refinement_starts:
        if time.time() > deadline:
            break
        refined_seq, refined_val = _refine(start, max_iters=20000)
        if refined_val < best_val:
            best_val, best_seq, best_n = refined_val, refined_seq.copy(), len(refined_seq)

    # --------------------------------------------------------------
    # 3) Resolution lifting (optional, time‑permitting)
    # --------------------------------------------------------------
    # Upsample the current best to a larger grid and refine again.
    # This often yields a lower objective because the theoretical bound
    # improves with n.
    up_factors = [2, 4]   # 2× and 4× the current length
    for factor in up_factors:
        if time.time() > deadline:
            break
        new_n = best_n * factor
        if new_n > 20000:               # keep work manageable
            continue
        # linear interpolation to the larger grid
        x_old = np.arange(best_n, dtype=np.float64)
        x_new = np.linspace(0, best_n - 1, new_n)
        a_up = np.interp(x_new, x_old, best_seq)
        a_up = _project_simplex(a_up)
        print(f"[lift] upsampling {best_n} -> {new_n}")

        refined_seq, refined_val = _refine(a_up, max_iters=15000)
        if refined_val < best_val:
            best_val, best_seq, best_n = refined_val, refined_seq.copy(), new_n

    # ------------------------------------------------------------------
    # Return result (as plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform on a modest size
        n = 1024
        best_seq = np.ones(n, dtype=np.float64) / n
    return best_seq.tolist()
