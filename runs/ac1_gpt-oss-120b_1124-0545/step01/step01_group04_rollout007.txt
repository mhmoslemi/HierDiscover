        conv = np.fft.irfft(np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len), n=fft_len)
        conv = conv[:2 * n - 1]
        max_val = conv.max()
        # identify active indices
        # choose indices where conv >= max_val - active_epsilon (or relative)
        tol = max_val * 1e-6 + 1e-12
        active_idx = np.where(conv >= max_val - tol)[0]
        # compute gradient
        grad = np.zeros_like(a)
        for k in active_idx:
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            idx = np.arange(start, end + 1)
            grad[idx] += 2 * a[k - idx]  # as before
        # subgradient step
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, None)  # nonnegativity
        sum_a = a_new.sum()

            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new /= sum_a   # normalization to sum=1 (maintain scale)
        a = a_new

        if iter % 10 == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[grad] iter {iter} best {best_score}")
        step *= step_decay
        iter += 1

    # then random perturbation local search (simulated annealing)
    temperature = 0.1
    eps = 0.2
    while time.time() < deadline:
        # Randomly pick some mass to redistribute across random positions
        cand = a.copy()
        # pick number of moves maybe 2-5
        m = np.random.randint(1, 4)

            i = np.random.randint(n)
            j = np.random.randint(n)

                continue

            cand[j] += delta

""" 
Gradient descent on L2 variance of self-convolution followed by max-subgradient 
and simulated annealing to lower the max self‑convolution bound.
"""

import numpy as np
import time

def compute_catalan_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    a /= a.sum()
    return a

def fft_length(min_len):
    return 1 << ((min_len - 1).bit_length())

def full_convolution_fft(x, y, fft_len):
    # compute linear convolution of real vectors x and y using FFT of length fft_len
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    # length = len(x) + len(y) - 1
    return conv[:len(x) + len(y) - 1]

def max_convolution_fft(a, fft_len):
    conv = full_convolution_fft(a, a, fft_len)
    max_val = np.max(conv)
    return max_val, conv

def objective_from_vector(a):
    n = len(a)
    max_val = np.max(full_convolution_fft(a, a, fft_len=fft_length(2*n-1)))
    # sum a
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_val / (sum_a * sum_a))

def gradient_l2_step(a, step, fft_len):
    n = len(a)
    # convolution a*a
    conv_aa = full_convolution_fft(a, a, fft_len)
    avg = (a.sum()**2) / (2*n - 1)
    diff = conv_aa - avg
    # reverse a
    a_rev = a[::-1]
    # convolution diff * a_rev
    conv_da = full_convolution_fft(diff, a_rev, fft_len)
    # gradient slice
    grad = 4.0 * conv_da[n-1:n-1+n]   # length n
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= s
    return a_new

def gradient_max_step_active(a, step, fft_len, eps=1e-12):
    n = len(a)
    conv = full_convolution_fft(a, a, fft_len)
    max_val = np.max(conv)
    # active indices within a relative tolerance
    rel_tol = 1e-6
    thresh = max_val - max(rel_tol * max_val, eps)
    act_idx = np.where(conv >= thresh)[0]
    grad = np.zeros_like(a)
    for k in act_idx:
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        idx = np.arange(start, end + 1)
        grad[idx] += 2.0 * a[k - idx]
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= s
    return a_new, max_val

def random_move(a, eps):
    n = len(a)
    i = np.random.randint(n)
    j = np.random.randint(n)
    if i == j:
        return a
    delta = eps * a[i]
    if delta <= 0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new

                best = a.copy()

        step *= decay

    # ---------------------------------------------------------------

        it2 += 1

    # Phase 3 – stochastic local search (simulated annealing)
    # ---------------------------------------------------------------
    eps = 0.15
    temperature = 0.1
    it3 = 0
    while time.time() < deadline:
        cand = random_move(a, eps)
        cand_score = objective_from_vector(cand)
        # Metropolis acceptance
        if cand_score < best_score or np.random.rand() < np.exp(-(cand_score - best_score) / max(temperature, 1e-12)):
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                print(f"[SA] iter {it3:5d} improved score {best_score:.6f}")
        # decay parameters
        eps = max(eps * 0.9999, 1e-5)
        temperature *= 0.9998
        it3 += 1
    
    final_seq = np.clip(best, 0.0, 1000.0)
    return final_seq.tolist()

"""
Propose a step‑function that minimises the evaluation metric.

The algorithm has three stages:

1. **Smooth L2‑phase** – minimise the variance of the self‑convolution
   (i.e.  ∑(a*a‑avg)² ) by projected gradient descent.  The gradient can be
   expressed as a convolution, so a single FFT of size ≈ 3 n gives the
   exact gradient.  This rapidly flattens the whole autocorrelation curve.

2. **Active‑set max‑phase** – switch to a sub‑gradient step that only
   touches the (few) indices where the convolution is maximal.  A decaying
   step‑size together with a tiny tolerance gradually pushes the peak
   down.

3. **Stochastic local search** – a simulated‑annealing loop that moves a
   small fraction of mass from a random source index to a random target
   index.  Upside moves are occasionally accepted to escape shallow plateaus.

All vectors are kept non‑negative and normalised to sum = 1 (the objective
is invariant under scaling).  The best vector seen so far is returned as a
plain Python list; values are clipped to the interval [0, 1000] to satisfy
the harness’s safety checks.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def compute_catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan‑type coefficients, normalised to sum = 1.

    The coefficients are (2i choose i)/4**i for i=0…n‑1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def next_pow2(x: int) -> int:
    """Small helper returning the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of real vectors x and y using FFT of length ``fft_len``.
    The returned array has length len(x)+len(y)-1.
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def max_convolution_fft(a: np.ndarray, fft_len: int) -> tuple[float, np.ndarray]:
    """
    Compute the full self‑convolution a*a via FFT and return (max, conv).
    """
    conv = full_convolution_fft(a, a, fft_len)
    return float(conv.max()), conv


def objective_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Exact objective value 2·n·max(conv)/(sum_a²)."""
    max_val = float(conv.max())
    return 2.0 * n * max_val / (sum_a * sum_a)


def objective(a: np.ndarray) -> float:
    """Convenient wrapper that computes the metric by a full convolution."""
    n = len(a)
    conv = np.convolve(a, a)          # O(n²) but cheap for occasional calls
    return objective_from_conv(conv, n, a.sum())


def gradient_l2_step(a: np.ndarray, step: float, fft_len: int) -> np.ndarray:
    """
    One projected gradient step for the *L₂* loss
        ∑_k ( (a*a)[k] - avg )² .
    The gradient can be written as 4·( (conv‑avg) * a_rev ), where
    a_rev is the reversed vector.
    """
    n = len(a)
    # 1️⃣  a*a
    conv_aa = full_convolution_fft(a, a, fft_len)          # length 2n‑1
    # 2️⃣  average value of the convolution (sum^2)/(2n‑1)
    avg = (a.sum() ** 2) / (2 * n - 1)
    diff = conv_aa - avg                                   # length 2n‑1

    # 3️⃣  gradient = 4·conv(diff, a_rev)  → pick centre slice of length n
    a_rev = a[::-1]
    conv_da = full_convolution_fft(diff, a_rev, fft_len)   # length 3n‑2
    grad = 4.0 * conv_da[n - 1 : n - 1 + n]                # indices [n‑1 … 2n‑2]

    # 4️⃣  projected sub‑gradient step on the simplex
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= s
    return a_new


def gradient_max_step_active(a: np.ndarray,
                             step: float,
                             fft_len: int,
                             rel_tol: float = 1e-6,
                             abs_eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient step that only uses the (near) maximal entries of the convolution.
    Returns the new vector and the current maximal convolution value.
    """
    n = len(a)
    conv = full_convolution_fft(a, a, fft_len)   # length 2n‑1
    max_val = float(conv.max())
    # active set: all indices whose value is within a small tolerance of the max
    thresh = max_val - max(rel_tol * max_val, abs_eps)
    active = np.where(conv >= thresh)[0]

    grad = np.zeros_like(a)
    for k in active:
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        idx = np.arange(start, end + 1, dtype=np.int64)
        grad[idx] += 2.0 * a[k - idx]   # ∂(a*a)_k/∂a_i = 2·a_{k‑i}
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= s
    return a_new, max_val


def random_perturb(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Transfer a fraction ``eps`` of the mass of a randomly chosen source
    index to a different random target index.  The result stays normalised.
    """
    n = len(a)
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    delta = eps * a[i]
    if delta <= 0.0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new

                best_score = cur
                best = a.copy()
                print(f"[L2]  iter {it:5d}  score {best_score:.6f}")
        step *= decay
        it += 1

    # ------------------------------------------------------------------
    # 2️⃣  Phase 2 – directly drive down the maximal convolution entry.
    # ------------------------------------------------------------------
    fft_len_max = next_pow2(2 * n)           # for a*a
    step = 0.2
    decay = 0.9997
    it = 0
    while time.time() < deadline and step > 1e-7:
        a, max_val = gradient_max_step_active(a, step, fft_len_max)
        if it % 10 == 0:
            cur = objective(a)
            if cur < best_score:
                best_score = cur
                best = a.copy()
                print(f"[max] iter {it:5d}  max={max_val:.6f}  score={best_score:.6f}")
        step *= decay
        it += 1

    # ------------------------------------------------------------------
    # 3️⃣  Phase 3 – stochastic local search (simulated annealing).
    # ------------------------------------------------------------------
    eps = 0.15
    temperature = 0.1
    it = 0
    while time.time() < deadline:
        cand = random_perturb(a, eps)
        cand_score = objective(cand)

        # Metropolis criterion
        if cand_score < best_score or np.random.rand() < np.exp(
                -(cand_score - best_score) / max(temperature, 1e-12)):
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                print(f"[SA]  iter {it:5d}  improved score {best_score:.6f}")

        # gradual cooling / shrinkage of the move size
        eps = max(eps * 0.9999, 1e-5)
        temperature *= 0.9998
        it += 1

    # ------------------------------------------------------------------
    # Return the best vector seen (clipped to the allowed range).
    # ------------------------------------------------------------------
    final_seq = np.clip(best, 0.0, 1000.0)
    print(f"[done] final score ≈ {best_score:.6f}")
    return final_seq.tolist()
