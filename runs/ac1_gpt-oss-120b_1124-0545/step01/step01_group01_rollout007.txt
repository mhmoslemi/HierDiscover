def _subgradient_step(seq, T, target_sum):
    conv = np.convolve(seq, seq)
    k_max = int(np.argmax(conv))
    n = seq.size
    gradient = np.zeros_like(seq)
    idx = np.arange(n)
    partner = k_max - idx
    mask = (partner >= 0) & (partner < n)
    gradient[mask] = seq[partner[mask]]
    eta = target_sum * 0.2 * T
    new_seq = seq - eta * gradient
    # clip negative after step (set to 0)
    new_seq = np.maximum(new_seq, 0.0)

"""
Population‑based stochastic optimisation for the step‑function problem.

The objective is  max(conv(a,a))  (the evaluation function returns exactly this
value, because the ℓ₁–norm is kept fixed to √(2·n)).  The algorithm works as
follows:

* The length `n` and the target ℓ₁‑norm `S = sqrt(2·n)` are taken from the
  previously found construction `height_sequence_1` if it exists,
  otherwise a safe default length (~2000) is used.

* A small population (≈15) of candidate vectors on the simplex
  `{a_i ≥ 0, Σ a_i = S}` is created.  The population contains the old
  best solution, a uniform vector and a few random Dirichlet draws together
  with a couple of simple handcrafted shapes (exponential tails).

* Main loop (runs until the time budget expires):
    – a temperature `T` (linear/exponential cooling) controls the amplitude
      of mutations;
    – two parents are selected by tournament (3 random members, the best wins);
    – a child is produced by convex combination (crossover) of the parents;
    – the child undergoes several mutation operators:
        * a random “transfer” of mass between two coordinates,
        * a small global Gaussian perturbation,
      after which it is projected back onto the simplex.
    – a single sub‑gradient step targeting the currently largest convolution
      entry is applied (the gradient is `g_i = a_{k_max-i}`); the step size
      is proportional to `T`.
    – the child's objective value (maximum convolution entry) is evaluated
      using a fast NumPy convolution.
    – if the child improves the global best, the best solution is updated;
      otherwise it replaces the worst member of the population when it is
      better than that worst member.

* After the time limit the best sequence found is returned as a plain
  Python list.  The routine never calls `evaluate_sequence` during the search
  (the same computation is performed locally), keeping the inner loop very
  fast.  The returned list complies with all security checks of the
  evaluation harness.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of v onto the simplex {w ≥ 0, Σ w = s}."""
    n = v.size
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform vector
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution(seq: np.ndarray) -> float:
    """Return max(conv(seq, seq)).  Equivalent to the public evaluation."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


# ----------------------------------------------------------------------
def _subgradient_step(seq: np.ndarray, T: float, target_sum: float) -> np.ndarray:
    """
    One sub‑gradient descent step on the maximal convolution entry.
    k_max = argmax(conv(seq,seq)),
    gradient_i = seq[k_max - i] (zero when out of range).
    """
    conv = np.convolve(seq, seq)
    k_max = int(conv.argmax())
    n = seq.size
    idx = np.arange(n)
    partner = k_max - idx
    mask = (partner >= 0) & (partner < n)
    gradient = np.zeros_like(seq)
    gradient[mask] = seq[partner[mask]]
    # step size – decays with temperature
    eta = target_sum * 0.15 * T
    new_seq = seq - eta * gradient
    new_seq = np.maximum(new_seq, 0.0)
    new_seq = _simplex_projection(new_seq, target_sum)
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function height vector that minimises the
    maximum entry of its self‑convolution.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    **kwargs : dict
        Ignored – present for compatibility with the harness.

    Returns
    -------
    list[float]
        The best coefficient sequence found within the time budget.
    """
    # ------------------------------------------------------------------
    # Fix random state
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        # use the provided vector as a starting point
        start_from_known = np.array(height_sequence_1, dtype=float)
    else:
        n = 2000                      # reasonable default
        start_from_known = None

    target_sum = math.sqrt(2.0 * n)   # ℓ₁‑norm after normalisation

    # ------------------------------------------------------------------
    # Helper to generate a random point on the simplex
    def _rand_simplex():
        x = np.random.exponential(scale=1.0, size=n)
        return _simplex_projection(x, target_sum)

    # ------------------------------------------------------------------
    # Initialise a small elite population
    pop_size = 15
    population = []

    # 1) Uniform vector (already on the simplex)
    uniform = np.full(n, target_sum / n)
    population.append((uniform, _max_convolution(uniform)))

    # 2) The previously discovered best (if any)
    if start_from_known is not None:
        # make sure it satisfies the sum constraint exactly
        known = _simplex_projection(start_from_known, target_sum)
        population.append((known, _max_convolution(known)))

    # 3) A few random Dirichlet samples
    for _ in range(4):
        v = np.random.dirichlet(alpha=np.ones(n)) * target_sum
        population.append((v, _max_convolution(v)))

    # 4) Simple exponential‑tail shapes (biased towards the ends)
    for scale in (0.1, 0.3, 0.6):
        t = np.exp(-np.arange(n) / (scale * n))
        t = _simplex_projection(t, target_sum)
        population.append((t, _max_convolution(t)))

    # Keep only the best `pop_size` individuals
    population.sort(key=lambda x: x[1])
    population = population[:pop_size]

    best_seq, best_score = population[0]
    print(f"Initial best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin
    iteration = 0

    # Temperature schedule (exponential cooling)
    T_start = 1.0
    T_end = 1e-3

    # ------------------------------------------------------------------
    # Tournament selection (size 3)
    def _tournament():
        contestants = random.sample(population, 3)
        contestant = min(contestants, key=lambda x: x[1])
        return contestant[0]

    while time.time() < deadline:
        # Temperature
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        T = T_start * ((T_end / T_start) ** frac)   # exponential cooling

        # ---------- Crossover ----------
        parent1 = _tournament()
        parent2 = _tournament()
        alpha = random.random()
        child = alpha * parent1 + (1.0 - alpha) * parent2

        # ---------- Mutations ----------
        # (a) Transfer a random amount between two coordinates
        i, j = np.random.randint(n, size=2)
        if i != j:
            delta = np.random.normal(loc=0.0, scale=target_sum * 0.02 * T)
            delta = max(-child[i], min(delta, child[i]))   # stay non‑negative
            child[i] -= delta
            child[j] += delta

        # (b) Small global Gaussian perturbation
        child += np.random.normal(loc=0.0,
                                  scale=target_sum * 0.005 * T,
                                  size=n)

        # Ensure non‑negativity and project onto simplex
        child = np.maximum(child, 0.0)
        child = _simplex_projection(child, target_sum)

        # ---------- One sub‑gradient refinement ----------
        child = _subgradient_step(child, T, target_sum)

        # ---------- Evaluate ----------
        child_score = _max_convolution(child)

        # ---------- Keep the global best ----------
        if child_score < best_score:
            best_score = child_score
            best_seq = child.copy()
            print(f"[{iteration:6d}] New best score: {best_score:.6f} (time {elapsed:.1f}s)")

        # ---------- Population replacement ----------
        # Replace the worst individual if the child is better
        if child_score < population[-1][1]:
            population[-1] = (child, child_score)
            population.sort(key=lambda x: x[1])

        # Occasionally inject a fresh random individual to keep diversity
        if iteration % 500 == 0 and iteration > 0:
            fresh = _rand_simplex()
            fresh_score = _max_convolution(fresh)
            if fresh_score < population[-1][1]:
                population[-1] = (fresh, fresh_score)
                population.sort(key=lambda x: x[1])

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best vector as a plain Python list.
    return best_seq.tolist()
