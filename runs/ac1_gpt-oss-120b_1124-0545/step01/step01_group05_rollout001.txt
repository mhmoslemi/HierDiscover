def simplex_projection(v):
    """Projects onto the probability simplex (sum=1, v>=0)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a high‑quality step‑function using a smoothed max‑convolution
gradient descent with simplex projection and occasional random restarts /
crossover.  The objective is 2·n·max(conv(a,a))/sum(a)^2 which, after
normalising sum(a)=1, reduces to 2·n·max(conv(a,a)).  We minimise a smooth
approximation   L(a)=log‑sum‑exp(α·conv(a,a))   (α grows from a modest
value to a large one) and use the exact evaluator only for improving
candidates.  The algorithm proceeds as follows:

1. Initialise a from the provided `height_sequence_1` (or uniform).
2. Repeatedly:
   * compute the full self‑convolution,
   * form soft‑max weights wₖ ∝ exp(α·convₖ),
   * obtain the gradient ∇L = 2·conv_valid(w, a)  (valid convolution),
   * take a projected gradient‑descent step onto the probability simplex,
   * increase α gradually and decay the learning‑rate,
   * keep the best sequence seen (checked with `evaluate_sequence`).
3. Occasionally restart from a random Dirichlet sample or from a blended
   version of the current best and a random two‑block shape to avoid
   stagnation.
4. Finish with a short local perturbation phase (tiny mass moves).

The method stays well within the 1000 s budget and typically improves
the bound below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> float:
    """Maximum value of the linear convolution a*a (full mode)."""
    conv = np.convolve(a, a)
    return float(conv.max())


def _fast_score(a: np.ndarray) -> float:
    """Fast surrogate for evaluate_sequence when sum(a)=1."""
    n = a.shape[0]
    return 2.0 * n * _conv_max(a)


def _make_two_block(n: int, m: int, high_frac: float) -> np.ndarray:
    """
    Two‑block step function.
    First `m` entries share `high_frac` of the total mass,
    the remaining entries share the rest.
    """
    if m <= 0 or m >= n:
        raise ValueError("invalid block size")
    high = high_frac / m
    low = (1.0 - high_frac) / (n - m)
    a = np.empty(n, dtype=float)
    a[:m] = high
    a[m:] = low
    return a


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → project to uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, seed: int = 42, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation score.  Returns the best sequence (as a plain Python list)
    found before the time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    # safety margin so we finish cleanly before the external timeout
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() > 0:
            a0 = a0 / a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        n = 2000
        a0 = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Initialise state
    # ------------------------------------------------------------------
    a = a0.copy()
    best_a = a.copy()
    try:
        best_score = evaluate_sequence(best_a.tolist())
    except Exception:
        best_score = np.inf
    # Occasionally the initial exact evaluation can be a bit noisy
    print(f"[0.0s] initial score: {best_score:.6f}")

    # Gradient‑descent hyper‑parameters
    lr = 0.5                 # initial learning rate
    min_lr = 0.002           # floor learning rate
    lr_decay = 0.9995        # exponential decay per iteration
    alpha = 10.0             # start smoothing parameter
    alpha_max = 80.0         # final smoothing (approaches true max)
    alpha_inc = (alpha_max - alpha) / 2500  # linear increase schedule
    beta = 0.9               # momentum coefficient
    momentum = np.zeros_like(a)

    it = 0
    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ---- update schedules -------------------------------------------------
        lr = max(min_lr, lr * lr_decay)
        alpha = min(alpha_max, alpha + alpha_inc)

        # ---- compute convolution and soft‑max weights -------------------------
        conv = np.convolve(a, a)                     # length 2n‑1
        # stability shift
        max_c = conv.max()
        exp_vals = np.exp(alpha * (conv - max_c))
        w = exp_vals / exp_vals.sum()                # probability vector, sum=1

        # ---- gradient of the smoothed objective --------------------------------
        # gradient_i = 2 * Σ_j w_{i+j} a_j  -> valid convolution
        grad = 2.0 * np.convolve(w, a, mode='valid')  # length n

        # ---- momentum + projected gradient step --------------------------------
        momentum = beta * momentum - lr * grad
        a_candidate = a + momentum
        a_candidate = _simplex_projection(a_candidate)

        # ---- fast surrogate evaluation -----------------------------------------
        cand_score = _fast_score(a_candidate)

        # ---- accept if we see any improvement (exact check) --------------------
        if cand_score < best_score - 1e-12:
            try:
                exact = evaluate_sequence(a_candidate.tolist())
                if exact < best_score:
                    best_score = exact
                    best_a = a_candidate.copy()
                    elapsed = time.time() - start
                    print(
                        f"[{elapsed:6.1f}s] iter={it} new best: {best_score:.6f}"
                    )
                    # keep the improved point as the new centre
                    a = a_candidate.copy()
                else:
                    # keep previous a if exact evaluation not better
                    a = a_candidate.copy()
            except Exception:
                # on unexpected error just keep moving
                a = a_candidate.copy()
        else:
            # always move to the candidate (helps escaping plateaus)
            a = a_candidate.copy()

        it += 1

        # ---- occasional random restart / crossover to diversify ----------------
        if it % 4500 == 0:
            if np.random.rand() < 0.4:
                # fresh sparse Dirichlet
                alpha_dir = np.random.uniform(0.1, 4.0)
                a = np.random.dirichlet(np.full(n, alpha_dir))
            else:
                # blend best with a random two‑block shape
                m = np.random.randint(int(0.1 * n), int(0.9 * n))
                high_frac = np.random.rand()
                block = _make_two_block(n, m, high_frac)
                blend = 0.5 * best_a + 0.5 * block
                a = _simplex_projection(blend)

        # ---- periodic progress report ------------------------------------------
        if it % 2000 == 0:
            elapsed = time.time() - start
            remaining = max(0.0, deadline - time.time())
            print(
                f"[{elapsed:6.1f}s] iter={it} best={best_score:.6f} "
                f"time left≈{int(remaining)}s"
            )

    # ------------------------------------------------------------------
    # Final local refinement (tiny mass shuffling)
    # ------------------------------------------------------------------
    a = best_a.copy()
    for _ in range(3000):
        i, j = np.random.choice(n, 2, replace=False)
        # move a small fraction of mass from i to j (or vice‑versa)
        delta = a[i] * np.random.uniform(-0.04, 0.04)
        delta = np.clip(delta, -a[i] + 1e-14, np.inf)
        a[i] -= delta
        a[j] += delta
        a = _simplex_projection(a)

        surf = _fast_score(a)
        if surf < best_score - 1e-12:
            try:
                exact = evaluate_sequence(a.tolist())
                if exact < best_score:
                    best_score = exact
                    best_a = a.copy()
                    elapsed = time.time() - start
                    print(
                        f"[{elapsed:6.1f}s] final refine improved to {best_score:.6f}"
                    )
            except Exception:
                pass

        if time.time() >= deadline:
            break

    # Return result as plain Python list
    return best_a.tolist()
