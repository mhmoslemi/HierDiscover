import numpy as np
import time

def compute_catalan_sequence(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a

def compute_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * len(a) - 1]

def project_simplex(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s

def weighted_gradient(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    # w length = 2*n-1
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len)
    fw = np.fft.rfft(w_rev, n=fft_len)
    h = np.fft.irfft(fa * fw, n=fft_len)
    n = len(a)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * h[idx]
    return grad

def mirror_descent_step(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    # multiplicative update that stays on simplex
    a_new = a * np.exp(-eta * grad)
    a_new = project_simplex(a_new)
    return a_new

def random_perturb(a: np.ndarray, eps: float) -> np.ndarray:
    n = len(a)
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    delta = eps * a[i]
    if delta <= 0.0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = project_simplex(a_new)
    return a_new

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Gradient‑based minimisation of 2·n·max(conv(a,a)) on the probability simplex.
    Starts from the Catalan sequence, applies a smooth‑max (soft‑max) sub‑gradient
    with mirror‑descent on the simplex, and finishes with a short stochastic
    local‑search phase.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a margin

    # ------------------------------------------------------------------
    # Choose length
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 2000

    # ------------------------------------------------------------------
    # Pre‑compute FFT size
    # ------------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Initialise – Catalan shape (already normalised to sum 1)
    # ------------------------------------------------------------------
    a = compute_catalan_sequence(n)
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] Catalan baseline score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smoothed sub‑gradient (mirror) descent
    # ------------------------------------------------------------------
    beta = 2.0          # smoothing for soft‑max, will be increased
    beta_max = 50.0
    momentum = 0.0      # location for optional momentum term (unused)
    it = 0
    eval_interval = 20
    while time.time() < deadline:
        # current convolution
        conv = compute_convolution_fft(a, fft_len)
        max_conv = conv.max()

        # soft‑max weights (the max term gets weight 1, others decay)
        w = np.exp(beta * (conv - max_conv))

        # weighted gradient of the smooth max
        grad = weighted_gradient(a, w, fft_len)

        # adaptive step size – we limit the L∞ change of a to at most 0.5·η
        max_abs = np.max(np.abs(grad))
        if max_abs == 0:
            eta = 0.0
        else:
            eta = 0.5 / max_abs     # ensures update magnitude ≤0.5

        # mirror‑descent update on the simplex
        a = mirror_descent_step(a, grad, eta)

        # occasional evaluation
        if it % eval_interval == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[grad] iter {it:5d} – improved score {best_score:.6f}")

        # schedule – increase smoothing, slowly decay step scale
        if beta < beta_max:
            beta = min(beta * 1.001, beta_max)

        it += 1
        # safety break if we have already converged to very low improvement
        if it > 20000 and time.time() > deadline - 30:
            # give a small window for final local search
            break

    # ------------------------------------------------------------------
    # Phase 2 – stochastic local search (annealing style)
    # ------------------------------------------------------------------
    eps = 0.08                # initial amount of mass transferred
    temperature = 0.08
    it = 0
    while time.time() < deadline:
        cand = random_perturb(a, eps)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand
            print(f"[pert] iter {it:5d} – improved score {best_score:.6f}")
        else:

            prob = np.exp(-(cand_score - best_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:

        # decay parameters

        temperature *= 0.9997
        it += 1

    return final_seq.tolist()

"""
Propose a step‑function (non‑negative sequence) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

which is exactly the value returned by ``evaluate_sequence``.
The algorithm consists of three phases:

1. **Initialisation** – start from the Catalan ( (1‑x)^‑½ ) coefficients,
   which already give a very good bound (≈π/2≈1.5708).

2. **Smoothed sub‑gradient descent on the simplex** – we minimise a smooth
   approximation of the max‑convolution using a mirror‑descent update.
   The smooth‑max is obtained with a soft‑max weight
   w_k = exp(β·(conv_k – max(conv))) .  The gradient of the weighted sum
   is computed with a single FFT (O(n log n)).  An adaptive step size
   keeps the update stable and the projection onto the probability simplex
   guarantees non‑negativity and a unit 𝓁₁‑norm.

3. **Stochastic local search** – after the deterministic phase we perform
   cheap random mass transfers (simulated‑annealing style) to escape any
   remaining plateau.

The whole routine respects the 1000 s time budget, works for any length
(uses the length of ``height_sequence_1`` when available) and returns the
best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def compute_catalan_sequence(n: int) -> np.ndarray:
    """Catalan‑type coefficients (1‑x)^‑½, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def compute_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using the FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * len(a) - 1]


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the non‑negative simplex (∑v = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


def weighted_gradient(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a*a)_k  w.r.t. a.
    The formula is   g_i = 2·Σ_k w_k·a_{k-i} .
    Computed efficiently via a single convolution with the reversed weight.
    """
    n = len(a)
    w_rev = w[::-1]                       # reverse the weight vector
    fa = np.fft.rfft(a, n=fft_len)
    fw = np.fft.rfft(w_rev, n=fft_len)
    h = np.fft.irfft(fa * fw, n=fft_len)   # linear convolution a * w_rev
    idx = (2 * n - 2) - np.arange(n)       # positions L‑i, L=2n‑2
    grad = 2.0 * h[idx]
    return grad


def mirror_descent_step(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Multiplicative (mirror‑descent) update on the simplex:
        a_i ← a_i·exp(‑η·grad_i)  and renormalise.
    """
    a_new = a * np.exp(-eta * grad)
    a_new = project_simplex(a_new)
    return a_new


def random_perturb(a: np.ndarray, eps: float) -> np.ndarray:
    """Transfer a fraction *eps* of the mass from one random index to another."""
    n = len(a)
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    delta = eps * a[i]
    if delta <= 0.0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)
