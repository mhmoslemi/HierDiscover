def project_simplex(v):
    """Projects vector v onto probability simplex (sum=1, v>=0)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho = max i such that u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose‑candidate for the “minimum max‑autocorrelation” challenge.

The algorithm combines three complementary ideas:

1. **Smoothed gradient descent** –  
   The non‑smooth objective  max(a∗a)  is approximated by a
   log‑sum‑exp (temperature ε).  The gradient of the smoothed
   objective can be written in closed form:
        ∇_a  ε·log Σ_k exp(b_k/ε) = 2·correlate(w, a, "valid")
   with  w_k ∝ exp(b_k/ε).  After each step the vector is projected
   onto the probability simplex (non‑negative, sum = 1).

2. **Linear‑programming “gap‑filler’’** –  
   For the current sequence `a` let M = max(a∗a).  Solving the LP

        maximise   Σ_j g_j
        subject to  (a ∗ g)_k ≤ M   for all k
                    g_j ≥ 0

   yields a feasible direction `g` that respects the present
   convolution bound.  After normalising `g` to sum = 1 we blend it
   with `a` (optimal blend found by a cheap golden‑section search).
   This step strictly reduces the bound on the peak autocorrelation.

3. **Restart / exploration** –  
   If no improvement is observed for a while the algorithm restarts
   from a fresh random simplex point.

All heavy linear algebra uses NumPy (FFT for fast convolutions) and
SciPy’s HiGHS linear‑programming backend.  The routine stops at the
time budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small utility to get the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Convolution a*a via FFT (returns length 2·n‑1 array)."""
    n = a.shape[0]
    m = 2 * n - 1
    size = _next_pow2(m)
    A = np.fft.rfft(a, n=size)
    B = A * A
    b = np.fft.irfft(B, n=size)[:m]
    return b


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective 2·n·max(conv(a,a)) / (sum(a)^2).
    The implementation uses an FFT‑based convolution for speed.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    b = _conv_fft(a)
    max_b = b.max()
    return float(2.0 * a.shape[0] * max_b / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, Σ x = 1}.
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the rho (last index where condition holds)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero; return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def smoothed_gradient_step(a: np.ndarray, eps: float, step: float) -> np.ndarray:
    """
    One gradient step on the ε‑smoothed objective.
    The gradient is 2·correlate(w, a, "valid") where
    w_k ∝ exp(b_k/ε) with b = a∗a.
    """
    b = _conv_fft(a)                     # a∗a
    # Stable soft‑max weights
    max_b = b.max()
    exp_vals = np.exp((b - max_b) / eps)
    w = exp_vals / exp_vals.sum()
    # Gradient: 2·correlate(w, a, mode='valid')
    grad = 2.0 * np.correlate(w, a, mode='valid')
    a_new = a - step * grad
    return project_simplex(a_new)


def solve_lp_gap_filler(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP that maximises Σ g_j subject to (a∗g)_k ≤ M and g ≥ 0.
    Returns the optimal g (not yet normalised to sum = 1).
    """
    n = a.shape[0]
    rows = 2 * n - 1

    # Build the dense constraint matrix A where A[k, j] = a[k-j] if
    # 0 ≤ k-j < n else 0.
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a

    c = -np.ones(n)                     # maximise Σ g_j  → minimise -Σ g_j
    bounds = [(0.0, None)] * n
    b_ub = np.full(rows, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        # If the LP fails, fall back to the current a (feasible by construction)
        return a.copy()
    return res.x


def optimal_mix(a: np.ndarray, g: np.ndarray) -> np.ndarray:
    """
    Find the best convex combination (1‑t)·a + t·g (t∈[0,1])
    that minimises the true score.  A cheap golden‑section search is used.
    """
    def f(t_):
        mixed = (1.0 - t_) * a + t_ * g
        return compute_score(mixed)

    lo, hi = 0.0, 1.0
    phi = (1.0 + 5**0.5) / 2.0
    for _ in range(25):                 # 25 iterations → high precision
        t1 = hi - (hi - lo) / phi
        t2 = lo + (hi - lo) / phi
        f1, f2 = f(t1), f(t2)
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    t_opt = (lo + hi) / 2.0
    mixed = (1.0 - t_opt) * a + t_opt * g
    return project_simplex(mixed)

    """
    np.random.seed(seed)
    start = time.time()

    best_seq = a.copy()

    eps_schedule = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]  # smoothing temperatures
