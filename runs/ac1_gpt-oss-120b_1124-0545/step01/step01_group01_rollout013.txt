```
def lp_optimize(f, M, target_sum):
    n = len(f)
    from scipy.sparse import csr_matrix
    from scipy.optimize import linprog
    # Build A (2n-1 x n)
    rows = []
    cols = []
    data = []
    for i in range(n):
        fi = f[i]
        if fi == 0:
            continue
        r_start = i
        r_end = i + n
        rows.extend(range(r_start, r_end))
        cols.extend([i] * n)
        data.extend([fi] * n)
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum(g) => minimize -sum(g)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError("LP failed")
    g = res.x
    # Scale to target_sum
    sum_g = g.sum()
    if sum_g == 0:
        return np.full_like(f, target_sum / n)
    scale = target_sum / sum_g
    g_scaled = g * scale
    # Clip small negative due to numerical error
    g_scaled = np.clip(g_scaled, 0, 1000)
    # Project onto simplex to ensure exact sum (optional)
    g_scaled = _simplex_projection(g_scaled, target_sum)
    return g_scaled
```

"""
Hybrid optimisation for the step‑function heights.

The objective returned by `evaluate_sequence` is the maximum entry of
the self‑convolution once the ℓ¹‑norm is fixed to √(2 n).  This program
searches for a sequence with a smaller maximum by combining three ideas:

1. **Smoothed‑max gradient descent** – minimise a log‑sum‑exp
   approximation of the max, compute its exact gradient via a single
   convolution, and project back onto the probability simplex.

2. **Occasional linear‑programming boosts** – for the current sequence
   `f` we solve the LP
        maximise   Σ g_j
        s.t.      f * g ≤ M   (element‑wise, M = max(f*f))
        g ≥ 0
   using `scipy.optimize.linprog`.  The solution `g` is rescaled to the
   required ℓ¹‑norm and used as a new candidate; this often jumps out of
   shallow local minima.

3. **Random restarts & occasional perturbations** – if no improvement
   is seen for a while the algorithm restarts from a fresh Dirichlet
   sample, and small random mass transfers are injected to keep the
   search stochastic.

The routine respects the 1000 s time budget, prints progress and returns
the best sequence found (as a plain Python list).  All helper functions
are defined at the top level, use no closures or lambdas, and rely only
on NumPy, SciPy and the provided `evaluate_sequence`.
"""

import time
import math
import random
import numpy as np

# optional imports – SciPy is required for the LP step
try:
    from scipy.optimize import linprog
    from scipy.sparse import diags
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project `v` onto the simplex { w ≥ 0 , Σ w = s }."""
    n = v.size
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_gradient(seq: np.ndarray, tau: float):
    """
    Compute the gradient of the log‑sum‑exp (soft‑max) of the self‑convolution.
    Returns (gradient, conv), where ``conv = np.convolve(seq, seq)``.
    """
    conv = np.convolve(seq, seq)                     # length 2·n‑1
    shift = np.max(conv)                             # subtraction for stability
    # soft‑max weights (numerically stable)
    exp_vals = np.exp((conv - shift) / tau)
    w = exp_vals / np.sum(exp_vals)                  # shape (2·n‑1,)
    # gradient = Σ_k w_k * ∂conv_k/∂seq  ;  ∂conv_k/∂seq_i = seq_{k‑i}
    rev_w = w[::-1]
    grad_full = np.convolve(seq, rev_w)              # length 3·n‑2
    n = seq.size
    # extract the required n entries: grad_i = grad_full[n‑1‑i]
    grad = grad_full[n - 1::-1][:n]
    return grad, conv


def _lp_optimize(f: np.ndarray, M: float, target_sum: float) -> np.ndarray:
    """
    Linear‑programming step: maximise Σ g_j subject to
        f * g ≤ M   (element‑wise) , g ≥ 0 .
    The solution is scaled to have ℓ¹‑norm = ``target_sum`` and projected
    back onto the simplex.
    """
    if not _SCIPY_AVAILABLE:
        raise RuntimeError("SciPy not available – cannot run LP step")

    n = f.size
    # Build the (2·n‑1) × n convolution matrix using sparse diagonals.
    # For offset -i the diagonal consists of the constant f_i.
    offsets = -np.arange(n)                     # 0, -1, -2, …, -(n‑1)
    data = [np.full(n, f_i, dtype=np.float64) for f_i in f]
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    b = np.full(2 * n - 1, float(M), dtype=np.float64)

    # maximise Σ g_j  -->  minimise –Σ g_j
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Use the HiGHS solver (fast sparse support)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        raise RuntimeError("LP failed: " + str(res.message))

    g = res.x
    total = g.sum()
    if total == 0.0:
        # fallback to uniform distribution
        return np.full_like(f, target_sum / n)

    scale = target_sum / total
    g_scaled = g * scale
    g_scaled = np.clip(g_scaled, 0.0, 1000.0)
    g_scaled = _simplex_projection(g_scaled, target_sum)
    return g_scaled

    """
    Search for a non‑negative step‑function height sequence that minimises
    the evaluation function.  The routine runs for up to ``budget_s`` seconds

    """
    np.random.seed(seed)
    random.seed(seed)

    # ----------------------------------------------------------------------

    target_sum = math.sqrt(2.0 * n)          # the normalising ℓ¹‑norm

    init = _simplex_projection(init, target_sum)

    best_seq = init.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # a random Dirichlet start (often a good diverse seed)

    rand_score = evaluate_sequence(rand_seq.tolist())

            max_delta = target_sum * 0.02 * tau

                    print(f"[{iteration:6d}] New best (grad): {best_score:.6f}")
                    no_improve = 0

                    no_improve += 1
            else:

            cur_seq, cur_score = cand, cand_score
