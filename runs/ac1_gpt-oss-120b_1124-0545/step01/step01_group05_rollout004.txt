def projected_gradient_descent(a, max_iters, time_budget):
    best_a = a.copy()
    best_val = fast_score(a)
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_budget:
            break
        # compute conv
        conv = _conv_max_and_full(a)  # returns full conv array
        i_max = np.argmax(conv)
        # gradient (subgrad) at max
        # compute gradient = 2 * shifted version of a
        # For each j, gradient_j = 2*a[i_max - j] if 0 <= i_max - j < n else 0
        grad = np.zeros_like(a)
        # vectorized:
        # indices j where i_max - j in [0,n)
        lo = max(0, i_max - (n-1))
        hi = min(n-1, i_max)
        # j runs from lo to hi
        # then partner = i_max - j
        grad[lo:hi+1] = 2 * a[i_max - np.arange(lo, hi+1)]
        # step size maybe adapt
        eta = 0.1 / np.sqrt(it+1) # diminishing steps
        a_new = a - eta * grad
        # project onto simplex
        a_new = project_simplex(a_new)
        # evaluate fast score
        cur_score = _fast_score(a_new)
        if cur_score < best_val:
            best_val = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_val
```

def project_simplex(v):
    # input: numpy array
    # returns projection onto simplex (non-negative, sum=1)
    n = v.shape[0]
    # algorithm: sort descending, find rho, compute theta.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The evaluation is homogeneous, so we keep the vector normalised to sum=1.
The algorithm works in three stages under the given time budget:

1. **Broad random search** over several parametrised families
   (centered exponentials, Gaussians, cosine‑exponentials,
   two‑block symmetric, Dirichlet, mixture of exponentials and
   small perturbations of the current best).  Each candidate is
   evaluated with a fast FFT‑based surrogate; an improvement is kept
   as the new best.

2. **Projected sub‑gradient descent** on the objective.
   For the index `i_max` where the convolution attains its maximum we
   use the sub‑gradient `∇_i_max (a*a) = 2·shift_i_max(a)`.  A diminishing
   step size is taken and the result is projected back onto the
   probability simplex (non‑negative, sum = 1).  Whenever the surrogate
   score improves we record the new best.

3. **Fine‑grained random tweaks** (pairwise mass transfers) to polish
   the solution.

All heavy work uses FFT‑based convolution (O(n log n)), so even for
vectors of several thousand entries the method easily stays within the
1000 s limit.  The final vector is returned as a plain Python list;
its score equals the official `evaluate_sequence` result because the
vector is normalised to sum = 1 and all entries are far below the
1000‑clamp.
"""

import time
import math

# ----------------------------------------------------------------------
# Helper utilities

def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (real FFT, fast)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution of a."""
    return float(_conv_full(a).max())


def _fast_score(a: np.ndarray) -> float:
    """Surrogate of `evaluate_sequence` when sum(a)=1."""
    n = a.shape[0]
    return 2.0 * n * _conv_max(a)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, sum x_i = 1}
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive; return uniform vector
        return np.full(n, 1.0 / n, dtype=float)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def resample_to_length(seq: np.ndarray, target_len: int) -> np.ndarray:
    """Linear interpolation of a 1‑D sequence to a new length."""
    old_len = seq.shape[0]
    if old_len == target_len:
        out = seq.astype(float).copy()
    else:
        x_old = np.linspace(0.0, 1.0, old_len)
        x_new = np.linspace(0.0, 1.0, target_len)
        out = np.interp(x_new, x_old, seq).astype(float)
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out = np.full(target_len, 1.0 / target_len, dtype=float)
    else:
        out /= s
    return out

def make_centered_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·|i - centre|).  lam ≥ 0."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * np.abs(i - centre))
    a /= a.sum()
    return a


def make_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-(i-centre)^2 / (2σ²)).  σ > 0."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def make_cosine_exponential(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ exp(alpha * cos(2π i / n)).
    For α>0 the distribution peaks at i=0, for α<0 it is flatter.
    """
    i = np.arange(n, dtype=float)
    a = np.exp(alpha * np.cos(2.0 * math.pi * i / n))
    a /= a.sum()
    return a


def make_two_block_symmetric(n: int, m: int, high_frac: float) -> np.ndarray:
    """
    Symmetric two‑block shape.
    First `m` entries and last `m` entries receive the same high value,
    the interior receives the remaining mass.
    `high_frac` ∈ [0,1] controls how much total mass goes to the two ends.
    """
    if m <= 0 or 2 * m > n:
        raise ValueError("invalid block size for symmetric two‑block")
    high = high_frac / (2 * m)
    low = (1.0 - high_frac) / (n - 2 * m)
    a = np.full(n, low, dtype=float)
    a[:m] = high
    a[-m:] = high
    a /= a.sum()
    return a


def make_mix_exponential(n: int, w: float, lam1: float, lam2: float) -> np.ndarray:
    """Mixture of two centered exponentials."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = w * np.exp(-lam1 * np.abs(i - centre)) + (1.0 - w) * np.exp(-lam2 * np.abs(i - centre))
    a /= a.sum()
    return a

# Main search routine

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function with a low evaluation score.
    The routine runs for at most `budget_s` seconds and returns the best
    sequence it discovered (as a regular Python list of floats).
    """
    np.random.seed(seed)
    start = time.time()
    # safety margin so we can always return a result
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Choose a working length.
    #    Larger n gives us more freedom to flatten the auto‑correlation.
    # ------------------------------------------------------------------
    n = 4000                     # feel free to adjust; 4000 is safe within budget
    # initialise with the uniform distribution
    best_a = np.full(n, 1.0 / n, dtype=float)
    best_score = _fast_score(best_a)

    # ------------------------------------------------------------------
    # 2. If the platform already supplied a construction, up‑sample it.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old_seq = np.array(height_sequence_1, dtype=float)
        cand = resample_to_length(old_seq, n)
        sc = _fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print("[init] Imported previous best, score:", best_score)

    # ------------------------------------------------------------------
    # 3. Broad random search over parametrised families.
    # ------------------------------------------------------------------
    families = ["centered_exp", "gaussian", "cosine_exp",
                "two_block_sym", "dirichlet", "mix_exp", "perturb"]
    probs = [0.13, 0.13, 0.13, 0.15, 0.20, 0.13, 0.13]   # sum to 1
    it = 0
    while time.time() < deadline * 0.70:   # spend about 70 % of the time here
        it += 1
        method = np.random.choice(families, p=probs)

        if method == "centered_exp":
            lam = np.random.exponential(scale=0.5)   # heavy‑tailed draw
            # clamp lam to a reasonable interval
            lam = min(lam, 10.0)
            a = make_centered_exponential(n, lam)

        elif method == "gaussian":
            sigma = np.random.uniform(0.05 * n, 0.4 * n)
            a = make_gaussian(n, sigma)

        elif method == "cosine_exp":
            alpha = np.random.uniform(-5.0, 5.0)
            a = make_cosine_exponential(n, alpha)

        elif method == "two_block_sym":
            m = np.random.randint(int(0.01 * n), int(0.25 * n) + 1)
            high_frac = np.random.uniform(0.0, 0.9)
            a = make_two_block_symmetric(n, m, high_frac)

        elif method == "dirichlet":
            alpha = np.random.uniform(0.1, 5.0)
            a = np.random.dirichlet([alpha] * n)

        elif method == "mix_exp":
            w = np.random.uniform(0.0, 1.0)
            lam1 = np.random.exponential(scale=0.5)
            lam2 = np.random.exponential(scale=0.5)
            lam1 = min(lam1, 10.0)
            lam2 = min(lam2, 10.0)
            a = make_mix_exponential(n, w, lam1, lam2)

        else:   # "perturb" – small random move from the current best
            a = best_a.copy()
            i, j = np.random.choice(n, 2, replace=False)
            delta = a[i] * np.random.uniform(-0.3, 0.3)
            delta = np.clip(delta, -a[i] + 1e-12, 1e12)
            a[i] -= delta
            a[j] += delta
            a = project_simplex(a)

        # sanity: non‑negativity + normalisation
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s

        # fast surrogate evaluation
        score = _fast_score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            elapsed = time.time() - start
            print(f"[{elapsed:6.1f}s] New best (random) = {best_score:.6f} "
                  f"iter={it} method={method}")

        if it % 2000 == 0:
            elapsed = time.time() - start
            remaining = max(0.0, deadline - time.time())
            print(f"[{elapsed:6.1f}s] iter={it} best={best_score:.6f} "
                  f"time left≈{remaining:.0f}s")

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient descent on the max‑convolution objective.
    # ------------------------------------------------------------------
    print("[phase] Starting sub‑gradient descent")
    grad_it = 0
    while time.time() < deadline * 0.95:   # leave ~5 % for polishing
        grad_it += 1
        conv = _conv_full(best_a)
        i_max = int(np.argmax(conv))

        # compute sub‑gradient g_i = 2 * a_{i_max - i} (if in range)
        g = np.zeros_like(best_a)
        lo = max(0, i_max - (n - 1))
        hi = min(i_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        g[idx] = 2.0 * best_a[i_max - idx]

        # step size schedule – diminishing
        eta = 0.5 / math.sqrt(grad_it + 1.0)

        # try a few shrink factors to guarantee improvement
        improved = False
        cur_score = _fast_score(best_a)
        for factor in (1.0, 0.5, 0.25):
            a_candidate = best_a - eta * factor * g
            a_candidate = project_simplex(a_candidate)
            cand_score = _fast_score(a_candidate)
            if cand_score < cur_score - 1e-12:
                best_a = a_candidate
                best_score = cand_score
                improved = True
                elapsed = time.time() - start
                print(f"[{elapsed:6.1f}s] Descend iter={grad_it} "
                      f"score={best_score:.6f} i_max={i_max}")
                break
        # if none of the shrink factors improve, simply take a tiny move
        if not improved:
            # add a tiny random jitter to escape possible stagnation
            i, j = np.random.choice(n, 2, replace=False)
            delta = best_a[i] * np.random.uniform(-0.01, 0.01)
            delta = np.clip(delta, -best_a[i] + 1e-12, 1e12)
            a_jitter = best_a.copy()
            a_jitter[i] -= delta
            a_jitter[j] += delta
            a_jitter = project_simplex(a_jitter)
            jitter_score = _fast_score(a_jitter)
            if jitter_score < best_score:
                best_a = a_jitter
                best_score = jitter_score
                elapsed = time.time() - start
                print(f"[{elapsed:6.1f}s] Jitter iter={grad_it} score={best_score:.6f}")

        if grad_it % 1000 == 0:
            elapsed = time.time() - start
            remaining = max(0.0, deadline - time.time())
            print(f"[{elapsed:6.1f}s] sub‑grad iter={grad_it} best={best_score:.6f} "
                  f"time left≈{remaining:.0f}s")

    # ------------------------------------------------------------------
    # 5. Final fine‑grained random tweaks (pairwise transfers).
    # ------------------------------------------------------------------
    print("[phase] Final polishing")
    fine_it = 0
    while time.time() < deadline:
        fine_it += 1
        i, j = np.random.choice(n, 2, replace=False)
        # move a small fraction of mass from i to j
        delta = best_a[i] * np.random.uniform(-0.03, 0.03)
        delta = np.clip(delta, -best_a[i] + 1e-12, 1e12)
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = project_simplex(a_new)
        sc = _fast_score(a_new)
        if sc < best_score - 1e-12:
            best_a = a_new
            best_score = sc
            elapsed = time.time() - start
            print(f"[{elapsed:6.1f}s] Polishing iter={fine_it} score={best_score:.6f}")

        if fine_it % 5000 == 0:
            elapsed = time.time() - start
            remaining = max(0.0, deadline - time.time())
            print(f"[{elapsed:6.1f}s] polish iter={fine_it} best={best_score:.6f} "
                  f"time left≈{remaining:.0f}s")

    # ------------------------------------------------------------------
    # 6. Return the best sequence found (as a Python list)
    # ------------------------------------------------------------------
    print(f"[done] Best score = {best_score:.6f} after {time.time() - start:.1f}s")
    return best_a.tolist()
