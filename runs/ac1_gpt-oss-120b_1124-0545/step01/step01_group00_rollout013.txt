def solve_lp(a, M):
    n = len(a)
    rows = []
    cols = []
    data = []
    b_ub = np.full(2*n-1, M)
    for k in range(2*n-1):
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        # indices j in [lo, hi]
        # coefficient for g_j is a_{k-j}
        idx = np.arange(lo, hi+1)
        coeff = a[k - idx]
        rows.extend([k]*len(idx))
        cols.extend(idx.tolist())
        data.extend(coeff.tolist())
    A_ub = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)  # maximize sum => minimize -sum
    bounds = [(0, None)]*n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback: return current a maybe
        return a.copy()
    g0 = res.x
    # scale to sum = 1 (or original sum)
    s = g0.sum()
    if s <= 0:
        return np.ones_like(a) / len(a)
    g = g0 / s
    return g
```

def simplex_projection(v):
    # returns projection onto {x >=0, sum x = 1}
    # Implementation from Duchi et al. (2008).
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Projected gradient descent on a smooth approximation of the max‑convolution.
==========================================================================

The objective to minimise is

    f(a) = 2·n·max_k (a * a)[k] / (∑a)^2

where ``*`` is the discrete convolution.  
We keep the vector normalised (`∑a = 1`) and enforce non‑negativity
by projecting every iterate onto the probability simplex.

The non‑differentiable max is approximated by a temperature‑controlled
soft‑max (log‑sum‑exp).  For a given temperature β

    w_k(β) ∝ exp(β·(b_k – max(b)))   with   b = a * a

and the gradient of the smooth objective is

    ∇≈ f(a) = 2·corr(a, w)                (corr = linear cross‑correlation)

Both the convolution `b` and the correlation are evaluated by FFTs,
so each iteration costs O(n log n).  The temperature β is increased
linearly with the elapsed fraction of the time budget, which makes the
updates first exploratory (low β → smooth landscape) and later
exploitation‑oriented (high β → true max).

A simple diminishing stepsize `η = η0 / sqrt(t+1)` together with a
projection onto the simplex yields a fast, memory‑light search.  The
best sequence seen during the run is returned.

The routine respects the 1000 s budget, prints progress information
and works for any `n` (default 2000).  If a previously found sequence
named ``height_sequence_1`` exists it is used as a warm start.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact score used by the challenge (∑a is not assumed to be 1)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    b = np.convolve(a, a)
    return float(2.0 * len(a) * b.max() / (s * s))


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex:
        {x : x_i >= 0, sum_i x_i = 1}
    Implementation of the algorithm from
    Duchi et al., "Efficient Projections onto the L1‑Ball for
    Learning in High Dimensions", ICML 2008.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:          # all entries are non‑positive
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (max_b, grad) where
        max_b = max_k (a*a)[k]
        grad   = ∇_a  (smooth max of a*a with temperature beta)
    Both the convolution and the correlation needed for the gradient
    are computed with FFTs.  The length of the transforms is the next
    power‑of‑two >= 3·n‑2, which is enough for a * a (length 2n‑1) and
    the correlation with the soft‑max weights (length 2n‑1).
    """
    n = a.shape[0]
    m = 2 * n - 1                     # length of a*a

    # FFT length (power of two for speed) covering correlation length 3n‑2
    L = 1 << ((3 * n - 2) - 1).bit_length()

    # forward transform of a (zero‑padded to L)
    A = np.fft.rfft(a, n=L)

    # a * a  (auto‑correlation)  – only first (2n‑1) entries are needed
    b_full = np.fft.irfft(A * A, n=L)[:m]

    max_b = b_full.max()

    # soft‑max weights over the 2n‑1 positions (stable by subtracting max)
    shifted = b_full - max_b                 # <= 0
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:                         # numerical safety
        w = np.full_like(w, 1.0 / m)
    else:
        w /= w_sum

    # correlation a ⋆ w  (linear cross‑correlation)
    # linear correlation can be obtained from A * conj(W) in the frequency domain
    W = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(A * np.conj(W), n=L)

    # index offset: correlation lag j corresponds to index (len(w)-1)+j
    start = m - 1               # = 2n‑2
    grad = 2.0 * corr_full[start:start + n]   # length n

    return float(max_b), grad

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function within the given time budget.
    Uses projected gradient descent on a smooth approximation of the
    max‑convolution.  The best vector found is returned as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # initialisation (warm‑start if a previous solution is available)
    # ------------------------------------------------------------------
    a = None
    if "height_sequence_1" in globals():
        seq = globals()["height_sequence_1"]
        if isinstance(seq, (list, tuple, np.ndarray)):
            seq_arr = np.asarray(seq, dtype=float)
            if seq_arr.size >= n:
                a = seq_arr[:n].copy()
            else:
                # pad with a tiny uniform mass to reach length n
                a = np.zeros(n, dtype=float)
                a[:seq_arr.size] = seq_arr
    if a is None:
        a = np.random.rand(n)

    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.random.rand(n)
    a /= a.sum()                     # normalise to the simplex

    best_seq = a.copy()
    best_score = compute_score(best_seq)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # hyper‑parameters (tuned for the 1000 s budget)
    # ------------------------------------------------------------------
    eta0 = 0.5          # initial stepsize (will be divided by sqrt(t+1))
    beta_start = 1.0    # start temperature for the smooth max
    beta_end = 31.0     # temperature at the end of the run
    stagnation_timeout = 45.0            # seconds without improvement → restart
    last_improve = start_time
    iteration = 0

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # schedule (linear increase of beta, diminishing eta)
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 10.0)          # in [0, 1]
        beta = beta_start + (beta_end - beta_start) * progress
        eta = eta0 / np.sqrt(iteration + 1.0)

        # --------------------------------------------------------------
        # compute smooth max and its gradient
        # --------------------------------------------------------------
        max_b, grad = smooth_max_and_grad(a, beta)

        # current true score (exact, not the smoothed one)
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{iteration:6d}] new best = {best_score:.6f}  t={elapsed:.1f}s")

        # --------------------------------------------------------------
        # gradient descent step + projection onto the simplex
        # --------------------------------------------------------------
        a_candidate = a - eta * grad
        a = simplex_projection(a_candidate)

        # --------------------------------------------------------------
        # occasional random perturbation to escape flat regions
        # --------------------------------------------------------------
        if np.random.rand() < 0.005:          # 0.5 % chance
            noise = np.random.randn(n) * 1e-4
            a = simplex_projection(a + noise)

        # --------------------------------------------------------------
        # stagnation handling – restart from the best seen so far
        # --------------------------------------------------------------
        if time.time() - last_improve > stagnation_timeout:
            print(f"[{iteration:6d}] stagnation → restart")
            a = best_seq.copy()
            # add a tiny perturbation to avoid cycling
            a = simplex_projection(a + 1e-6 * np.random.rand(n))
            last_improve = time.time()

        iteration += 1

    total_time = time.time() - start_time
    print(f"[done] iters={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
    return best_seq.tolist()
