```
def project_simplex(v):
    # From https://arxiv.org/abs/1309.1541
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function sequence with a lower evaluation score.

The algorithm works in three stages:

1. **Initialisation** – start from the analytically optimal central‑binomial
   sequence (or from a previously found global `height_sequence_1` if it
   exists) and normalise it so that its L1‑norm equals 1.  The evaluation
   function is invariant under scaling, therefore fixing the sum simplifies
   the objective to minimising the maximum self‑convolution value `M`.

2. **Sub‑gradient descent on the maximum convolution** – each iteration
   computes the full convolution `c = a*a`.  The indices where `c` attains its
   maximum are identified and a uniform weight `w` on those indices is built.
   The sub‑gradient of `max(c)` w.r.t. the coefficient vector `a` is
   `grad = 2 * (w_rev * a)` (where `*` denotes convolution and `w_rev` the
   reversed weight vector).  A line‑search style step length is adapted
   greedily: a trial update `a - step*grad` is projected onto the simplex
   (non‑negative entries, sum = 1) and accepted only if it reduces the
   maximum convolution.  The step size is increased on success and halved on
   failure, allowing fast progress initially and fine‑grained adjustments later.

3. **Fine‑grained stochastic hill‑climb** – after the deterministic phase a
   lightweight random walk (moving a moderate fraction of mass from one
   index to another) is performed to capture any remaining local improvements.

All operations use fast FFT‑based convolutions; the whole search respects the
global time budget (default 1000 s).  The returned list is ready for the
provided `evaluate_sequence` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: build the truncated central binomial sequence.
# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """Return a = C(2i,i)/4**i for i=0..length‑1 (exact recurrence)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, a_i >= 0).
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][rho - 1] / rho
    w = np.maximum(v - theta, 0.0)
    # due to numerical errors the sum may deviate slightly; renormalise
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Fast full convolution using a power‑of‑two FFT length.
# ----------------------------------------------------------------------
def full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of `x` and `y` using real FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    return conv


# ----------------------------------------------------------------------
# Main entry point required by the framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a
    lower `evaluate_sequence` score than the baseline binomial construction.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # 1) initialise the coefficient vector `a`
    # --------------------------------------------------------------
    n = 5000                      # a few thousand items give a good trade‑off
    try:
        # Use the previously discovered global if it exists
        init = np.array(height_sequence_1, dtype=np.float64)
        if init.size < n:
            init = np.pad(init, (0, n - init.size), mode="constant")
        else:
            init = init[:n].copy()
        a = init
        print("[init] using global height_sequence_1")
    except Exception:
        a = binomial_sequence(n)
        print("[init] using binomial seed")

    # Normalise to sum = 1 (evaluation is scale‑invariant)
    a = a / a.sum()
    a = project_simplex(a)                # ensure exact sum = 1

    # Initial score (fast: 2*n*max(conv) because sum=1)
    conv = full_convolution(a, a)
    M = conv.max()
    best_score = 2.0 * n * M
    best_a = a.copy()
    print(f"[init] n={n}, score={best_score:.6f}, max_conv={M:.8e}")

    # --------------------------------------------------------------
    # 2) Sub‑gradient descent on the maximum of the self‑convolution
    # --------------------------------------------------------------
    step = 0.5          # initial step length (will be adapted)
    min_step = 1e-8
    max_step = 5.0
    no_improve_cnt = 0
    max_no_improve = 200          # stagnation tolerance

    while time.time() < deadline and no_improve_cnt < max_no_improve:
        # --- compute current convolution and the set of maximal indices ---
        conv = full_convolution(a, a)
        M = conv.max()
        max_mask = conv == M
        cnt_max = max_mask.sum()
        # uniform weight on all maximal positions
        w = np.zeros_like(conv)
        w[max_mask] = 1.0 / cnt_max

        # --- gradient of max(conv) w.r.t. `a` (sub‑gradient) ---
        # grad_i = 2 * sum_k w_k * a_{k-i}  (out‑of‑range indices give 0)
        w_rev = w[::-1]
        # convolution w_rev * a   (full length = 3*n-2)
        grad_full = full_convolution(w_rev, a)
        # take the slice (2*n-2) - i, i = 0..n-1
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # --- try an update -------------------------------------------------
        a_candidate = a - step * grad
        a_candidate = project_simplex(a_candidate)

        conv_cand = full_convolution(a_candidate, a_candidate)
        M_cand = conv_cand.max()
        score_cand = 2.0 * n * M_cand

        if M_cand < M:                     # strict improvement
            a = a_candidate
            M = M_cand
            if score_cand < best_score:
                best_score = score_cand
                best_a = a_candidate.copy()
                print(f"[subgrad] iter improving: score={best_score:.6f}, "
                      f"step={step:.4f}, max={M:.8e}")
            # increase step size modestly
            step = min(max_step, step * 1.05)
            no_improve_cnt = 0
        else:
            # no improvement – shrink step size
            step = max(min_step, step * 0.5)
            no_improve_cnt += 1

    # --------------------------------------------------------------
    # 3) Light stochastic hill‑climb (biased moves)
    # --------------------------------------------------------------
    attempts = 0
    max_attempts = 300_000
    a = best_a.copy()                # start hill‑climb from the best found
    while time.time() < deadline and attempts < max_attempts:
        attempts += 1
        # bias source towards the early half (large contribution to max)
        i = np.random.randint(n // 2)
        if a[i] <= 1e-12:
            continue
        # bias target towards the later half
        j = np.random.randint(n // 2, n)
        if i == j:
            continue

        # move a modest fraction (up to 5 %) of the source mass
        frac = np.random.rand() * 0.05
        delta = frac * a[i]
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # projection not required because total mass is preserved,
        # but we enforce non‑negativity explicitly
        if new_a[i] < 0:
            continue

        conv = full_convolution(new_a, new_a)
        M = conv.max()
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best_a = new_a.copy()
            a = new_a
            print(f"[hill] iter={attempts}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final validation with the official evaluator
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        official = evaluate_sequence(final_seq)
        print(f"[final] official evaluate_sequence = {official:.6f}")
    except Exception as exc:          # defensive: should never happen
        print("[final] evaluation failed:", exc)

    return final_seq
