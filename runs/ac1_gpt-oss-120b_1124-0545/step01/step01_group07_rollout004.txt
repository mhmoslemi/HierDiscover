"""
Propose a low‑score step‑function.

Algorithm
---------
1. **Warm‑start** – if a historic solution (`height_sequence_1`) exists it is
   evaluated first.
2. **Parametric sweep** – a small set of classic analytic shapes (constant,
   linear, quadratic, triangular, valley, sinusoidal, exponential‑decay,
   power‑law) is tested for several lengths.  This reproduces the earlier
   exploration but at negligible cost.
3. **Random‑shape search** – the core improvement.  For the remaining budget
   a large number of *uniformly random* non‑negative vectors are generated.
   For a vector of length `n` the average of its self‑convolution is
   `sum(a)^2/(2n)` and the max of the convolution concentrates around this
   average with a relative fluctuation ≈ O(1/√n).  Consequently, for
   moderate‑size vectors (e.g. `n≈3·10⁴`) the score is typically close to the
   theoretical optimum 1.  Each random candidate is evaluated with
   `evaluate_sequence`; the best one is retained.
4. **Local refinement** – starting from the best random candidate a fast
   simulated‑annealing / hill‑climbing loop (Gaussian perturbations of a few
   entries) tries to shave off the remaining excess.

The whole routine respects the supplied time budget (`budget_s`, default
1000 s).  Progress and intermediate best scores are printed.

The function returns the best list of non‑negative heights found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas – top‑level as required)
# ----------------------------------------------------------------------


def _ensure_valid(seq: np.ndarray) -> np.ndarray:
    """
    Clip the sequence to the limits imposed by ``evaluate_sequence`` and
    guarantee a non‑zero sum.  The operation does not affect the final score
    because the evaluation is homogeneous.
    """
    a = np.clip(seq, 0.0, 999.0)               # stay below the 1000 clamp
    s = a.sum()
    if s < 0.01:                               # avoid the “sum too close to zero” guard
        if s == 0.0:
            a += 0.01 / a.size
        else:
            a *= 0.01 / s
    return a


def _random_candidate(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Produce a random non‑negative shape of length ``n``.
    Uniform[0, 1) entries give a very low variance of the convolution and
    therefore scores close to 1 for sufficiently large ``n``.
    """
    return _ensure_valid(rng.random(n))


def _candidate_shapes(n: int):
    """
    Deterministic families used in the original sweep.
    """
    cand = []

    # constant
    cand.append(np.ones(n, dtype=float))

    # linear
    cand.append(np.arange(n, 0, -1, dtype=float))
    cand.append(np.arange(1, n + 1, dtype=float))

    # quadratic
    cand.append(np.square(np.arange(n, 0, -1, dtype=float)))
    cand.append(np.square(np.arange(1, n + 1, dtype=float)))

    # triangular “mountain”
    t = np.arange(1, n + 1, dtype=float)
    cand.append(np.minimum(t, n - t + 1))

    # valley (large at the ends, small in the centre)
    centre = (n - 1) / 2.0
    cand.append(1.0 / (1.0 + ((np.arange(n, dtype=float) - centre) ** 2)))

    # sine‑squared shape
    cand.append(np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) / (n + 1)) ** 2)

    # exponential decays
    for r in (0.95, 0.9, 0.8, 0.7, 0.6, 0.5):
        cand.append(np.power(r, np.arange(n, dtype=float)))

    # power‑law decays
    for p in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n + 1, dtype=float)
        cand.append(1.0 / (i ** p))

    return cand


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimizes ``evaluate_sequence``.
    The function returns the best sequence that was found before the time
    budget expires.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1          # small safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        seq = _ensure_valid(np.asarray(hs1, dtype=float))
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[{time.time() - start_time:.1f}s] Warm‑start "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic families (very cheap)
    # ------------------------------------------------------------------
    lengths = [64, 128, 256, 512, 1024, 2048, 4096, 8192]
    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _candidate_shapes(n):
            if time.time() >= deadline:
                break
            seq = _ensure_valid(cand)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"[{time.time() - start_time:.1f}s] "
                      f"deterministic ({n} terms) new best score={best_score:.6f}")
            if best_score <= 1.20:                  # early break – random part will beat this easily
                break

    # ------------------------------------------------------------------
    # Phase 2 – random‑shape exploration (core improvement)
    # ------------------------------------------------------------------
    # Choose a set of lengths that balance speed and quality.
    random_lengths = [2048, 4096, 8192, 16384, 32768]
    # Keep a short list of the best random candidates for later refinement.
    best_randoms = []

    while time.time() < deadline:
        # pick a length (favor larger ones as time progresses)
        if time.time() - start_time < budget_s * 0.3:
            n = rng.choice(random_lengths[:3])          # early stage – faster evaluations
        else:
            n = rng.choice(random_lengths)

        seq = _random_candidate(rng, n)
        score = evaluate_sequence(seq.tolist())

        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[{time.time() - start_time:.1f}s] Random n={n} "
                  f"new best score={best_score:.6f}")

        # remember a few top random runs for the final local search
        if len(best_randoms) < 5:
            best_randoms.append((score, seq.copy()))
        else:
            # keep the 5 smallest scores
            worst_idx = max(range(5), key=lambda i: best_randoms[i][0])
            if score < best_randoms[worst_idx][0]:
                best_randoms[worst_idx] = (score, seq.copy())

        # stop if we already reached a near‑optimal bound
        if best_score <= 1.02:
            print("Score close to theoretical optimum – stopping random search.")
            break

    # If nothing was found (unlikely) fall back to a random vector
    if best_seq is None:
        best_seq = rng.random(128).tolist()
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Phase 3 – local refinement (simulated annealing on the best random)
    # ------------------------------------------------------------------
    cur = np.asarray(best_seq, dtype=float)

    # linear temperature schedule
    start_temp, end_temp = 1.0, 0.001

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = start_temp * (1 - elapsed / total) + end_temp * (elapsed / total)

        # Choose a mutation type
        if rng.random() < 0.70:
            # Gaussian perturbation of a random 1 % of the entries
            m = max(1, int(0.01 * len(cur)))
            idx = rng.choice(len(cur), size=m, replace=False)
            delta = rng.standard_normal(m) * cur.mean() * 0.1
            new = cur.copy()
            new[idx] += delta
        else:
            # Reverse a short random block (re‑ordering)
            start_idx = rng.integers(0, len(cur))
            block_len = rng.integers(1, min(200, len(cur) - start_idx) + 1)
            new = cur.copy()
            new[start_idx:start_idx + block_len] = new[start_idx:start_idx + block_len][::-1]

        new = _ensure_valid(new)

        cand_score = evaluate_sequence(new.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = new.tolist()
            cur = new
            print(f"[{time.time() - start_time:.1f}s] Improved (local) score={best_score:.6f}")
            if best_score <= 1.01:
                print("Target close to 1 reached – terminating.")
                break
        else:
            # Simulated‑annealing acceptance
            prob = np.exp(-(cand_score - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur = new

    return best_seq
