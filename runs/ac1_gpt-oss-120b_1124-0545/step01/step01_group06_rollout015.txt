"""Improved search for a low‑value step function.

The algorithm consists of two phases.

Phase 1 – a fine parametric sweep of geometric families.
    a_i = r**i  (normalised to sum 1) with r = 1 – c/n.
    The sweep uses many values of c (0.1 … 10) on a length of 4000,
    keeping the best candidate.

Phase 2 – hard‑max exponentiated‑gradient descent.
    At each iteration the convolution a*a is computed, the index
    k_max of the current maximum is found and the sub‑gradient of
    that term (2·a_{k_max‑i}) is used.  An exponentiated update
    a_i ← a_i·exp(‑η·grad_i) is performed and the vector is projected
    back onto the probability simplex (∑a_i=1, a_i≥0).  The learning
    rate η decays like 1/√t.  The best sequence seen during the run
    is returned.

The whole routine respects the supplied time budget (default 1000 s)
and prints intermediate progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w≥0, sum w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    mask = u - cssv / ind > 0
    if not np.any(mask):
        theta = cssv[-1] / n
    else:
        rho = np.where(mask)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# families of parametric sequences (all normalised to unit L1‑norm)
# ----------------------------------------------------------------------
def _geom_normalized(n: int, r: float) -> np.ndarray:
    """Geometric series a_i = r**i, normalised to sum 1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _power_normalized(n: int, p: float) -> np.ndarray:
    """Power‑law a_i = 1/(i+1)^p, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s == 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _sym_geom_normalized(n: int, r: float) -> np.ndarray:
    """Symmetric geometric a_i = r**i + r**(n‑1‑i), normalised."""
    a = r ** np.arange(n, dtype=np.float64) + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    a /= a.sum()
    return a


def _evaluate_normed(a: np.ndarray) -> float:
    """
    Objective value when a is already a probability vector (sum = 1):
    2·n·max(conv(a,a)).
    """
    conv = _conv_fft(a, a)
    return float(2.0 * len(a) * np.max(conv))

    """

    # --------------------------------------------------------------

            print(f"[sweep] geom n={n:4d} c={c:6.3f}  val={val:.6f}")

    # --------------------------------------------------------------

    a = best_seq.copy()

        a = _project_simplex(a)

    # final sanity check – make sure we return plain Python floats
    return best_seq.tolist()
