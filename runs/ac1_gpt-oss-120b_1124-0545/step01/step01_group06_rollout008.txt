def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np

    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    best_seq = None
    best_val = np.inf

    # Helper to evaluate on normalized sequence (sum=1)
    def evaluate_norm(a):
        conv = _conv_fft(a, a)
        val = 2.0 * len(a) * np.max(conv)
        return float(val)

    # Parameter sweep
    lengths = [2000, 4000, 8000]  # increasing sizes
    for n in lengths:
        # keep at most maximal candidates.
        # uniform baseline
        a = np.ones(n, dtype=np.float64) / n
        val = evaluate_norm(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
        # geometric families
        for c in np.linspace(0.5, 20.0, 40):  # c bigger
            r = 1.0 - c / n
            if r <= 0:
                continue
            a = _geom(n, r)
            val = evaluate_norm(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
        # power families
        for p in np.linspace(0.0, 5.0, 30):
            a = _power(n, p)
            val = evaluate_norm(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
        # geom*power
        for c in np.linspace(0.5, 20.0, 20):
            r = 1.0 - c / n
            if r <= 0:
                continue
            for p in np.linspace(0.0, 5.0, 20):
                a = _geom_power(n, r, p)
                val = evaluate_norm(a)
                if val < best_val:
                    best_val, best_seq = val, a.copy()
        # random mixtures
        for _ in range(200):
            r1 = 1.0 - np.random.rand() * 0.9  # random r close to 1 maybe
            r2 = 1.0 - np.random.rand() * 0.9
            alpha = np.random.rand()
            idx = np.arange(n, dtype=np.float64)
            a = alpha * (r1 ** idx) + (1 - alpha) * (r2 ** idx)
            s = a.sum()
            if s == 0:
                continue
            a /= s
            val = evaluate_norm(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
        if time.time() >= deadline:
            break

    # Use best sequence for refinement via Adam with softmax
    if best_seq is None:
        # fallback
        n = 4000
        best_seq = np.ones(n, dtype=np.float64) / n
        best_val = evaluate_norm(best_seq)

    # Convert to log-odds variables for softmax
    y = np.log(best_seq + 1e-12)

    n = len(best_seq)
    # Adam hyperparameters
    lr = 0.1
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    max_iters = 20000
    t = 0
    beta_start = 5.0
    beta_end = 200.0
    # We'll increase beta exponentially to get sharper max
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        # compute a as softmax(y)
        exp_y = np.exp(y - np.max(y))  # mitigate overflow
        a = exp_y / np.sum(exp_y)

        # compute convolution via FFT
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[refine] iter {it:5d} new best {cur_val:.6f}")

        # smoothed max with temperature parameter beta
        beta = beta_start * ((beta_end / beta_start) ** (it / max_iters))
        # to avoid overflow, subtract max
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = np.sum(w)
        if w_sum == 0:
            # fallback to hard max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w = w / w_sum

        # gradient w.r.t a: derivative of sum_i w_i * conv_i is 2*(a_rev * w)
        grad_a = 2.0 * _conv_fft(a[::-1], w)[n-1:n-1+n]

        # backprop to y (softmax)
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    # Clip any negative small numbers (softmax ensures nonnegative)
    final_seq = best_seq.tolist()
    print(f"Best value found: {best_val}")
    return final_seq

"""
Propose a step‐function (non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three stages.

1.  A fast parametric sweep over several families (geometric, power‑law,
    geometric·power, a random mixture of two exponentials, and a symmetric
    geometric) for a few lengths (2000, 4000, 8000).  Every candidate is
    normalised to sum = 1 and evaluated with an FFT based linear convolution.
    The best candidate from the sweep is kept.

2.  The best candidate is refined with an Adam optimiser on the *soft‑max*
    parametrisation `a = softmax(y)`.  The loss is a smoothed‐max
    approximation `log‑sum‑exp(beta * conv)`.  The temperature ``beta`` is
    increased from a small value (≈5) to a large value (≈200) so that the
    optimiser ultimately minimises the true maximum.  The Adam update works
    directly on the log‑probabilities `y`; the simplex constraint is automatically
    satisfied by the soft‑max.

3.  During refinement the current best value is stored and printed.  The whole
    routine respects a time budget (default 1000 s) and returns the best
    sequence found as a plain list.
"""

import time
import numpy as np

# ---------------------------------------------------------------
# Basic building blocks
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution via FFT (full length = len(x)+len(y)-1).
    Uses real FFT for speed.
    """
    L = x.shape[0] + y.shape[0] - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ---------------------------------------------------------------
# Parametric families (all return a *normalized* probability vector)
# ---------------------------------------------------------------

def _geom(n: int, r: float) -> np.ndarray:
    """Geometric progression a_i ∝ r^i."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _power(n: int, p: float) -> np.ndarray:
    """Power law a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s == 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Combined geometric·power a_i ∝ r^i / (i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    if s == 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric: a_i = r^i + r^{n-1-i}."""
    a = r ** np.arange(n, dtype=np.float64) + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    a /= a.sum()
    return a


def _mix_geom(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    """Mixture of two exponentials: a_i ∝ α r1^i + (1-α) r2^i."""
    idx = np.arange(n, dtype=np.float64)
    a = alpha * (r1 ** idx) + (1.0 - alpha) * (r2 ** idx)
    s = a.sum()
    if s == 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


# ---------------------------------------------------------------
# Helper for evaluating a *normalised* candidate (sum = 1)
# ---------------------------------------------------------------

def _eval_norm(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a sum‑to‑one vector."""
    conv = _conv_fft(a, a)
    return float(2.0 * a.shape[0] * np.max(conv))


# ---------------------------------------------------------------
# Main search routine
# ---------------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that minimises
    `evaluate_sequence`.  The routine runs for approximately ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0      # small safety margin

    best_seq = None
    best_val = np.inf

    # -----------------------------------------------------------------
    # 1) Parametric sweep (very cheap compared to the later optimisation)
    # -----------------------------------------------------------------
    lengths = [2000, 4000, 8000]          # modest sizes – larger = more freedom
    # geometric decay parameter c (r = 1 - c/n) – explore a wider range
    c_vals = np.linspace(0.5, 20.0, 36)    # step ≈0.55
    # power‑law exponent
    p_vals = np.linspace(0.0, 6.0, 31)    # up to p=6

    for n in lengths:
        # uniform baseline
        a = np.ones(n, dtype=np.float64) / n
        val = _eval_norm(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[sweep] uniform  n={n:5d}  val={val:.6f}")

        # try a few families
        for c in c_vals:
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            # pure geometric
            a = _geom(n, r)
            val = _eval_norm(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[sweep] geom     n={n:5d} c={c:5.2f}  val={val:.6f}")

            # symmetric geometric
            a = _sym_geom(n, r)
            val = _eval_norm(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[sweep] sym‑g    n={n:5d} c={c:5.2f}  val={val:.6f}")

        for p in p_vals:
            # pure power law
            a = _power(n, p)
            val = _eval_norm(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[sweep] power    n={n:5d} p={p:5.2f}  val={val:.6f}")

        # combined geometric·power
        for c in c_vals:
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in p_vals:
                a = _geom_power(n, r, p)
                val = _eval_norm(a)
                if val < best_val:
                    best_val, best_seq = val, a.copy()
                    print(f"[sweep] g×p      n={n:5d} c={c:5.2f} p={p:5.2f}  val={val:.6f}")

        # random mixtures of two exponentials
        for _ in range(120):
            r1 = 1.0 - np.random.rand() * 0.95          # in (0.05,1]
            r2 = 1.0 - np.random.rand() * 0.95
            alpha = np.random.rand()
            a = _mix_geom(n, r1, r2, alpha)
            val = _eval_norm(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[sweep] mix‑g    n={n:5d} r1={r1:.3f} r2={r2:.3f} α={alpha:.3f}  val={val:.6f}")

        if time.time() >= deadline:
            break

    # If a global variable ``height_sequence_1`` is present, use it as an
    # additional candidate (the benchmark usually provides it).
    try:
        hs1 = np.array(height_sequence_1, dtype=np.float64)
        if hs1.size > 0:
            hs1 = hs1 / hs1.sum()
            val = _eval_norm(hs1)
            if val < best_val:
                best_val, best_seq = val, hs1.copy()
                print(f"[sweep] global   n={hs1.shape[0]:5d}  val={val:.6f}")
    except NameError:
        pass

    # -----------------------------------------------------------------
    # 2) Refinement – Adam on the soft‑max parameters
    # -----------------------------------------------------------------
    if best_seq is None:
        # fallback – should never happen
        n = 4000
        best_seq = np.ones(n, dtype=np.float64) / n
        best_val = _eval_norm(best_seq)

    n = best_seq.shape[0]
    y = np.log(best_seq + 1e-12)                      # log‑soft‑max parameters

    # Adam hyper‑parameters
    lr = 0.07
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    max_iters = 18000                                 # plenty within the budget
    beta_start = 5.0
    beta_end = 250.0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # soft‑max ↦ probability vector a (guaranteed non‑negative, sum=1)
        y_shift = y - np.max(y)                       # avoid overflow
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()

        # convolution and current score
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[refine] iter {it:5d} new best val={cur_val:.6f}")

        # ----- smoothed max (log‑sum‑exp) -----
        # geometric schedule for β (smooth → hard)
        beta = beta_start * ((beta_end / beta_start) ** (it / max_iters))
        shift = cur_max                                # numerical stabiliser
        w = np.exp(beta * (conv - shift))
        w_sum = np.sum(w)
        if w_sum == 0.0:
            # fallback to hard max (should rarely happen)
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient w.r.t. a (2 * conv(a_rev, w))
        grad_a = 2.0 * _conv_fft(a[::-1], w)[n - 1:n - 1 + n]

        # back‑prop through soft‑max: ∂L/∂y = a ⊙ (grad_a - 〈a,grad_a〉)
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)

        # Adam update on y
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional random jitter to escape shallow plateaus
        if (it + 1) % 3000 == 0:
            jitter = 0.2 * np.random.randn(n)
            y = np.log(best_seq + 1e-12) + jitter

    # -----------------------------------------------------------------
    # Return best sequence found
    # -----------------------------------------------------------------
    print(f"[final] best value = {best_val:.6f} with length {len(best_seq)}")
    return best_seq.tolist()
