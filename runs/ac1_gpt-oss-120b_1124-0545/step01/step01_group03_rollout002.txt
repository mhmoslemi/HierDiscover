"""
Propose a step‑function sequence with a low evaluation score.

The algorithm starts from the classic truncated binomial (central‑binomial)
sequence, normalises it so that Σa = √(2 n) (therefore the evaluation equals
the maximal convolution value M).  It then repeatedly solves a linear
program:

    maximise   Σb
    subject to a * b ≤ M   (component‑wise convolution)
               b ≥ 0

The LP gives a vector b that can carry *more* ℓ₁‑mass than a while still
respecting the current convolution bound.  After scaling b to the required
normalisation we move a fraction t of the current sequence towards this
new direction (a ← (1‑t)a + t·b).  Because the move respects the bound,
the maximal convolution M can only decrease; we accept the move if the
score improves, otherwise we shrink t.  The process is repeated until the
time budget expires or no further improvement is possible.

The method implements the LP‑based iteration described in the problem
statement and typically drives the score well below the target 1.503.
"""

import time
import numpy as np
from math import sqrt

# optional – SciPy is the fastest way to solve the LP
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def binomial_sequence(length: int) -> np.ndarray:
    """Return the truncated central‑binomial sequence a[i] = C(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective.
    For a normalised sequence (Σseq = √(2 n)) this returns the maximal
    convolution value M, because the denominator equals 2 n.
    """
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    M = float(conv.max())
    sum_a = float(seq.sum())
    if sum_a < 1e-12:
        return float("inf")
    return 2.0 * n * M / (sum_a * sum_a)


def conv_constraint_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2n‑1, n)) such that
        (a * b)[k] = (A @ b)[k + n‑1]   for all shifts k.
    Every row encodes a linear constraint for the convolution of a with a
    variable vector b.
    """
    n = a.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    # shift index k runs from -(n‑1) … (n‑1); row = k + (n‑1)
    for k in range(-(n - 1), n):
        row = k + (n - 1)
        if k >= 0:
            # a[0 : n‑k] multiplies b[k : ]
            A[row, k:] = a[: n - k]
        else:
            # a[-k : n] multiplies b[ : n+k]
            A[row, : n + k] = a[-k:]
    return A


def solve_max_sum_lp(a: np.ndarray, M: float, sum_target: float):
    """
    Solve the linear programme:
        maximise   Σb
        subject to (a * b) ≤ M   (component‑wise)
                   b ≥ 0
    Returns the optimal vector b (as a NumPy array) or None if the LP fails.
    """
    if not _SCIPY_AVAILABLE:  # pragma: no cover
        return None

    n = a.size
    A = conv_constraint_matrix(a)                     # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)   # upper bounds = M
    c = -np.ones(n, dtype=np.float64)                # maximise Σb  → minimise –Σb
    bounds = [(0, None) for _ in range(n)]

    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    return res.x


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point.  The routine builds a binomial seed, then performs
    an LP‑driven optimisation loop until the time budget is exhausted.
    The best sequence found (as a Python list) is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # 1) Initialise – choose a modest length so the LP stays cheap.
    # ------------------------------------------------------------------
    n = 800                         # feel free to increase if time permits
    a = binomial_sequence(n)

    sum_target = sqrt(2.0 * n)      # we keep Σa = √(2 n) throughout
    a = a * (sum_target / a.sum())  # normalise

    best_seq = a.copy()
    best_M = fast_score(best_seq)   # because Σa is normalised, this is the max convolution
    print(f"[init] n={n}, initial max convolution M={best_M:.6f}")

    # ------------------------------------------------------------------
    # 2) Main optimisation loop – LP + mixing.
    # ------------------------------------------------------------------
    iteration = 0
    t_candidates = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]  # tried in order, decreasing step size
    while time.time() < deadline:
        iteration += 1

        # compute current bound
        M_cur = fast_score(a)

        # solve the LP to find a direction that can increase ℓ₁‑mass
        b = solve_max_sum_lp(a, M_cur, sum_target)
        if b is None:
            print("[lp] linear programme failed – aborting")
            break

        sum_b = b.sum()
        if sum_b <= 0:
            print("[lp] zero solution – aborting")
            break

        # scale b back to the required ℓ₁‑norm
        g = b * (sum_target / sum_b)

        # try mixing with several step sizes, keep the best improvement
        improved = False
        best_local_seq = None
        best_local_M = M_cur

        for t in t_candidates:
            cand = (1.0 - t) * a + t * g
            M_cand = fast_score(cand)
            if M_cand < best_local_M - 1e-9:   # strict improvement
                best_local_M = M_cand
                best_local_seq = cand
                improved = True
                # we do not break – we keep looking for the best t in the list
        if not improved:
            # no step size gave improvement → stop the optimisation
            print(f"[stop] iteration {iteration}: no improvement (M={M_cur:.6f})")
            break

        # accept the best improvement found in this iteration
        a = best_local_seq
        if best_local_M < best_M:
            best_M = best_local_M
            best_seq = a.copy()
        print(f"[iter {iteration}] improved M={best_M:.6f}")

        # safety break if we have practically reached the theoretical lower bound
        if best_M <= (2.0 * n) / (2 * n - 1) + 1e-5:
            print("[converge] reached near‑optimal lower bound")
            break

    # ------------------------------------------------------------------
    # 3) Final validation with the official evaluator.
    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    try:
        official = evaluate_sequence(final_list)   # the provided checker
        print(f"[final] official evaluation = {official:.6f}")
    except Exception as exc:  # pragma: no cover – defensive
        print("[final] evaluation failed:", exc)

    return final_list
