"""
Propose a step‑function sequence that minimises the metric used by ``evaluate_sequence``.
The algorithm combines three ideas:

1. **Catalan start** – the coefficients of (1‑x)^‑½ (normalised to sum = 1)
   give a good baseline (objective ≈ π/2).

2. **Projected sub‑gradient descent** – minimise the maximal entry of the
   self‑convolution while staying in the probability simplex.

3. **Linear‑programming refinement** – given the current sequence *a*
   (with peak M = max(a*a)), solve a linear program that finds a non‑negative
   sequence *g* with the *same* peak but maximal total mass.
   Mixing a convex combination of *a* and the normalised *g* strictly reduces
   the peak.  A small line‑search on the mixing weight is performed on each
   LP step.

4. **Stochastic local‑search** – final perturbations (mass transfer between
   random indices) with occasional uphill moves (simulated‑annealing).

The whole search respects the 1000 s time budget; the best sequence seen so
far is returned.  If ``scipy`` is unavailable the algorithm falls back to the
gradient + random phases only.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (pure, top‑level)
# ----------------------------------------------------------------------


def compute_catalan_sequence(n: int) -> np.ndarray:
    """Catalan coefficients (1‑x)^‑½, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def max_convolution_fft(a: np.ndarray, fft_len: int) -> float:
    """Full linear convolution a*a via FFT and return its maximum value."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * len(a) - 1]
    return float(conv.max())


def gradient_step(a: np.ndarray, step: float, fft_len: int) -> np.ndarray:
    """Projected sub‑gradient step for max (a*a)."""
    n = len(a)
    # locate a shift where the convolution attains its max
    conv = np.fft.irfft(np.fft.rfft(a, n=fft_len) ** 2, n=fft_len)
    conv = conv[: 2 * n - 1]
    k = int(conv.argmax())
    # indices that contribute to the maximal shift
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    idx = np.arange(start, end + 1, dtype=np.int64)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]          # sub‑gradient of the active term
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)       # projection onto non‑negative orthant
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / n
    else:
        a_new /= s
    return a_new


def random_perturb(a: np.ndarray, eps: float) -> np.ndarray:
    """Move a fraction *eps* of mass from a random index to another."""
    n = len(a)
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    delta = eps * a[i]
    if delta <= 0.0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix ``A`` with entries
    A[k, j] = a[k‑j]  (zero whenever the index is out of range).
    This matrix encodes the constraints   (a * g)[k] ≤ M .
    """
    n = len(a)
    K = 2 * n - 1
    A = np.zeros((K, n), dtype=np.float64)
    # column j contributes a[·] to rows j … j+n‑1
    for j in range(n):
        A[j:j + n, j] = a
    return A


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the provided ``evaluate_sequence`` metric.  The routine respects the time
    budget and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # tiny safety margin

    # --------------------------------------------------------------
    # 1. Choose a starting length and initialise the sequence
    # --------------------------------------------------------------
    # Prefer the previously found best sequence if it exists.
    global height_sequence_1  # may be defined by the harness
    if 'height_sequence_1' in globals():
        # keep only a manageable prefix (the first 800 entries) – the tail is
        # negligible for the objective and keeps the LP dimension moderate.
        n = min(800, len(height_sequence_1))
        a = np.array(height_sequence_1[:n], dtype=np.float64)
        if a.sum() == 0.0:
            a = compute_catalan_sequence(n)
    else:
        n = 800
        a = compute_catalan_sequence(n)

    # Normalise to a probability vector (sum = 1). The metric is scale‑invariant.
    a = a / a.sum()

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}  score={best_score:.6f}")

    # FFT length for fast convolutions (next power of two ≥ 2·n)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # 2. Projected sub‑gradient descent (cheap, many iterations)
    # --------------------------------------------------------------
    step = 0.25
    it = 0
    while time.time() < deadline and step > 1e-5 and it < 5000:
        a = gradient_step(a, step, fft_len)
        if it % 100 == 0:
            cur = evaluate_sequence(a.tolist())
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[grad] it={it:4d} score={best_score:.6f}")
        step *= 0.9996
        it += 1

    # --------------------------------------------------------------
    # 3. Linear‑programming refinement (few but costly iterations)
    # --------------------------------------------------------------
    try:
        from scipy.optimize import linprog
    except Exception as exc:          # scipy not available – skip LP phase
        linprog = None
        print("[lp] scipy not available, skipping LP phase:", exc)

    if linprog is not None:
        lp_iters = 8                 # how many LP updates we attempt
        mix_t = 0.3                  # base mixing weight
        while time.time() < deadline and lp_iters > 0:
            # compute current peak of the self‑convolution
            M = max_convolution_fft(a, fft_len)

            # build the constraint matrix  (each row = a‑convolution with a canonical basis vector)
            A = build_conv_matrix(a)
            K = 2 * n - 1
            c = -np.ones(n)                     # maximise sum(g)  <=>  minimise -sum(g)
            bounds = [(0, None)] * n
            b_ub = np.full(K, M, dtype=np.float64)

            # solve the LP
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                print("[lp] linprog failed:", res.message)
                break

            g0 = res.x
            sum_g0 = g0.sum()
            if sum_g0 <= 1.0 + 1e-12:            # no slack – cannot improve further
                print("[lp] no improvement possible, breaking.")
                break

            # normalise g so that both vectors have sum = 1
            g = g0 / sum_g0

            # ------------------------------------------------------
            # 3a. Choose a mixing coefficient that most reduces the peak
            # ------------------------------------------------------
            conv_aa = np.convolve(a, a)
            conv_gg = np.convolve(g, g)
            conv_ag = np.convolve(a, g)

            best_t = 0.0
            best_Mnew = M
            # coarse grid search on [0,1] (skip 0 and 1 – they give the old vectors)
            for t in np.linspace(0.0, 1.0, 21)[1:-1]:
                mixed = (1 - t) * (1 - t) * conv_aa + t * t * conv_gg + 2 * t * (1 - t) * conv_ag
                Mnew = mixed.max()
                if Mnew < best_Mnew:
                    best_Mnew = Mnew
                    best_t = t

            # if the grid failed to improve we fall back to a small fixed step
            if best_t == 0.0:
                best_t = mix_t

            # update the current iterate
            a = (1 - best_t) * a + best_t * g
            a /= a.sum()          # re‑normalise (numerical drift)

            cur = evaluate_sequence(a.tolist())
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[lpmix] it={8 - lp_iters:2d} t={best_t:.3f} score={best_score:.6f}")

            lp_iters -= 1

    # --------------------------------------------------------------
    # 4. Stochastic local search / simulated‑annealing finish
    # --------------------------------------------------------------
    eps = 0.12
    temperature = 0.12
    it = 0
    while time.time() < deadline:
        cand = random_perturb(a, eps)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand
            print(f"[pert] it={it:5d}  IMPROVED score={best_score:.6f}")
        else:
            # occasional uphill moves
            prob = np.exp(-(cand_score - best_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                a = cand
        # decay parameters slowly
        eps = max(eps * 0.9999, 1e-4)
        temperature *= 0.9997
        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    final_seq = np.clip(best_seq, 0.0, 1000.0)   # safety clamp
    print(f"[final] best score = {best_score:.6f}  length = {len(final_seq)}")
    return final_seq.tolist()
