def propose_candidate(seed=42, budget_s=1000, **kwargs):

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = int(kwargs.get('n', 2000))

    # init from known sequence if exists and length matches
    global_vars = globals()
    init_seq = global_vars.get('height_sequence_1')
    if init_seq is not None and isinstance(init_seq, (list, tuple, np.ndarray)):
        init_arr = np.array(init_seq, dtype=float)
        if len(init_arr) == n:
            a = init_arr.copy()
        else:
            a = np.random.rand(n)
    else:
        a = np.random.rand(n)

    a = np.clip(a, 0, None)
    if a.sum() == 0:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] score={best_score:.6f}")

    iteration = 0
    last_improve_time = start

    while time.time() < deadline:
        # compute M
        conv_f = np.convolve(a, a)
        M = conv_f.max()

        # Build A matrix
        # faster: using broadcasting?
        # Use for loop
        rows = 2*n - 1
        A = np.empty((rows, n), dtype=np.float64)
        # Use zeros then fill slices
        A.fill(0.0)
        for i in range(n):
            A[i:i+n, i] = a[i]  # fill column i

        # Solve LP: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(rows, M, dtype=np.float64)

        # Use high-performance method
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs', options={'presolve': True, 'time_limit': deadline - time.time()})
        # the time_limit option is may exist; not sure.
        if not res.success:
            # can't solve; break or fallback to random move
            # fallback: random move
            a = random_move(a, 0.05)
            continue

        g = res.x

        if sum_g == 0:
            # nothing; try random move
            a = random_move(a, 0.05)
            continue

        # scale g to same sum as a

        else:

"""
LP‑guided refinement of a step‑function sequence.

We start from a known good sequence (or a random one) and repeatedly
solve a linear program that maximises the total mass under the constraint
that the convolution f∗g does not exceed the current maximal autocorrelation
M = ‖f∗f‖∞.  The LP solution g contains “extra” mass placed where it does
not increase the peak.  By mixing f with g (several mixing coefficients and
both with/without normalising the total mass) we obtain a new sequence with
a smaller ‖f∗f‖∞ and therefore a lower evaluation score.
The process is repeated until the time budget is exhausted,
occasionally performing random perturbations to escape stagnation.
"""

import time
import numpy as np
from scipy.optimize import linprog

def compute_score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (same as evaluate_sequence without checks)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    b = np.convolve(a, a)
    return float(2.0 * len(a) * b.max() / (s * s))

def random_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """Transfer a small amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # renormalise to avoid drift
    a_new /= a_new.sum()
    return a_new

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence with a low evaluation score.

    The method is LP‑guided: at each iteration we compute M = max(conv(f,f)),
    solve   max Σ g_i   s.t.   conv(f,g) ≤ M,  g_i ≥ 0,
    and then mix the current sequence f with the LP solution g.
    Random perturbations and restarts are used to avoid stagnation.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1.  Initialise the sequence
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    # try to start from the best construction we already have
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        init_arr = np.asarray(init_seq, dtype=float)
        if init_arr.shape[0] == n:
            a = init_arr.copy()
        else:
            a = np.random.rand(n)
    else:
        a = np.random.rand(n)

    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.ones(n) / n
    else:
        a /= a.sum()                       # normalise to unit sum (scale does not matter)

    best_seq = a.copy()
    best_score = compute_score(best_seq)
    print(f"[init] length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    last_improve = start
    rows = 2 * n - 1

    # pre‑allocate the constraint matrix to avoid repeated allocations
    A = np.empty((rows, n), dtype=np.float64)

    # helper for building the Toeplitz‑like matrix for conv(f,g) ≤ M
    def build_constraint_matrix(f: np.ndarray):
        A.fill(0.0)
        # column i contributes f[i] to rows i … i+n‑1
        for i in range(n):
            A[i:i + n, i] = f[i]
        return A

    # candidate mixing coefficients we will try each iteration
    mix_factors = (0.25, 0.5, 0.75)

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 2.1  Compute current peak M
        # --------------------------------------------------------------
        conv_f = np.convolve(a, a)
        M = conv_f.max()

        # --------------------------------------------------------------
        # 2.2  Solve the linear programme
        # --------------------------------------------------------------
        # Build the (dense) constraint matrix for conv(a, g) ≤ M
        build_constraint_matrix(a)

        c = -np.ones(n, dtype=np.float64)   # maximise Σg  ⇔ minimise –Σg
        b_ub = np.full(rows, M, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={"presolve": True})
        if not res.success:
            # LP failed – fall back to a cheap random perturbation
            a = random_move(a, 0.05)
            iteration += 1
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            # degenerate LP – try a random move
            a = random_move(a, 0.05)
            iteration += 1
            continue

        # --------------------------------------------------------------
        # 2.3  Generate a few mixed candidates and keep the best one
        # --------------------------------------------------------------
        # we will try both the raw LP solution (which may increase the total mass)
        # and a version that is re‑scaled to the original total mass.
        g_scaled = g * (a.sum() / sum_g)

        best_local = a
        best_local_score = best_score

        # list of (candidate, score) to evaluate
        for t in mix_factors:
            # mixing with raw g
            cand_raw = (1 - t) * a + t * g
            cand_raw_score = compute_score(cand_raw)

            if cand_raw_score < best_local_score:
                best_local_score = cand_raw_score
                best_local = cand_raw

            # mixing with mass‑preserving g_scaled
            cand_scaled = (1 - t) * a + t * g_scaled
            cand_scaled_score = compute_score(cand_scaled)

            if cand_scaled_score < best_local_score:
                best_local_score = cand_scaled_score
                best_local = cand_scaled

        # --------------------------------------------------------------
        # 2.4  Accept improvement (or occasional uphill move)
        # --------------------------------------------------------------
        if best_local_score < best_score - 1e-9:
            a = best_local
            best_seq = a.copy()
            best_score = best_local_score
            last_improve = time.time()
            print(f"[{iteration:6d}] new best = {best_score:.6f}  time={last_improve - start:.1f}s")
        else:
            # no improvement – perform a small random perturbation every few steps
            if iteration % 200 == 0:
                a = random_move(a, 0.02)

        # --------------------------------------------------------------
        # 2.5  Stagnation handling
        # --------------------------------------------------------------
        if time.time() - last_improve > 30.0:
            # restart from a fresh random distribution
            a = np.random.rand(n)
            a /= a.sum()
            print(f"[{iteration:6d}] restart (stagnation)")
            last_improve = time.time()

        iteration += 1

    # ------------------------------------------------------------------
    # 3.  Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    total_time = time.time() - start
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
    return best_seq.tolist()

LP‑guided refinement of a non‑negative step function.

The algorithm works as follows.

1.  Initialise a sequence `f` (either the known good sequence
    `height_sequence_1` or a fresh random Dirichlet‑like vector) and
    normalise it to unit sum (the objective is scale‑invariant).

2.  Let M = max(conv(f,f)).  Build the dense linear constraint matrix
    A for the convolution `conv(f,g) ≤ M` (A_ub @ g ≤ M).  Solve the

        maximise   Σ g_i
        subject to A_ub·g ≤ M·1 ,   g_i ≥ 0

    into positions that do **not** increase the peak of the autocorrelation.

3.  From `g` we generate several mixed candidates:

                best_local = cand_raw

            # (ii) mass‑preserving version
            cand_scaled = (1.0 - t) * a + t * g_scaled
            cand_scaled_score = compute_score(cand_scaled)
            if cand_scaled_score < best_local_score:
                best_local_score = cand_scaled_score
                best_local = cand_scaled

        # --------------------------------------------------------------
        # 2.4 accept improvement (or occasionally a random perturbation)
        # --------------------------------------------------------------
        if best_local_score < best_score - 1e-9:
            a = best_local
            best_seq = a.copy()
            best_score = best_local_score
            last_improve = time.time()
            print(f"[{iteration:6d}] new best = {best_score:.6f}  time={last_improve - start:.1f}s")
        else:
            # no improvement – perform a tiny random move every few steps
            if iteration % 200 == 0:
                a = random_move(a, 0.02)

        # --------------------------------------------------------------
        # 2.5 stagnation handling
        # --------------------------------------------------------------
        if time.time() - last_improve > 30.0:
            # restart from scratch
            a = np.random.rand(n)
            a /= a.sum()
            print(f"[{iteration:6d}] restart (stagnation)")
            last_improve = time.time()

        iteration += 1

    # ------------------------------------------------------------------
    # 3. return the best sequence found
    # ------------------------------------------------------------------
    total_time = time.time() - start
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
    return best_seq.tolist()
