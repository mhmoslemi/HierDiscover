def _project_to_simplex(a):
    # ensure nonnegative and sum to 1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        # assign uniform positive
        a[:] = 1.0 / a.size
    else:

def _max_convolution(a):
    n = a.size
    size = 2*n - 1
    # Use next power of two for efficient FFT
    fft_len = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    conv = np.fft.irfft(B, n=fft_len)[:size]
    return conv.max()

"""
Gradient‑descent + stochastic refinement search for a step‑function
that minimises ``evaluate_sequence``.

The evaluation is homogeneous, therefore we keep every candidate
normalised to sum = 1 (the probability simplex).  For a fixed length
``n`` the score reduces to

    score = 2 * n * max(conv(a, a))

so the optimisation problem is to minimise the maximal entry of the
auto‑convolution.  The gradient of the maximal convolution entry
(at lag ``k_max``) is simply ``2 * a[ k_max - i ]`` for the admissible
indices ``i``; a projected gradient step therefore pushes large
coefficients away from each other.

The search proceeds in three phases:

* **Warm‑start** – if a global ``height_sequence_1`` exists it is used
  as an initial candidate.
* **Phase 1 – random initialise + gradient descent** – for a set of
  lengths (up to 16384) many Dirichlet random vectors are generated.
  Each vector is refined by a few hundred inexpensive gradient steps,
  followed by occasional random perturbations (Gaussian noise and
  block reversals) to escape local minima.
* **Phase 2 – simulated‑annealing fine‑tuning** – the current best
  sequence is further improved by a simple annealing loop that tries
  small Gaussian perturbations and block reversals, accepting worse
  moves with a temperature‑controlled probability.

The algorithm respects the supplied ``budget_s`` (default 1000 s) and
returns the best list of non‑negative heights found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(x: np.ndarray) -> np.ndarray:
    """
    Project an array onto the probability simplex (non‑negative, sum = 1).
    """
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s <= 0.0:
        # fallback to uniform distribution
        x[:] = 1.0 / x.size
    else:
        x /= s
    return x


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution of ``a`` with itself using FFT.
    Returns a real array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n - 1
    # next power of two for efficient FFT
    fft_len = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    conv = np.fft.irfft(B, n=fft_len)[:size]
    # numerical noise may produce tiny negatives – clip safely
    conv = np.maximum(conv, 0.0)
    return conv


def _max_convolution(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Return (max_value, full_convolution) for the given array.
    """
    conv = _full_convolution(a)
    return float(conv.max()), conv


def _gradient_step(a: np.ndarray, k_max: int, lr: float) -> np.ndarray:
    """
    Perform a projected gradient descent step on ``a`` aimed at reducing
    the convolution entry at lag ``k_max``.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    # indices that actually participate in the k_max term
    start = max(0, k_max - (n - 1))
    end = min(n - 1, k_max)
    idx = np.arange(start, end + 1, dtype=int)
    partner = k_max - idx
    # gradient is 2 * a[partner] (see derivation in the docstring)
    g[idx] = 2.0 * a[partner]

    a_new = a - lr * g
    a_new = np.clip(a_new, 0.0, None)
    return _project_to_simplex(a_new)


def _random_dirichlet(rng: np.random.Generator, n: int, alpha: float = 0.5) -> np.ndarray:
    """Draw a random point on the simplex using a Gamma/Dirichlet distribution."""
    x = rng.gamma(shape=alpha, scale=1.0, size=n)
    return _project_to_simplex(x)

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative coefficient list that minimises
    ``evaluate_sequence`` within the given time budget.  The returned
    list is a plain Python list of floats (the evaluator will clip /
    normalise it internally).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # small safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")
    TARGET = 1.503

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction, if available
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        a0 = np.asarray(hs1, dtype=float)
        a0 = _project_to_simplex(a0)          # sum = 1
        max_b0, _ = _max_convolution(a0)
        score0 = 2 * a0.size * max_b0
        if score0 < best_score:
            best_score = score0
            best_seq = a0.tolist()
            print(f"[{time.time() - start_time:.1f}s] Warm‑start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – random Dirichlet initialisations + gradient descent
    # ------------------------------------------------------------------
    # lengths to explore – larger lengths give a lower theoretical bound
    lengths = [4096, 8192, 16384]
    for n in lengths:
        if time.time() >= deadline:
            break

        # a few restarts per length (the exact number is time‑driven)
        restarts = 6
        for r in range(restarts):
            if time.time() >= deadline:
                break

            # initialise a random point on the simplex (heavy‑tailed)
            a = _random_dirichlet(rng, n, alpha=0.5)

            # gradient descent parameters
            max_iters = 2000
            lr0 = 0.12  # initial learning rate

            for it in range(max_iters):
                if time.time() >= deadline:
                    break

                # evaluate current max convolution
                max_b, conv = _max_convolution(a)
                score = 2.0 * n * max_b

                # keep global best
                if score < best_score:
                    best_score = score
                    best_seq = a.tolist()
                    print(f"[{time.time() - start_time:.1f}s] New best (n={n}) score={best_score:.6f}")
                    if best_score <= TARGET:
                        print("Target reached – exiting early.")
                        return best_seq

                # gradient step targeting the worst lag
                k_max = int(np.argmax(conv))
                lr = lr0 / np.sqrt(it + 1.0)          # decreasing step size
                a = _gradient_step(a, k_max, lr)

                # occasional stochastic perturbations to avoid stagnation
                if (it + 1) % 50 == 0:
                    if rng.random() < 0.30:
                        # add tiny Gaussian noise
                        a += rng.standard_normal(n) * 0.01
                        a = _project_to_simplex(a)
                    if rng.random() < 0.20:
                        # reverse a short random block (changes ordering)
                        start_block = rng.integers(0, n)
                        block_len = rng.integers(1, min(300, n - start_block) + 1)
                        a[start_block:start_block + block_len] = a[
                            start_block:start_block + block_len][::-1]

    # ------------------------------------------------------------------
    # If nothing was found (unlikely) fall back to a simple uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 4096
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=float)
        best_seq = a.tolist()
        best_score = 2.0 * n_fallback * (1.0 / n_fallback)  # =2.0

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing fine‑tuning of the current best
    # ------------------------------------------------------------------
    cur = np.asarray(best_seq, dtype=float)
    n_cur = cur.size
    start_temp, end_temp = 1.0, 0.001
    cur_score = best_score

    while time.time() < deadline:
        elapsed = time.time() - start_time
        T = start_temp * (1.0 - elapsed / budget_s) + end_temp * (elapsed / budget_s)

        # ---- propose a move -------------------------------------------------
        if rng.random() < 0.65:          # Gaussian perturbation of 1% of entries
            m = max(1, int(0.01 * n_cur))
            idx = rng.choice(n_cur, size=m, replace=False)
            delta = rng.standard_normal(m) * cur.mean() * 0.05
            cand = cur.copy()
            cand[idx] += delta
            cand = np.clip(cand, 0.0, None)
        else:                             # short block reversal
            pos = rng.integers(0, n_cur)
            bl = rng.integers(1, min(400, n_cur - pos) + 1)
            cand = cur.copy()
            cand[pos:pos + bl] = cand[pos:pos + bl][::-1]

        cand = _project_to_simplex(cand)
        max_b_cand, _ = _max_convolution(cand)
        cand_score = 2.0 * n_cur * max_b_cand

        # ---- acceptance ----------------------------------------------------
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.tolist()
                print(f"[{time.time() - start_time:.1f}s] Improved (SA) score={best_score:.6f}")
                if best_score <= TARGET:
                    print("Target reached – finishing search.")
                    break
        else:
            # Metropolis criterion
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_score = cand_score

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain list of floats)
    # ------------------------------------------------------------------
    return best_seq
