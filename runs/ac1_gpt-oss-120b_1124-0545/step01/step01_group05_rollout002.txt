def proj_simplex(v):
    """Project onto probability simplex (sum=1, v>=0)."""
    # using algorithm: sort v descending, compute cumulative sum, find rho, etc.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative sequence) with a low value of

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines:
* a rich set of random parametric families (exponential, power‑law,
  two‑block, mixtures, Gaussian, Dirichlet) to generate diverse starting
  points;
* a projected sub‑gradient descent that directly reduces the current
  maximum of the self‑convolution; the step is projected onto the
  probability simplex (non‑negative, sum = 1);
* occasional random perturbations of the current best solution to
  escape local plateaus;
* a final cheap hill‑climbing phase.

All heavy work uses NumPy/FFT (O(n log n) per iteration).  The routine
stops before the given time budget and returns the best sequence it has
found (as a plain Python list).  `evaluate_sequence` is assumed to be
available in the global namespace.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x >= 0, sum(x) = 1}
    Implements the algorithm of Wang & Carreira‑Perpinán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (returns full length)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def _conv_max_and_idx(a: np.ndarray) -> tuple[float, int]:
    """Return (max value, index of the max) of a*a."""
    conv = _conv_full_fft(a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _fast_score(a: np.ndarray) -> float:
    """Fast surrogate for evaluate_sequence when sum(a)=1."""
    n = a.shape[0]
    max_conv, _ = _conv_max_and_idx(a)
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# Parametric families for random starts
# ----------------------------------------------------------------------


def _make_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def _make_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = i ** p
    a /= a.sum()
    return a


def _make_two_block(n: int, m: int, high_frac: float) -> np.ndarray:
    """First m entries receive high_frac of the total mass."""
    high = high_frac / m
    low = (1.0 - high_frac) / (n - m)
    a = np.empty(n, dtype=float)
    a[:m] = high
    a[m:] = low
    return a


def _make_mix_exponential(n: int, w: float, lam1: float, lam2: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = w * np.exp(-lam1 * i) + (1.0 - w) * np.exp(-lam2 * i)
    a /= a.sum()
    return a


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric discrete Gaussian centred in the middle of the vector."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def _make_triangular(n: int) -> np.ndarray:
    """Peak in the centre, linear rise/fall."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0, None)
    a /= a.sum()
    return a


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet([alpha] * n)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent for a fixed start point
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    n: int,
    max_iter: int,
    step0: float,
    min_step: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Perform a simple projected sub‑gradient descent on
        f(a) = max(conv(a,a))
    (the sum of a is fixed to 1 by the simplex projection).
    Returns the best vector seen during the run and its fast score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _fast_score(a)

    for t in range(max_iter):
        if time.time() >= deadline:
            break

        # convolution and location of the current maximum
        conv = _conv_full_fft(a)
        max_idx = int(np.argmax(conv))

        # gradient of the active entry (sub‑gradient)
        g = np.zeros_like(a)
        start = max(0, max_idx - n + 1)
        end = min(max_idx, n - 1)
        if start <= end:
            idx = np.arange(start, end + 1)
            g[idx] = 2.0 * a[max_idx - idx]

        # diminishing step size
        step = step0 / np.sqrt(t + 1.0)
        if step < min_step:
            step = min_step

        # projected update
        a_candidate = _proj_simplex(a - step * g)

        cand_score = _fast_score(a_candidate)

        # accept if we improve; otherwise keep the old point
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
        else:
            # still move to the candidate (helps to escape plateaus) with a small probability
            if np.random.rand() < 0.02:
                a = a_candidate

    return best_a, best_score

    Search for a non‑negative step‑function of length n that minimizes the
    evaluation function.  The routine runs for at most `budget_s` seconds
    and returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin for finalisation

    # ------------------------------------------------------------------
    # Determine the problem size and initialise the baseline
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)               # use the length of the known construction
    else:
        n = 2000                                  # fallback size

    # uniform baseline (score = 2.0)
    best_a = np.full(n, 1.0 / n, dtype=float)
    best_score = _fast_score(best_a)

    # if a known construction exists, try it first
    if "height_sequence_1" in globals():
        cand = np.array(height_sequence_1, dtype=float)
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s > 0.0:
            cand /= s
            sc = _fast_score(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                print(f"[{time.time()-start_time:6.1f}s] Starting from known seq, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper to generate a random start from a parametric family
    # ------------------------------------------------------------------
    def _random_start() -> np.ndarray:
        method = np.random.choice(
            ["exp", "power", "two_block", "gauss", "triangular", "dirichlet", "mix_exp"],
            p=[0.12, 0.12, 0.16, 0.12, 0.08, 0.28, 0.12],
        )
        if method == "exp":
            lam = np.random.uniform(0.0, 8.0 / n)
            return _make_exponential(n, lam)
        if method == "power":
            p = np.random.uniform(-6.0, 6.0)
            return _make_power_law(n, p)
        if method == "two_block":
            m = np.random.randint(int(0.05 * n), int(0.95 * n) + 1)
            high_frac = np.random.uniform(0.0, 1.0)
            return _make_two_block(n, m, high_frac)
        if method == "gauss":
            sigma = np.random.uniform(0.1 * n, 0.5 * n)
            return _make_gaussian(n, sigma)
        if method == "triangular":
            return _make_triangular(n)
        if method == "dirichlet":
            alpha = np.random.uniform(0.05, 5.0)
            return _make_dirichlet(n, alpha)
        if method == "mix_exp":
            w = np.random.uniform(0.0, 1.0)
            lam1 = np.random.uniform(0.0, 8.0 / n)
            lam2 = np.random.uniform(0.0, 8.0 / n)
            return _make_mix_exponential(n, w, lam1, lam2)
        # fallback
        return np.full(n, 1.0 / n, dtype=float)

    # ------------------------------------------------------------------
    # Main stochastic search (random families + projected sub‑gradient)
    # ------------------------------------------------------------------
    outer_iters = 0
    while time.time() < deadline:
        outer_iters += 1

        # Occasionally perturb the current best to try to leave a basin
        if np.random.rand() < 0.22:
            a_start = best_a.copy()
            # add a small Dirichlet‑style jitter
            jitter = np.random.rand(n) * np.random.uniform(0.001, 0.015)
            a_start = np.clip(a_start + jitter, 0.0, None)
            a_start /= a_start.sum()
        else:
            a_start = _random_start()

        # Quick filter: if even the fast score is not better, skip descent
        start_score = _fast_score(a_start)
        if start_score < best_score * 0.995:           # give a tiny leeway
            # run a limited sub‑gradient descent (early exit if time runs out)
            max_iter = 4000
            a_opt, sc_opt = _subgradient_descent(
                a_start,
                n=n,
                max_iter=max_iter,
                step0=0.6,
                min_step=1e-5,
                deadline=deadline,
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
                elapsed = time.time() - start_time
                print(
                    f"[{elapsed:6.1f}s] New best {best_score:.6f} "
                    f"(outer #{outer_iters}, start={start_score:.6f})"
                )
        else:
            # Still try a tiny local tweak when we are very close
            if start_score < best_score * 1.005:
                # one short descent (few iterations) to see if we can improve
                a_opt, sc_opt = _subgradient_descent(
                    a_start,
                    n=n,
                    max_iter=800,
                    step0=0.4,
                    min_step=1e-5,
                    deadline=deadline,
                )
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_a = a_opt.copy()
                    elapsed = time.time() - start_time
                    print(
                        f"[{elapsed:6.1f}s] New best (tight) {best_score:.6f} "
                        f"(outer #{outer_iters})"
                    )

        # Periodic status report
        if outer_iters % 40 == 0:
            elapsed = time.time() - start_time
            remaining = max(0.0, deadline - time.time())
            print(
                f"[{elapsed:6.1f}s] outer={outer_iters} best={best_score:.6f} "
                f"time left≈{int(remaining)}s"
            )

    # ------------------------------------------------------------------
    # Final fine‑grained hill climbing (tiny mass transfers)
    # ------------------------------------------------------------------
    a = best_a.copy()
    for _ in range(3000):
        if time.time() >= deadline:
            break
        i, j = np.random.choice(n, 2, replace=False)
        # move a tiny fraction of mass from i to j
        delta = a[i] * np.random.uniform(-0.03, 0.03)
        delta = np.clip(delta, -a[i] + 1e-12, 1e12)
        a[i] -= delta
        a[j] += delta
        a = np.clip(a, 0.0, None)
        a /= a.sum()
        sc = _fast_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:6.1f}s] Final tweak improvement: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    # The official evaluator accepts any non‑negative list; our vectors sum to 1,
    # so the returned score coincides with evaluate_sequence.
    return best_a.tolist()
