"""
Propose a high‑quality non‑negative step‑function sequence.

The algorithm proceeds in three stages:

1. **Initialisation** – take the previously found construction
   (`height_sequence_1` if available) or fall back to the classic
   1/√(i+1) root sequence.
2. **Shape optimisation** – sample many power‑law shapes
   a_i ∝ (i+α)^‑p (with p≈0.5, α≥0) and keep the best one according
   to the evaluation score.  Fast convolution via FFT is used to
   compute the score.
3. **Refinement** – a hybrid of simulated annealing (random mass
   transfers) and projected gradient descent that reduces the
   maximum entry of the self‑convolution.  Gradient steps are taken
   in the direction that most reduces the current maximal convolution
   entry, followed by a projection onto the simplex (non‑negative,
   unit sum).

The whole search runs inside the given time budget (default 1000 s)
and returns the best sequence encountered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _initial_root_sequence(n: int) -> np.ndarray:
    """Root law a_i ∝ 1/√(i+1) normalised to unit sum."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def _power_law_sequence(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Generate a normalised sequence a_i ∝ (i + α + 1)^‑p.
    Adding 1 ensures the denominator never vanishes when alpha=0.
    """
    idx = np.arange(n, dtype=float)
    a = (idx + alpha + 1.0) ** (-p)
    s = a.sum()
    if s == 0:
        return _initial_root_sequence(n)
    return a / s


def _conv_max_and_arg(seq: np.ndarray) -> tuple[float, int]:
    """
    Return (max(conv(seq,seq)), argmax) using an FFT based convolution.
    """
    n = seq.size
    size = 2 * n - 1
    # next power of two for a fast real FFT
    m = 1 << ((size - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:size]
    arg = int(np.argmax(conv))
    return float(conv[arg]), arg


def _score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective:
        2 * n * max(conv(seq,seq)) / (sum(seq) ** 2)
    The sequence is assumed to be non‑negative; values are kept small
    so the clipping performed inside `evaluate_sequence` never triggers.
    """
    n = seq.size
    max_b, _ = _conv_max_and_arg(seq)
    s = seq.sum()
    if s < 0.01:               # mirrors the safety check in evaluate_sequence
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _projected_gradient_step(seq: np.ndarray, step: float) -> np.ndarray:
    """
    One gradient step that tries to reduce the maximal entry of
    the self‑convolution.  The gradient of the active entry is
    seq[k_max-i] (up to a constant factor).  After the step we
    clip to non‑negative values and renormalise to unit sum.
    """
    n = seq.size
    _, k_max = _conv_max_and_arg(seq)

    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    indices = np.arange(lo, hi + 1, dtype=int)

    grad = np.zeros_like(seq)
    grad[indices] = seq[k_max - indices]

    # keep us on the hyperplane sum(seq)=1
    grad -= grad.mean()

    new_seq = seq - step * grad
    new_seq = np.maximum(new_seq, 0.0)
    total = new_seq.sum()
    if total > 0:
        new_seq /= total
    else:                         # fall back to the previous sequence (should never happen)
        new_seq = seq.copy()
    return new_seq

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,

    Search for a non‑negative step‑function sequence that minimises the
    provided `evaluate_sequence` (or its equivalent).  The function
    respects the given time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # --------------------------------------------------------------
    # 1) Initialise from a known good construction or the root law
    # --------------------------------------------------------------
    try:

        init = np.array(height_sequence_1, dtype=float)  # type: ignore
        init = np.maximum(init, 0.0)                     # enforce non‑negativity
        if init.size == 0 or init.sum() == 0:
            raise ValueError
        init /= init.sum()
        n = init.size
        print(f"[init] using global height_sequence_1 (n={n})")
    except Exception:
        n = 5000
        init = _initial_root_sequence(n)
        print(f"[init] fallback to root sequence (n={n})")

    best_seq = init.copy()
    best_score = _score(best_seq)
    print(f"[baseline] score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Shape optimisation – random power‑law sampling
    # --------------------------------------------------------------

    # The exponent is expected to be near 0.5, we explore a modest band.
    max_trials = 3000
    for trial in range(max_trials):
        if time.time() >= deadline:
            break
        # Sample parameters
        p = np.random.uniform(0.35, 0.70)          # exponent
        alpha = np.random.uniform(0.0, 5.0)        # shift
        candidate = _power_law_sequence(n, p, alpha)
        cand_score = _score(candidate)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = candidate.copy()
            print(f"[shape] trial={trial:4d} p={p:.4f} α={alpha:.4f} → score={cand_score:.6f}")

    seq = best_seq.copy()
    cur_score = best_score
    it = 0

    # --------------------------------------------------------------
    # 3) Refinement – simulated annealing + gradient descent
    # --------------------------------------------------------------
    # Annealing schedule (linear in elapsed time)
    T0, Tf = 1.0, 1e-4
    # Gradient step size schedule
    step = 0.25
    step_decay = 0.99997

    while time.time() < deadline:
        it += 1

        # ----- random mass transfer (majority of iterations) -----
        if np.random.rand() < 0.7:
            i, j = np.random.choice(n, 2, replace=False)
            # move a modest fraction of the mass at i towards j
            eps = seq[i] * np.random.uniform(0.0005, 0.0100)
            eps = min(eps, seq[i])
            cand = seq.copy()
            cand[i] -= eps
            cand[j] += eps
        else:
            # ----- occasional projected gradient step -----
            cand = _projected_gradient_step(seq, step)

        cand_score = _score(cand)

        # ----- acceptance rule (simulated annealing) -----
        if cand_score < cur_score:
            accept = True
        else:
            elapsed = time.time() - start_time
            fraction = elapsed / (deadline - start_time)
            T = T0 * (1.0 - fraction) + Tf * fraction   # linear cooling
            prob = math.exp(-(cand_score - cur_score) / T) if T > 0 else 0.0
            accept = np.random.rand() < prob

        if accept:
            seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[improve] iter={it:6d} score={best_score:.6f}")

        # decaying gradient step size
        if it % 500 == 0:
            step = max(step * step_decay, 1e-6)

        # early exit if we have reached the target bound
        if best_score < 1.503:
            break

    print(f"[final] iterations={it}, best_score={best_score:.6f}")
    # Return the best sequence found as a plain list of Python floats
    return best_seq.tolist()
