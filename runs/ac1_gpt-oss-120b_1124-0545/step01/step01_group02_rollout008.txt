def _project_simplex(v):
    """Project vector v onto the probability simplex (non-negative, sum=1)."""
    # Equivalent to algorithm by Wang & Carreira-Perpinan (2013)
    # Works for any shape.
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1  # zero-index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _simplex_projection(v):
    n = len(v)
    if np.sum(v) == 1 and np.all(v >= 0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n+1))[0]
    if len(rho) == 0:
        theta = 0
    else:

        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑based simplex projection optimizer.

The algorithm starts from the classic 1/√(i+1) root‑sequence (normalized
to sum = 1) and then performs a continuous‑approximation of the
∞‑norm of the self‑convolution.  At each iteration we:

1. Compute the full self‑convolution using an FFT (O(n log n)).
2. Build a soft‑max weight vector w that concentrates on the largest
   convolution entries (parameter α grows slowly during the run).
3. Compute the gradient of the smoothed objective
   f(a)=log‑sum‑exp(α·(a∗a)) by another FFT‑based convolution.
4. Apply an Adam‑style update and project the result onto the
   probability simplex (non‑negative and sum = 1).
5. Keep the best sequence seen (according to the true evaluation
   function 2 n · max(a∗a)).

The search runs until the time budget expires (≈ 1000 s) and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x≥0, Σx=1}."""
    v = v.astype(float, copy=False)
    n = v.shape[0]
    # Fast algorithm (Wang & Carreira‑Perpiñán, 2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero or negative -> uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
def _initial_root_sequence(n: int) -> np.ndarray:
    """Return a normalised 1/√(i+1) sequence of length n (sum = 1)."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function height sequence that minimises
    ``evaluate_sequence``.  The routine runs for ``budget_s`` seconds
    (leaving a small safety margin) and returns the best sequence it
    discovered.
    """
    # ------------------------------------------------------------------
    # Basics
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Length of the sequence – a larger n gives a finer approximation.
    # 20 000 works comfortably within the time budget.
    n = 20000

    # Try to reuse a previously computed construction if available.
    global height_sequence_1
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, (list, tuple)):
        seq = np.array(height_sequence_1, dtype=float)
        if seq.size >= n:
            seq = seq[:n]                     # truncate if longer
        else:
            # pad the tail with the tail of the root‑sequence to reach n
            extra = _initial_root_sequence(n - seq.size)
            seq = np.concatenate([seq, extra])
        if seq.sum() == 0:
            seq = _initial_root_sequence(n)
        else:
            seq = seq / seq.sum()
    else:
        seq = _initial_root_sequence(n)

    # Adam state
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr = 0.5                     # base learning rate

    # Miscellaneous scheduling parameters
    alpha_start = 5.0
    alpha_end = 200.0
    alpha_ramp_iters = 60000    # number of iterations over which α grows

    # Pre‑compute FFT length (powers of two for speed)
    L = int(2 ** np.ceil(np.log2(3 * n)))   # >= 3n‑2

    # ------------------------------------------------------------------
    # Track the best solution seen
    # The true score is 2·n·max(conv) because we keep Σa = 1.
    conv_full = np.fft.irfft(np.fft.rfft(seq, n=L) * np.fft.rfft(seq, n=L), n=L)[:2 * n - 1]
    best_score = 2.0 * n * conv_full.max()
    best_seq = seq.copy()
    print(f"[init] n={n}, score={best_score:.6f}")

    iteration = 0
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- 1. Self‑convolution via FFT ---------------------------------
        fft_seq = np.fft.rfft(seq, n=L)
        conv_full = np.fft.irfft(fft_seq * fft_seq, n=L)[:2 * n - 1]

        # ----- 2. Current objective value ---------------------------------
        M = conv_full.max()
        current_score = 2.0 * n * M / (seq.sum() ** 2)   # sum is ≈1
        if current_score < best_score:
            best_score = current_score
            best_seq = seq.copy()
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # ----- 3. Soft‑max weighting (smooth max) -------------------------
        # α grows slowly, turning the soft‑max into a true max later.
        if iteration <= alpha_ramp_iters:
            alpha = alpha_start + (alpha_end - alpha_start) * (iteration / alpha_ramp_iters)
        else:
            alpha = alpha_end

        # Normalised soft‑max to keep the gradient magnitude bounded.
        exp_vals = np.exp(alpha * (conv_full - M))          # all ≤ 1
        w = exp_vals / exp_vals.sum()                       # shape (2n‑1,)

        # ----- 4. Gradient of the smoothed objective ----------------------
        w_rev = w[::-1]                                      # reverse for convolution
        fft_w_rev = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(fft_seq * fft_w_rev, n=L)    # length L
        # the relevant slice (indices 2n‑2 … 3n‑3) contains ∇
        start_idx = 2 * n - 2
        grad = 2.0 * grad_full[start_idx:start_idx + n]      # shape (n,)

        # ----- 5. Adam update + projection onto simplex -------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        seq_new = seq - step
        seq = _simplex_projection(seq_new)

        # ----- 6. Occasional random “mass‑swap’’ to escape plateaus ------
        if iteration % 1000 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(seq[i], 0.001) * np.random.rand()
            seq[i] -= delta
            seq[j] += delta
            seq = _simplex_projection(seq)

    # ------------------------------------------------------------------
    # Final report
    final_score = 2.0 * n * np.convolve(best_seq, best_seq).max() / (best_seq.sum() ** 2)
    print(f"[final] iterations={iteration}, best_score={final_score:.6f}")
    # Return the best sequence as a plain Python list (as required).
    return best_seq.tolist()

The evaluation function is  

\[
\text{score}(a)=\frac{2\,n\,\max (a*a)}{(\sum_i a_i)^2}.
\]

Because the score is *scale‑invariant* we may (and will) keep the vector
normalised to \(\sum_i a_i = 1\); then  

\[
\text{score}=2 n \; \max (a*a).
\]

So the task is to find a non‑negative vector whose **self‑convolution**
is as flat as possible – i.e. minimise the ∞‑norm of the convolution.

The continuous optimum is the function \(a(x)=c/\sqrt{x}\); discretising
gives the start‑vector  

\[
a_i\propto \frac{1}{\sqrt{i+1}} .
\]

From there we perform a **smooth optimisation**:

* the true objective \(\max (a*a)\) is replaced by a smooth
  approximation – the *soft‑max*  

  \[
  f_\alpha(a)=\frac{1}{\alpha}\log\Bigl(\sum_k e^{\alpha\,(a*a)_k}\Bigr).
  \]

* The gradient of \(f_\alpha\) is a convolution of the current vector
  with a *weight* vector that concentrates on the largest convolution
  entries.  
  Using the identity  
  \(\displaystyle\frac{\partial (a*a)_k}{\partial a_i}=2\,a_{k-i}\) we get  

  \[
  \nabla f_\alpha(a)=2\; w * a\quad\text{with}\quad
  w_k=\frac{e^{\alpha\,(a*a)_k}}{\sum_j e^{\alpha\,(a*a)_j}} .
  \]

* Both the convolution \(a*a\) and the gradient convolution are computed
  with **FFT** – \(O(n\log n)\) even for \(n=20000\).

* The descent step follows the **Adam** optimiser (momentum + adaptive
  step size).  After each step the vector is projected onto the
  probability simplex (non‑negative, sum = 1).

* The smoothing parameter \(\alpha\) starts small (a gentle landscape)
  and is increased linearly up to a large value (the soft‑max becomes
  the true max).

* Every 1000 iterations a tiny random mass‑swap is performed to help
  escaping flat plateaus.

The algorithm runs until the allotted time (≈ 1000 s) expires, constantly
remembering the best vector seen (its true score is evaluated with the
provided `evaluate_sequence`).  The returned list satisfies all
