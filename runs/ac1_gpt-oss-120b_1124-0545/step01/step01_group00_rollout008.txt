def project_onto_simplex(v):
    # Project onto probability simplex: sum=1, v_i >=0
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a height sequence that (hopefully) beats the previous best
(≈ 1.524) for the objective

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three stages.

1. **Initialisation** – start from the best public construction
   (``height_sequence_1`` if it exists) or from a random Dirichlet
   point, normalised to sum = 1.

2. **Smoothed‑max gradient descent** – we minimise a smooth
   approximation of the maximal autocorrelation entry.
   For a temperature‑like parameter ``α`` we define

        Φα(a) = (1/α)·log Σk exp(α·(a*a)[k])

   The gradient of Φα is a weighted sum of the ordinary gradients of
   the individual convolution entries.  After each gradient step the
   vector is projected onto the probability simplex (non‑negative,
   sum = 1).  ``α`` is increased slowly from a small value (≈ 0.5)
   to a larger one (≈ 30) so that the optimisation first smooths the
   landscape and later focuses on the true maximum.

3. **Fine‑grained local search** – in the last few seconds we
   fall back to the targeted‑move / random‑move hill‑climbing from the
   original solution (the same moves that gave 1.524).  These moves
   exploit the exact maximal convolution index and often shave the
   last bits of the objective.

The whole routine respects the 1000 s budget, works with any length
``n`` (default 2000) and returns a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact evaluation of the objective for a non‑negative sequence."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    b = np.convolve(a, a)
    return float(2.0 * len(a) * b.max() / (s * s))


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def smoothed_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of  Φα(a) = (1/α)·log Σk exp(α·(a*a)[k])
    with respect to a (assuming a ≥ 0).  The gradient is computed via a
    convolution with the softmax weights over the autocorrelation
    entries.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                       # length = 2n‑1
    max_c = conv.max()
    # softmax over the convolution entries (stability shift by max)
    shifted = alpha * (conv - max_c)
    # avoid overflow/underflow
    exp_shifted = np.exp(shifted - np.max(shifted))
    probs = exp_shifted / exp_shifted.sum()        # Σ probs = 1

    # gradient: g_i = Σk probs_k * (∂/∂a_i (a*a)[k])
    # where ∂(a*a)[k]/∂a_i = 2·a_{k-i}  (double unless i = k‑i)
    a_rev = a[::-1]
    # convolution of reversed a with probs gives Σj a_j·probs_{i+j}
    tmp = np.convolve(a_rev, probs)                # length = 3n‑2
    grad = 2.0 * tmp[n - 1:n - 1 + n]               # 2·Σj a_j·p_{i+j}
    # subtract the diagonal over‑count (i = k‑i)
    diag_idx = 2 * np.arange(n)
    grad -= a * probs[diag_idx]
    return grad


def targeted_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Move a small amount of mass away from indices that contribute most
    to the current maximal autocorrelation entry.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k_max = int(conv.argmax())

    # contributions w_i = a_i * a_{k_max-i}
    w = np.zeros(n)
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1)
    w[idx] = a[idx] * a[k_max - idx]

    if w.sum() == 0.0:
        i = np.random.randint(n)
    else:
        probs = w / w.sum()
        i = np.random.choice(n, p=probs)

    # destination index j ≠ i
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1

    max_transfer = a[i]
    if max_transfer == 0.0:
        return a.copy()
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_onto_simplex(a_new)


def random_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another.
    """
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)

    max_transfer = a[i]
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_onto_simplex(a_new)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The routine mixes a smoothed‑max gradient
    descent (majority of the time) with a final short hill‑climbing
    phase.  The best sequence found before the time budget expires is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a safety margin so the process is not killed abruptly
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    # ---- initial point ------------------------------------------------
    # Prefer the public construction if it exists, otherwise random.
    init_seq = globals().get("height_sequence_1")
    if init_seq is not None:
        a = np.array(init_seq[:n], dtype=float).copy()
        if a.shape[0] < n:               # pad with small random noise
            pad = np.random.rand(n - a.shape[0])
            a = np.concatenate([a, pad])
    else:
        a = np.random.rand(n)

    # safety: enforce non‑negativity and normalise to sum = 1
    a = np.clip(a, 0.0, None)
    a = a / a.sum()

    best_seq = a.copy()
    best_score = compute_score(best_seq)

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ---- schedules ----------------------------------------------------
    # α controls the smoothness of the max‑approximation (small → smooth)
    alpha_start, alpha_end = 0.5, 30.0
    # learning rate (step size) for the gradient update
    lr_start, lr_end = 0.8, 0.001
    # step scale for occasional random moves / targeted moves
    move_step_start, move_step_end = 0.15, 0.002

    # ----- state for momentum (optional) --------------------------------
    momentum = np.zeros_like(a)
    mu = 0.9  # momentum coefficient

    iteration = 0
    last_improve = start

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / (budget_s - 10.0)          # in [0, 1]

        # schedules that evolve linearly with progress
        alpha = alpha_start * (1.0 - progress) + alpha_end * progress
        lr = lr_start * (1.0 - progress) + lr_end * progress
        move_step = move_step_start * (1.0 - progress) + move_step_end * progress

        # time left – switch to pure hill‑climbing in the final ~30 s
        time_left = deadline - time.time()
        if time_left < 30.0:
            # a few cheap local moves; this part runs very fast
            if np.random.rand() < 0.6:
                cand = targeted_move(a, move_step)
            else:
                cand = random_move(a, move_step)
            cand_score = compute_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:6d}] NEW BEST (local) = {best_score:.6f}  time={elapsed:.1f}s")
                a = cand
            else:
                # small chance to accept worse move (simulated annealing style)
                temp = 0.01 + 0.3 * (1.0 - progress)
                if np.random.rand() < np.exp(-(cand_score - best_score) / max(temp, 1e-12)):
                    a = cand
            iteration += 1
            continue

        # --------------------------------------------------------------
        # 1. Gradient step on smoothed max
        # --------------------------------------------------------------
        grad = smoothed_max_gradient(a, alpha)

        # momentum update (helps traversing flat regions)
        momentum = mu * momentum + (1.0 - mu) * grad
        cand = a - lr * momentum
        cand = project_onto_simplex(cand)

        cand_score = compute_score(cand)

        # --------------------------------------------------------------
        # 2. Occasional random perturbation to escape plateaus
        # --------------------------------------------------------------
        if np.random.rand() < 0.07:
            cand2 = random_move(a, move_step)
            cand2_score = compute_score(cand2)
            if cand2_score < cand_score:
                cand, cand_score = cand2, cand2_score

        # --------------------------------------------------------------
        # 3. Acceptance logic
        # --------------------------------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            last_improve = time.time()
            print(f"[{iteration:6d}] NEW BEST = {best_score:.6f}  time={elapsed:.1f}s")
            a = cand
        else:
            # Metropolis‑style acceptance to allow rare uphill moves
            temp = 0.02 + 0.5 * (1.0 - progress)   # slowly cools
            if np.random.rand() < np.exp(-(cand_score - best_score) / max(temp, 1e-12)):
                a = cand

        # ----------------------------------------------------------------
        # Restart if we have been stuck for a while (helps avoid deep local minima)
        # ----------------------------------------------------------------
        if iteration % 8000 == 0 and (time.time() - last_improve) > 35.0:
            a = np.random.rand(n)
            a /= a.sum()
            momentum[:] = 0.0
            print(f"[{iteration:6d}] restart due to stagnation")

        iteration += 1

    total_time = time.time() - start
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
    # ensure the result is a plain list of python floats
    return best_seq.tolist()
