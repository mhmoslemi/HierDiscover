Propose a step‑function sequence with a lower evaluation score than the
baseline “central‑binomial” construction.

The algorithm proceeds in three stages:

1. **Initialisation** – start from the previously discovered optimal
   sequence (available as the global variable ``height_sequence_1``) or,
   if not present, from the centred‑binomial sequence of length *n*.
   A tiny random jitter is added to break the perfect algebraic symmetry.

2. **Targeted mass redistribution** – repeatedly move a random fraction
   of the mass from a “high‑contribution” index (an index that strongly
   participates in the current maximal convolution value) to a
   low‑contribution index.  The move is evaluated with a fast FFT‑based
   scorer; only strictly improving moves are kept.  By shifting mass away
   from the first coefficient (which dominates all convolution lags) the
   maximal convolution value can be reduced without sacrificing the L1‑norm,
   thereby lowering the final score.

3. **Occasional global reshaping** – every few thousand iterations a
   larger “bulk” move is tried: a substantial part of the mass at the
   current maximal‑contributor is spread uniformly over the rest of the
   vector.  This often yields a noticeable drop in the maximal convolution.

The loop is stopped when the time budget (default 1000 s) is exhausted.
The best sequence found is finally checked with the official
``evaluate_sequence`` function and returned as a plain list of floats.

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: central‑binomial sequence (a_i = C(2i,i)/4**i)
# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """Return the truncated central‑binomial sequence of the given length."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # a[i] = a[i‑1] * (2·i‑1) / (2·i)
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def conv_fft(seq: np.ndarray) -> np.ndarray:
    """Return the full linear convolution seq * seq using FFT."""
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    return conv


def max_conv(seq: np.ndarray) -> tuple[float, int, np.ndarray]:
    """
    Compute the maximal convolution value, its index and the full convolution.
    Returns (max_value, argmax_index, conv_array).
    """
    conv = conv_fft(seq)
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


def objective(seq: np.ndarray, max_val: float = None) -> float:
    """Return the evaluation objective: 2·n·max(conv) / (sum(seq)²)."""
    n = len(seq)
    if max_val is None:
        max_val, _, _ = max_conv(seq)
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)

        # pick a source index that contributes most to the max lag
        contribution = a * weight
        source = int(np.argmax(contribution))
        if a[source] <= 1e-12:
            continue

        # pick a target among a few randomly sampled low‑contribution indices
        # (avoid the source itself)
        sample = np.random.choice(n, size=min(15, n), replace=False)
        sample = sample[sample != source]
        if sample.size == 0:
            continue
        target = int(sample[np.argmin(weight[sample])])

        # pick a fraction of the source mass to move (larger early on)
        # a linear schedule makes the moves coarser at the beginning
        frac = np.random.uniform(min_frac, max_frac)
        delta = frac * a[source]

        # perform the move
        new_a = a.copy()
        new_a[source] -= delta
        new_a[target] += delta

        # evaluate new objective (fast, uses FFT)
        new_max, _, _ = max_conv(new_a)
        new_obj = objective(new_a, new_max)

        # accept only if strictly better (we keep a simple hill‑climb)
        if new_obj < best_val:
            best_val = new_obj
            best_seq = new_a.copy()
            a = new_a
            print(f"[improve] it={it:,} obj={best_val:.6f} max={new_max:.6f}")

        # every few thousand iterations try a larger “bulk” redistribution
        if it % 5000 == 0:
            # move a substantial fraction of the current dominant entry
            src = int(np.argmax(a))               # usually the first coefficient
            if a[src] > 0.0:
                bulk = 0.5 * a[src]               # move half of its mass
                rest = np.arange(n)
                rest = rest[rest != src]
                share = bulk / rest.size
                trial = a.copy()
                trial[src] -= bulk
                trial[rest] += share
                trial_max, _, _ = max_conv(trial)
                trial_obj = objective(trial, trial_max)
                if trial_obj < best_val:
                    best_val = trial_obj
                    best_seq = trial.copy()
                    a = trial
                    print(f"[global] it={it:,} obj={best_val:.6f} max={trial_max:.6f}")

        # slowly decay the allowed move size to fine‑tune near the end
        if it % 10000 == 0:
            max_frac = max(min_frac, max_frac * 0.9)

    # ------------------------------------------------------------------
    # 3) Final validation with the official evaluator
    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    try:
        official_score = evaluate_sequence(final_list)
        print(f"[final] official evaluate_sequence = {official_score:.6f}")
    except Exception as exc:
        print("[final] evaluation failed:", exc)

    return final_list
