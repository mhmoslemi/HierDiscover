def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, scipy.sparse as sp, scipy.optimize as opt
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    # Step1: choose length n based on time and memory
    n = 1000
    # parametric families
    lengths = [n]  # only one length
    best_seq = None
    best_val = np.inf

    c_vals = np.linspace(0.5, 5.0, 21)
    p_vals = np.linspace(0.0, 2.0, 21)

    for c in c_vals:
        r = 1 - c / n
        if r <= 0:
            continue
        a = r ** np.arange(n, dtype=np.float64)
        a /= a.sum()
        val = _evaluate(a)
        if val < best_val:
            best_seq, best_val = a.copy(), val

    for p in p_vals:
        idx = np.arange(1, n+1, dtype=np.float64)
        a = 1.0 / (idx ** p)
        a /= a.sum()
        val = _evaluate(a)
        if val < best_val:
            best_seq, best_val = a.copy(), val

    for c in c_vals:
        r = 1 - c / n
        if r <=0:
            continue
        for p in p_vals:
            idx = np.arange(1, n+1, dtype=np.float64)
            a = (r ** (idx-1)) / (idx ** p)
            a /= a.sum()
            val = _evaluate(a)
            if val < best_val:
                best_seq, best_val = a.copy(), val

    # After sweep, we also evaluate symmetric geometric
    for c in c_vals:
        r = 1 - c / n
        if r <= 0:
            continue
        a = r ** np.arange(n, dtype=np.float64) + r ** np.arange(n-1, -1, -1, dtype=np.float64)
        a /= a.sum()
        val = _evaluate(a)
        if val < best_val:
            best_seq, best_val = a.copy(), val

    # print best found
    print(f"Initial best value {best_val:.6f}")

    f = best_seq
    best_f = f.copy()
    best_val = 2 * n * np.max(_conv_fft(f, f))

    iter = 0
    while time.time() < deadline and iter < 50:
        iter += 1
        # compute conv_f_f and M
        conv_f = _conv_fft(f, f)
        M = conv_f.max()
        cur_val = 2 * n * M
        if cur_val < best_val:
            best_val, best_f = cur_val, f.copy()
            print(f"[iter {iter}] new best val={cur_val:.6f}")
        # Build constraint matrix A for f
        # Using sparse diags
        data = [np.full(n, f[i]) for i in range(n)]  # each diagonal length n
        offsets = list(range(n))
        A = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')
        # Solve LP max sum(b) s.t. A @ b <= M
        c_obj = -np.ones(n)  # minimize -sum b
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        # Using HiGHS
        res = opt.linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs', options={'presolve': True})
        if not res.success:
            print(f"[iter {iter}] LP failed: {res.message}")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            # no improvement possible
            print(f"[iter {iter}] sum_b {sum_b:.6f} ≤ 1, stop")
            break
        g = b / sum_b
        # mixture step
        t = min(0.5, 1.0 / np.sqrt(iter+1))
        f = (1 - t) * f + t * g
        # avoid negative due to numerical round-off
        f = np.maximum(f, 0.0)
        f /= f.sum()  # reproject onto simplex

    # return best_f
    return best_f.tolist()

"""
Algorithm Overview
------------------

1. **Parameter sweep** –  For a moderate length `n` (default 1000) we try a few
   parametric families (geometric, power‑law and a product of both) and keep the
   best candidate w.r.t. the exact objective  
   `2*n*max(conv(a, a))` (the same objective used by `evaluate_sequence` when
   the vector is normalised to sum 1).

2. **Majorisation–Minimisation (LP step)** –  Let `f` be the current probability
   vector and let `M = max(conv(f, f))`.  We solve the linear programme

        maximise   1ᵀ b
        subject to  (f ∗ b)ₖ ≤ M   for all k = 0 … 2n‑2
                    b ≥ 0

   The matrix of the linear system is the (rectangular) convolution matrix
   built from `f`.  Because `f` is non‑negative the constraints are linear.
   The LP is solved with SciPy’s HiGHS interior‑point method.

   The solution `b` typically has a total mass `Σb > 1`.  After normalising
   `g = b / Σb` the bound becomes `f ∗ g ≤ M / Σb < M`.  Updating the
   distribution by a short convex step

        f ← (1‑τ)·f + τ·g    (τ ≈ 0.2 initially, slowly decreasing)

   provably does **not increase** `max(conv(f, f))`.  Repeating the LP step
   drives the maximal autocorrelation down.

3. **Termination** –  The loop runs until the time budget is exhausted or no
   further increase in total mass is possible.  The best vector found (lowest
   objective) is returned as a plain Python list.

The whole routine stays well within the 1000 s budget for `n≈1000`.  If SciPy
is not available we fall back to the gradient‑descent code from the previous
submission.
"""

import time
import numpy as np

# ---------------------------- helper functions ----------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than *x*."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using FFT; returns the full length vector."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective value when `a` already sums to 1."""
    return float(2.0 * len(a) * np.max(_conv_fft(a, a)))


def _build_conv_matrix(f: np.ndarray):
    """
    Construct the rectangular convolution matrix A such that
        (A @ b) = conv(f, b)
    The matrix has shape (2*n‑1, n) and is returned as a CSR sparse matrix.
    """
    import scipy.sparse as sp
    n = f.shape[0]
    # each diagonal i (offset i) contains the value f[i] repeated n times
    data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=int)            # 0,1,…,n‑1
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


# ------------------------------ main routine -----------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a non‑negative sequence of heights that minimises
    ``evaluate_sequence``.  The routine runs for roughly ``budget_s`` seconds
    and returns the best sequence found as a plain Python list.
    """
    import scipy.optimize as opt
    import scipy.sparse as sp

    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ----------- 1. small parametric sweep (length = 1000) -------------
    n = 1000
    best_seq = None
    best_val = np.inf

    # families
    c_vals = np.linspace(0.5, 5.0, 21)          # controls geometric ratio
    p_vals = np.linspace(0.0, 2.0, 21)          # power‑law exponent

    # geometric
    for c in c_vals:
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        a = r ** np.arange(n, dtype=np.float64)
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[sweep] geom  n={n} c={c:5.2f}  val={val:.6f}")

    # pure power‑law
    for p in p_vals:
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / (idx ** p)
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[sweep] power n={n} p={p:5.2f}  val={val:.6f}")

    # product geometric × power
    for c in c_vals:
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in p_vals:
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = (r ** (idx - 1)) / (idx ** p)
            a /= a.sum()
            val = _objective(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[sweep] g×p   n={n} c={c:5.2f} p={p:5.2f}  val={val:.6f}")

    # symmetric geometric (mirrored)
    for c in c_vals:
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        a = r ** np.arange(n, dtype=np.float64) + r ** np.arange(
            n - 1, -1, -1, dtype=np.float64
        )
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[sweep] sym‑g n={n} c={c:5.2f}  val={val:.6f}")

    if best_seq is None:          # fallback – should never happen
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)

    print(f"[init] best sweep value = {best_val:.6f}")

    # ---------- 2. LP based majorisation–minimisation ------------------
    f = best_seq.copy()
    best_f = f.copy()
    best_val = _objective(f)

    iteration = 0
    while time.time() < deadline and iteration < 50:
        iteration += 1

        # current maximal autocorrelation
        conv_f = _conv_fft(f, f)
        M = conv_f.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val, best_f = cur_val, f.copy()
            print(f"[iter {iteration:03d}] new best val = {cur_val:.6f}")

        # build the convolution matrix for the LP (A @ b <= M)
        A = _build_conv_matrix(f)                # sparse (2n‑1) × n

        # linear programme: maximise sum(b)  ⇔ minimise -sum(b)
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # solve with HiGHS (fast interior‑point)
        try:
            res = opt.linprog(
                c_obj,
                A_ub=A,
                b_ub=b_ub,
                bounds=(0, None),
                method="highs",
                options={"presolve": True, "lpmethod": "interior"},
            )
        except Exception as e:
            print(f"[iter {iteration}] LP failed (exception): {e}")
            break

        if not res.success:
            print(f"[iter {iteration}] LP failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()

        # if the LP cannot increase total mass – we are at a fixed point
        if sum_b <= 1.0 + 1e-12:
            print(f"[iter {iteration}] sum_b = {sum_b:.6f} ≤ 1 – stop")
            break

        # normalise to a probability vector
        g = b / sum_b

        # convex combination – start with a moderate step and decay slowly
        tau = min(0.5, 1.0 / np.sqrt(iteration + 1.0))
        f_new = (1.0 - tau) * f + tau * g
        f_new = np.maximum(f_new, 0.0)           # protect against tiny negatives
        f_new /= f_new.sum()                     # re‑project onto the simplex

        # very cheap sanity check – we never want the objective to increase
        new_val = _objective(f_new)
        if new_val > cur_val * (1.0 + 1e-8):
            # if it deteriorates, shrink the step size and retry once
            tau *= 0.5
            f_new = (1.0 - tau) * f + tau * g
            f_new = np.maximum(f_new, 0.0)
            f_new /= f_new.sum()
            new_val = _objective(f_new)

        f = f_new

    # --------------------------------------------------------------------
    print(f"[final] best value = {best_val:.6f} (length {len(best_f)})")
    return best_f.tolist()
