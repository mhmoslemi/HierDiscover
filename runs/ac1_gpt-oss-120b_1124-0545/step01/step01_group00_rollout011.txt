```
def fft_convolve(x, y):
    size = np.fft.next_fast_len(len(x) + len(y) - 1)
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    z = np.fft.irfft(X * Y, size)
    return z[:len(x)+len(y)-1]
```

```
    start = time.time()

    n = int(kwargs.get('n', 2000))

            if init.shape[0] > n:

        a = np.maximum(a, 0)

        if a_sum == 0:
```

```
def project_simplex(v):
    # algorithm from https://arxiv.org/abs/1309.1541
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
```

"""
Propose a step‑function (non‑negative sequence) that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines three ideas:

1. **Edge‑biased initialization** – a deterministic 1/(i+1)+1/(n‑i) shape,
   optionally blended with a previously found construction (``height_sequence_1``).

2. **Lp‑smooth gradient descent** – the objective max(conv) is approximated by
   the Lp‑norm of the convolution (p grows from 2 up to 32).  The exact gradient
   is cheap because  ∂Ck/∂ai = 2·a_{k‑i}, so a convolution with the weight
   w_k = Ck^{p‑1} yields the gradient in O(n log n) via FFTs.

3. **Exploration moves** – a targeted mass‑transfer move that removes mass from
   indices that contribute most to the current peak, a random transfer move,
   and a simple stochastic acceptance rule (simulated‑annealing).

The search runs until the time budget (default 1000 s) expires and returns the
best sequence encountered.  All operations keep the vector non‑negative and
renormalised to sum = 1, which leaves the evaluation function invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Real‑to‑real convolution using a power‑of‑two FFT.
    Returns the full convolution (len(x)+len(y)-1) as a 1‑D float array.
    """
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())        # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    # Numerical noise may produce tiny negatives – clip them.
    return np.maximum(conv, 0.0)


def normalise(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex {v≥0, sum(v)=1} by simple clipping+renorm."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        v = np.ones_like(v) / v.size
    else:
        v = v / s
    return v


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a vector a that already sums to 1.
    (The denominator is then 1, so the score is just 2·n·max(conv(a,a)).)
    """
    n = a.size
    c = fft_convolve(a, a)
    return float(2.0 * n * c.max())


def random_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """Transfer a tiny random amount of mass from one index to another."""
    n = a.size
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    delta = step_scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return normalise(a_new)


def targeted_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Move mass away from indices that contribute most to the current maximal
    convolution entry.  The source index is sampled proportionally to
    a[i]·a[k_max‑i]; the destination is uniform among the remaining indices.
    """
    n = a.size
    conv = fft_convolve(a, a)
    k_max = int(conv.argmax())

    # contributions w_i = a_i * a_{k_max-i} (zero if out of range)
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    w = np.zeros(n)
    w[mask] = a[idx[mask]] * a[j[mask]]

    if w.sum() == 0.0:
        src = np.random.randint(n)
    else:
        probs = w / w.sum()
        src = np.random.choice(n, p=probs)

    # pick a destination different from src
    dst = np.random.randint(n - 1)
    if dst >= src:
        dst += 1

    delta = step_scale * a[src] * np.random.rand()
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    return normalise(a_new)


def gradient_step(a: np.ndarray, p: int, step_scale: float) -> np.ndarray:
    """
    Gradient descent on the Lp‑norm of the convolution (p≥2).
    The gradient is 2·(w * a_rev) where w = conv^{p‑1}.
    """
    n = a.size
    conv = fft_convolve(a, a)                     # C = a*a
    w = conv ** (p - 1)                           # weight for Lp‑norm
    rev_a = a[::-1]
    grad_full = fft_convolve(w, rev_a)            # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]     # strip to length n

    # simple forward Euler step, then renormalise onto the simplex
    a_new = a - step_scale * grad
    return normalise(a_new)


def lp_schedule(progress: float) -> int:
    """Return the current Lp exponent based on elapsed fraction of the budget."""
    if progress < 0.20:
        return 2
    elif progress < 0.40:
        return 4
    elif progress < 0.60:
        return 8
    elif progress < 0.80:
        return 16
    else:
        return 32

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The routine mixes targeted, random and
    gradient‑descent moves, uses a simple annealing temperature,
    and returns the best sequence found before the time limit.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5                     # safety margin

    # --------------------------------------------------------------
    # Length of the step function – can be overridden via kwargs
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 5000))

    # --------------------------------------------------------------
    # Initial distribution
    # --------------------------------------------------------------
    try:
        # use the previously discovered construction if it exists
        init = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if init.size > n:
            a = init[:n]
        elif init.size < n:
            pad_before = (n - init.size) // 2
            pad_after = n - init.size - pad_before
            a = np.concatenate([np.zeros(pad_before), init, np.zeros(pad_after)])
        else:
            a = init.copy()
    except Exception:
        # deterministic edge‑heavy shape: 1/(i+1) + 1/(n-i)
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
    a = normalise(a)

    best_seq = a.copy()
    best_score = compute_score(best_seq)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Annealing / schedule parameters
    # --------------------------------------------------------------
    init_temp = 1.0
    final_temp = 0.001
    init_step = 0.20          # max relative transfer in moves
    final_step = 0.001

    iteration = 0
    last_improve = start

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / (budget_s - 10.0)          # range roughly [0,1]

        # temperature for the Metropolis rule
        temp = init_temp * (1.0 - progress) + final_temp * progress
        # step size for transfer moves
        step_scale = init_step * (1.0 - progress) + final_step * progress

        # ----------------------------------------------------------
        # Evaluate current state
        # ----------------------------------------------------------
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score

        move_type = np.random.rand()
        if move_type < 0.35:                     # targeted move

        elif move_type < 0.70:                   # random move

        # Metropolis acceptance (allow occasional uphill moves)
        # ----------------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:

            prob = np.exp(-(cand_score - cur_score) / max(temp, 1e-12))

    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
