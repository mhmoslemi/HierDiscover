"""
propose_candidate – two‑phase optimisation for the self‑convolution peak.

Phase 1 – soft‑max “exponential‑gradient’’ descent.
    * The objective  max_k (a*a)[k] is approximated by the smooth
      function f_α(a)= (1/α)·log Σ_k exp(α·b_k)  with b=conv(a,a).
    * ∂f_α/∂a = 2·correlate(a,w) where w is the soft‑max weight vector.
    * An exponentiated‑gradient update a_i ← a_i·exp(−η·g_i) keeps the
      vector non‑negative and, after normalising, stays on the simplex.
    * α is increased from 1 to ≈10⁵ (log‑spaced) while the learning rate
      η is slowly annealed from 0.5 to 0.001.  The schedule forces the
      search to first smooth the distribution and then concentrate on
      the current peak, mimicking a deterministic version of the
      “targeted_move’’ used before but without noisy acceptance steps.

Phase 2 – short simulated‑annealing / hill‑climbing refinement.
    * Three cheap moves (targeted_move, random_move, gradient_step)
      are mixed, with a Metropolis acceptance rule.
    * The best sequence found in phase 1 is used as the starting point,
      then the annealer performs a fast local search for a few seconds.

Both phases keep the globally best sequence seen so far (according to
the exact `compute_score`).  The routine respects the 1000 s budget,
prints progress, and finally returns the best list of non‑negative
heights (sum ≈ 1, which is irrelevant for the final `evaluate_sequence`
call because the objective is homogeneous).  If a previously found
construction `height_sequence_1` exists and has the same length, it is
used as an alternative initialisation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(a,a)) / (sum(a)^2)."""
    s = a.sum()
    if s < 1e-12:  # should never happen
        return np.inf
    b = np.convolve(a, a)
    max_b = b.max()
    return float(2.0 * len(a) * max_b / (s * s))


def softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """
    Return the soft‑max probability vector w_k = exp(α·b_k)/Σ.
    Numerically stable: subtract max before exponentiation.
    """
    # b is real; α·b may be large → use subtraction of max.
    z = alpha * (b - b.max())
    w = np.exp(z)
    w_sum = w.sum()
    if w_sum == 0.0:          # degenerated case – fall back to uniform
        return np.ones_like(w) / w.size
    return w / w_sum


def exp_gradient_step(a: np.ndarray, alpha: float, eta: float) -> np.ndarray:
    """
    One exponentiated‑gradient update for the soft‑max approximation.
    a is assumed non‑negative and already normalised (∑a = 1).

    Returns a new normalised candidate.
    """
    # b = a * a  (full convolution)
    b = np.convolve(a, a)

    # soft‑max probabilities over convolution entries
    w = softmax_weights(b, alpha)

    # gradient g = 2·correlate(a, w)  (valid mode → length = len(a))
    # np.correlate returns Σ_j a[j]·w[i+j] exactly the needed term.
    g = 2.0 * np.correlate(a, w, mode='valid')

    # exponentiated update (keeps positivity)
    a_new = a * np.exp(-eta * g)

    # normalise back to the simplex
    total = a_new.sum()
    if total == 0.0:
        # extreme case – fall back to uniform
        a_new = np.ones_like(a) / a.size
    else:
        a_new /= total
    return a_new


# ----------------------------------------------------------------------
# Moves used in the final refinement (identical to the previous SA code)
# ----------------------------------------------------------------------


def random_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new


def targeted_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    n = a.shape[0]
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))

    # contribution of each index to the current peak
    w = np.zeros(n)
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1)
    w[idx] = a[idx] * a[k_max - idx]

    if w.sum() == 0.0:
        i = np.random.randint(n)
    else:
        probs = w / w.sum()
        i = np.random.choice(n, p=probs)

    # destination
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1

    max_transfer = a[i]
    if max_transfer == 0.0:
        return a.copy()
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new


def gradient_step(a: np.ndarray, step: float) -> np.ndarray:
    n = a.shape[0]
    b = np.convolve(a, a)
    k = int(np.argmax(b))

    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]  # ∂b_k/∂a_i

    d = -g + g.mean()  # keep sum unchanged
    a_new = a + step * d
    a_new = np.clip(a_new, 0.0, None)
    tot = a_new.sum()
    if tot == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= tot
    return a_new
