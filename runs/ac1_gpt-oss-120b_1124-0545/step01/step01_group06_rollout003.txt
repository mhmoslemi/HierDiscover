def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv
```

"""
Non‑negative step‑function optimisation.

The returned sequence minimises the evaluation function

    2 * n * max(conv(a, a)) / (sum(a)**2)

where ``conv`` is the linear convolution.
The algorithm works as follows

1.  Generate a modest set of seed sequences for a chosen length
    (geometric, power‑law, geometric × power, a smooth sine window,
    a symmetric geometric shape and a few random vectors).  All seeds are
    normalised to sum‑to‑one, i.e. they lie on the probability simplex.

2.  For each seed an Adam optimiser works on the log‑softmax
    parametrisation  a_i = softmax(v)_i .
    The loss is a smoothed max of the autocorrelation obtained
    with the log‑sum‑exp trick.  The smoothing parameter β is increased
    geometrically (starting at 5 and multiplied by 1.02 each iteration)
    – this slowly turns the smooth loss into the true max.
    The gradient of the smoothed loss is computed by an FFT‑based
    convolution, and the Adam update keeps the iterates on the simplex
    automatically via the softmax.

3.  The best sequence (lowest true objective) seen during the whole run
    is kept.  The loop respects the supplied time budget (default 1000 s).

The procedure is repeated for two lengths (4000 and, if time permits,
8000).  The best sequence across all runs is returned as a plain Python
list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (FFT convolution, simplex utilities, parametric families)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns the full result of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


# ----------------------------------------------------------------------
# Core optimiser for a single length
# ----------------------------------------------------------------------


def _optimise_length(
    n: int,
    seeds: list[np.ndarray],
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam on each seed for the given length `n`.
    Returns the best sequence and the associated objective value.
    """
    best_seq = None
    best_val = float("inf")

    # Adam hyper‑parameters (these work well for the problem)
    lr0 = 0.6          # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    # Smoothing (log‑sum‑exp) schedule
    beta_start = 5.0
    beta_mul = 1.02   # geometric increase per iteration

    # maximum number of iterations per restart – limited only by time
    max_iters_per_restart = 8000

    # ------------------------------------------------------------------
    # Try each seed independently
    # ------------------------------------------------------------------
    for seed_idx, a0 in enumerate(seeds):
        # initialise log‑softmax parameters
        v = np.log(a0 + 1e-30)
        m = np.zeros_like(v)       # first moment
        v2 = np.zeros_like(v)      # second moment
        t = 0

        # Run the optimiser
        for it in range(max_iters_per_restart):
            if time.time() > deadline:
                return best_seq, best_val

            t += 1
            # ---- probability vector (always on the simplex) ----
            a = _softmax(v)

            # ---- evaluate true objective (fast version) ----
            conv = _conv_fft(a, a)
            cur_max = np.max(conv)
            cur_val = 2.0 * n * cur_max
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(
                    f"[opt] len={n:5d} seed={seed_idx:2d} it={it:4d} "
                    f"new best={cur_val:.6f}"
                )

            # ---- smoothed‑max weights ----
            beta = beta_start * (beta_mul ** it)
            shift = cur_max
            # the exponent range is tiny (conv values ≈ 1/n); shift protects overflow
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0.0:
                # numerical disaster – skip this iteration
                continue
            w /= w_sum

            # ---- gradient w.r.t. a (smoothed loss) ----
            # grad = 2 * (rev(a) * w)   (slice of length n)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                     # length 3n-2
            grad_a = 2.0 * full[n - 1 : n - 1 + n]         # central n entries

            # ---- convert to gradient w.r.t. v (softmax) ----
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # ---- Adam update on v ----
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            # learning‑rate schedule: slowly decay with √t
            lr = lr0 / np.sqrt(t)

            v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # small safety margin

    # ------------------------------------------------------------------
    # Lengths to try – start with 4000, then 8000 if time permits
    # ------------------------------------------------------------------
    lengths = [4000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        # --------------------------------------------------------------
        # Build a modest collection of seed sequences for this length
        # --------------------------------------------------------------
        seed_seqs = []

        # Geometric families (r = 1 - c/n)
        c_vals = np.linspace(0.5, 5.0, 12)
        for c in c_vals:
            r = 1.0 - c / n
            if r > 0.0:
                seed_seqs.append(_geom(n, r))

        # Pure power‑law families
        p_vals = np.linspace(0.0, 2.2, 12)
        for p in p_vals:
            seed_seqs.append(_power(n, p))

        # Combined geometric × power families
        for c in np.linspace(0.5, 4.0, 8):
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in np.linspace(0.0, 2.0, 8):
                seed_seqs.append(_geom_power(n, r, p))

        # Symmetric geometric
        for c in np.linspace(0.5, 4.5, 8):
            r = 1.0 - c / n
            if r > 0.0:
                seed_seqs.append(_sym_geom(n, r))

        # Smooth sine window (often a good baseline)
        seed_seqs.append(_sine_window(n))

        # A few random initial points (logits are Gaussian)
        for _ in range(5):
            v_rand = np.random.randn(n)
            a_rand = _softmax(v_rand)
            seed_seqs.append(a_rand)

        # If a previous construction is available, use it
        prev = globals().get("height_sequence_1")
        if prev is not None:
            try:
                a_prev = np.asarray(prev, dtype=np.float64)
                if a_prev.shape == (n,):
                    a_prev = a_prev / a_prev.sum()
                    seed_seqs.append(a_prev)
                else:
                    # resize by simple truncation / padding
                    a_adj = np.zeros(n, dtype=np.float64)
                    m = min(len(a_prev), n)
                    a_adj[:m] = a_prev[:m]
                    a_adj = a_adj / a_adj.sum()
                    seed_seqs.append(a_adj)
            except Exception:
                pass

        # --------------------------------------------------------------
        # Quick scan with the fast internal evaluator to pick a first best
        # --------------------------------------------------------------
        for a in seed_seqs:
            cur_val = 2.0 * n * np.max(_conv_fft(a, a))
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()

        # --------------------------------------------------------------
        # Run the Adam optimiser on every seed (time‑aware)
        # --------------------------------------------------------------
        best_seq_len, best_val_len = _optimise_length(n, seed_seqs, deadline)

        if best_seq_len is not None and best_val_len < global_best_val:
            global_best_val = best_val_len
            global_best_seq = best_seq_len.copy()
            print(f"[overall] new global best (len={n}) = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never happen)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = np.ones(n, dtype=np.float64) / n
        print("[fallback] using uniform distribution")

    # Return as a plain Python list
    return global_best_seq.tolist()
