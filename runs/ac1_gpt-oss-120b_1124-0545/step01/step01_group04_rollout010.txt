def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    end_time = time.time() + budget_s - 0.5

    # length:
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2500

    # Precompute FFT length
    L = 1 << ((3*n - 2).bit_length())

    # initial a
    a = compute_catalan_sequence(n)  # normalized to sum 1

    # add small random noise to avoid symmetry as random
    a += 0.001 * np.random.rand(n)
    a = np.clip(a, 0, None)
    a /= a.sum()

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule
    lr = 0.05
    alpha = 0.5  # temperature for softmax
    alpha_max = 500.0
    # pace of increase: alpha *= 1.001 per iteration

    # best tracking
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, init score={best_score:.6f}")

    iter = 0
    # main loop
    while time.time() < end_time:
        # compute FFT of a
        Fa = np.fft.rfft(a, n=L)
        # convolution a*a
        conv_full = np.fft.irfft(Fa * Fa, n=L)[:2*n-1]

        # compute softmax weights
        max_conv = conv_full.max()
        # temperature scaling alpha -> softmax exponent factor
        # compute scaled conv: diff = conv_full - max_conv
        # In exponent we want exp(alpha * (conv - max_conv))
        exp_vals = np.exp(alpha * (conv_full - max_conv))
        p = exp_vals / exp_vals.sum()

        # compute gradient

        # reverse a
        a_rev = a[::-1]
        Fa_rev = np.fft.rfft(a_rev, n=L)
        Fp = np.fft.rfft(p, n=L)
        conv2_full = np.fft.irfft(Fa_rev * Fp, n=L)[:(3*n - 2)]
        grad = 2.0 * conv2_full[n-1 : n-1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        bias_correction1 = 1 - beta1 ** (iter + 1)
        bias_correction2 = 1 - beta2 ** (iter + 1)
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # projection
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new /= s

        a = a_new

        # schedule updates
        lr *= 0.9995
        if alpha < alpha_max:
            alpha *= 1.001

        # occasional random noise
        if iter % 200 == 0:
            # small perturb    
            a = random_perturb(a, eps=0.02)  # use previously defined random_perturb

        # evaluate periodically
        if iter % 50 == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score - 1e-9:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[iter {iter}] improved score: {best_score:.6f}")
        iter += 1

    # final evaluation to ensure best_seq is best
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {final_score:.6f}")

    # clip and return
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    return final_seq.tolist()

Propose a non‑negative step‑function that minimizes the provided
evaluate_sequence() metric.

Algorithm
---------
1.  Initialise the sequence with the Catalan‑type coefficients
    (the coefficients of (1‑x)^‑½) which already give a low value
    (≈π/2 ≈ 1.5708).  A tiny random perturbation breaks symmetry.
2.  Perform a large number of Adam‑optimised gradient steps on a
    *soft‑max* surrogate of the objective ``max(conv(a,a))``.  For a
    given temperature α we set

        p_k ∝ exp(α·(conv_k – max(conv))),

    which concentrates the gradient on the large convolution entries
    while still keeping the update smooth.  The gradient of this
    surrogate is

        ∇_i = 2·∑_k p_k·a_{k‑i},

    i.e. a cross‑correlation of the weight vector p with the reversed
    sequence a.  All convolutions are performed with a single FFT of
    length ≥ 3·n–2, so each iteration is O(n log n) and very fast.
3.  After every few iterations the true metric is evaluated (via the
    supplied ``evaluate_sequence``) and the best sequence seen so far
    is stored.
4.  Occasionally a tiny random mass transfer between two coordinates
    (``random_perturb``) is applied to help escape shallow plateaus.
5.  The learning‑rate decays slowly, while the soft‑max temperature
    α is increased gradually – first the optimisation smooths the
    whole convolution, then it focuses on the remaining peaks.
6.  When the time budget (≈1000 s) is exhausted the best sequence is
    returned (clipped to the allowed range).

The method works entirely with NumPy and needs no external solvers,
yet it typically pushes the score well below the Catalan baseline
(≈1.5709), approaching the target < 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def compute_catalan_sequence(n: int) -> np.ndarray:
    """
    Return the first n Catalan‑type coefficients

        c_i = (2i choose i) / 4**i   (i = 0 … n‑1)

    and normalise them so that they sum to one.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence  c_i = c_{i-1} * (2i‑1) / (2i)
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def random_perturb(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Transfer a fraction *eps* of the mass from one random index to another.
    The vector stays normalised and non‑negative.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    delta = eps * a[i]
    if delta <= 0.0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolve_fft(u: np.ndarray, v: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution of real vectors u and v using FFT of length L.
    The result is the first len(u)+len(v)-1 entries.
    """
    fu = np.fft.rfft(u, n=L)
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fu * fv, n=L)
    return conv[: u.shape[0] + v.shape[0] - 1]

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Choose length of the sequence
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)      # use length of the previous best if present
    except NameError:
        n = 2500                         # reasonable default

    # --------------------------------------------------------------
    # Pre‑compute FFT length (must accommodate both a*a and a_rev*p)
    # --------------------------------------------------------------
    L = next_pow2(3 * n - 2)             # ≥ (2n‑1)+(n‑1)

    # --------------------------------------------------------------
    # Initial sequence: Catalan distribution + tiny random noise
    # --------------------------------------------------------------
    a = compute_catalan_sequence(n)
    a += 0.001 * np.random.rand(n)      # break exact symmetry
    a = np.clip(a, 0.0, None)
    a /= a.sum()                         # normalise to sum = 1

    # Adam bookkeeping
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Schedule parameters
    lr = 0.08                # initial learning‑rate
    alpha = 0.5              # soft‑max temperature (low → smooth)
    alpha_max = 3000.0       # final temperature (very sharp)
    lr_decay = 0.9995
    alpha_inc = 1.001

    # Tracking the best sequence
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    it = 0
    while time.time() < deadline:
        # ---------- convolution a*a ----------
        Fa = np.fft.rfft(a, n=L)
        conv_aa = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]

        # ---------- soft‑max over the convolution ----------
        max_conv = conv_aa.max()
        # (conv - max) ≤ 0, stable exponentiation
        exp_vals = np.exp(alpha * (conv_aa - max_conv))
        p = exp_vals / exp_vals.sum()                     # length 2n‑1, sums to 1

        # ---------- gradient of the soft‑max surrogate ----------
        a_rev = a[::-1]                                    # reversed a
        # convolution p * a_rev (full length = 3n‑2)
        # use the same FFT length L (L ≥ 3n‑2)
        Fa_rev = np.fft.rfft(a_rev, n=L)
        Fp = np.fft.rfft(p, n=L)
        conv_pa = np.fft.irfft(Fa_rev * Fp, n=L)[: 3 * n - 2]
        # gradient entry i corresponds to index (n‑1)+i of the full convolution
        grad = 2.0 * conv_pa[n - 1 : n - 1 + n]

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        bias_correction1 = 1.0 - beta1 ** (it + 1)
        bias_correction2 = 1.0 - beta2 ** (it + 1)
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)          # projection onto the non‑negative orthant
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new /= s

        a = a_new

        # ---------- schedule updates ----------
        lr *= lr_decay
        if alpha < alpha_max:
            alpha *= alpha_inc

        # occasional small random move to avoid stagnation
        if it % 200 == 0:
            a = random_perturb(a, eps=0.015)

        # evaluate periodically (cheap compared to the total budget)
        if it % 50 == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[iter {it:5d}] improved score = {best_score:.6f}")

        it += 1

    # --------------------------------------------------------------
    # Final sanity check and return
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {final_score:.6f}")

    # Safety clipping (evaluate_sequence already does this, but we keep it)
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    return final_seq.tolist()
