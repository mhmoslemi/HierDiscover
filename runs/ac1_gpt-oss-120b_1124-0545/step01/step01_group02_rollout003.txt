"""
propose_candidate
=================

Search for a low‑score non‑negative step sequence.

The program works in three stages:

1. **Coarse parametric sweep** –  sequences of the form
   `a_i ∝ (i+β)^(-α)` (α≈0.5, β>0) are generated for several
   `(α,β)` pairs (length 4000).  The best pair is remembered.

2. **Large‑scale refinement** –  the best parameters are used to build
   a longer sequence (≈8000 points).  From this start point a
   *smooth‑max* gradient descent is performed.  Instead of the hard
   ``max`` in the objective we minimise the log‑sum‑exp
   `J_τ(a)=τ·log Σ exp((a*a)[k]/τ)`.  For τ large the gradient roughly
   follows the ordinary sub‑gradient of the max; τ is slowly reduced
   so that the optimisation focuses on the true maximum.  The gradient
   is evaluated efficiently with FFTs (three forward / two inverse
   transforms per iteration).

3. **Escapes & finalisation** –  every few hundred iterations a tiny
   random mass transfer between two coordinates is performed to help
   escape shallow local minima.  The best sequence seen (according to
   the exact `evaluate_sequence` formula) is returned.

All operations keep the vector non‑negative and renormalise it to
unit L1‑norm, which leaves the evaluation score invariant.
The routine respects the global 1000 s time budget and prints progress
messages.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper functions (top‑level – no closures)
# --------------------------------------------------------------

def _linear_convolution_fft(x: np.ndarray, length: int) -> np.ndarray:
    """
    Return the linear convolution `x * x` (full length = 2*len(x)-1)
    using an FFT of size `length` (must be ≥ 2*len(x)-1).
    """
    fx = np.fft.rfft(x, n=length)
    conv = np.fft.irfft(fx * fx, n=length)
    return conv[: 2 * x.size - 1]


def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Soft‑max weights for the smooth‑max objective.
    conv : 1‑D array of the convolution values.
    tau  : temperature (τ>0).  Smaller τ gives a tighter approximation
           of the true max.
    """
    # Values are tiny (≈1e‑4), thus exp is safe without extra shifting.
    return np.exp(conv / tau)


def _gradient_via_fft(seq: np.ndarray, weights: np.ndarray, length: int) -> np.ndarray:
    """
    Compute the gradient of the smooth‑max objective:
        grad_i = Σ_k w_k * seq_{k-i}
    using FFTs.  ``weights`` has length 2*len(seq)-1.
    """
    rev = seq[::-1]                     # reversed sequence
    fw = np.fft.rfft(weights, n=length)
    fr = np.fft.rfft(rev, n=length)
    conv = np.fft.irfft(fw * fr, n=length)          # full linear conv
    # Slice the part that corresponds to grad_i (see analysis)
    n = seq.size
    start = n - 1
    return conv[start:start + n]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Simple implementation: set negatives to zero
    and renormalise – sufficient for the small steps used here.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v /= s
    else:                     # should never happen, fall back to uniform
        v[:] = 1.0 / v.size
    return v


def _score_from_conv(conv: np.ndarray, total_mass: float, n: int) -> float:
    """Exact score = 2·n·max(conv) / (total_mass**2).  total_mass≈1."""
    return 2.0 * n * conv.max() / (total_mass * total_mass)

# Main entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative numbers that (approximately) minimise

        2 * n * max(a * a) / (sum(a)^2)

    where ``a * a`` is the discrete linear convolution of the list with
    itself.  The routine runs for at most ``budget_s`` seconds and
    prints progress information.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Coarse parametric sweep (power‑law families)
    # ------------------------------------------------------------------
    n_small = 4000
    best_alpha = None
    best_beta = None
    best_score = np.inf
    best_seq = None

    # parameter grids
    alphas = np.linspace(0.44, 0.56, 25)                     # around 1/2
    betas = np.logspace(-1, 1, 30)                         # 0.1 … 10

    for α in alphas:
        for β in betas:
            # a_i ∝ (i + β)^(-α)
            idx = np.arange(n_small, dtype=float)
            seq = (idx + β) ** (-α)
            seq_sum = seq.sum()
            if seq_sum == 0:
                continue
            seq /= seq_sum                               # enforce Σa = 1

            # fast exact score using np.convolve (small n)
            conv = np.convolve(seq, seq)
            score = _score_from_conv(conv, seq.sum(), n_small)

            if score < best_score:
                best_score = score
                best_alpha = α
                best_beta = β
                best_seq = seq.copy()
                print(f"[grid] α={α:.4f}, β={β:.4f}, score={score:.6f}")

    # 2. Large‑scale refinement with smooth‑max gradient descent
    # ------------------------------------------------------------------
    # Use a larger n for a finer discretisation
    n = 8000
    # rebuild the best parametric sequence at the new length
    idx = np.arange(n, dtype=float)
    seq = (idx + best_beta) ** (-best_alpha)
    seq /= seq.sum()                     # L1‑norm = 1

    # compute conv with FFT (fast)

    conv = _linear_convolution_fft(seq, fft_len)
    best_score = _score_from_conv(conv, seq.sum(), n)
    best_seq = seq.copy()

    tau_min = 0.004
    tau_decay = 0.9998

    step_min = 1e-5
    step_decay = 0.9999

        conv = _linear_convolution_fft(seq, fft_len)

        # ---- smooth‑max gradient ------------------------------------
        weights = _softmax_weights(conv, tau)          # w_k = exp(c_k/τ)
        grad = _gradient_via_fft(seq, weights, fft_len)

        # make the gradient tangent to the hyperplane Σa = 1
        grad -= grad.mean()

        # gradient step + projection onto simplex
        seq = _project_onto_simplex(seq - step * grad)

        # ---- occasionally perform a tiny random redistribution ----
        if iteration % 150 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = 0.002 * seq[i]                     # move ≈0.2 % of mass
            delta = min(delta, seq[i])
            seq[i] -= delta
            seq[j] += delta
            seq = _project_onto_simplex(seq)

        # ---- evaluate (fast version) --------------------------------
        conv = _linear_convolution_fft(seq, fft_len)
        cur_score = _score_from_conv(conv, seq.sum(), n)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] iter={iteration}, τ={tau:.5f}, step={step:.5f}, "
                  f"score={best_score:.6f}")

        # ---- schedule updates ----------------------------------------
        tau = max(tau * tau_decay, tau_min)
        step = max(step * step_decay, step_min)

        # occasional full re‑evaluation with the provided function to guard
        # against any numerical drift (only every ~2000 iters)
        if iteration % 2000 == 0:
            # the official evaluator does its own clipping; we already stay
            # inside the allowed range, so this is just a sanity check.
            official_score = None
            try:
                official_score = evaluate_sequence(seq.tolist())
            except Exception:                     # pragma: no cover
                official_score = None
            if official_score is not None and official_score < best_score:
                best_score = official_score
                best_seq = seq.copy()
                print(f"[official‑improve] iter={iteration}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Return the best discovered sequence
    # ------------------------------------------------------------------
    total_time = time.time() - start_time
    print(f"[final] iterations={iteration}, best_score={best_score:.6f}, "
          f"time={total_time:.1f}s")
    # ensure the return type matches the expected list[float]
    return best_seq.tolist()
