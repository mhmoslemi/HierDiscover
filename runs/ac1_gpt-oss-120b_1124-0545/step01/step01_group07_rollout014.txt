Implementation of LP:

```
def _lp_max_sum(f, M):
    n = len(f)
    # Build A_ub matrix: shape (2n-1, n)
    rows = []
    for k in range(2*n - 1):
        row = np.zeros(n, dtype=float)
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(2*n-1, M, dtype=float)
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None
```

"""
Propose a step‑function sequence with a low evaluation score.

The algorithm consists of three stages:

1. **Warm‑start** – evaluate a few analytically‑defined families (constant,
   linear, quadratic, triangular, valley, sine‑squared, exponential,
   power‑law).  The best of these is used as the initial sequence.

2. **Linear‑programming boost** – for the current sequence `f` we compute
   `M = max(f*f)`.  Using a linear program we find a new non‑negative
   vector `g` that maximises the total sum `∑g` subject to the linear
   constraints `f * g ≤ M` (convolution with `f` never exceeds the current
   bound).  This step can raise the ℓ₁‑norm of the sequence without
   increasing the worst convolution entry, thus strictly decreasing the
   evaluation score.

   After solving the LP we move a fraction `t` towards `g`
   (`f ← (1‑t)·f + t·g`).  The new sequence is re‑normalised, its score
   is evaluated and, if improved, becomes the current best.  The process
   repeats while time permits.

3. **Fine‑grained stochastic refinement** – once LP improvements stagnate
   we fall back to the simulated‑annealing loop from the original approach
   to polish the solution.

The whole search respects the supplied time budget (default 1000 s) and
always returns the best sequence found before termination.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _normalise_seq(a: np.ndarray) -> np.ndarray:
    """
    Scale the array so that its maximum is 0.9 (well below the 1000 clip)
    and guarantee a total sum of at least 0.01 (otherwise evaluate_sequence
    returns ∞).  The scaling does not affect the final score because the
    evaluation is homogeneous.
    """
    a = np.asarray(a, dtype=float).copy()
    max_v = a.max()
    if max_v > 0:
        a = a * (0.9 / max_v)          # keep values < 1
    s = a.sum()
    if s < 0.01:                      # avoid the “sum too close to zero” guard
        a = a * (0.01 / s) if s > 0 else a + (0.01 / a.size)
    return a


def _candidate_shapes(n: int):
    """Yield a list of simple analytic shape vectors of length `n`."""
    cand = []

    # 1. constant
    cand.append(np.ones(n, dtype=float))

    # 2. linear (decreasing / increasing)
    cand.append(np.arange(n, 0, -1, dtype=float))
    cand.append(np.arange(1, n + 1, dtype=float))

    # 3. quadratic (decreasing / increasing)
    cand.append(np.square(np.arange(n, 0, -1, dtype=float)))
    cand.append(np.square(np.arange(1, n + 1, dtype=float)))

    # 4. triangular “mountain”
    t = np.arange(1, n + 1, dtype=float)
    cand.append(np.minimum(t, n - t + 1))

    # 5. valley (large at the ends, small in the centre)
    centre = (n - 1) / 2.0
    cand.append(1.0 / (1.0 + ((np.arange(n, dtype=float) - centre) ** 2)))

    # 6. sine‑squared shape
    cand.append(np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) / (n + 1)) ** 2)

    # 7. exponential decays
    for r in (0.99, 0.95, 0.9, 0.85, 0.8, 0.7):
        cand.append(np.power(r, np.arange(n, dtype=float)))

    # 8. power‑law decays
    for p in (0.5, 0.6, 0.7):
        i = np.arange(1, n + 1, dtype=float)
        cand.append(1.0 / (i ** p))

    return cand


def _lp_max_sum(f: np.ndarray, M: float):
    """
    Solve the linear program:
        maximise   ∑ g_j
        subject to  (f * g)[k] ≤ M  for all k
                    g_j ≥ 0
    Returns the optimal g as a NumPy array, or None on failure.
    """
    n = f.size
    # Build constraint matrix A_ub (shape (2n‑1, n))
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n, dtype=float)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs from i_start … i_end
        # corresponding g‑index j = k - i
        idx = np.arange(i_start, i_end + 1)
        row[k - idx] = f[idx]
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(2 * n - 1, M, dtype=float)

    # Objective: maximise sum(g)  <=> minimise -sum(g)
    c = -np.ones(n, dtype=float)

    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception as e:          # linprog may raise on bad inputs
        print("  [LP] Exception:", e)
        return None

    if res.success:
        return np.maximum(res.x, 0.0)   # numerical safety
    else:
        print("  [LP] No feasible solution:", res.message)
        return None

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative coefficient list that minimises
    `evaluate_sequence`.  The function respects the time budget; after the
    budget expires it returns the best sequence found so far.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1          # small safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        seq = _normalise_seq(np.asarray(hs1, dtype=float))
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[{time.time() - start_time:.1f}s] Start from height_sequence_1 "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – systematic exploration of analytic families
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]   # modest sizes – LP is O(n²)
    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _candidate_shapes(n):
            if time.time() >= deadline:
                break
            seq = _normalise_seq(cand)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"[{time.time() - start_time:.1f}s] "
                      f"Phase‑1 new best ({n} terms) score={best_score:.6f}")
            if best_score <= 1.503:   # early exit if target already reached
                break

    # Fallback in the unlikely case that no candidate was found yet
    if best_seq is None:
        n = 512
        best_seq = rng.random(n).tolist()
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – linear‑programming boost
    # ------------------------------------------------------------------
    cur = np.asarray(best_seq, dtype=float)
    t = 0.4                         # initial mixing fraction
    while time.time() < deadline:
        # recompute current worst convolution entry
        conv_ff = np.convolve(cur, cur)
        M = float(conv_ff.max())
        # solve the LP to maximise ℓ₁‑norm under the current bound M
        g = _lp_max_sum(cur, M)
        if g is None:
            break  # LP failed – give up on this phase

        # mix towards the LP solution
        new = (1.0 - t) * cur + t * g
        new_seq = _normalise_seq(new)
        new_score = evaluate_sequence(new_seq.tolist())

        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.tolist()
            cur = new
            print(f"[{time.time() - start_time:.1f}s] LP improved "
                  f"score={best_score:.6f} (t={t:.3f})")
            # after a successful step we can afford a larger step size
            t = min(0.9, t * 1.2)
            if best_score <= 1.503:
                print("Target reached – stopping search.")
                break
        else:
            # no improvement – shrink step size
            t *= 0.5
            if t < 1e-3:
                # step too small, stop LP phase
                break

    # ------------------------------------------------------------------
    # Phase 3 – stochastic fine‑tuning (fallback if needed)
    # ------------------------------------------------------------------
    cur = np.asarray(best_seq, dtype=float)
    start_temp, end_temp = 1.0, 0.001

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = start_temp * (1 - elapsed / total) + end_temp * (elapsed / total)

        # ----- mutation -------------------------------------------------
        if rng.random() < 0.70:
            # Gaussian perturbation of a random 2 % of the entries
            m = max(1, int(0.02 * len(cur)))
            idx = rng.choice(len(cur), size=m, replace=False)
            delta = rng.standard_normal(m) * cur.mean() * 0.05
            new = cur.copy()
            new[idx] += delta
            new = np.clip(new, 0.0, 999.0)
        else:
            # Reverse a short random block
            start_idx = rng.integers(0, len(cur))
            block_len = rng.integers(1, min(200, len(cur) - start_idx) + 1)
            new = cur.copy()
            new[start_idx:start_idx + block_len] = new[start_idx:start_idx + block_len][::-1]

        if new.sum() < 0.01:
            new += (0.01 - new.sum()) / len(new)

        cand_score = evaluate_sequence(new.tolist())

        # ----- acceptance ------------------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best_seq = new.tolist()
            cur = new
            print(f"[{time.time() - start_time:.1f}s] SA improved score={best_score:.6f}")
            if best_score <= 1.503:
                print("Target reached – stopping search.")
                break
        else:
            prob = np.exp(-(cand_score - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur = new

    # ------------------------------------------------------------------
    # Return the best sequence discovered
    # ------------------------------------------------------------------
    return best_seq
