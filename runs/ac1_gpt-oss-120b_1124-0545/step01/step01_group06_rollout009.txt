def refine_adam(a_init, time_deadline):
    n = len(a_init)
    y = np.log(a_init + 1e-30)   # log of a
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_val = float('inf')
    best_a = a_init.copy()
    beta = 1.0
    beta_mul = 1.03
    max_beta = 500.0
    while time.time() < time_deadline:
        t += 1
        a = np.exp(y)
        a = a / a.sum()
        # compute gradient of smoothed max
        conv = _conv_fft(a, a)
        cur_max = conv.max()
        # evaluate actual objective for logging
        val_obj = 2.0 * n * cur_max  # because sum=1
        if val_obj < best_val:
            best_val, best_a = val_obj, a.copy()
            print(f"[adam] it {t} new best {best_val:.6f}")
        # smoothed weights:
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0:
            # fallback to uniform weights
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # gradient wrt a
        grad_full = _conv_fft(a[::-1], w)
        grad_a = 2.0 * grad_full[n-1:n-1+n]
        # gradient wrt y (softmax parameters)
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # learning rate schedule: alpha_t = alpha0 / sqrt(t)
        alpha0 = 0.05  # tuned
        step = alpha0 / np.sqrt(t)
        y = y - step * m_hat / (np.sqrt(v_hat) + eps)
        # increase beta gradually
        beta = min(beta * beta_mul, max_beta)
    return best_a, best_val
```

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    '''
    Search for step function heights minimizing evaluate_sequence using
    parametric sweep followed by Adam optimization on smoothed max.
    '''
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    best_seq = None
    best_val = float('inf')
    candidates = []   # store (val, a)

    lengths = [2000, 4000, 8000]  # adapt if time
    # families parameters
    r_vals = np.linspace(0.1, 0.99, 15)
    p_vals = np.linspace(0.1, 5.0, 20)
    gamma_vals = np.linspace(-2.0, -0.1, 20)  # negative
    w_fracs = np.linspace(0.01, 0.3, 8)   # fraction of n for edge region
    ratio_vals = np.linspace(1.5, 10.0, 8)

    for n in lengths:
        # uniform baseline
        a = np.ones(n) / n
        val = _evaluate(a)
        candidates.append((val, a.copy()))
        # symmetric geometric
        for r in r_vals:
            a = _sym_geom(n, r)
            val = _evaluate(a)
            candidates.append((val, a.copy()))
        # symmetric power
        for p in p_vals:
            a = _sym_power(n, p)
            val = _evaluate(a)
            candidates.append((val, a.copy()))
        # beta shape (edges heavy)
        for gamma in gamma_vals:
            a = _beta_shape(n, gamma)
            val = _evaluate(a)
            candidates.append((val, a.copy()))
        # edge constant shape
        for w_frac in w_fracs:
            for ratio in ratio_vals:
                a = _edge_const(n, w_frac, ratio)
                val = _evaluate(a)
                candidates.append((val, a.copy()))
        # random Dirichlet draws
        for _ in range(15):
            a = np.random.dirichlet(np.ones(n))
            # ensure sum=1
            val = _evaluate(a)
            candidates.append((val, a.copy()))
        # check time
        if time.time() >= deadline:
            break

    # keep top K candidates
    candidates.sort(key=lambda x: x[0])
    top_k = 10
    top_candidates = candidates[:top_k]

    # Update best_seq based on these
    best_val, best_seq = top_candidates[0]
    best_seq = best_seq.copy()
    print(f"[init] best initial val={best_val:.6f}, length={len(best_seq)}")

    # Refinement using Adam for each top candidate until time runs out
    remaining = deadline - time.time()
    # allocate time per candidate
    per_candidate_time = remaining / len(top_candidates) if len(top_candidates)>0 else 0
    for idx, (init_val, a0) in enumerate(top_candidates):
        # adjust time limit for each candidate
        cur_deadline = time.time() + per_candidate_time
        if time.time() >= deadline:
            break
        print(f"[refine] start candidate {idx+1}/{len(top_candidates)} init_val={init_val:.6f}")
        a_refined, val_refined = _adam_optimize(a0, cur_deadline)
        if val_refined < best_val:
            best_val, best_seq = val_refined, a_refined.copy()
            print(f"[refine] new global best val={best_val:.6f}")

    # Final safety: evaluate using provided evaluate_sequence to ensure correct
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[final] final evaluated value={final_val:.6f}")
    except Exception as e:
        print("[final] error evaluating final sequence:", e)
        final_val = None

    # Return best sequence as list of floats
    return best_seq.tolist()

def _adam_optimize(a_init, deadline):

    # start from log probabilities
    y = np.log(a_init + 1e-30)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha0 = 0.02
    beta = 1.0
    beta_mul = 1.03
    max_beta = 400.0
    best_val = float('inf')
    best_a = a_init.copy()
    t = 0
    while time.time() < deadline:
        t += 1
        # compute a
        a = np.exp(y)
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum):
            a = np.ones_like(y) / len(y)
        else:
            a /= a_sum
        # compute conv and objective
        conv = _conv_fft(a, a)
        cur_max = conv.max()
        val = 2.0 * n * cur_max
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # compute smoothed gradient
        shift = cur_max
        # choose beta based on schedule
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # gradient wrt a
        grad_full = _conv_fft(a[::-1], w)
        grad_a = 2.0 * grad_full[n-1:n-1+n]
        # gradient wrt y
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha0 / np.sqrt(t)
        y = y - step * m_hat / (np.sqrt(v_hat) + eps)
        # increase beta
        beta = min(beta * beta_mul, max_beta)
    return best_a, best_val

def propose_candidate(seed=....):
    np.random.seed(seed)
    deadline = time.time() + budget_s -5
    lengths = [2000,4000,8000]
    best_val = inf
    best_seq = None

    for n in lengths:
        # Sweep families
        cand_vals = []
        # uniform
        a = np.ones(n)/n
        cand_vals.append((a, _evaluate(a)))
        # symmetric geometric
        for r in np.geomspace(0.03, 0.99, num=30):
            a = _sym_geom(n, r)
            cand_vals.append((a, _evaluate(a)))
        # symmetric power
        for p in np.linspace(0.1,10, num=30):
            a = _sym_power(n, p)
            cand_vals.append((a, _evaluate(a)))
        # beta shape
        for gamma in np.linspace(-5, -0.1, num=30):
            a = _beta_shape(n, gamma)
            cand_vals.append((a, _evaluate(a)))
        # random Dirichlet draws
        for _ in range(20):
            a = np.random.dirichlet(np.ones(n))
            cand_vals.append((a, _evaluate(a)))
        # keep top few
        cand_vals.sort(key=lambda x: x[1])
        top = cand_vals[:3]  # few
        # refine each
        for a0,val0 in top:
            # allocate time for refinement based on remaining
            if time.time() >= deadline: break
            cur_deadline = time.time() + min( (deadline - time.time())/len(top), 300)
            a_ref,best_val_ref = _adam_optimize(a0, cur_deadline)
            if best_val_ref < best_val:
                best_val = best_val_ref
                best_seq = a_ref
    # final eval using evaluate_sequence
    final_val = evaluate_sequence(best_seq.tolist())
    print(f"[final] best value after evaluate_sequence: {final_val}")
    return best_seq.tolist()

"""
Propose a step‑function (non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in two stages:

1. **Broad parametric sweep** – many simple families (symmetric geometric,
   power‑law, beta‑shape, edge‑constant) are evaluated for a few lengths
   (2000, 4000, 8000).  The best few candidates are kept.
2. **Refinement** – each kept candidate is optimised with Adam on the
   soft‑max (log‑sum‑exp) smooth approximation of the maximum.  The
   variables are the logits of a soft‑max, guaranteeing non‑negativity
   and sum‑to‑one.  The smoothing parameter β is increased gradually so
   that the optimisation first smooths the objective and then focuses
   on the true maximum.  The best sequence found during the whole run
   (including the final call to ``evaluate_sequence``) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution via FFT (real‑valued signals).
    Returns the full vector of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _evaluate(a: np.ndarray) -> float:
    """
    Fast evaluation for a sequence that already sums to 1.
    Returns 2 * n * max(conv(a, a)), i.e. the objective without the
    denominator (since sum(a)=1).
    """
    conv = _conv_fft(a, a)
    return float(2.0 * len(a) * np.max(conv))


# ----------------------------------------------------------------------
# Parametric families (all return a probability vector, sum=1)
# ----------------------------------------------------------------------
def _sym_geom(n: int, r: float) -> np.ndarray:
    """a_i = r^i + r^{n-1-i}, normalised."""
    base = np.power(r, np.arange(n, dtype=np.float64))
    a = base + base[::-1]
    a /= a.sum()
    return a


def _sym_power(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^{-p} + (n-i)^{-p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (idx ** (-p)) + ((n - idx + 1) ** (-p))
    a /= a.sum()
    return a


def _beta_shape(n: int, gamma: float) -> np.ndarray:
    """a_i = (i+1)^{γ} * (n-i)^{γ}, normalised.  γ<0 gives edge‑heavy shape."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (idx ** gamma) * ((n - idx + 1) ** gamma)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= a_sum
    return a


def _edge_const(n: int, w_frac: float, ratio: float) -> np.ndarray:
    """
    Two‑level constant shape.
    The first ``w`` entries on each side have height ``ratio * a_mid``,
    the middle part has height ``a_mid``.  The whole vector is normalised.
    """
    w = max(1, int(round(w_frac * n)))
    if 2 * w >= n:                 # safety: at least one middle element
        w = n // 2 - 1
    middle_len = n - 2 * w
    a_mid = 1.0 / (middle_len + 2.0 * w / ratio)
    a_edge = ratio * a_mid
    a = np.full(n, a_mid, dtype=np.float64)
    a[:w] = a_edge
    a[-w:] = a_edge
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed‑max objective
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimise a starting distribution ``a0`` (sum=1, non‑negative) using
    Adam on the log‑sum‑exp approximation of the maximum convolution entry.
    Returns the best distribution seen and its (exact) objective value.
    """
    n = a0.shape[0]
    eps = 1e-12

    # initialise logits (softmax parameters) safely
    y = np.log(np.clip(a0, eps, None))
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    beta1, beta2 = 0.9, 0.999
    alpha0 = 0.05          # initial learning rate for logits
    beta = 0.5             # start smoothing parameter
    beta_mul = 1.07
    max_beta = 400.0

    best_val = float("inf")
    best_a = a0.copy()
    it = 0

    while time.time() < deadline:
        it += 1
        # soft‑max to obtain a valid probability vector
        y_shift = y - np.max(y)          # for numerical stability
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()

        # current convolution and exact objective (sum(a)=1)
        conv = _conv_fft(a, a)
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[adam] it {it:5d} best={best_val:.6f}")

        # smoothed‑max (log‑sum‑exp) weights
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective w.r.t. a
        grad_full = _conv_fft(a[::-1], w)
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

        # transform gradient to logits space (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha0 / np.sqrt(it)
        y -= step * m_hat / (np.sqrt(v_hat) + 1e-8)

        # make the smoothing sharper
        beta = min(beta * beta_mul, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The routine respects the time
    budget, performs a broad parametric sweep, refines the best few
    candidates with Adam, and finally returns the best sequence found.
    """
    np.random.seed(seed)
    safety = 5.0                      # keep a small margin before timeout
    deadline = time.time() + budget_s - safety

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 1. PARAMETRIC SWEEP
    # --------------------------------------------------------------
    lengths = [2000, 4000, 8000]       # keep the sweep cheap but covering sizes
    # grids for the families
    r_vals = np.geomspace(0.01, 0.99, num=30)          # geometric ratio
    p_vals = np.linspace(0.1, 10.0, num=30)           # power‑law exponent
    gamma_vals = np.linspace(-5.0, -0.1, num=30)      # beta‑shape exponent
    w_fracs = np.linspace(0.01, 0.20, num=6)          # edge width as fraction of n
    ratio_vals = np.linspace(2.0, 15.0, num=6)        # edge‑to‑middle ratio

    # store (value, sequence) tuples
    all_candidates = []

    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"[sweep] length = {n}")

        # uniform baseline
        a = np.ones(n, dtype=np.float64) / n
        val = _evaluate(a)
        all_candidates.append((val, a.copy()))
        print(f"  uniform val={val:.6f}")

        # symmetric geometric
        for r in r_vals:
            a = _sym_geom(n, r)
            val = _evaluate(a)
            all_candidates.append((val, a.copy()))

        # symmetric power‑law
        for p in p_vals:
            a = _sym_power(n, p)
            val = _evaluate(a)
            all_candidates.append((val, a.copy()))

        # beta‑shape (edge‑heavy)
        for g in gamma_vals:
            a = _beta_shape(n, g)
            val = _evaluate(a)
            all_candidates.append((val, a.copy()))

        # two‑level constant edge shape
        for wf in w_fracs:
            for rat in ratio_vals:
                a = _edge_const(n, wf, rat)
                val = _evaluate(a)
                all_candidates.append((val, a.copy()))

        # random Dirichlet draws (provide diverse starting points)
        for _ in range(20):
            a = np.random.dirichlet(np.ones(n, dtype=np.float64))
            val = _evaluate(a)
            all_candidates.append((val, a.copy()))

        # keep the sweep fast
        if time.time() >= deadline:
            break

    # keep only the top few (to spend time on refinement)
    all_candidates.sort(key=lambda x: x[0])
    top_k = min(10, len(all_candidates))
    top_candidates = all_candidates[:top_k]

    if top_candidates:
        best_val, best_seq = top_candidates[0]
        best_seq = best_seq.copy()
        print(f"[init] best initial value = {best_val:.6f} (length {len(best_seq)})")

    # --------------------------------------------------------------
    # 2. REFINEMENT WITH ADAM
    # --------------------------------------------------------------
    # Distribute remaining time among the top candidates
    remaining_time = deadline - time.time()
    if remaining_time > 0 and top_candidates:
        per_candidate = remaining_time / len(top_candidates)
        for idx, (init_val, a0) in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            cur_deadline = time.time() + min(per_candidate, deadline - time.time())
            print(f"[refine] candidate {idx + 1}/{len(top_candidates)} start (init val={init_val:.6f})")
            a_ref, val_ref = _adam_optimize(a0, cur_deadline)
            if val_ref < best_val:
                best_val, best_seq = val_ref, a_ref.copy()
                print(f"[global] new best after refinement = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. FINAL CHECK (use the provided evaluate_sequence)
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returns {final_val:.6f}")
    except Exception as exc:
        # Should never happen, but keep a safe fallback
        print("[final] error during final evaluation:", exc)
        final_val = None

    # Return as plain Python list (the evaluator expects this)
    return best_seq.tolist()
