"""
Hybrid search for a low‑autocorrelation step function.

The algorithm consists of three stages:

1. **Parametric sweep** – evaluates a few families of simple monotone
   sequences (geometric, power‑law, combined, and pure exponential) for a
   handful of lengths (up to ~8000).  The best candidate from this cheap
   sweep is taken as a starting point.

2. **Momentum exponentiated‑gradient refinement** – the objective
   ``2·n·max(conv(a,a))`` (with ``sum(a)=1``) is approximated by a
   smooth log‑sum‑exp of the convolution.  The gradient of this smooth
   loss is obtained by a single FFT‑based convolution.  An
   exponentiated‑gradient update with Nesterov‑style momentum is applied,
   followed by a Euclidean projection onto the probability simplex.
   The smoothing parameter β is increased geometrically during the run,
   which drives the iterates toward the true max.

3. **Stochastic local search** – after the gradient phase a lightweight
   random‑swap procedure tries to push the value a little further.
   Small amounts of mass are moved between two random positions and the
   objective is recomputed; accepted moves strictly improve the best
   known value.

The whole search respects the supplied time budget (default 1000 s).  All
numbers are kept non‑negative and the sequence is always renormalised to
unit sum, therefore the returned list can be fed directly to
``evaluate_sequence``.  The code runs on a single thread but can be
executed on up to two CPUs because the heavy work (FFT) is vectorised.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using the FFT.
    Returns a vector of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x >= 0, sum(x) = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    mask = u - cssv / ind > 0
    if np.any(mask):
        rho = np.where(mask)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i = r**i, normalised to sum 1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i = 1/(i+1)^p, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s == 0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Combined geometric × power‑law."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    if s == 0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Pure exponential a_i = exp(-lam*i), normalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    if s == 0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Fast evaluation (sum(a) = 1) – returns 2·n·max(conv)
# ----------------------------------------------------------------------
def _fast_value(a: np.ndarray) -> float:
    """Object value for a sequence already normalised to sum 1."""
    conv = _conv_fft(a, a)
    return float(2.0 * len(a) * np.max(conv))


# ----------------------------------------------------------------------
# Parametric sweep – cheap but broad exploration
# ----------------------------------------------------------------------
def _parametric_sweep(deadline: float):
    """Search a few classic families; return (best_seq, best_val)."""
    best_seq = None
    best_val = math.inf

    # a few lengths – larger lengths give more room for improvement
    lengths = [2000, 4000, 6000, 8000]
    # geometric / exponential decay parameters
    r_vals = np.linspace(0.90, 0.9995, 25)          # for geometric
    lam_vals = np.linspace(0.0005, 0.005, 25)      # for pure exponential
    # power‑law exponents
    p_vals = np.linspace(0.0, 2.5, 26)
    # combined parameters
    c_vals = np.linspace(0.5, 5.0, 20)             # used to set r = 1 - c/n

    for n in lengths:
        if time.time() > deadline:
            break

        # ---- pure geometric ------------------------------------------------
        for r in r_vals:
            a = _geom(n, r)
            val = _fast_value(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[sweep] geom n={n:5d} r={r:.5f} val={val:.6f}")

        # ---- pure exponential ---------------------------------------------
        for lam in lam_vals:
            a = _exp_decay(n, lam)
            val = _fast_value(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[sweep] exp n={n:5d} lam={lam:.5f} val={val:.6f}")

        # ---- power‑law -----------------------------------------------------
        for p in p_vals:
            a = _power(n, p)
            val = _fast_value(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[sweep] pow n={n:5d} p={p:5.2f} val={val:.6f}")

        # ---- combined geometric × power‑law --------------------------------
        for c in c_vals:
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in p_vals:
                a = _geom_power(n, r, p)
                val = _fast_value(a)
                if val < best_val:
                    best_val, best_seq = val, a.copy()
                    print(f"[sweep] g×p n={n:5d} c={c:5.2f} p={p:5.2f} val={val:.6f}")

        # early manual injection: use the previously found sequence if
        # available (global variable `height_sequence_1`)
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.ndim == 1 and prev.size > 1:
                prev = prev / prev.sum()
                val = _fast_value(prev)
                if val < best_val:
                    best_val, best_seq = val, prev.copy()
                    print(f"[sweep] injected previous best val={val:.6f}")
        except NameError:
            pass

    # fallback – uniform distribution
    if best_seq is None:
        n = 4000
        best_seq = np.ones(n, dtype=np.float64) / n
        best_val = _fast_value(best_seq)

    print(f"[sweep] best after sweep: val={best_val:.6f} (len={len(best_seq)})")
    return best_seq, best_val


# ----------------------------------------------------------------------
# Momentum exponentiated‑gradient refinement (smooth‑max)
# ----------------------------------------------------------------------
def _refine(a_init: np.ndarray, deadline: float):
    """
    Refine a single candidate with an exponentiated‑gradient scheme.
    Returns (best_seq, best_val) seen during the run.
    """
    a = a_init.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _fast_value(a)

    # hyper‑parameters (tuned empirically)
    beta0 = 5.0          # initial smoothing strength
    beta_mul = 1.02      # geometric increase per iteration
    eta0 = 0.5           # base step size
    mu = 0.9             # momentum coefficient
    m = np.zeros_like(a)  # momentum accumulator

    it = 0
    max_iters = 12000    # generous upper bound; loop will stop earlier

    while time.time() < deadline and it < max_iters:
        # ---------- convolution ----------
        conv = _conv_fft(a, a)                      # length 2n‑1
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()
            print(f"[refine] iter {it:5d} new best {cur_val:.6f}")

        # ---------- smooth‑max ----------
        beta = beta0 * (beta_mul ** it)
        if beta > 2000.0:
            beta = 2000.0
        # stable exponentiation (subtract the max)
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            it += 1
            continue
        w /= w_sum

        # ---------- gradient ----------
        rev = a[::-1]
        grad_full = _conv_fft(rev, w)               # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]     # slice of length n

        # ---------- momentum ----------
        m = mu * m + (1.0 - mu) * grad

        # ---------- update ----------
        eta = eta0 / math.sqrt(it + 1.0)
        a = a * np.exp(-eta * m)                    # multiplicative EG step
        a = _project_simplex(a)                     # stay on the simplex

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stochastic local improvement – tiny random swaps
# ----------------------------------------------------------------------
def _local_search(a_start: np.ndarray, deadline: float):
    """
    Perform cheap random moves (mass transfer between two indices) while
    improving the objective.  Returns the best sequence seen.
    """
    a = a_start.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _fast_value(a)

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move a random fraction of the mass at i to j
        frac = np.random.rand() * 0.05           # at most 5 % of a[i]
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # simplex projection is not needed – sum unchanged, non‑negative
        val_new = _fast_value(a_new)
        if val_new < best_val:
            best_val, best_seq = val_new, a_new.copy()
            a = a_new
            print(f"[local] improved to {val_new:.6f}")
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimizes
    ``evaluate_sequence``.  The routine runs for roughly ``budget_s``
    seconds and returns the best sequence found (as a plain list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0    # small safety margin

    # ------------------------------------------------------------------
    # 1) cheap parametric sweep
    # ------------------------------------------------------------------
    best_seq, best_val = _parametric_sweep(deadline)

    # ------------------------------------------------------------------
    # 2) gradient refinement – several random restarts
    # ------------------------------------------------------------------
    n_best = len(best_seq)
    restarts = 4
    for r in range(restarts):
        if time.time() >= deadline:
            break
        if r == 0:
            init = best_seq.copy()
        else:
            # random Dirichlet start (same length)
            init = np.random.dirichlet(alpha=np.ones(n_best))
        seq_refined, val_refined = _refine(init, deadline)
        if val_refined < best_val:
            best_val, best_seq = val_refined, seq_refined.copy()
            print(f"[restart {r}] new best after refinement {val_refined:.6f}")

    # ------------------------------------------------------------------
    # 3) very light stochastic local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_local, val_local = _local_search(best_seq, deadline)
        if val_local < best_val:
            best_val, best_seq = val_local, seq_local.copy()
            print(f"[local] final best {val_local:.6f}")

    print(f"=== finished === best value ≈ {best_val:.6f} length {len(best_seq)}")
    # Convert to plain Python list for the caller
    return best_seq.tolist()
