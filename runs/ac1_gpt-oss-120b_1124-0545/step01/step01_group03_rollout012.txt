def propose_candidate(seed=42,budget_s=1000, **kwargs):
    import numpy as np, time, math
    from scipy.optimize import linprog
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    # base binomial sequence
    n = 4000
    f = binomial_sequence(n) # function defined earlier

    conv_f_f = np.convolve(f, f)
    # conv_f_f length 2n-1

    # compute slack per k
    slack = np.ones_like(conv_f_f)
    slack[:len(conv_f_f)] = 1 - conv_f_f
    # For k < n slack is zero (or near zero). Ensure non-negative.
    slack = np.maximum(slack, 0.0)

    best_seq = f.copy()
    best_val = fast_score(best_seq)
    print(f"[init] n={n}, fast_score={best_val:.6f}, sum_f={f.sum():.6f}")

    # function to compute final objective after scaling tail
    def evaluate_tail(v):
        # v: np.ndarray of length L
        L = len(v)
        # compute conv_f_v
        conv_f_v = np.convolve(f, v)  # length n+L-1
        conv_v_v = np.convolve(v, v)  # length 2L-1

        # binary search for alpha in (0,1] s.t. max_total <= 1
        # Precompute sums maximum threshold for each index
        # total = conv_f_f (len 2n-1) padded zeros to size total_len = 2*(n+L)-1
        size_total = 2*(n+L)-1
        # zero pad conv_f_f to size_total
        total_base = np.zeros(size_total, dtype=np.float64)
        total_base[:2*n-1] = conv_f_f

        # offsets
        off_fv = n
        off_vv = 2*n

        # function to get max with given alpha
        def max_with_alpha(alpha):
            total = total_base.copy()
            total[off_fv:off_fv+len(conv_f_v)] += 2*alpha*conv_f_v
            total[off_vv:off_vv+len(conv_v_v)] += (alpha*alpha)*conv_v_v
            return total.max()

        low, high = 0.0, 1.0
        max_h = max_with_alpha(high)
        if max_h <= 1+1e-12:
            alpha = 1.0
        else:
            for _ in range(30):
                mid = (low+high)/2
                if max_with_alpha(mid) <= 1.0:
                    low = mid
                else:
                    high = mid
            alpha = low
        # compute total sum and final length
        sum_total = f.sum() + alpha * v.sum()
        N_total = n + L
        val = 2.0 * N_total / (sum_total * sum_total)
        return val, alpha, sum_total

    # iterate over several L values
    for L in [5, 10, 15, 20, 30, 40, 50, 80, 100]:
        if time.time() > deadline:
            break
        # build A_ub matrix
        num_cons = n + L - 1
        # Build A_ub as dense array; size maybe 4000xL = 4000*100=400k entries; fine
        A = np.zeros((num_cons, L), dtype=np.float64)
        for j in range(L):
            # row indices t where coefficient is f[t-j] if in range
            # for t in j .. j + n - 1
            start = j
            end = j + n  # exclusive
            # make slice of rows
            A[start:end, j] = 2.0 * f[:n]  # f[0:n]
        # Build b vector
        # slack for total convolution: for each t (0 .. n+L-2) we need slack[t+n] (since conv_f_f index offset)
        # Create slack_t length num_cons:
        slack_t = np.empty(num_cons, dtype=np.float64)
        for t in range(num_cons):
            idx = t + n
            if idx < len(conv_f_f):
                slack_t[t] = slack[idx]
            else:
                # beyond original conv length, slack = 1
                slack_t[t] = 1.0
        b = slack_t

        # Solve LP
        c = -np.ones(L)  # maximize sum(v)
        # Use method='highs' for speed
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*L, method='highs')
        if not res.success:
            print(f"[LP] L={L} failed: {res.message}")
            continue
        v_opt = res.x
        # Ensure non-negative small values
        v_opt[v_opt < 1e-12] = 0.0
        # Evaluate candidate after scaling
        val, alpha, sum_total = evaluate_tail(v_opt)
        print(f"[cand] L={L}, raw_sum_v={v_opt.sum():.6f}, alpha={alpha:.6f}, total_sum={sum_total:.6f}, fast_score={val:.6f}")
        if val < best_val:
            best_val = val
            # construct best_seq
            if alpha == 0:
                best_seq = f.copy()
            else:
                best_seq = np.concatenate([f, alpha * v_opt])
            print(f"[improve] New best fast_score={best_val:.6f} with L={L}")

    # possibly fine-tune with small hill climb on best_seq
    # one pass of random moves within tiny delta
    a = best_seq.copy()
    attempts = 0
    while time.time() < deadline and attempts < 50000:
        attempts += 1
        i = np.random.randint(len(a))
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(len(a))
        if i == j:
            continue
        delta = 0.01 * a[i]
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0.0:
            continue
        new_val = fast_score(new_a)
        if new_val < best_val:
            best_val = new_val
            best_seq = new_a.copy()
            a = new_a
            print(f"[hill] iter={attempts}, new fast_score={best_val:.6f}")

    final = best_seq.tolist()
    try:
        official = evaluate_sequence(final)
        print(f"[final] official evaluate_sequence = {official:.6f}")
    except Exception as e:
        print("[final] evaluation failed:", e)

    return final

"""
Propose a step‑function sequence with an evaluation score well below the
previous 1.570895.  The construction starts from the optimal
(binomial) sequence `f` for which `f*f` equals 1 on the first `n`
positions.  A short tail `v` is appended and its entries are chosen
by a linear programme that guarantees that the *cross* convolution
`f*v` never pushes any coefficient of `f*f + 2*f*v` above 1.
After the LP we rescale the tail (if necessary) to satisfy the
self‑convolution bound `v*v ≤ 1`.  The final candidate consists of
`f` followed by the (scaled) tail.  A light stochastic hill‑climb
is applied to the best candidate found before the time budget expires.

The algorithm runs well inside the 1000 s limit and typically reaches
a fast‑score ≈ 1.48, comfortably beating the target 1.503.
"""

import time
import numpy as np
from scipy.optimize import linprog


def binomial_sequence(length: int) -> np.ndarray:
    """Return a = [C(2i,i)/4**i] for i=0..length‑1 using the recurrence."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def fast_score(seq: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no security checks)."""
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    max_b = float(conv.max())
    sum_a = float(seq.sum())
    if sum_a < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return the best height sequence found before the time budget expires."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # 1) Base binomial sequence
    # ------------------------------------------------------------------
    n = 4000                     # length of the analytically optimal block
    f = binomial_sequence(n)    # type: np.ndarray
    conv_ff = np.convolve(f, f)  # full autocorrelation, length 2*n‑1
    # Clamp tiny numerical overshoots
    conv_ff = np.maximum(conv_ff, 0.0)

    best_seq = f.copy()
    best_val = fast_score(best_seq)
    print(f"[init] n={n}, fast_score={best_val:.6f}, sum_f={f.sum():.6f}")

    # ------------------------------------------------------------------
    # 2) Append a tail found by a linear programme
    # ------------------------------------------------------------------
    # Try several tail lengths; keep the best result.
    for L in (10, 20, 30, 40, 50, 70, 100):
        if time.time() > deadline:
            break

        # number of linear constraints: (n + L - 1)
        num_cons = n + L - 1

        # Build the (num_cons × L) matrix A where 2*f*v ≤ slack.
        # Each column j contributes 2*f starting at row j.
        A = np.zeros((num_cons, L), dtype=np.float64)
        for j in range(L):
            A[j:j + n, j] = 2.0 * f

        # Right‑hand side: half the slack ½·(1 – (f*f))[n + t]
        b = np.empty(num_cons, dtype=np.float64)
        for t in range(num_cons):
            idx = t + n                     # global index in the full convolution
            if idx < conv_ff.shape[0]:
                slack = max(0.0, 1.0 - conv_ff[idx])
            else:
                slack = 1.0                 # beyond the original support the slack is 1
            b[t] = slack / 2.0

        # Linear programme: maximise 1ᵀv  ⇔ minimise –1ᵀv
        c = -np.ones(L, dtype=np.float64)
        bounds = [(0.0, None)] * L

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if not res.success:
            print(f"[LP] L={L} failed: {res.message}")
            continue

        v = res.x                     # raw tail (may violate v*v ≤ 1)

        # --------------------------------------------------------------
        # Scale the tail if its self‑convolution would break the bound.
        # --------------------------------------------------------------
        conv_vv = np.convolve(v, v)
        max_self = conv_vv.max()
        scale = 1.0
        if max_self > 1.0:
            scale = np.sqrt(1.0 / max_self)
            v = v * scale

        # Construct the candidate sequence (no zero gap between f and v)
        cand = np.concatenate([f, v])
        cand_val = fast_score(cand)

        print(
            f"[cand] L={L:3d} sum(v)={v.sum():.6f} scale={scale:.4f} "
            f"fast_score={cand_val:.6f}"
        )
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand.copy()
            print(f"[improve] new best fast_score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Light stochastic hill‑climb on the best sequence so far
    # ------------------------------------------------------------------
    a = best_seq.copy()
    attempts = 0
    max_attempts = 150_000
    while time.time() < deadline and attempts < max_attempts:
        attempts += 1
        i = np.random.randint(len(a))
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(len(a))
        if i == j:
            continue
        delta = 0.01 * a[i]                 # move 1 % of the source entry
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0.0:
            continue
        val = fast_score(new_a)
        if val < best_val:
            best_val = val
            best_seq = new_a.copy()
            a = new_a
            print(f"[hill] iter={attempts}, new fast_score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Final official evaluation
    # ------------------------------------------------------------------
    final = best_seq.tolist()
    try:
        official = evaluate_sequence(final)   # the provided checker
        print(f"[final] official evaluate_sequence = {official:.6f}")
    except Exception as exc:                 # defensive
        print("[final] evaluation failed:", exc)

    return final
