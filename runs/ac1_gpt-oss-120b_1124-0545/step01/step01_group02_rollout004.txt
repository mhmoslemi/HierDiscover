def project_onto_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u > cssv / np.arange(1, n+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def project_onto_simplex(v):
    # v is 1D numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        # all v <= 0
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
A fast projected gradient descent on a smooth approximation of the
objective  max(conv(a,a)).  The algorithm works on the simplex
{a_i ≥ 0, Σ a_i = 1}.  At each iteration we
  • compute the self‑convolution a*a via FFT,
  • form a soft‑max weighting w_k ∝ exp(λ·c_k) (λ grows during the run),
  • obtain the gradient of the smoothed objective as the cross‑correlation
    of a and w,
  • take a step and project back onto the simplex,
  • keep the best sequence seen (score = 2·n·max(c) because Σa=1).

The use of FFT makes each iteration O(n log n) and allows many
iterations within the 1000 s budget, often driving the score well below
the previous 1.552 bound.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # All entries are non‑positive – return the uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_root_sequence(n: int) -> np.ndarray:
    """
    Initialise a_i ∝ 1/√(i+1) and normalise so that Σ a_i = 1.
    This is the classic continuous‑model starting point.
    """
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    a /= a.sum()
    return a


def _smoothed_gradient(a: np.ndarray,
                       lam: float,
                       L: int,
                       fft_a: np.ndarray,
                       conv_full: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of  M_λ(a) = (1/λ) log Σ exp(λ·c_k)  where
    c = a*a (linear convolution).  The gradient is
        ∇M = 2 * (a ⋆ w)   (⋆ = cross‑correlation)
    with w_k ∝ exp(λ·c_k).
    """
    n = a.shape[0]

    # c = a*a (linear part of the circular convolution)
    c = conv_full[: 2 * n - 1]

    # soft‑max weights (avoid overflow)
    w = np.exp(lam * c)
    w_sum = w.sum()
    if w_sum == 0.0:                     # numerical safeguard
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum

    # FFT of w (same length L)
    fft_w = np.fft.rfft(w, L)

    # cross‑correlation a ⋆ w = ifft( FFT(a) * conj(FFT(w)) )
    corr = np.fft.irfft(fft_a * np.conj(fft_w), L)

    # The correlation offset is (len(w)-1)
    offset = w.shape[0] - 1
    grad = corr[offset: offset + n] * 2.0   # factor 2 from ∂c/∂a_i

    return grad, c


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises ``evaluate_sequence``.  The method is a projected gradient
    descent on a smooth approximation of the max‑convolution objective.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for graceful termination
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Parameters / initialisation
    # ------------------------------------------------------------------
    n = 5000                                 # length of the sequence
    a = _initial_root_sequence(n)            # start from 1/√(i+1)

    # pre‑compute the FFT length (≥ 3·n to avoid circular wrap‑around)
    L = _next_pow2(3 * n)

    # optimisation hyper‑parameters
    step = 0.5                               # initial step size
    step_decay = 0.9999                      # exponential decay
    lam = 2.0                                 # initial smoothing parameter
    lam_mul = 1.001                           # slowly increase λ
    lam_max = 30.0

    # store the best solution seen so far
    # compute its true score via the same formula used by evaluate_sequence
    conv0 = np.convolve(a, a)
    best_score = 2.0 * n * np.max(conv0)      # Σa = 1 ⇒ score = 2·n·max(c)
    best_seq = a.copy()
    print(f"[init] n={n}, score={best_score:.6f}")

    iteration = 0
    # ------------------------------------------------------------------
    # 2. Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- FFT of the current vector (re‑used for conv & corr) -----
        fft_a = np.fft.rfft(a, L)

        # circular convolution a*a (linear part is the first 2n‑1 entries)
        conv_full = np.fft.irfft(fft_a * fft_a, L)

        # ----- gradient of the smoothed max‑convolution -----
        grad, c = _smoothed_gradient(a, lam, L, fft_a, conv_full)

        # ----- gradient step + simplex projection -----
        a_new = a - step * grad
        a = _project_onto_simplex(a_new)

        # ----- evaluate (cheap) current objective -----
        cur_score = 2.0 * n * np.max(c)        # Σa = 1, so this matches evaluate_sequence
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # ----- schedule updates -----
        step = max(step * step_decay, 1e-8)   # never let step become exactly zero
        lam = min(lam * lam_mul, lam_max)

        # occasional random jitter to help escape flat regions
        if iteration % 5000 == 0:
            eps = step * 0.1
            idx_src, idx_tgt = np.random.choice(n, 2, replace=False)
            transfer = min(eps, a[idx_src])
            a[idx_src] -= transfer
            a[idx_tgt] += transfer
            a = _project_onto_simplex(a)     # re‑project after the perturbation

    # ------------------------------------------------------------------
    # 3. Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    print(f"[final] iterations={iteration}, best_score={best_score:.6f}")
    return best_seq.tolist()
