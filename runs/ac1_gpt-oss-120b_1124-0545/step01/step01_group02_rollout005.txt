"""
Gradient descent on a soft‑max relaxation of the max‑convolution objective.

The evaluation function is   score = 2·n·max(a*a)  (with ∑a = 1).
We minimise a smooth surrogate  L(α) = log‑sum‑exp(α·(a*a)) :

    ∂L/∂a_j = 2·∑_k w_k·a_{k-j},
    w_k = exp(α·c_k) / ∑_ℓ exp(α·c_ℓ),  c = a*a.

Using an exponential (multiplicative) update
    a ← a·exp(‑η·∂L/∂a)   and renormalising to ∑a = 1
keeps the variables non‑negative.  The temperature α starts small
(≈0.5) and is slowly increased (up to ≈20) so that the optimisation
first smooths the whole convolution and then focuses on the true
maximum.  A tiny linear‑step switch (random mass exchange) is applied
periodically to escape flat regions.  The best sequence found during
the time budget is returned.

The implementation is fully vectorised (numpy) and works for a
moderately large length (n≈5000) within the 1000 s budget.
"""

import time
import numpy as np

def _initial_root_sequence(n: int) -> np.ndarray:
    """Return the normalised 1/√(i+1) sequence of length n."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Stable soft‑max over the convolution values."""
    # subtract the maximum for numerical stability
    shifted = alpha * (conv - conv.max())
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    return w


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence``.  Returns the best sequence discovered before
    the time budget expires.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total time budget in seconds (the harness gives us 1000 s).
    kwargs : dict
        Ignored – kept for compatibility.

    Returns
    -------
    list[float]
        The best sequence (non‑negative, sum > 0).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------
    # 1.  Initialise the sequence
    # ------------------------------------------------------------
    # Prefer the previously found construction if it exists,
    # otherwise start from the analytically motivated 1/√x shape.
    if 'height_sequence_1' in globals():
        seq = np.array(height_sequence_1, dtype=float)
        if seq.ndim != 1 or np.any(seq < 0):
            seq = _initial_root_sequence(5000)
    else:
        seq = _initial_root_sequence(5000)

    # Normalise to sum = 1 (scale is irrelevant for the score)
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total == 0:
        seq = _initial_root_sequence(seq.size)
        total = seq.sum()
    seq /= total

    n = seq.size

    # Evaluate the initial score (exact formula, avoids extra overhead)
    conv = np.convolve(seq, seq)
    best_score = 2.0 * n * conv.max()
    best_seq = seq.copy()
    print(f"[init] n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------
    # 2.  Hyper‑parameters for the optimisation loop
    # ------------------------------------------------------------
    eta = 0.5          # initial step size for the multiplicative update
    eta_min = 1e-6
    eta_decay = 0.9999

    alpha = 0.5        # temperature of the soft‑max (low → smoother)
    alpha_max = 20.0
    alpha_inc = 1.001  # multiplicative increase each iteration

    eps_swap = 0.02    # size of the occasional random mass swap
    swap_interval = 100   # how often to perform a random swap
    log_interval = 200    # reporting frequency

    iteration = 0
    # ------------------------------------------------------------
    # 3.  Main optimisation loop
    # ------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- compute self‑convolution and the current score ----------
        conv = np.convolve(seq, seq)          # length 2n‑1
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        # ----- keep the best sequence seen so far -----------------------
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # ----- soft‑max weights over convolution entries ----------------
        w = _softmax_weights(conv, alpha)     # shape 2n‑1, sums to 1

        # ----- gradient of the soft‑max surrogate ----------------------
        # g_j = 2 * Σ_i w_i * a_{i-j}   (zero‑padding outside [0,n‑1])
        corr = np.correlate(w, seq, mode='full')
        start = len(w) - 1           # index of zero lag in the correlation result
        grad = 2.0 * corr[start:start + n]   # length n

        # ----- multiplicative (exponential) update --------------------
        # a ← a * exp(-η·grad)   (keeps positivity)
        seq = seq * np.exp(-eta * grad)

        # Guard against numerically vanishing entries
        eps = 1e-14
        seq = np.maximum(seq, eps)

        # Renormalise to keep Σ a = 1 (the score is scale‑invariant)
        seq /= seq.sum()

        # ----- occasional random mass exchange to break plateaus -------
        if iteration % swap_interval == 0:
            i, j = np.random.choice(n, 2, replace=False)
            # move a small fraction of the mass from i to j
            delta = eps_swap * seq[i]
            delta = min(delta, seq[i])  # stay non‑negative
            seq[i] -= delta
            seq[j] += delta
            seq /= seq.sum()

        # ----- optional symmetrisation (helps flatten the centre) -----
        if iteration % (swap_interval * 10) == 0:
            seq = (seq + seq[::-1]) * 0.5
            seq /= seq.sum()

        # ----- schedule updates for hyper‑parameters -------------------
        eta = max(eta * eta_decay, eta_min)
        alpha = min(alpha * alpha_inc, alpha_max)

        # ----- occasional logging --------------------------------------
        if iteration % log_interval == 0:
            elapsed = time.time() - start_time
            print(f"[log] iter={iteration}, score={cur_score:.6f}, "
                  f"eta={eta:.2e}, alpha={alpha:.2f}, time={elapsed:.1f}s")

    # ------------------------------------------------------------
    # 4.  Final report and return
    # ------------------------------------------------------------
    final_conv = np.convolve(best_seq, best_seq)
    final_score = 2.0 * n * final_conv.max()
    print(f"[final] iterations={iteration}, best_score={best_score:.6f}, "
          f"final_score_check={final_score:.6f}")

    return best_seq.tolist()
