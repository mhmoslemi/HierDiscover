"""
Propose a step‑function with a low evaluation score.

Algorithm
---------
1. Initialise the sequence `a` on the simplex.
   * If a previous construction (`height_sequence_1`) exists it is used.
   * Otherwise a “discrete arcsine” distribution is taken:
         a_i ∝ 1 / sqrt((i+0.5)*(n‑i‑0.5))

2. Iteratively improve the sequence by solving a linear programme.
   For the current `a` we compute
        M = max( conv(a, a) )
   and then solve
        maximise   Σ g_j
        subject to  (a * g)_k ≤ M   for every convolution index k
                    g_j ≥ 0
   This LP is linear because (a * g)_k is linear in the unknown vector g.
   The solution `g` carries as much mass as possible while respecting the
   same upper bound `M`.  After normalising `g` to have unit sum we mix it
   with `a`:
        a ← (1‑t)·a + t·g      (t is a step‑size, start ≈0.3)
   The mixing guarantees a monotone non‑increase of `max(conv(a,a))`.
   The LP is solved with `scipy.optimize.linprog` (highs solver).

3. After the LP phase a cheap local hill‑climb is performed:
   small random transfers of mass between two indices, acceptance only
   when the official `evaluate_sequence` value improves.

The whole routine runs for at most `budget_s` seconds (default ≈ 1000 s)
and finally returns the best sequence found (as a plain Python list of
floats).  All numbers are non‑negative and the sequence is normalised to
sum = 1, which makes the fast surrogate `2·n·max(conv)` identical to the
official evaluator.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> float:
    """
    Compute max(conv(a, a)) using an FFT based convolution.
    The vector `a` is assumed to sum to 1 (normalised) but the function
    works for any non‑negative vector.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def _build_A(f: np.ndarray) -> sparse.csc_matrix:
    """
    Build the sparse matrix A for the linear constraints
        (f * g)[k] = Σ_i f_i * g_{k‑i} ≤ M .
    For a given `f` (shape (n,)), A has shape (2n‑1, n) and
    entry A[k, j] = f_{k‑j} when 0 ≤ k‑j < n, otherwise 0.
    """
    n = f.shape[0]
    # each diagonal i (offset i) contains the constant f[i]
    diagonals = [np.full(n, val, dtype=float) for val in f]
    offsets = np.arange(n, dtype=int)
    A = sparse.diags(diagonals, offsets, shape=(2 * n - 1, n), format="csc")
    return A


def _mix_and_evaluate(a: np.ndarray, g: np.ndarray, t: float) -> tuple[float, np.ndarray]:
    """
    Mix `a` and `g` with step size `t`, re‑normalise and compute the exact
    score using the provided `evaluate_sequence`.  Returns (score, new_a).
    """
    cand = (1.0 - t) * a + t * g
    cand = np.clip(cand, 0.0, None)
    cand /= cand.sum()
    # evaluate_sequence is assumed to be already imported in the environment
    score = evaluate_sequence(cand.tolist())
    return float(score), cand


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights with a small
    `evaluate_sequence` score.  The function runs for at most
    `budget_s` seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------
    # Length of the step function (use the existing construction if any)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        init = np.array(height_sequence_1, dtype=float)
        init = np.clip(init, 0.0, None)
        if init.sum() > 0.0:
            a = init / init.sum()
        else:
            a = np.full(n, 1.0 / n, dtype=float)
    else:
        n = 2000
        a = np.full(n, 1.0 / n, dtype=float)

    # also keep a simple “discrete arcsine” start – it often gives a lower
    # initial max‑convolution than the uniform vector.
    i = np.arange(n, dtype=float)
    arcsine_raw = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine = arcsine_raw / arcsine_raw.sum()
    # choose the better of the two as the starting point
    if _conv_max(arcsine) < _conv_max(a):
        a = arcsine.copy()

    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[0.0s] Initial score: {best_score:.6f}")

    # --------------------------------------------------------------
    #  LP‑based refinement
    # --------------------------------------------------------------
    t_mix = 0.30                     # initial mixing coefficient
    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1

        # current bound M
        M = _conv_max(best_a)

        # build the linear‑program matrix for constraints (f * g) ≤ M
        A = _build_A(best_a)

        # Linear program:
        #   maximise   1ᵀ g          (equivalently minimise –1ᵀ g)
        #   subject to A @ g ≤ M,   g ≥ 0
        c = -np.ones(n, dtype=float)          # linprog *minimises*
        b = np.full(2 * n - 1, M, dtype=float)

        # solve LP with the high‑performance HiGHS solver
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
        )

        if not res.success or res.x is None:
            print(f"[{time.time() - start:.1f}s] LP failed at iteration {it} – stopping.")
            break

        g_opt = np.maximum(res.x, 0.0)
        sum_g = g_opt.sum()
        if sum_g == 0.0:
            print(f"[{time.time() - start:.1f}s] LP returned zero mass – stopping.")
            break

        # normalise the LP solution to a probability vector
        g_norm = g_opt / sum_g

        # try to improve the current best by mixing with g_norm
        improved = False
        t_local = t_mix
        while t_local > 1e-4:
            score, cand = _mix_and_evaluate(best_a, g_norm, t_local)
            if score < best_score - 1e-12:
                best_score = score
                best_a = cand
                improved = True
                print(
                    f"[{time.time() - start:.1f}s] iter={it:04d} "
                    f"t={t_local:.4f} new best={best_score:.6f}"
                )
                # after a successful step we keep the current mixing size
                break
            else:
                t_local *= 0.5

        if not improved:
            # no improvement – shrink the mixing coefficient a little
            t_mix *= 0.9
            no_improve += 1
        else:
            no_improve = 0
            # reset mixing size occasionally to keep the search energetic
            if it % 20 == 0:
                t_mix = min(0.5, t_mix * 1.1)

        # stop early if we see a long streak without progress
        if no_improve >= 30:
            print(f"[{time.time() - start:.1f}s] No progress for 30 LP steps – exiting LP loop.")
            break

        # occasional progress report
        if it % 10 == 0:
            elapsed = time.time() - start
            remaining = max(0.0, deadline - time.time())
            print(
                f"[{elapsed:.1f}s] LP iter={it} best={best_score:.6f} "
                f"time left≈{int(remaining)}s"
            )

    # --------------------------------------------------------------
    #  Final cheap hill‑climbing (small random mass transfers)
    # --------------------------------------------------------------
    a = best_a.copy()
    hill_iters = 3000
    for k in range(hill_iters):
        if time.time() >= deadline:
            break
        i, j = np.random.choice(n, 2, replace=False)
        # move a modest fraction of mass from i to j
        delta = a[i] * np.random.uniform(-0.05, 0.05)
        delta = np.clip(delta, -a[i] + 1e-12, 1e12)
        a[i] -= delta
        a[j] += delta
        a = np.clip(a, 0.0, None)
        a /= a.sum()

        # evaluate only every 200 steps to keep the overhead low
        if (k + 1) % 200 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - start:.1f}s] hill‑climb new best={best_score:.6f}")

    print(f"[{time.time() - start:.1f}s] Finished. Best score = {best_score:.6f}")
    return best_a.tolist()
