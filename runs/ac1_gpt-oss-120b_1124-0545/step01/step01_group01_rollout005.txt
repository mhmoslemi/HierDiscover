def peak_reduction_move(a, m, target_sum, T):
    idx = np.arange(len(a))
    partner = m - idx
    mask = (partner >= 0) & (partner < len(a))
    contributions = np.zeros_like(a)
    contributions[mask] = a[idx[mask]] * a[partner[mask]]
    # find i with maximum contribution
    i_candidates = np.argpartition(-contributions, 10)[:10]
    i = i_candidates[np.argmax(contributions[i_candidates])]
    # find j with minimal contribution, but also not the same index
    j_candidates = np.argpartition(contributions, 10)[:10]
    # pick j from those not equal to wanted maybe also partner invalid
    # but we can also choose j from invalid partner (if any)
    invalid = ~mask
    if np.any(invalid):
        # choose random j from invalid positions (zero contribution)
        j = np.random.choice(np.where(invalid)[0])
    else:
        # choose among low contributions
        j = j_candidates[np.argmin(contributions[j_candidates])]
    # compute delta as fraction of a[i] and also scaled by T
    scale = T * 0.5  # parameter
    delta = min(a[i] * scale, target_sum * 0.01)
    # ensure positive:
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # projection onto simplex
    a_new = _simplex_projection(a_new, target_sum)
    return a_new

            # gradient step
            conv = np.convolve(a_cur, a_cur)
            m = int(np.argmax(conv))
            idx = np.arange(n)
            partner = m - idx
            mask = (partner >= 0) & (partner < n)
            grad = np.zeros_like(a_cur)
            grad[mask] = a_cur[partner[mask]]
            grad *= 2.0
            eta = target_sum * 0.2 * T
            a_prop = a_cur - eta * grad
        # projection
        a_prop = _simplex_projection(a_prop, target_sum)

        # evaluate objective
        new_score = np.max(np.convolve(a_prop, a_prop))

        # acceptance
        if new_score < cur_score or random.random() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
            a_cur = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a_prop.copy()
                print(f"[{iteration}] New best: {best_score:.6f} time {elapsed:.1f}s")
        # occasional random restart if no improvement for many iterations
        # Could check iteration % 50000?
        iteration += 1

        # optionally to avoid stagnation:

            # if cur_score is not improving we restart

                a_cur = _simplex_projection(a_cur, target_sum)
                cur_score = np.max(np.convolve(a_cur, a_cur))

    return best_seq.tolist()

"""
Propose a step‑function (non‑negative vector) that minimizes
   E(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

because of the normalisation sum(a)=sqrt(2n) the objective reduces to
the maximal entry of the self‑convolution.  The optimiser is a
simulated‑annealing / hill‑climbing loop that works directly on the
simplex:

* The vector is always projected onto the probability simplex
  {x≥0, sum(x)=sqrt(2n)} by a fast O(n log n) algorithm.
* Four types of elementary moves are mixed:
    – random transfer of a small amount of mass between two coordinates,
    – swapping two coordinates (permutes the multiset of values),
    – a “peak‑reduction’’ move that transfers mass from a coordinate
      that contributes most to the current maximal convolution entry
      to a coordinate that contributes little,
    – a sub‑gradient step that reduces the entry attaining the maximum.
* An exponential cooling schedule controls the acceptance probability.
* The best vector seen so far is remembered and printed whenever it
  improves.

The routine respects the 1000 s budget, returns a plain list of floats
and never performs any I/O.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project a vector `v` onto the simplex {x >= 0, sum(x)=s}.
    Implementation of the algorithm by Wang & Carreira‑Perpiñán (2013).
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – should happen only when s=0
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, nfft: int) -> float:
    """
    Return the maximal entry of the linear convolution a*a.
    Uses an FFT of length `nfft` (next power of two larger than 2*len(a)-1).
    """
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * a.size - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _peak_reduction_move(a: np.ndarray, max_idx: int,
                         target_sum: float, temperature: float) -> np.ndarray:
    """
    Move a small amount of mass from a coordinate that heavily contributes
    to the current maximal convolution entry to a coordinate that contributes
    little (or not at all).  The move is followed by a simplex projection.
    """
    n = a.size
    idx = np.arange(n)
    partner = max_idx - idx
    valid = (partner >= 0) & (partner < n)

    # contribution of each coordinate to the maximal entry
    contrib = np.zeros_like(a)
    contrib[valid] = a[idx[valid]] * a[partner[valid]]

    # source: one of the top contributors
    top_k = 10
    src_candidates = np.argpartition(-contrib, top_k)[:top_k]
    src = src_candidates[np.argmax(contrib[src_candidates])]

    # destination: preferably a coordinate that does NOT contribute
    noncontrib = np.where(~valid)[0]
    if noncontrib.size > 0:
        dst = int(np.random.choice(noncontrib))
    else:
        # fallback – pick a low‑contribution coordinate
        low_k = 10
        dst_candidates = np.argpartition(contrib, low_k)[:low_k]
        dst = int(np.random.choice(dst_candidates))

    # amount to move – scaled with temperature, but never larger than the
    # source coordinate and never smaller than a tiny epsilon.
    max_delta = min(a[src], target_sum * 0.02)
    delta = max_delta * temperature * random.random()
    # guard against moving zero
    if delta < 1e-12:
        return a.copy()

    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    # projection restores the exact simplex constraint
    a_new = _simplex_projection(a_new, target_sum)
    return a_new

                     budget_s: int = 1000,

    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the evaluation function defined in the problem statement.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence found.
    """
    # ------------------------------------------------------------------
    # Initialise RNGs
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size – reuse a known construction if present
    if "height_sequence_1" in globals():

    else:

        a = np.random.exponential(scale=1.0, size=n)

    # Normalise to the conventional ℓ₁‑norm √(2 n)
    target_sum = math.sqrt(2.0 * n)
    a = a / a.sum() * target_sum

    # FFT length (next power of two for efficient convolution)
    nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Initial score (max entry of the self‑convolution)
    cur_score = _max_convolution(a, nfft)
    best_seq = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin
    iteration = 0

    T_start = 1.0
    T_end = 1e-6

    # ------------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        # exponential decay: T = T_start * (T_end/T_start)**frac
        temperature = T_start * ((T_end / T_start) ** frac)

        # --------------------------------------------------------------
        # Choose a move type
        move_type = random.random()
        a_prop = a.copy()

        if move_type < 0.30:
            # 1) Random transfer of a small amount of mass i -> j
            i, j = np.random.randint(n, size=2)
            if i != j:
                max_delta = min(a_prop[i], target_sum * 0.04)
                delta = max_delta * temperature * random.random()
                if delta > 1e-12:
                    a_prop[i] -= delta
                    a_prop[j] += delta

        elif move_type < 0.55:
            # 2) Simple swap of two coordinates (permutes the multiset)
            i, j = np.random.randint(n, size=2)
            if i != j:
                a_prop[i], a_prop[j] = a_prop[j], a_prop[i]

        elif move_type < 0.80:
            # 3) Peak‑reduction move (target the current maximal conv entry)
            conv = np.convolve(a, a)
            max_idx = int(np.argmax(conv))
            a_prop = _peak_reduction_move(a, max_idx, target_sum, temperature)

        else:
            # 4) Sub‑gradient step aimed at the current maximal entry
            conv = np.convolve(a, a)
            max_idx = int(np.argmax(conv))
            idx = np.arange(n)
            partner = max_idx - idx
            mask = (partner >= 0) & (partner < n)
            grad = np.zeros_like(a)
            # gradient of max entry (factor 2 is irrelevant for the direction)
            grad[mask] = a[partner[mask]]
            eta = target_sum * 0.20 * temperature
            a_prop -= eta * grad

        # --------------------------------------------------------------
        # Ensure feasibility: non‑negative and correct ℓ₁‑norm
        a_prop = _simplex_projection(a_prop, target_sum)

        # --------------------------------------------------------------
        # Evaluate the new candidate (fast internal evaluation)
        new_score = _max_convolution(a_prop, nfft)

        # --------------------------------------------------------------
        # Acceptance rule (simulated annealing)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # uphill move accepted with probability exp(-(Δ)/T)
            prob = math.exp(-(new_score - cur_score) / (temperature + 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            a = a_prop
            cur_score = new_score
            # Track the global best
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{iteration:7d}] New best score: {best_score:.6f} (time {elapsed:.1f}s)")

        # Occasionally (every ~30 000 iterations) perform a small random
        # perturbation of the current best to escape possible stalls.
        if iteration % 30000 == 0 and iteration > 0:
            a = best_seq + np.random.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a, 0.0)
            a = _simplex_projection(a, target_sum)
            cur_score = _max_convolution(a, nfft)

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list.
    return best_seq.tolist()
