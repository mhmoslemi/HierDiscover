    n = a.shape[0]

    b_ub = np.full(2*n-1, M, dtype=float)
    c = -np.ones(n)  # maximize sum => minimize negative sum

"""
Propose a step‑function that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is a deterministic improvement loop based on the
linear‑programming scheme described in the statement.  Starting from a
normalised seed (the previous construction if present, otherwise uniform)
we repeatedly

1. compute M = max(conv(a,a));
2. solve the LP   maximise 1ᵀg   s.t.   a∗g ≤ M , g ≥ 0
   (the LP matrix is the convolution matrix of ``a``);
3. mix the current distribution with the LP solution:
        a ← (1–t)·a + t·g   (t is chosen by a cheap line‑search that
   minimises the new maximal convolution);
4. keep the best exact score (via ``evaluate_sequence``).

When the LP no longer yields improvement we finish with a short random
local‑search hill‑climb.  The whole routine respects the 1000 s time limit
and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based – fast for the sizes we use)
# ----------------------------------------------------------------------
def _conv_full(x: np.ndarray, y: np.ndarray | None = None) -> np.ndarray:
    """Full linear convolution of two 1‑D arrays using FFT."""
    if y is None:
        y = x
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 2).bit_length())        # next power‑of‑2
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[: n + m - 1]
    return conv


def _conv_max(x: np.ndarray) -> float:
    """Maximum entry of the full (non‑circular) convolution x∗x."""
    return float(_conv_full(x).max())


# ----------------------------------------------------------------------
# LP matrix construction
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray):
    """
    Return the matrix A (shape (2n‑1, n)) such that
        (A @ g)[k] == (a∗g)[k]   for all k.
    The matrix is built as a collection of shifted copies of ``a`` and
    returned as a CSR sparse matrix.
    """
    from scipy.sparse import diags
    n = a.shape[0]
    # one diagonal for each possible shift 0 … n‑1,
    # each diagonal contains the whole vector a
    offsets = np.arange(n, dtype=int)          # 0,1,…,n‑1
    # diags expects a list of arrays – we repeat the same array `a`
    data = [a] * n
    return diags(data, offsets, shape=(2 * n - 1, n), format="csr")


# ----------------------------------------------------------------------
# Simple line search for the optimal mixing weight t∈(0,1)
# ----------------------------------------------------------------------
def _best_mix_t(aa, ag, gg, cur_max):
    """
    Given the three convolution vectors
        aa = a∗a, ag = a∗g, gg = g∗g,
    find a step size t∈(0,1) that minimises
        max_k ((1‑t)²·aa[k] + 2·t·(1‑t)·ag[k] + t²·gg[k]).

    Uses a cheap coarse grid and a refined search around the best grid point.
    """
    # coarse grid – 0 is excluded because it gives no improvement
    t_grid = np.linspace(0.0, 0.5, 41)[1:]  # up to 0.5 is enough in practice
    best_t, best_val = 0.0, cur_max

    for t in t_grid:
        mixed = (1 - t) ** 2 * aa + 2 * t * (1 - t) * ag + (t ** 2) * gg
        cur = mixed.max()
        if cur < best_val:
            best_val, best_t = cur, t

    # refine locally if a promising t was found
    if best_t > 0.0:
        lo = max(0.0, best_t - 0.05)
        hi = min(0.5, best_t + 0.05)
        t_grid2 = np.linspace(lo, hi, 31)[1:]  # skip 0 again
        for t in t_grid2:
            mixed = (1 - t) ** 2 * aa + 2 * t * (1 - t) * ag + (t ** 2) * gg
            cur = mixed.max()
            if cur < best_val:
                best_val, best_t = cur, t
    return best_t, best_val
