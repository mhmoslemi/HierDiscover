def propose_candidate(seed=0, budget_s=1000, **kwargs):

    np.random.seed(seed)
    random.seed(seed)

    # Determine n: maybe start with 5000 or double existing length.
    base_n = len(height_sequence_1) if "height_sequence_1" in globals() else 1500
    target_n = max(base_n * 2, 5000)
    n = int(target_n)

    S = math.sqrt(2.0 * n)

    # Starting sequences: we can create:
    # - uniform
    # - scaled version of existing sequence padded
    # - random exponential
    # We'll evaluate each and set best.
    seqs = []
    uniform = np.full(n, S / n)
    seqs.append(uniform)

    # if previous seq exists, upsample by repeating each element (or linear interpolation)
    if "height_sequence_1" in globals():
        old = np.array(height_sequence_1, dtype=float)
        # upsample: repeat each element via np.repeat until length n, trim.
        repeat_factor = int(np.ceil(n / old.size))
        up = np.repeat(old, repeat_factor)[:n]
        up = up / up.sum() * S
        seqs.append(up)

    # random exponential
    random_seq = np.random.exponential(scale=1.0, size=n)
    random_seq = random_seq / random_seq.sum() * S
    seqs.append(random_seq)

    best_seq = None
    best_score = float('inf')
    for seq in seqs:
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"Initial seq score {score:.6f}")

    # hyperparameters
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    # schedule params
    T_start = 1.0
    T_end = 1e-5

    # momentum
    momentum = np.zeros(n)

    iteration = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        T = T_start * ((T_end / T_start) ** frac)

        proposal = best_seq.copy()

        # Random mass transfer (two indices)
        i, j = np.random.randint(n, size=2)
        if i != j:
            # delta scale with T
            max_delta = proposal[i] * 0.5  # cannot move more than half of it
            delta = np.random.uniform(-max_delta, max_delta) * T
            # incorporate to ensure nonneg after
            delta = max(-proposal[i], min(delta, proposal[i]))
            proposal[i] -= delta
            proposal[j] += delta

        # Subgradient for active max conv indices (stretch)
        # Compute conv using FFT for speed
        # Choose next power of two size for speed
        n_fft = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(proposal, n_fft)
        conv = np.fft.irfft(a_fft * a_fft, n_fft)[:2 * n - 1]
        max_b = conv.max()
        # compute active indices within 1e-12 absolute tolerance
        tol = max_b * 1e-12
        active = np.where(conv >= max_b - tol)[0]

        # compute gradient (subgrad)
        grad = np.zeros_like(proposal)
        for k in active:
            idx = np.arange(n)
            partners = k - idx
            mask = (partners >= 0) & (partners < n)
            grad[idx[mask]] += proposal[partners[mask]]

        # combine gradient with momentum
        beta = 0.7
        momentum = beta * momentum + (1 - beta) * grad
        # step size
        step = T * S * 0.1  # scaling factor
        proposal = proposal - step * momentum

        # Project onto simplex
        proposal = _simplex_projection(proposal, S)

        # Evaluate
        try:
            score = evaluate_sequence(proposal.tolist())
        except Exception:
            continue

        if score < best_score - 1e-12:
            best_score = score
            best_seq = proposal.copy()
            print(f"[{iteration:6d}] New best: {best_score:.6f} (t={elapsed:.1f}s)")
        else:
            # maybe accept with probability (to escape)
            if np.random.rand() < np.exp(-(score - best_score) / (T + 1e-12)):
                best_seq = proposal.copy()

        # Periodically (every 1000 iterations) do a random restart to avoid stagnation
        if iteration % 1000 == 0 and iteration > 0:
            # small random perturbation from best
            pert = best_seq + np.random.normal(scale=S*0.01, size=n)
            pert = np.maximum(pert, 0)
            pert = _simplex_projection(pert, S)
            # Possibly accept if better
            s2 = evaluate_sequence(pert.tolist())
            if s2 < best_score:
                best_score = s2
                best_seq = pert.copy()
                print(f"[{iteration:6d}] New best after perturb: {best_score:.6f}")
        iteration += 1

    print(f"Final best score: {best_score:.6f}")
    return best_seq.tolist()
