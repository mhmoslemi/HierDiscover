    """Exact evaluation of a sequence (the same formula as evaluate_sequence)."""
    s = a.sum()
    if s < 1e-12:                     # should never happen – defensive
        return np.inf
    b = np.convolve(a, a)
    max_b = b.max()
    return float(2.0 * len(a) * max_b / (s * s))


def random_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from a random source to a random target."""
    n = len(a)
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)

    max_transfer = a[i]
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # re‑normalise to eliminate round‑off drift
    a_new /= a_new.sum()
    return a_new


def targeted_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Move a little mass away from the index that contributes most to the
    current maximal convolution entry.
    Probability of picking an index is proportional to its contribution
    a[i]·a[k_max‑i].
    """
    n = len(a)
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))

    # contribution of each i to b[k_max]
    w = np.zeros(n)
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1)
    w[idx] = a[idx] * a[k_max - idx]

    if w.sum() == 0.0:
        i = np.random.randint(n)
    else:
        probs = w / w.sum()
        i = np.random.choice(n, p=probs)

    # pick a destination index different from i
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1

    max_transfer = a[i]
    if max_transfer == 0.0:
        return a.copy()

    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new


def gradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    Sub‑gradient descent on a smooth approximation of max(conv).
    The direction d = -g + mean(g) keeps the sum unchanged.
    """
    n = len(a)
    b = np.convolve(a, a)

    # use a smooth proxy for the max – log‑sum‑exp with a moderate β
    beta = 30.0
    # stable log‑sum‑exp
    max_b = b.max()
    lse = max_b + np.log(np.exp(beta * (b - max_b)).sum()) / beta
    # softmax weights for the proxy gradient
    w = np.exp(beta * (b - max_b))
    w /= w.sum()

    # gradient of Σ w_k * (a * a)[k] w.r.t. a_i:
    #   ∂/∂a_i Σ w_k (Σ_j a_j a_{k-j})
    #   = Σ_k w_k (2·a_{k-i})   (the factor 2, except when k=2i)
    g = np.zeros_like(a)
    for i in range(n):
        # k ranges where a_{k-i} is defined
        lo = i
        hi = i + n - 1
        ks = np.arange(lo, hi + 1)
        # a_{k-i} = a[ks - i]   always inside [0,n)
        contrib = a[ks - i] * w[ks]
        # double count except on the diagonal (k = 2*i)
        diag = (ks == 2 * i)
        g[i] = 2.0 * contrib.sum() - contrib[diag].sum()
    # project the update to keep the sum of a unchanged
    d = -g + g.mean()
    a_new = a + step * d
    a_new = np.clip(a_new, 0.0, None)

    total = a_new.sum()
    if total == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= total
    return a_new

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a good height sequence within ``budget_s`` seconds.
    Returns the best sequence found as a plain list of Python floats.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a tiny safety margin
    deadline = start + budget_s - 0.5

    # length of the step function – can be overridden via kwargs
    n = int(kwargs.get("n", 2000))

    # --------------------------------------------------------------
    # Initialise – use the global warm start if it exists
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = globals()["height_sequence_1"]
        if isinstance(hs, (list, tuple, np.ndarray)) and len(hs) >= n:
            a = np.array([float(x) for x in hs[:n]])
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s < 0.01:                     # fallback to random if the warm start is too small
                a = np.random.rand(n)
                a /= a.sum()
            else:
                a /= s
        else:
            a = np.random.rand(n)
            a /= a.sum()
    else:
        a = np.random.rand(n)
        a /= a.sum()

    best_seq = a.copy()
    best_score = compute_score(best_seq)
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # annealing parameters
    init_temp = 1.0
    final_temp = 0.001
    init_step = 0.20          # relative to current mass
    final_step = 0.0005

    iteration = 0
    last_improve = start

    # --------------------------------------------------------------
    # Main SA / hill‑climbing loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / (budget_s - 10.0)   # keep a small safety buffer

        # temperature for Metropolis acceptance
        temp = init_temp * (1.0 - progress) + final_temp * progress
        # step size (relative transfer factor)
        step_scale = init_step * (1.0 - progress) + final_step * progress

        # pick a move
        r = np.random.rand()
        if r < 0.35:                     # targeted move – most important
            cand = targeted_move(a, step_scale)
        elif r < 0.70:                   # random mass transfer
            cand = random_move(a, step_scale)
        elif r < 0.85:                   # gradient‑like sub‑gradient step
            cand = gradient_step(a, step_scale)
        else:                            # simple swap – zero‑error perturbation
            i, j = np.random.randint(n, size=2)
            while i == j:
                j = np.random.randint(n)
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]

        cand_score = compute_score(cand)

        # Metropolis acceptance (lower score is better)
        if cand_score < best_score or np.random.rand() < np.exp(-(cand_score - best_score) / max(temp, 1e-12)):
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:6d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # restart if we have been idle for a while
        if iteration % 4000 == 0 and (time.time() - last_improve) > 25.0:
            a = np.random.rand(n)
            a /= a.sum()
            print(f"[{iteration:6d}] restart due to stagnation")

        iteration += 1

    total_time = time.time() - start
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
    return best_seq.tolist()
