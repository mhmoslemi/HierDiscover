"""
Gradient‑descent on a soft‑max of the self‑convolution.

The evaluation function (with the ℓ₁‑norm fixed to √(2 n)) is simply
the maximum entry of the autocorrelation.  We minimise a smooth
approximation   fβ(a)= (1/β)·log Σk exp(β·(a*a)[k])   using projected
gradient descent on the probability simplex.  β grows from a small
value (smooth) to a large value (sharp) while the step size shrinks.
Occasional random restarts prevent stagnation.  The best vector found
within the time budget is returned.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex { w ≥ 0 , sum(w) = s }."""
    n = v.size
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (vector) that minimises
    max(np.convolve(a, a)).  The algorithm uses a temperature‑controlled
    soft‑max of the convolution together with projected gradient descent
    on the simplex.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        Ignored (present for compatibility).

    Returns
    -------
    list[float]
        The best coefficient sequence found within the time budget.
    """
    random.seed(seed)
    np.random.seed(seed)

    # --------------------------------------------------------------
    # Problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        start_vec = np.array(height_sequence_1, dtype=np.float64)
        # re‑normalise to the canonical ℓ₁‑norm
        start_vec = start_vec / start_vec.sum() * math.sqrt(2.0 * n)
    else:
        n = 2500                     # safe default when no prior data
        start_vec = None

    target_sum = math.sqrt(2.0 * n)   # the normalising ℓ₁‑norm

    # --------------------------------------------------------------
    # Initial candidates
    uniform = np.full(n, target_sum / n, dtype=np.float64)

    # pick a decent start (uniform is a reliable baseline)
    if start_vec is not None:
        cur = start_vec.copy()
    else:
        cur = np.random.exponential(scale=1.0, size=n)
        cur = cur / cur.sum() * target_sum

    # keep the best we have seen so far
    best_seq = uniform.copy()
    best_score = np.max(np.convolve(best_seq, best_seq))
    # compare with the random start as well
    rand_score = np.max(np.convolve(cur, cur))
    if rand_score < best_score:
        best_seq = cur.copy()
        best_score = rand_score

    # --------------------------------------------------------------
    # Scheduling parameters
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    beta_start = 0.5          # soft‑max temperature at the beginning
    beta_end = 200.0          # temperature at the end (sharp max)
    step_start = target_sum * 0.4   # initial gradient‑step size
    step_min = target_sum * 1e-4    # minimum step size

    # Fixed FFT length (next power of two ≥ n + (2n‑1) – 1)
    m = 2 * n - 1
    L = 1 << ((n + m - 1).bit_length())

    iteration = 0
    last_improve = start_time

    # --------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)   # 0 … 1

        # temperature and step size schedules
        beta = beta_start * (beta_end / beta_start) ** frac
        step = step_start * (1.0 - frac) + step_min

        # ---------- self‑convolution (FFT) ----------
        fft_a = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(fft_a * fft_a, n=L)[:m]   # length 2n‑1

        # ---------- soft‑max weights ----------
        scaled = beta * conv_full
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)          # avoid overflow
        probs = exp_scaled / exp_scaled.sum()             # Σ probs = 1

        # ---------- gradient of the soft‑max ----------
        # correlation a ⋆ probs  (equivalent to conv(a, probs[::-1]))
        fft_w_rev = np.fft.rfft(probs[::-1], n=L)
        corr_full = np.fft.irfft(fft_a * fft_w_rev, n=L)[:n + m - 1]
        start_idx = m - 1                                 # index of lag 0
        grad = 2.0 * corr_full[start_idx:start_idx + n]   # ∂f/∂a

        # ---------- gradient step ----------
        proposal = cur - step * grad

        # occasional jitter to escape tiny plateaus
        if np.random.rand() < 0.02:
            proposal += step * 0.05 * np.random.randn(n)

        # project back onto the simplex (keeps sum = target_sum)
        proposal = _simplex_projection(proposal, target_sum)

        # ---------- evaluate ----------
        # because sum = √(2 n) the evaluation is simply max(conv)
        score = np.max(np.convolve(proposal, proposal))

        # ---------- keep the best ----------
        if score < best_score:
            best_score = score
            best_seq = proposal.copy()
            print(f"[{iteration:6d}] New best {best_score:.6f} "
                  f"(t={elapsed:.1f}s, β={beta:.2f})")
            last_improve = time.time()

        # move on – we keep the latest proposal as the current point
        cur = proposal

        # ---------- stagnation handling ----------
        if time.time() - last_improve > 60.0:   # one minute without improvement
            cur = np.random.exponential(scale=1.0, size=n)
            cur = cur / cur.sum() * target_sum
            last_improve = time.time()
            print(f"[{iteration:6d}] Restarted (t={elapsed:.1f}s)")

        iteration += 1

    # --------------------------------------------------------------
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
