def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    # Determine length n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        base_seq = np.array(height_sequence_1, dtype=float)
    else:
        n = 2000
        base_seq = _random_simplex(n, math.sqrt(2.0 * n))

    target_sum = math.sqrt(2.0 * n)

    # Initialize population
    pop_size = 5
    population = []
    scores = []

    # include base_seq as one of the population
    population.append(base_seq.copy())
    scores.append(evaluate_sequence(base_seq.tolist()))

    # random others
    for _ in range(pop_size - 1):
        seq = _random_simplex(n, target_sum)
        pop = seq.copy()
        population.append(pop)
        scores.append(evaluate_sequence(pop.tolist()))

    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    start = time.time()
    deadline = start + budget_s - 2.0

    # temperature schedule
    T0 = 1.0
    T_end = 1e-4

    iteration = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = T0 * ((T_end / T0) ** frac)

        # select parent
        # probability weighted by inverse score (better get selected more)
        # Use softmax over negative scores to give higher chance to better
        weights = np.exp(-np.array(scores) / (T + 1e-12))
        weights /= weights.sum()
        parent_idx = np.random.choice(len(population), p=weights)
        parent = population[parent_idx].copy()
        # choose move type random
        move_type = np.random.rand()
        if move_type < 0.4:
            # random transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                sigma = target_sum * 0.02 * T
                delta = np.random.normal(0, sigma)
                delta = max(-parent[i], min(delta, parent[i]))
                parent[i] -= delta
                parent[j] += delta
                parent = _simplex_projection(parent, target_sum)
        elif move_type < 0.7:
            # subgradient step
            conv = np.convolve(parent, parent)
            k_max = int(np.argmax(conv))
            idx = np.arange(n)
            partner = k_max - idx
            mask = (partner >= 0) & (partner < n)
            gradient = np.zeros_like(parent)
            gradient[mask] = parent[partner[mask]]
            eta = target_sum * 0.1 * T
            parent -= eta * gradient
            parent = _simplex_projection(parent, target_sum)
        elif move_type < 0.9:
            # targeted shift
            conv = np.convolve(parent, parent)
            k_max = int(np.argmax(conv))
            idx = np.arange(n)
            partner = k_max - idx

            # contributions as weight
            contrib = np.zeros_like(parent)
            contrib[mask] = parent[idx[mask]] * parent[partner[mask]]
            if contrib.sum() > 0:
                # choose i proportionally to contrib
                probs = contrib / contrib.sum()
                i = np.random.choice(n, p=probs)
            else:
                i = np.random.choice(np.where(mask)[0]) if mask.any() else np.random.randint(n)
            # choose j outside overlapping region

            nonmask_idxs = np.where(~mask)[0]

            best_seq = parent.copy()
            print(f"[{iteration:6d}] New global best: {best_score:.6f} time {elapsed:.1f}s")
        iteration += 1

"""
Hybrid evolutionary+gradient optimizer for the maximal self‑convolution
problem.

The algorithm works on a fixed length `n` (default 5000) and keeps the
ℓ₁‑norm of the vector fixed to √(2 n).  A small population of candidate
vectors is maintained.  In each iteration a parent is selected, a move
is applied (random mass transfer, a sub‑gradient step on the current
largest convolution entry, or a targeted shift that moves mass out of
the indices that contribute to that entry), and the new vector is
projected onto the simplex.  The objective (the maximal entry of the
auto‑convolution) is evaluated locally (no call to the expensive
`evaluate_sequence` wrapper).  Accepted improvements replace the parent
and the global best is updated.  The temperature is annealed
exponentially, which controls the step sizes and the probability of
up‑hill moves.  The method runs until the time budget expires and
returns the best sequence found.
"""

import time
import math
import random
import numpy as np

def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x>=0, sum(x)=s}."""
    # Implementation of the algorithm by Wang & Carreira‑Perpiñán (2013)
    if s < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.size
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find the last index where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero; return uniform vector
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _random_simplex(n: int, total: float) -> np.ndarray:
    """Draw a random point on the simplex of dimension n and L1 sum `total`."""
    # Using exponential distribution (Dirichlet with α=1)
    x = np.random.exponential(scale=1.0, size=n)
    x = x / x.sum() * total
    return x

def _objective(seq: np.ndarray) -> float:
    """Return the maximal entry of the self‑convolution of `seq`."""
    # seq is assumed to be a 1‑D numpy array of non‑negative floats

    T_end = 1e-4
    iteration = 0

    while time.time() < deadline:

                if i_candidates.size == 0:
                    i = np.random.randint(n)
                else:
                    i = int(np.random.choice(i_candidates))
            # Choose a destination index j that does *not* overlap for the same k_max
            non_overlap = np.where(~mask)[0]
            if non_overlap.size == 0:
                j = np.random.randint(n)
            else:
                j = int(np.random.choice(non_overlap))
            # Transfer a fraction of the source mass
            max_delta = parent[i]
            delta = max_delta * 0.5 * T   # up to half of the source, scaled by temperature
            if delta > 0:
                parent[i] -= delta
                parent[j] += delta
                # Project only to guard against tiny numerical drift
                parent = _simplex_projection(parent, target_sum)

            fresh = _random_simplex(n, target_sum)
            alpha = T   # at high temperature we mix more aggressively

            parent = _simplex_projection(parent, target_sum)

        cand_score = _objective(parent)

        # Acceptance test (simulated annealing style)
        # Always keep a better solution; otherwise accept with probability
        # exp(-(Δ) / T) where Δ = cand_score - old_score
        old_score = scores[parent_idx]
        if cand_score < old_score or random.random() < math.exp(-(cand_score - old_score) / (T + 1e-12)):

            population[parent_idx] = parent

            if cand_score < best_score:
                best_score = cand_score
                best_seq = parent.copy()
                print(f"[{iteration:6d}] New best score: {best_score:.6f} (time {elapsed:.1f}s)")

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list.
    # A final safeguard: make sure the sum is exactly the target (within tolerance)
    best_seq = _simplex_projection(best_seq, target_sum)
    return best_seq.tolist()

"""
Hybrid evolutionary+gradient optimizer for the maximal self‑convolution
problem.

The algorithm works on a fixed length `n` (default 5000) and keeps the
ℓ₁‑norm of the vector fixed to √(2 n).  A small population of candidate
vectors is maintained.  In each iteration a parent is selected, a move
is applied (random mass transfer, a sub‑gradient step on the current
largest convolution entry, or a targeted shift that moves mass out of
the indices that contribute to that entry), and the new vector is
projected onto the simplex.  The objective (the maximal entry of the
auto‑convolution) is evaluated locally (no call to the expensive
`evaluate_sequence` wrapper).  Accepted improvements replace the parent
and the global best is updated.  The temperature is annealed
exponentially, which controls the step sizes and the probability of
up‑hill moves.  The method runs until the time budget expires and
returns the best sequence found.
"""

import time
import math
import random
import numpy as np

def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto {x≥0, sum(x)=s} (Wang‑Carreira‑Perpiñán, 2013)."""
    if s < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]          # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):          # all entries will be zero → uniform vector
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _random_simplex(n: int, total: float) -> np.ndarray:
    """Draw a random point on the simplex of dimension n with L1 sum `total`."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / x.sum() * total

def _objective(seq: np.ndarray) -> float:
