def _project_simplex(v):
   # ensure non-negative and sum=1
   v = np.maximum(v, 0)
   s = v.sum()
   if s == 0:
       return np.full_like(v, 1.0/v.size)
   else:
       return v / s

"""
Hybrid shape optimiser for the step‑function problem.

The evaluation function is homogeneous, therefore we normalise every
candidate to have unit sum (the score then equals 2·n·max(conv)).
The algorithm works in three stages:

1. **Quick parametric sweep** – a few analytical families (constant,
   linear, quadratic, triangular, sinusoidal, exponential, power‑law)
   are evaluated for several lengths.  The best shape found is kept as a
   warm‑start.

2. **Projected gradient descent on a smooth max‑approximation** – we
   replace the non‑smooth max (conv) by a log‑sum‑exp with a temperature
   β.  For a given β the gradient is
       ∇a ≈ 2·(w ⋆ a_rev)   with   w_k ∝ exp(β·(conv_k−max(conv))),
   where `a_rev` is the reversed vector.  After each step we project
   back onto the probability simplex (non‑negative, unit sum).  β is
   increased gradually (annealing) while the learning rate is decayed.

3. **Fine‑grained stochastic polishing** – a short simulated‑annealing
   loop (Gaussian perturbations, block reversals) is run on the current
   best sequence to escape possible local minima.

The whole search respects the supplied time budget (default 1000 s) and
returns the best sequence discovered.  All numbers are non‑negative,
the sum stays well above the 0.01 guard, and the entries remain far
below the 1000 clipping limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _normalise_seq(arr: np.ndarray) -> np.ndarray:
    """Scale `arr` to have unit sum and stay safely below the 1000‑clip."""
    a = np.array(arr, dtype=float)
    if a.max() > 0:
        a = a * (0.9 / a.max())                # keep values < 1
    s = a.sum()
    if s < 0.01:
        a = a * (0.01 / s) if s > 0 else a + (0.01 / a.size)
    a /= a.sum()
    return a


def _candidate_shapes(n: int):
    """Analytic families of non‑negative shapes of length `n`."""
    cand = []

    # constant
    cand.append(np.ones(n, dtype=float))

    # linear
    cand.append(np.arange(n, 0, -1, dtype=float))
    cand.append(np.arange(1, n + 1, dtype=float))

    # quadratic
    cand.append(np.square(np.arange(n, 0, -1, dtype=float)))
    cand.append(np.square(np.arange(1, n + 1, dtype=float)))

    # triangular “mountain”
    t = np.arange(1, n + 1, dtype=float)
    cand.append(np.minimum(t, n - t + 1))

    # valley (large at ends, small in centre)
    centre = (n - 1) / 2.0
    cand.append(1.0 / (1.0 + (np.arange(n, dtype=float) - centre) ** 2))

    # sine‑squared
    cand.append(np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) /
                       (n + 1)) ** 2)

    # exponential decays
    for r in (0.95, 0.9, 0.8, 0.7, 0.6, 0.5):
        cand.append(r ** np.arange(n, dtype=float))

    # power‑law decays
    for p in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n + 1, dtype=float)
        cand.append(1.0 / (i ** p))

    return cand


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    # Ensure non‑negativity first
    v = np.maximum(v, 0.0)
    if v.sum() == 0.0:
        return np.full_like(v, 1.0 / v.size)
    # Simple scaling – good enough because all entries are already ≥0
    return v / v.sum()


def _smooth_gradient(a: np.ndarray, beta: float):
    """
    Return (grad, max_conv) where `grad` is the gradient of the
    log‑sum‑exp approximation of max(conv) at temperature `beta`,
    and `max_conv` = max(a * a).
    """
    conv = np.convolve(a, a)               # length 2·n‑1
    max_c = conv.max()
    # softmax weights (always ≤1, sum to 1)
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2·(w ⋆ a_rev) limited to the original length
    a_rev = a[::-1]
    tmp = np.convolve(w, a_rev)            # length 3·n‑2
    n = a.size
    grad = 2.0 * tmp[n - 1:n - 1 + n]      # slice the central n entries
    return grad, max_c

        # cheap evaluation using the true objective
        conv_new = np.convolve(new, new)
        cand_score = 2 * cur.size * conv_new.max()

        # acceptance test
        if cand_score < best_score:
            best_score = cand_score
            best_seq = new.tolist()
            cur = new
            print(f"[{time.time() - start_time:.1f}s] SA improved score={best_score:.6f}")
            if best_score <= 1.503:
                print("[target reached] early termination of SA")
                break
        else:
            prob = np.exp(-(cand_score - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur = new

    # ------------------------------------------------------------------
    # Final sanity check (evaluate via the provided function)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    # The direct score may be infinitesimally different because evaluate_sequence
    # applies clipping – we keep the sequence anyway.
    print(f"[{time.time() - start_time:.1f}s] Finished – final score={final_score:.6f}")

    return best_seq
