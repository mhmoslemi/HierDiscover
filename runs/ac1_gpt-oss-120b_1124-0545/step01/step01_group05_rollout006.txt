"""
Hill‑climbing for the step‑function evaluation problem.

The algorithm combines two complementary strategies:

1. **Targeted local moves** – we keep the current best sequence `a`
   (initialised from the provided `height_sequence_1` or from a uniform
   distribution) and repeatedly reduce the maximal convolution value.
   The shift `k*` where `conv = a * a` attains its maximum is found.
   All contributions to that shift come from pairs `(i, k*‑i)`.
   We pick the pair with the largest product `a[i]·a[k*‑i]` and move a
   small amount of mass from the larger index to a distant location.
   The convolution is updated *incrementally* (O(n) slice operations)
   instead of recomputing it from scratch, which makes millions of
   iterations feasible inside the 1000 s budget.

2. **Exploratory random families** – periodically a fresh candidate is
   drawn from a parametrised family (exponential, power‑law,
   two‑block, Dirichlet, mixture of exponentials, sparse random).  The
   fast surrogate score `2·n·max(conv)` is evaluated and the exact
   `evaluate_sequence` is called only when an improvement over the
   current best is observed.

The loop runs until the time budget expires (or the target score
`1.503` is reached).  A short final refinement with tiny random tweaks
is applied before returning the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full, length 2·len(a)‑1)."""
    return np.convolve(a, a)


def _fast_score_from_conv(conv: np.ndarray, n: int) -> float:
    """Fast surrogate of `evaluate_sequence` when sum(a)=1."""
    return 2.0 * n * conv.max()


def _make_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def _make_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = i ** p
    a /= a.sum()
    return a


def _make_two_block(n: int, m: int, high_frac: float) -> np.ndarray:
    if not (0 < m < n):
        raise ValueError("invalid block size")
    high = high_frac / m
    low = (1.0 - high_frac) / (n - m)
    a = np.empty(n, dtype=float)
    a[:m] = high
    a[m:] = low
    return a


def _make_mix_exponential(n: int, w: float, lam1: float, lam2: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = w * np.exp(-lam1 * i) + (1.0 - w) * np.exp(-lam2 * i)
    a /= a.sum()
    return a


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    a = np.random.dirichlet([alpha] * n)
    return a


def _make_sparse_random(n: int, k: int) -> np.ndarray:
    """k uniformly random locations receive i.i.d. positive weights."""
    a = np.zeros(n, dtype=float)
    idx = np.random.choice(n, k, replace=False)
    vals = np.random.rand(k) + 1e-6  # avoid exact zeros
    a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Incremental convolution update (used by the local hill‑climber)
# ----------------------------------------------------------------------


def _update_conv_incremental(conv: np.ndarray,
                             a: np.ndarray,
                             src: int,
                             dst: int,
                             delta: float) -> np.ndarray:
    """
    Return a new convolution array after moving `delta` mass from index
    `src` to index `dst`.  `a` is the *old* distribution (before the move).
    The update is performed with O(n) slice operations.
    """
    n = a.shape[0]
    conv_new = conv.copy()

    # removal of `delta` from src
    conv_new[src:src + n] -= 2.0 * delta * a

    # addition of `delta` to dst
    conv_new[dst:dst + n] += 2.0 * delta * a

    # quadratic corrections for the two diagonal terms
    conv_new[2 * src] += delta * delta
    conv_new[2 * dst] += delta * delta

    return conv_new


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence (list of floats) found before the time
    budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the outer environment
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # Initialise sequence (prefer the provided construction)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a = np.clip(np.array(height_sequence_1, dtype=float), 0.0, None)
        if a.sum() == 0.0:
            a = np.full(n, 1.0 / n, dtype=float)
        else:
            a = a / a.sum()
    else:
        n = 2000
        a = np.full(n, 1.0 / n, dtype=float)

    # Normalise exactly to sum = 1 (the score is scale‑invariant)
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n, dtype=float)
    else:
        a = a / a_sum

    # Full convolution and current best score
    conv = _conv_full(a)
    best_conv = conv.copy()
    best_score = _fast_score_from_conv(conv, n)
    best_a = a.copy()
