def _interpolate_sequence(seq, new_n):
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    # Use linear interpolation across [0, old_n-1] to new positions
    old_pos = np.arange(old_n) / (old_n - 1) if old_n > 1 else np.array([0.0])
    new_pos = np.arange(new_n) / (new_n - 1)
    new_seq = np.interp(new_pos, old_pos, seq)
    new_seq = np.maximum(new_seq, 0.0)
    total = new_seq.sum()
    if total > 0:
        new_seq /= total
    return new_seq

"""
Gradient‑descent on a smooth approximation of the max‐convolution.
====================================================================

The evaluation function is

    score = 2·n·max(conv(a,a)) / (sum(a)**2)          (a_i ≥ 0)

Scaling a does not change the score, therefore we keep the vector
normalised to sum = 1.  The continuous optimum a(x)∝1/√x yields a *flat*
self‑convolution of height ≈π/2≈1.571.  To beat this bound we minimise a
*smoothed* version of the max using a log‑sum‑exp (soft‑max) surrogate,
which gives a dense gradient that moves all entries at once.  The
gradient of the surrogate can be written as a correlation, which we
compute efficiently with FFTs.  After each gradient step we project onto
the feasible set:

* non‑negativity,
* monotone decreasing shape (isotonic regression, the continuous optimum
  is decreasing), and
* the affine hyperplane sum = 1.

A small amount of momentum and occasional random mass‑swaps help escape
flat regions.  The smoothing parameter is increased gradually so that the
algorithm first spreads the mass globally and later focuses on the true
maximum.  The routine runs for the allotted time (default 1000 s) and
returns the best sequence found.

The implementation uses only NumPy and the provided ``evaluate_sequence``
function and therefore respects the execution constraints.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _initial_root_sequence(n: int) -> np.ndarray:
    """Return a normalised sequence a_i ∝ 1/√(i+1) of length n."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def _fft_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of a and b using real FFT (fast for repeated calls)."""
    m = len(a) + len(b) - 1
    size = 1 << (m - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)
    return conv[:m]


def _fft_cross_correlation(w: np.ndarray, seq: np.ndarray) -> np.ndarray:
    """
    Valid cross‑correlation of w (length 2n‑1) with seq (length n).
    Result has length n and equals   corr[i] = Σ_j w[i+j]·seq[j].
    """
    n = len(seq)
    rev_seq = seq[::-1]
    m = len(w) + len(rev_seq) - 1
    size = 1 << (m - 1).bit_length()
    fw = np.fft.rfft(w, size)
    fseq = np.fft.rfft(rev_seq, size)
    conv = np.fft.irfft(fw * fseq, size)
    start = n - 1
    return conv[start:start + n]


def _isotonic_decreasing(x: np.ndarray) -> np.ndarray:
    """
    Projection of a vector onto the cone of non‑negative, monotone decreasing
    sequences (PAVA on -x).  Returns a new array of the same shape.
    """
    n = len(x)
    # work with y = -x, which must be non‑decreasing.
    y = -x.astype(np.float64).copy()
    # PAVA: maintain block averages and counts.
    avg = list(y)
    cnt = [1] * n
    i = 0
    while i < len(avg) - 1:
        if avg[i] > avg[i + 1] + 1e-12:          # violation of monotonicity
            # merge blocks i and i+1
            tot = cnt[i] + cnt[i + 1]
            merged = (avg[i] * cnt[i] + avg[i + 1] * cnt[i + 1]) / tot
            avg[i] = merged
            cnt[i] = tot
            del avg[i + 1]
            del cnt[i + 1]
            i = max(i - 1, 0)
        else:
            i += 1
    # expand the block averages back to a full vector
    out = np.empty(n, dtype=np.float64)
    pos = 0
    for a, c in zip(avg, cnt):
        out[pos:pos + c] = a
        pos += c
    # revert the sign (back to decreasing) and enforce non‑negativity
    out = -out
    out[out < 0] = 0.0
    return out


def _smooth_max_weights(conv: np.ndarray, sigma: float) -> np.ndarray:
    """
    Return a soft‑max weight vector w_i = exp((conv_i - max(conv))/sigma)
    normalised to sum = 1.  ``sigma`` controls the amount of smoothing:
    large sigma → uniform weights, small sigma → concentrates on the
    true maximum.
    """
    mx = conv.max()
    # subtract mx for numerical stability; sigma > 0
    diff = (conv - mx) / sigma
    # avoid overflow – the largest exponent is 0, the rest are ≤0
    w = np.exp(diff, dtype=np.float64)
    s = w.sum()
    if s == 0.0:                     # fallback (should not happen)
        w = np.ones_like(conv) / conv.size
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the supplied
    evaluation function.  The algorithm uses gradient descent on a
    smoothed maximum of the self‑convolution, isotonic projection,
    momentum and occasional random swaps.  It runs until the time budget
    expires and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin so the harness does not kill us
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1) Choose problem size and initial sequence
    # ------------------------------------------------------------------
    # Use an existing global sequence if present; otherwise start from the
    # classic 1/√x shape.
    try:
        # ``height_sequence_1`` may be defined in the surrounding environment.
        init = np.array(height_sequence_1, dtype=np.float64)
        n = len(init)
        if n < 2000:                     # protect against accidental tiny seeds
            raise ValueError
        seq = init.copy()
    except Exception:
        n = 5000                          # default length used by the reference code
        seq = _initial_root_sequence(n)

    # Normalise to sum = 1 (scale does not affect the score)
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total > 0:
        seq /= total
    else:
        seq = _initial_root_sequence(n)

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Parameters for the optimiser
    # ------------------------------------------------------------------
    step = 0.25                     # initial step size for the gradient move
    step_decay = 0.99996           # exponential decay (slow)
    min_step = 1e-6

    beta = 0.9                     # momentum coefficient
    velocity = np.zeros_like(seq)

    sigma = 1e-3                   # initial smoothing (large → smoother)
    sigma_min = 1e-8
    sigma_decay = 0.99995          # geometric decay towards a sharper max

    eval_every = 10                # evaluate the expensive objective only every N‑th iter
    swap_every = 200               # frequency of random mass swaps
    iteration = 0

    # ------------------------------------------------------------------
    # 3) Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ---- compute soft‑max weighted gradient -----------------------
        conv = _fft_convolution(seq, seq)                 # length 2n‑1
        w = _smooth_max_weights(conv, sigma)              # soft‑max weights, sum = 1
        grad = 2.0 * _fft_cross_correlation(w, seq)       # ∇ L_soft (size n)

        # ---- momentum + projected gradient step -----------------------
        velocity = beta * velocity - step * grad
        new_seq = seq + velocity

        # enforce non‑negativity
        new_seq = np.clip(new_seq, 0.0, None)

        # isotonic (monotone decreasing) projection
        new_seq = _isotonic_decreasing(new_seq)

        # re‑normalise to sum = 1 (scale invariance of the score)
        s = new_seq.sum()
        if s > 0:
            new_seq /= s
        else:
            new_seq = _initial_root_sequence(n)

        seq = new_seq

        # ---- schedule updates -----------------------------------------
        step = max(step * step_decay, min_step)
        sigma = max(sigma * sigma_decay, sigma_min)

        # ---- occasional random mass swaps (helps escape plateaus) -----
        if iteration % swap_every == 0:
            i, j = np.random.choice(n, 2, replace=False)
            # move a small fraction of mass from i to j
            eps = 0.03 * seq[i]               # 3 % of the source entry
            eps = min(eps, seq[i])
            seq[i] -= eps
            seq[j] += eps
            seq = np.clip(seq, 0.0, None)
            seq = _isotonic_decreasing(seq)
            seq /= seq.sum()                   # keep the normalisation

        # ---- occasional evaluation of the true objective -------------
        if iteration % eval_every == 0:
            cur_score = evaluate_sequence(seq.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                elapsed = time.time() - start_time
                print(f"[improve] iter={iteration:6d}, score={best_score:.6f}, "
                      f"sigma={sigma:.2e}, step={step:.2e}, time={elapsed:.1f}s")

    # ------------------------------------------------------------------
    # 4) Finish
    # ------------------------------------------------------------------
    print(f"[final] iterations={iteration}, best_score={best_score:.6f}")
    return best_seq.tolist()
