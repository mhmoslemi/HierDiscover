def _optimize_length(n, time_budget):
    import time, math, random, numpy as np

    target_sum = math.sqrt(2.0 * n)

    # Simple projection defined above.

    # generate initial candidates
    candidates = []

    uniform = np.full(n, target_sum / n)
    candidates.append(uniform)

    exp_seq = np.random.exponential(scale=1.0, size=n)
    exp_seq = exp_seq / exp_seq.sum() * target_sum
    candidates.append(exp_seq)

    if 'height_sequence_1' in globals():
        hs = np.array(height_sequence_1, dtype=float)
        # resize to n using linear interpolation if lengths differ
        if len(hs) != n:
            # simple linear interpolation
            xs = np.linspace(0, 1, len(hs))
            xs_new = np.linspace(0, 1, n)
            hs_resized = np.interp(xs_new, xs, hs)
            hs_resized = hs_resized / hs_resized.sum() * target_sum
            candidates.append(hs_resized)
        else:
            candidates.append(hs / hs.sum() * target_sum)

    # evaluate candidates and pick best as starting point
    best_local = None
    best_score_local = float('inf')
    for cand in candidates:
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score_local:
            best_score_local = sc
            best_local = cand.copy()

    # main optimization
    seq = best_local.copy()
    cur_score = best_score_local
    best_seq = seq.copy()
    best_score = cur_score

    start = time.time()
    deadline = start + time_budget
    it = 0
    no_improve = 0

    while time.time() < deadline:
        # compute convolution
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        # update best if improvement
        if max_val < best_score:
            best_score = max_val
            best_seq = seq.copy()
            print(f"[{it:6d}] n={n} New best {best_score:.6f} (time {time.time()-start:.1f}s)")
            no_improve = 0
        else:
            no_improve += 1

        # temperature and step schedule
        elapsed = time.time() - start
        frac = elapsed / time_budget
        # cooling schedule
        T = 1.0 * ((1e-4 / 1.0) ** frac)  # from 1 to 1e-4

        # number of top elements to consider
        top_k = max(3, int(30 * (1 - frac)))  # start with ~30, end with 3

        # get indices of top_k conv entries
        # use argpartition for efficiency
        if top_k >= len(conv):
            idxs = np.arange(len(conv))
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]

        # compute gradient
        grad = np.zeros_like(seq)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            idx_j = k - idx_i
            # accumulate
            grad[idx_i] += seq[idx_j]

        grad_norm = np.linalg.norm(grad) + 1e-12
        # step magnitude
        step_initial = target_sum * 0.05
        step_size = step_initial * (1.0 - frac) / grad_norm

        proposal = seq - step_size * grad

        # occasional random perturbation
        if random.random() < 0.005:
            proposal += np.random.normal(scale=target_sum * 0.01, size=n)

        # project onto simplex to keep sum and non-negativity
        proposal = _simplex_projection(proposal, target_sum)

        # evaluate proposal
        try:
            score = evaluate_sequence(proposal.tolist())
        except Exception:
            continue

        # acceptance (simulated annealing) based on current conv max (cur_score)
        if score < cur_score or random.random() < math.exp(-(score - cur_score) / (T + 1e-12)):

            seq = np.random.exponential(scale=1.0, size=n)
            seq = seq / seq.sum() * target_sum
            cur_score = evaluate_sequence(seq.tolist())
            no_improve = 0

        it += 1

    return best_seq, best_score

"""
Propose a step‑function height vector that minimises
    E(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)
where the sum is fixed to √(2n).  The program runs a projected
gradient–descent on the (smoothed) ℓ∞‑norm of the self‑convolution,
combined with a simple simulated‑annealing acceptance rule and occasional
random restarts.  Several problem sizes are tried (the previous best
length is always included) and the best sequence among them is returned.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex { w ≥ 0 , sum(w)=s } (Wang–Carreira‑Perpiñán)."""
    n = v.size
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(v.shape, s / n, dtype=v.dtype)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _optimize_length(n: int, time_budget: float) -> tuple[np.ndarray, float]:
    """
    Run the optimiser for a fixed length `n` for at most `time_budget` seconds.
    Returns the best found sequence and its evaluation (which equals max(conv)).
    """
    target_sum = math.sqrt(2.0 * n)

    # ---------- initial points ------------------------------------------------
    cand = []

    # uniform start
    uniform = np.full(n, target_sum / n, dtype=float)
    cand.append(uniform)

    # exponential start
    exp_seq = np.random.exponential(scale=1.0, size=n)
    exp_seq = exp_seq / exp_seq.sum() * target_sum
    cand.append(exp_seq)

    # Dirichlet (α=0.5) – fairly sparse
    dir_seq = np.random.dirichlet(alpha=np.full(n, 0.5))
    dir_seq = dir_seq * target_sum
    cand.append(dir_seq)

    # if a previous construction exists, resize it by linear interpolation
    if "height_sequence_1" in globals():
        old = np.asarray(height_sequence_1, dtype=float)
        if old.size != n:
            xs_old = np.linspace(0.0, 1.0, old.size)
            xs_new = np.linspace(0.0, 1.0, n)
            resized = np.interp(xs_new, xs_old, old)
            resized = resized / resized.sum() * target_sum
            cand.append(resized)
        else:
            cand.append(old / old.sum() * target_sum)

    # pick the best of the crude starts
    best_seq = None
    best_score = float("inf")
    for c in cand:
        scr = float(evaluate_sequence(c.tolist()))
        if scr < best_score:
            best_score = scr
            best_seq = c.copy()

    # ---------- main optimisation loop ---------------------------------------
    seq = best_seq.copy()
    cur_score = best_score
    best_seq_local = seq.copy()
    best_score_local = best_score

    start = time.time()
    deadline = start + time_budget
    it = 0
    no_improve = 0

    while time.time() < deadline:
        # current convolution and its maximum
        conv = np.convolve(seq, seq)
        max_val = conv.max()

        # record improvement
        if max_val < best_score_local:
            best_score_local = max_val
            best_seq_local = seq.copy()
            print(f"[{it:6d}] n={n:<4d} new best {best_score_local:.6f} (t={time.time()-start:.1f}s)")
            no_improve = 0
        else:
            no_improve += 1

        # ------- schedule ----------------------------------------------------
        elapsed = time.time() - start
        frac = elapsed / time_budget                     # 0 … 1
        # simulated‑annealing temperature (exponential cooling)
        T = 1.0 * ((1e-4 / 1.0) ** frac)

        # number of active convolution entries used to form the sub‑gradient
        top_k = max(3, int(30 * (1.0 - frac)))          # start with ~30, finish with 3

        # indices of the `top_k` largest convolution entries
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]

        # ------- sub‑gradient of the (smoothed) max -------------------------
        grad = np.zeros_like(seq)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            idx_j = k - idx_i
            grad[idx_i] += seq[idx_j]          # ∂conv[k]/∂a_i = a_{k‑i}

        grad_norm = np.linalg.norm(grad) + 1e-12

        # step size – large at the beginning, vanishing at the end
        step_initial = target_sum * 0.05                # absolute scale
        step = step_initial * (1.0 - frac) / grad_norm

        # propose a new point
        proposal = seq - step * grad

        # occasional random “kick” to escape plateaus
        if random.random() < 0.005:
            proposal += np.random.normal(scale=target_sum * 0.01, size=n)

        # enforce simplex constraints (non‑negative, correct ℓ1‑norm)
        proposal = _simplex_projection(proposal, target_sum)

        # evaluate the candidate (the evaluation equals max(conv) because sum is fixed)
        try:
            new_score = float(evaluate_sequence(proposal.tolist()))
        except Exception:
            continue

        # simulated‑annealing style acceptance
        if new_score < cur_score or random.random() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
            seq = proposal
            cur_score = new_score

        # restart if we have been stuck for a while
        if no_improve >= 2000:
            seq = np.random.exponential(scale=1.0, size=n)
            seq = seq / seq.sum() * target_sum
            cur_score = float(evaluate_sequence(seq.tolist()))
            no_improve = 0

        it += 1

    return best_seq_local, best_score_local


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function height vector that minimises the
    given evaluation function.  The routine tries several problem sizes,
    runs a projected gradient descent with a smoothed ℓ∞‑objective and
    returns the best sequence found before the time limit expires.
    """
    np.random.seed(seed)
    random.seed(seed)

    # problem sizes to explore – always include the length of the previous
    # construction (if any) and a few larger values.
    lengths = []
    if "height_sequence_1" in globals():
        lengths.append(len(height_sequence_1))
    # add a few larger candidates (still fast enough for the budget)
    lengths.extend([3000, 4000, 5000])
    lengths = sorted(set(lengths))

    # allocate the time budget uniformly over the chosen lengths
    total_budget = max(0.0, budget_s - 5.0)          # reserve a small safety margin
    per_len_budget = total_budget / len(lengths)

    best_overall_seq = None
    best_overall_score = float("inf")

    for n in lengths:
        print(f"--- Optimising length {n} (budget {per_len_budget:.1f}s) ---")
        seq, score = _optimize_length(n, per_len_budget)
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = seq.copy()
            print(f">>> New overall best: {best_overall_score:.6f} (n={n})")

    # fallback – should never happen, but keep the harness happy
    if best_overall_seq is None:
        n = lengths[0]
        best_overall_seq = np.full(n, math.sqrt(2.0 * n) / n)

    # return a plain Python list as required by the harness
    return best_overall_seq.tolist()
