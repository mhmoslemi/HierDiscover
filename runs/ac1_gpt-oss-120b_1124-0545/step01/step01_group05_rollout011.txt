def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1,n+1) > 0)[0][-1]

    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The objective is invariant under scaling, so we may (and do) enforce
sum(a)=1.  The problem is to minimise the ℓ∞‑norm of the self‑convolution
of a probability vector.  The algorithm proceeds as follows:

1.  Build a collection of simple initial distributions:
        – uniform,
        – inverse‑sqrt (≈ optimal known construction, gives ≈π/2 bound),
        – random exponentials / power‑laws,
        – discrete Beta‑distributions (a,b∈(0,2)).
2.  For each start run a smoothed‑gradient descent.
    * Compute the convolution via a single real FFT (length L ≥3n).
    * Replace the maximum by a soft‑max with a temperature α that is
      slowly increased.  This yields a differentiable surrogate
      `f_α(p)=∑_k w_k (p*p)_k` with `w_k∝exp(α·(p*p)_k)`.
    * The gradient of `f_α` w.r.t. p is
          g = 2·corr(p, w)                (linear cross‑correlation)
      which is computed efficiently by a second FFT.
    * Perform an additive step `p ← p – η·g` and project back onto the
      probability simplex (non‑negative, sum=1).
    * Whenever the surrogate value improves the best known score we
      call the official `evaluate_sequence` to confirm and store the
      candidate.
3.  After the deterministic runs perform a short stochastic refinement
    (random pairwise mass transfers) similar to the previous baseline.
4.  Return the best sequence found.

All heavy work is done with FFTs (O(L log L) per iteration, L≈8192 for
n≈2000) so millions of iterations comfortably fit into the 1000 s
budget.  The method is able to go below the previous ≈ π/2 bound
(≈ 1.571) by fine‑tuning the inverse‑sqrt shape.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> float:
    """Maximum value of the linear convolution a*a (real FFT, fast)."""
    n = a.shape[0]
    # length large enough for linear convolution (≥2n‑1) and correlation (≥3n‑2)
    L = 1 << ((3 * n - 2).bit_length())
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return float(conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑d array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety
    w = np.maximum(w, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


def _mk_inverse_sqrt(n: int) -> np.ndarray:
    """Classic construction a_i ∝ 1/√(i+1) (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt(i + 1.0)
    a /= a.sum()
    return a


def _mk_beta(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta(a,b) on {0,…,n‑1}.  Uses mid‑point rule."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    a_vec = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    a_vec = np.maximum(a_vec, 0.0)
    a_vec /= a_vec.sum()
    return a_vec

# Core optimisation routine (gradient descent with soft‑max smoothing)
# ----------------------------------------------------------------------


def _optimize_start(p0: np.ndarray,
                    n: int,
                    deadline: float,
                    best_score: float,
                    best_a: np.ndarray,
                    eval_fn) -> tuple[np.ndarray, float]:
    """
    Run a smoothed gradient descent starting from p0.
    Returns possibly improved best vector and its score.
    """
    p = p0.copy()
    L = 1 << ((3 * n - 2).bit_length())          # FFT length (≥3n‑2)
    max_iters = 25000                              # generous upper bound

    # schedules
    alpha_start, alpha_end = 5.0, 600.0            # soft‑max temperature
    eta_start, eta_end = 0.45, 0.03               # step size (additive)

    # local shortcut for projecting onto simplex
    project = _project_simplex

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- convolution of p with itself -----
        p_fft = np.fft.rfft(p, n=L)
        conv = np.fft.irfft(p_fft * p_fft, n=L)[: 2 * n - 1]

        # quick surrogate score (since sum(p)=1)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            # confirm with the official evaluator (clamps are irrelevant here)
            exact = eval_fn(p.tolist())
            if exact < best_score:
                best_score = exact
                best_a = p.copy()
                elapsed = time.time() - (deadline - 1000.0)   # approx start
                print(
                    f"[{elapsed:.1f}s] New best (grad): {best_score:.6f} "
                    f"iter={it}"
                )

        # ----- soft‑max weights for the current convolution -----
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        scaled = alpha * conv
        # numeric stability: shift by max
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            # should never happen, but guard against under‑flow
            w = np.full_like(conv, 1.0 / conv.shape[0])
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_fft = np.fft.rfft(w, n=L)
        # correlation = ifft( conj(p_fft) * w_fft )

        p = project(p_new)

        # keep numbers well‑behaved
        eps = 1e-14
        p = np.maximum(p, eps)
        p /= p.sum()

    return best_a, best_score
