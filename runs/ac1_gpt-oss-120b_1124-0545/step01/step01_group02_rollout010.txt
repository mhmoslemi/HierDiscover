```
def smooth_gradient_descent_step(seq, step, alpha):
    conv = np.convolve(seq, seq)
    # compute weights
    # To avoid overflow, subtract max.
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    weights = np.exp(scaled - max_scaled)
    weights /= np.sum(weights)
    grad = 2 * np.correlate(weights, seq, mode='valid')
    # Project gradient onto sum=0 hyperplane: grad -= np.mean(grad)
    grad -= np.mean(grad)
    new_seq = seq - step * grad
    new_seq = np.maximum(new_seq, 0.0)
    total = np.sum(new_seq)
    if total > 0:
        new_seq /= total
    else:
        new_seq = seq # fallback
    return new_seq
```

    """
    Gradient descent with a smoothed max objective and occasional hill‑climbing.
    Starts from the 1/√i discretisation, then repeatedly reduces the maximum
    entry of the self‑convolution while keeping the coefficients non‑negative
    and normalised.  A momentum term and an exponential weight (controlled by
    ``alpha``) give a smooth approximation of the nondifferentiable max.
    After the main descent phase a short simulated‑annealing style local‑search
    moves small amounts of mass between random coordinates.
    The best sequence found before the time budget expires is returned.
    """
    import numpy as np
    import time
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin

    # length: choose a compromise between accuracy and speed
    n = 8000                      # you can tune, 8000 < 10000 fits in memory
    # initial sequence a_i ∝ 1/√(i+1), normalised to sum = 1
    idx = np.arange(1, n+1, dtype=np.float64)

    # pre‑computed FFT sizes (next power of two)
    L_conv = 1 << ((2*n-1).bit_length())
    L_corr = 1 << ((3*n-2).bit_length())

    # current score

        fft = np.fft.rfft(s, n=L_conv)
        conv = np.fft.irfft(fft*fft, n=L_conv)[:2*n-1]
        return 2.0 * n * conv.max(), conv

    best_seq = seq.copy()
    best_score, _ = score_of(best_seq)
    print(f"[init] n={n}, score={best_score:.6f}")

    # parameters for the descent
    step = 0.5                 # learning rate
    alpha = 1.0                # smoothing parameter for the max
    beta = 0.9                 # momentum coefficient
    momentum = np.zeros_like(seq)

    iteration = 0
    # -------- gradient descent phase ----------
    while time.time() < deadline:
        iteration += 1

        # compute convolution and current score
        fft_seq = np.fft.rfft(seq, n=L_conv)
        conv = np.fft.irfft(fft_seq*fft_seq, n=L_conv)[:2*n-1]
        max_conv = conv.max()

        # update global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # --- smoothed gradient of max ---
        # exponential weights concentrate on indices where conv is large
        scaled = alpha * conv
        # subtract max to keep exponentials in a safe range

        # remove the component that would change the total sum (project to
        # the simplex hyperplane sum(seq)=1)
        grad -= grad.mean()

        # momentum update

        # gradient step

        seq_new = np.maximum(seq_new, 0.0)
        total = seq_new.sum()
        if total > 0:

        else:

        # but with a tiny probability to avoid getting stuck.
        # For simplicity we always move.

    # convert to plain Python list (the harness expects a list of floats)
    return best_seq.tolist()

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Smoothed max‑gradient descent with momentum + a short hill‑climbing phase.

    * Starts from the classic discretisation a_i ∝ 1/√(i+1) (sum normalised to 1).
    * In each iteration the self‑convolution is computed with an FFT (O(n log n)).
    * A soft‑max over the convolution entries concentrates the gradient on the
      largest values; the soft‑max temperature (``alpha``) is increased slowly so
      that the method first performs a global flattening and later focuses on the
      true maximum.
    * Momentum (``beta``) stabilises the noisy sub‑gradient while the step size
      decays.
    * Every few hundred steps a tiny random mass transfer is performed to
      escape flat regions.
    * When the time budget is almost exhausted a cheap hill‑climbing phase
      (random pairwise swaps) tries to polish the best solution found.
    * The best sequence discovered before the deadline is returned (as a plain
      ``list[float]``).  All coefficients are non‑negative and the sum is 1,
      which is the scale used by ``evaluate_sequence``.
    """
    import time
    import numpy as np

    # --------------------------------------------------------------
    # 1.  Basic setup
    # --------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # small safety margin

    # length of the step function – a compromise between quality and speed
    n = 8000

    # a_i ∝ 1/√(i+1), normalised to sum = 1
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = 1.0 / np.sqrt(idx)
    seq /= seq.sum()

    # FFT sizes (next power of two – best for NumPy’s FFT implementation)
    L_conv = 1 << ((2 * n - 1).bit_length())          # ≥ 2n‑1
    L_corr = 1 << ((3 * n - 2).bit_length())          # ≥ 3n‑2 (convolution + correlation)

    # --------------------------------------------------------------
    # 2.  Helper to compute the score quickly (no extra Python loops)
    # --------------------------------------------------------------
    def compute_score_and_conv(x: np.ndarray):
        """Return (score, full convolution) for a normalised sequence x."""
        # single FFT of the current vector (zero‑padded to L_corr)
        F = np.fft.rfft(x, n=L_corr)
        # linear convolution (real part) – only the first 2n‑1 entries are needed
        conv = np.fft.irfft(F * F, n=L_corr)[: 2 * n - 1]
        score = 2.0 * n * conv.max()
        return score, conv, F

    # keep the best solution seen so far
    best_seq = seq.copy()
    best_score, best_conv, _ = compute_score_and_conv(best_seq)
    print(f"[init] n={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Parameters for the smoothed gradient descent
    # --------------------------------------------------------------
    step = 0.5          # learning rate
    alpha = 1.0         # soft‑max temperature (will be increased)
    beta = 0.9          # momentum coefficient
    momentum = np.zeros_like(seq)

    iteration = 0

    # --------------------------------------------------------------
    # 4.  Main gradient‑descent loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- current convolution, score and FFT of seq -----
        cur_score, conv, F_seq = compute_score_and_conv(seq)

        # update global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            best_conv = conv.copy()
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # ----- soft‑max weights over convolution entries -----
        #   w_i ∝ exp(alpha * conv_i)  (shifted for numerical stability)
        scaled = alpha * conv
        shifted = scaled - scaled.max()
        w = np.exp(shifted)
        w /= w.sum()

        # ----- gradient of the smoothed max -----
        # correlation w ⋆ seq (valid part) via FFT:
        #   corr = ifft( fft(w) * conj(fft(seq)) )
        F_w = np.fft.rfft(w, n=L_corr)
        corr = np.fft.irfft(F_w * np.conj(F_seq), n=L_corr)
        # valid slice length = n, starts at index n‑1
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # project gradient onto the hyperplane sum(seq)=1  (i.e. zero‑mean)
        grad -= grad.mean()

        # ----- momentum update & gradient step -----
        momentum = beta * momentum + (1.0 - beta) * grad
        seq_new = seq - step * momentum
        seq_new = np.maximum(seq_new, 0.0)          # keep non‑negative
        total = seq_new.sum()
        if total > 0.0:
            seq_new /= total                        # renormalise to sum = 1
        else:                                       # extremely unlikely
            seq_new = seq.copy()

        seq = seq_new

        # ----- slowly adapt hyper‑parameters -----
        step = max(step * 0.9997, 1e-6)              # decay learning rate
        alpha = min(alpha * 1.0005, 50.0)            # sharpen soft‑max

        # occasional tiny random redistribution to break plateaus
        if iteration % 200 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = 0.01 * seq[i]                     # 1 % of the source mass
            eps = min(eps, seq[i])
            seq[i] -= eps
            seq[j] += eps
            seq /= seq.sum()

        if best_score < 1.503:

    # --------------------------------------------------------------
    # 5.  Short hill‑climbing / simulated‑annealing phase
    # --------------------------------------------------------------
    # Use the remaining time for cheap random pairwise swaps.
    while time.time() < deadline:

            best_seq = cand.copy()
            seq = cand                    # keep the improvement
            print(f"[hill] improve iter={iteration}, score={best_score:.6f}")

    print(f"[final] iterations={iteration}, best_score={best_score:.6f}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
