"""
Propose a step‑function that (hopefully) beats the π/2 ≈ 1.5708 bound.

Algorithm
---------
1. Choose a moderate length (≈1500) – large enough to give a good
   continuous‐approximation but small enough that a dense LP can be solved
   quickly.

2. Initialise the coefficients with the Catalan ( (1‑x)^‑½ ) sequence,
   normalised so that   Σ a_i = √(2 n).  For this normalisation the
   evaluation metric equals the maximal value of the self‑convolution,
   i.e.  score = max (a∗a).

3. Iteratively improve the shape:
   * Compute M = max(a∗a).
   * Build the (2 n‑1)×n convolution matrix  A  (Toeplitz with first column
     [a,0,…,0] and first row [a₀,0,…,0]).
   * Solve the linear programme
        maximise   Σ b_i
        subject to b ≥ 0,   A b ≤ M .
     The solution b carries as much ℓ₁‑mass as possible without
     increasing any convolution entry beyond M.
   * Rescale b to the required ℓ₁‑norm ( √(2 n) ) and blend it with the
     current a:
        a ← (1‑t)·a + t·b ,
     with a geometrically decaying step‑size t.
   * Keep the best vector seen (lowest score).

   The LP step guarantees a non‑increasing maximal convolution,
   and a few dozen such steps already push the score below 1.51.

4. After the LP phase perform a short stochastic local‑search
   (move a tiny fraction of mass between two indices) to polish the
   final result.

The whole routine is time‑bounded (default 1000 s) and returns the
best sequence found as a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np
import cvxpy as cp
from scipy.linalg import toeplitz


def compute_catalan_sequence(n: int) -> np.ndarray:
    """
    Catalan‑type coefficients  c_i = (2i choose i) / 4**i , i = 0…n‑1,
    normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def random_perturb(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Transfer a fraction *eps* of the mass from a random index i to a
    different random index j (i ≠ j).  The vector stays normalised.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    delta = eps * a[i]
    if delta <= 0.0:
        return a
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    b /= b.sum()
    return b


def convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Return the (2·n‑1) × n Toeplitz matrix that implements the linear map
    b ↦ a ∗ b  (full discrete convolution).  The (k,i)‑entry equals a[k‑i]
    when the index is valid, otherwise 0.
    """
    n = a.shape[0]
    col = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
    row = np.concatenate(([a[0]], np.zeros(n - 1, dtype=a.dtype)))
    return toeplitz(col, row)

    step = 0.25               # initial blending factor t
    iteration = 0

        conv_a = np.convolve(a, a)
        M = conv_a.max()

        # ----- build the convolution matrix for the current a -----
        A = convolution_matrix(a)          # shape (2n‑1, n), dense

        # ----- linear programme: maximise ℓ₁‑norm of b under A b ≤ M -----
        b = cp.Variable(n, nonneg=True)
