def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np, scipy.optimize, scipy.sparse as sp
    np.random.seed(seed)

    deadline = time.time() + budget_s - 5.0

    n = 2000
    a = binomial_sequence(n)   # as defined earlier
    # normalize
    target_sum = np.sqrt(2.0*n)
    a = a * (target_sum / a.sum())
    def max_conv(arr):
        size = 1 << ((len(arr)*2 - 1).bit_length())
        fft = np.fft.rfft(arr, n=size)
        conv = np.fft.irfft(fft * fft, n=size)[: 2*len(arr)-1]
        return conv.max()
    M = max_conv(a)
    best_seq = a.copy()
    best_val = M
    print(f"[init] n={n}, M={M:.6f}")

    # keep list of line search steps
    t_vals = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # Build convolution matrix A (f * b <= M)
        # Use sparse diagonal construction
        offsets = np.arange(n, dtype=int)
        data = [a for _ in range(n)]   # each is same reference but fine
        A = sp.diags(data, offsets, shape=(2*n-1, n), format='csc')
        # LP
        c = -np.ones(n)
        bounds = [(0, None)] * n
        ub = np.full(2*n-1, M, dtype=np.float64)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
        if not res.success:
            print("[lp] iteration", iteration, "LP failed", res.message)
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= target_sum * 1.000001:
            print(f"[lp] iteration {iteration} no sum increase, sum_b={sum_b:.6f}")
            break
        # scale to target sum
        scale = target_sum / sum_b
        g = b * scale
        # line search for mixing weight t
        improved = False
        for t in t_vals:
            f_new = (1 - t) * a + t * g
            M_new = max_conv(f_new)
            if M_new < M - 1e-9:
                a = f_new
                M = M_new
                if M < best_val:
                    best_val = M
                    best_seq = a.copy()
                print(f"[mix] iter={iteration}, t={t}, M_new={M:.6f}")
                improved = True
                break
        if not improved:
            print(f"[mix] iteration {iteration} no improvement with line search, break")
            break
        # optional: if M already below target threshold break
        if M < 1.503:
            print("[target] reached target")
            break

    # Final local hill climb (small random moves)
    attempts = 0
    max_attempts = 200_000
    while time.time() < deadline and attempts < max_attempts:
        attempts += 1
        i = np.random.randint(n)
        if best_seq[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = 0.02 * best_seq[i]  # 2% move
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0:
            continue
        M_new = max_conv(new_seq)
        if M_new < best_val:
            best_val = M_new
            best_seq = new_seq
            print(f"[hill] iter={attempts}, new_M={best_val:.6f}")

    # final official evaluation
    final_list = best_seq.tolist()
    try:
        official = evaluate_sequence(final_list)
        print("[final] official evaluate_sequence =", official)
    except Exception as e:
        print("[final] evaluation error:", e)
    return final_list

"""
search for a step‑function with a low evaluation score.

The algorithm starts from the classic truncated binomial (central
coefficients) sequence, which already gives a score ≈ π/2 ≈ 1.571.
A small linear‑programming (LP) sub‑routine is then used to find a
new sequence `b` that maximises the L1‑norm while keeping the
convolution `f * b` bounded by the current maximal auto‑convolution
value `M`.  After rescaling `b` to the required L1‑norm we line‑search
a convex blend `f←(1‑t)f + t b`.  The blend strictly reduces `M` in
most cases, after which the process repeats.  When the LP stops
producing progress we finish with a lightweight stochastic hill‑climb
that moves a tiny amount of mass between two indices whenever it
lowers the maximal convolution.
The routine respects the given time budget and returns the best
sequence found (as a plain Python list) before the deadline.
"""

import time
import numpy as np

# optional SciPy imports – if unavailable we simply skip the LP stage
try:
    from scipy.optimize import linprog
    from scipy import sparse as sp
except Exception:                     # pragma: no cover
    linprog = None
    sp = None


def binomial_sequence(length: int) -> np.ndarray:
    """Return a truncated central‑binomial sequence a[i] = C(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2.0 * i - 1) / (2.0 * i)
    return a


def max_convolution(u: np.ndarray, v: np.ndarray) -> float:
    """Maximum value of the (real) convolution of two non‑negative vectors."""
    n = len(u)
    m = len(v)
    size = 1 << ((n + m - 2).bit_length())
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(fu * fv, n=size)[: n + m - 1]
    return float(conv.max())


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # small safety margin

    # ------------------------------------------------------------------
    # 1) initialise with the classic binomial construction
    # ------------------------------------------------------------------
    n = 2000                         # size – moderate to keep LP fast
    a = binomial_sequence(n)
    target_sum = np.sqrt(2.0 * n)    # required L1‑norm
    a = a * (target_sum / a.sum())  # normalise
    M = max_convolution(a, a)        # current maximal auto‑convolution
    best_seq = a.copy()
    best_val = M
    print(f"[init] n={n}, M={M:.6f}")

    # ------------------------------------------------------------------
    # 2) iterative LP‑based improvement (if SciPy is available)
    # ------------------------------------------------------------------
    if linprog is not None and sp is not None:
        # line‑search coefficients for the convex blend
        t_candidates = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        iteration = 0
        while time.time() < deadline:
            iteration += 1

            # Build the convolution matrix A such that (A @ b)_k = (a * b)[k]
            offsets = np.arange(n, dtype=int)
            # each column i carries the vector a shifted down by i rows
            data = [a.copy() for _ in range(n)]
            A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")

            # Linear program: maximise sum(b)  ⇔ minimise -sum(b)
            c = -np.ones(n, dtype=np.float64)
            bounds = [(0, None)] * n
            ub = np.full(2 * n - 1, M, dtype=np.float64)

            lp_res = linprog(
                c,
                A_ub=A,
                b_ub=ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": max(0, deadline - time.time())},
            )

            if not lp_res.success:
                print(f"[lp] iteration {iteration} – LP failed: {lp_res.message}")
                break

            b = lp_res.x
            sum_b = b.sum()

            # If we cannot increase the L1‑norm, stop the LP phase
            if sum_b <= target_sum * (1.0 + 1e-9):
                print(f"[lp] iteration {iteration} – no L1‑gain (sum_b={sum_b:.6f})")
                break

            # Rescale b to keep the L1‑norm fixed (required for evaluation)
            scale = target_sum / sum_b
            g = b * scale

            # --------------------------------------------------------------
            # Blend f←(1‑t)f + t g – try a few blend weights
            # --------------------------------------------------------------
            improved = False
            for t in t_candidates:
                f_new = (1.0 - t) * a + t * g
                M_new = max_convolution(f_new, f_new)
                if M_new < M - 1e-9:
                    a = f_new
                    M = M_new
                    if M < best_val:
                        best_val = M
                        best_seq = a.copy()
                    print(f"[blend] it={iteration}, t={t:.3f}, M={M:.6f}")
                    improved = True
                    break
            if not improved:
                print(f"[blend] iteration {iteration} – no improvement, stop LP loop")
                break

            # early exit if we already beat the target
            if M < 1.503:
                print("[target] achieved required bound")
                break
    else:
        print("[info] SciPy not available – skipping LP stage")

    # ------------------------------------------------------------------
    # 3) final stochastic hill‑climb (tiny mass shifts)
    # ------------------------------------------------------------------
    attempts = 0
    max_attempts = 250_000
    while time.time() < deadline and attempts < max_attempts:
        attempts += 1
        i = np.random.randint(n)
        if best_seq[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move a modest fraction of mass from i to j
        delta = 0.02 * best_seq[i]            # 2 % move
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue

        M_new = max_convolution(new_seq, new_seq)
        if M_new < best_val - 1e-9:
            best_val = M_new
            best_seq = new_seq
            print(f"[hill] iter={attempts}, M={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) return the best candidate (list of Python floats)
    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    try:
        official = evaluate_sequence(final_list)  # imported by the framework
        print(f"[final] official evaluate_sequence = {official:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluation failed:", exc)

    return final_list
