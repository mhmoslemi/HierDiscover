def project_onto_simplex(v):
    # Duchi algorithm: project onto {x >=0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
optimise_step_sequence.py

Optimises a non‑negative step function (list of heights) for the
`evaluate_sequence` metric.

The algorithm works in three phases:

1. **Initialisation** – start from the Catalan (arcsine) distribution,
   which gives a flat self‑convolution in the infinite limit.

2. **Smooth‑max gradient descent** – minimise the maximum of the
   self‑convolution while keeping the coefficients on the probability
   simplex (non‑negative, sum = 1).  The true max is approximated by a
   soft‑max (log‑sum‑exp) with a temperature β that is increased
   gradually.  Gradients are obtained analytically and evaluated with
   FFTs for speed.  An Adam optimiser is used and the iterates are
   projected onto the simplex after each step.

3. **Fine stochastic local search** – short simulated‑annealing style
   perturbations to escape shallow plateaus and possibly improve the
   solution further.

The whole routine is time‑boxed (default 1000 s).  The best sequence
found before the deadline is returned as a plain Python list.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def compute_catalan_sequence(n: int) -> np.ndarray:
    """
    Return the first *n* Catalan‑type coefficients
        c_i = (2i choose i) / 4**i   (i = 0 … n‑1)
    normalised to sum to one.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def conv_linear_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear (full) convolution of real vectors x and y using FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    prod = fx * fy
    conv = np.fft.irfft(prod, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    { w >= 0, sum w = 1 } (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def random_perturb(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Transfer a fraction *eps* of the mass at a random index to another
    random index (i ≠ j).  The vector stays normalised.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    delta = eps * a[i]
    if delta <= 0.0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Produce a non‑negative step‑function (list of heights) that tries to
    minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Determine the working length
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # use length of the previous best if present
    except NameError:
        n = 2500                     # a reasonably large size

    # --------------------------------------------------------------
    # 0.  Initialise with the Catalan (arcsine) distribution
    # --------------------------------------------------------------
    a = compute_catalan_sequence(n)    # already normalised to sum = 1
    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())
    print(f"[init] score = {best_score:.6f}")

    # pre‑compute FFT lengths (power of two, a little larger than needed)
    fft_len1 = 1 << ((2 * n - 1).bit_length())      # for a*a
    fft_len2 = 1 << ((3 * n - 2).bit_length())      # for w * a_rev

    # --------------------------------------------------------------
    # 1.  Smooth‑max gradient descent (Adam optimiser)
    # --------------------------------------------------------------
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05                     # base learning rate
    it = 0

    # smoothing parameter for the soft‑max approximation of max(conv)
    beta_smooth = 2.0
    beta_target = 100.0
    # schedule for beta – linear increase over a large number of steps
    smooth_steps = 200_000

    while time.time() < deadline:
        # ---- compute convolution a*a ---------------------------------
        conv = conv_linear_fft(a, a, fft_len1)          # shape (2n-1,)
        max_conv = conv.max()

        # ---- soft‑max weights ----------------------------------------
        # w_k ∝ exp(β (conv_k - max_conv))  (ensures numerical stability)
        w = np.exp(beta_smooth * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth‑max -------------------------------
        # grad_i = 2 * Σ_k w_k * a_{k-i}
        grad_full = conv_linear_fft(w, a[::-1], fft_len2)   # length 3n-2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ---- Adam update ---------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- projection onto probability simplex -----------------------
        a = project_onto_simplex(a)

        # ---- occasional real evaluation --------------------------------
        if it % 50 == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[adam] iter {it:6d} – improved score {best_score:.6f}")

        # ---- increase smoothing parameter gradually --------------------
        if it < smooth_steps:
            beta_smooth = 2.0 + (beta_target - 2.0) * (it / smooth_steps)
        else:
            beta_smooth = beta_target

        # ---- tiny random jump every few thousand iterations -----------
        if it % 2000 == 0:
            a = random_perturb(a, eps=0.03)

    # --------------------------------------------------------------
    # 2.  Short stochastic local‑search / simulated‑annealing phase
    # --------------------------------------------------------------
    eps = 0.12
    temperature = 0.12
    it_ls = 0
    while time.time() < deadline:
        cand = random_perturb(best_seq, eps)
        cand_score = evaluate_sequence(cand.tolist())

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[pert] LS iter {it_ls:6d} – improved score {best_score:.6f}")
        else:
            # uphill move with decreasing temperature
            prob = np.exp(-(cand_score - best_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                best_seq = cand

        # decay parameters
        eps = max(eps * 0.9999, 1e-4)
        temperature *= 0.9997
        it_ls += 1

    # --------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # --------------------------------------------------------------
    final_seq = np.clip(best_seq, 0.0, 1000.0)   # safeguard against accidental overflow
    print(f"[final] best score = {best_score:.6f}  length = {n}")
    return final_seq.tolist()
