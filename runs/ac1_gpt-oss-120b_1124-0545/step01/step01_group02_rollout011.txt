def conv_max(seq):
    n = len(seq)
    m = 1 << (2*n-1).bit_length()
    # compute using FFT
    f = np.fft.rfft(seq, m)
    conv = np.fft.irfft(f * f, m)
    max_val = conv[:2*n-1].max()
    return max_val

score = 2*n * max_val # because sum=1.

"""
A two‑stage optimiser for the step‑function evaluation problem.

Stage 1 – cheap parametrised scan:
    Build candidate sequences a_i ∝ (i + c + 1)^‑p  (p≈0.5, c≥0) and keep the
    best one according to the analytical score 2·n·max(a*a) (the true
    evaluate_sequence() is scale‑invariant).  A coarse grid is scanned and
    then refined locally.

Stage 2 – constrained sub‑gradient / simulated‑annealing search:
    Starting from the best parametrised sequence we perform many iterations.
    In each iteration we either
        * take a sub‑gradient step that reduces the currently maximal
          convolution entry (projected onto the simplex), or
        * propose a random mass‑transfer move between two indices.
    Improvements are always accepted; worsening moves may be accepted with a
    temperature that decays over time (simulated annealing).  The step size
    and temperature also decay, allowing fine‑grained convergence.

The algorithm respects all security checks (non‑negative, finite numbers) and
fits well inside the 1000 s budget.
"""

import time
import numpy as np

# evaluate_sequence is assumed to be available in the execution environment


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(seq: np.ndarray, fft_len: int) -> float:
    """Return max entry of the linear convolution seq*seq using FFT."""
    # real FFT is enough – the input is real.
    f = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(f * f, fft_len)
    # convolution length is 2·len(seq)-1
    n = seq.size
    return conv[: 2 * n - 1].max()


def _score_fast(seq: np.ndarray, fft_len: int) -> float:
    """Fast surrogate for evaluate_sequence (scale‑invariant)."""
    max_c = _conv_max(seq, fft_len)
    return 2.0 * seq.size * max_c


def _project_nonneg_normalise(x: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    y = np.maximum(x, 0.0)
    s = y.sum()
    if s == 0.0:
        # fallback to a uniform distribution
        return np.full_like(y, 1.0 / y.size)
    return y / s


def _subgradient_step(seq: np.ndarray, step: float, fft_len: int) -> np.ndarray:
    """One sub‑gradient step w.r.t. the current maximal convolution entry."""
    n = seq.size
    # compute full convolution and locate the active index
    f = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(f * f, fft_len)[: 2 * n - 1]
    k_max = int(np.argmax(conv))

    # gradient of (a*a)[k_max] w.r.t. a_i is a_{k_max-i} (if inside the support)
    grad = np.zeros_like(seq)
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad[idx] = seq[k_max - idx]

    # move opposite to the gradient and stay on the simplex (sum = 1)
    cand = seq - step * grad
    cand = _project_nonneg_normalise(cand)
    return cand


def _random_transfer(seq: np.ndarray, step: float) -> np.ndarray:
    """Transfer a small amount of mass from one index to another."""
    n = seq.size
    i, j = np.random.choice(n, 2, replace=False)
    eps = step * seq[i]
    eps = min(eps, seq[i])
    cand = seq.copy()
    cand[i] -= eps
    cand[j] += eps
    cand = _project_nonneg_normalise(cand)
    return cand


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    supplied evaluate_sequence() function.  Returns the best list found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # a few seconds safety margin

    # --------------------------------------------------------------
    # 0. Basic configuration
    # --------------------------------------------------------------
    n = 5000                                 # length of the sequence
    fft_len = _next_pow2(2 * n - 1)           # size for FFT‑based convolutions

    # --------------------------------------------------------------
    # 1. Coarse parameter scan (p, offset)
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    best_params = None

    # Coarse grid
    p_vals = np.linspace(0.48, 0.52, 9)       # around the theoretical 0.5
    c_vals = np.arange(0, 401, 20)           # offset in [0, 400]

    for p in p_vals:
        for c in c_vals:
            idx = np.arange(n, dtype=float) + c + 1.0
            seq = idx ** (-p)
            seq = seq / seq.sum()            # normalise to L1 = 1
            sc = _score_fast(seq, fft_len)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                best_params = (p, c)
    print(f"[grid coarse] best p={best_params[0]:.4f}, c={best_params[1]}, score={best_score:.6f}")

    # Refine around the best coarse point
    p_center, c_center = best_params
    p_vals_fine = np.linspace(p_center - 0.01, p_center + 0.01, 11)
    c_vals_fine = np.arange(max(0, c_center - 30), c_center + 31, 2)

    for p in p_vals_fine:
        for c in c_vals_fine:
            idx = np.arange(n, dtype=float) + c + 1.0
            seq = idx ** (-p)
            seq = seq / seq.sum()
            sc = _score_fast(seq, fft_len)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                best_params = (p, c)
    print(f"[grid fine]   best p={best_params[0]:.5f}, c={best_params[1]}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Iterative local optimisation (sub‑gradient + annealing)
    # --------------------------------------------------------------
    seq = best_seq.copy()
    cur_score = best_score

    step = 0.12               # magnitude of mass transfer / gradient step
    step_decay = 0.99994
    temperature = 1e-3
    temp_decay = 0.9995

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # Choose the move type
        if iteration % 7 == 0:
            # deterministic sub‑gradient move
            candidate = _subgradient_step(seq, step, fft_len)
        else:
            # stochastic mass‑transfer move
            candidate = _random_transfer(seq, step)

        # Fast evaluation of the candidate
        cand_score = _score_fast(candidate, fft_len)

        # Acceptance test (simulated annealing)
        if cand_score < cur_score:
            seq = candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = candidate.copy()
                print(f"[improve] iter={iteration}, score={best_score:.6f}")
        else:
            # possibly accept a worsening move
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                seq = candidate
                cur_score = cand_score

        # decay step size and temperature
        step = max(step * step_decay, 5e-6)
        temperature = max(temperature * temp_decay, 1e-9)

    print(f"[final] iterations={iteration}, best_score={best_score:.6f}")
    # Convert to plain Python list for the harness
    return best_seq.tolist()
