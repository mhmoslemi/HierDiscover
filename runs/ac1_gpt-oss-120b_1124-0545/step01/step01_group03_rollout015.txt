    n = 4000
    a = binomial_sequence(n)   # initial
    best_seq = a.copy()
    best_val = fast_score(a)

        conv = np.fft.irfft(np.fft.rfft(a, n = size) * np.fft.rfft(a, n=size), n=size)[: 2*n-1]
        M = conv.max()
        if M <= 0: break
        t = conv.argmax()
        # compute gradient contributions from source side
        grad = np.zeros_like(a)
        # indices i where 0 <= t-i < n
        low_i = max(0, t - (n-1))
        high_i = min(t, n-1)
        # for i in [low_i, high_i], grad[i] = a[t-i]  (or 2*a[t-i])
        idx = np.arange(low_i, high_i+1)
        grad[idx] = a[t - idx]   # shape matches
        # Since derivative *2 magnitude factor not needed; we can treat as grad.
        # choose source
        # compute contribution = grad * a (like each index contribution to M)
        contrib = grad * a
        # pick source i as max contrib (maybe top)
        source_candidates = np.where(contrib > 0)[0]
        if len(source_candidates) == 0:
            break
        i = source_candidates[np.argmax(contrib[source_candidates])]
        # choose target j > t (or random in later half)
        if t < n-1:
            possible_targets = np.arange(t+1, n)
        else:
            possible_targets = np.arange(n//2, n) # if t near end, choose earlier?
        j = np.random.choice(possible_targets)
        # transfer delta
        delta = 0.1 * a[i]   # maybe 10% (or adapt)
        if delta < 1e-12:
            break
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # enforce non-negativity
        if new_a[i] < 0:
            new_a[i] = 0.0
        score = fast_score(new_a)
        if score < best_val:
            best_val = score
            best_seq = new_a.copy()
            a = new_a
            print(f"[improve] M={M:.6f} new_score={best_val:.6f}")
        else:
            # optional: try smaller delta or revert

"""
search_step_function.py

A lightweight optimizer that looks for a step‑function sequence with a low
evaluation score.  The approach consists of three phases:

1.  Initialise a baseline (the classic binomial construction).
2.  Sweep a two‑parameter “power‑law” family a_i = (i+α)^‑p and keep the
    best candidate found by the fast FFT based scorer.
3.  Perform a small directed hill‑climb: repeatedly shift a tiny amount of
    mass from a point that contributes most to the current maximal
    convolution value towards a point that does not influence that maximum.
    Only moves that strictly improve the score are kept.

The routine respects the given time budget (default 1000 s) and finally
returns the best sequence as a plain Python list.  All numbers are
non‑negative and the official ``evaluate_sequence`` is called once at the end
to obtain the official score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: binomial (central) sequence a_i = C(2i,i)/4**i.
# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """Return the truncated binomial sequence of given length."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i-1] * (2*i-1) / (2*i)
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Helper: power‑law family a_i = (i+α)^‑p   (α>0, p>0).
# ----------------------------------------------------------------------
def powerlaw_sequence(length: int, alpha: float, p: float) -> np.ndarray:
    """Return a power‑law sequence (i+α)^-p of the given length."""
    idx = np.arange(length, dtype=np.float64) + alpha
    return idx ** (-p)


# ----------------------------------------------------------------------
# Fast scorer (identical to evaluate_sequence but without security checks).
# ----------------------------------------------------------------------
def fast_score(seq: np.ndarray) -> float:
    """Compute the official objective using an FFT‑based convolution."""
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    max_b = float(conv.max())
    sum_a = float(seq.sum())
    if sum_a < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Compute full linear convolution of a real vector with itself (FFT version).
# ----------------------------------------------------------------------
def full_self_conv(seq: np.ndarray, fft_size: int) -> np.ndarray:
    """Return the linear convolution seq*seq (length 2*len(seq)-1)."""
    f = np.fft.rfft(seq, n=fft_size)
    conv = np.fft.irfft(f * f, n=fft_size)[: 2 * len(seq) - 1]
    return conv


# ----------------------------------------------------------------------
# Main entry point required by the framework.
# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
):
    """
    Search for a step‑function sequence that minimises ``evaluate_sequence``.
    Returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # --------------------------------------------------------------
    # 1) Baseline construction (binomial)
    # --------------------------------------------------------------
    n = 4000                         # length; large enough but still fast
    a_bin = binomial_sequence(n)
    best_seq = a_bin.copy()
    best_val = fast_score(best_seq)
    print(f"[init] baseline (binomial) score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2) Grid search over the power‑law family a_i = (i+α)^‑p
    # --------------------------------------------------------------
    # Parameter ranges – chosen to explore a wide variety of shapes.
    alphas = np.logspace(-0.5, 2.0, num=30)          # 0.3 … 100
    ps = np.linspace(0.52, 0.90, num=39)            # 0.52 … 0.90

    for alpha in alphas:
        if time.time() > deadline:
            break
        # pre‑compute the shifted index array once per α
        idx = np.arange(n, dtype=np.float64) + alpha
        for p in ps:
            # construct candidate
            a = idx ** (-p)
            val = fast_score(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[grid] α={alpha:.3g}, p={p:.3f} → score={val:.6f}")
            # time check
            if time.time() > deadline:
                break

    # --------------------------------------------------------------
    # 3) Optional linear blend between baseline and the best power‑law.
    #    Simple sweep over mixing coefficient λ∈[0,1].
    # --------------------------------------------------------------
    for lam in np.linspace(0.0, 1.0, 11):
        if time.time() > deadline:
            break
        mixed = (1.0 - lam) * a_bin + lam * best_seq
        val = fast_score(mixed)
        if val < best_val:
            best_val = val
            best_seq = mixed.copy()
            print(f"[mix] λ={lam:.2f} → score={val:.6f}")

    # --------------------------------------------------------------
    # 4) Light directed hill‑climb.
    #    Each iteration moves a small fraction of mass from a "high‑impact"
    #    index to a "low‑impact" index, measured w.r.t. the current
    #    maximal convolution entry.
    # --------------------------------------------------------------
    a = best_seq.copy()
    fft_size = 1 << ((2 * n - 1).bit_length())
    attempts = 0
    max_attempts = 200_000
    # step size starts at 5 % of the source entry and decays slowly
    while time.time() < deadline and attempts < max_attempts:
        attempts += 1

        # compute convolution and locate the active index t
        conv = full_self_conv(a, fft_size)
        t = int(conv.argmax())
        # gradient of conv[t] w.r.t. each entry a[i] is 2*a[t-i] (if in‑range)
        grad = np.zeros_like(a)
        i_min = max(0, t - (n - 1))
        i_max = min(t, n - 1)
        indices = np.arange(i_min, i_max + 1, dtype=np.int64)
        grad[indices] = a[t - indices]                     # factor 2 omitted (common)

        # source candidates – large a[i] * grad[i] are most responsible for M
        impact = grad * a
        src = int(np.argmax(impact))

        # choose a target that does NOT affect the current maximal entry.
        # indices > t have zero gradient; if none exist, pick the minimal grad.
        if t + 1 < n:
            possible_targets = np.arange(t + 1, n, dtype=np.int64)
        else:
            # fallback: any index with the smallest gradient contribution
            possible_targets = np.where(grad == grad.min())[0]
        if possible_targets.size == 0:
            continue
        tgt = int(np.random.choice(possible_targets))

        # move a modest fraction of mass from src to tgt
        step_frac = 0.03                     # 3 % of the source value
        delta = step_frac * a[src]
        if delta <= 0.0:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[tgt] += delta
        # enforce non‑negativity (numerical safety)
        if new_a[src] < 0.0:
            continue

        new_val = fast_score(new_a)
        if new_val < best_val:
            best_val = new_val
            best_seq = new_a.copy()
            a = new_a
            print(f"[climb] iter={attempts}, score={new_val:.6f}")

        # gradually shrink the step fraction to allow fine tuning
        if attempts % 5000 == 0 and step_frac > 0.005:
            step_frac *= 0.9

    # --------------------------------------------------------------
    # 5) Final validation with the official evaluator.
    # --------------------------------------------------------------
    final_list = best_seq.tolist()
    try:
        official = evaluate_sequence(final_list)   # the provided checker
        print(f"[final] official evaluate_sequence = {official:.6f}")
    except Exception as exc:  # pragma: no cover – defensive
        print("[final] evaluation failed:", exc)

    return final_list
