def _candidate_shapes_extended(n):
    shapes = []
    # constant
    shapes.append(np.ones(n, dtype=float))
    # linear up
    shapes.append(np.arange(1, n+1, dtype=float))
    # linear down
    shapes.append(np.arange(n, 0, -1, dtype=float))
    # quadratic up
    shapes.append(np.square(np.arange(1,n+1,dtype=float)))
    # quadratic down
    shapes.append(np.square(np.arange(n,0,-1,dtype=float)))
    # triangular
    t = np.arange(1, n+1, dtype=float)
    shapes.append(np.minimum(t, n - t + 1))
    # valley
    centre = (n-1)/2.0
    shapes.append(1.0/(1.0+((np.arange(n, dtype=float)-centre)**2)))
    # sine^2
    shapes.append(np.sin(np.pi * (np.arange(1, n+1, dtype=float)) / (n+1))**2)
    # exponential decays with various r
    for r in (0.9, 0.8, 0.7, 0.6, 0.5):
        shapes.append(np.power(r, np.arange(n, dtype=float)))
    # power-law decays
    for p in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n+1, dtype=float)
        shapes.append(1.0/(i**p))
    # Gaussian shapes
    xs = np.arange(n, dtype=float)
    centre = (n-1)/2.0
    for sigma_factor in np.linspace(0.05, 0.5, 9):
        sigma = sigma_factor * n
        a = np.exp(-((xs - centre)**2)/(2*sigma**2))
        shapes.append(a)
    # Laplacian
    for scale_factor in np.linspace(0.05, 0.5, 9):
        scale = scale_factor * n
        a = np.exp(-np.abs(xs - centre) / scale)
        shapes.append(a)
    # Beta-like symmetric power shape
    for alpha in np.linspace(0.5, 5.0, 9):
        u = xs/(n-1)  # [0,1]
        a = (u**alpha) * ((1-u)**alpha)
        shapes.append(a)
    # Raised cosine 
    for width_factor in np.linspace(0.2, 1.0, 5):
        width = width_factor * (n-1)/2
        a = np.zeros(n)
        d = np.abs(xs - centre)
        mask = d <= width
        a[mask] = 0.5 * (1 + np.cos(np.pi * d[mask] / width))
        shapes.append(a)
    return shapes

def _normalise_seq(arr):
    """Return a non‑negative array with total sum equal to 1."""
    a = np.array(arr, dtype=float)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a, dtype=float)
        s = a.sum()
    a = a / s               # now sum == 1
    return a

def _fast_score(seq):
    """Fast evaluation of 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = len(seq)
    # next power of two ≥ 2n‑1 for efficient FFT
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2 * n - 1]
    return 2.0 * n * conv.max()

def _convolve_fft(seq):
    """Return full convolution a*a using FFT (real output)."""
    n = len(seq)
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2 * n - 1]
    return conv

def _candidate_shapes_extended(n):
    """Generate a list of diverse non‑negative shapes of length n."""
    shapes = []
    xs = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0

    # 1. constant
    shapes.append(np.ones(n, dtype=float))

    # 2. linear (inc / dec)
    shapes.append(np.arange(1, n + 1, dtype=float))
    shapes.append(np.arange(n, 0, -1, dtype=float))

    # 3. quadratic (inc / dec)
    shapes.append(np.square(np.arange(1, n + 1, dtype=float)))
    shapes.append(np.square(np.arange(n, 0, -1, dtype=float)))

    # 4. triangular “mountain”
    t = np.arange(1, n + 1, dtype=float)
    shapes.append(np.minimum(t, n - t + 1))

    # 5. valley (large at ends, small centre)
    shapes.append(1.0 / (1.0 + (xs - centre) ** 2))

    # 6. sine‑squared
    shapes.append(np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) / (n + 1)) ** 2)

    # 7. exponential decays
    for r in (0.9, 0.8, 0.7, 0.6, 0.5):
        shapes.append(r ** xs)

    # 8. power‑law decays
    for p in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n + 1, dtype=float)
        shapes.append(1.0 / (i ** p))

    # 9. Gaussian blobs
    for sigma_factor in np.linspace(0.05, 0.5, 9):
        sigma = sigma_factor * n
        a = np.exp(-((xs - centre) ** 2) / (2 * sigma ** 2))
        shapes.append(a)

    # 10. Laplacian (double‑exponential) decay
    for scale_factor in np.linspace(0.05, 0.5, 9):
        scale = scale_factor * n
        a = np.exp(-np.abs(xs - centre) / scale)
        shapes.append(a)

    # 11. Symmetric beta‑like power shape
    for alpha in np.linspace(0.5, 5.0, 9):
        u = xs / (n - 1)  # in [0,1]
        a = (u ** alpha) * ((1 - u) ** alpha)
        shapes.append(a)

    # 12. Raised‑cosine (smooth cap)
    for width_factor in np.linspace(0.2, 1.0, 5):
        width = width_factor * (n - 1) / 2.0
        d = np.abs(xs - centre)
        a = np.zeros_like(xs)
        mask = d <= width
        a[mask] = 0.5 * (1 + np.cos(np.pi * d[mask] / width))
        shapes.append(a)

    return shapes

    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method consists of three stages:
      1. Warm‑start from an existing construction if available.
      2. Systematic exploration of many analytical families (constant,
         linear, quadratic, triangular, Gaussian, Laplacian, beta‑like, …)
         for several lengths (up to 32768).  Fast FFT based scoring is used.
      3. Gradient‑based refinement on the current best shape.  The gradient
         of the dominant convolution entry is taken, a step opposite to it
         is performed and the sequence is projected back onto the
         simplex (non‑negative, sum = 1).  Step size adapts and occasional
         random perturbations keep the search escaping shallow minima.
    The routine respects the supplied time budget and returns the best
    sequence found as a plain Python list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered sequence (if any)
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        seq = _normalise_seq(hs1)
        score = _fast_score(seq)
        best_seq = seq.copy()
        best_score = score

              f"score≈{best_score:.6f}")

    # Phase 1 – systematic scan of parametrised families
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384, 32768]
    for n in lengths:
        if time.time() >= deadline:
            break
        for shape in _candidate_shapes_extended(n):
            if time.time() >= deadline:
                break
            seq = _normalise_seq(shape)
            # fast evaluation (exact because sum=1)
            score = _fast_score(seq)
            if score < best_score:
                best_seq = seq.copy()
                best_score = score
                print(f"[{time.time() - start_time:.1f}s] New best (n={n}) score≈{best_score:.6f}")
                # early exit if we already beat the target by a comfortable margin
                if best_score <= 1.503:
                    print("Target already reached in Phase 1.")
                    return best_seq.tolist()

    # If for some reason we have no candidate (should not happen)
    if best_seq is None:
        best_seq = rng.random(1024)
        best_seq = _normalise_seq(best_seq)
        best_score = _fast_score(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – gradient­‑based refinement with occasional stochastic jumps
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_score
    eta = 0.1                     # initial step size
    max_eta = 0.5
    min_eta = 1e-8

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # Occasionally inject a random perturbation to escape plateaus
        if rng.random() < 0.07:
            # Gaussian noise on a random subset (≈2 % of entries)
            m = max(1, int(0.02 * len(cur)))
            idx = rng.choice(len(cur), size=m, replace=False)
            noisy = cur.copy()
            sigma = cur.mean() * 0.05
            noisy[idx] += rng.normal(scale=sigma, size=m)
            noisy = np.maximum(noisy, 0.0)
            noisy = (noisy + noisy[::-1]) / 2.0     # keep symmetry
            noisy /= noisy.sum()
            new = noisy
        else:
            # Gradient step aimed at the current maximal convolution entry
            conv = _convolve_fft(cur)
            k_max = int(np.argmax(conv))
            n = len(cur)
            idx = np.arange(n)
            j = k_max - idx
            mask = (j >= 0) & (j < n)
            grad = np.zeros_like(cur)
            grad[mask] = cur[j[mask]]
            if k_max % 2 == 0:
                diag = k_max // 2
                if 0 <= diag < n:
                    grad[diag] += cur[diag]            # double contribution on the diagonal
            # enforce symmetry on the gradient itself
            grad = (grad + grad[::-1]) / 2.0

            new = cur - eta * grad
            new = np.maximum(new, 0.0)
            new = (new + new[::-1]) / 2.0
            new_sum = new.sum()
            if new_sum > 0:
                new /= new_sum
            else:
                new = cur

        new_score = _fast_score(new)

        # Acceptance logic
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            # increase step size a little after a success
            eta = min(eta * 1.07, max_eta)
            if new_score < best_score:
                # verify with the official evaluator (tiny overhead)
                real_score = evaluate_sequence(new.tolist())
                if real_score < best_score:
                    best_score = real_score
                    best_seq = new.copy()
                    print(f"[{time.time() - start_time:.1f}s] Global improvement to ≈{best_score:.6f} "
                          f"(iteration {iteration})")

        if best_score <= 1.503:

            break

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    return best_seq.tolist()
