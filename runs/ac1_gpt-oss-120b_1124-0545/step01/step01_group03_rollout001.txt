"""
Propose a step‑function sequence with a low evaluation score.

The algorithm works in three phases:

1. **Initialisation** – a truncated central‑binomial (Catalan) sequence
   of length *n* is used as a good starting point (score ≈ π/2).

2. **Linear‑programming boost** – for the current sequence *f* we compute
   its convolution maximum   M = max(f∗f).  We then solve the LP

       maximise   Σ_j b_j
       subject to  (f ∗ b)[k] ≤ M   for all k = 0 … 2n‑2
                   b_j ≥ 0

   The LP finds a vector *b* that puts as much mass as possible into
   indices where the convolution with *f* is still slack.  This step
   dramatically increases the L¹‑norm without raising the peak
   convolution value.

3. **Mix & refine** – the new candidate is taken as a convex combination
   (1‑t) f + t b.  Several values of *t* are tried and the best is kept.
   Afterwards a light stochastic hill‑climb (tiny mass transfers) is
   applied to squeeze a little extra improvement.

The whole procedure repeats while time permits (a few iterations are
typically enough).  The final sequence is returned as a plain Python
list; it is also evaluated with the official `evaluate_sequence` function
for logging.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper: truncated Catalan / central‑binomial sequence
# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """
    Return a length‑`length` array a with
        a[i] = C(2*i, i) / 4**i .
    Recurrence: a[i+1] = a[i] * (2*i+1) / (2*(i+1)).
    All values lie in (0, 1].
    """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but without safety checks)
# ----------------------------------------------------------------------
def fast_score(seq: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(seq,seq)) / (sum(seq)**2) using FFT.
    """
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(F * F, n=size)[: 2 * n - 1]
    max_b = float(conv.max())
    sum_a = float(seq.sum())
    if sum_a < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Convolution maximum (used for the LP bound)
# ----------------------------------------------------------------------
def max_convolution(seq: np.ndarray) -> float:
    """Return max(f∗f) using FFT."""
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(F * F, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Build the sparse matrix that encodes (f ∗ b)[k] ≤ M   (k = 0 … 2n‑2)
# ----------------------------------------------------------------------
def build_constraint_matrix(f: np.ndarray, rows: np.ndarray, cols: np.ndarray) -> sp.csr_matrix:
    """
    Given a coefficient vector f (length n) and the pre‑computed
    row/col index arrays, return the CSR matrix A with
        A @ b = f ∗ b .
    """
    data = np.repeat(f, f.size)          # each f[i] appears n times (one per column)
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * f.size - 1, f.size))

            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            sc = fast_score(new_a)
            if sc < local_best_val:
                local_best_val = sc
                local_best_seq = new_a.copy()
                a = new_a
                print(f"[hill] iter {iteration}, attempt {attempts}, score={sc:.6f}")

        # ---- Update global best -------------------------------------------------
        if local_best_val < best_val:
            best_val = local_best_val
            best_seq = local_best_seq.copy()
            print(f"[update] new global best = {best_val:.6f}")

        # Prepare f for the next iteration – use the locally best candidate
        f = local_best_seq.copy()

        # optional early break if improvements become tiny
        if iteration >= 2 and best_val > 1.55:   # still above target, keep going
            continue

    # ------------------------------------------------------------------
    # 3) final official evaluation
    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    try:
        official = evaluate_sequence(final_list)   # already in the environment
        print(f"[final] official evaluate_sequence = {official:.6f}")
    except Exception as exc:   # defensive
        print("[final] evaluation failed:", exc)

    return final_list
```
