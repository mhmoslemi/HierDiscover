"""
Gradient‑descent on a smoothed maximum of the self‑convolution.

The original objective is
    E(a) = 2·n·max(conv(a,a)) / (∑a)² .
Because it is scale‑invariant we keep the ℓ₁‑norm fixed to
    ∑a = √(2·n)
and minimise the maximum entry of the convolution.

The algorithm works in three phases:

1. **Initialisation** – reuse an existing construction if available,
   otherwise start from a uniform vector and a random exponential vector.
2. **Smoothed optimisation** – minimise a smooth approximation
   of `max(conv)` given by the log‑sum‑exp function.
   The gradient of this surrogate can be computed analytically using
   two convolutions.  Adam‑style adaptive steps are applied and the
   iterate is projected onto the probability simplex to keep the
   ℓ₁‑norm fixed.
3. **Random perturbations / restarts** – every few hundred iterations a
   mixture of the current best solution and a fresh random vector is
   tried to avoid being trapped in a local minimum.

The whole routine respects the 1000 s time budget, constantly records
the best sequence found and finally returns it as a plain Python list.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex { w ≥ 0 , ∑ w = s }."""
    n = v.size
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – all entries equal
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function height vector that minimises
    `evaluate_sequence`.  The routine runs a projected Adam‑style gradient
    descent on a log‑sum‑exp surrogate for the maximum of the
    self‑convolution, interleaved with occasional random perturbations.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size (use the existing construction if it exists)
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        base = np.asarray(height_sequence_1, dtype=float)
    else:
        n = 2500                     # safe default
        base = np.full(n, 1.0)

    target_sum = math.sqrt(2.0 * n)          # the normalisation ∑a = √(2n)

    # Normalise the base vector to the required ℓ₁‑norm
    base = base / base.sum() * target_sum

    # ------------------------------------------------------------------
    # Assemble a few simple starters and pick the best one as the first iterate
    uniform = np.full(n, target_sum / n)
    rand_exp = np.random.exponential(scale=1.0, size=n)
    rand_exp = rand_exp / rand_exp.sum() * target_sum

    starters = [base, uniform, rand_exp]
    best_seq = None
    best_score = np.inf
    for cand in starters:
        conv = np.convolve(cand, cand)
        score = float(conv.max())          # because ∑a = √(2n) ⇒ score = max(conv)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()

    # Current iterate (will be updated)
    a = best_seq.copy()

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    t = 0

    # Hyper‑parameters for the smoothed objective and the step‑size
    eps_start, eps_end = 0.5, 0.01          # smoothing (log‑sum‑exp) schedule
    lr_start = target_sum * 0.001          # initial learning rate
    lr_end   = target_sum * 0.00005        # final learning rate

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # leave a tiny safety margin
    it = 0
    last_improve = 0

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ---- schedules -------------------------------------------------
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)   # 0 → 1 over the run
        eps = eps_start * ((eps_end / eps_start) ** frac)
        lr  = lr_start  * ((lr_end  / lr_start ) ** frac)

        # ---- current objective -----------------------------------------
        conv = np.convolve(a, a)
        cur_score = float(conv.max())

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = it
            print(f"[{it:6d}] New best  max(conv) = {best_score:.6f} "
                  f"(t={elapsed:.1f}s, eps={eps:.4f})")

        # ---- soft‑max weights for the smooth maximum ------------------
        # stability shift
        conv_shift = conv - conv.max()
        w_raw = np.exp(conv_shift / eps)
        w_sum = w_raw.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            # fall back to a hard max (one‑hot)
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = w_raw / w_sum                     # shape (2n‑1,)

        # ---- gradient of the smoothed objective ------------------------
        # moving‑sum σ_j = Σ_{i=j}^{j+n‑1} w_i  (length n)
        cumsum_w = np.cumsum(w)
        sigma = cumsum_w[n-1:2*n-1] - np.concatenate(([0.0], cumsum_w[:n-1]))

        # first term = Σ_{t=0}^{n‑1} w_{j+t} * a_t   (length n)
        a_rev = a[::-1]
        conv_wa = np.convolve(w, a_rev)           # length 3n‑2
        first = conv_wa[(n-1):(n-1+n)]

        # full gradient (vector of size n)
        grad = first + a * sigma

        # ---- Adam update ------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # gradient descent step (minimise)
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)                # keep non‑negative
        a_new = _simplex_projection(a_new, target_sum)

        a = a_new

        # ---- occasional random perturbations to escape plateaus ------
        if it % 500 == 0 and it > 0:
            # blend the best known solution with a fresh random one
            rnd = np.random.exponential(scale=1.0, size=n)
            rnd = rnd / rnd.sum() * target_sum
            mix = np.random.rand()
            a_pert = mix * best_seq + (1.0 - mix) * rnd
            a_pert = _simplex_projection(np.maximum(a_pert, 0.0), target_sum)

            # evaluate the perturbed candidate
            conv_p = np.convolve(a_pert, a_pert)
            score_p = float(conv_p.max())
            if score_p < best_score:
                best_score = score_p
                best_seq = a_pert.copy()
                last_improve = it
                print(f"[{it:6d}] New best after perturbation: {best_score:.6f}")

            # accept the perturbation if it improves the current point
            if score_p < cur_score:
                a = a_pert

        # ---- restart if we have been stagnant for a long while ------
        if it - last_improve > 3000:
            # re‑initialise from a mixture of the current best and noise
            noise = np.random.exponential(scale=1.0, size=n)
            noise = noise / noise.sum() * target_sum
            a = 0.5 * best_seq + 0.5 * noise
            a = _simplex_projection(a, target_sum)
            last_improve = it  # reset counter

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    return best_seq.tolist()
