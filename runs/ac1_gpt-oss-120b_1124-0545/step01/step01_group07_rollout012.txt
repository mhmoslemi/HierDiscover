"""
Search for a step‑function (non‑negative list) that minimizes
```
evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)
```
The objective is homogeneous, therefore only the *shape* matters.
The algorithm proceeds in three stages:

1. **Initialisation** – generate a few candidate lengths.  For each length
   we create a heavy‑tailed random vector, sort it descending and interleave
   the largest and smallest values (large values end up far apart, reducing
   the overlap with the reversed copy).

2. **Fast local optimisation** – for the current shape we repeatedly
   apply cheap stochastic moves that try to lower the current maximal
   convolution value.  Moves are of two kinds:
      * random swap of two positions,
      * targeted mass transfer: locate the lag `k` with the largest
        convolution value, pick the pair `(i, k‑i)` that contributes most,
        move a fraction of its mass to two random locations.
   After each move the sequence is projected onto the probability simplex
   (cut to non‑negative and renormalised).  The maximal convolution and
   its lag are obtained by an FFT based convolution, which is ≈ O(n log n)
   and therefore fast enough for thousands of iterations.

   Accepted moves are those that improve the *fast* score; worse moves are
   accepted with a simple temperature schedule (simulated‑annealing).

3. **Verification** – whenever a move yields a fast score lower than the
   best full score seen so far, the expensive `evaluate_sequence` function
   is called and the global best is updated.

The search respects the time budget (default 1000 s) and always returns
the best sequence found.  All helper functions are defined at module level
as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(seq: np.ndarray):
    """Return (max_value, argmax_index) of the linear convolution seq*seq."""
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    # Length of linear convolution = 2*n-1.  Choose power‑of‑2 for FFT.
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _fast_score(seq: np.ndarray) -> float:
    """Fast surrogate for `evaluate_sequence`: 2·n·max_b / sum^2."""
    n = seq.shape[0]
    s = seq.sum()
    if s <= 0.0:
        return float("inf")
    max_b, _ = _max_convolution_fft(seq)
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Shape helpers
# ----------------------------------------------------------------------
def _interleave_desc(values: np.ndarray) -> np.ndarray:
    """Sort descending and interleave largest/smallest (large‑far‑apart)."""
    v = np.sort(values)[::-1]            # descending
    n = v.size
    res = np.empty_like(v)
    lo, hi = 0, n - 1
    toggle = True
    for val in v:
        if toggle:
            res[lo] = val
            lo += 1
        else:
            res[hi] = val
            hi -= 1
        toggle = not toggle
    return res


def _random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    """Sample a random positive vector and normalise it to sum = 1."""
    # Gamma(1) == Exp(1) gives a Dirichlet(1,…,1) (uniform on simplex)
    a = rng.gamma(1.0, size=n)
    a = a / a.sum()
    return a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative coefficient list that minimises
    `evaluate_sequence`.  The function respects the time budget and returns
    the best sequence found so far.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional warm‑start from previously discovered construction
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        cand = np.asarray(hs1, dtype=float)
        # normalise to keep scale irrelevant (max < 1)
        cand = cand / np.max(cand) * 0.9
        cand = cand / cand.sum()
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.tolist()
            print(f"[{time.time() - start_time:.1f}s] Warm‑start from height_sequence_1 "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – larger lengths give a lower theoretical bound.
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- initialise a random shape --------------------------------
        cur = _interleave_desc(_random_dirichlet(rng, n))

        cur_score = _fast_score(cur)

        # occasional full evaluation to update the global best
        full = evaluate_sequence(cur.tolist())
        if full < best_score:
            best_score = full
            best_seq = cur.tolist()
            print(f"[{time.time() - start_time:.1f}s] Initial best (len={n}) "
                  f"score={best_score:.6f}")

        # ---- local stochastic optimisation ----------------------------
        # parameters for the inner loop
        it = 0
        max_iters = 200_000          # generous upper bound; loop will break on time

        while it < max_iters and time.time() < deadline:
            it += 1

            # temperature schedule (linear cooling)
            elapsed = time.time() - start_time
            total = deadline - start_time
            T = max(0.001, 1.0 - elapsed / total)

            # ----- propose a move ---------------------------------------
            if rng.random() < 0.55:
                # --- random swap -------------------------------------------------
                i, j = rng.choice(n, size=2, replace=False)
                new = cur.copy()
                new[i], new[j] = new[j], new[i]
            else:
                # --- targeted mass transfer --------------------------------------
                max_b, k = _max_convolution_fft(cur)          # lag with maximal overlap
                # collect the contributing pair with the largest product
                if k < n:
                    idx = np.arange(k + 1)
                else:
                    start = k - (n - 1)
                    idx = np.arange(start, n)
                prod = cur[idx] * cur[k - idx]
                i_max = idx[np.argmax(prod)]
                j_max = k - i_max
                eps = min(cur[i_max], cur[j_max]) * 0.3 * rng.random()
                if eps <= 0.0:
                    continue
                new = cur.copy()
                new[i_max] -= eps
                new[j_max] -= eps
                # redistribute the taken mass to two random locations
                p1 = rng.integers(0, n)
                p2 = rng.integers(0, n)
                new[p1] += eps / 2.0
                new[p2] += eps / 2.0

            # ----- keep feasibility ---------------------------------------
            new = np.clip(new, 0.0, None)
            total_mass = new.sum()
            if total_mass == 0.0:
                continue
            new = new / total_mass          # project onto simplex (scale invariance)

            # ----- cheap evaluation ----------------------------------------
            new_score = _fast_score(new)

            # ----- acceptance ---------------------------------------------
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                # simulated‑annealing acceptance of worse moves
                prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur = new
                cur_score = new_score

                if new_score < best_score * 0.99:   # only when promising
                    full_score = evaluate_sequence(new.tolist())
                    if full_score < best_score:

                      f"cur_score≈{cur_score:.6f} best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fall back to a simple uniform sequence if nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1024
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    print(f"[{time.time() - start_time:.1f}s] Search finished – best score={best_score:.6f}")
    return best_seq
