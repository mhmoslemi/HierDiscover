"""
Propose a sequence that minimizes the evaluation function.

Algorithm
---------
1. **Exact “1/√i” seed** – use the closed‑form coefficients
   a_k = C_k = binom(2k,k) / 4**k  (the discrete analogue of a(x)=1/√x).
   This shape already gives a score close to the theoretical constant π/2.

2. **Smoothed max‑gradient descent** – minimise a smooth surrogate of the
   objective  max(conv(a,a)).  For a temperature λ the surrogate is

        F_λ(a) = (1/λ)·log Σ_k exp(λ·c_k)   ,   c = a * a

   Its gradient w.r.t. a is
        ∇F_λ(a) = 2·conv(w , a_rev)   ,   w_k ∝ exp(λ·c_k)

   where a_rev is a reversed copy of a.  The gradient is projected onto the
   simplex (non‑negative entries with unit sum) and a momentum update
   (Adam‑style) is performed.

3. **Annealing schedule** – λ is increased slowly (softmax becomes sharper)
   while the learning‑rate is geometrically decayed.  Periodic random
   mass‑swaps help escape flat plateaus.

4. **Multiple restarts** – a few different initial shapes (pure 1/√i,
   a power law a_i∝(i+1)^‑p, a random Dirichlet) are tried; the best result
   of all runs is returned.

All heavy work (convolutions) is performed with FFTs (O(n log n)) so that
hundreds of thousands of iterations fit easily into the 1000 s budget.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full (linear) convolution of two real vectors using FFT."""
    n = len(x) + len(y) - 1
    L = _next_pow2(n)
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)[:n]
    return conv


def _smooth_max_gradient(seq: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(seq,seq)).
    Returns a vector of the same size as seq.
    """
    # 1. convolution a*a
    conv = _full_convolve_fft(seq, seq)                     # length 2n‑1
    # 2. soft‑max weights
    max_c = conv.max()
    # stabilise the exponentials
    exp_vals = np.exp(lam * (conv - max_c))
    w = exp_vals / exp_vals.sum()                           # length 2n‑1, sums to 1

    # 3. gradient = 2 * conv(w, rev(seq))
    rev = seq[::-1]
    tmp = _full_convolve_fft(w, rev)                        # length 3n‑2
    grad = 2.0 * tmp[len(seq) - 1: len(seq) - 1 + len(seq)]

    return grad


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    # we only need a positive scaling because we never have negative entries after clipping.
    s = v.sum()
    if s > 0:
        return v / s
    # fallback – uniform distribution
    n = v.shape[0]
    return np.full_like(v, 1.0 / n)


def _initial_binom_sequence(n: int) -> np.ndarray:
    """Exact discrete 1/√i sequence: a_k = binom(2k,k)/4**k, normalised to sum=1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2 * k - 1) / (2 * k)
    a /= a.sum()
    return a


def _initial_power_sequence(n: int, p: float, shift: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+shift+1)^‑p, normalised."""
    idx = np.arange(n, dtype=np.float64) + shift + 1.0
    a = idx ** (-p)
    a /= a.sum()
    return a


def _initial_dirichlet_sequence(n: int, seed: int) -> np.ndarray:
    """Random positive vector from a Dirichlet distribution."""
    rng = np.random.default_rng(seed)
    a = rng.gamma(1.0, 1.0, size=n)
    a /= a.sum()
    return a


def _score(seq: np.ndarray) -> float:
    """Fast replica of evaluate_sequence (no clamping because seq stays in [0,1000])."""
    n = len(seq)
    conv = _full_convolve_fft(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))

            seq = new_seq
            cur_score = _score(seq)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[polish] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a simple normalised 1/√i if everything failed
        best_seq = _initial_power_sequence(n, 0.5)
        print("[fallback] returning simple 1/√i sequence.")

    print(f"[final] best_score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
