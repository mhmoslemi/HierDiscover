"""
Gradient‑descent (soft‑max) optimisation of the step‑function heights.

The objective to minimise is 2 · n · max(conv(a,a)) / (sum(a)²).  Since the
objective is scale‑invariant we keep the ℓ₁‑norm fixed to
sqrt(2 n) and minimise the maximum entry of the self‑convolution.
We smooth the max using a soft‑max (log‑sum‑exp) with a temperature
parameter γ that is increased during the run.  The gradient of the
smoothed objective is computed analytically (a correlation of the
soft‑max weights with the current vector).  After each gradient step the
vector is projected back onto the simplex {a ≥ 0, sum(a)=√(2 n)}.
A simple momentum term and occasional random jitter help escape
plateaus.  Multiple random restarts are performed until the time budget
expires; the best sequence found is returned.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector v onto the simplex { w ≥ 0 , sum(w)=s }."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.size
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find rho = max { j : u_j - cssv_j / j > 0 }
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Fallback (should not happen for s>0)
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function height vector that minimises
    the evaluation function.  The routine runs for ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    random.seed(seed)

    # --------------------------------------------------------------
    # Problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 1500                     # sensible default

    target_sum = math.sqrt(2.0 * n)   # ℓ₁‑norm we keep fixed

    # --------------------------------------------------------------
    # Hyper‑parameters (annealing schedules)
    start_lr = target_sum * 0.5      # initial learning rate
    end_lr   = target_sum * 0.001    # final learning rate
    start_gamma = 0.001              # start with a very smooth max
    end_gamma   = 10.0               # end with a sharp max
    momentum_beta = 0.9              # momentum coefficient
    jitter_interval = 200            # every ``jitter_interval`` iters add noise
    jitter_scale = target_sum * 0.02 # magnitude of the random jitter

    # --------------------------------------------------------------
    # Timing
    start_time = time.time()
    deadline   = start_time + budget_s - 0.5   # safety margin
    best_seq   = None
    best_score = float("inf")
    iteration  = 0

    # --------------------------------------------------------------
    # Helper to evaluate without the extra Python wrapper (faster)
    def _fast_score(a_vec: np.ndarray) -> float:
        conv = np.convolve(a_vec, a_vec)
        max_b = np.max(conv)
        # evaluate_sequence does the same scaling, but we avoid its overhead
        return float(2.0 * n * max_b / (target_sum ** 2))

    # --------------------------------------------------------------
    # Main optimisation loop – many random restarts
    while time.time() < deadline:
        # --- initialise a fresh start --------------------------------
        a = np.random.exponential(scale=1.0, size=n)
        a = a / a.sum() * target_sum          # project onto simplex
        momentum = np.zeros_like(a)

        # Run inner gradient descent for a bounded number of steps
        inner_max_iters = 3000
        inner_start = time.time()

        while (time.time() < deadline and
               iteration < inner_max_iters):
            frac = (time.time() - start_time) / (deadline - start_time)

            # Linear schedules for learning rate and gamma
            lr = start_lr * ((end_lr / start_lr) ** frac)
            gamma = start_gamma * ((end_gamma / start_gamma) ** frac)

            # --- compute convolution and soft‑max weights ---------------
            conv = np.convolve(a, a)                # length 2n‑1
            max_conv = np.max(conv)                 # only for numerical stability
            # soft‑max (log‑sum‑exp) over conv entries
            exp_term = np.exp(gamma * (conv - max_conv))
            weights = exp_term / exp_term.sum()     # shape (2n‑1,)

            # --- gradient of the smoothed max (correlation) -----------
            # gradient_j = 2 * Σ_k weights_k * a_{k‑j}
            grad = 2.0 * np.correlate(weights, a, mode='valid')

            # --- momentum & gradient step -------------------------------
            momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad
            a_proposed = a - lr * momentum

            # --- occasional random jitter to escape plateaus -----------
            if iteration % jitter_interval == 0 and iteration > 0:
                jitter = np.random.normal(scale=jitter_scale, size=n)
                a_proposed += jitter

            # --- projection back onto simplex ----------------------------
            a = _simplex_projection(a_proposed, target_sum)

            # --- evaluate (full evaluate_sequence is cheap enough) -----
            curr_score = _fast_score(a)
            if curr_score < best_score:
                best_score = curr_score
                best_seq   = a.copy()
                elapsed = time.time() - start_time
                print(f"[{iteration:6d}] New best score: {best_score:.6f} (time {elapsed:.1f}s)")

            iteration += 1

            # early break of inner loop if we are close to the deadline
            if time.time() >= deadline:
                break

    # ------------------------------------------------------------------
    # Fallback: if nothing was found (should not happen), return uniform
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    # Return a plain Python list as required by the harness
    return best_seq.tolist()
