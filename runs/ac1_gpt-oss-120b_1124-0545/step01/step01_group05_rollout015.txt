def _conv_full(a):
   n = len(a)
   size = 1 << ((2*n -1).bit_length())
   A = np.fft.rfft(a, n=size)
   conv = np.fft.irfft(A * A, n=size)[:2*n-1]
   return conv

def _project_simplex(v):
    """Project onto probability simplex (sum=1, non-negative)."""
    # algorithm from Duchi et al.
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho where u_i - (cssv_i / i) > 0
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _guided_perturb(a, conv, k_max, step_frac):
    n = len(a)
    # compute contributions for each i: contrib_i = a[i] * a[kmax-i] if valid else 0
    contributions = np.zeros_like(a)
    i_min = max(0, k_max-(n-1))
    i_max = min(n-1, k_max)
    idx = np.arange(i_min, i_max+1)
    contributions[idx] = a[idx] * a[k_max - idx]
    # pick source index i from contributions proportional to contributions.
    total_contrib = contributions.sum()
    if total_contrib == 0:
        # fallback: random
        i = np.random.randint(n)
    else:
        prob = contributions / total_contrib
        i = np.random.choice(n, p=prob)
    # pick random destination j not equal i
    j = np.random.choice(np.delete(np.arange(n), i))
    delta = a[i] * step_frac * np.random.uniform(0,1)
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = np.clip(a_new,0, None)
    s = a_new.sum()
    if s == 0: return a
    a_new /= s
    return a_new

"""
Propose a step‑function (non‑negative heights) with a low evaluation score.
The algorithm consists of three phases:

1.  Broad random search over many parametrised families (exponential,
    mixtures, beta‑shapes, piece‑wise linear, Dirichlet etc.).  Each candidate
    is scored with a fast FFT based surrogate; the official evaluator is
    called only when an improvement is observed.

2.  A targeted local search that repeatedly reduces the most critical
    convolution entry.  For the current best sequence the convolution
    index k with maximal value is found; the algorithm moves a small
    amount of mass from positions that contribute heavily to that entry
    towards positions that contribute little.  Moves are accepted only
    when they improve the fast score, and the exact score is recomputed
    on acceptance.

3.  A simple projected‑gradient refinement: the sub‑gradient of the
    maximal convolution term is taken (g_i = 2·a_{k‑i}) and a step in the
    opposite direction is projected back onto the probability simplex.
    This provides a smooth “hill‑climbing’’ phase that can escape shallow
    plateaus.

The whole routine respects the 1000 s time budget, prints occasional
progress reports and finally returns the best sequence found (as a plain
list of floats).  All helper functions are top‑level and contain no
closures.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Fast convolution helpers (FFT based, real input)
# --------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of a * a."""
    return float(_conv_full(a).max())


def _fast_score(a: np.ndarray) -> float:
    """Fast surrogate of evaluate_sequence (assumes sum(a)==1)."""
    n = a.shape[0]
    return 2.0 * n * _conv_max(a)


# --------------------------------------------------------------
# Simplex projection (Duchi et al.)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find the last index where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Families of candidate sequences
# --------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=float)


def _make_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def _make_mix_exponential(n: int, w: float, lam1: float, lam2: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = w * np.exp(-lam1 * i) + (1.0 - w) * np.exp(-lam2 * i)
    a /= a.sum()
    return a


def _make_mix_unif_exp(n: int, w: float, lam: float) -> np.ndarray:
    """Mixture of uniform (weight w) and exponential (weight 1‑w)."""
    uniform = _make_uniform(n)
    expo = _make_exponential(n, lam)
    a = w * uniform + (1.0 - w) * expo
    a /= a.sum()
    return a


def _make_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = i ** p
    a /= a.sum()
    return a


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^α · (n‑i)^β   (α,β can be negative)."""
    i = np.arange(n, dtype=float) + 1.0
    left = i ** alpha
    right = (n - i + 1) ** beta
    a = left * right
    a_sum = a.sum()
    if a_sum == 0.0:
        # fallback to uniform
        return _make_uniform(n)
    a /= a_sum
    return a


def _make_two_block(n: int, m: int, high_frac: float) -> np.ndarray:
    if m <= 0 or m >= n:
        raise ValueError("invalid block size")
    high = high_frac / m
    low = (1.0 - high_frac) / (n - m)
    a = np.empty(n, dtype=float)
    a[:m] = high
    a[m:] = low
    return a


def _make_linear_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ max(0, 1 - lam·i).  lam chosen so that at least one entry stays >0."""
    i = np.arange(n, dtype=float)
    a = 1.0 - lam * i
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        return _make_uniform(n)
    a /= s
    return a


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    a = np.random.dirichlet([alpha] * n)
    return a


def _make_sorted_dirichlet(n: int, alpha: float) -> np.ndarray:
    a = np.random.dirichlet([alpha] * n)
    a = np.sort(a)[::-1]               # descending order
    a /= a.sum()
    return a


# --------------------------------------------------------------
# Gradient of the maximal convolution entry
# --------------------------------------------------------------
def _grad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the max‑convolution term at shift k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    if i_min <= i_max:
        idx = np.arange(i_min, i_max + 1, dtype=int)
        g[idx] = 2.0 * a[k - idx]
    return g


# --------------------------------------------------------------
# Targeted perturbation (moves mass away from the worst shift)
# --------------------------------------------------------------
def _guided_perturb(a: np.ndarray, conv: np.ndarray, k_max: int, step_frac: float) -> np.ndarray:
    """Move a small amount of probability from indices that heavily
    contribute to the worst convolution entry to a random low‑contributing
    index.  Returns a new (still normalised) candidate."""
    n = a.shape[0]
    # contribution of each index i to conv[k_max] = a[i] * a[k_max-i]
    contrib = np.zeros_like(a)
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    if i_min <= i_max:
        idx = np.arange(i_min, i_max + 1, dtype=int)
        contrib[idx] = a[idx] * a[k_max - idx]

    total = contrib.sum()
    if total == 0.0:
        # nothing to guide – just pick a random source
        src = np.random.randint(n)
    else:
        prob = contrib / total
        src = np.random.choice(n, p=prob)

    # destination: choose a random index that is not the source
    dst_choices = np.delete(np.arange(n), src)
    dst = np.random.choice(dst_choices)

    # amount to move – a fraction of the source mass
    delta = a[src] * step_frac * np.random.uniform(0.0, 1.0)
    if delta <= 0.0:
        return a

    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    # re‑normalise (the total stays 1, but rounding errors may appear)
    a_new = np.clip(a_new, 0.0, None)
    a_sum = a_new.sum()
    if a_sum == 0.0:
        return a
    a_new /= a_sum
    return a_new


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the supplied `evaluate_sequence` score.  The routine respects the
    supplied time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # small safety margin

    # -----------------------------------------------------------------
    # Length of the pattern – use the existing construction if present.
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2500                # a reasonable default

    # -----------------------------------------------------------------
    # Baseline – uniform distribution.
    # -----------------------------------------------------------------
    best_a = _make_uniform(n)
    best_score = _fast_score(best_a)
    # also store the exact value (costs almost nothing for the first candidate)
    try:
        exact = evaluate_sequence(best_a.tolist())
        best_score = exact
    except Exception:
        pass

    # -----------------------------------------------------------------
    # Random search over a broad set of families.
    # -----------------------------------------------------------------
    families = [
        "exp", "mix_exp", "mix_unif_exp", "power",
        "beta", "two_block", "linear", "dirichlet",
        "sorted_dirichlet", "perturb"
    ]
    probs = np.array([0.12, 0.12, 0.10, 0.10, 0.10, 0.15, 0.07, 0.09, 0.07, 0.08])
    probs /= probs.sum()

    it = 0
    rand_deadline = start_time + budget_s * 0.78   # leave ~22 % for refinement
    while time.time() < rand_deadline:
        it += 1
        method = np.random.choice(families, p=probs)

        if method == "exp":
            lam = np.random.uniform(0.0, 5.0 / n)
            a = _make_exponential(n, lam)

        elif method == "mix_exp":
            w = np.random.uniform(0.0, 1.0)
            lam1 = np.random.uniform(0.0, 8.0 / n)
            lam2 = np.random.uniform(0.0, 8.0 / n)
            a = _make_mix_exponential(n, w, lam1, lam2)

        elif method == "mix_unif_exp":
            w = np.random.uniform(0.0, 1.0)
            lam = np.random.uniform(0.0, 8.0 / n)
            a = _make_mix_unif_exp(n, w, lam)

        elif method == "power":
            p = np.random.uniform(-8.0, 8.0)
            a = _make_power_law(n, p)

        elif method == "beta":
            alpha = np.random.uniform(-3.0, 5.0)
            beta = np.random.uniform(-3.0, 5.0)
            a = _make_beta(n, alpha, beta)

        elif method == "two_block":
            m = np.random.randint(int(0.05 * n), int(0.95 * n) + 1)
            high_frac = np.random.uniform(0.0, 1.0)
            a = _make_two_block(n, m, high_frac)

        elif method == "linear":
            lam = np.random.uniform(0.0, 2.0 / n)   # ensure at least one positive entry
            a = _make_linear_decay(n, lam)

        elif method == "dirichlet":
            alpha = np.random.uniform(0.1, 5.0)
            a = _make_dirichlet(n, alpha)

        elif method == "sorted_dirichlet":
            alpha = np.random.uniform(0.1, 5.0)
            a = _make_sorted_dirichlet(n, alpha)

        else:              # "perturb" – a small random move from the current best
            a = best_a.copy()
            i, j = np.random.choice(n, 2, replace=False)
            delta = a[i] * np.random.uniform(-0.3, 0.3)
            delta = np.clip(delta, -a[i] + 1e-12, 1e12)
            a[i] -= delta
            a[j] += delta

        # enforce constraints and normalise
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s

        # fast surrogate score
        fast = _fast_score(a)

        if fast < best_score:
            # verify with the official evaluator
            try:
                exact = evaluate_sequence(a.tolist())
                if exact < best_score:
                    best_score = exact
                    best_a = a.copy()
                    elapsed = time.time() - start_time
                    print(f"[{elapsed:6.1f}s] New best (rand): {best_score:.6f} "
                          f"(iter {it}, method={method})")
            except Exception:
                pass

        if it % 5000 == 0:
            elapsed = time.time() - start_time
            remaining = max(0.0, deadline - time.time())
            print(f"[{elapsed:6.1f}s] iter={it} best={best_score:.6f} "
                  f"time left≈{remaining:.0f}s")

    # -----------------------------------------------------------------
    # Focused refinement: guided perturbations around the current best.
    # -----------------------------------------------------------------
    a = best_a.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_full(a)
        k_max = int(np.argmax(conv))

        # tiny move that is directed at the worst shift
        a_candidate = _guided_perturb(a, conv, k_max, step_frac=0.03)

        fast = _fast_score(a_candidate)
        if fast < best_score:
            try:
                exact = evaluate_sequence(a_candidate.tolist())
                if exact < best_score:
                    best_score = exact
                    best_a = a_candidate.copy()
                    a = a_candidate.copy()
                    elapsed = time.time() - start_time
                    print(f"[{elapsed:6.1f}s] New best (guided): {best_score:.6f} "
                          f"(iter {it})")
                    continue  # accepted – skip gradient step for this round
            except Exception:
                pass

        # If guided move not accepted, fall back to a modest projected‑gradient step
        grad = _grad_max_conv(a, k_max)
        step = 0.5 / np.sqrt(it)                      # decreasing step size
        a_tmp = _project_simplex(a - step * grad)

        fast = _fast_score(a_tmp)
        if fast < best_score:
            try:
                exact = evaluate_sequence(a_tmp.tolist())
                if exact < best_score:
                    best_score = exact
                    best_a = a_tmp.copy()
                    a = a_tmp.copy()
                    elapsed = time.time() - start_time
                    print(f"[{elapsed:6.1f}s] New best (grad): {best_score:.6f} "
                          f"(iter {it})")
                    continue
            except Exception:
                pass

        # otherwise keep the old a (may still be useful for next iteration)
        if it % 2000 == 0:
            elapsed = time.time() - start_time
            remaining = max(0.0, deadline - time.time())
            print(f"[{elapsed:6.1f}s] refinement iter={it} best={best_score:.6f} "
                  f"time left≈{remaining:.0f}s")

    # -----------------------------------------------------------------
    # Return the best sequence (plain Python list)
    # -----------------------------------------------------------------
    return best_a.tolist()
