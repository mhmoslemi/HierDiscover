    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i iterates over i where a_i contributes: a_i * b_{k-i}
        # So set A[k, j] = a_{k-j} where j index of b.
        # So for j in [i_start, i_end], we have i = k - j, so A[k, j] = a[i]
        # Let's vectorize:
        # No T needed.
        # Loop over j:
        for j in range(i_start, i_end+1):
            A[k, j] = a[k - j]
    # Solve LP:
    c = -np.ones(n)  # minimize -sum(b)
    bounds = [(0, None)] * n
    b_ub = np.full(rows, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs-ds')
    if res.success:
        b = res.x
        # Rescale sum to 1:
        if b.sum() > 0:
            b_norm = b / b.sum()
            return b_norm
    # fallback: return a
    return a

"""
Algorithm:
1. Choose initial sequence either from provided global `height_sequence_1`
   or via a quick parametric sweep over several families (geometric,
   power-law, geometric·power, symmetric geometric, Gaussian, tent).
2. Optimize the sequence on the probability simplex using an
   Adam‑style adaptive gradient descent on a smoothed max of the
   autocorrelation (log‑sum‑exp).  The smoothing parameter beta is
   increased slowly, the step size is scaled by the Adam moments,
   and after each update the vector is projected onto the simplex.
   Random perturbations are injected when progress stalls.
3. The best sequence seen during the search is returned.
The whole routine respects the supplied time budget.
"""
import time
import numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _project_simplex(v):
    """ Euclidean projection onto the probability simplex. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    mask = u - cssv / ind > 0
    if not np.any(mask):
        theta = cssv[-1] / n
    else:
        rho = np.where(mask)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# families ---------------------------------------------------------------

def _geom(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a

def _power(n, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a

def _geom_power(n, r, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    if s == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a

def _sym_geom(n, r):
    a = r ** np.arange(n, dtype=np.float64) + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    a /= a.sum()
    return a

def _gaussian(n, sigma):
    mu = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    s = a.sum()
    if s == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a

def _tent(n):
    # symmetric triangular shape
    half = (n + 1) // 2
    a = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        a = np.concatenate([a, a[::-1]])
    else:
        a = np.concatenate([a, a[-2::-1]])
    a /= a.sum()
    return a

# evaluation --------------------------------------------------------------

def _objective(a):
    # a assumed normalised to sum=1
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))

        conv = _conv_fft(a, a)
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:

            best_seq = a.copy()

            print(f"[adam] iter {it:5d} new best val={cur_val:.6f}")

        beta = beta0 * (beta_mul ** it)
        if beta > 500.0:
            beta = 500.0
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()

        else:
            w /= w_sum

        grad_full = _conv_fft(a[::-1], w)   # length = 3n-2

    print(f"[final] best value observed = {best_val:.6f} with length {best_seq.size}")

Algorithm:
1. Obtain a good start – if the notebook already defines a variable
   ``height_sequence_1`` we reuse it, otherwise we perform a cheap
   parametric sweep over several families (geometric, power‑law,
   geometric·power, symmetric geometric, Gaussian and a simple tent).
   Every candidate is normalised to sum = 1 and the objective
   ``2·n·max(conv(a,a))`` is evaluated with an FFT based convolution.

2. Refine the start using an Adam‑style adaptive gradient descent on a
   smoothed version of the max (log‑sum‑exp).  The smoothing parameter
   ``beta`` grows slowly so that the gradient initially sees a soft
   maximum and later focuses on the true peak.  After each additive step
   the vector is projected onto the probability simplex.
   If progress stalls for a while, a small relative Gaussian perturbation
   is injected to escape shallow local minima.

3. The best sequence seen before the time budget expires is returned as a
   plain Python list.  All operations respect a user‑supplied time budget
   (default = 1000 s) and no external I/O is performed.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution via FFT.  Returns the full (len(x)+len(y)-1) vector.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    mask = u - cssv / ind > 0
    if not np.any(mask):
        theta = cssv[-1] / n
    else:
        rho = np.where(mask)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    s = a.sum()
    if s == 0:

    else:
        a /= s
    return a

    s = a.sum()
    if s == 0:

    a /= a.sum()
    return a


def _gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle, normalised."""
    mu = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    s = a.sum()
    if s == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _tent(n: int) -> np.ndarray:
    """Symmetric triangular (tent) shape."""
    half = (n + 1) // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        a = np.concatenate([inc, inc[::-1]])
    else:
        a = np.concatenate([inc, inc[-2::-1]])
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Objective (sum = 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(convolution) for a normalised to sum = 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The function respects the supplied time budget and returns the best
    sequence (list of floats) found.
    """
    import time
    import numpy as np

    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    # ------------------------------------------------------------------
    # 1.  Initialise – reuse previous construction if present
    # ------------------------------------------------------------------
    init = None
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0:
                init = cand.copy()
                init = np.maximum(init, 0.0)
                s = init.sum()
                if s > 0:
                    init /= s
                else:
                    init = None
        except Exception:
            init = None

    N_target = init.shape[0] if init is not None else 4000
    print(f"[setup] target length = {N_target}")

    best_seq = None
    best_val = np.inf
    best_n = None

    if init is not None:
        val = _objective(init)
        best_seq = init.copy()
        best_val = val
        best_n = init.shape[0]
        print(f"[init] provided sequence objective = {val:.6f}")

    # ------------------------------------------------------------------
    # 2.  Light parametric sweep (covers a variety of shapes)
    # ------------------------------------------------------------------
    sweep_lengths = [256, 512, 1024, 2048, N_target]
    c_vals = np.linspace(0.5, 5.0, 15)                # geometric families
    p_vals = np.linspace(0.0, 2.0, 15)                # power families
    sigma_vals = np.linspace(0.2, 0.45, 10) * N_target  # Gaussian sigmas

    for n in sweep_lengths:
        if time.time() > deadline:
            break

        # geometric
        for c in c_vals:
            r = 1.0 - c / n
            if r <= 0:
                continue
            a = _geom(n, r)
            val = _objective(a)
            if val < best_val:
                best_val, best_seq, best_n = val, a.copy(), n
                print(f"[sweep] geom  n={n:4d} c={c:5.2f} val={val:.6f}")

        # power‑law
        for p in p_vals:
            a = _power(n, p)
            val = _objective(a)
            if val < best_val:
                best_val, best_seq, best_n = val, a.copy(), n
                print(f"[sweep] power n={n:4d} p={p:5.2f} val={val:.6f}")

        # geometric·power
        for c in c_vals:
            r = 1.0 - c / n
            if r <= 0:
                continue
            for p in p_vals:
                a = _geom_power(n, r, p)
                val = _objective(a)
                if val < best_val:
                    best_val, best_seq, best_n = val, a.copy(), n
                    print(f"[sweep] g×p   n={n:4d} c={c:5.2f} p={p:5.2f} val={val:.6f}")

        # symmetric geometric
        for c in c_vals:
            r = 1.0 - c / n
            if r <= 0:
                continue
            a = _sym_geom(n, r)
            val = _objective(a)
            if val < best_val:
                best_val, best_seq, best_n = val, a.copy(), n
                print(f"[sweep] sym‑g n={n:4d} c={c:5.2f} val={val:.6f}")

        # Gaussian (only for the final length to keep work modest)
        if n == N_target:
            for sigma in sigma_vals:
                a = _gaussian(n, sigma)
                val = _objective(a)
                if val < best_val:
                    best_val, best_seq, best_n = val, a.copy(), n
                    print(f"[sweep] gauss n={n:4d} sigma={sigma:.1f} val={val:.6f}")

        # simple tent shape
        a = _tent(n)
        val = _objective(a)
        if val < best_val:
            best_val, best_seq, best_n = val, a.copy(), n
            print(f"[sweep] tent  n={n:4d} val={val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Resize to the final length if necessary
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.ones(N_target, dtype=np.float64) / N_target
        best_val = _objective(best_seq)
        best_n = N_target
        print("[fallback] uniform sequence used")

    if best_n != N_target:
        x_old = np.linspace(0, 1, best_n)
        x_new = np.linspace(0, 1, N_target)
        best_seq = np.interp(x_new, x_old, best_seq)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq /= best_seq.sum()
        print(f"[resize] upsampled from {best_n} to {N_target}")

    # ------------------------------------------------------------------
    # 4.  Adam‑style SGD on a smoothed max of the autocorrelation
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.4
    beta0 = 5.0
    beta_mul = 1.025

    # stagnation handling
    last_improve = 0
    patience = 800
    max_iters = 12000

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            print("[time] budget exhausted – stopping optimisation")
            break

        conv = _conv_fft(a, a)
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = it
            print(f"[adam] iter {it:5d} new best val={cur_val:.6f}")

        # smoothed max via log‑sum‑exp
        beta = beta0 * (beta_mul ** it)
        if beta > 500.0:
            beta = 500.0
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        grad_full = _conv_fft(a[::-1], w)          # length = 3·n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update (additive)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

        # random perturbation if no improvement for a while
        if it - last_improve > patience:
            noise = np.random.randn(n) * 0.02
            a = a + noise * a
            a = np.maximum(a, 0.0)
            a /= a.sum()
            last_improve = it
            print(f"[perturb] iter {it:5d} – injected noise")

    # ------------------------------------------------------------------
    # 5.  Return the best sequence found
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.ones_like(best_seq) / best_seq.size

    print(f"[final] best observed value = {best_val:.6f}, length = {best_seq.size}")
    return best_seq.tolist()
