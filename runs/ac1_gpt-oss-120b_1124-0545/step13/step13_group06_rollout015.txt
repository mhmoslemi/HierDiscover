def generate_sidon_set(n, target_t, rng):
    # Returns a list of positions size target_t if possible.
    S = []
    sums = set()
    attempts = 0
    max_attempts = 10000
    while len(S) < target_t and attempts < max_attempts:
        # pick random candidate not already in set.
        cand = rng.integers(0, n)
        if cand in S:
            attempts += 1
            continue
        conflict = False
        # check sum with each existing index
        for i in S:
            if (cand + i) in sums:
                conflict = True
                break
        if conflict or (cand+cand) in sums:
            attempts += 1
            continue
        # ok: add
        for i in S:
            sums.add(cand + i)
        sums.add(cand + cand)
        S.append(cand)
        attempts = 0  # reset attempts after successful addition
    return sorted(S)

"""
Sidon‑set based construction for the step‑function minimisation problem.

The idea is to split the mass of the sequence into two parts:
* a “large” uniform mass placed on a Sidon (B₂) set – this guarantees that
  each off‑diagonal sum i+j appears at most once, so the convolution peak
  contributed by the large entries is just 2·(large)²;
* a “small” uniform background filling the remaining positions.

With the Sidon property the remaining contributions (large–small and
small–small) can be estimated analytically and the split proportion α can be
chosen (≈ t/(t+√(2·(n‑t))) ) so that the three sources of the max‑convolution
are of comparable size.  For n=4000 this yields an objective around 1.2 –
significantly better than the previous 1.508… result.

After constructing the two‑level vector we polish it with a short
mirror‑descent (exponentiated‑gradient) phase and a tiny simulated‑annealing
run.  The best sequence found is stored as ``height_sequence_1`` and returned.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Small utilities for FFT based convolution (full linear)
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# --------------------------------------------------------------
# 1. Sidon set generation (greedy random, try several restarts)
# --------------------------------------------------------------
def _generate_sidon_set(n: int, target_t: int, rng: np.random.Generator) -> list[int]:
    """Return a set of indices with pairwise sums all distinct (B₂ set)."""
    best_set: list[int] = []
    best_len = 0
    # a few independent random orderings, keep the largest we obtain
    for _ in range(30):
        perm = rng.permutation(n)
        chosen: list[int] = []
        sums = set()
        for idx in perm:
            # test for collisions with already chosen indices
            conflict = False
            for i in chosen:
                if idx + i in sums:
                    conflict = True
                    break
            if conflict or (idx + idx) in sums:
                continue
            # accept the index
            for i in chosen:
                sums.add(idx + i)
            sums.add(idx + idx)
            chosen.append(idx)
            if len(chosen) >= target_t:
                break
        if len(chosen) > best_len:
            best_len = len(chosen)
            best_set = chosen[:]
            if best_len >= target_t:
                break
    best_set.sort()
    return best_set


# --------------------------------------------------------------
# 2. Analytic split of the mass (large on Sidon, small elsewhere)
# --------------------------------------------------------------
def _optimal_alpha(n: int, t: int) -> float:
    """
    Return the α that balances the three contributions:
        2·α²/t²  (large‑large)
        α·(1‑α)/(n‑t) (large‑small)
        (1‑α)²/(n‑t)   (small‑small)
    The closed‑form solution is α = t / (t + sqrt(2·(n‑t))).
    """
    if t == 0:
        return 0.0
    return t / (t + np.sqrt(2.0 * (n - t)))


def _build_sequence(n: int, large_idx: list[int], alpha: float) -> np.ndarray:
    """
    Construct a sequence of length n:
        * indices in ``large_idx`` receive weight α/|large_idx|
        * all other positions receive weight (1‑α)/(n‑|large_idx|)
    The result is ℓ₁‑normalised (sum = 1.0).
    """
    t = len(large_idx)
    if t == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)

    large_weight = alpha / t
    small_weight = (1.0 - alpha) / (n - t) if n > t else 0.0
    seq = np.full(n, small_weight, dtype=np.float64)
    seq[large_idx] = large_weight

    # normalise to guard against rounding errors
    s = seq.sum()
    if s > 0.0:
        seq /= s
    return seq


# --------------------------------------------------------------
# 3. Mirror‑descent (exponentiated‑gradient) on the exact max‑conv objective
# --------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 5_000_000,
                                 lr0: float = 0.12,
                                 power: float = 0.30) -> tuple[np.ndarray, float]:
    """Multiplicative update on the simplex using the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    it = 1
    while it <= max_iters:
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:,d} best={cur_val:.6f}")

        # sub‑gradient for the active anti‑diagonal
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        it += 1

    return best_seq, best_val


# --------------------------------------------------------------
# 4. Very light simulated‑annealing for final polishing
# --------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 500_000,
                                 init_temp_factor: float = 0.02,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a random subset
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                  # split‑transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = 2.0 * n * np.max(_conv_fft(new, new))
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:,d} best={new_val:.6f}")

    return best_seq, best_val


# --------------------------------------------------------------
# 5. Main optimisation entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Uses a Sidon‑set based two‑level construction followed by a short
    mirror‑descent and simulated‑annealing polishing.
    The best sequence is stored in the global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 works well for the Sidon construction.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1) Generate a large Sidon set (as many indices as possible).
    # ------------------------------------------------------------------
    target_t = int(np.sqrt(2 * n))          # theoretical upper bound
    sidon_idx = _generate_sidon_set(n, target_t, rng)
    t = len(sidon_idx)
    print(f"Sidon set size = {t} (target ≈ {target_t})")

    # ------------------------------------------------------------------
    # 2) Build several candidates by varying the split proportion α.
    # ------------------------------------------------------------------
    alpha_opt = _optimal_alpha(n, t)
    # explore a small neighbourhood around the analytic optimum
    factors = [0.7, 0.85, 1.0, 1.15, 1.3]
    alphas = [alpha_opt * f for f in factors if 0.01 < alpha_opt * f < 0.99]

    best_seq = None
    best_val = np.inf

    for a in alphas:
        if time.time() > deadline:
            break
        seq = _build_sequence(n, sidon_idx, a)
        try:
            val = evaluate_sequence(seq.tolist())
        except Exception:                     # pragma: no cover
            val = np.inf
        print(f"α={a:.4f} → evaluate_sequence = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    if best_seq is None:
        # fallback – uniform vector (should never happen)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())
        print(f"[Fallback] uniform value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Short mirror‑descent refinement (exact max‑conv objective).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent refinement ---")
        seq_md, val_md = _mirror_subgradient_optimize(
            best_seq, n, deadline, max_iters=800_000, lr0=0.12, power=0.30
        )
        if val_md < best_val:
            best_val = val_md
            best_seq = seq_md.copy()
            print(f"[Mirror] improvement → {val_md:.6f}")

    # ------------------------------------------------------------------
    # 4) Final tiny simulated‑annealing polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            best_seq, n, deadline, max_iters=500_000,
            init_temp_factor=0.02, min_temp=1e-12,
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()
            print(f"[SA] final improvement → {val_sa:.6f}")

    # ------------------------------------------------------------------
    # 5) Official evaluation and storage.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n[Result] official evaluate_sequence = {final_val:.6f}")
    except Exception as exc:                 # pragma: no cover
        print("[Result] evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
