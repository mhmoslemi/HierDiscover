def _adam_optimize(a0, n, deadline, max_iters):
    # a0 is normalized sum to 1; but we need logs
    eps = 1e-30
    v = np.log(a0 + eps)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.1
    best_seq = a0
    best_val = evaluate_sequence(a0.tolist())
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # compute subgradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # gradient w.r.t logits
        dot = np.dot(g, a)
        grad_v = a * (g - dot)
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat_corr = v_hat / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat_corr) + 1e-8)
        v = v - step
        # optional: every 10000 it, evaluate objective
        if t % 10000 == 0:
            cur_seq = _softmax(v)
            cur_val = evaluate_sequence(cur_seq.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[Adam] iteration {t} new best {best_val:.6f}")
    return best_seq, best_val

"""
Propose a non‑negative step function that (approximately) minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

The implementation builds a rich set of seeds (including DPSS windows),
runs a continuation L‑BFGS optimisation on a smooth log‑sum‑exp
approximation of the max‑convolution, and finally refines the best
candidate with a long Adam‑style mirror‑descent on the exact (non‑smooth)
objective.  All vectors are kept on the probability simplex (ℓ₁‑norm = 1);
the evaluation function is invariant to scaling, so the final vector can
be returned directly.  The algorithm respects the 1000 s wall‑time budget
and stores the best sequence in the global name ``height_sequence_1``.
"""

import time
import numpy as np
import scipy.optimize as _opt
from scipy.signal import windows as _windows

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)).
    With sum(seq)=1 this equals the true evaluate_sequence value.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)          # Modified Bessel function I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window – positive, well‑behaved."""
    a = _windows.dpss(n, nw, sym=True)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Low‑frequency cosine mixture, projected onto the simplex."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection (enriched)
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised seed vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # DPSS windows – known to have low sidelobes
    for nw in (2.0, 2.5, 3.0, 3.5, 4.0):
        try:
            seeds.append(_dpss_window(n, nw))
        except Exception:
            pass

    # Random Dirichlet / exponential
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(4):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency curves and mixtures
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution, if any
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final clean‑up (ensure strict ℓ₁ normalisation)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimises a smooth surrogate of the max‑convolution with L‑BFGS‑B.
    Returns a ℓ₁‑normalised probability vector.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0]

    # Initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)                # exact max
            # smooth max via log‑sum‑exp with temperature β
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            smooth = shift + np.log(sum_exp) / beta_val
            obj = 2.0 * n * smooth

            # weights for the smooth max
            w = exp_beta / sum_exp

            # gradient w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule: d obj/d v = J_softmax·grad_a
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Adam‑style mirror‑descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 500_000,
                   lr: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Adam on the logits of a probability vector, using the true max‑convolution
    as objective.  Returns the best sequence seen and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_seq + eps)               # logits
    m = np.zeros_like(v)                      # first moment
    v_hat = np.zeros_like(v)                  # second moment
    beta1 = 0.9
    beta2 = 0.999
    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active autocorrelation entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat_corr = v_hat / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat_corr) + 1e-8)
        v -= step

        # occasional bookkeeping
        if t % 20000 == 0:
            cur_seq = _softmax(v)
            cur_val = _objective_surrogate(cur_seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[Adam] iteration {t:,} new best {best_val:.6f}")

    # final evaluate
    final_seq = _softmax(v)
    final_val = _objective_surrogate(final_seq)
    if final_val < best_val:
        best_seq, best_val = final_seq, final_val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence (list of floats) found before the time budget expires.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.5  # small safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # Quick surrogate evaluation of all seeds
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print("Best surrogate values among seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS on the very best seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate from seeds = {global_best_val:.6f}")

    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS seeds.")
            break
        # allocate a modest number of inner iterations (more if time permits)
        remaining = deadline - time.time()
        max_iter_per_beta = int(min(8000, max(1000, remaining * 0.015 * 1e5)))
        print(f"\n--- L‑BFGS run on seed #{idx + 1} (max_iter/beta={max_iter_per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=max_iter_per_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Long Adam‑style mirror‑descent refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_adam_iters = int(min(5_000_000, remaining * 1500))
        if max_adam_iters > 0:
            print("\n--- Adam‑style refinement ---")
            seq_adam, val_adam = _adam_optimize(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_adam_iters,
                                                lr=0.07)
            print(f"[Adam] best surrogate after refinement = {val_adam:.6f}")
            if val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[Update] improved global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
