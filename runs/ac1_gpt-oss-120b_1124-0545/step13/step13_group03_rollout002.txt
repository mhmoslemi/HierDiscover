"""
Optimise a non‑negative step‑function to minimise

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a))²

The algorithm builds many diverse seeds (analytic windows,
random Dirichlet draws, low‑frequency cosine sums, mixtures),
keeps the most promising ones using the exact surrogate
2·n·max(conv) (equal to the true objective when ∑a = 1),
and then refines the best seed with three stages:

1. **Fast Exponentiated‑Gradient (EG) descent** – a mirror‑descent on the
   probability simplex.  Only a few ×10⁴ iterations are needed to get
   into a good basin.

2. **Soft‑max Adam optimisation** – minimises a smooth surrogate
   ``Fₚ(a)=1/p·log Σexp(p·convₖ(a))``.  The smoothing parameter *p*
   starts modest (≈1) and is increased slowly (factor 1.5 every
   200 k steps).  This lets the optimiser explore a smooth landscape
   for a long time before it tightens to the true max‑convolution.
   Adam’s adaptive step‑size together with an ℓ₁‑projection keeps the
   iterate feasible.

3. **Final EG polishing** – a cheap multiplicative‑update pass that
   cleans up any remaining slack.

All stages respect the global deadline (default 1000 s).  The length
of the window is chosen as a power of two (4096) to make the FFT cheap.
The final sequence is stored in ``globals()["height_sequence_1"]`` for
reuse in later runs.

The whole search is deliberately stochastic (different seeds, random
mixtures) and runs many short restarts if time permits, which helps
escape shallow local minima and typically pushes the bound below the
target 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate (equal to evaluate_sequence when sum(seq)==1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection (large and diverse)
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families with richer parameter sweeps
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # sine windows – sweep gamma
    for gamma in np.linspace(0.5, 4.0, 8):
        seeds.append(_sine_window(n, float(gamma)))

    # Gaussian – sweep sigma fraction
    for sigma in (0.12, 0.18, 0.24, 0.30, 0.36, 0.42):
        seeds.append(_gaussian(n, sigma))

    # Exponential decay – sweep rate
    for rate in (0.90, 0.94, 0.96, 0.98, 0.99, 0.995, 0.998):
        seeds.append(_exponential_decay(n, rate))

    # power law – sweep exponent
    for p in (0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 4.0):
        seeds.append(_power(n, p))

    # beta shape – a grid
    for alpha in (0.3, 0.6, 1.0, 2.0, 3.0):
        for beta in (0.3, 0.6, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, alpha, beta))

    # Tukey windows – sweep alpha
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, float(alpha)))

    # random Dirichlet draws – varied concentration
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform – add a bias towards flatness
    for mix_alpha in (0.1, 0.3, 0.5, 0.7):
        for nf in (2, 3, 5):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # keep previous solution (if any) as a seed
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final sanity check – drop zero‑vectors
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (multiplicative update on simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 200_000,
                      eta0: float = 0.30) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the simplex with KL‑divergence."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # decreasing stepsize – √t schedule
        eta = eta0 / math.sqrt(it)

        # multiplicative update (clip exponent for safety)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # projection onto simplex – simple ℓ₁ scaling
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# Adam optimiser on soft‑max‑smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_smooth_optimize(start: np.ndarray, n: int, deadline: float,
                          init_lr: float = 0.07,
                          beta1: float = 0.9, beta2: float = 0.999,
                          eps: float = 1e-8,
                          p0: float = 1.0,
                          p_factor: float = 1.5,
                          p_step: int = 200_000,
                          max_total_iters: int = 4_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth surrogate
        F_p(a) = (1/p)·log Σ exp(p·convₖ(a)).
    The smoothing parameter p is increased geometrically every `p_step`
    iterations.
    """
    a = start.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    p = p0

    while time.time() < deadline and t < max_total_iters:
        # --- 1) exact convolution and objective ---------------------------------
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # light progress report
            print(f"[Adam] it={t:7d} best={best_val:.6f}  p={p:.2f}")

        # --- 2) soft‑max weights over convolution entries -----------------------
        shifted = conv - max_conv          # these are ≤ 0
        exp_arg = p * shifted
        exp_arg = np.clip(exp_arg, -700, 0)   # avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:                # extreme under‑flow fallback
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # --- 3) gradient of the smooth surrogate (central slice) ---------------
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1: n - 1 + n]   # shape (n,)

        # --- 4) Adam update ----------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = _uniform(n)

        # --- 5) schedule for the smoothing parameter ---------------------------
        if t % p_step == 0:
            p *= p_factor
            print(f"[Adam] increased p to {p:.2f}")

    # final exact evaluation of the best vector found
    final_val = _objective_surrogate(best_a)
    return best_a, final_val


# ----------------------------------------------------------------------
# Simulated‑annealing (lightweight) – optional final tweak
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 300_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99993, min_temp)

        if rng.random() < 0.55:
            # reshuffle a random small subset (Dirichlet re‑allocation)
            k = rng.integers(3, 14)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny mass transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Length choice – a power of two makes FFT cheap and gives enough
    # degrees of freedom for a low autocorrelation peak.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Stage 0 – generate a very diverse set of seeds.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # Evaluate all seeds with the exact surrogate and keep the elite ones.
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    elite_k = 20
    best_seeds = [vec for vec, _ in scored[:elite_k]]
    print(f"Top {elite_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:elite_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – quick EG refinements on a few elite seeds.
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Quick EG refinement (fast pass) ---")
        for idx, seed_vec in enumerate(best_seeds[:5]):   # five fast runs
            rem = deadline - time.time()
            if rem <= 0:
                break
            # allocate a modest number of iterations (≈10 % of remaining time)
            max_it = int(min(120_000, rem * 1500))
            cand, val = _expgrad_optimize(seed_vec, n, deadline,
                                          max_iters=max_it, eta0=0.30)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Quick EG] seed #{idx+1} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long Adam optimisation with a slowly increasing soft‑max.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam smooth optimisation (long run) ---")
        remaining = deadline - time.time()
        # Determine a generous iteration budget based on remaining seconds.
        # Empirically about 12 000 Adam steps per second are affordable.
        max_adam_iters = int(min(4_000_000, remaining * 12_000))
        best_seq, best_val = _adam_smooth_optimize(
            best_seq, n, deadline,
            init_lr=0.07,
            p0=1.0,
            p_factor=1.5,
            p_step=200_000,
            max_total_iters=max_adam_iters)
        print(f"[Adam] best surrogate after long run = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final EG polishing (cheap, stabilises the simplex point).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing ---")
        remaining = deadline - time.time()
        max_it = int(min(60_000, remaining * 3000))
        if max_it > 0:
            best_seq, best_val = _expgrad_optimize(
                best_seq, n, deadline,
                max_iters=max_it,
                eta0=0.25)
            print(f"[Final EG] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – optional simulated‑annealing tweak if time permits.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing final tweak ---")
        remaining = deadline - time.time()
        max_it = int(min(300_000, remaining * 1000))
        if max_it > 0:
            best_seq, best_val = _simulated_annealing(
                best_seq, n, deadline,
                max_iters=max_it)
            print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse in a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
