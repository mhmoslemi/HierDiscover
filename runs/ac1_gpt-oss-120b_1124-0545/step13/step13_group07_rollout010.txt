def mirror_descent(start_seq, deadline, max_iters=10**7):
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _surrogate_obj(a)
    lr0 = 0.5
    power = 0.30
    it = 0
    while it < max_iters and time.time() < deadline:
        conv = _conv_fft(a, a)
        max_idx = np.argmax(conv)
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # gradient g: same approach
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        lr = lr0 / ((it+1) ** power)
        a = a * np.exp(-lr * g)
        a_sum = a.sum()
        if a_sum == 0:
            a = _uniform(n) # fallback
        else:
            a = a / a_sum
        it += 1
    return best_seq, best_val

"""
Propose a step‑function (non‑negative vector) that minimises the official
``evaluate_sequence`` objective.

Approach
--------
1. **Rich seed pool** – deterministic analytic families (uniform,
   triangular, cosine, Gaussian, beta, exponential, power‑law) together
   with many random Dirichlet draws.  Every candidate is ℓ₁‑normalised
   (∑aᵢ=1) so that the official objective is simply
   ``2·n·max(conv(a,a))``.

2. **Select the best few seeds** by the cheap surrogate objective
   ``2·n·max(conv)``.

3. **Long mirror‑descent** – an exponentiated‑gradient sub‑gradient method
   on the exact max‑convolution.  The update is
   ``a ← a·exp(−η·g)`` where ``g`` is the gradient of the active
   convolution entry.  The step‑size η decays as ``η₀·t^{−α}``.
   Each run is limited only by the remaining wall‑clock time.

4. **Random restarts** – after the first mirror run we perturb the best
   sequence slightly, renormalise and run a short mirror descent again.
   This helps escape shallow local minima.

5. **Light SA polishing** – a simple simulated‑annealing that swaps,
   redistributes or transfers a tiny amount of mass between random
   positions, accepting moves with the Metropolis rule.  It runs for the
   remainder of the budget.

The whole routine respects the 1000 s wall‑clock limit, prints progress
and finally registers the best list of floats in the global name
``height_sequence_1`` before returning it.

import time
import numpy as np
from typing import List, Tuple, Optional

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate_obj(seq: np.ndarray, n: int) -> float:
    """
    Fast surrogate objective (Σ seq = 1):
        2·n·max(conv(seq,seq))
    """
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random 3‑segment piece‑wise linear shape."""
    breaks = np.sort(rng.choice(n, size=3, replace=False))
    heights = rng.random(3) + 0.01
    x = np.arange(n, dtype=np.float64)
    a = np.interp(x, breaks, heights)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few randomly positioned Gaussians."""
    a = np.zeros(n, dtype=np.float64)
    num = rng.integers(2, 6)
    for _ in range(num):
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.02 * n, 0.15 * n)
        x = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, nf: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth curve from a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, nf + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: Optional[np.ndarray],
) -> List[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: List[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))

    # parameter sweeps
    for sigma in np.linspace(0.05, 0.35, 8):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 4.5, 10):
        seeds.append(_power_law(n, p))

    alphas = np.linspace(0.2, 3.0, 9)
    betas = np.linspace(0.2, 3.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for lam in np.linspace(0.0005, 0.015, 10):
        seeds.append(_exponential_decay(n, lam))

    # random Dirichlet draws
    concs = np.logspace(-4, 0.5, 12)   # 1e‑4 … ~3.2
    for conc in concs:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixtures and other random shapes
    for _ in range(5):
        seeds.append(_gaussian_mixture(n, rng))

    for _ in range(5):
        seeds.append(_piecewise_linear(n, rng))

    for nf in (2, 3, 5, 7):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mix uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4, 6):
            curve = _lowfreq_curve(n, nf, rng)
            mixed = (1.0 - mix_alpha) * _uniform(n) + mix_alpha * curve
            mixed = mixed / mixed.sum()
            seeds.append(mixed)

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0.0:
            seeds.append(avg / avg.sum())

    return seeds

def _mirror_descent(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    lr0: float = 0.5,
    power: float = 0.30,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    # safety: guarantee exact sum‑1
    a_sum = a.sum()
    if a_sum <= 0.0:
        a = _uniform(n)
    else:
        a = a / a_sum

    best_seq = a.copy()
    best_val = _surrogate_obj(a, n)

    it = 0
    while it < max_iters and time.time() < deadline:
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 250_000 == 0:
                print(f"[Mirror] iter={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / ((it + 1) ** power)
        a = a * np.exp(-lr * g)

        # renormalise onto simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        it += 1

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    init_temp_factor: float = 0.06,
    min_temp: float = 1e-14,
) -> Tuple[np.ndarray, float]:
    """Lightweight SA that swaps or redistributes mass."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate_obj(cur, n)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        # exponential cooling
        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:
            # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            # redistribute mass on a small random subset
            k = rng.integers(2, 14)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total == 0.0:
                it += 1
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            # pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total == 0.0:
                it += 1
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _surrogate_obj(new, n)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 200_000 == 0:
                    print(f"[SA] iter={it:7d} best={new_val:.6f}")

        it += 1

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective.  Returns the best sequence as a plain
    list of floats and registers it globally as ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ---------------------------------------------------------------
    # Problem size – a good trade‑off between resolution and speed
    # ---------------------------------------------------------------
    n = 4000

    # load previous solution if present
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size != n:
                if previous.size > n:
                    previous = previous[:n]
                else:
                    previous = np.pad(previous, (0, n - previous.size), constant_values=0.0)
        except Exception:
            previous = None
    else:
        previous = None

    # ---------------------------------------------------------------
    # Stage 0 – seed generation
    # ---------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _surrogate_obj(vec, n)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _surrogate_obj(global_best_seq, n)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ---------------------------------------------------------------
    # Stage 1 – Mirror‑descent on several promising seeds
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        per_seed_time = remaining / min(len(best_seeds), 5)  # at most 5 seeds
        print("\n--- Mirror‑descent on top seeds ---")
        for idx, seed_vec in enumerate(best_seeds[:5], 1):
            seed_deadline = time.time() + per_seed_time
            seq_m, val_m = _mirror_descent(
                seed_vec,
                n,
                seed_deadline,
                max_iters=5_000_000,
                lr0=0.7,
                power=0.30,
            )
            if val_m < global_best_val:
                global_best_seq = seq_m.copy()
                global_best_val = val_m
                print(f"[Mirror‑seed {idx}] new global best = {val_m:.6f}")
            if time.time() >= deadline:
                break

    # ---------------------------------------------------------------
    # Stage 2 – Extra long mirror‑descent from the current best
    # ---------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long mirror‑descent from current best ---")
        seq_m2, val_m2 = _mirror_descent(
            global_best_seq,
            n,
            deadline,
            max_iters=10_000_000,
            lr0=0.5,
            power=0.33,
        )
        if val_m2 < global_best_val:
            global_best_seq = seq_m2.copy()
            global_best_val = val_m2
            print(f"[Long Mirror] new global best = {val_m2:.6f}")

    # ---------------------------------------------------------------
    # Stage 3 – Small random perturbations + short mirror runs
    # ---------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random‑restart mirror‑descent polishing ---")
        num_restarts = 3
        for r in range(num_restarts):
            if time.time() >= deadline:
                break
            # perturb current best slightly
            pert = global_best_seq + 0.001 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            pert = pert / pert.sum()
            sub_deadline = time.time() + (deadline - time.time()) * 0.3 / (num_restarts - r)
            seq_r, val_r = _mirror_descent(
                pert,
                n,
                sub_deadline,
                max_iters=2_000_000,
                lr0=0.6,
                power=0.30,
            )
            if val_r < global_best_val:
                global_best_seq = seq_r.copy()
                global_best_val = val_r
                print(f"[Restart {r+1}] new global best = {val_r:.6f}")

    # ---------------------------------------------------------------
    # Stage 4 – Simulated‑annealing polishing
    # ---------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq,
            n,
            deadline,
            max_iters=3_000_000,
            init_temp_factor=0.08,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] new global best = {val_sa:.6f}")

    # ---------------------------------------------------------------
    # Final official evaluation (for information only)
    # ---------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
