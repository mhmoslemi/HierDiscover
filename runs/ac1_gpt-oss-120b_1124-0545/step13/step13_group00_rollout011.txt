def _lp_refine(a: np.ndarray, n: int, deadline: float, max_rounds: int = 5, t_frac: float = 0.5) -> np.ndarray:
    '''
    Perform a few iterations of linear programming refinement.
    Each iteration solves:
      minimize t
      s.t. conv(a, g) <= t   (for a subset of k)
           sum(g) == 1
           g >= 0
    The solution g becomes the new a (optionally blended with previous a using t_frac).
    '''
    cur = a.copy()
    cur_surrogate = _objective_surrogate(cur)
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        conv = _conv_fft(cur, cur)
        M = np.max(conv)
        # Select active constraints: indices with conv >= M * 0.99
        threshold = M * 0.99
        active_idxs = np.where(conv >= threshold)[0]
        if len(active_idxs) == 0:
            active_idxs = np.argsort(-conv)[:10]
        # Limit to at most K constraints
        K = min(len(active_idxs), 30)
        # Take top K
        active_idxs = active_idxs[np.argsort(-conv[active_idxs])[:K]]
        # Build LP
        num_vars = n + 1
        c = np.zeros(num_vars, dtype=np.float64)
        c[-1] = 1.0   # minimize t
        # Build A_ub and b_ub
        A_ub = np.zeros((K, num_vars), dtype=np.float64)
        for row_idx, k in enumerate(active_idxs):
            start = max(0, k - n + 1)
            end = min(k, n - 1)
            # columns start..end correspond to g_j
            # coefficient a_{k - j}
            # j indices = range(start, end+1)
            j_range = np.arange(start, end+1, dtype=np.int64)
            coeffs = cur[k - j_range]   # a_{k - j}
            A_ub[row_idx, start:end+1] = coeffs
            # coefficient for t
            A_ub[row_idx, -1] = -1.0
        b_ub = np.zeros(K, dtype=np.float64)
        # Equality sum(g) == 1
        A_eq = np.zeros((1, num_vars), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0], dtype=np.float64)
        # Bounds
        bounds = [(0, None)] * n + [(0, None)]  # t >= 0
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception as e:
            print(f"[LP] linprog exception: {e}")
            break
        if not res.success:
            print("[LP] linprog failed")
            break
        g_new = res.x[:n]
        t_new = res.x[-1]
        # optionally blend
        new_seq = (1 - t_frac) * cur + t_frac * g_new
        # renorm to sum 1 (should be already)
        new_seq = new_seq / new_seq.sum()
        new_surrogate = _objective_surrogate(new_seq)
        if new_surrogate < cur_surrogate - 1e-6:
            cur = new_seq
            cur_surrogate = new_surrogate
            print(f"[LP] round {r+1}: improved surrogate to {cur_surrogate:.6f}, t={t_new:.6f}")
        else:
            print(f"[LP] round {r+1}: no improvement")
            break
    return cur

"""
Propose a high‑quality step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²) ,

i.e. the ℓ∞‑norm of the aperiodic autocorrelation of an ℓ₁‑normalised vector.
The routine combines a rich set of analytic seeds, a continuation L‑BFGS
optimiser on a smooth log‑sum‑exp surrogate, exponentiated‑gradient polishing,
a cheap projected sub‑gradient descent, a small linear‑programming
refinement (LP) that directly reduces the peak cross‑correlation, and finally a
short simulated‑annealing stage.  All stages respect the global 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    _opt = None
    linprog = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² for a (possibly un‑normalised) vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = exact objective for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families of ℓ₁‑normalised vectors (used as seeds)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # Modified Bessel (approximation)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed collection (rich pool)
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a large, diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    # Gaussians
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    seeds.append(_double_gaussian(n, sigma_frac=0.10, shift_frac=0.30))
    seeds.append(_double_gaussian(n, sigma_frac=0.15, shift_frac=0.25))

    # Power‑law
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta‑family grid
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Special alternating‑extrema Dirichlet seed
    seeds.append(_alternating_dirichlet(n, rng))

    # Include previous solution if provided
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Final normalisation pass
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """L‑BFGS‑B on a log‑sum‑exp smoothed max‑convolution objective."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 3000.0, 5000.0]

    # initialise logits (= log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)          # stays on the simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smoothed surrogate (log‑sum‑exp)
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp          # softmax weights over conv entries
            # gradient w.r.t. a: 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)      # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline hit during β={beta_val}")
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Sub‑gradient descent (projected on simplex)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step‑size (tuned a bit higher for faster motion)
        lr = 0.35 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (softmax parametrisation)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (cross‑correlation bound)
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray,
               n: int,
               deadline: float,
               max_rounds: int = 5,
               t_frac: float = 0.3) -> np.ndarray:
    """
    Small LP that directly minimises the peak cross‑correlation
    conv(a, g) while keeping sum(g)=1 and g≥0.  The returned vector replaces
    ``a`` if it yields a better surrogate value.
    """
    if linprog is None:
        print("[LP] SciPy linprog not available – skipping LP refinement.")
        return a

    cur = a.copy()
    cur_val = _objective_surrogate(cur)

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break

        conv = _conv_fft(cur, cur)
        M = np.max(conv)

        # Choose active constraints – those very close to the maximum.
        thresh = M * 0.99
        active_idxs = np.where(conv >= thresh)[0]
        if active_idxs.size == 0:
            active_idxs = np.argsort(-conv)[:10]
        # Limit to at most 30 constraints for speed
        K = min(active_idxs.size, 30)
        active_idxs = active_idxs[np.argsort(-conv[active_idxs])[:K]]

        # Build LP data (variables = g[0..n-1] + t)
        num_vars = n + 1
        c = np.zeros(num_vars, dtype=np.float64)
        c[-1] = 1.0                     # minimise t

        A_ub = np.zeros((K, num_vars), dtype=np.float64)
        for row, k in enumerate(active_idxs):
            start = max(0, k - n + 1)
            end = min(k, n - 1)
            j_range = np.arange(start, end + 1, dtype=np.int64)
            coeffs = cur[k - j_range]          # a_{k-j}
            A_ub[row, start:end + 1] = coeffs
            A_ub[row, -1] = -1.0                # -t

        b_ub = np.zeros(K, dtype=np.float64)

        # Equality: sum(g) = 1
        A_eq = np.zeros((1, num_vars), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0], dtype=np.float64)

        bounds = [(0, None)] * n + [(0, None)]   # t ≥ 0

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception as e:
            print(f"[LP] linprog raised {e}")
            break

        if not res.success:
            print("[LP] linprog failed – status:", res.message)
            break

        g_new = res.x[:n]
        # Combine with current vector (convex blend)
        new_seq = (1.0 - t_frac) * cur + t_frac * g_new
        new_seq = new_seq / new_seq.sum()   # renormalise (should stay ≈1)

        new_val = _objective_surrogate(new_seq)
        if new_val < cur_val - 1e-7:
            print(f"[LP] round {rnd+1}: improved surrogate to {new_val:.6f}")
            cur = new_seq
            cur_val = new_val
        else:
            print(f"[LP] round {rnd+1}: no improvement")
            break

    return cur

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Seed generation
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seed vectors.")

    # cheap surrogate evaluation (exact for ℓ₁‑normalised vectors)
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"Best surrogate among seeds: {evaluated[0][1]:.6f}")

    # --------------------------------------------------------------
    # Initialise global best
    # --------------------------------------------------------------
    global_best = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best)
    print(f"[Init] best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Continuation L‑BFGS on the most promising seeds
    # --------------------------------------------------------------
    if _opt is not None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 3000.0, 5000.0]
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                print("[Time] deadline reached before finishing all L‑BFGS runs.")
                break
            print(f"\n--- L‑BFGS run on seed #{idx+1} ---")
            seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline,
                                             beta_schedule=beta_schedule,
                                             max_iter_per_beta=2500)
            val_opt = _objective_surrogate(seq_opt)
            print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

            if val_opt < global_best_val:
                global_best = seq_opt.copy()
                global_best_val = val_opt
                print(f"[Update] new global best surrogate = {global_best_val:.6f}")

            # short exponentiated‑gradient polishing
            if time.time() < deadline - 0.5:
                max_exp = int(min(150_000,
                                  (deadline - time.time()) * 0.06 * 1e5))
                seq_exp, val_exp = _exp_gradient_opt(global_best, n,
                                                     deadline,
                                                     max_iters=max_exp)
                if val_exp < global_best_val:
                    global_best = seq_exp.copy()
                    global_best_val = val_exp
                    print(f"[ExpGrad] improved to {global_best_val:.6f}")
    else:
        print("[Info] SciPy optimiser not available – skipping L‑BFGS stage.")

    # --------------------------------------------------------------
    # Long sub‑gradient polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(600_000, remaining * 0.07 * 1e5))
        print("\n--- Sub‑gradient polishing ---")
        seq_sub, val_sub = _subgradient_opt(global_best, n,
                                            deadline,
                                            max_iters=max_sub)
        if val_sub < global_best_val:
            global_best = seq_sub.copy()
            global_best_val = val_sub
            print(f"[SubGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Linear‑programming refinement (cross‑correlation bound)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- LP refinement ---")
        seq_lp = _lp_refine(global_best, n, deadline,
                            max_rounds=5, t_frac=0.3)
        lp_val = _objective_surrogate(seq_lp)
        if lp_val < global_best_val:
            global_best = seq_lp.copy()
            global_best_val = lp_val
            print(f"[LP] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Optional final simulated‑annealing style refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(300_000, remaining * 0.06 * 1e5))
        if max_sa > 0:
            print("\n--- Simulated‑annealing refinement ---")
            logits = np.log(global_best + 1e-30)
            cur_seq = global_best.copy()
            cur_val = global_best_val
            best_seq = cur_seq.copy()
            best_val = cur_val

            for it in range(1, max_sa + 1):
                if time.time() > deadline:
                    break
                progress = it / max_sa
                T = 0.06 * (1.0 - progress) + 1e-6
                sigma = 0.20 * (1.0 - progress) + 1e-4

                proposal_logits = logits + sigma * rng.normal(size=n)
                proposal_seq = _softmax(proposal_logits)
                proposal_val = _objective_surrogate(proposal_seq)

                if (proposal_val < cur_val or
                    math.exp(-(proposal_val - cur_val) / T) > rng.random()):
                    logits = proposal_logits
                    cur_seq = proposal_seq
                    cur_val = proposal_val
                    if cur_val < best_val:
                        best_val = cur_val
                        best_seq = cur_seq.copy()

                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

            if best_val < global_best_val:
                global_best = best_seq.copy()
                global_best_val = best_val
                print(f"[SA] final best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best.tolist()
    return global_best.tolist()
