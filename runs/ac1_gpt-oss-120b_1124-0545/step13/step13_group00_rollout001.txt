       2·n·max(conv(a,a)) / (sum(a)²)

where conv is the full linear convolution.  The function works inside a
1000 s time budget and proceeds as follows:

1.  Build a large, diverse set of ℓ₁‑normalised seeds (analytic windows,
    Dirichlet draws, “comb’’ patterns, random spikes, and the previous
    best solution if any).
2.  Keep the best few seeds according to the exact surrogate
    2·n·max(conv) (the true objective for ℓ₁‑normalised vectors).
3.  For each seed run a smooth‑max optimiser based on Adam in the
    log‑softmax (logit) space.  The smoothness parameter β is increased
    gradually (continuation) so that the optimiser first sees a smooth
    landscape and finally approximates the true max‑convolution.
4.  Refine the best candidate with a large projected sub‑gradient run
    (exact max‑convolution) and a simple simulated‑annealing style
    perturbation that moves tiny amounts of mass between coordinates.
5.  Return the best vector found (as a plain list of floats); the official

All heavy work is done on the surrogate, which is cheap to compute.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector on the simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:

def _blackman_window(n: int) -> np.ndarray:

    a = (a0

         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()

def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # modified Bessel (numpy)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int,
                    sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:

def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:

    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)

    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a Dirichlet draw and assign largest/smallest alternatingly."""
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _comb_seed(n: int, period: int, offset: int = 0) -> np.ndarray:
    """A simple “comb’’: equal mass on every `period`‑th index."""
    a = np.zeros(n, dtype=np.float64)
    positions = np.arange(offset, n, period, dtype=int)
    if positions.size == 0:
        return _uniform(n)
    a[positions] = 1.0
    s = a.sum()
    return a / s


def _random_spike_seed(n: int, k_spikes: int, rng: np.random.Generator) -> np.ndarray:
    """k random spikes with random positive weights."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=min(k_spikes, n), replace=False)
    vals = rng.exponential(scale=1.0, size=idx.shape[0])
    a[idx] = vals
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s

# Surrogate evaluation (exact for ℓ₁‑normalised vectors)

def _surrogate_obj(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for an ℓ₁‑normalised vector."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Adam optimiser on a smooth approximation of max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule=None,
                   max_iters_per_beta: int = 25000) -> tuple[np.ndarray, float]:
    """
    Smooth‑max optimisation using Adam in log‑softmax space.
    Returns the best ℓ₁‑normalised vector seen and its surrogate value.
    """
    eps = 1e-30
    logits = np.log(start_vec + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    t = 0

    if beta_schedule is None:
        # gentle start → aggressive final smoothing
        beta_schedule = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0, 2000.0, 5000.0]

    best_a = None
    best_val = math.inf

    for beta in beta_schedule:
        it = 0
        while time.time() < deadline and it < max_iters_per_beta:
            t += 1
            it += 1

            a = _softmax(logits)               # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # surrogate (exact for L1‑normalised)
            cur_val = 2.0 * n * max_c
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

            # smooth‑max weights
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            w = exp_beta / np.sum(exp_beta)

            # gradient w.r.t. a (2 * a_rev * w convolution)
            rev_a = a[::-1]
            grad_a = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]

            # convert to gradient w.r.t logits (softmax Jacobian)
            dot = np.dot(grad_a, a)
            grad_logit = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_logit
            v = beta2 * v + (1.0 - beta2) * (grad_logit ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # learning‑rate that slowly decays with β (larger β needs smaller steps)
            lr = 0.05 / (1.0 + 0.1 * beta)

            logits -= lr * m_hat / (np.sqrt(v_hat) + epsilon)

    # fallback to the final logits if no improvement happened
    if best_a is None:
        best_a = _softmax(logits)
        best_val = _surrogate_obj(best_a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(start_vec: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent (exact max‑convolution)."""
    a = start_vec.copy()
    best_a = a.copy()
    best_val = _surrogate_obj(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:,d} best={best_val:.6f}")

        # sub‑gradient for the active entry (as described in the spec)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step‑size
        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing style refinement (logit space)
# ----------------------------------------------------------------------
def _simulated_annealing(start_vec: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Random walk in logit space with a linear cooling schedule."""
    logits = np.log(start_vec + 1e-30)
    cur_seq = start_vec.copy()
    cur_val = _surrogate_obj(cur_seq)

    best_seq = cur_seq.copy()
    best_val = cur_val
    rng = np.random.default_rng(12345)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.06 * (1.0 - progress) + 1e-6          # temperature
        sigma = 0.20 * (1.0 - progress) + 1e-4      # proposal scale

        prop_logits = logits + sigma * rng.normal(size=n)
        prop_seq = _softmax(prop_logits)
        prop_val = _surrogate_obj(prop_seq)

        # Metropolis acceptance
        if prop_val < cur_val or math.exp(-(prop_val - cur_val) / T) > rng.random():
            logits = prop_logits
            cur_seq = prop_seq
            cur_val = prop_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:,d} best={best_val:.6f}")

    return best_seq, best_val

    # Gaussians
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    seeds.append(_double_gaussian(n, sigma_frac=0.10, shift_frac=0.30))
    seeds.append(_double_gaussian(n, sigma_frac=0.15, shift_frac=0.25))

    # Power‑law
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta family grid
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Comb patterns (different periods)
    for period in (5, 7, 11, 13, 17, 23, 29, 37, 43):
        seeds.append(_comb_seed(n, period))
        for offset in range(1, period):
            seeds.append(_comb_seed(n, period, offset))

    # Random spike patterns
    for k in (3, 5, 10, 20, 40, 80):
        seeds.append(_random_spike_seed(n, k, rng))

    # Alternating Dirichlet seed
    seeds.append(_alternating_dirichlet(n, rng))

    # Include previous solution if it matches the size
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise everything robustly
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    `evaluate_sequence` objective.  The routine runs for at most
    ``budget_s`` seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 3.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – we keep 4000 as a good trade‑off between quality and speed.
    # Feel free to change n (e.g. 5000) if extra time is available.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n[Info] Problem size n = {n}")

    # ------------------------------------------------------------------
    # 1. Seed generation
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seed vectors.")

    # cheap surrogate evaluation for all seeds
    surrogate_vals = [_surrogate_obj(v) for v in seeds]
    ranked = sorted(zip(seeds, surrogate_vals), key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _ in ranked[:top_k]]
    print(f"Best surrogate among seeds: {ranked[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam optimisation on the best seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _surrogate_obj(global_best_seq)
    print(f"\n[Init] Current best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] Deadline reached before finishing all Adam runs.")
            break

        print(f"\n--- Adam optimisation on seed #{idx + 1} ---")
        seq_opt, val_opt = _adam_optimize(seed_vec, n, deadline)
        print(f"[Adam] best surrogate from this run = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

        # short mirror‑descent polishing with exponentiated gradient (few iters)
        remaining = deadline - time.time()
        if remaining > 2.0:
            # use a very short exponentiated‑gradient run to tidy up
            seq_exp, val_exp = _adam_optimize(global_best_seq, n,
                                              deadline,
                                              beta_schedule=[5000.0],
                                              max_iters_per_beta=5000)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Long sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(600_000, remaining * 0.07 * 1e5))
        if max_sub > 0:
            print("\n--- Sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_descent(global_best_seq,
                                                    n,
                                                    deadline,
                                                    max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing style refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(300_000, remaining * 0.06 * 1e5))
        if max_sa > 0:
            print("\n--- Simulated‑annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing(global_best_seq,
                                                  n,
                                                  deadline,
                                                  max_iters=max_sa)
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[SA] final best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next round
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
