Key new function: `_find_optimal_exponential(n, deadline)` performing golden-section search:

```
def _exponential_obj(r, n):
    # compute conv max as described
    eps = 1-r
    if eps <= 0 or eps >= 1:
        return float('inf')
    # compute const term
    rn = r**n
    const = (eps**2) / ((1 - rn) ** 2)
    # find k_star
    # using -1/log(r) - 1.
    k_star = int(np.floor(-1.0 / np.log(r) - 1.0))
    k_star = max(0, min(n-1, k_star))
    # evaluate at k_star and maybe k_star+1
    # define helper to compute product
    def val_at(k):
        return (k+1) * (r**k)
    v1 = val_at(k_star)
    v2 = val_at(min(k_star+1, n-1))
    max_val = max(v1, v2)
    conv_max = const * max_val
    return 2.0 * n * conv_max
```

"""
Propose a high‑quality step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Rich seed generation** – create many ℓ₁‑normalised candidates from
   analytically‑motivated families (uniform, triangular, sine‑window,
   Gaussian, power‑law, Beta, exponential).  For the exponential family
   a fast 1‑D golden‑section search is performed to obtain the (near‑optimal)
   decay factor *r* for the given length *n*.

2. **Cheap surrogate ranking** – evaluate the cheap surrogate
   ``2·n·max(conv(a,a))`` (exact for ℓ₁‑normalised vectors) on all seeds
   and keep the best *k* of them.

3. **Long Adam optimisation** – starting from each of the best seeds run a
   log‑softmax Adam optimiser that smooths the max with a slowly increasing
   temperature (log‑sum‑exp).  The total number of Adam iterations is
   limited only by the remaining time budget.

4. **Sub‑gradient / exponentiated‑gradient polishing** – use exact
   sub‑gradients of the true max‑convolution objective with Euclidean
   projection onto the simplex, followed by mirror‑descent (exponentiated
   gradient) to further decrease the objective.

5. **Simulated‑annealing final sweep** – aggressive random reshuffling of
   the mass on small subsets, accepting uphill moves with a cooling schedule.

6. **Final evaluation** – call ``evaluate_sequence`` on the best vector,
   store it in the global ``height_sequence_1`` for reuse and return it.

All steps respect the global 1000 s deadline (a small safety margin is kept
for the final evaluation).  The implementation uses only NumPy and the
standard library; no I/O is performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities (FFT‑based convolution, simplex ops)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector
    (∑seq = 1).  The true ``evaluate_sequence`` value equals this surrogate
    because the denominator is 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_shape(n: int, r: float) -> np.ndarray:
    """Monotone exponential a_i ∝ r**i, ℓ₁‑normalised."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Optimisation primitives (identical to the baseline, with a few tweaks)
# ----------------------------------------------------------------------
def _logspace_adam_optimize_long(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam on the log‑softmax parametrisation with a log‑sum‑exp smoothing of
    the max (β increases up to 2000).  Returns the best vector seen and its
    surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5               # start temperature
    beta_inc = 0.5
    beta_max = 2000.0

    lr0 = 1.0                # initial learning‑rate in log‑space

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 10000 == 0:
                print(f"[Adam] iter={it:7d} best={cur_val:.6f}")

        # smoothed max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:   # should not happen
            continue
        w = w / w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2 (unused tail trimmed)
        grad_a = 2.0 * full[n - 1: n - 1 + n]

        # Jacobian of log‑softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / math.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Euclidean projection onto the simplex after each step.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubProj] iter={it:7d} best={cur_val:.6f}")

        # active sub‑gradient (only overlapping part)
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100000 == 0:
                print(f"[ExpGrad] iter={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
    init_temp_factor: float = 0.06,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Aggressive annealing that reshuffles mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cool down
        T = max(T * 0.9999, min_temp)

        if rng.random() < 0.72:          # reshuffle a random subset
            k = rng.integers(2, 12)          # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                           # simple pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)  # ≤5 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100000 == 0:
                    print(f"[SA] iter={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponential family helper – fast analytic surrogate
# ----------------------------------------------------------------------
def _exp_surrogate_obj(r: float, n: int) -> float:
    """Exact surrogate 2·n·max(conv) for an exponential vector a_i ∝ r^i."""
    if not (0.0 < r < 1.0):
        return float("inf")
    eps = 1.0 - r
    # constant factor due to ℓ₁‑normalisation on a truncated exponential
    rn = r**n
    const = (eps * eps) / ((1.0 - rn) * (1.0 - rn))

    # location of the maximal term (continuous approx)
    # k* ≈ 1/eps - 1  (clamped to [0, n‑1])
    k_star = int(min(n - 1, max(0, math.floor(1.0 / eps - 1.0))))

    # evaluate at k_star and its neighbour (discrete effect)
    def prod(k):
        return (k + 1) * (r**k)

    val0 = prod(k_star)
    val1 = prod(min(k_star + 1, n - 1))
    max_term = max(val0, val1)

    conv_max = const * max_term
    return 2.0 * n * conv_max


def _find_optimal_exponential(n: int, deadline: float) -> tuple[float, float]:
    """Golden‑section search for the r ∈ (0,1) that minimises the exponential surrogate."""
    # Search bracket – the optimum is always very close to 1.
    lo = 0.5
    hi = 0.9999995
    phi = (math.sqrt(5) - 1) / 2   # 0.618...

    # Initial points
    x1 = lo + (1 - phi) * (hi - lo)
    x2 = lo + phi * (hi - lo)
    f1 = _exp_surrogate_obj(x1, n)
    f2 = _exp_surrogate_obj(x2, n)

    while time.time() < deadline - 0.5 and hi - lo > 1e-12:
        if f1 > f2:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + phi * (hi - lo)
            f2 = _exp_surrogate_obj(x2, n)
        else:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = lo + (1 - phi) * (hi - lo)
            f1 = _exp_surrogate_obj(x1, n)

    r_opt = (lo + hi) / 2.0
    best_val = _exp_surrogate_obj(r_opt, n)
    return r_opt, best_val

# Seed generation – exhaustive sweep over families
# ----------------------------------------------------------------------
def _collect_rich_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Sine‑window family (fine grid)
    for gamma in np.linspace(0.3, 3.0, 30):
        seeds.append(_sine_window(n, gamma=gamma))

    # Gaussian family
    for sigma in np.linspace(0.05, 0.35, 20):
        seeds.append(_gaussian(n, sigma_frac=sigma))

    # Power‑law family
    for p in np.linspace(0.2, 5.0, 30):
        seeds.append(_power(n, p=p))

    # Beta‑shape grid (log‑space)
    alphas = np.logspace(np.log10(0.2), np.log10(5.0), 12)
    for a in alphas:
        for b in alphas:
            seeds.append(_beta_shape(n, alpha=a, beta=b))

    # Exponential – find near‑optimal r
    r_opt, _ = _find_optimal_exponential(n, time.time() + 3.0)  # quick search
    # generate a handful of r around the optimum
    for mul in (0.9, 0.95, 1.0, 1.05, 1.1):
        r = max(0.0, min(0.999999, r_opt * mul))
        seeds.append(_exponential_shape(n, r))

    # Random Dirichlet draws – add diversity
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # Random exponential draws (un‑normalised)
    for _ in range(5):
        a = rng.exponential(scale=1.0, size=n)
        a /= a.sum()
        seeds.append(a)

    # Previous solution if present
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Final normalisation guard
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
