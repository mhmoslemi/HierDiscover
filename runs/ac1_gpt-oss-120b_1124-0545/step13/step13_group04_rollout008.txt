"""
Propose a step‑function (non‑negative heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The evaluation routine is homogeneous, so we may work with a
probability vector (∑a = 1).  The algorithm consists of two phases:

1. **Randomised sampling**
   – generate many sparse candidates with random support sizes and
     random positive weights (exponential law);
   – keep the best candidate according to the fast surrogate
     2·n·max(conv).

2. **Smooth‑gradient refinement**
   – minimise a smoothed version of the max‑convolution:
       L(a) = (1/λ)·log∑_k exp(λ·(a*a)_k) .
     Its gradient is
       ∇L = 2·(a ⋆ w) ,   w_i = exp(λ·c_i)/∑_j exp(λ·c_j)
     where c = a*a is the self‑convolution and «⋆» denotes
     a linear cross‑correlation.
   – the gradient is obtained by FFTs, after which a projected
     (Euclidean) gradient step keeps the vector on the simplex.
   – λ is increased slowly so the optimisation starts with a
     smooth landscape and ends focusing on the true maximum.

The whole routine respects the time budget (default 1000 s) and
stores the best sequence globally as ``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric primitives
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _cross_corr(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (non‑circular) of real vectors.
    (x ⋆ y)[i] = Σ_j x[j]·y[i+j] .
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    corr = np.fft.irfft(fx * np.conj(fy), n=N)[:L]
    return corr


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (∑v = 1, v ≥ 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _surrogate(a: np.ndarray, n: int) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


def _smooth_gradient(a: np.ndarray, lam: float):
    """
    Compute the value of max(conv) and the gradient of the
    smoothed max (log‑sum‑exp) for the current vector ``a``.
    Returns (max_conv, grad) where ``grad`` has the same shape as ``a``.
    """
    # self‑convolution
    conv = _conv_fft(a, a)               # length 2·n‑1
    max_conv = np.max(conv)

    # soft‑max weights over the convolution entries
    shift = conv - np.max(conv)           # for numerical stability
    w = np.exp(lam * shift)
    w /= np.sum(w)                        # shape (2·n‑1,)

    # gradient: 2·(a ⋆ w)
    grad = 2.0 * _cross_corr(a, w)[: len(a)]
    return max_conv, grad

        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and np.sum(arr) > 0.0:
                arr = arr / np.sum(arr)          # normalise to a probability vector
                val = _surrogate(arr, n)
                best_seq, best_surrogate = arr, val
                print(f"[Init] previous solution surrogate = {val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Uniform distribution as a cheap baseline
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    val_uni = _surrogate(uniform, n)
    if val_uni < best_surrogate:
        best_seq, best_surrogate = uniform, val_uni
        print(f"[Init] uniform surrogate = {val_uni:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – random sparse candidates (fast surrogate evaluation)
    # ------------------------------------------------------------------
    sample_iter = 0
    # leave about 150 s for the refinement phase
    sample_deadline = deadline - 150.0
    while time.time() < sample_deadline:
        # random support size between 30 % and 90 % of n
        frac = rng.uniform(0.30, 0.90)
        S = max(2, int(frac * n))
        # random support and random positive weights (exponential law)
        idx = rng.choice(n, size=S, replace=False)
        weights = rng.exponential(scale=1.0, size=S)
        a = np.zeros(n, dtype=np.float64)
        a[idx] = weights
        a /= a.sum()                     # ℓ₁‑normalise
        val = _surrogate(a, n)
        sample_iter += 1
        if val < best_surrogate:
            best_surrogate = val
            best_seq = a.copy()
            print(f"[Sample] iter={sample_iter:6d} new best surrogate = {val:.6f}")
        if sample_iter % 5000 == 0:
            elapsed = time.time() - start_time
            print(f"[Sample] {sample_iter} candidates, best={best_surrogate:.6f}, elapsed={elapsed:.1f}s")

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑gradient refinement on the best candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        # should never happen, but fall back to uniform
        best_seq = uniform.copy()
        best_surrogate = _surrogate(best_seq, n)

    a = best_seq.copy()
    best_local_seq = a.copy()
    best_local_val = best_surrogate

    lam = 5.0           # initial smoothness parameter
    lam_max = 50.0
    step0 = 0.5         # base step size

    it = 0
    print("\n--- Smooth‑gradient refinement (≈ remaining time) ---")
    while time.time() < deadline - 1.0:
        it += 1
        max_conv, grad = _smooth_gradient(a, lam)
        cur_surrogate = 2.0 * n * max_conv
        if cur_surrogate < best_local_val:
            best_local_val = cur_surrogate
            best_local_seq = a.copy()
            if it % 1000 == 0:
                print(f"[Refine] iter={it:6d} new best surrogate = {cur_surrogate:.6f}")

        # diminishing step size
        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = _proj_simplex(a)          # keep on the probability simplex

        # slowly increase the smoothness (focus on the true max)
        if lam < lam_max:
            lam = min(lam_max, lam * 1.001)

    # keep any improvement obtained during refinement
    if best_local_val < best_surrogate:
        best_surrogate = best_local_val
        best_seq = best_local_seq.copy()
        print(f"[Refine] final best surrogate = {best_local_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (the only expensive call)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_surrogate:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
