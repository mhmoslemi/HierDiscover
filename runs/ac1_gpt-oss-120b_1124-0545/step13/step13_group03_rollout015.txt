def build_sidon_set(n, rng, max_trials=...):
    best_set = []
    best_len = 0
    for trial in range(num_trials):
        candidate_indices = list(range(n))
        rng.shuffle(candidate_indices)
        sums_taken = np.zeros(2*n-1, dtype=bool)
        current_set = []
        for idx in candidate_indices:
            # compute sums with existing indices.
            collision = False
            # check 2*idx (self-sum)
            if sums_taken[2*idx]:
                collision = True
            else:
                for j in current_set:
                    if sums_taken[idx + j]:
                        collision = True
                        break
            if not collision:
                current_set.append(idx)
                sums_taken[2*idx] = True
                for j in current_set[:-1]: # all previous j
                    sums_taken[idx + j] = True
        if len(current_set) > best_len:
            best_set = current_set
            best_len = len(current_set)
            # optionally break early if close to theoretical max.
    return best_set

def find_large_sidon(n, rng, time_limit):
    best_set = []
    best_len = 0
    attempts = 0
    while time.time() < time_limit:
        # random order
        indices = np.arange(n)
        rng.shuffle(indices)
        used = np.zeros(2*n-1, dtype=bool)
        S = []
        for idx in indices:
            # quick test: check used[2*idx]:
            if used[2*idx]:
                continue
            collision = False
            for j in S:
                if used[idx + j]:
                    collision = True
                    break
            if not collision:
                # add idx
                S.append(idx)
                used[2*idx] = True
                for j in S[:-1]:
                    used[idx + j] = True
                # optional early break if we reached theoretical max
                if len(S) == (int(np.floor((-1 + np.sqrt(1+16*n-8))/2))):
                    break
        if len(S) > best_len:
            best_len = len(S)
            best_set = list(S)
            print(f"[Sidon] New best size {best_len} after {attempts} attempts")
            if best_len >= int(np.ceil(np.sqrt(4*n/1.503))):
                # already good enough perhaps break? Keep trying for better.
                pass
        attempts += 1
    return best_set

"""
Propose a step‑function with a very low evaluation value.

Algorithm (Sidon‑type construction)
------------------------------------
1. **Goal** – minimise 2·n·max(conv(a,a)) / (sum(a))².
   For a vector `a` with total mass 1 the objective is
        2·n·max(conv(a,a)).
   In the convolution each entry `k` contains the sum of all ordered
   products a_i·a_{k‑i}.  If we restrict the support of `a` to a set
   `S` of indices and assign the same weight `w = 1/|S|` to every
   selected position, then for a sum `k` the contribution is
        C_k · w²
   where `C_k` is the number of *ordered* pairs (i,j)∈S×S with i+j=k.
   If we keep the number of *unordered* pairs per sum bounded by a
   small constant `C_MAX` then   max(conv) ≤ 2·C_MAX·w².

2. **Combinatorial construction** – we build a large set `S ⊂ {0,…,n‑1}`
   such that every unordered pair has a distinct (or almost distinct)
   sum.  Concretely we maintain an array `cnt[s]` of how many unordered
   pairs currently use the sum `s`.  A new index `i` is accepted only if
   `cnt[i+i] < C_MAX` and `cnt[i+j] < C_MAX` for all already chosen
   `j`.  This guarantees `C_k ≤ C_MAX` for every `k`.

3. **Random greedy search** – many random permutations of the indices
   are tried while time permits; the largest feasible set found is kept.
   The theoretical upper bound for a given `C_MAX` is
        m·(m+1)/2 ≤ C_MAX·(2·n‑1)
   (because there are `2·n‑1` possible sums).  With `n=4000` and
   `C_MAX=2` the bound is `m ≤ 178`.  Sets with `m≈150` already give
   an objective ≈ 1.4, comfortably beating the previous 1.508 result.

4. **Final sequence** – a uniform mass `w = 1/m` is placed on the
   indices of the best set.  This leaves the sum exactly 1, respects
   the non‑negativity and clipping constraints, and yields a provably
   low `max(conv)`.  The sequence is returned as a plain list.

The whole routine respects the global 1000 s deadline, prints progress
and the final evaluation, and stores the result in the global variable
`height_sequence_1` for possible reuse in later runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: build a large set with bounded pair‑sum multiplicity
# ----------------------------------------------------------------------
def _greedy_bounded_sum_set(n: int, c_max: int, rng: np.random.Generator,
                            deadline: float, max_attempts: int = 5000):
    """
    Random‑greedy construction of a subset S⊆{0,…,n‑1} such that
    every unordered pair {i,j} (i≤j) contributes to the sum i+j at most
    `c_max` times.  Returns the best set found before `deadline`.
    """
    best_S = []
    # theoretical maximum for given c_max (simple counting bound)
    max_possible = int((math.isqrt(1 + 8 * c_max * (2 * n - 1)) - 1) // 2)

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        attempts += 1
        # random order of candidates
        order = np.arange(n, dtype=np.int32)
        rng.shuffle(order)

        cnt = np.zeros(2 * n - 1, dtype=np.int16)   # how many unordered pairs per sum
        S = []

        for idx in order:
            # self‑sum (i,i) uses the slot 2*idx
            if cnt[idx + idx] >= c_max:
                continue

            # check all pair sums with already selected indices
            ok = True
            for j in S:
                if cnt[idx + j] >= c_max:
                    ok = False
                    break
            if not ok:
                continue

            # add idx to the set and update counters
            for j in S:
                cnt[idx + j] += 1          # one more unordered pair using this sum
            cnt[idx + idx] += 1
            S.append(idx)

            # early stop when we cannot improve beyond the absolute bound
            if len(S) == max_possible:
                break

        if len(S) > len(best_S):
            best_S = list(S)   # copy
            print(f"[Greedy] attempt {attempts:5d} – new best size {len(best_S)}")
            # if we already hit the theoretical bound we can stop
            if len(best_S) >= max_possible:
                print("[Greedy] reached theoretical maximum – stopping search.")
                break

    return best_S
