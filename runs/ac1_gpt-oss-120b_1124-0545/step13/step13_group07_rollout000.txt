"""
search_proposal.py

A new search routine for the “step‑function” optimisation problem.

Algorithm Overview
------------------
1. **Seed preparation** – reuse the previous best solution (if any) and a
   uniform distribution as the initial log‑probability vector.

2. **Simple Evolution Strategy (ES)** – work in the log‑softmax (probability
   simplex) space.  In each generation we draw λ≈80 offspring,
   evaluate the cheap surrogate `2·n·max(conv(a,a))`, keep the best 20 % and
   replace the mean log‑vector by the average of those elites.
   The step‑size σ is adapted by the 1/5‑success rule.

3. **Mirror‑descent polishing** – start the exact sub‑gradient exponentiated‑gradient
   optimiser on the best ES candidate (the same routine that was used in the
   previous state‑of‑the‑art).

4. **LP polishing** – a handful of linear‑programming refinements as in the
   reference implementation (maximise Σ g_i under a convolution bound).

5. **Final Simulated‑Annealing** – lightweight SA for a last set of local
   improvements.

All stages respect the global time‑budget (default 1000 s).  The best
sequence found is stored in the global name ``height_sequence_1`` and
returned as a plain list of ``float``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Convolution utilities (FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate_obj(seq: np.ndarray) -> float:
    """
    Fast surrogate objective for a sequence that is ℓ₁‑normalised (Σseq = 1):

        2·n·max(conv(seq,seq))

    This is exactly the official objective when Σseq = 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple probability helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _uniform(n: int) -> np.ndarray:
    """Uniform probability vector of length n."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Optimisers from the reference solution (kept unchanged)
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                max_iters: int = 4_000_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror) sub‑gradient descent on the exact max."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    lr0 = 0.25
    power = 0.28

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _lp_refine(seq: np.ndarray, deadline: float,
               max_iters: int = 12, tau: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Perform several LP refinements.

    Each iteration solves:
        maximise   Σ g_i
        subject to conv(seq, g) ≤ M   (where M = max conv(seq, seq))
                  g_i ≥ 0
    The solution g is mixed back into the current sequence with weight τ
    and renormalised.
    """
    f = seq.copy()
    n = len(f)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        M = _conv_fft(f, f).max()

        # Build convolution matrix for current f (sparse, exactly n² non‑zeros)
        rows = np.empty(n * n, dtype=np.int32)
        cols = np.empty(n * n, dtype=np.int32)
        data = np.empty(n * n, dtype=np.float64)

        idx = 0
        for k in range(n):
            start = k
            end = k + n
            rows[idx:idx + n] = np.arange(start, end, dtype=np.int32)
            cols[idx:idx + n] = k
            data[idx:idx + n] = f
            idx += n

        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if not res.success:
            print(f"[LP] iteration {it} failed – stopping LP stage.")
            break

        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g <= sum_f + 1e-12:
            # no increase in ℓ₁‑norm → nothing to gain
            break

        f_new = (1.0 - tau) * f + tau * g
        f_new = f_new / f_new.sum()
        f = f_new

        cur_obj = _surrogate_obj(f)
        print(f"[LP] iteration {it} – objective = {cur_obj:.6f}")

    return f, _surrogate_obj(f)


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                 max_iters: int = 2_500_000,
                                 init_temp_factor: float = 0.06,
                                 min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """Lightweight SA with three move types."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate_obj(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # gentle exponential cooling
        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:
            # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            # redistribute mass on a small random subset
            k = rng.integers(2, 14)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total == 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            # pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total == 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _surrogate_obj(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Evolution Strategy (ES) core
# ----------------------------------------------------------------------
def _evolution_strategy(mu: np.ndarray, sigma: float, n: int,
                        deadline: float, best_seq: np.ndarray,
                        best_val: float) -> tuple[np.ndarray, float]:
    """
    Simple (μ/λ,λ)‑ES on the log‑probability simplex.
    Returns the best sequence and its surrogate value.
    """
    rng = np.random.default_rng()
    generation = 0
    eps = 1e-12

    while time.time() < deadline:
        # number of offspring – fixed (tunable)
        lam = 80
        cand_vals = []
        cand_vecs = []

        # generate offspring
        for _ in range(lam):
            l = mu + sigma * rng.standard_normal(n)
            a = _softmax(l)
            val = _surrogate_obj(a)
            cand_vecs.append(a)
            cand_vals.append(val)
            if val < best_val:
                best_val = val
                best_seq = a.copy()

        # elite selection
        elite_frac = 0.20
        elite_k = max(1, int(lam * elite_frac))
        elite_idx = np.argsort(cand_vals)[:elite_k]
        elite_logs = np.log(np.stack([cand_vecs[i] + eps for i in elite_idx], axis=0))
        mu = np.mean(elite_logs, axis=0)

        # success‑rate adaptation (1/5 rule)
        cur_mu_seq = _softmax(mu)
        cur_mu_val = _surrogate_obj(cur_mu_seq)
        successes = sum(1 for v in cand_vals if v < cur_mu_val)
        success_rate = successes / lam
        if success_rate > 0.20:
            sigma *= 1.5
        elif success_rate < 0.20:
            sigma *= 0.85
        sigma = max(sigma, 0.01)

        generation += 1
        if generation % 5 == 0:
            print(f"[ES] gen={generation:4d} best={best_val:.6f} sigma={sigma:.4f}")

        # safety – break if we are very close to the deadline
        if time.time() > deadline - 2.0:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective.

    Returns the best sequence (list of float) and registers it globally as
    ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a small safety margin

    # problem size – 4000 works well for the benchmark
    n = 4000

    # ------------------------------------------------------------------
    # Initialise from a previous solution if it exists, otherwise by uniform.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                if prev_arr.size > n:
                    prev_arr = prev_arr[:n]
                else:
                    prev_arr = np.pad(prev_arr, (0, n - prev_arr.size), constant_values=0.0)
            s = prev_arr.sum()
            if s > 0.0:
                best_seq = prev_arr / s
                best_val = _surrogate_obj(best_seq)
                print(f"[Init] Loaded previous solution, surrogate = {best_val:.6f}")
            else:
                raise ValueError
        except Exception:
            previous = None

    if previous is None:
        best_seq = _uniform(n)          # start from uniform (Σ = 1)
        best_val = _surrogate_obj(best_seq)
        print(f"[Init] Starting from uniform, surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Evolution Strategy on log‑softmax parameters
    # ------------------------------------------------------------------
    # initialise log‑mean (mu) – log‑softmax of the current best point
    mu = np.log(np.maximum(best_seq, 1e-12))
    sigma = 1.0                        # initial std‑dev in log‑space

    print("\n--- Evolution Strategy (ES) ---")
    best_seq, best_val = _evolution_strategy(mu, sigma, n, deadline, best_seq, best_val)
    print(f"[ES] Global best surrogate after ES = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Mirror‑descent refinement (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent (exact max) ---")
        # allocate remaining time proportionally
        remaining = deadline - time.time()
        max_iters = int(remaining * 120)   # ~120 iterations per second (empirical)
        best_seq, val_md = _mirror_subgradient_optimize(best_seq, n,
                                                        deadline,
                                                        max_iters=max_iters)
        if val_md < best_val:
            best_val = val_md
            print(f"[Mirror] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Linear‑programming polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Linear‑programming polishing ---")
        best_seq, val_lp = _lp_refine(best_seq, deadline,
                                      max_iters=8, tau=0.12)
        if val_lp < best_val:
            best_val = val_lp
            print(f"[LP] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        best_seq, val_sa = _simulated_annealing_optimize(best_seq, n,
                                                         deadline,
                                                         max_iters=1_500_000)
        if val_sa < best_val:
            best_val = val_sa
            print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (for information only)
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # store globally and return
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
