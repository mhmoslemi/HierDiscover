"""
Improved step‑function search for the min‑max convolution objective.

The algorithm expands the seed pool (extra windows, more Dirichlet draws),
runs a longer multi‑restart projected sub‑gradient / Adam optimisation,
adds a lightweight CMA‑ES‑style perturbation phase and finishes with a
long simulated‑annealing polish.  All work is done on the probability simplex
and respects the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for the lengths we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        # All entries are equal – return uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a ℓ₁‑normalised vector: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic window families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42
         - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float = 14.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    x = (i - alpha) / alpha
    a = np.i0(beta * np.sqrt(1.0 - x * x)) / np.i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip negative parts, ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

        s = arr.sum()

# ----------------------------------------------------------------------
# Exponentiated‑gradient (multiplicative) optimiser
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 0.30) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the simplex with KL‑divergence (multiplicative)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)          # √t‑schedule (robust)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)      # safety against overflow
        x = x * np.exp(delta)                # multiplicative update
        x = _proj_simplex(x)                 # back to simplex

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max‑smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_smooth_optimize(start: np.ndarray, n: int, deadline: float,
                          init_lr: float = 0.07,
                          beta1: float = 0.9, beta2: float = 0.999,
                          eps: float = 1e-8,
                          p0: float = 5.0,
                          p_factor: float = 2.0,
                          p_step: int = 25000,
                          max_total_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed surrogate:
        F_p(a) = (1/p)·log Σ exp(p·conv_k(a))
    The smoothness parameter p is increased geometrically.
    Returns the best vector seen and its exact surrogate value.
    """
    a = start.copy()
    best_a = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    p = p0

    while time.time() < deadline and t < max_total_iters:
        # ----- current convolution & surrogate (exact) -----
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam] it={t:7d} best={best_val:.6f}  p={p:.1f}")

        # ----- soft‑max weights -----
        shifted = conv - max_conv                # ≤ 0
        exp_arg = p * shifted
        exp_arg = np.clip(exp_arg, -700, 0)      # under‑flow guard
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            # extreme under‑flow – fallback to hard max indicator
            mask = conv == max_conv
            w = mask.astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of smooth surrogate -----
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]   # central slice → shape (n,)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 0.0 else _uniform(n)

        # ----- schedule of p -----
        if t % p_step == 0:
            p *= p_factor
            print(f"[Adam] increased p to {p:.1f}")

    final_val = 2.0 * n * np.max(_conv_fft(best_a, best_a))
    return best_a, final_val


# ----------------------------------------------------------------------
# Simulated‑annealing polish
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 600_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.55:
            # reshuffle a modest subset
            k = rng.integers(4, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.04)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Perturbation + short Adam – simple CMA‑ES‑style restarts
# ----------------------------------------------------------------------
def _perturb_and_refine(base: np.ndarray, n: int, rng: np.random.Generator,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Create a small random perturbation of ``base`` and run a brief Adam."""
    eps = 0.02  # perturbation strength
    # Dirichlet noise mixed with the current solution
    noisy = np.maximum(base + eps * (rng.dirichlet(np.full(n, 1.0)) - base), 0.0)
    noisy = noisy / noisy.sum()
    # short Adam run – we keep it lightweight
    remaining = deadline - time.time()
    max_iters = int(min(150_000, remaining * 1500))
    if max_iters <= 0:
        return base, _objective_surrogate(base)
    return _adam_smooth_optimize(noisy, n, deadline,
                                 init_lr=0.05,
                                 p0=5.0,
                                 p_factor=2.0,
                                 p_step=20000,
                                 max_total_iters=max_iters)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Length – power of two gives cheap FFTs.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Seed generation.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,) and arr.sum() > 0:
                prev_arr = arr
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # keep a moderate number of the best seeds (surrogate)
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 30
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Multi‑restart optimisation over the top seeds.
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        # allocate a fair portion of the remaining budget
        remaining = deadline - time.time()
        # give at most ~10 % of remaining time to each seed in early stages
        alloc = min(remaining * 0.10, remaining / (len(best_seeds) - idx))

        sub_deadline = time.time() + alloc

        # ---- Stage A: quick EG refinement ----
        eg_iters = int(min(250_000, alloc * 2000))
        cand, val = _expgrad_optimize(seed_vec, n, sub_deadline,
                                      max_iters=eg_iters, eta0=0.35)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Seed {idx+1}] EG improvement to {best_val:.6f}")

        if time.time() >= deadline:
            break

        # ---- Stage B: Adam smooth optimisation (longer) ----
        remaining2 = deadline - time.time()
        if remaining2 <= 0:
            break
        adam_iters = int(min(800_000, remaining2 * 1500))
        cand, val = _adam_smooth_optimize(best_seq, n, deadline,
                                          init_lr=0.07,
                                          p0=5.0,
                                          p_factor=2.0,
                                          p_step=25000,
                                          max_total_iters=adam_iters)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Seed {idx+1}] Adam improvement to {best_val:.6f}")

        if time.time() >= deadline:
            break

        # ---- Stage C: short EG polish ----
        remaining3 = deadline - time.time()
        if remaining3 <= 0:
            break
        eg2_iters = int(min(150_000, remaining3 * 2000))
        cand, val = _expgrad_optimize(best_seq, n, deadline,
                                      max_iters=eg2_iters, eta0=0.30)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Seed {idx+1}] Final EG improvement to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Perturbation‑restart phase (CMA‑ES‑like).  A few random nudges.
    # ------------------------------------------------------------------
    print("\n--- Perturbation‑restart phase ---")
    for r in range(5):
        if time.time() >= deadline:
            break
        cand, val = _perturb_and_refine(best_seq, n, rng, deadline)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Restart {r+1}] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing final polish.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(600_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(best_seq, n, deadline,
                                                  max_iters=max_iters)
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Very short EG polish (stabilisation).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(50_000, remaining * 2000))
        if max_iters > 0:
            best_seq, best_val = _expgrad_optimize(best_seq, n, deadline,
                                                   max_iters=max_iters,
                                                   eta0=0.25)
            print(f"[Final EG] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
