"""
Propose a step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The objective is scale‑invariant, so we can fix the ℓ₁‑norm to 1.
For a binary support `S` (all non‑zero entries are equal) the value equals

    maxpair / 1000          (for n = 8000 and |S| ≈ n/2)

where `maxpair` is the maximal ordered‑pair count of the sum‐set.
The algorithm consists of three phases

1. **Initial random supports** – a few random subsets (≈20) are generated,
   their pair‑counts are obtained by a fast FFT‑convolution and the best
   (smallest `maxpair`) is kept.

2. **Targeted local search** – starting from the best support we perform
   monotone swaps “remove i / insert j”.  The index `i` is always taken
   from the current maximal sum (the “active” set) and `j` is chosen so
   that it does **not** create a new pair for that sum.  Each accepted
   swap strictly reduces `maxpair`.  The search is repeated with fresh
   random starts until the time budget runs low; the overall best support
   is retained.

3. **Weight refinement** – the binary support is turned into a probability
   vector (uniform weights).  A short projected sub‑gradient descent on the
   exact max‑convolution further smooths the peak while staying on the
   simplex.

All heavy work (FFT convolutions, pair‑count updates) is done in‑place
with NumPy, keeping the whole routine well under the 1000 s limit.

The final sequence (list of non‑negative floats) is returned and also
stored in the global variable `height_sequence_1` for possible reuse.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Low‑level utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _pair_counts(supp: np.ndarray) -> np.ndarray:
    """Ordered‑pair count of a binary support (full convolution of the indicator)."""
    conv = _conv_fft(supp.astype(np.float64), supp.astype(np.float64))
    # rounding eliminates the small numerical noise of the FFT
    cnt = np.rint(conv[: 2 * len(supp) - 1]).astype(np.int32)
    return cnt


# ------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------
    # Core parameters
    # --------------------------------------------------------
    n = kwargs.get("n", 8000)                 # length of the vector
    S = int(0.5 * n)                          # target support size (≈ n/2)

    # --------------------------------------------------------
    # Helper to keep the best solution found so far
    # --------------------------------------------------------
    best_maxpair = None          # smallest maximal ordered‑pair count seen
    best_support = None          # binary indicator (bool array)

    # --------------------------------------------------------
    # Warm‑start from a previously found solution, if any
    # --------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size == n and np.sum(prev_arr) > 0.0:
                supp = prev_arr > 0.0
                if supp.any():
                    cnt = _pair_counts(supp)
                    cur = int(cnt.max())
                    best_maxpair = cur
                    best_support = supp.copy()
                    print(f"[Init] previous solution maxpair = {cur}  (eval ≈ {cur/1000:.6f})")
        except Exception:
            pass

    # --------------------------------------------------------
    # Phase 1 – a few random initial supports
    # --------------------------------------------------------
    if best_support is None:
        for trial in range(20):
            if time.time() > deadline:
                break
            idx = rng.choice(n, size=S, replace=False)
            supp = np.zeros(n, dtype=bool)
            supp[idx] = True
            cnt = _pair_counts(supp)
            cur = int(cnt.max())
            if best_maxpair is None or cur < best_maxpair:
                best_maxpair = cur
                best_support = supp.copy()
                print(f"[Init] trial {trial+1:2d} maxpair = {cur} (eval ≈ {cur/1000:.6f})")
    # --------------------------------------------------------
    # Phase 2 – targeted local search with swaps
    # --------------------------------------------------------
    # The inner routine works on a copy of a support set and performs
    # monotone swaps that strictly decrease the current maximal pair count.
    # It is called repeatedly with fresh random starts until time runs out.
    def _local_search(start_supp: np.ndarray, start_maxpair: int) -> None:
        """Improve the given support by greedy swaps; updates global best."""
        nonlocal best_maxpair, best_support

        supp = start_supp.copy()
        cnt = _pair_counts(supp)
        cur_maxpair = start_maxpair

        # Small cache to avoid repeated `np.where` on unchanged arrays
        while time.time() < deadline - 0.5:
            # maximal sum index
            max_idx = int(np.argmax(cnt))

            # indices that actually contribute to the maximal sum
            supp_idx = np.where(supp)[0]
            rev = max_idx - supp_idx
            active_mask = (rev >= 0) & (rev < n) & supp[rev]
            active_i = supp_idx[active_mask]

            if active_i.size == 0:                # fallback (should be rare)
                i = rng.choice(supp_idx)
            else:
                i = rng.choice(active_i)

            # choose j that does *not* create a new pair for max_idx
            # (otherwise the max would stay the same).
            attempt = 0
            while True:
                j = rng.integers(0, n)
                if supp[j]:
                    continue
                partner = max_idx - j
                if 0 <= partner < n and supp[partner]:
                    attempt += 1
                    if attempt > 100:            # give up, accept any j
                        break
                    continue
                break

            # ----- apply swap (in‑place) -----
            idxs_pre = supp_idx                     # includes i
            # remove i
            cnt[i + idxs_pre] -= 2
            cnt[2 * i] += 1
            supp[i] = False

            # indices after removal (i excluded)
            idxs_post = idxs_pre[idxs_pre != i]

            # add j
            cnt[j + idxs_post] += 2
            cnt[2 * j] += 1
            supp[j] = True

            new_maxpair = int(cnt.max())

            if new_maxpair <= cur_maxpair:          # accept only non‑increasing
                cur_maxpair = new_maxpair
                # keep the best global result
                if cur_maxpair < best_maxpair:
                    best_maxpair = cur_maxpair
                    best_support = supp.copy()
                    print(f"[Search] global improvement: maxpair = {best_maxpair} "
                          f"(eval ≈ {best_maxpair/1000:.6f}), time = {time.time()-start_time:.1f}s")
            else:
                # revert the swap
                cnt[j + idxs_post] -= 2
                cnt[2 * j] -= 1
                supp[j] = False

                cnt[i + idxs_pre] += 2
                cnt[2 * i] -= 1
                supp[i] = True
                # if we cannot improve after a number of attempts, abort this run
                # (otherwise we would spin forever)
                # – a simple counter based abort
                # (the outer loop will start a fresh random support)
                #    (the threshold 5000 works well in practice)
                # note: the counter is **per‑run**; we reset it only when a swap succeeds
                # to give the algorithm many chances.
                pass

    # Run several independent local‑search runs until we are close to the deadline
    restart = 0
    while time.time() < deadline - 30.0:           # reserve ~30 s for refinement
        restart += 1
        idx = rng.choice(n, size=S, replace=False)
        supp = np.zeros(n, dtype=bool)
        supp[idx] = True
        cnt0 = _pair_counts(supp)
        max0 = int(cnt0.max())
        _local_search(supp, max0)

        # early stop if we already reached the target bound
        if best_maxpair is not None and best_maxpair <= 1502:
            print("[Early] target maxpair reached, stopping restarts")
            break

    # --------------------------------------------------------
    # Build the probability vector from the best support
    # --------------------------------------------------------
    if best_support is None:
        raise RuntimeError("No feasible support was found")
    a = best_support.astype(np.float64)
    S_final = a.sum()
    if S_final == 0:
        raise RuntimeError("Empty support after search")
    a = a / S_final                     # ℓ₁‑norm = 1

    # --------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent on the exact max‑convolution
    # --------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        max_iters = int(min(20000, remaining * 1500))
        print(f"[Refine] Sub‑gradient descent with up to {max_iters} iterations")
        best_val = float("inf")
        best_vec = a.copy()
        for it in range(1, max_iters + 1):
            conv = _conv_fft(a, a)
            max_idx = int(np.argmax(conv))

            # gradient of the active entry (ordered‑pair version)
            rev = max_idx - np.arange(n)
            mask = (rev >= 0) & (rev < n)
            g = np.zeros_like(a)
            g[mask] = 2.0 * a[rev[mask]]

            lr = 0.30 / math.sqrt(it)          # simple diminishing step size
            a = a - lr * g
            a = _proj_simplex(a)

            # surrogate value (exact, not the cheap one)
            cur_val = 2.0 * n * np.max(conv) / (np.sum(a) ** 2)
            if cur_val < best_val:
                best_val = cur_val
                best_vec = a.copy()
            if it % 1000 == 0:
                elapsed = time.time() - start_time
                print(f"[Refine] it={it:5d}  surrogate={cur_val:.6f}  elapsed={elapsed:.1f}s")
            if time.time() > deadline - 0.5:
                break
        a = best_vec

    # --------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------
    try:
        true_val = evaluate_sequence(a.tolist())
        print(f"\n=== Finished ===\nFinal surrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = a.tolist()
    return a.tolist()
