def _lp_improve(start, n, deadline, max_iter=20, t=0.1):
    import scipy.sparse as sp
    import scipy.optimize as opt

    a = start.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(max_iter):
        if time.time() > deadline:
            break
        # compute convolution and max
        conv = _conv_fft(a, a)
        M = float(np.max(conv))

        # build linear constraints A_ub * b <= M_vec
        # Build sparse matrix A
        # Use data generation loops.
        rows = []
        cols = []
        data = []
        # for each column i, rows i .. i+n-1
        # Outer loop
        for i in range(n):
            start_row = i
            end_row = i + n  # exclusive
            rows.extend(range(start_row, end_row))
            cols.extend([i] * n)
            data.extend([a[i]] * n)
        A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # RHS
        b_ub = np.full(2 * n - 1, M)

        # objective: minimize -sum(b)
        c = -np.ones(n)

        # bounds
        bounds = [(0, None)] * n

        # solve
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"LP iteration {it} failed: {res.message}")
            break

        b = res.x
        b_sum = b.sum()
        if b_sum <= 0:
            print(f"LP iteration {it} yields zero sum")
            break

        # normalize b to sum 1
        b_norm = b / b_sum

        # convex combination
        a = (1 - t) * a + t * b_norm

        # project onto simplex (should remain sum 1 but numerical)
        a_sum = a.sum()
        if a_sum != 0:
            a = a / a_sum
        # evaluate
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LP] iter {it+1} improved to {best_val:.6f}")

    return best_seq, best_val

Algorithm
---------

        maximise   ∑ bⱼ
        subject to  (a∗b)[k] ≤ M    for all k
                    bⱼ ≥ 0

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(p,p)) for a distribution p (∑p=1)."""
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)

def _triangular(n: int) -> np.ndarray:

    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:

    s = a.sum()

    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

    idx = np.arange(1, n + 1, dtype=np.float64)

    s = a.sum()
    return a / s if s > 0 else _uniform(n)

        return _uniform(n)

    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0

                (2 * i / (alpha * (n - 1)) - 1)))

            a[i] = 1.0
        else:

                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

    i = np.arange(n, dtype=np.float64)

# LP‑based polishing (core of the new search)
# ----------------------------------------------------------------------
def _lp_polish(start: np.ndarray,
               n: int,
               deadline: float,
               max_iter: int = 15,
               mix_factor: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Iteratively improve ``start`` using a linear‑program that maximises
    the ℓ₁‑mass of a partner vector while honouring the current
    convolution peak.  The partner is mixed into the current vector.
    """
    try:
        import scipy.sparse as sp
        import scipy.optimize as opt
    except Exception as exc:  # pragma: no cover
        print("SciPy unavailable – skipping LP polish:", exc)
        return start, _objective_surrogate(start)

    a = start.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(max_iter):
        if time.time() > deadline:
            break

        # current peak of the self‑convolution
        conv = _conv_fft(a, a)
        M = float(np.max(conv))

        # Build the Toeplitz‑like matrix A where (A @ b) = conv(a,b)
        # A has shape (2 n‑1, n); column i is a_i placed on diagonal i.
        offsets = np.arange(n, dtype=np.int32)          # 0 … n‑1
        # Broadcast a so each row i contains a_i repeated n times.
        data = np.broadcast_to(a[:, None], (n, n))
        A = sp.dia_matrix((data, offsets), shape=(2 * n - 1, n)).tocsr()

        # Linear program: maximise ∑b_j  <=> minimise -∑b_j
        c = -np.ones(n, dtype=np.float64)               # objective
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)  # RHS of A @ b ≤ M
        bounds = [(0.0, None)] * n                       # b ≥ 0

        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        if not res.success:
            print(f"[LP] iteration {it+1}: solver failure – {res.message}")
            break

        b = res.x
        s = b.sum()
        if s <= 0.0:
            print(f"[LP] iteration {it+1}: zero‑mass solution")
            break

        # Normalise the partner vector and mix it back.
        b_norm = b / s
        a = (1.0 - mix_factor) * a + mix_factor * b_norm

        # Projection onto the probability simplex (tiny numerical drift)
        a_sum = a.sum()
        if a_sum > 0.0:
            a = a / a_sum
        else:
            a = _uniform(n)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LP] iter {it+1:2d} improved → {best_val:.6f}")

    return best_seq, best_val

                           alpha_end: float = 2000.0,
                           lr: float = 0.12) -> tuple[np.ndarray, float]:
    """Adam optimisation on the smooth surrogate L(p) = (1/α)·log Σ exp(α·(p∗p))."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    N = _next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # full self‑convolution
        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        # smooth max (log‑sum‑exp)
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        # gradient of the surrogate (correlation with soft)
        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # projection onto the simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if t % 20000 == 0:
                print(f"[Adam] it={t:7d} best={best_val:.6f} α={alpha:.1f}")

    return best_x, best_val

# Main entry point – the function required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the official
    ``evaluate_sequence`` objective within ``budget_s`` seconds.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0     # safety margin

    # Length choice – 2000 provides a good trade‑off between quality
    # and the cost of the LP sub‑problem.
    n = 2000
    print(f"\n=== Begin search (n={n}) ===")

    # ------------------------------------------------------------------
    # Stage 0 – generate a library of seeds
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed candidates.")

    # Evaluate and keep the most promising ones
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 8
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – LP polishing on the top seeds
    # ------------------------------------------------------------------
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- LP polishing start (seed #{idx+1}) ---")
        cand_seq, cand_val = _lp_polish(seed_vec, n, deadline,
                                        max_iter=12, mix_factor=0.15)
        print(f"[LP] seed #{idx+1} finished with surrogate {cand_val:.6f}")

        if cand_val < global_best_val:
            global_best_val = cand_val
            global_best_seq = cand_seq.copy()
            print(f"[Update] new global best → {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – optional Adam fine‑tuning (uses the smooth surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(120_000, remaining * 140))
        print("\n--- Adam fine‑tuning ---")
        cand_seq, cand_val = _adam_softmax_optimize(global_best_seq, n,
                                                    deadline,
                                                    max_iters=max_iters,
                                                    alpha_end=1800.0,
                                                    lr=0.12)
        if cand_val < global_best_val:
            global_best_val = cand_val
            global_best_seq = cand_seq.copy()
            print(f"[Adam] improved global best → {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official helper
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible reuse by a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
