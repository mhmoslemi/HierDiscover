"""
Search for a step‑function that minimizes
    2·n·max(conv(a,a)) / (sum(a))²
using a combination of diverse seeds, Adam optimisation on a
smooth‑max surrogate, exponentiated‑gradient (mirror) descent on the
exact max‑convolution and a final simulated‑annealing polish.
"""

import time
import numpy as np
import scipy.optimize

# Helper functions
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smooth max (log‑sum‑exp) of the self‑convolution.
    Returns (smooth, gradient w.r.t. a).
    """
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum
    smooth = shift + np.log(w_sum) / beta

    # gradient of Σ w_k * (a*a)_k  w.r.t a = 2 * (rev(a) ⊛ w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad


def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 5000,
                   init_lr: float = 0.5,
                   beta_schedule: list[float] = None,
                   inner_iters: int = 200) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    The temperature β is stepped through ``beta_schedule``.
    ``inner_iters`` controls how many Adam steps are taken at each β.
    """
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 40, 80, 150, 300, 600, 1000, 2000, 4000, 8000]

    eps = 1e-12
    # log‑it representation of the simplex (soft‑max)
    v = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999

    it = 1
    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for beta in beta_schedule:
        for _ in range(inner_iters):
            if time.time() > deadline:
                return best_seq, best_val

            a = _softmax(v)

            smooth, grad_a = _smooth_max_and_grad(a, beta)
            loss = 2.0 * n * smooth

            # gradient w.r.t. logits (soft‑max chain rule)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update on logits
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat_corr = v_hat / (1.0 - beta2 ** it)

            v -= init_lr * m_hat / (np.sqrt(v_hat_corr) + eps)

            # keep track of best encountered (exact objective)
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                # occasional progress report
                if it % 500 == 0:
                    print(f"[Adam] it={it} β={beta:.0f} best={best_val:.6f}")

            it += 1

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 10_000_000,
                                 lr0: float = 0.25,
                                 power: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) descent directly on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 1
    while it <= max_iters:
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 500_000 == 0:
                print(f"[Mirror] it={it:,d} best={cur_val:.6f}")

        # sub‑gradient of the active anti‑diagonal
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / (it ** power)
        a = a * np.exp(-lr * g)

        # re‑normalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        it += 1

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 3_000_000,
                                 init_temp_factor: float = 0.05,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:  # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:  # Dirichlet redistribute a random subset
            k = rng.integers(2, 15)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:  # split‑transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 500_000 == 0:
                    print(f"[SA] it={it:,d} best={new_val:.6f}")

    return best_seq, best_val


# Analytic shape families – all ℓ₁‑normalised
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = n - i
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _u_shape(n: int) -> np.ndarray:
    """U‑shaped distribution: large at the ends, small in the centre."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = 1.0 / (i * (n - i + 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_linear_decreasing(n))
    seeds.append(_u_shape(n))

    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))

    alphas = np.concatenate((np.linspace(0.01, 0.3, 6), np.linspace(0.3, 1.0, 6)))
    betas = np.concatenate((np.linspace(1.0, 5.0, 5), np.linspace(5.0, 12.0, 5)))
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for lam in np.linspace(0.0005, 0.020, 12):
        seeds.append(_exponential_decay(n, lam))

    for nf in (2, 3, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Dirichlet draws – a few concentration values
    for conc in (0.02, 0.05, 0.1, 0.3):
        for _ in range(4):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw order
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending

    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

"""
Search for a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a))²

where conv is the full linear convolution.  The optimiser works in three
phases:

1.  **Seed generation** – a large diverse pool of ℓ₁‑normalised
    candidates (uniform, linear, power‑law, β‑shape, exponential,
    low‑frequency sinusoids, a hand‑crafted “U‑shape”, Dirichlet draws,
    the previous best solution).

2.  **Smooth‑max Adam optimisation** – each of the best‑looking seeds is
    refined by Adam on a log‑sum‑exp surrogate of the max‑convolution.
    The temperature β is increased gradually (5 → … → 8000) and the
    optimiser works on the logits of the simplex via a soft‑max map.
    Random perturbations of the current best are also re‑optimised.

3.  **Exact‑max polishing** – a long mirror‑descent (exponentiated‑gradient)
    run on the true max‑convolution, followed by a lightweight simulated
    annealing for the final touch.

All stages respect the global deadline (default 1000 s).  The best
sequence found is stored as the global name ``height_sequence_1`` and
returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.optimize  # imported for completeness; not strictly required

# ----------------------------------------------------------------------
# FFT‑based convolution (real‑valued, full length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Soft‑max helpers (used to stay on the simplex)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = n - i
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth random curve made from a few low‑frequency sinusoids."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _u_shape(n: int) -> np.ndarray:
    """U‑shaped distribution: big at the ends, tiny in the centre."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = 1.0 / (i * (n - i + 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Surrogate objective (fast, identical ordering to the official one)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – works on ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Return (smooth_max, grad_a) for the log‑sum‑exp surrogate.
    """
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum
    smooth = shift + np.log(w_sum) / beta

    # gradient of Σ w_k * (a*a)_k  w.r.t a  ->  2 * (rev(a) ⊛ w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad_a = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 5_000,
                   init_lr: float = 0.6,
                   beta_schedule: list[float] | None = None,
                   inner_iters: int = 250) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    The temperature β is stepped through ``beta_schedule``; for each β
    we perform ``inner_iters`` Adam updates.
    Returns the best sequence seen (exact objective) together with its
    exact‑objective value.
    """
    if beta_schedule is None:
        beta_schedule = [5, 15, 30, 60, 120, 250, 500, 1_000, 2_000, 4_000, 8_000]

    eps = 1e-12
    v = np.log(start_seq + 1e-30)          # logits → simplex via soft‑max
    m = np.zeros_like(v)                   # Adam first moment
    v2 = np.zeros_like(v)                  # Adam second moment
    beta1, beta2 = 0.9, 0.999

    it = 1
    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for beta in beta_schedule:
        for _ in range(inner_iters):
            if time.time() > deadline:
                return best_seq, best_val

            a = _softmax(v)

            smooth, grad_a = _smooth_max_and_grad(a, beta)
            loss = 2.0 * n * smooth

            # gradient w.r.t. logits (soft‑max chain rule)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update on logits
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v2_hat = v2 / (1.0 - beta2 ** it)

            v -= init_lr * m_hat / (np.sqrt(v2_hat) + eps)

            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                if it % 500 == 0:
                    print(f"[Adam] it={it:,d} β={beta:.0f} best={best_val:.6f}")

            it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑conv objective
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 8_000_000,
                                 lr0: float = 0.23,
                                 power: float = 0.30) -> tuple[np.ndarray, float]:
    """Multiplicative update on the simplex using the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 1
    while it <= max_iters:
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:,d} best={cur_val:.6f}")

        # sub‑gradient for the active anti‑diagonal
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / (it ** power)
        a = a * np.exp(-lr * g)

        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light simulated‑annealing for final polishing
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 3_000_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a random subset
            k = rng.integers(2, 15)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                  # split‑transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 500_000 == 0:
                    print(f"[SA] it={it:,d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation (rich and diverse)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_linear_decreasing(n))
    seeds.append(_u_shape(n))

    # power‑law family
    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))

    # β‑shape – wide asymmetric grid
    alphas = np.concatenate((np.linspace(0.01, 0.3, 6), np.linspace(0.3, 1.0, 6)))
    betas = np.concatenate((np.linspace(1.0, 5.0, 5), np.linspace(5.0, 12.0, 5)))
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # exponential decay
    for lam in np.linspace(0.0005, 0.020, 12):
        seeds.append(_exponential_decay(n, lam))

    # low‑frequency random curves
    for nf in (2, 3, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Dirichlet draws with several concentration parameters
    for conc in (0.02, 0.05, 0.1, 0.3):
        for _ in range(4):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw ordering
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending

    # previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

# Random perturbation + short Adam re‑run (helps escape local minima)
# ----------------------------------------------------------------------
def _perturb_and_reoptimize(best_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           trials: int = 8) -> tuple[np.ndarray, float]:
    """Apply a few random multiplicative perturbations and re‑optimise."""
    best_val = _objective(best_seq)
    cur_best = best_seq.copy()

    for t in range(trials):
        if time.time() > deadline:
            break
        # multiplicative Gaussian noise, keep non‑negative, renormalise
        noisy = cur_best * (1.0 + 0.02 * rng.standard_normal(n))
        noisy = np.maximum(noisy, 0.0)
        s = noisy.sum()
        if s <= 0.0:
            continue
        noisy = noisy / s

        seq_opt, val_opt = _adam_optimize(
            noisy, n, deadline,
            max_iters=1_500,
            init_lr=0.4,
            beta_schedule=[5, 30, 120, 500, 2_000],
            inner_iters=150,
        )
        if val_opt < best_val:
            best_val = val_opt
            cur_best = seq_opt.copy()
            print(f"[Perturb] trial {t+1} → new best {val_opt:.6f}")

    return cur_best, best_val

        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=8_000_000,
            lr0=0.23,
            power=0.30,
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Simulated‑annealing polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=3_000_000,
            init_temp_factor=0.04,
            min_temp=1e-12,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (information only).
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Eval] evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for possible reuse by a later run.
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
