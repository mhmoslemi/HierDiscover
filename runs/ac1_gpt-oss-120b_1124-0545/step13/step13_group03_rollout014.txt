def _double_block(shape_func, n, fraction, param): # shape_func(n_sub, param)
    s = int(fraction * n)
    if s <= 1 or s*2 >= n: continue
    left_shape = shape_func(s, param)
    # ensure shape sum=1 for block; later scale to block mass 0.5 each.
    left_shape_norm = left_shape / left_shape.sum()
    block = np.concatenate([left_shape_norm * 0.5,
                            np.zeros(n - 2*s, dtype=np.float64),
                            (left_shape_norm[::-1]) * 0.5])
    return block

"""
Propose a step‑function (non‑negative vector) that minimises
    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a))²

Algorithm (≈ 1000 s)
--------------------
1. **Length selection** – we use a longer vector (n≈8000) which gives a
   finer discretisation of the continuous problem and therefore a smaller
   asymptotic constant.

2. **Rich seed pool** – analytic windows (uniform, triangular, sine,
   Gaussian, power, beta, Tukey, exponential), a dense grid over the
   beta‑parameters (α,β) and the power exponent *p*, plus a few random
   Dirichlet draws.  All seeds are ℓ₁‑normalised (∑a = 1).

3. **Surrogate ranking** – for every seed we evaluate the cheap surrogate
   2·n·max(conv) (the scaling by the sum is irrelevant because the sum is 1)
   and keep the 12 best candidates.

4. **Smooth Adam optimisation** – we optimise the smoothed objective

       Fₚ(a) = (1/p)·log Σₖ exp(p·convₖ(a))

   using Adam on the *softmax* parametrisation
       a = softmax(z) .
   The gradient of Fₚ is
       ∇ₐFₚ = 2·conv( softmaxₚ(conv) , a_rev )
   and the chain rule gives the gradient w.r.t. *z*.
   The smoothness parameter *p* starts small (≈5) and is multiplied by 2
   every 25 k iterations, allowing the optimiser to first move in a
   smooth landscape and then focus on the true max‑convolution.
   The best vector seen is recorded.

5. **Final polish** – a short Exponentiated‑Gradient (multiplicative) run
   on the best‑found vector, similarly to the original implementation,
   to squeeze a little more out of the solution.

6. **Official evaluation** – the final vector is fed to the provided
   ``evaluate_sequence`` function, stored globally as ``height_sequence_1``,
   and returned.

All stages respect the global deadline (default 1000 s) and use only NumPy.
"""

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(z)
    e = np.exp(z - m)
    s = e.sum()
    return e / s


def _surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for sum(seq)=1: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Seed collection (adds a dense beta / power grid)
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, g))
    for sf in (0.15, 0.20, 0.25, 0.30):
        seeds.append(_gaussian(n, sf))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.2, 0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))

    # fine beta‑grid
    alphas = np.linspace(0.4, 4.0, 9)
    betas = np.linspace(0.4, 4.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # keep previous solution as a seed
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final safety normalisation
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        clean.append(v / s if s > 0 else _uniform(n))
    return clean

# Adam optimiser on the smoothed max‑convolution (soft‑max)
# ----------------------------------------------------------------------
def _adam_softmax_optimize(start_a: np.ndarray, n: int,
                           deadline: float,
                           init_lr: float = 0.02,
                           beta1: float = 0.9,
                           beta2: float = 0.999,
                           eps: float = 1e-8,
                           p0: float = 5.0,
                           p_factor: float = 2.0,
                           p_step: int = 25000,
                           max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth surrogate
         Fₚ(a) = (1/p)·log Σ exp(p·convₖ(a))
    using a softmax‑parameterisation a = softmax(z).
    Returns the best vector seen (according to the *exact* surrogate).
    """
    # initialise logits
    z = np.log(start_a + 1e-18)          # avoid log(0)
    best_a = start_a.copy()
    best_val = _surrogate(best_a)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    p = p0

    while time.time() < deadline and t < max_iters:
        # --------------------------------------------------------------
        # 1) probability vector a
        # --------------------------------------------------------------
        a = _softmax(z)

        # --------------------------------------------------------------
        # 2) convolution and soft‑max weights
        # --------------------------------------------------------------
        conv = _conv_fft(a, a)                     # length 2n‑1
        max_c = conv.max()
        shifted = conv - max_c                     # ≤0
        exp_arg = p * shifted
        exp_arg = np.clip(exp_arg, -700, 0)        # safety
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            # numerical underflow – fallback to the (possibly several) max positions
            mask = conv == max_c
            w = mask.astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # --------------------------------------------------------------
        # 3) gradient w.r.t. a  (see derivation in the docstring)
        # --------------------------------------------------------------
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)           # length 3n‑2
        grad_a = 2.0 * grad_full[n - 1: n - 1 + n]

        # --------------------------------------------------------------
        # 4) chain rule: gradient w.r.t logits z
        #    ∂a_i/∂z_j = a_i (δ_{ij} - a_j)
        # --------------------------------------------------------------
        mean = np.dot(grad_a, a)                  # scalar ⟨grad_a , a⟩
        grad_z = a * (grad_a - mean)              # shape (n,)

        # --------------------------------------------------------------
        # 5) Adam update on logits
        # --------------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        # --------------------------------------------------------------
        # 6) keep the best exact surrogate seen so far
        # --------------------------------------------------------------
        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # occasional progress report
            if t % 50000 == 0:
                print(f"[Adam] it={t:7d} best surrogate={best_val:.6f}   p={p:.1f}")

        # --------------------------------------------------------------
        # 7) smoothness schedule
        # --------------------------------------------------------------
        if t % p_step == 0:
            p *= p_factor
            # print schedule change
            print(f"[Adam] increased smoothing p to {p:.1f}")

    # final exact value of the best vector
    final_val = _surrogate(best_a)
    return best_a, final_val

# Exponentiated‑gradient optimiser (mirror‑descent on the simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int,
                      deadline: float,
                      max_iters: int = 150_000,
                      eta0: float = 0.30) -> tuple[np.ndarray, float]:
    """Multiplicative (KL‑mirror) descent on the simplex."""
    x = start.copy()
    best_x = x.copy()
    best_val = _surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # stepsize (√t schedule)
        eta = eta0 / math.sqrt(it)

        # multiplicative update (clip to avoid overflow)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # projection onto simplex (Euclidean)
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur = _surrogate(x)
        if cur < best_val:
            best_val = cur
            best_x = x.copy()
    return best_x, best_val

    print(f"\n--- Using vector length n={n} ---")

    # --------------------------------------------------------------
    # Seed generation
    # --------------------------------------------------------------
    print("\n--- Seed generation ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # evaluate surrogate for all seeds
    scored = [(s, _surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"\nTop {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # --------------------------------------------------------------
    # Quick Exponential Gradient refinement on a few top seeds
    # --------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Quick EG refinement (few seeds) ---")
        for idx, seed_vec in enumerate(best_seeds[:4]):
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            # allocate a modest number of iterations per seed
            iters = int(min(80_000, remaining * 1800))
            cand, val = _expgrad_optimize(seed_vec, n, deadline,
                                          max_iters=iters, eta0=0.30)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Quick EG] seed #{idx+1} improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Main Adam soft‑max optimisation on the current best seed
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam smooth optimisation (soft‑max) ---")
        remaining = deadline - time.time()
        max_iters = int(min(400_000, remaining * 2000))
        best_seq, best_val = _adam_softmax_optimize(best_seq, n, deadline,
                                                    init_lr=0.02,
                                                    p0=5.0,
                                                    p_factor=2.0,
                                                    p_step=25000,
                                                    max_iters=max_iters)
        print(f"[Adam] best surrogate after polishing = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final brief EG polishing (helps to settle on a simplex corner)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(30_000, remaining * 1800))
        if iters > 0:
            best_seq, best_val = _expgrad_optimize(best_seq, n, deadline,
                                                   max_iters=iters,
                                                   eta0=0.25)
            print(f"[Final EG] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
