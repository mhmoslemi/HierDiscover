"""
Propose a step‑function minimizing
    2·n·max(conv(a, a)) / (sum(a)²)

The algorithm proceeds in three stages:

1. **Rich seed generation** – a large pool of ℓ₁‑normalised vectors is
   created from analytic windows (Beta, Gaussian, power‑law, Tukey …),
   random Dirichlet draws and a few handcrafted “alternating‑extrema”
   patterns.  The cheap surrogate `2·n·max(conv)` (exact when the sum is 1)
   is used to keep the best few seeds.

2. **Continuation Adam optimisation** – for each promising seed we optimise
   a smooth log‑sum‑exp approximation of the max‑convolution.  The variables
   are logits `z` (softmax `a = softmax(z)`) which automatically enforce
   non‑negativity and the simplex constraint.  A schedule of growing
   inverse‑temperature β (1 → … → 5 000) is combined with an Adam optimiser.
   The best vector seen during the run is retained.

3. **Intensive simulated‑annealing polish** – the current best logits are
   perturbed with Gaussian noise while an annealed Metropolis acceptance rule
   drives the search for a final tiny improvement.

All steps respect the global 1000 s wall‑clock budget.  The final
sequence is stored in the global variable ``height_sequence_1`` and also
returned as a plain list of floats.
"""

import time
import math

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (efficient for long vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (Euclidean projection)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Arrange a Dirichlet draw alternatingly from the largest to the smallest."""
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_linear(n: int) -> np.ndarray:
    """U‑shaped inverse‑linear: a_i ∝ 1/(i+1) + 1/(n‑i)  (symmetrised)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (i + 1.0) + 1.0 / (n - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / ((i + 1.0) * (n - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Seed pool construction

def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised seed vectors (diverse families)."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_inverse_linear(n))
    seeds.append(_inverse_quadratic(n))

    # Beta family (grid, now covering very small parameters)
    alphas = np.concatenate((np.array([0.05, 0.10, 0.15]), np.linspace(0.20, 5.0, 13)))
    betas = np.concatenate((np.array([0.05, 0.10, 0.15]), np.linspace(0.20, 5.0, 13)))
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, float(a), float(b)))

    # Power‑law shapes
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, float(p)))

    # Gaussian families
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, float(sigma)))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, float(conc), rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Special alternating‑extrema Dirichlet seed
    seeds.append(_alternating_dirichlet(n, rng))

    # Previous solution, if available
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise robustly (guards against accidental zero sum)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Smoothed max‑convolution gradient (log‑sum‑exp) – Adam optimiser
# ----------------------------------------------------------------------
def _smooth_objective_and_grad(logits: np.ndarray,
                               n: int,
                               beta: float):
    """
    Returns (obj, grad_logits) where ``obj`` is the smooth surrogate
    2·n·(shift + log Σexp(β·(conv‑shift))/β)  and ``grad_logits`` is
    the gradient w.r.t. the logits.
    """
    a = _softmax(logits)                     # stays on the simplex
    conv = _conv_fft(a, a)                   # length 2n‑1
    max_c = np.max(conv)
    shift = max_c
    # smoothed weights (softmax over β·(conv‑shift))
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    # surrogate objective (same as in the original script)
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_beta / sum_exp                  # weight vector, sums to 1

    # gradient of Σ w_k·conv_k w.r.t. a:
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)              # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]    # slice of length n

    # chain rule through the softmax
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)        # shape (n,)
    return float(obj), grad_logits


def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule=None,
                   steps_per_beta: int = 15000) -> np.ndarray:
    """
    Adam optimisation on the smoothed max‑convolution surrogate.
    Returns the best (lowest surrogate) ℓ₁‑normalised vector seen.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1200.0, 2500.0, 5000.0]

    # Initialise logits (log‑softmax parameters) safely
    logits = np.log(start_vec + 1e-30)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = _softmax(logits)
    best_val = 2.0 * n * np.max(_conv_fft(best_a, best_a))

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                break
            t += 1
            obj, grad = _smooth_objective_and_grad(logits, n, beta)

            # Keep track of the best *exact* surrogate encountered
            a_curr = _softmax(logits)
            cur_val = 2.0 * n * np.max(_conv_fft(a_curr, a_curr))
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_curr.copy()

            # Adam update on logits
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            # A mildly decaying stepsize (larger early, smaller later)
            lr = 0.3 / math.sqrt(t)
            logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

    # Return the best vector seen (already ℓ₁‑normalised)
    return best_a

# Simple simulated‑annealing polishing (works directly on logits)
# ----------------------------------------------------------------------
def _simulated_annealing(logits: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300_000) -> np.ndarray:
    """
    Perturb ``logits`` with Gaussian noise while accepting moves
    according to a cooling schedule.  Returns the best vector discovered.
    """
    cur_logits = logits.copy()
    cur_seq = _softmax(cur_logits)
    cur_val = 2.0 * n * np.max(_conv_fft(cur_seq, cur_seq))
    best_seq = cur_seq.copy()
    best_val = cur_val

    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # linear cooling – temperature falls to near zero
        progress = it / max_iters
        T = max(1e-6, 0.05 * (1.0 - progress))
        sigma = max(1e-5, 0.15 * (1.0 - progress))

        proposal_logits = cur_logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = 2.0 * n * np.max(_conv_fft(proposal_seq, proposal_seq))

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            cur_logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] iter={it:7d} best surrogate={best_val:.6f}")

    return best_seq

        if time.time() < deadline:
            remaining = deadline - time.time()
            # give a short SA window (≈20 % of remaining time)
            sa_deadline = time.time() + max(10.0, 0.20 * remaining)
            logits = np.log(global_best_seq + 1e-30)
            polished = _simulated_annealing(logits, n, sa_deadline,
                                            max_iters=150_000)
            polished_val = 2.0 * n * np.max(_conv_fft(polished, polished))
            if polished_val < global_best_val:
                global_best_seq = polished.copy()
                global_best_val = polished_val
                print(f"[SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Final long‑run simulated‑annealing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(400_000, remaining * 300))  # adapt to remaining time
        print("\n--- Final long simulated‑annealing refinement ---")
        final_logits = np.log(global_best_seq + 1e-30)
        final_seq = _simulated_annealing(final_logits, n, deadline,
                                         max_iters=max_iters)
        final_val = 2.0 * n * np.max(_conv_fft(final_seq, final_seq))
        if final_val < global_best_val:
            global_best_seq = final_seq.copy()
            global_best_val = final_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Evaluate with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
