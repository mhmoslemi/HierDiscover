def propose_candidate(seed=123, budget_s=1000):
    """Search for a non‑negative step‑function (length n) minimising the
    max‑autoconvolution objective.  The algorithm generates a diverse pool
    of ℓ₁‑normalised seeds, refines the best with Adam on a smooth
    approximation of the max, then performs a long exponentiated‑gradient
    (mirror‑descent) phase on the exact objective, and finally a short
    simulated‑annealing polishing step.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    # Choose problem size.  Larger n gives a lower theoretical bound.
    n = 8000  # try a larger problem size for a better optimum

    # Load previous best if available
    previous = globals().get('height_sequence_1')
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # Seed generation
    seeds = _collect_seeds(n, rng, previous)
    # Evaluate surrogate (fast objective) which is 2*n*max(conv) for ℓ₁‑normalised vectors.
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec,_ in evaluated[:top_k]]
    print(f"Generated {len(seeds)} seeds, keeping top {top_k}")

    # Global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"[Init] best surrogate = {global_best_val:.6f}")

    # Stage 1: Adam optimisation on smooth max for each top seed
    beta_schedule = [5, 10, 20, 40, 80, 160, 320, 640, 1280, 2560]
    for idx, seed_vec in enumerate(best_seeds, start=1):
        if time.time() > deadline:
            break
        seq = seed_vec.copy()
        seq_opt, val_opt = _adam_optimize(seq, n, deadline, beta_schedule, iter_per_beta=200)
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_seq = seq_opt.copy()
            print(f"[Adam] seed #{idx} → new best {val_opt:.6f}")

    # Stage 2: Mirror descent (exponentiated gradient) on the exact objective
    if time.time() < deadline:
        print("\n--- Mirror‑descent (exponentiated‑gradient) ---")
        seq_mir, val_mir = _mirror_subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=6_000_000,
            lr0=0.24,
            power=0.32,
        )
        if val_mir < global_best_val:
            global_best_val = val_mir
            global_best_seq = seq_mir.copy()
            print(f"[Mirror] new global best = {val_mir:.6f}")

    # Stage 3: Simulated annealing polishing
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=2_000_000,
            init_temp_factor=0.04,
            min_temp=1e-12,
        )
        if val_sa < global_best_val:
            global_best_val = val_sa
            global_best_seq = seq_sa.copy()
            print(f"[SA] final global best = {val_sa:.6f}")

    # Final evaluation with official function
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("[Eval] evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} (length {len(global_best_seq)})")
    globals()['height_sequence_1'] = global_best_seq.tolist()
    return global_best_seq.tolist()

def _adam_optimize(start_seq, n, deadline, beta_schedule, iter_per_beta=200):
    eps=1e-12
    v = np.log(start_seq + eps)  # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    t = 0
    best_seq = None
    best_val = np.inf
    for beta in beta_schedule:
        for _ in range(iter_per_beta):
            t+=1
            if time.time()>deadline:
                break
            a = _softmax(v)
            smooth, grad_a = _smooth_max_and_grad(a, beta)
            loss = 2.0 * n * smooth
            # compute gradient w.r.t v
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_v
            v_sq = beta2 * v_sq + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v_sq / (1 - beta2 ** t)
            lr = 0.5 / (np.sqrt(t) + 1e-8)  # base step decays
            v = v - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            # optional projection via softmax after update path? Already mapped via a later.
            # compute actual objective for the current a
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                # optionally print progress
    if best_seq is None:
        best_seq = start_seq.copy()
        best_val = _objective(best_seq)
    return best_seq, best_val

def _gaussian_mixture(n, num_components, rng):
    a = np.zeros(n)
    xs = np.arange(n)
    for _ in range(num_components):
        mu = rng.uniform(0, n-1)
        sigma = rng.uniform(0.02*n, 0.1*n)
        weight = rng.uniform(0.1, 1.0)
        a += weight * np.exp(-0.5 * ((xs - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s>0 else _uniform(n)

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)

Define _smooth_max_and_grad:

We need to compute conv, weight softmax with temperature beta, then gradient.

Implementation careful: we compute conv = _conv_fft(a,a). Then shift = max(conv). Compute exp_term = np.exp(beta*(conv - shift)). weights = exp_term / exp_term.sum(). Weighted conv = sum_k weights_k * conv_k approximates smooth max.

def _adam_optimize(start_seq, n, deadline, beta_schedule, iter_per_beta=200):
    eps = 1e-30
    v = np.log(start_seq + eps)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    t = 0
    best_seq = None
    best_val = np.inf
    for beta in beta_schedule:
        for _ in range(iter_per_beta):
            if time.time() > deadline:
                break
            t += 1
            a = _softmax(v)
            smooth, grad_a = _smooth_max_and_grad(a, beta)
            loss = 2.0 * n * smooth
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            # Adam moments update
            m = beta1 * m + (1 - beta1) * grad_v
            v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1 ** t)
            v2_hat = v2 / (1 - beta2 ** t)
            lr = 0.5 / (np.sqrt(t) + 1e-8)  # base step decays slowly
            v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-8)
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    if best_seq is None:
        best_seq = start_seq.copy()
        best_val = _objective(best_seq)
    return best_seq, best_val

def _mirror_subgradient_optimize(start_seq, n, deadline,
                                 max_iters=2_000_000, lr0=0.18, power=0.30):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    it = 1
    while it <= max_iters:
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        max_idx = int(np.argmax(conv))
        if max_val < best_val:
            best_val = max_val
            best_seq = a.copy()
        # build subgradient g
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # step size
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)
        # renormalize to simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s
        it += 1
    return best_seq, best_val

"""
Optimise the non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a))²

by a three‑stage scheme:

1.  Build a large, highly diverse pool of ℓ₁‑normalised seeds
    (analytic families, random low‑frequency curves, Dirichlet draws,
    Gaussian mixtures, etc.).  Keep the ~20 best according to the fast
    surrogate 2·n·max(conv).

2.  Refine every top seed with an ADAM optimiser on a smooth
    approximation of the max (log‑sum‑exp) – temperature (β) is
    increased gradually.  The parameters live on the simplex via a
    soft‑max (log‑its) representation.

3.  Perform a long mirror‑descent (exponentiated‑gradient) phase on
    the exact max‑convolution objective, followed by a short
    L‑BFGS fine‑tune and a lightweight simulated‑annealing polish.

All work is bounded by the supplied time budget (default 1000 s) and
the best sequence found is stored in the global name ``height_sequence_1``.
"""

import time

# FFT‑based full linear convolution (real‑valued)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """Fast surrogate: 2·n·max(conv(seq,seq)) for ℓ₁‑normalised sequences."""
    conv = _conv_fft(seq, seq)
    return float(2.0 * len(seq) * np.max(conv))


def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Log‑sum‑exp smooth approximation of max(conv(a,a)).
    Returns (smooth_max, grad_a) where grad_a = ∂smooth/∂a.
    """
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    w_sum = np.sum(exp_term)
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum
    smooth = shift + np.log(w_sum) / beta

    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    n = len(a)
    grad_a = 2.0 * full[n - 1 : n - 1 + n]
    return smooth, grad_a

# ----------------------------------------------------------------------
# ADAM optimisation on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule: list[float],
                   iter_per_beta: int = 200) -> tuple[np.ndarray, float]:
    """Run ADAM on the log‑sum‑exp surrogate.  Returns best ℓ₁‑normalised seq."""
    eps = 1e-30
    v = np.log(start_seq + eps)          # logits → simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    t = 0
    best_seq = None
    best_val = np.inf

    for beta in beta_schedule:
        for _ in range(iter_per_beta):
            if time.time() > deadline:
                break
            t += 1
            a = _softmax(v)
            smooth, grad_a = _smooth_max_and_grad(a, beta)
            loss = 2.0 * n * smooth
            # chain rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # ADAM moments
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            # bias‑corrected
            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            # learning‑rate schedule (slow decay)
            lr = 0.5 / (np.sqrt(t) + 1e-8)

            v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-8)

            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    if best_seq is None:
        best_seq = start_seq.copy()
        best_val = _objective(best_seq)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact max‑convolution mirror‑descent (multiplicative updates)
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 2_000_000,
                                 lr0: float = 0.18,
                                 power: float = 0.30) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 1
    while it <= max_iters:
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        max_idx = int(np.argmax(conv))

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:,d} best={cur_val:.6f}")

        # sub‑gradient for the active anti‑diagonal
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / (it ** power)
        a = a * np.exp(-lr * g)

        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light L‑BFGS on the smooth‑max surrogate (final fine‑tune)
# ----------------------------------------------------------------------
def _lbfgs_optimize(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iter: int = 200,
                    temperature: float = 2000.0) -> tuple[np.ndarray, float]:
    """Wrap SciPy’s L‑BFGS‑B on the smooth‑max surrogate."""
    eps = 1e-30
    v = np.log(start_seq + eps)

    def fun(v):
        a = _softmax(v)
        smooth, grad_a = _smooth_max_and_grad(a, temperature)
        loss = 2.0 * n * smooth
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)          # ∂loss/∂v
        return loss, (2.0 * n * grad_v).astype(np.float64)

    try:
        res = scipy.optimize.minimize(
            fun,
            v,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
    except Exception:  # pragma: no cover
        return start_seq, _objective(start_seq)

    a_opt = _softmax(res.x)
    return a_opt, _objective(a_opt)


# ----------------------------------------------------------------------
# Light simulated‑annealing polishing
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 500_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:  # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:  # redistribute a small random subset
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:  # split‑transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 200_000 == 0:
                    print(f"[SA] it={it:,d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective.  The algorithm follows the three‑stage
    scheme described in the module docstring and stores the best
    sequence in the global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – larger n gives a lower theoretical bound.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Load a previously found solution (if any) for seeding.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – generate a big and diverse seed pool.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # Fast surrogate ranking – keep the most promising seeds.
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # Initialise the global best.
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – ADAM optimisation on the smooth‑max surrogate.
    # ------------------------------------------------------------------
    beta_schedule = [5, 10, 20, 40, 80, 160, 320, 640, 1280, 2560]
    for idx, seed_vec in enumerate(best_seeds, start=1):
        if time.time() > deadline:
            break
        seq_opt, val_opt = _adam_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule,
            iter_per_beta=200,
        )
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_seq = seq_opt.copy()
            print(f"[Adam] seed #{idx} → new best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Mirror‑descent sub‑gradient optimisation on the exact max.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent (exponentiated‑gradient) ---")
        seq_mir, val_mir = _mirror_subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=3_000_000,
            lr0=0.24,
            power=0.32,
        )
        if val_mir < global_best_val:
            global_best_val = val_mir
            global_best_seq = seq_mir.copy()
            print(f"[Mirror] new global best = {val_mir:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Brief L‑BFGS fine‑tune on the smooth surrogate.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- L‑BFGS fine‑tune ---")
        seq_lb, val_lb = _lbfgs_optimize(
            global_best_seq,
            n,
            deadline,
            max_iter=200,
            temperature=3000.0,
        )
        if val_lb < global_best_val:
            global_best_val = val_lb
            global_best_seq = seq_lb.copy()
            print(f"[L‑BFGS] new global best = {val_lb:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=500_000,
            init_temp_factor=0.04,
            min_temp=1e-12,
        )
        if val_sa < global_best_val:
            global_best_val = val_sa
            global_best_seq = seq_sa.copy()
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (informational only).
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Eval] evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for possible reuse by a later run.
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
