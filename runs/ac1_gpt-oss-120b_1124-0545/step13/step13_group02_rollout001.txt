    best_x = x.copy()
    best_val = _objective_surrogate(x)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))
        # compute subgradient
        grad = np.zeros_like(x)
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        idx = np.arange(i_start, i_end+1, dtype=int)
        rev = k - idx
        grad[idx] = 2.0 * x[rev]
        eta = eta0 / math.sqrt(it)
        y = x - eta * grad
        # project onto simplex
        if y.sum() <= 0:
            y = _uniform(n)
        else:
            y = _proj_simplex(y)
        x = y
        val = _objective_surrogate(x)
        if val < best_val:
            best_val = val
            best_x = x.copy()
    return best_x, best_val

"""
Propose a non‑negative step function (list of heights) that minimizes
the official ``evaluate_sequence`` objective.

Algorithm (≈ 1000 s budget)
----------------------------
0️⃣  **Seed generation** – a large pool of analytically‑derived and
    random shapes (uniform, triangular, sine‑windows, Gaussians,
    beta‑shapes, Tukey windows, exponentials, Dirichlet draws,
    low‑frequency cosine sums, mixtures).  All seeds are ℓ₁‑normalised
    (∑ = 1).  The best 30 seeds (by the fast surrogate 2·n·max(conv))
    initialise a population.

🌀  **Evolutionary exploration** – while time permits, repeatedly
    select a parent from the current population, apply a cheap
    mutation (random Dirichlet reshuffle on a small subset + tiny
    Gaussian noise), re‑normalise onto the simplex, and optionally run
    a few hundred sub‑gradient steps (additive mirror descent) to
    polish.  If the offspring improves the surrogate it replaces the
    worst individual; the global best is tracked.

⚙️  **Deep polishing** – the current best vector undergoes three
    increasingly precise refinements:

    • **Soft‑max Adam** – optimises a smooth approximation of max
      (log‑sum‑exp) with an annealed temperature α that grows from 20
      to 3000.

    • **Exponential‑gradient (multiplicative mirror descent)** –
      directly minimises the true max‑convolution using the sub‑gradient
      at the active shift.

    • **Simulated‑annealing** – a final stochastic fine‑tuning that
      reshuffles small subsets and performs tiny mass transfers,
      accepting uphill moves with a slowly cooling temperature.

🧪  **Evaluation** – the final surrogate value is printed together with
    the true ``evaluate_sequence`` value (the latter is the score
    reported to the competition).

All stages respect the global deadline (budget s − 2 s safety margin)
and the code runs on a single thread.  The resulting sequence is stored
in the global name ``height_sequence_1`` for possible warm‑starts in
future runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """Fast surrogate: 2·n·max(conv(p,p)) for a distribution p (∑p=1)."""
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    """Monotonically decreasing exponential, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (0.5, 1.0, 1.5, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.1, 0.2, 0.3, 0.4):
        seeds.append(_gaussian(n, sigma))
    for rate in (0.9998, 0.9995, 0.999):
        seeds.append(_exponential_decay(n, rate))
    for p in (0.3, 0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.4, 0.6, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws (different concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution if supplied and of matching size
    if previous is not None:
        if previous.shape == (n,) and previous.sum() > 0:
            seeds.append(previous / previous.sum())

    # final sanity‑check / normalisation
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Multiplicative mirror‑descent (exponentiated‑gradient)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient at active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # avoid overflow/underflow
        x = x * np.exp(delta)

        # Euclidean projection onto simplex
        s = x.sum()
        if s <= 0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


def _adam_softmax_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    alpha_start: float = 20.0,
    alpha_end: float = 2000.0,
    lr: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        L(p) = (1/α)·log Σₖ exp(α·(p*p)[k]).
    α is linearly annealed from ``alpha_start`` to ``alpha_end``.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    N = _next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution p * p (full) via FFT
        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        # soft‑max over the convolution
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        # correlation p ⋆ soft (gradient of L)
        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # projection onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


def _simulated_annealing(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.6:
            # reshuffle a random subset (size 3‑12)
            k_sub = rng.integers(3, 13)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
    return best, best_val


def _subgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 50_000,
    eta0: float = 0.2,
) -> tuple[np.ndarray, float]:
    """Additive mirror‑descent (sub‑gradient) for the true max‑convolution."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient at active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        y = x - eta * g

        # projection onto simplex
        s = y.sum()
        if s <= 0.0:
            y = _uniform(n)
        else:
            y = _proj_simplex(y)

        x = y
        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    the official ``evaluate_sequence`` objective within the given budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Length selection – 5000 gives a reasonable granularity without
    # making each iteration too expensive.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Load previous solution if it matches the current length.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(prev, (list, tuple)):
        arr = np.asarray(prev, dtype=np.float64)
        if arr.shape == (n,) and np.isfinite(arr).all() and arr.sum() > 0:
            prev_arr = arr / arr.sum()

    # ------------------------------------------------------------------
    # Stage 0 – seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # Score seeds with the fast surrogate and keep the best few
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    popsize = 30
    population = [vec for vec, _ in scored[:popsize]]
    pop_vals = [val for _, val in scored[:popsize]]

    best_idx = int(np.argmin(pop_vals))
    best_seq = population[best_idx].copy()
    best_val = pop_vals[best_idx]
    print(f"Initial best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – evolutionary exploration with cheap sub‑gradient polish
    # ------------------------------------------------------------------
    print("\n--- Evolutionary exploration ---")
    gen = 0
    while time.time() < deadline - 10.0:   # keep ~10 s for final polishing
        gen += 1
        # parent from the top half of the population
        parent_idx = rng.integers(popsize // 2)
        parent = population[parent_idx]

        # ---- mutation -------------------------------------------------
        child = parent.copy()

        # a) random Dirichlet reshuffle on a small random subset
        k_sub = rng.integers(3, 13)
        idx = rng.choice(n, size=k_sub, replace=False)
        tot = child[idx].sum()
        if tot > 0.0:
            child[idx] = rng.dirichlet(np.full(k_sub, 1.0)) * tot

        # b) tiny Gaussian noise (ensuring non‑negativity)
        noise = rng.normal(scale=0.005, size=n)
        noise = np.where(child + noise > 0, noise, -child)  # clamp to >=0
        child = child + noise

        # renormalise onto the simplex
        s = child.sum()
        if s <= 0.0:
            child = _uniform(n)
        else:
            child = child / s

        # ---- cheap sub‑gradient polish (few thousand steps) ----------
        if time.time() < deadline - 10.0:
            child, _ = _subgrad_optimize(child, n, deadline, max_iters=4000, eta0=0.15)

        # evaluate surrogate
        val = _objective_surrogate(child)

        # replace worst individual if improved
        worst_idx = int(np.argmax(pop_vals))
        if val < pop_vals[worst_idx]:
            population[worst_idx] = child
            pop_vals[worst_idx] = val
            if val < best_val:
                best_val = val
                best_seq = child.copy()
                print(f"[Gen {gen:4d}] new global best surrogate = {best_val:.6f} "
                      f"(time {time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # Stage 2 – deep polishing (Adam → EG → SA)
    # ------------------------------------------------------------------
    # 2.1 Soft‑max Adam optimisation
    if time.time() < deadline - 2.0:
        print("\n--- Soft‑max Adam optimisation (final) ---")
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 200))
        best_seq, best_val = _adam_softmax_optimize(
            best_seq, n, deadline,
            max_iters=max_iters,
            alpha_start=20.0,
            alpha_end=3000.0,
            lr=0.12,
        )
        print(f"[Adam] best surrogate after Adam = {best_val:.6f}")

    # 2.2 Exponential‑gradient polishing
    if time.time() < deadline - 2.0:
        print("\n--- Exponential‑gradient polishing (final) ---")
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 1500))
        best_seq, best_val = _expgrad_optimize(
            best_seq, n, deadline, max_iters=max_iters, eta0=0.25
        )
        print(f"[EG] best surrogate after EG = {best_val:.6f}")

    # 2.3 Simulated‑annealing fine‑tuning
    if time.time() < deadline - 2.0:
        print("\n--- Simulated‑annealing polishing (final) ---")
        remaining = deadline - time.time()
        max_iters = int(min(600_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(
            best_seq, n, deadline, max_iters=max_iters
        )
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (the score reported to the platform)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for warm‑start in a future run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
