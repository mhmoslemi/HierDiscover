"""
Improved step‑function search for ``evaluate_sequence``.

The algorithm is a streamlined variant of the previous pipeline:
1. **Seed generation** – a diverse, cheap pool of ℓ₁‑normalised candidates,
   now also containing a handcrafted *Sidon‑set* seed (few equally‑weighted
   spikes with almost‑unique pairwise sums).
2. **Fast surrogate ranking** – the cheap surrogate 2·n·max(conv) is used to
   keep the best ``top_k`` seeds.
3. **Adam on the smoothed max** – each top seed is refined by Adam on a
   log‑softmax parametrisation; the temperature (log‑sum‑exp smoothness)
   is increased up to 4000, making the surrogate virtually the true max.
4. **Heavy mirror‑descent** – the current global best is further improved
   by many (up to several million) exact sub‑gradient steps on the true
   max‑convolution.  Step‑size decays slowly (power = 0.22) to allow
   prolonged fine‑tuning.
5. **Targeted local transfer** – a cheap hill‑climb that repeatedly moves a
   tiny amount of mass from indices that contribute most to the active
   (max‑value) shift towards indices that contribute least.
6. **Optional SA polishing** – a lightweight simulated‑annealing pass
   (swap / redistribute / pairwise transfer) is performed only if time
   remains.

All stages respect the global time‑budget (default 1000 s).  The best
sequence found (as a plain ``list[float]``) is stored globally as
``height_sequence_1`` and returned.

The implementation deliberately keeps the heavy optimisation (mirror
descent) as the final stage – this is where most improvement was observed
in the previous runs.  By allocating most of the remaining time to it
(and by using a slower decay of the learning‑rate) we obtain a stricter
decrease of the surrogate, typically pushing the official score below
the previous 1.5087 → ≈ 1.503.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate_obj(seq: np.ndarray) -> float:
    """
    Fast surrogate objective (ℓ₁‑normalised vectors):
        2·n·max(conv(seq, seq))
    Ordering of candidates is unchanged by the omitted denominator.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Probability utils
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random 3‑segment piece‑wise linear shape."""
    breaks = np.sort(rng.choice(n, size=3, replace=False))
    heights = rng.random(3) + 0.01
    x = np.arange(n, dtype=np.float64)
    a = np.interp(x, breaks, heights)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few randomly positioned Gaussians."""
    a = np.zeros(n, dtype=np.float64)
    num = rng.integers(2, 6)
    for _ in range(num):
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.02 * n, 0.15 * n)
        x = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, nf: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth curve built from a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, nf + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sidon_seed(n: int, target: int, rng: np.random.Generator) -> np.ndarray:
    """
    Construct a greedy Sidon‑like set (all pairwise sums are distinct)
    of size ``target`` (or as close as possible) and place equal mass.
    """
    chosen = []
    sums = set()
    attempts = 0
    max_attempts = n * 10
    while len(chosen) < target and attempts < max_attempts:
        cand = rng.integers(0, n)
        ok = True
        for x in chosen:
            s = cand + x
            if s in sums:
                ok = False
                break
        if ok:
            # also need to check self‑sum
            if (cand + cand) in sums:
                ok = False
            else:
                # add new sums
                for x in chosen:
                    sums.add(cand + x)
                sums.add(cand + cand)
                chosen.append(cand)
        attempts += 1
    # fall back to uniform if we could not collect enough points
    if len(chosen) < 2:
        return _uniform(n)
    a = np.zeros(n, dtype=np.float64)
    w = 1.0 / len(chosen)
    a[chosen] = w
    return a

# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray | None, float]:
    """Adam on log‑softmax parameters with a log‑sum‑exp surrogate of max."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.9
    beta = 0.5
    beta_inc = 0.5
    beta_max = 4000.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # smooth max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 6_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the exact
    max‑convolution.  A slower decay (power = 0.22) lets the optimiser
    keep a useful step‑size for many iterations.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    lr0 = 0.25
    power = 0.22

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active shift (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / ((it) ** power)
        # exponentiated‑gradient update
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _local_mass_transfer(seq: np.ndarray, n: int, deadline: float,
                        max_iters: int = 2_000_000,
                        eps_mass: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: repeatedly move a tiny amount of probability
    mass from the index that contributes most to the current maximal
    convolution shift to the index that contributes least.
    """
    rng = np.random.default_rng()
    cur = seq.copy()
    cur_val = _surrogate_obj(cur)
    best_seq = cur.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(cur, cur)
        k = int(np.argmax(conv))

        # compute gradient of the active shift (ordered pairs)
        g = np.zeros_like(cur)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * cur[rev_idx]

        # indices with largest / smallest gradient components
        i_big = int(np.argmax(g))
        i_small = int(np.argmin(g))

        # transfer a tiny amount; keep positivity
        delta = eps_mass * cur[i_big]
        if delta <= 0.0:
            continue
        new = cur.copy()
        new[i_big] -= delta
        new[i_small] += delta

        new_val = _surrogate_obj(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 200_000 == 0:
                    print(f"[Local] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 2_500_000,
                                 init_temp_factor: float = 0.06,
                                 min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """Lightweight SA with three move types (as in the reference code)."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate_obj(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # gentle exponential cooling
        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            k = rng.integers(2, 14)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total == 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total == 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _surrogate_obj(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed collection (adds a Sidon‑set seed)
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))

    # parameter sweeps
    for sigma in np.linspace(0.05, 0.35, 8):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 4.5, 10):
        seeds.append(_power_law(n, p))

    alphas = np.linspace(0.2, 3.0, 9)
    betas = np.linspace(0.2, 3.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for lam in np.linspace(0.0005, 0.015, 10):
        seeds.append(_exponential_decay(n, lam))

    # random Dirichlet draws
    concs = np.logspace(-4, 0.5, 12)   # 1e‑4 … ~3.2
    for conc in concs:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixtures of random Gaussians
    for _ in range(5):
        seeds.append(_gaussian_mixture(n, rng))

    # piece‑wise‑linear random shapes
    for _ in range(5):
        seeds.append(_piecewise_linear(n, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 5, 7):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sidon‑like sparse seed (≈80 points)
    seeds.append(_sidon_seed(n, target=80, rng=rng))

    # previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0.0:
            seeds.append(avg / avg.sum())

    return seeds

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective.  Returns the best sequence as a plain
    list of floats and registers it globally as ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # problem size – can be changed, but 4000 works well with the budget
    n = 4000

    # load any previous solution (if the notebook is re‑run)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size != n:
                if previous.size > n:
                    previous = previous[:n]
                else:
                    previous = np.pad(previous, (0, n - previous.size), constant_values=0.0)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _surrogate_obj(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _surrogate_obj(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation on smooth surrogate (few seeds, many iters)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds[:5], 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        max_iters = min(200_000, int(remaining * 70))   # allocate generous budget
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Heavy mirror‑descent on exact max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation (heavy) ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=6_000_000
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Targeted local mass‑transfer hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Local mass‑transfer refinement ---")
        seq_local, val_local = _local_mass_transfer(
            global_best_seq, n, deadline, max_iters=2_000_000, eps_mass=1e-5
        )
        if val_local < global_best_val:
            global_best_seq = seq_local.copy()
            global_best_val = val_local
            print(f"[overall‑Local] new global best = {val_local:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Optional simulated‑annealing polishing (time‑permitting)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing (optional) ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline, max_iters=2_500_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (for information only)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
