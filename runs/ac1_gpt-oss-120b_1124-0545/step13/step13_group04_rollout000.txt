    while time.time() < deadline - 0.1:

        # increase beta slowly up to target
        beta = min(beta_target, beta * 1.001)

        # compute conv
        conv = _conv_fft(a, a)

"""
Propose a step‑function (probability vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

using a smoothed‐max (log‑sum‑exp) surrogate and exponentiated‑gradient
(Adam) descent on the probability simplex.  The smoothing parameter β is
started small and increased slowly so that early iterations see a smooth
objective (many “large” convolution entries receive weight) while later
iterations focus on the true maximum.  The algorithm runs as many
iterations as possible inside the supplied time budget, keeps the best
vector seen, and finally evaluates it with the official ``evaluate_sequence``.
If a previous solution (global ``height_sequence_1``) exists it is used as a
warm start.

The implementation uses only NumPy (FFT for fast convolutions) and works
with any length `n` (default 8000).  It respects the 1000 s budget and
returns a plain list of floats.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Low‑level utilities (FFT convolution, softmax, etc.)
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT (O(N log N)).
    The result length is len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    max_log = np.max(logits)
    exps = np.exp(logits - max_log)
    s = np.sum(exps)
    return exps / s


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises the ``evaluate_sequence`` objective within ``budget_s`` seconds.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final expensive call
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------
    # Problem size – keep the same as the previous best solution
    # --------------------------------------------------------
    n = 8000

    # --------------------------------------------------------
    # Initialise a (probability vector) – warm start if possible
    # --------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            a0 = np.asarray(prev, dtype=np.float64)
            if a0.shape == (n,) and np.sum(a0) > 0.0:
                a = a0 / np.sum(a0)
                print("[Init] Warm start from previous solution.")
            else:
                raise ValueError
        except Exception:
            a = np.full(n, 1.0 / n, dtype=np.float64)
            print("[Init] Falling back to uniform start (previous invalid).")
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)
        print("[Init] Uniform start.")

    # log‑parameters for exponentiated gradient (Adam)
    logits = np.log(a + 1e-30)               # ensure positivity
    m = np.zeros_like(logits)                # first moment
    v = np.zeros_like(logits)                # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.5

    # smoothing (log‑sum‑exp) parameter β
    beta = 1.0           # start smooth
    beta_target = 2000.0 # final – close to true max
    beta_mul = 1.001     # slow geometric growth

    # store the best vector seen so far (surrogate = 2·n·max conv)
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)
    best_a = a.copy()
    last_improve_iter = 0

    it = 0
    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline - 0.5:
            break
        it += 1

        # ----------------------------------------------------
        # 1) compute convolution of current a
        # ----------------------------------------------------
        conv = _conv_fft(a, a)                     # length 2n-1
        max_conv = np.max(conv)

        # ----------------------------------------------------
        # 2) smooth approximation of max: w_k ∝ exp(β·conv_k)
        # ----------------------------------------------------
        # shift by max to avoid overflow
        shifted = beta * (conv - max_conv)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        w /= w_sum                                 # now Σ w = 1

        # ----------------------------------------------------
        # 3) gradient of the smoothed objective w.r.t. a
        #    ∂f/∂a_i = 2 * Σ_k w_k * a_{k-i}
        #    This is a convolution of w with the reversed a.
        # ----------------------------------------------------
        a_rev = a[::-1]
        conv_w_rev = _conv_fft(w, a_rev)          # length = (2n-1)+n-1 = 3n-2
        # extract indices n-1-i (i from 0..n-1)
        idx = (n - 1) - np.arange(n)
        grad = 2.0 * conv_w_rev[idx]               # shape (n,)

        # ----------------------------------------------------
        # 4) Adam update on logits (exponentiated‑gradient step)
        # ----------------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # descent direction (we minimise)
        logits -= base_lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----------------------------------------------------
        # 5) map back to the simplex via softmax
        # ----------------------------------------------------
        a = _softmax(logits)

        # ----------------------------------------------------
        # 6) evaluate surrogate (using the freshly computed conv)
        # ----------------------------------------------------
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val - 1e-9:  # significant improvement
            best_val = cur_val
            best_a = a.copy()
            last_improve_iter = it
            if it % 100 == 0:
                elapsed = now - start_time
                print(f"[Iter] {it:7d} new best surrogate = {best_val:.6f} "
                      f"elapsed={elapsed:.1f}s β={beta:.1f}")

        # ----------------------------------------------------
        # 7) schedule for β (smoothness)
        # ----------------------------------------------------
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        # ----------------------------------------------------
        # 8) occasional random perturbation to escape stagnation
        # ----------------------------------------------------
        if it % 5000 == 0 and (it - last_improve_iter) > 5000:
            # add a small Gaussian noise to logits
            noise = rng.normal(scale=0.02, size=n)
            logits += noise
            print(f"[Perturb] iteration {it} – added noise to logits.")

    # --------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_a.tolist())
        print("\n=== Finished ===")
        print(f"Best surrogate ≈ {best_val:.6f}")
        print(f"True evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for reuse
    globals()["height_sequence_1"] = best_a.tolist()
    return best_a.tolist()
