"""
search for a step‑function that minimises
    eval(a) = 2·n·max(conv(a,a)) / (sum(a))**2
with a_i ≥ 0.
The strategy is simple but surprisingly effective:

* Randomly sample a huge number of points on the probability simplex.
  We use several families:
    – Dirichlet(α) for a few values of α (α=1 is just normalised
      Exponential, α>1 gives flatter vectors, α<1 gives sparser vectors)
    – Uniform weight on a random subset of indices (the “subset‑uniform”
      family).  This yields extremely low pairwise‑sum multiplicities.
* Keep the best candidate according to a fast FFT‑based surrogate:
      2·n·max(conv(a,a))   (the denominator cancels because we keep
      sum(a)=1 during the search).
* When the time budget is almost exhausted, polish the best vector with a
  short exponentiated‑gradient (mirror‑descent) run that directly minimises
  the true max‑convolution.

The whole routine respects the 1000 s budget, prints progress and finally
stores the best sequence in the global name ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns a vector with non‑negative entries summing to 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate objective (exact for ∑seq = 1):
       2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative updates)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # stepsize (√t schedule)
        eta = eta0 / math.sqrt(it)

        # multiplicative update (clip exponent for safety)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # project onto simplex
        s = x.sum()
        if s <= 0.0:
            x = np.full_like(x, 1.0 / n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val

# Sampling helpers
# ----------------------------------------------------------------------
def _sample_dirichlet(alpha: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) point by normalising Gamma(α,1) samples."""
    if alpha == 1.0:
        # exponential is faster than generic gamma for α=1
        a = rng.exponential(scale=1.0, size=n)
    else:
        a = rng.gamma(shape=alpha, scale=1.0, size=n)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _sample_subset_uniform(n: int, rng: np.random.Generator,
                           min_frac: float = 0.4, max_frac: float = 1.0) -> np.ndarray:
    """Uniform weight on a random subset of the coordinates."""
    # pick size m ∈ [min_frac·n , max_frac·n]
    lo = max(1, int(min_frac * n))
    hi = max(lo, int(max_frac * n))
    m = rng.integers(lo, hi + 1)
    idx = rng.choice(n, size=m, replace=False)
    seq = np.zeros(n, dtype=np.float64)
    seq[idx] = 1.0 / m
    return seq
