    # Evaluate surrogate (2*n * max(conv) for each L1-normalized)
    survals = [_objective_surrogate(s) for s in seeds]
    # Determine top seeds
    top_idx = np.argsort(survals)[:8]
    top_seeds = [seeds[i] for i in top_idx]

    # keep best globally
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print("Initial best surrogate:", best_val)

    # Define gradient descent optimizer function
    def adam_optimize(start_a, max_iters, beta_start, beta_end):
        # start from logit of start_a
        logits = np.log(start_a + 1e-30)
        m = np.zeros_like(logits)
        v = np.zeros_like(logits)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        for it in range(1, max_iters + 1):
            # update beta linearly
            beta = beta_start + (beta_end - beta_start) * (it - 1) / (max_iters - 1)
            # compute a
            a = _softmax(logits)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            # compute softmax weights over conv
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp
            # gradient wrt a
            rev_a = a[::-1]
            grad_a = 2.0 * _conv_fft(rev_a, w)[n-1:n-1+n]
            # gradient wrt logits
            dot = np.dot(grad_a, a)
            grad_logits = a * (grad_a - dot)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_logits
            v = beta2 * v + (1 - beta2) * (grad_logits ** 2)
            m_hat = m / (1 - beta1**it)
            v_hat = v / (1 - beta2**it)
            lr = 0.5 / (np.sqrt(it) + 1e-8)   # diminishing
            logits = logits - lr * m_hat / (np.sqrt(v_hat) + eps)

            if it % 5000 == 0:
               cur_val = _objective_surrogate(_softmax(logits))
               # optional early stop if no improvement
            # timer check:
            if time.time() > deadline:
                break
        return _softmax(logits)

    # For each top seed run Adam with schedule
    for i, seed in enumerate(top_seeds):
        if time.time() > deadline: break
        # allocate iterations based on remaining time
        remaining = deadline - time.time()
        # Assuming each iteration ~0.0005 s, allocate accordingly
        max_iters = int(min(200_000, remaining * 1200))  # arbitrary scaling

        print(f"Optimizing seed {i+1} with {max_iters} iterations")
        # Beta schedule from 1 to 5000
        optimized = adam_optimize(seed, max_iters, beta_start=1.0, beta_end=5000.0)
        val_opt = _objective_surrogate(optimized)

        if val_opt < best_val:
            best_seq = optimized.copy()
            best_val = val_opt
            print(f"New global best surrogate {best_val:.6f}")

    # Use 300k steps or until time runs out

        sa_steps = int(min(500_000, remaining * 2500))
        print(f"Running SA final polishing with {sa_steps} steps")
        logits = np.log(best_seq + 1e-30)
        cur_seq = best_seq
        cur_val = best_val
        best_seq_sa = best_seq.copy()
        best_val_sa = best_val
        # temperature schedule linearly decreasing from 0.1 to 0
        for it in range(1, sa_steps + 1):
            if time.time() > deadline:
                break
            progress = it / sa_steps
            T = 0.1 * (1 - progress) + 1e-6
            sigma = 0.15 * (1 - progress) + 1e-4
            proposal_logits = logits + sigma * rng.normal(size=n)
            proposal_seq = _softmax(proposal_logits)
            proposal_val = _objective_surrogate(proposal_seq)
            if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
                logits = proposal_logits
                cur_seq = proposal_seq
                cur_val = proposal_val
                if cur_val < best_val_sa:
                    best_val_sa = cur_val
                    best_seq_sa = cur_seq.copy()
            if it % 50000 == 0:
                print(f"[SA] step {it} best surrogate {best_val_sa:.6f}")
        if best_val_sa < best_val:
            best_seq = best_seq_sa.copy()
            best_val = best_val_sa
            print(f"After SA best surrogate {best_val:.6f}")

    # Evaluate true objective
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print("True evaluate_sequence:", true_val)
    except Exception as e:
        print("Error evaluating:", e)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

"""
Propose a step function with a smaller evaluation value.

Algorithm
---------
1.  Build a diverse pool of ℓ¹‑normalised candidate sequences of length `n=4000`
    (uniform, triangular, sine windows, Gaussians, power‑law, Beta‑shapes,
    random Dirichlet / exponential draws, and the previous best if present).

2.  Rank the pool with the cheap surrogate `2·n·max(conv(a,a))` (exact for
    ℓ¹‑normalised vectors) and keep the best seed.

3.  Polish the best seed with a *smooth‑max* Adam optimiser.
    The smooth max is the log‑sum‑exp of the convolution:
        f_β(a)=2·n·( max_c + log(∑exp(β·(c−max_c))) / β )
    Its gradient w.r.t. the softmax logits `z` is computed analytically.
    β is increased linearly from 1 to a large value (≈5000) while the
    Adam step‑size decays ∝ 1/√t.  This directly minimises the true
    objective on the simplex without the need for projections.

4.  If time permits, a final simulated‑annealing walk in the logit space
    refines the result further.

5.  Return the best sequence found and store it in the global variable
    ``height_sequence_1`` for the next round.

The whole routine respects the 1000 s time budget, prints progress and
uses only NumPy (no external solvers).
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a *ℓ¹‑normalised* sequence:
        2·n·max(conv(seq,seq))
    This is the quantity minimised during the search.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (always ℓ¹‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta PDF sampled at integer points 1…n (then normalised)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Optimisation utilities (smooth‑max Adam)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_a: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int,
                     beta_start: float = 1.0,
                     beta_end: float = 5000.0) -> np.ndarray:
    """
    Adam optimiser in the log‑softmax space.
    ``start_a`` must be ℓ¹‑normalised.
    """
    # initialise logits (log‑softmax parameters)
    logits = np.log(start_a + 1e-30)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear schedule for β
        beta = beta_start + (beta_end - beta_start) * (it - 1) / (max_iters - 1)

        a = _softmax(logits)                     # stays on simplex
        conv = _conv_fft(a, a)                    # full convolution, length 2n‑1
        max_c = np.max(conv)
        shift = max_c

        # softmax weights over the convolution entries
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = np.sum(exp_beta)
        w = exp_beta / sum_exp

        # gradient w.r.t. a : 2 * conv(reverse(a), w)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_logits = a * (grad_a - dot)

        # clipping prevents occasional exploding updates
        grad_logits = np.clip(grad_logits, -20.0, 20.0)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # simple diminishing base learning rate
        lr = 0.6 / (math.sqrt(it) + 1.0)

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report
        if it % 25000 == 0 or it == max_iters:
            cur_val = _objective_surrogate(_softmax(logits))
            print(f"    [Adam] it={it:7d} β={beta:8.1f} surrogate={cur_val:.6f}")

    return _softmax(logits)


def _simulated_annealing(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         max_steps: int) -> np.ndarray:
    """
    Simple SA on the logit space (softmax to keep feasibility).
    """
    rng = np.random.default_rng()
    logits = np.log(start_a + 1e-30)
    cur_seq = start_a.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for step in range(1, max_steps + 1):
        if time.time() > deadline:
            break

        progress = step / max_steps
        # linear cooling, small final temperature
        T = 0.08 * (1.0 - progress) + 1e-6
        sigma = 0.12 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        # Metropolis acceptance
        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if step % 50000 == 0:
            print(f"    [SA] step={step:7d} best surrogate={best_val:.6f}")

    return best_seq

# Candidate proposer

    deadline = start_time + budget_s - 2.0   # small safety margin

    n = 4000

    # ------------------------------------------------------------------
    seeds = []

        print(f"Polishing with Adam for {max_iters:,} iterations "
              f"(β from 1 → 5000)...")
        polished = _adam_smooth_opt(best_seed, n, deadline, max_iters,
                                    beta_start=1.0, beta_end=5000.0)
        polished_val = _objective_surrogate(polished)
        print(f"    After Adam surrogate = {polished_val:.6f}")

        if polished_val < best_val:
            best_seed = polished
            best_val = polished_val
            print(f"    Updated global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final simulated‑annealing refinement (optional)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10:
        max_steps = int(min(300_000, remaining * 2000))
        print(f"Running simulated‑annealing for {max_steps:,} steps...")
        final_seq = _simulated_annealing(best_seed, n, deadline, max_steps)
        final_val = _objective_surrogate(final_seq)
        print(f"    After SA surrogate = {final_val:.6f}")

        if final_val < best_val:
            best_seed = final_seq
            best_val = final_val
            print(f"    Updated global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seed.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose the result for the next round
    globals()["height_sequence_1"] = best_seed.tolist()
    return best_seed.tolist()
