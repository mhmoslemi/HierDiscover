def propose_candidate(seed=123, budget_s=1000):
    import time, numpy as np, scipy.optimize as opt

    start_time = time.time()
    deadline = start_time + budget_s - 2

    n = 4000

    # load previous best if exists
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except:
            prev_arr = None

    rng = np.random.default_rng(seed)

    # generate seeds
    seeds = _collect_seeds_extended(n, rng, prev_arr)
    # evaluate surrogate of each (or true) to pick top k
    scores = [(vec, _objective_surrogate(vec)) for vec in seeds]
    scores.sort(key=lambda x: x[1])
    top_seeds = [vec for vec,_ in scores[:3]] # maybe 3

    best_seq = top_seeds[0]
    best_val = _objective_surrogate(best_seq)

    # L-BFGS Phase
    beta_sched = [1.,5.,20.,80.,200.,500.,1000.,2000.,5000.]
    for seed_idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        max_per_beta = int(min(15000, (deadline - time.time()) * 0.12 * 1e5))
        seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=max_per_beta)
        val_opt = _objective_surrogate(seq_opt)
        if val_opt < best_val:
            best_seq = seq_opt
            best_val = val_opt

    # further refinement with high beta
    if time.time() < deadline:
        # start from best_seq
        # 1) extra LBFGS with many iter and high beta
        beta_sched2 = [5000.,10000.]
        extra_iters = int(min(30000, (deadline - time.time()) * 0.2 * 1e5))
        seq_ref = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                         beta_schedule=beta_sched2,
                                         max_iter_per_beta=extra_iters)
        val_ref = _objective_surrogate(seq_ref)
        if val_ref < best_val:
            best_seq = seq_ref
            best_val = val_ref

    # exponentiated gradient refinement
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(500_000, remaining * 0.3 * 1e5))
        seq_exp, val_exp = _exp_gradient_optimize(best_seq, n, deadline,
                                                  max_iters=max_exp_iters)
        if val_exp < best_val:
            best_seq = seq_exp
            best_val = val_exp

    # random local search
    if time.time() < deadline:
        # Use simple hill climbing
        cur_seq = best_seq
        cur_val = best_val
        # Ensure normalized to sum=1
        cur_seq = cur_seq / cur_seq.sum()
        # Determine temperature schedule
        it = 0
        while time.time() < deadline:
            it += 1
            # noise magnitude decreasing with sqrt of iter or time
            sigma = 0.1 / np.sqrt(it)  # adjust
            # propose new sequence
            log_cur = np.log(cur_seq + 1e-30)
            perturb = rng.normal(scale=sigma, size=n)
            new_log = log_cur + perturb
            new_seq = _softmax(new_log)  # automatically normalised
            new_val = _objective_surrogate(new_seq)
            if new_val < cur_val:
                cur_seq = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_seq = new_seq
                    best_val = new_val
            # optionally reduce sigma after some steps

    # final evaluation via evaluate_sequence

    globals()["height_sequence_1"] = best_seq.tolist()

"""
Propose a non‑negative step‑function (list of floats) that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

within a 1000 s budget.

The algorithm combines:
  1. A rich pool of analytic and random seeds.
  2. Continuation L‑BFGS on a smoothed “log‑sum‑exp” surrogate of the max‑convolution.
  3. An Adam‑style mirror‑descent on the exact surrogate with a very high β.
  4. A cheap stochastic hill‑climbing phase (log‑softmax perturbations).
All vectors are ℓ₁‑normalised, never contain negatives and are kept on the probability
simplex via the softmax parametrisation.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(a: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(a,a)) for an ℓ₁‑normalised vector.
    (If sum(a)==1 then the true objective equals this value.)
    """
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _objective_exact(a: np.ndarray) -> float:
    """Exact objective used for the final check (identical to evaluate_sequence)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_b = np.max(conv)
    sum_a = a.sum()
    if sum_a < 0.01:        # should never happen – just a guard
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic shape families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection (diverse pool of ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_tukey_window(n, alpha=0.4))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_gaussian(n, sigma_frac=0.35))
    seeds.append(_power(n, p=0.5))
    seeds.append(_power(n, p=1.0))
    seeds.append(_power(n, p=2.0))

    # beta grid (finer than before)
    alphas = np.linspace(0.2, 5.0, 11)
    betas = np.linspace(0.2, 5.0, 11)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (different concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final safety normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector (softmax of the logits).
    """
    if beta_schedule is None:
        # start small, finish extremely sharp
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0]

    # initialise logits
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                # probability simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                  # max(conv)
            # stable log‑sum‑exp of β·(conv–shift)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            # smoothed objective
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

            # gradient w.r.t. a
            w = exp_beta / sum_exp                # softmax‑like weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule (softmax Jacobian)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] Time limit reached during beta={beta}")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] Exception {e} at beta={beta}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Adam‑style mirror‑descent on the smoothed surrogate
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta: float = 5000.0,
                   max_iters: int = 300_000,
                   lr: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Adam on logits using the smooth max (log‑sum‑exp) with a large β.
    Returns the best sequence seen together with its surrogate value.
    """
    # initialise logits
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = exp_beta.sum()
        w = exp_beta / sum_exp

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_corr = m / (1.0 - beta1 ** it)
        v_corr = v_hat / (1.0 - beta2 ** it)

        v -= lr * m_corr / (np.sqrt(v_corr) + eps)

        # occasional progress check
        if it % 2000 == 0:
            cur_seq = _softmax(v)
            cur_val = _objective_surrogate(cur_seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[Adam] iter={it} new best surrogate={best_val:.6f}")

    # final evaluate
    final_seq = _softmax(v)
    final_val = _objective_surrogate(final_seq)
    if final_val < best_val:
        best_seq = final_seq
        best_val = final_val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient refinement (mirror‑descent on the exact max)
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient at the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / np.sqrt(it)          # decreasing stepsize
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] iter={it} best surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing on logits (accept‑only‑improving)
# ----------------------------------------------------------------------
def _stochastic_hc(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 200_000,
                   init_sigma: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Tiny hill‑climbing: add decaying Gaussian noise to the logits,
    accept only if the surrogate improves.
    """
    rng = np.random.default_rng()
    log_cur = np.log(start_seq + 1e-30)
    cur_seq = _softmax(log_cur)
    cur_val = _objective_surrogate(cur_seq)

    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        sigma = init_sigma / np.sqrt(it)   # annealing
        perturb = rng.normal(scale=sigma, size=n)
        new_log = log_cur + perturb
        new_seq = _softmax(new_log)
        new_val = _objective_surrogate(new_seq)
        if new_val < cur_val:
            log_cur = new_log
            cur_seq = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new_seq.copy()
        # occasional progress dump
        if it % 50000 == 0:
            print(f"[HC] iter={it} best surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # Initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Choose a working resolution (balance quality vs speed)
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # cheap surrogate ranking
    seed_vals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in seed_vals[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(seed_vals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS on the best seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] surrogate = {global_best_val:.6f}")

    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1000.0, 2000.0, 5000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before all L‑BFGS runs.")
            break
        # try to give each run a decent iteration budget
        remaining = deadline - time.time()
        max_iter_beta = max(5000, int(min(15000, remaining * 0.07 * 1e5)))
        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter/beta={max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=max_iter_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimiser on a very sharp surrogate (β = 5000)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam‑style refinement (β=5000) ---")
        remaining = deadline - time.time()
        max_adam_iters = int(min(300_000, remaining * 0.25 * 1e5))
        seq_adam, val_adam = _adam_optimize(global_best_seq,
                                            n,
                                            deadline,
                                            beta=5000.0,
                                            max_iters=max_adam_iters,
                                            lr=0.5)
        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient refinement on the exact max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponentiated‑gradient refinement (exact max) ---")
        remaining = deadline - time.time()
        max_exp_iters = int(min(500_000, remaining * 0.30 * 1e5))
        seq_exp, val_exp = _exp_gradient_optimize(global_best_seq,
                                                  n,
                                                  deadline,
                                                  max_iters=max_exp_iters)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Tiny stochastic hill‑climbing on logits
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Stochastic hill‑climbing on logits ---")
        remaining = deadline - time.time()
        max_hc_iters = int(min(200_000, remaining * 0.20 * 1e5))
        seq_hc, val_hc = _stochastic_hc(global_best_seq,
                                         n,
                                         deadline,
                                         max_iters=max_hc_iters,
                                         init_sigma=0.9)
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[HC] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible next run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
