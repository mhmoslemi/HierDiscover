"""
Improved step‑function search for minimise evaluate_sequence.

The algorithm proceeds in four stages:

1️⃣  Seed generation – a very rich family of analytic shapes is
    instantiated (uniform, triangular, sine‑window, Gaussian,
    power‑law, beta‑distribution, exponential, *symmetric* exponential,
    plus many random Dirichlet / exponential draws).  Parameter sweeps
    are dense (≈ 300 candidates) so that the best ℓ₁‑normalised seed is
    already very close to the optimum (surrogate ≈ 1.48 for n=8000).

2️⃣  Long Adam optimisation on the log‑softmax parametrisation.
    The optimiser uses a temperature‑controlled smooth‑max
    (log‑sum‑exp) with a slowly increasing inverse temperature, so that
    the true max‑convolution objective is approached faithfully.

3️⃣  Sub‑gradient and exponentiated‑gradient refinements – short
    passes that exploit the exact max‑convolution sub‑gradient while
    projecting onto the simplex.

4️⃣  Aggressive simulated‑annealing that not only reshuffles mass but
    also *swaps* large coefficients.  Swapping breaks up
    correlations between the biggest entries and quickly drives the
    max‑convolution down.

All stages respect the global 1000 s deadline.  The best sequence
found is stored in the global variable ``height_sequence_1`` for
possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised
    vector (∑seq = 1).  The true evaluate_sequence value equals this
    surrogate because the denominator is 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_shape(n: int, r: float) -> np.ndarray:
    """Monotone exponential a_i ∝ r**i, ℓ₁‑normalised."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_offset_shape(n: int, offset: int, r: float) -> np.ndarray:
    """Zero prefix of length *offset* followed by an exponential tail."""
    if offset >= n:
        return _uniform(n)
    tail_len = n - offset
    if r <= 0.0:
        tail = np.full(tail_len, 1.0 / tail_len, dtype=np.float64)
    else:
        i = np.arange(tail_len, dtype=np.float64)
        tail = np.power(r, i)
        tail /= tail.sum()
    a = np.concatenate([np.zeros(offset, dtype=np.float64), tail])
    a /= a.sum()
    return a


def _symmetric_exponential_shape(n: int, r: float) -> np.ndarray:
    """Symmetric exponential a_i ∝ r**|i-c| (c = centre)."""
    if r <= 0.0:
        return _uniform(n)
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Optimisation primitives
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _logspace_adam_optimize_long(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray | None, float]:
    """Adam on the log‑softmax parametrisation with a smoothed max."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5               # start temperature (smooth‑max strength)
    beta_inc = 0.5
    beta_max = 3000.0

    lr0 = 1.0                # initial learning‑rate in log‑space

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[AdamLong] it={it:7d} best={cur_val:.6f}")

        # smoothed max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1: n - 1 + n]

        # Jacobian of log‑softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / math.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Euclidean projection onto the simplex after each step.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _simulated_annealing_permutations(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
    init_temp_factor: float = 0.06,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Aggressive annealing that mixes reshuffles, swaps and pairwise transfers."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cool down
        T = max(T * 0.9999, min_temp)

        op = rng.random()
        if op < 0.4:
            # swap two positions – keeps the multiset unchanged
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = cur[j], cur[i]
        elif op < 0.8:
            # reshuffle a small random subset
            k = rng.integers(2, 12)          # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            # simple pairwise mass transfer (as in the original SA)
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)  # ≤5 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SAperm] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation (expanded)
# ----------------------------------------------------------------------
def _collect_rich_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a large and diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic classical families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in np.linspace(0.2, 5.0, 20):
        seeds.append(_sine_window(n, gamma=gamma))
    for sigma in (0.10, 0.15, 0.20, 0.30):
        seeds.append(_gaussian(n, sigma_frac=sigma))
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        seeds.append(_power(n, p=p))

    # beta‑distribution grid (log‑space)
    alphas = np.geomspace(0.05, 10.0, 12)
    betas = np.geomspace(0.05, 10.0, 12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # exponential families (pure, offset, symmetric)
    base_eps = 2.0 / n
    eps_factors = np.linspace(0.5, 3.0, 30)
    for fac in eps_factors:
        eps = fac * base_eps
        r = max(0.0, 1.0 - eps)
        seeds.append(_exponential_shape(n, r))

    # symmetric exponential sweep
    for fac in eps_factors:
        eps = fac * base_eps
        r = max(0.0, 1.0 - eps)
        seeds.append(_symmetric_exponential_shape(n, r))

    # offset exponentials – explore larger offsets than before
    for offset_frac in (0.0, 0.05, 0.10, 0.15, 0.20, 0.30):
        offset = int(offset_frac * n)
        tail_len = n - offset
        if tail_len <= 0:
            continue
        base_eps_off = 2.0 / tail_len
        for fac in np.linspace(0.5, 2.5, 12):
            eps = fac * base_eps_off
            r = max(0.0, 1.0 - eps)
            seeds.append(_exponential_offset_shape(n, offset, r))

    # random Dirichlet draws (various concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(4):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # random exponential draws
    for _ in range(6):
        a = rng.exponential(scale=1.0, size=n)
        a /= a.sum()
        seeds.append(a)

    # previous solution, if any
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final sanity – ensure ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allocated time budget.
    Returns the best sequence (list of floats).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a safety margin for the final exact evaluation
    overall_deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Phase 0 – generate a huge pool of analytic seeds
    # ------------------------------------------------------------------
    n = 8000
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_rich_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} candidate seeds.")

    # cheap surrogate ranking
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in evals[:top_k]]

    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # initialise global best with the absolute best seed
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long Adam optimisation (distributed over top seeds)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 60.0:
        # allocate about half the remaining time to Adam
        it_per_seed = int(min(400_000, remaining * 4000) // top_k) + 1
        print(f"\n--- Adam optimisation (≈{it_per_seed} it per seed) ---")
        for idx, seed_vec in enumerate(best_seeds):
            seq_adam, val_adam = _logspace_adam_optimize_long(
                seed_vec, n, overall_deadline, max_iters=it_per_seed
            )
            if seq_adam is not None and val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[Adam‑seed{idx+1}] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient projection
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 30.0:
        max_iters_sub = int(min(2_000_000, remaining * 2500))
        print(f"\n--- Sub‑gradient projection (max_iters={max_iters_sub}) ---")
        seq_sub, val_sub = _subgradient_proj(
            global_best_seq, n, overall_deadline, max_iters=max_iters_sub
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[SubProj] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 20.0:
        max_iters_exp = int(min(800_000, remaining * 1500))
        print(f"\n--- Exponentiated‑gradient refinement (max_iters={max_iters_exp}) ---")
        seq_exp, val_exp = _exp_gradient_optimize(
            global_best_seq, n, overall_deadline, max_iters=max_iters_exp
        )
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] new global best = {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – intensive simulated‑annealing with swaps
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 10.0:
        max_iters_sa = int(min(2_000_000, remaining * 2500))
        print(f"\n--- Simulated annealing with permutations (max_iters={max_iters_sa}) ---")
        seq_sa, val_sa = _simulated_annealing_permutations(
            global_best_seq, n, overall_deadline, max_iters=max_iters_sa
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA‑perm] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
