"""
Hybrid parametric + stochastic optimizer for the step‑function bound.

The objective (after ℓ₁‑normalisation) is to minimise
    2·n·max(conv(a,a)),
i.e. to make the self‑convolution as flat as possible.
The algorithm proceeds in four stages:

0.  Rich seed generation.  In addition to the classic families, we
    explore a two‑parameter “exponential‑power’’ family
        a_i ∝ (i+1)^{p}·r^{i}
    (optionally preceded by a zero prefix).  A dense grid and a few
    random draws are used; only the best `k` seeds are retained.

1.  Long Adam optimisation on the log‑softmax (smooth log‑sum‑exp
    approximation of the max).  The temperature β is driven up to
    5000, allowing a very sharp focus on the true max.  Only the top
    few seeds are refined, each for many thousands of iterations.

2.  Exact sub‑gradient descent with Euclidean projection onto the
    probability simplex.  This stage reduces the true max directly.

3.  Aggressive simulated‑annealing polish.  Random Dirichlet
    redistributions on subsets of size up to 30, plus pairwise mass
    transfers, are accepted with a cooling schedule that quickly
    drives the temperature to zero.

Throughout we keep the globally best sequence (its surrogate value
and the true `evaluate_sequence` value after the final polishing).

The function returns the best plain list of floats, and also stores it
in the global variable ``height_sequence_1`` for possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using an FFT (real, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate objective 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_shape(n: int, r: float) -> np.ndarray:
    """Monotone exponential a_i ∝ r**i, ℓ₁‑normalised."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_offset_shape(n: int, offset: int, r: float) -> np.ndarray:
    """Zero prefix of length *offset* followed by an exponential tail."""
    if offset >= n:
        return _uniform(n)
    tail_len = n - offset
    if r <= 0.0:
        tail = np.full(tail_len, 1.0 / tail_len, dtype=np.float64)
    else:
        i = np.arange(tail_len, dtype=np.float64)
        tail = np.power(r, i)
        tail /= tail.sum()
    a = np.concatenate([np.zeros(offset, dtype=np.float64), tail])
    a /= a.sum()
    return a


def _exp_power_shape(n: int, r: float, p: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^{p}·r^{i}, ℓ₁‑normalised.
    """
    i = np.arange(n, dtype=np.float64)
    a = np.power(i + 1.0, p) * np.power(r, i)
    s = a.sum()
    if s <= 0.0 or np.isnan(s):
        return _uniform(n)
    return a / s


def _exp_power_offset_shape(n: int, offset: int, r: float, p: float) -> np.ndarray:
    """Zero prefix followed by a (power·exponential) tail."""
    if offset >= n:
        return _uniform(n)
    tail_len = n - offset
    i = np.arange(tail_len, dtype=np.float64)
    tail = np.power(i + 1.0, p) * np.power(r, i)
    s = tail.sum()
    if s <= 0.0:
        tail = np.full(tail_len, 1.0 / tail_len, dtype=np.float64)
    else:
        tail = tail / s
    a = np.concatenate([np.zeros(offset, dtype=np.float64), tail])
    a /= a.sum()
    return a


def _mix_exponential(n: int, r1: float, r2: float, w: float) -> np.ndarray:
    """Convex mixture w·r1ⁱ + (1‑w)·r2ⁱ, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = w * np.power(r1, i) + (1.0 - w) * np.power(r2, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _logspace_adam_optimize_long(start_vec: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 500_000) -> tuple[np.ndarray | None, float]:
    """Adam on log‑softmax with a log‑sum‑exp smoothed max."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5
    beta_inc = 0.5
    beta_max = 5000.0               # sharper than the original

    lr0 = 1.0                       # start learning rate in log‑space

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 5000 == 0:
                print(f"[AdamLong] it={it:7d} best={cur_val:.6f}")

        # smoothed max via log‑sum‑exp (temperature = beta)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1: n - 1 + n]

        # Jacobian of log‑softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / math.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # active sub‑gradient (only overlapping part)
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 2_000_000,
                         init_temp_factor: float = 0.06,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Aggressive annealing that reshuffles mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.99995, min_temp)

        # two types of move
        if rng.random() < 0.65:       # Dirichlet reshuffle on a random subset
            k = rng.integers(2, 30)   # subset size 2 … 29
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                          # simple pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)   # ≤7 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _generate_seeds(n: int,
                    rng: np.random.Generator,
                    previous: np.ndarray | None,
                    top_k: int = 30) -> list[np.ndarray]:
    """Return the `top_k` seeds (ℓ₁‑normalised) having the lowest surrogate."""
    # Helper for maintaining a sorted top‑k list
    def _push_candidate(val, vec):
        if len(top) < top_k:
            top.append((val, vec.copy()))
            top.sort(key=lambda x: x[0])
        elif val < top[-1][0]:
            top[-1] = (val, vec.copy())
            top.sort(key=lambda x: x[0])

    top: list[tuple[float, np.ndarray]] = []

    # ---------- deterministic families ----------
    families = [
        _uniform,
        _triangular,
        lambda n: _sine_window(n, gamma=0.3),
        lambda n: _sine_window(n, gamma=0.5),
        lambda n: _sine_window(n, gamma=0.8),
        lambda n: _sine_window(n, gamma=1.0),
        lambda n: _sine_window(n, gamma=2.0),
        lambda n: _gaussian(n, sigma_frac=0.15),
        lambda n: _gaussian(n, sigma_frac=0.25),
        lambda n: _power(n, p=1.0),
        lambda n: _power(n, p=2.0),
        lambda n: _beta_shape(n, alpha=2.0, beta=5.0),
        lambda n: _beta_shape(n, alpha=5.0, beta=2.0),
    ]

    for fn in families:
        vec = fn(n)
        val = _objective_surrogate(vec)
        _push_candidate(val, vec)

    # ---------- exponential families ----------
    base_eps = 2.0 / n
    eps_factors = np.linspace(0.2, 5.0, 30)   # sweep around the theoretical optimum
    for fac in eps_factors:
        eps = fac * base_eps
        r = max(0.0, 1.0 - eps)
        vec = _exponential_shape(n, r)
        val = _objective_surrogate(vec)
        _push_candidate(val, vec)

    # offsets + exponential
    offsets = [0, int(0.02 * n), int(0.05 * n), int(0.10 * n), int(0.15 * n), int(0.20 * n)]
    for off in offsets:
        L = n - off
        if L <= 0:
            continue
        base_eps_off = 2.0 / L
        for fac in np.linspace(0.2, 4.0, 20):
            eps = fac * base_eps_off
            r = max(0.0, 1.0 - eps)
            vec = _exponential_offset_shape(n, off, r)
            val = _objective_surrogate(vec)
            _push_candidate(val, vec)

    # ---------- power‑exponential families ----------
    # systematic grid over (offset, r, p)
    p_grid = np.linspace(-3.0, 3.0, 31)   # 31 values
    for off in offsets:
        L = n - off
        if L <= 0:
            continue
        base_eps_off = 2.0 / L
        for fac in np.linspace(0.2, 4.0, 18):
            eps = fac * base_eps_off
            r = max(0.0, 1.0 - eps)
            for p in p_grid:
                vec = _exp_power_offset_shape(n, off, r, p)
                val = _objective_surrogate(vec)
                _push_candidate(val, vec)

    # ---------- random mixtures of exponentials ----------
    for _ in range(200):
        r1 = np.exp(-rng.uniform(0.00005, 0.008))
        r2 = np.exp(-rng.uniform(0.00005, 0.008))
        w = rng.random()
        vec = _mix_exponential(n, r1, r2, w)
        val = _objective_surrogate(vec)
        _push_candidate(val, vec)

    # ---------- Dirichlet random draws ----------
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            vec = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            val = _objective_surrogate(vec)
            _push_candidate(val, vec)

    # ---------- previous solution ----------
    if previous is not None and previous.size == n and previous.sum() > 0:
        vec = previous / previous.sum()
        val = _objective_surrogate(vec)
        _push_candidate(val, vec)

    # Return only the vectors (sorted by surrogate value)
    print(f"\n--- Seed generation complete: kept top {top_k} candidates ---")
    for i, (_, v) in enumerate(top):
        print(f"  seed #{i+1:2d} surrogate = {_objective_surrogate(v):.6f}")
    return [vec for _, vec in top]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    overall_deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – rich seed generation (n = 8000)
    # ------------------------------------------------------------------
    n = 8000
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    print(f"\n=== Phase 0: seed generation (n={n}) ===")
    top_seeds = _generate_seeds(n, rng, prev_arr, top_k=30)

    # initialise global best
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Seed #1 surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam refinement (focus on the very best seeds)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 30.0:
        # allocate the majority of the time to Adam on the top 5 seeds
        max_total_iters = int(min(1_300_000, remaining * 3000))
        top_for_adam = top_seeds[:5]
        it_per_seed = max_total_iters // len(top_for_adam)
        print(f"\n=== Phase 1: Adam optimisation (total it≈{max_total_iters}) ===")
        for idx, seed_vec in enumerate(top_for_adam, 1):
            seq_adam, val_adam = _logspace_adam_optimize_long(
                seed_vec, n, overall_deadline, max_iters=it_per_seed
            )
            if seq_adam is not None and val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[Adam‑seed{idx}] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient projection
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 20.0:
        max_iters_sub = int(min(1_500_000, remaining * 2000))
        print(f"\n=== Phase 2: Sub‑gradient projection (max_iters={max_iters_sub}) ===")
        seq_sub, val_sub = _subgradient_proj(
            global_best_seq, n, overall_deadline, max_iters=max_iters_sub
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[SubProj] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing polishing (longer run)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 10.0:
        max_iters_sa = int(min(3_000_000, remaining * 2500))
        print(f"\n=== Phase 3: Simulated annealing (max_iters={max_iters_sa}) ===")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, overall_deadline, max_iters=max_iters_sa
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
