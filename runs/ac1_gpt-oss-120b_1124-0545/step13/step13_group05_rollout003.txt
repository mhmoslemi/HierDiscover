"""
Optimise a non‑negative step‑function (vector a, sum(a)=1) for the
objective

    OBJ(a) = 2·n·max(conv(a,a))

where conv is the full linear convolution.  The algorithm works in
four stages within the 1000 s budget:

1.  Build a diverse pool of analytic “seed’’ vectors (uniform,
    exponential families, power‑law, beta, sine‑window, Dirichlet,
    previous best, …) for a *small* problem size (n_small≈2000).  This
    makes each optimisation iteration cheap.

2.  Run a long “soft‑max’’ Adam optimiser on the log‑softmax
    parametrisation of the seeds.  The max in the objective is
    smoothed by a temperature β that is slowly increased to a large
    value (≈10 000) so the optimiser eventually focuses on the true
    ℓ∞‑norm.

3.  Perform a few hundred thousand steps of exact sub‑gradient descent
    (projected onto the simplex) to tighten the solution.

4.  Upscale the best small‑n vector to the target length (n_big = 8000)
    by repeating each entry a fixed number of times, renormalise and
    polish the full‑size vector with a short simulated‑annealing run.

All stages respect the global deadline; the best vector found is
returned as a plain list of floats and stored in the global variable
``height_sequence_1`` for possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) for a ℓ₁‑normalised vector."""
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_decay(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_shape(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i, 0<r<1."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_offset_shape(n: int, offset: int, r: float) -> np.ndarray:
    """Zero prefix of length offset, then exponential tail."""
    if offset >= n:
        return _uniform(n)
    tail_len = n - offset
    if r <= 0.0:
        tail = np.full(tail_len, 1.0 / tail_len, dtype=np.float64)
    else:
        i = np.arange(tail_len, dtype=np.float64)
        tail = np.power(r, i)
        tail = tail / tail.sum()
    a = np.concatenate([np.zeros(offset, dtype=np.float64), tail])
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers (log‑softmax Adam with smoothed max, sub‑gradient, SA)
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 300_000,
                   beta_max: float = 10_000.0) -> tuple[np.ndarray, float]:
    """Adam on log‑softmax parametrisation with Log‑Sum‑Exp smoothing."""
    eps = 1e-30
    v = np.log(start_vec + eps)           # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8
    beta = 0.5               # start temperature
    beta_inc = (beta_max - beta) / max_iters

    best_seq = None
    best_val = np.inf

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ---- smoothed max (log‑sum‑exp) ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)   # length = 2n-1
        grad_a = 2.0 * full[n - 1: n - 1 + n]

        # Jacobian of softmax (log‑softmax back‑prop)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = 1.0 / math.sqrt(it)          # decaying learning‑rate
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule
        beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_optimize(start_vec: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution objective."""
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the max‑entry (only the overlapping part)
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing_optimize(start_vec: np.ndarray,
                                  n: int,
                                  deadline: float,
                                  max_iters: int = 200_000,
                                  init_temp_factor: float = 0.06,
                                  min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Aggressive SA that reshuffles mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_vec.copy()
    cur_val = _objective_surrogate(cur, n)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cool down
        T = max(T * 0.9999, min_temp)

        if rng.random() < 0.7:                       # reshuffle a random subset
            k = rng.integers(2, 12)                 # subset size 2..11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                         # simple pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new, n)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()

    return best_seq, best_val

# Seed generation (small problem)

def _collect_seeds_small(n: int,
                         rng: np.random.Generator,
                         previous: np.ndarray | None) -> list[np.ndarray]:
    """Build a rich set of ℓ₁‑normalised vectors for the *small* problem."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian(n, sigma_frac=0.22))
    seeds.append(_power_decay(n, p=0.8))
    seeds.append(_power_decay(n, p=1.2))
    seeds.append(_power_decay(n, p=2.0))

    # beta‑family sweep
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, alpha=a, beta=b))

    # exponential families (including offsets)
    base_eps = 2.0 / n
    for fac in np.linspace(0.4, 3.0, 20):
        eps = fac * base_eps
        r = max(0.0, 1.0 - eps)
        seeds.append(_exponential_shape(n, r))

    offsets = [0, int(0.03 * n), int(0.07 * n), int(0.12 * n)]
    for off in offsets:
        L = n - off
        if L <= 0:
            continue
        base_eps_off = 2.0 / L
        for fac in np.linspace(0.4, 2.5, 10):
            eps = fac * base_eps_off
            r = max(0.0, 1.0 - eps)
            seeds.append(_exponential_offset_shape(n, off, r))

    # random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.6, 1.0, 2.0):
        for _ in range(4):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # exponential random draws
    for _ in range(5):
        a = rng.exponential(scale=1.0, size=n)
        a /= a.sum()
        seeds.append(a)

    # previous solution, if any
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up – guarantee ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


def _upscale_vector(v_small: np.ndarray, n_big: int) -> np.ndarray:
    """Repeat each entry of v_small enough times to reach length n_big."""
    n_small = v_small.shape[0]
    factor = n_big // n_small
    remainder = n_big - factor * n_small
    # repeat entries
    v_big = np.repeat(v_small, factor)
    if remainder > 0:
        # distribute the leftover by appending the first 'remainder' entries
        v_big = np.concatenate([v_big, v_small[:remainder]])
    # normalise (sum must stay 1)
    v_big /= v_big.sum()
    return v_big

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted time budget.
    Returns the best sequence (list of floats).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a small safety margin for the final exact evaluation
    overall_deadline = start_time + budget_s - 1.5

    # ------------------------------------------------------------------
    # Phase 0 – cheap seed generation (small problem)
    # ------------------------------------------------------------------
    n_small = 2000                      # cheap dimension for heavy optimisation
    n_big = 8000                        # final dimension required by the competition

    previous = globals().get("height_sequence_1")
    prev_small = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n_small:
                prev_small = arr.copy()
            else:
                # if the stored vector has a different length, down‑sample it
                step = max(1, arr.size // n_small)
                prev_small = arr[::step][:n_small]
                prev_small /= prev_small.sum()
        except Exception:
            prev_small = None

    print(f"\n--- Seed generation (n_small={n_small}) ---")
    seeds_small = _collect_seeds_small(n_small, rng, prev_small)
    print(f"Generated {len(seeds_small)} seeds.")

    # cheap surrogate ranking (objective for the small problem)
    evals = [(vec, _objective_surrogate(vec, n_small)) for vec in seeds_small]
    evals.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds_small = [vec for vec, _ in evals[:top_k]]

    print(f"Top {top_k} surrogate values (small n):")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f" #{i+1:2d}: {val:.6f}")

    # initialise global best (still on small problem)
    global_best_small = best_seeds_small[0].copy()
    global_best_val = _objective_surrogate(global_best_small, n_small)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long Adam optimisation on the small problem
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 60.0:
        # allocate roughly half of the remaining budget to Adam
        max_total_iters = int(min(1_200_000, remaining * 10_000))
        it_per_seed = max_total_iters // top_k
        print(f"\n--- Adam optimisation (small n, total it≈{max_total_iters}) ---")
        for idx, seed_vec in enumerate(best_seeds_small):
            seq_adam, val_adam = _adam_optimize(
                seed_vec,
                n_small,
                overall_deadline,
                max_iters=it_per_seed,
                beta_max=10_000.0
            )
            if seq_adam is not None and val_adam < global_best_val:
                global_best_small = seq_adam.copy()
                global_best_val = val_adam
                print(f"[Adam‑seed{idx+1}] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent (small n)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 25.0:
        max_iters_sub = int(min(500_000, remaining * 5_000))
        print(f"\n--- Sub‑gradient (small n) ---")
        seq_sub, val_sub = _subgradient_optimize(
            global_best_small,
            n_small,
            overall_deadline,
            max_iters=max_iters_sub
        )
        if val_sub < global_best_val:
            global_best_small = seq_sub.copy()
            global_best_val = val_sub
            print(f"[Sub‑grad] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – upscale to the target size and polish
    # ------------------------------------------------------------------
    print("\n--- Upscaling to full size ---")
    global_best_big = _upscale_vector(global_best_small, n_big)
    global_best_big_val = _objective_surrogate(global_best_big, n_big)
    print(f"[Upscale] surrogate on big n = {global_best_big_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final polishing (big n)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 10.0:
        # a short run of sub‑gradient followed by SA
        max_iters_sub_big = int(min(300_000, remaining * 2000))
        print(f"\n--- Sub‑gradient (big n) ---")
        seq_sub_big, val_sub_big = _subgradient_optimize(
            global_best_big,
            n_big,
            overall_deadline,
            max_iters=max_iters_sub_big
        )
        if val_sub_big < global_best_big_val:
            global_best_big = seq_sub_big.copy()
            global_best_big_val = val_sub_big
            print(f"[Sub‑grad‑big] new best = {val_sub_big:.6f}")

    remaining = overall_deadline - time.time()
    if remaining > 5.0:
        max_iters_sa_big = int(min(500_000, remaining * 2500))
        print(f"\n--- Simulated annealing (big n) ---")
        seq_sa_big, val_sa_big = _simulated_annealing_optimize(
            global_best_big,
            n_big,
            overall_deadline,
            max_iters=max_iters_sa_big
        )
        if val_sa_big < global_best_big_val:
            global_best_big = seq_sa_big.copy()
            global_best_big_val = val_sa_big
            print(f"[SA‑big] new best = {val_sa_big:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_big.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_big_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:            # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse by a later run
    globals()["height_sequence_1"] = global_best_big.tolist()
    return global_best_big.tolist()
