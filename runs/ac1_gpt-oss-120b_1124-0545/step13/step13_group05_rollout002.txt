"""
Hybrid search for the step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm proceeds as follows:

1.  A rich pool of analytic “seed’’ vectors (uniform, exponential,
    beta, power, sine‑window, Gaussian, etc.) is generated for a
    chosen length *n*.  All seeds are ℓ₁‑normalised (∑aᵢ=1).

2.  The cheap surrogate 2·n·max(conv(a,a)) is evaluated for every seed.
    The top‑K seeds (K≈20) are kept.

3.  Each kept seed is refined with a sequence of L‑BFGS‑B optimisations.
    The true max‑convolution is approximated by a log‑sum‑exp
    smoothing with a temperature β that is increased gradually
    (β∈{5,20,100,500,2000}).  The optimisation works on the
    log‑softmax parametrisation (v with a=softmax(v)) so that the
    non‑negativity and ℓ₁‑normalisation constraints are satisfied
    automatically.

4.  The best vector found by L‑BFGS is optionally polished with a
    long Adam optimiser on the same smoothed objective (the Adam code
    is taken from the previous baseline).

5.  The final vector is returned as a plain list of floats and stored
    in the global variable ``height_sequence_1`` for possible reuse.

All heavy work respects the global deadline (default 1000 s).  The
implementation only uses NumPy, SciPy and the provided
``evaluate_sequence`` function.
"""

import time
import math
import numpy as np
import functools

# ----------------------------------------------------------------------
# Low‑level numeric utilities (FFT‑based convolution, simplex ops)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised
    vector (∑seq = 1).  The true ``evaluate_sequence`` value equals this
    surrogate because the denominator is 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_shape(n: int, r: float) -> np.ndarray:
    """Monotone exponential a_i ∝ r**i, ℓ₁‑normalised."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_offset_shape(n: int, offset: int, r: float) -> np.ndarray:
    """Zero prefix of length *offset* followed by an exponential tail."""
    if offset >= n:
        return _uniform(n)
    tail_len = n - offset
    if r <= 0.0:
        tail = np.full(tail_len, 1.0 / tail_len, dtype=np.float64)
    else:
        i = np.arange(tail_len, dtype=np.float64)
        tail = np.power(r, i)
        tail /= tail.sum()
    a = np.concatenate([np.zeros(offset, dtype=np.float64), tail])
    a /= a.sum()
    return a


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Optimisation primitives (Adam on the smoothed max‑convolution)
# ----------------------------------------------------------------------
def _logspace_adam_optimize_long(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray | None, float]:
    """Adam on the log‑softmax parametrisation with a log‑sum‑exp
    smoothing of the max (β increases up to 2000).  Returns the best
    vector seen and its surrogate value."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5               # start temperature
    beta_inc = 0.5
    beta_max = 2000.0

    lr0 = 1.0                # initial learning‑rate in log‑space

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report (can be silenced)
            if it % 5000 == 0:
                print(f"[AdamLong] it={it:7d} best={cur_val:.6f}")

        # smoothed max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2 (unused tail trimmed)
        grad_a = 2.0 * full[n - 1: n - 1 + n]

        # Jacobian of log‑softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / math.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


# ----------------------------------------------------------------------
# L‑BFGS helper (smooth max with adjustable β)
# ----------------------------------------------------------------------
def _obj_grad_for_lbfgs(v: np.ndarray, n: int, beta: float):
    """
    Smoothed objective and its gradient w.r.t. v (logits) for L‑BFGS.
    Uses the log‑sum‑exp approximation of max(conv(a,a)).
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)

    shift = np.max(conv)
    # subtract shift for numerical stability, clip to avoid overflow
    exp_arg = beta * (conv - shift)
    exp_arg = np.clip(exp_arg, -700.0, 0.0)   # conv‑shift ≤ 0, keep within exp range
    w = np.exp(exp_arg)
    sum_w = np.sum(w)
    w = w / sum_w

    # gradient of Σ w_k·conv_k w.r.t. a
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # gradient w.r.t. logits v (chain rule through softmax)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    # smooth approximation of max(conv)
    smooth_max = shift + (1.0 / beta) * np.log(sum_w)
    obj = 2.0 * n * smooth_max
    return obj, grad_v


def _lbfgs_refine(v_init: np.ndarray, n: int, deadline: float):
    """
    Run a short schedule of L‑BFGS optimisations on the smoothed
    max‑convolution, increasing the temperature β gradually.
    Returns the refined logits and the corresponding surrogate value.
    """
    import scipy.optimize as opt

    v = v_init.copy()
    # temperature schedule (coarse → fine)
    beta_schedule = [5.0, 20.0, 100.0, 500.0, 2000.0]

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # partial fixes n and beta, leaving only the variable v
        fun = functools.partial(_obj_grad_for_lbfgs, n=n, beta=beta)
        try:
            res = opt.minimize(fun,
                               v,
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': 200,
                                        'disp': False})
            v = res.x
        except Exception as exc:  # pragma: no cover
            print("[L‑BFGS] optimisation failed:", exc)
            break

    # final surrogate value
    a = _softmax(v)
    final_val = _objective_surrogate(a)
    return v, final_val

# Seed generation (diverse analytic families)

def _collect_rich_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic classic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_power(n, p=1.0))
    seeds.append(_power(n, p=2.0))

    # systematic exponential sweep
    base_eps = 2.0 / n
    for fac in np.linspace(0.5, 3.0, 25):
        eps = fac * base_eps
        r = max(0.0, 1.0 - eps)
        seeds.append(_exponential_shape(n, r))

    # offset exponential families (more offsets than before)
    offsets = [int(frac * n) for frac in (0.0, 0.02, 0.05, 0.10, 0.15, 0.25, 0.35)]
    for off in offsets:
        L = n - off
        if L <= 0:
            continue
        base_eps_off = 2.0 / L
        for fac in np.linspace(0.5, 3.0, 12):
            eps = fac * base_eps_off
            r = max(0.0, 1.0 - eps)
            seeds.append(_exponential_offset_shape(n, off, r))

    # beta family grid
    for alpha in np.linspace(0.3, 5.0, 7):
        for beta in np.linspace(0.3, 5.0, 7):
            seeds.append(_beta_shape(n, alpha, beta))

    # random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # random exponential draws
    for _ in range(3):
        a = rng.exponential(scale=1.0, size=n)
        a /= a.sum()
        seeds.append(a)

    # previous solution if available
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final clean‑up – guarantee ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        cleaned.append(v / s if s > 0 else _uniform(n))
    return cleaned

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin for the final exact evaluation
    overall_deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – a medium size gives many cheap iterations
    # ------------------------------------------------------------------
    n = 4000   # feel free to adjust (larger n gives more expressive power,
               # smaller n yields faster iterations)

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_rich_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} candidate seeds.")

    # cheap surrogate ranking
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = min(20, len(evals))
    best_seeds = [vec for vec, _ in evals[:top_k]]

    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS refinement of the top seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > overall_deadline:
            break
        print(f"\n[Phase L‑BFGS] Refining seed #{idx+1}")
        v0 = np.log(seed_vec + 1e-30)
        v_opt, val_opt = _lbfgs_refine(v0, n, overall_deadline)
        a_opt = _softmax(v_opt)
        print(f"   surrogate after L‑BFGS = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_seq = a_opt.copy()
            print(f"   >>> New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – optional long Adam polish on the current best vector
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 30.0:
        print("\n--- Adam polishing (remaining time {:.1f}s) ---".format(remaining))
        # Allocate a sizeable number of iterations but respect the deadline
        max_iters = int(min(600_000, remaining * 5000))
        seq_adam, val_adam = _logspace_adam_optimize_long(
            global_best_seq, n, overall_deadline, max_iters=max_iters
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam] new global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
