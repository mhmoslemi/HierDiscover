"""
Differential‑Evolution + local Adam polishing.

The search works on a probability vector p (∑p=1, p_i≥0).  The
objective that the official evaluator uses (with the sum‑normalisation
cancelling out) is

    L(p) = 2·n·max(conv(p,p))

where conv(p,p) is the full linear convolution.  We minimise L with a
population‑based differential‑evolution (DE) that works directly on the
simplex: mutation → crossover → simplex projection → selection.  The best
individual is periodically refined with a smooth‑max Adam optimiser (the
same technique that was used in the previous state‑of‑the‑art solution)
and, if time permits, a short exponential‑gradient polishing step.

The algorithm is deliberately lightweight so it fits comfortably into the
1000 s budget while still exploring a large variety of shapes (analytic
windows, random Dirichlet draws, random binary masks, low‑frequency
curves, …).  The final vector is stored in the global name
``height_sequence_1`` for possible reuse in a subsequent run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Surrogate objective (exact, but called “surrogate” for historic reasons)
# ----------------------------------------------------------------------
def _objective(p: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv(p,p)) for a probability vector."""
    return float(2.0 * n * np.max(_conv_fft(p, p)))


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max (log‑sum‑exp) – same code as before
# ----------------------------------------------------------------------
def _adam_softmax_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    alpha_start: float = 20.0,
    alpha_end: float = 2000.0,
    lr: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        L(p) = (1/α)·log Σₖ exp(α·(p*p)[k]).

    Returns the best vector encountered and its exact objective value.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective(x, n)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    N = _next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective(x, n)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# Exponential‑gradient (mirror‑descent) polishing – same as before
# ----------------------------------------------------------------------
def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Mirror‐descent with KL divergence (multiplicative update)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective(x, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        s = x.sum()
        if s <= 0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective(x, n)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()

    return best_x, best_val


# ----------------------------------------------------------------------
# Differential Evolution (population based search)
# ----------------------------------------------------------------------
def _differential_evolution(
    pop: list[np.ndarray],
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_generations: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Simple DE on the simplex.  Returns the best individual found.
    """
    popsize = len(pop)
    obj_vals = [_objective(v, n) for v in pop]
    best_idx = int(np.argmin(obj_vals))
    best = pop[best_idx].copy()
    best_val = obj_vals[best_idx]

    generation = 0
    F0, CR0 = 0.7, 0.9

    while time.time() < deadline and generation < max_generations:
        generation += 1
        # gradually shrink the mutation factor so that we converge
        time_frac = (deadline - time.time()) / (deadline - (deadline - 1000))
        F = F0 * (0.5 + 0.5 * max(0.0, time_frac))
        CR = CR0

        new_pop = []
        new_obj = []
        for i in range(popsize):
            # choose three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a, b, c = pop[a_idx], pop[b_idx], pop[c_idx]

            mutant = a + F * (b - c)
            mutant = np.maximum(mutant, 0.0)                     # stay non‑negative

            # crossover (vectorised)
            mask = rng.random(n) < CR
            jrand = rng.integers(0, n)
            mask[jrand] = True
            trial = np.where(mask, mutant, pop[i])
            trial = _proj_simplex(trial)

            trial_val = _objective(trial, n)
            if trial_val <= obj_vals[i]:
                new_pop.append(trial)
                new_obj.append(trial_val)
                if trial_val < best_val:
                    best_val, best = trial_val, trial.copy()
                    print(f"[DE] gen={generation:4d} new best={best_val:.6f}")
            else:
                new_pop.append(pop[i])
                new_obj.append(obj_vals[i])

        pop, obj_vals = new_pop, new_obj

        # occasional local polishing with Adam (if we still have time)
        if generation % 20 == 0:
            remaining = deadline - time.time()
            if remaining > 4.0:
                cand, val = _adam_softmax_optimize(
                    best, n, deadline, max_iters=int(min(30_000, remaining * 1_500)),
                    alpha_start=20.0, alpha_end=2000.0, lr=0.12
                )
                if val < best_val:
                    best_val, best = val, cand.copy()
                    # replace the worst individual with the improved one
                    worst = int(np.argmax(obj_vals))
                    pop[worst] = best
                    obj_vals[worst] = best_val
                    print(f"[Adam polish] gen={generation:4d} improved to {best_val:.6f}")

        # early exit if we are already extremely close to the theoretical lower bound
        if best_val < 1.02:
            break

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    the official ``evaluate_sequence`` objective within the allotted time.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Choose a moderate length – large enough to allow a low bound,
    # small enough to keep the optimisation cheap.
    # --------------------------------------------------------------
    n = 2000                # can be changed; 2000 gives a good trade‑off

    # --------------------------------------------------------------
    # 0️⃣ Seed generation – diverse deterministic families + random draws
    # --------------------------------------------------------------
    seeds = [
        _uniform(n),
        _triangular(n),
        _sine_window(n, 1.0),
        _gaussian(n, 0.3),
    ]

    # previous solution (if any)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # Random Dirichlet draws with several concentration parameters
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(rng.dirichlet(np.full(n, conc)))

    # Random binary masks (sparse or moderately dense)
    for _ in range(5):
        k = rng.integers(1, n // 2)                     # number of ones
        mask = np.zeros(n, dtype=np.float64)
        mask[rng.choice(n, size=k, replace=False)] = 1.0
        mask = mask / mask.sum()
        seeds.append(mask)

    # Low‑frequency random curves
    for _ in range(5):
        seeds.append(_lowfreq_curve(n, rng.integers(1, 5), rng))

    # Ensure everything lives on the simplex
    population = [_proj_simplex(s) for s in seeds[:]]
    popsize = max(30, len(population))                # at least 30 individuals
    # If we have fewer than popsize, fill up with fresh Dirichlet draws
    while len(population) < popsize:
        population.append(rng.dirichlet(np.full(n, 1.0)))
    population = population[:popsize]

    # --------------------------------------------------------------
    # 1️⃣ Differential‑Evolution global search
    # --------------------------------------------------------------
    print(f"\n--- DE global search (n={n}, popsize={popsize}) ---")
    best_vec, best_val = _differential_evolution(population, n, deadline, rng)

    print(f"[DE] finished with surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Final polishing – a short Adam run + optional exp‑grad
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        print("\n--- Final Adam polishing ---")
        max_iters = int(min(60_000, remaining * 1_500))
        best_vec, best_val = _adam_softmax_optimize(
            best_vec, n, deadline, max_iters=max_iters,
            alpha_start=20.0, alpha_end=2000.0, lr=0.12
        )
        print(f"[Adam] post‑polish surrogate = {best_val:.6f}")

    remaining = deadline - time.time()
    if remaining > 1.0:
        print("\n--- Optional exponential‑gradient polishing ---")
        max_iters = int(min(300_000, remaining * 1500))
        best_vec, best_val = _expgrad_optimize(
            best_vec, n, deadline, max_iters=max_iters, eta0=0.25
        )
        print(f"[ExpGrad] final surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Official evaluation
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_vec.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_vec.tolist()
    return best_vec.tolist()
```

This implementation replaces the previous two‑stage optimisation with a global differential‑evolution search that explores many qualitatively different shapes, followed by a focused Adam smoothing and a short exponential‑gradient refinement.  It stays well within the 1000 s budget, continually reports progress, and stores the best sequence for reuse.
