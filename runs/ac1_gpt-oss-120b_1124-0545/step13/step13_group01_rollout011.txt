def _grad_and_obj(v, beta, n):
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    # compute exp_beta
    # careful: conv-shift <=0
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_beta)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    start = n - 1
    grad_a = 2.0 * full[start:start + n]
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
    return a, grad_v, obj
```

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    2·n·max(conv(a,a)) / (sum(a)²)

The strategy combines three ideas:

1. **Smooth‑max Adam optimisation** – optimise a differentiable
   surrogate of the max‑convolution using Adam on the log‑softmax
   (logits) representation.  A β‑schedule gradually sharpens the
   approximation to the true max.

2. **Exact sub‑gradient (exponentiated‑gradient) polishing** – after the
   smooth stage we perform many mirror‑descent steps w.r.t. the true
   max‑convolution to fine‑tune the solution.

3. **Local mass‑transfer search and optional up‑sampling** – simple hill‑
   climbing moves that swap a small amount of probability mass between
   two indices, followed by an optional refinement on a doubled grid
   if time permits.

All steps respect the ℓ₁‑normalisation (sum = 1) via the softmax
parameterisation, guaranteeing that the evaluation function is
scale‑invariant.  The routine is time‑bounded (≈1000 s) and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector.
    If sum(seq)=1, this equals the true objective.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (used as fall‑backs)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _grad_and_obj(v: np.ndarray, beta: float, n: int):
    """
    Compute (a, grad_v, surrogate) for the smooth‑max objective.
    a = softmax(v) is ℓ₁‑normalised.
    The gradient is w.r.t. the logits v.
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)                 # full autocorrelation
    shift = np.max(conv)                   # max value = max(conv)
    # smooth‑max via log‑sum‑exp
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_beta)
    w = exp_beta / sum_exp                 # weights for the smooth max

    # gradient w.r.t. a : 2 * conv(rev_a, w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)            # length 2n‑1
    start = n - 1
    grad_a = 2.0 * full[start:start + n]

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    # surrogate objective value
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
    return a, grad_v, obj


def _adam_optimize(start_a: np.ndarray, n: int, deadline: float):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best vector (ℓ₁‑normalised) and its surrogate value.
    """
    v = np.log(start_a + 1e-30)               # logits
    best_a = start_a.copy()
    best_sur = _objective_surrogate(best_a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    bet1_pow = 1.0
    bet2_pow = 1.0
    t = 0

    # β‑schedule (smoothness gradually increased)
    beta_schedule = [5, 20, 80, 200, 500, 1000]

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # rough estimate: one iteration ≈ 0.00015 s at n≈4000
        max_iter = int(min(200_000, max(0.0, remaining / 0.00015)))
        if max_iter <= 0:
            break
        print(f"[Adam] beta={beta:.0f}, allotted iterations={max_iter}")

        # learning‑rate scaled with β (larger β → sharper landscape)
        lr = 0.05 / np.sqrt(beta)

        for _ in range(max_iter):
            t += 1
            if time.time() > deadline:
                break

            a, grad_v, obj = _grad_and_obj(v, beta, n)

            if obj < best_sur:
                best_sur = obj
                best_a = a.copy()
                # occasional true evaluation for monitoring
                if t % 50000 == 0:
                    try:
                        true_val = evaluate_sequence(best_a.tolist())
                        print(f"[Adam] step={t}, β={beta}, surrogate={obj:.6f}, true={true_val:.6f}")
                    except Exception:
                        pass

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_v
            v2 = beta2 * v2 + (1 - beta2) * (grad_v * grad_v)
            bet1_pow *= beta1
            bet2_pow *= beta2
            m_hat = m / (1 - bet1_pow)
            v_hat = v2 / (1 - bet2_pow)
            v = v - lr * m_hat / (np.sqrt(v_hat) + eps)

        print(f"[Adam] after β={beta}, best surrogate={best_sur:.6f}")

    return best_a, best_sur


def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 500_000):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active lag
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size schedule (larger constant than before)
        eta = 0.7 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _local_search(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  max_attempts: int):
    """
    Simple hill‑climbing: repeatedly move a random fraction of probability mass
    from one index to another and keep the move if the surrogate improves.
    """
    a = start_seq.copy()
    best_sur = _objective_surrogate(a)

    for attempt in range(max_attempts):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer up to 5 % of the source mass
        delta = a[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0.0:
            continue
        new_sur = _objective_surrogate(a_new)
        if new_sur < best_sur:
            a = a_new
            best_sur = new_sur
    return a, best_sur


def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    s = new_seq.sum()
    if s == 0.0:
        return _uniform(new_n)
    return new_seq / s


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (plain list of floats).
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a small safety margin

    # ------------------------------------------------------------------
    # Initialise with the previous best or a simple analytic shape
    # ------------------------------------------------------------------
    n = 4000
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        a0 = np.asarray(prev, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else _uniform(n)
    else:
        # fall‑back to a deterministic seed
        a0 = _uniform(n)

    print(f"Initial surrogate = {_objective_surrogate(a0):.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation of a smooth surrogate
    # ------------------------------------------------------------------
    a_opt, sur_opt = _adam_optimize(a0, n, deadline)
    print(f"[Stage 1] after Adam – surrogate = {sur_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Exact exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(800_000, remaining * 200))   # generous budget
        a_opt, sur_opt = _exp_gradient_optimize(a_opt, n, deadline,
                                                max_iters=max_exp_iters)
        print(f"[Stage 2] after Exp‑Grad – surrogate = {sur_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Light local mass‑transfer search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_attempts = int(min(250_000, remaining * 2500))
        a_opt, sur_opt = _local_search(a_opt, n, deadline, rng, max_attempts)
        print(f"[Stage 3] after local search – surrogate = {sur_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Optional up‑sampling and refinement on a finer grid
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 30.0:
            new_n = n * 2
            print(f"\n[Stage 4] Upsampling to n={new_n}")
            up_seq = _upsample_sequence(a_opt, new_n)

            # Adam on the finer grid
            up_seq, up_sur = _adam_optimize(up_seq, new_n, deadline)
            print(f"[Upsample‑Adam] surrogate = {up_sur:.6f}")

            # Exp‑Grad on the finer grid
            if time.time() < deadline:
                remaining = deadline - time.time()
                max_exp_iters_fine = int(min(400_000, remaining * 200))
                up_seq, up_sur = _exp_gradient_optimize(up_seq, new_n, deadline,
                                                        max_iters=max_exp_iters_fine)
                print(f"[Upsample‑ExpGrad] surrogate = {up_sur:.6f}")

            # Keep the finer solution if it improves the surrogate
            if up_sur < sur_opt:
                a_opt = up_seq
                sur_opt = up_sur
                n = new_n
                print(f"[Update] new best after up‑sampling – surrogate = {sur_opt:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(a_opt.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {sur_opt:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                                           # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose the result for possible continuation runs
    globals()["height_sequence_1"] = a_opt.tolist()
    return a_opt.tolist()
