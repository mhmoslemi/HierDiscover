def _adam_optimize(start_vec, n, deadline, beta_schedule, steps_per_beta):

    logits = np.log(start_vec + 1e-30)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1=0.9; beta2=0.999; eps=1e-8; t=0

    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            t += 1
            if time.time() > deadline:
                return _softmax(logits)
            a = _softmax(logits)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_term = np.exp(beta * (conv - shift))
            sum_exp = exp_term.sum()
            # compute w
            w = exp_term / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            dot = np.dot(grad_a, a)
            grad_logits = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_logits
            v = beta2 * v + (1 - beta2) * (grad_logits ** 2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            lr = 0.1 / (np.sqrt(v_hat) + eps)  # element-wise learning rate
            logits -= lr * m_hat
    return _softmax(logits)

def _projected_subgradient(start_seq, n, deadline, max_iters):

    a = start_seq.copy()
    best = a.copy()
    best_val = _objective_surrogate(a)

    # momentum vector
    momentum = np.zeros_like(a)
    momentum_coeff = 0.9

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # gradient: g_i = 2*a[max_idx - i] for valid indices
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end+1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]
        # momentum update
        momentum = momentum_coeff * momentum + (1 - momentum_coeff) * g
        # step size
        eta = 0.5 / np.sqrt(it)
        # update a in Euclidean space
        a = a - eta * momentum
        # project onto simplex
        a = _proj_simplex(a)
        # evaluate
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best = a.copy()
    return best, best_val

"""
Hybrid heuristic to minimise 2·n·max(conv(a,a)) / (sum(a)^2).

The algorithm proceeds in several stages within the time budget:
  1. Generate a large, diverse pool of ℓ₁‑normalised seed vectors
     (analytic families, Dirichlet draws, low‑frequency curves, etc.).
  2. Rank seeds by a smooth surrogate (log‑sum‑exp of the convolution)
     and keep the best few.
  3. Refine each kept seed with a **Adam optimiser** on the smooth
     surrogate.  The smoothing parameter β is increased on a schedule,
     ending at a very large β (≈ 5000) so the surrogate approaches the
     true ∞‑norm.
  4. Perform a long **exponentiated‑gradient** (mirror‑descent) run on
     the exact max‑convolution objective.
  5. Further polish the result with a Euclidean **projected sub‑gradient**
     method (with momentum) that directly minimises the true max.
  6. If time permits, up‑sample the best sequence (linear interpolation)
     to double the resolution and repeat stages 3‑5 on the finer grid.
  7. A quick stochastic “local‑swap’’ search is finally applied to
     escape any remaining shallow minima.

All vectors are kept on the probability simplex (non‑negative,
ℓ₁‑normalised) – the evaluation ratio is invariant to scaling, so the
constraint sum = 1 is sufficient.  The best sequence found is stored in
the global name ``height_sequence_1`` and returned as a plain list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate objective (sum = 1 ⇒ true objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)          # Bessel I0 (numpy implementation)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Seed collection (richer than the original version)

def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a large pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian families
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    seeds.append(_double_gaussian(n, sigma_frac=0.10, shift_frac=0.30))
    seeds.append(_double_gaussian(n, sigma_frac=0.15, shift_frac=0.25))

    # Power‑law
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta family grid
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws (heavy‑tailed)
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise robustly
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule: list[float],
                   steps_per_beta: int) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective using Adam.
    Returns an ℓ₁‑normalised vector (softmax of the logits).
    """
    # initialise logits (log‑softmax parameters)
    logits = np.log(start_vec + 1e-30)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    for beta in beta_schedule:
        beta_val = float(beta)
        for _ in range(steps_per_beta):
            t += 1
            if time.time() > deadline:
                return _softmax(logits)

            a = _softmax(logits)                     # probability vector
            conv = _conv_fft(a, a)                    # full autocorrelation
            shift = np.max(conv)                      # max value (used for stability)

            # smooth max via log‑sum‑exp
            exp_term = np.exp(beta_val * (conv - shift))
            sum_exp = np.sum(exp_term)

            # weight vector for the smooth max
            w = exp_term / sum_exp

            # gradient wrt a : 2·conv(rev_a, w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 2·n‑1
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule: d obj / d logits = Jacobian(softmax)·grad_a
            dot = np.dot(grad_a, a)
            grad_logits = a * (grad_a - dot)

            # Adam update (per‑coordinate learning rate)
            m = beta1 * m + (1 - beta1) * grad_logits
            v = beta2 * v + (1 - beta2) * (grad_logits ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.1 / (np.sqrt(v_hat) + eps)
            logits -= lr * m_hat

    return _softmax(logits)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size η_t = 0.5 / sqrt(t)
        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected Euclidean sub‑gradient descent (with momentum)
# ----------------------------------------------------------------------
def _projected_subgradient_optimize(start_seq: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient method with Euclidean projection onto the simplex.
    Uses a Nesterov‑style momentum term.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    momentum = np.zeros_like(a)
    mom_coeff = 0.9

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # gradient of the active term (zero elsewhere)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]

        # momentum update
        momentum = mom_coeff * momentum + (1 - mom_coeff) * g

        # step‑size schedule
        eta = 0.5 / np.sqrt(it)

        a = a - eta * momentum
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ProjGrad] it={it} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Small stochastic local search to escape shallow minima
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_trials: int = 15_000,
                  delta_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Randomly move a tiny amount of mass between two positions.
    Accepts moves that improve the surrogate.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    rng = np.random.default_rng()

    for _ in range(max_trials):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a small fraction of the mass at i to j
        eps = delta_frac * a[i]
        if eps <= 0:
            continue
        new = a.copy()
        new[i] -= eps
        new[j] += eps
        # projection is not necessary – sum stays 1 and all entries stay ≥0
        cur_val = _objective_surrogate(new)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = new.copy()
            a = new
    return best_seq, best_val


# ----------------------------------------------------------------------
# Upsampling utilities
# ----------------------------------------------------------------------
def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = new_seq / new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    # --------------------------------------------------------------
    # Initialisation
    # --------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # small safety margin

    # --------------------------------------------------------------
    # Stage 0 – base resolution
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # --------------------------------------------------------------
    # Stage 1 – seed generation
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)

    # Add a few extra Dirichlet draws with varied concentration
    for conc in (0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    print(f"Generated {len(seeds)} seed vectors.")

    # --------------------------------------------------------------
    # Rank seeds by the smooth surrogate (fast)
    # --------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best surrogate among seeds: {evaluated[0][1]:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – Adam optimisation on the surrogate
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] current best surrogate = {global_best_val:.6f}")

    # β‑schedule that ends at a very large value (close to the true max)
    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all seeds.")
            break

        # adapt the number of Adam steps per β to the remaining time
        remaining = deadline - time.time()
        # Rough estimate: one Adam step ≈ 0.001 s, give at most 3000 steps
        steps_per_beta = int(min(3000, max(500, remaining * 250)))   # 250 steps per second approx.
        steps_per_beta = max(500, steps_per_beta)

        print(f"\n--- Adam optimisation on seed #{idx+1} (steps/β={steps_per_beta}) ---")
        seq_opt = _adam_optimize(seed_vec,
                                 n,
                                 deadline,
                                 beta_schedule=beta_sched,
                                 steps_per_beta=steps_per_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[Adam] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – long exponentiated‑gradient refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(1_500_000, remaining * 2500))      # about 2500 iters per second
        max_exp_iters = max(0, max_exp_iters)
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient refinement ---")
            seq_exp, val_exp = _exp_gradient_optimize(global_best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=max_exp_iters)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – projected Euclidean sub‑gradient polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_proj_iters = int(min(400_000, remaining * 800))       # aggressive but safe
        max_proj_iters = max(0, max_proj_iters)
        if max_proj_iters > 0:
            print("\n--- Projected sub‑gradient polishing ---")
            seq_proj, val_proj = _projected_subgradient_optimize(global_best_seq,
                                                                 n,
                                                                 deadline,
                                                                 max_iters=max_proj_iters)
            if val_proj < global_best_val:
                global_best_seq = seq_proj.copy()
                global_best_val = val_proj
                print(f"[ProjGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – small stochastic local search
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_trials = int(min(20_000, remaining * 300))           # ~300 trials per second
        max_trials = max(0, max_trials)
        if max_trials > 0:
            print("\n--- Local random search ---")
            seq_loc, val_loc = _local_search(global_best_seq,
                                             n,
                                             deadline,
                                             max_trials=max_trials,
                                             delta_frac=0.015)
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[LocalSearch] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 6 – optional up‑sampling to a finer grid
    # --------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 35.0:                     # enough time to meaningfully upscale
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} ---")
            up_seq = _upsample_sequence(global_best_seq, new_n)

            # Run a shorter Adam optimisation on the finer grid
            remaining = deadline - time.time()
            steps_per_beta_up = int(min(2000, max(300, remaining * 150)))
            steps_per_beta_up = max(300, steps_per_beta_up)

            up_opt = _adam_optimize(up_seq,
                                    new_n,
                                    deadline,
                                    beta_schedule=beta_sched,
                                    steps_per_beta=steps_per_beta_up)
            up_val = _objective_surrogate(up_opt)
            print(f"[Upsample‑Adam] surrogate = {up_val:.6f}")

            # Follow with a modest sub‑gradient run
            if time.time() < deadline:
                remaining = deadline - time.time()
                max_exp_up = int(min(300_000, remaining * 1200))
                max_exp_up = max(0, max_exp_up)
                seq_up_refine, val_up_refine = _exp_gradient_optimize(up_opt,
                                                                     new_n,
                                                                     deadline,
                                                                     max_iters=max_exp_up)
                if val_up_refine < global_best_val:
                    global_best_seq = seq_up_refine.copy()
                    global_best_val = val_up_refine
                    n = new_n
                    print(f"[Upsample‑Refine] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
