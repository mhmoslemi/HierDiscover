def _expgrad_multi(start, n, deadline, max_iters, eta0=0.25, tol=0.001):
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)
    for it in range(1, max_iters+1):
        if time.time() > deadline: break
        conv = _conv_fft(x, x)
        max_val = conv.max()
        # select indices where conv >= max_val*(1-tol)
        idxs = np.where(conv >= max_val * (1 - tol))[0]
        g = np.zeros_like(x)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx
            g[idx] += 2.0 * x[rev]
        # average gradient
        g = g / len(idxs) if len(idxs) > 0 else g
        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        # clip for exponent stability
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)
        # projection onto simplex
        x = _proj_simplex(x)
        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val, best_x = cur_val, x.copy()
    return best_x, best_val

"""
Propose a non‑negative step function that minimises the official
``evaluate_sequence`` objective.

Algorithm
---------
0️⃣  Generate a rich pool of ℓ₁‑normalised seed vectors (uniform,
    triangular, various window families, DPSS, random Fourier, Dirichlet
    draws, low‑frequency cosine sums, mixtures).  Keep the best few
    according to the exact surrogate 2·n·max(conv).

1️⃣  Soft‑max Adam refinement – minimise the smooth surrogate
        Lα(p)= (1/α)·log Σₖ exp(α·(p∗p)[k])
    with a slowly increasing temperature α (up to 1e4).  Several short
    restarts from the top seeds are performed; the best result is kept.

2️⃣  Multi‑active exponential‑gradient polish – a mirror‑descent on the
    probability simplex.  Instead of using a single active shift,
    all shifts whose convolution value is within a tiny relative
    tolerance of the current maximum are taken into account (the true
    sub‑gradient of the max‑function is a convex combination of those).
    The multiplicative update
        x←x·exp(−η·g)   with η≈η₀/√t
    is followed by Euclidean projection onto the simplex.

3️⃣  Simulated‑annealing final tweak – random mass reshuffles on small
    subsets and tiny pairwise transfers, accepting any move that
    improves the exact surrogate (or with a Boltzmann probability at
    temperature T).  Runs until the time budget is exhausted.

All stages stop before the deadline, the best vector found is stored
in ``height_sequence_1`` for reuse, and its true official score is
returned.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(p,p)) for a distribution with ∑p=1."""
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic window families (always ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _dpss_window(n: int, nw: float = 2.5) -> np.ndarray:
    """Discrete prolate spheroidal sequence (DPSS).  Falls back to uniform."""
    try:
        from scipy.signal import windows as win
        a = win.dpss(n, nw, sym=False)[0]
        a = np.maximum(a, 0.0)            # DPSS can be tiny negative due to numerics
        s = a.sum()
        if s > 0:
            return a / s
    except Exception:
        pass
    return _uniform(n)


def _random_fourier(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a real non‑negative vector by randomising phase in the Fourier
    domain, then taking absolute value.  Finally ℓ₁‑normalise.
    """
    # generate random complex spectrum with Hermitian symmetry
    N = _next_pow2(2 * n)
    phase = rng.uniform(0, 2 * np.pi, size=N // 2 + 1)
    mag = rng.exponential(scale=1.0, size=N // 2 + 1)
    spec = mag * np.exp(1j * phase)
    # enforce Hermitian symmetry for real inverse FFT
    spec_full = np.concatenate([spec, np.conj(spec[-2:0:-1])])
    a = np.fft.ifft(spec_full).real
    a = np.abs(a[:n])                     # ensure non‑negative
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _collect_seeds_extended(n: int, rng: np.random.Generator,
                           previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    seeds.append(_gaussian(n, 0.2))
    seeds.append(_gaussian(n, 0.3))
    seeds.append(_gaussian(n, 0.4))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # DPSS windows (different bandwidth parameters)
    for nw in (2.0, 2.5, 3.0, 4.0):
        seeds.append(_dpss_window(n, nw))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # random Fourier seeds
    for _ in range(4):
        seeds.append(_random_fourier(n, rng))

    # mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final safety normalisation
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        clean.append(v / s if s > 0 else _uniform(n))
    return clean

# Optimisers
# ----------------------------------------------------------------------
def _adam_softmax_optimize(start: np.ndarray, n: int,
                           deadline: float, max_iters: int,
                           alpha_start: float = 20.0,
                           alpha_end: float = 10000.0,
                           lr: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        L(p) = (1/α)·log Σₖ exp(α·(p∗p)[k]).
    α is linearly increased from ``alpha_start`` to ``alpha_end``.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    N = _next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear convolution via FFT (full length)
        fx = rfft(x, n=N)
        conv_full = irfft(fx * fx, n=N)[:2 * n - 1]

        # annealing schedule for α
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)

        # stable log‑sum‑exp (soft‑max)
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        # correlation = gradient of L wrt x (via convolution theorem)
        fs = rfft(soft, n=N)
        corr = irfft(np.conj(fx) * fs, n=N)[:n]   # Σ_j x_j·soft[i+j]
        grad = 2.0 * corr

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # projection onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val, best_x = cur_val, x.copy()
    return best_x, best_val


def _expgrad_multi(start: np.ndarray, n: int,
                   deadline: float, max_iters: int,
                   eta0: float = 0.2,
                   tol: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Multiplicative (mirror‑descent) optimiser.
    Uses *all* shifts whose convolution value is within ``tol`` of the
    current maximum to form a true sub‑gradient of the max‑function.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        max_val = conv.max()
        active = np.where(conv >= max_val * (1.0 - tol))[0]

        # accumulate sub‑gradient over the active set
        g = np.zeros_like(x)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx
            g[idx] += 2.0 * x[rev]
        if len(active) > 0:
            g /= len(active)

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)            # prevent overflow
        x = x * np.exp(delta)
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val, best_x = cur_val, x.copy()

    return best_x, best_val


def _simulated_annealing(start: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Short simulated‑annealing which works directly on the exact surrogate.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.6:
            # reshuffle a random small subset
            k_sub = rng.integers(3, 13)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()

    return best, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    the official ``evaluate_sequence`` objective within the allotted time.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0     # small safety margin

    # ------------------------------------------------------------------
    # Choose an appropriate length.  Larger n gives a tighter theoretical
    # bound but costs more work per iteration.
    # ------------------------------------------------------------------
    n = 6000   # works well within the 1000 s budget

    # ------------------------------------------------------------------
    # Stage 0 – Seed generation
    # ------------------------------------------------------------------
    print(f"\n=== Seed generation (n={n}) ===")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # Evaluate all seeds (surrogate) and keep the best few
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation on the smooth surrogate (multiple restarts)
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Soft‑max Adam optimisation (multiple restarts) ---")
        # allocate roughly 30 % of remaining budget to Adam
        remaining_total = deadline - time.time()
        per_restart = remaining_total * 0.3 / min(4, len(best_seeds[:4]))
        for idx, seed_vec in enumerate(best_seeds[:4]):
            if time.time() >= deadline:
                break
            stop = time.time() + per_restart
            max_iters = int(250_000 * (per_restart / remaining_total))
            cand, val = _adam_softmax_optimize(
                seed_vec, n, stop,
                max_iters=max_iters,
                alpha_start=20.0,
                alpha_end=10000.0,
                lr=0.06
            )
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Adam] restart #{idx+1} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Multi‑active exponential‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Multi‑active Exponential‑Gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(700_000, remaining * 1300))
        best_seq, best_val = _expgrad_multi(
            best_seq, n, deadline, max_iters=max_iters,
            eta0=0.22, tol=0.001
        )
        print(f"[ExpGrad] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Simulated‑annealing final tweak
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 1900))
        best_seq, best_val = _simulated_annealing(
            best_seq, n, deadline, max_iters=max_iters,
            init_temp_factor=0.07
        )
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse by the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
