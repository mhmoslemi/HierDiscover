    v = start_v.copy()
    m = np.zeros_like(v)
    vvar = np.zeros_like(v)
    t = 0
    best_a = None
    best_val = np.inf

    while time.time() < deadline and t < max_iter:
        t += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_idx = np.argmax(conv)
        # compute gradient as before for max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # compute grad wrt v
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        m = betas[0] * m + (1 - betas[0]) * grad_v

#!/usr/bin/env python
"""
Search for a non‑negative step‑function minimizing the
``evaluate_sequence`` objective.

The algorithm works in several stages:

1.  Generate a diverse pool of ℓ₁‑normalised candidates from analytic
    families (uniform, triangular, Gaussian, Beta, Tukey, power‑law,
    sine‑window, double‑Gaussian, random Dirichlet, etc.).
2.  Evaluate the smooth surrogate 2·n·max(conv) for all candidates and
    keep the top few.
3.  Refine each top candidate with a continuation L‑BFGS optimisation
    of a log‑sum‑exp smoothed version of the max‑convolution.
4.  Perform a large‑scale mirror‑descent (Adam on the logits) on the
    best refined candidate to directly minimise the exact max‑convolution.
5.  Finish with a short stochastic hill‑climbing pass that moves tiny
    amounts of mass between random coordinates.

The whole pipeline respects the global time budget (≈1000 s) and
stores the final sequence in the global name ``height_sequence_1``.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective for a ℓ₁‑normalised vector:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        s = a.sum()
        return a / s
    # else 0 < alpha < 1
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector (softmax of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0]

    v = np.log(start_vec + 1e-30)  # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # probability simplex
            conv = _conv_fft(a, a)                     # full autocorrelation
            shift = np.max(conv)                       # certainty shift
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                     # weights for smooth max
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            # time limit reached inside L‑BFGS
            break
        except Exception as e:  # pragma: no cover
            # Unexpected problem – abort this beta
            print(f"[L‑BFGS] exception {e} at beta={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam optimiser on the logits (exact max‑convolution objective)
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 300_000,
                   lr: float = 0.1,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   log_interval: int = 5000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam on the simplex using logits.
    Returns the best vector seen and its surrogate value.
    """
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    vvar = np.zeros_like(v)
    best_seq = None
    best_val = np.inf
    t = 0

    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(v)

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # gradient wrt logits
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * grad_v
        vvar = beta2 * vvar + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        vvar_hat = vvar / (1.0 - beta2 ** t)

        v -= lr * m_hat / (np.sqrt(vvar_hat) + eps)

        if (t % log_interval) == 0:
            cur_val = _objective_surrogate(_softmax(v))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = _softmax(v).copy()
                print(f"[Adam] it={t} best_surrogate={best_val:.6f}")

    if best_seq is None:
        best_seq = _softmax(v)
        best_val = _objective_surrogate(best_seq)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill climbing on the simplex
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                n: int,
                deadline: float,
                max_trials: int = 20000,
                step_frac: float = 0.02,
                rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Randomly moves a small fraction of mass between two coordinates,
    accepting only improvements (exact surrogate evaluation).
    """
    if rng is None:
        rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    for _ in range(max_trials):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        if best_seq[i] < 1e-12:
            continue
        delta = min(best_seq[i] * step_frac, best_seq[i])
        # Transfer delta from i to j
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cur_val = _objective_surrogate(cand)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cand
    return best_seq, best_val

    # Random exponential

"""
Search for a non‑negative step‑function that minimises the
``evaluate_sequence`` objective.

The algorithm works in several stages:

1.  Generate a diverse pool of ℓ₁‑normalised candidates from several
    analytic families (uniform, triangular, Gaussian, Beta, Tukey,
    power‑law, sine‑window, double‑Gaussian, Dirichlet, exponential).
2.  Evaluate a smooth surrogate 2·n·max(conv) for all candidates and keep
    the best few.
3.  Refine each of those seeds with a continuation L‑BFGS optimisation of
    a log‑sum‑exp smoothed version of the max‑convolution.
4.  Run a large‑scale Adam optimiser on the logits (mirror‑descent on
    the simplex) to directly minimise the exact max‑convolution.
5.  Finish with a short stochastic hill‑climbing pass that moves a tiny
    fraction of mass between random coordinates.

All stages respect the global time budget (≈1000 s) and the final
sequence is stored in the global variable ``height_sequence_1``.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective for a ℓ₁‑normalised vector:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 1500) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector (softmax of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0]

    v = np.log(start_vec + 1e-30)  # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at beta={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam optimiser on the logits (exact max‑convolution objective)
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 200_000,
                   lr: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   log_interval: int = 10000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam on the simplex using logits.
    Returns the best vector seen and its surrogate value.
    """
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    vvar = np.zeros_like(v)
    best_seq = None
    best_val = np.inf
    t = 0

    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(v)

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # gradient wrt logits
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_v
        vvar = beta2 * vvar + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        vvar_hat = vvar / (1.0 - beta2 ** t)

        v -= lr * m_hat / (np.sqrt(vvar_hat) + eps)

        if (t % log_interval) == 0:
            cur_val = _objective_surrogate(_softmax(v))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = _softmax(v).copy()
                print(f"[Adam] it={t} best_surrogate={best_val:.6f}")

    if best_seq is None:
        best_seq = _softmax(v)
        best_val = _objective_surrogate(best_seq)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill climbing on the simplex
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                n: int,
                deadline: float,
                max_trials: int = 20000,
                step_frac: float = 0.02,
                rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Randomly moves a small fraction of mass between two coordinates,
    accepting only improvements (exact surrogate evaluation).
    """
    if rng is None:
        rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    for _ in range(max_trials):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        if best_seq[i] < 1e-12:
            continue
        delta = min(best_seq[i] * step_frac, best_seq[i])
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cur_val = _objective_surrogate(cand)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cand
    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed collection – analytic families + random draws
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (0.5, 1.0, 1.5, 2.0):
        seeds.append(_sine_window(n, gamma=gamma))
    for sigma in (0.15, 0.20, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))
    for sigma in (0.12, 0.18, 0.24):
        for shift in (0.2, 0.3, 0.4):
            seeds.append(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))
    for p in (0.0, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))
    alphas = np.linspace(0.5, 4.0, 8)
    betas = np.linspace(0.5, 4.0, 8)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet
    for conc in (0.1, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,) and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise safety
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        cleaned.append(v / s if s > 0 else _uniform(n))
    return cleaned

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # ------------------------------------------------------------------
    # Problem size (the original code used 4000 – keep the same)
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # ------------------------------------------------------------------
    # Stage 1 – build a pool of seeds
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape != (n,):
                # simple linear interpolation if size differs
                prev_arr = np.interp(np.linspace(0, 1, n),
                                     np.linspace(0, 1, len(prev_arr)),
                                     prev_arr)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seed vectors.")

    # Evaluate surrogate for all seeds (fast)
    seed_vals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in seed_vals[:top_k]]
    print(f"Top {top_k} surrogate values (seed stage):")
    for i, (_, val) in enumerate(seed_vals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – L‑BFGS refinement on the best seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] surrogate = {global_best_val:.6f}")

    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all L‑BFGS seeds.")
            break
        remaining = deadline - time.time()
        # Allow a few thousand iterations per L‑BFGS run
        max_iter_beta = max(800, int(min(1500, remaining * 0.06 * 1e5)))
        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter/beta={max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=max_iter_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Adam optimisation on the best candidate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_adam_iters = int(min(200_000, remaining * 0.3 * 1e5))
        print(f"\n--- Adam optimisation (max_iters={max_adam_iters}) ---")
        seq_adam, val_adam = _adam_optimize(global_best_seq,
                                            n,
                                            deadline,
                                            max_iters=max_adam_iters,
                                            lr=0.2,
                                            log_interval=10000)
        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – short stochastic hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_trials = int(min(30000, remaining * 2500))
        print(f"\n--- Hill‑climbing refinement (max_trials={max_trials}) ---")
        seq_hc, val_hc = _hill_climb(global_best_seq,
                                     n,
                                     deadline,
                                     max_trials=max_trials,
                                     step_frac=0.02,
                                     rng=rng)
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[HillClimb] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store the sequence for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
