def conv_fft(a,b):
    L = len(a)+len(b)-1
    N = 1 << ((L - 1).bit_length())
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa*fb, n=N)[:L]
    return c

"""
Propose a non‑negative step function that minimises the official
``evaluate_sequence`` objective.

Algorithm
---------
1️⃣  **Seed generation** – a large, diverse set of analytically‑defined
    shapes (uniform, triangular, sine, Beta, Gaussian, mixtures of
    Gaussians, low‑frequency cosine sums, random Dirichlet draws, the
    previous best solution).  All seeds are ℓ₁‑normalised (∑ = 1) and the
    best few are kept.

2️⃣  **Fast ℓ∞‑convolution minimisation** – a simple sub‑gradient descent
    on the quantity ``M = max_k (a*a)[k]`` (the ℓ∞‑norm of the
    self‑convolution).  For a current peak index ``k`` the sub‑gradient
    is ``g_i = 2·a_{k‑i}`` for the indices that contribute to the peak.
    We apply a multiplicative (exponentiated‑gradient) update
    ``a_i ← a_i·exp(‑η·g_i)`` and re‑normalise onto the probability simplex.
    The step size η decays like 1/√t.  The loop runs until the time budget
    is exhausted or a fixed number of iterations.

3️⃣  **Final stochastic polishing** – a short simulated‑annealing phase
    (tiny mass transfers and small Dirichlet reshuffles) that can escape
    tiny local plateaus left by the deterministic gradient step.

The whole pipeline is time‑aware: each stage consumes a fraction of the
remaining budget, guaranteeing a valid answer before the 1000 s limit.
The function returns the best sequence found as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = a.shape[0] + b.shape[0] - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """Exact objective (since sum(a)=1, it equals 2·n·max(conv))."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape factories (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_mixture(n: int, centers, sigmas, weights) -> np.ndarray:
    """Sum of (weight·Gaussian) components, then normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for c, sgm, w in zip(centers, sigmas, weights):
        a += w * np.exp(-((i - c) ** 2) / (2.0 * sgm ** 2))
    tot = a.sum()
    return a / tot if tot > 0 else _uniform(n)


def _lowfreq_cosine_sum(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Random low‑frequency cosine series, clipped to non‑negative & normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for _ in range(num_freq):
        k = rng.integers(1, n // 4 + 1)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (0.5, 1.0, 2.0, 3.0, 4.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.08, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))
    # beta shapes – wide sweep over (α,β)
    alphas = np.logspace(np.log10(0.06), np.log10(5.0), num=15)
    betas = np.logspace(np.log10(0.06), np.log10(5.0), num=15)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, float(a), float(b)))
    # mixtures of two Gaussians – “double‑hump”
    for _ in range(80):
        c1 = rng.integers(int(0.12 * n), int(0.38 * n))
        c2 = rng.integers(int(0.62 * n), int(0.88 * n))
        sigma = rng.uniform(0.03 * n, 0.07 * n)
        seeds.append(_gaussian_mixture(
            n,
            centers=[c1, c2],
            sigmas=[sigma, sigma],
            weights=[0.5, 0.5],
        ))
    # low‑frequency cosine sums
    for nf in (2, 3, 4):
        for _ in range(10):
            seeds.append(_lowfreq_cosine_sum(n, nf, rng))
    # random Dirichlet draws
    for conc in (0.05, 0.15, 0.4, 1.0, 2.0):
        for _ in range(12):
            seeds.append(_random_dirichlet(n, conc, rng))
    # random exponential draws
    for _ in range(15):
        seeds.append(_random_exponential(n, rng))

    # previous best (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final safety normalisation (should already be normalised)
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ----------------------------------------------------------------------
# Sub‑gradient exponentiated‑gradient optimiser
# ----------------------------------------------------------------------
def _gd_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    eta0: float = 0.8,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Multiplicative sub‑gradient descent on max(self‑conv) with simplex projection."""
    x = start.copy()
    best = x.copy()
    best_val = _objective(x)
    cur_val = best_val

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # full convolution
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active peak
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # step size schedule
        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient update (multiplicative)
        mult = np.exp(-eta * g)
        x = x * mult
        # renormalise onto the probability simplex (actually just ℓ₁, because x≥0)
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective(x)
        if cur_val < best_val:
            best_val = cur_val
            best = x.copy()
            if it % 50000 == 0:
                print(f"[GD] it={it:7d} cur={cur_val:.6f} best={best_val:.6f}")
        # monotonic acceptance – keep the current point even if it is worse
        # (the multiplicative update is usually improving)
    return best, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (fast but effective)
# ----------------------------------------------------------------------
def _simulated_annealing(
    start: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-12,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # exponential cooling
        T = max(T * 0.99995, min_temp)

        # two simple move types
        if rng.random() < 0.6:
            # reshuffle a random tiny subset (3‑12 entries)
            k_sub = rng.integers(3, 13)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.06)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 200_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val
