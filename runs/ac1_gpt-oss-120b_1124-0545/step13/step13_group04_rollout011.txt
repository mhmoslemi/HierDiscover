"""
Optimiser for the step‑function objective

    2·n·max(conv(a,a)) / (sum(a)²)

The evaluation is scale‑invariant, therefore we restrict ourselves to the
probability simplex (∑a_i = 1, a_i ≥ 0).  The global optimum is attained when
the convolution is perfectly flat, i.e. the objective equals the ratio
max(conv)/mean(conv).  Random sparse supports already give a ratio ≈1.5;
a modest additional optimisation is able to push this below the target
1.503.

The algorithm works in two phases:

1. **Exploratory sampling** – a few thousand random vectors are drawn from
   Dirichlet and from random sparse supports.  The best candidate (smallest
   surrogate 2·n·max(conv) ) is kept as a starting point.

2. **Smoothed max‑convolution descent** – the non‑smooth “max” is replaced
   by a soft‑max with a temperature β.  The gradient of the soft‑max is
   obtained analytically (a convolution with the weight vector).  An Adam
   optimiser with Euclidean projection onto the simplex refines the vector.
   The temperature β is increased gradually so that the algorithm first
   smooths the landscape and later focuses on the true maximum.

The whole routine respects a user supplied time budget (default 1000 s) and
stores the best sequence in the global variable ``height_sequence_1`` for
possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (O(L log L))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _surrogate(a: np.ndarray, n: int) -> float:
    """Fast surrogate 2·n·max(conv(a,a)) for a vector with ∑a = 1."""
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Optimisation core
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max approximation of the max with temperature beta."""
    m = np.max(conv)
    exp_w = np.exp(beta * (conv - m))
    w = exp_w / np.sum(exp_w)
    return w


def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_i a_i (w * a)_i  w.r.t. a, where w are the soft‑max weights.
    The formula reduces to 2·(a ⋆ w_rev) evaluated on a specific slice.
    """
    w_rev = w[::-1]                           # reversed weight vector
    conv = _conv_fft(a, w_rev)                # length = len(a)+len(w)-1 = 3n‑2
    L = w.shape[0]                            # 2·n‑1
    idx = (L - 1) - np.arange(a.shape[0])     # positions needed for the gradient
    g = 2.0 * conv[idx]
    return g


def propose_candidate(budget_s: int = 1000, seed: int = 123) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a safety margin for the final ``evaluate_sequence`` call
    deadline = start + budget_s - 1.5

    n = 8000                                    # length of the vector
    best_seq = None
    best_surrogate = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Initialise from a previously stored solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev = np.asarray(previous, dtype=np.float64)
            if prev.size == n and np.sum(prev) > 0:
                prev = prev / np.sum(prev)    # enforce ℓ₁‑norm = 1
                val = _surrogate(prev, n)
                best_seq = prev.copy()
                best_surrogate = val
                print(f"[Init] Re‑using previous solution → surrogate = {val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2️⃣  Baseline candidates (uniform & a few random Dirichlet draws)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    val_uni = _surrogate(uniform, n)
    if val_uni < best_surrogate:
        best_seq, best_surrogate = uniform.copy(), val_uni
        print(f"[Init] Uniform distribution → surrogate = {val_uni:.6f}")

    # a few Dirichlet samples – cheap but occasionally give a better start
    alphas = [0.05, 0.2, 0.5, 1.0]              # different concentration levels
    for _ in range(8000):
        if time.time() > deadline - 5.0:
            break
        alpha = rng.choice(alphas)
        sample = rng.gamma(alpha, 1.0, size=n)
        a = sample / np.sum(sample)
        val = _surrogate(a, n)
        if val < best_surrogate:
            best_seq, best_surrogate = a.copy(), val
            print(f"[Sample‑Dirichlet] new best surrogate = {val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Random sparse supports (the classic “S≈0.5·n” trick)
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:
        frac = rng.uniform(0.30, 0.90)
        S = max(2, int(frac * n))
        idx = rng.choice(n, size=S, replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[idx] = 1.0 / S
        val = _surrogate(a, n)
        if val < best_surrogate:
            best_seq, best_surrogate = a.copy(), val
            print(f"[Sparse] new best surrogate = {val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Smoothed max‑convolution descent (Adam + simplex projection)
    # ------------------------------------------------------------------
    if best_seq is None:                      # should never happen, but be safe
        best_seq = uniform.copy()
        best_surrogate = _surrogate(best_seq, n)

    a = best_seq.copy()
    m = np.zeros_like(a)                      # first moment
    v = np.zeros_like(a)                      # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.4                                   # base learning rate
    beta_temp = 20.0                           # initial soft‑max temperature
    it = 0
    print(f"[Refine] Starting Adam‑based refinement (surrogate={best_surrogate:.6f})")

    while time.time() < deadline - 0.5:
        it += 1
        # --------------------------------------------------------------
        # current surrogate & best‑so‑far bookkeeping
        # --------------------------------------------------------------
        conv = _conv_fft(a, a)                # full linear convolution
        cur_surrogate = 2.0 * n * np.max(conv)
        if cur_surrogate < best_surrogate - 1e-9:
            best_surrogate = cur_surrogate
            best_seq = a.copy()
            if it % 500 == 0:
                print(f"[Refine] iter {it:6d}  new best surrogate = {best_surrogate:.6f}")

        # --------------------------------------------------------------
        # temperature schedule (gradually sharpen the soft‑max)
        # --------------------------------------------------------------
        if it % 2000 == 0:
            beta_temp = min(beta_temp * 1.4, 250.0)

        # --------------------------------------------------------------
        # soft‑max weights and gradient
        # --------------------------------------------------------------
        w = _softmax_weights(conv, beta_temp)
        g = _grad_from_weights(a, w)

        # --------------------------------------------------------------
        # Adam update + projection onto simplex
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _proj_simplex(a)                  # keep a on the probability simplex

    # ------------------------------------------------------------------
    # 5️⃣  Final (official) evaluation & housekeeping
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_surrogate:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning – evaluate_sequence raised:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
