Hybrid L‑BFGS‑B optimizer for the step‑function minimisation problem.

The algorithm proceeds as follows:

def _next_pow2(x: int) -> int:

    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for ‖seq‖₁ = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

    """Flat distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exponential_shape(n: int, r: float) -> np.ndarray:
    """Monotone exponential a_i ∝ r**i, ℓ₁‑normalised."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _exponential_offset_shape(n: int, offset: int, r: float) -> np.ndarray:
    """Zero prefix of length *offset* followed by an exponential tail."""
    if offset >= n:
        return _uniform(n)
    tail_len = n - offset
    if r <= 0.0:
        tail = np.full(tail_len, 1.0 / tail_len, dtype=np.float64)
    else:
        i = np.arange(tail_len, dtype=np.float64)
        tail = np.power(r, i)
        tail_sum = tail.sum()
        if tail_sum == 0.0:
            tail = np.full(tail_len, 1.0 / tail_len, dtype=np.float64)
        else:
            tail = tail / tail_sum
    a = np.concatenate([np.zeros(offset, dtype=np.float64), tail])
    a_sum = a.sum()
    if a_sum == 0.0:
        return _uniform(n)
    return a / a_sum

# ----------------------------------------------------------------------
# L‑BFGS‑B optimisation of the smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _lbfgs_optimize(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule=None,
                    max_iter_per_beta: int = 200) -> tuple[np.ndarray, float]:
    """
    Continuation L‑BFGS‑B optimiser.

    Parameters
    ----------
    start_seq : np.ndarray
        ℓ₁‑normalised start vector (shape (n,)).
    n : int
        Length of the vector.
    deadline : float
        Unix time after which optimisation must stop.
    beta_schedule : sequence of float, optional
        Sequence of smoothing parameters.  If omitted a default schedule
        spanning several decades is used.
    max_iter_per_beta : int, optional
        Maximum number of L‑BFGS iterations for each β value.

    Returns
    -------
    best_seq, best_val : tuple
        Best vector encountered (still ℓ₁‑normalised) and its exact surrogate
        value ``_objective_surrogate(best_seq)``.
    """
    if beta_schedule is None:
        beta_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2000.0]

    # initialise logits (log‑softmax representation)
    eps = 1e-30
    v = np.log(start_seq + eps)

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    # Core optimisation loop over smoothing parameters
    for beta in beta_schedule:
        # if time is exhausted, abort early
        if time.time() > deadline:
            break

        # define the smooth surrogate and its gradient
        def func_and_grad(v_flat: np.ndarray):
            if time.time() > deadline:
                raise TimeoutError

            a = _softmax(v_flat)                # ℓ₁‑normalised
            conv = _conv_fft(a, a)              # length 2n‑1
            shift = np.max(conv)                # max(conv)
            # smooth softmax over the convolution values
            exp_vec = np.exp(beta * (conv - shift))   # ≤ 1
            w_sum = np.sum(exp_vec)
            # in the unlikely case of underflow (w_sum == 0) we fallback to hard max
            if w_sum == 0.0:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
                w_sum = 1.0
            else:
                w = exp_vec / w_sum

            # gradient w.r.t. a (see analysis)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1: n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # smoothed log‑sum‑exp of the convolution
            logsumexp = shift + np.log(w_sum) / beta
            obj = 2.0 * n * logsumexp
            return obj, grad_v

        # L‑BFGS‑B call (no bounds needed – the softmax guarantees feasibility)
        try:
            res = minimize(fun=func_and_grad,
                           x0=v,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': max_iter_per_beta,
                                    'disp': False})
            v = res.x
        except TimeoutError:
            break
        except Exception as exc:  # pragma: no cover
            # In case SciPy raises an unexpected error we fall back to the current v.
            print(f"[LBFGS] warning: {exc}")
            # continue with current v

        # Evaluate the exact (non‑smoothed) surrogate on the new point
        cand_seq = _softmax(v)
        cand_val = _objective_surrogate(cand_seq)
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand_seq.copy()
            print(f"[LBFGS] beta={beta:.1f} new best surrogate = {cand_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple projected sub‑gradient descent (as a polishing step)
# ----------------------------------------------------------------------
def _subgradient_proj(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on max(conv) with Euclidean projection onto the simplex."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))               # index attaining the max
        cur_val = 2.0 * n * conv[idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # compute sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size schedule
        lr = 0.20 / math.sqrt(it)
        a = a - lr * g

        # Euclidean projection onto the probability simplex
        # (the implementation follows the classic O(n) algorithm)
        u = np.sort(a)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(len(a)) + 1
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond].max()
            theta = cssv[cond].max() / rho
            a = np.maximum(a - theta, 0.0)
        else:
            a = np.full_like(a, 1.0 / len(a))

    return best_seq, best_val

    for conc in (0.05, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution, if any
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final sanity – guarantee ℓ₁ normalisation
    final = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        final.append(v / s if s > 0 else _uniform(n))
    return final


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` within the allotted time budget.

    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    overall_deadline = start_time + budget_s - 1.5   # keep a tiny safety margin

    # ------------------------------------------------------------------
    # Choose a working problem size.
    # ------------------------------------------------------------------
    n = 8000
    if overall_deadline - time.time() < 250.0:
        n = 4000
        print("[Info] Very little time left – switching to n=4000.")

    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # cheap surrogate ranking (exact max‑convolution, but cheap enough)
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 5
    best_seeds = [vec for vec, _ in evals[:top_k]]

    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS‑B optimisation
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 30.0:
        print("\n--- L‑BFGS‑B optimisation phase ---")
        # allocate a modest number of L‑BFGS iterations per β
        max_iters_per_beta = max(50,
                                 int(remaining * 0.05 / (len(best_seeds) * 7)))
        for idx, seed_vec in enumerate(best_seeds):
            print(f"\n[LBFGS] Optimising seed #{idx+1}")
            seq_opt, val_opt = _lbfgs_optimize(seed_vec,
                                                n,
                                                overall_deadline,
                                                max_iter_per_beta=max_iters_per_beta)
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[LBFGS‑seed{idx+1}] new global best surrogate = {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – short sub‑gradient polishing
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 10.0:
        max_iters_sub = int(min(300_000, remaining * 20_000))
        print(f"\n--- Sub‑gradient polishing (max_iters={max_iters_sub}) ---")
        seq_sub, val_sub = _subgradient_proj(global_best_seq,
                                             n,
                                             overall_deadline,
                                             max_iters=max_iters_sub)
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[SubProj] new global best surrogate = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
