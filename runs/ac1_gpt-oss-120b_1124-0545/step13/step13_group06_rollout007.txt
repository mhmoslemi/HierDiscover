"""
Propose a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a))²

within the given time budget (≈1000 s).

The optimiser combines three ideas that improve on the reference
implementation:

1. **Richer seed pool** – additional β‑shape families (including
   α,β < 1, the classical arcsine shape) and a few extra random
   constructions.

2. **Longer, high‑temperature L‑BFGS refinement** – we run L‑BFGS on
   a smooth log‑sum‑exp surrogate at many temperatures (up to 2000)
   and with more iterations, giving a better warm‑start for later
   phases.

3. **Additive projected sub‑gradient descent** – after the
   multiplicative mirror‑descent stage we perform a classic
   projected sub‑gradient method on the *exact* max‑convolution
   objective.  An Euclidean projection onto the simplex is used,
   which (empirically) drives the peak a little lower than the
   exponentiated‑gradient update alone.

The final polishing is a light simulated‑annealing stage.
The best sequence found is stored in the global name
``height_sequence_1`` and returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Fast convolution (full linear convolution via real FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety
    w_sum = w.sum()
    if w_sum <= 0:
        w = np.full_like(v, 1.0 / n)
    else:
        w = w / w_sum
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = n - i
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Surrogate objective (identical ordering to the official one)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – works on ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max auxiliary (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """Return (smooth_max, grad) for the log‑sum‑exp surrogate."""
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum
    smooth = shift + np.log(w_sum) / beta

    # gradient: 2 * rev(a) ⊛ weights   (⊛ = linear convolution)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad


# ----------------------------------------------------------------------
# L‑BFGS on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     eps_time: float,
                     max_iter: int = 400,
                     temperature: float = 1000.0):
    """Run L‑BFGS on the smooth‑max surrogate at a fixed temperature."""
    v = np.log(start_seq + 1e-30)   # logits → simplex via softmax

    def fun(v):
        a = _softmax(v)
        smooth, grad_a = _smooth_max_and_grad(a, temperature)
        loss = 2.0 * n * smooth
        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return loss, (2.0 * n * grad_v).astype(np.float64)

    # honour the global deadline
    try:
        res = scipy.optimize.minimize(
            fun,
            v,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
    except Exception:  # pragma: no cover
        return start_seq, _objective(start_seq)

    a_opt = _softmax(res.x)
    return a_opt, _objective(a_opt)


# ----------------------------------------------------------------------
# Multiplicative mirror‑descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 5_000_000,
                                 lr0: float = 0.18,
                                 power: float = 0.30):
    """Exponentiated‑gradient updates on the exact max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 1
    while it <= max_iters:
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:,d} best={cur_val:.6f}")

        # sub‑gradient for the active anti‑diagonal
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # Project back onto the simplex
        s = a.sum()
        a = _uniform(n) if s <= 0.0 else a / s

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Additive projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _additive_subgradient_optimize(start_seq: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   max_iters: int = 2_000_000,
                                   eta0: float = 0.5,
                                   power: float = 0.50) -> tuple[np.ndarray, float]:
    """Euclidean projected sub‑gradient descent on the exact max‑conv objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 1
    while it <= max_iters:
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Additive] it={it:,d} best={cur_val:.6f}")

        # sub‑gradient for the active anti‑diagonal
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size schedule
        eta = eta0 / ((it) ** power)

        # gradient step + projection onto simplex
        a_new = a - eta * g
        a = _proj_simplex(a_new)

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light simulated‑annealing polishing
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 2_000_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a random subset
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                  # split‑transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 200_000 == 0:
                    print(f"[SA] it={it:,d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation (richer and more diverse)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_linear_decreasing(n))

    # Power‑law grid
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # β‑shape – dense grid, now also covering α,β in (0,1)
    alphas = np.concatenate((
        np.linspace(0.2, 1.5, 7),
        np.linspace(0.2, 1.5, 7) * 0.5,
    ))
    betas = np.concatenate((
        np.linspace(0.2, 1.5, 7),
        np.linspace(0.2, 1.5, 7) * 0.5,
    ))
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Exponential decay
    for lam in np.linspace(0.001, 0.020, 12):
        seeds.append(_exponential_decay(n, lam))

    # Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Dirichlet draws – several concentrations
    for conc in (0.01, 0.05, 0.1, 0.5, 1.0):
        for _ in range(4):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw ordering
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending

    # Random “sine‑plus‑offset” patterns – always non‑negative
    for amp in (0.5, 1.0, 2.0):
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n, dtype=np.float64)
        a = amp * (np.sin(2 * np.pi * t / n + phase) + 1.0)  # shift to be ≥0
        if a.sum() > 0:
            seeds.append(a / a.sum())

    # Previous best, if available
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Main optimiser entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    `evaluate_sequence` objective within the supplied time budget.
    Returns the best sequence (list of floats) and stores it as the
    global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – we use a moderate length that still allows many
    # optimisation iterations.
    # --------------------------------------------------------------
    n = 3000

    # --------------------------------------------------------------
    # Load any previous solution for seeding.
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # Stage 0 – build a large, diverse seed pool.
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # Fast surrogate ranking – keep the most promising candidates.
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 40
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – L‑BFGS refinement on the smooth‑max surrogate.
    # --------------------------------------------------------------
    temperature_schedule = [5, 10, 20, 40, 80, 150, 300, 600, 1200, 2000]
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        seq = seed_vec.copy()
        for temp in temperature_schedule:
            if time.time() > deadline:
                break
            seq_opt, val_opt = _l_bfgs_optimize(
                seq, n, deadline, max_iter=400, temperature=temp
            )
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[L‑BFGS] seed #{idx}, temp={temp} → new best {val_opt:.6f}")
            seq = seq_opt   # warm‑start the next temperature

    # --------------------------------------------------------------
    # Stage 2 – Mirror‑descent (exponentiated‑gradient) fine‑tuning.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=5_000_000,
            lr0=0.18,
            power=0.30,
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – Additive projected sub‑gradient descent (exact max).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Additive projected sub‑gradient optimisation ---")
        seq_add, val_add = _additive_subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=2_000_000,
            eta0=0.5,
            power=0.50,
        )
        if val_add < global_best_val:
            global_best_seq = seq_add.copy()
            global_best_val = val_add
            print(f"[Additive] new global best = {val_add:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – Simulated‑annealing polishing.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=2_000_000,
            init_temp_factor=0.04,
            min_temp=1e-12,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (for information only).
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Eval] evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Save for possible reuse.
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
