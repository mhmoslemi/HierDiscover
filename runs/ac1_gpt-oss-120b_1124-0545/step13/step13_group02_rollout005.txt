"""
Propose a non‑negative step function that minimises the official
``evaluate_sequence`` objective.

Algorithm Overview
------------------
* Initialise from the best known solution (``height_sequence_1``) or from a
  simple analytic seed.
* Compute the full self‑convolution once.
* Run a long simulated‑annealing (SA) phase where each move only touches a
  few coefficients.  The convolution is updated **incrementally** using the
  sparse change vector, so a single iteration costs O(n) instead of an FFT.
  This makes it possible to perform many millions of cheap moves within the
  1000 s time‑budget.
* The SA temperature is slowly cooled; worsening moves are sometimes
  accepted to escape local minima.
* The best sequence seen during the run is returned (as a plain list) and
  stored in the global ``height_sequence_1`` for the next run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on n points (ℓ₁‑normalised)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    """Discrete sine window a_i = sin(π·(i+1)/(n+1))^γ, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a decent starting shape when no previous solution exists.
    A mixture of a sine window and a low‑frequency random curve tends to give
    a low peak in the self‑convolution.
    """
    # sine window (γ≈2 gives a fairly flat autocorrelation)
    base = _sine_window(n, gamma=2.0)
    # add a tiny random low‑frequency perturbation
    t = np.arange(n, dtype=np.float64)
    curve = np.zeros(n, dtype=np.float64)
    for k in (1, 2, 3):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        curve += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    curve = np.maximum(curve, 0.0)
    # mix the two and renormalise
    mixed = 0.8 * base + 0.2 * curve
    mixed = np.maximum(mixed, 0.0)
    s = mixed.sum()
    if s <= 0:
        return base
    return mixed / s


# ----------------------------------------------------------------------
# Incremental convolution utilities (core of the fast SA)
# ----------------------------------------------------------------------
def _apply_delta_to_conv(conv: np.ndarray, p: np.ndarray,
                         delta: np.ndarray, idx: np.ndarray) -> None:
    """
    In‑place update of ``conv = p * p`` after ``p`` has been changed by
    ``delta`` (sparse).  ``idx`` contains the positions where delta ≠ 0.
    The update follows
        conv ← conv + 2·conv(p,Δ) + conv(Δ,Δ)
    where conv(·,·) denotes full linear convolution.
    """
    n = p.shape[0]

    # 2·conv(p,Δ) : for each modified index i add 2·Δ_i·p shifted by i
    for i in idx:
        d = delta[i]
        if d != 0.0:
            conv[i:i + n] += 2.0 * d * p

    # conv(Δ,Δ) : double loop over the sparse support (still O(m²))
    for a in idx:
        da = delta[a]
        if da == 0.0:
            continue
        for b in idx:
            db = delta[b]
            if db == 0.0:
                continue
            conv[a + b] += da * db


def _revert_delta_from_conv(conv: np.ndarray, p: np.ndarray,
                            delta: np.ndarray, idx: np.ndarray) -> None:
    """
    Undo the modification performed by ``_apply_delta_to_conv``.
    The same ``p`` (still the old one) and ``delta`` are used.
    """
    n = p.shape[0]

    for i in idx:
        d = delta[i]
        if d != 0.0:
            conv[i:i + n] -= 2.0 * d * p

    for a in idx:
        da = delta[a]
        if da == 0.0:
            continue
        for b in idx:
            db = delta[b]
            if db == 0.0:
                continue
            conv[a + b] -= da * db


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    the official ``evaluate_sequence`` objective within ``budget_s`` seconds.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # a tiny safety margin – we must return before the hard limit
    deadline = start_time + budget_s - 1.5

    # ------------------------------------------------------------------
    # 1️⃣ Initialise the distribution -------------------------------------------------
    # Use the solution from the previous run if present; otherwise fall back
    # to a handcrafted analytic shape.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        p = np.asarray(prev, dtype=np.float64).ravel()
        if p.size == 0:
            p = None
    else:
        p = None

    if p is None or p.ndim != 1 or p.size == 0:
        # no usable previous solution → build a fresh start
        n = 4000
        p = _initial_sequence(n, rng)
    else:
        n = p.shape[0]
        s = p.sum()
        if s <= 0.0:
            p = _uniform(n)
        else:
            p = p / s                 # ℓ₁‑normalise

    # ------------------------------------------------------------------
    # 2️⃣ Compute the initial full self‑convolution (once) ----------------------------
    # conv = p * p (full linear convolution, length = 2n‑1)
    # ------------------------------------------------------------------
    conv = np.convolve(p, p)          # numpy uses a fast implementation internally
    cur_max = float(conv.max())
    best_p = p.copy()
    best_conv = conv.copy()
    best_max = cur_max

    # print initial status
    init_obj = 2.0 * n * cur_max
    print(f"[Init] n={n}, initial objective = {init_obj:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Fast Simulated‑Annealing with incremental convolution updates ----------
    # ------------------------------------------------------------------
    # SA parameters – tuned to spend most of the time budget here.
    T = max(0.1 * cur_max, 1e-12)      # initial temperature
    cooling = 0.99997                  # exponential cooling factor
    min_temp = 1e-14
    it = 0
    report_every = 200_000            # progress reports

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # Choose a move type
        # --------------------------------------------------------------
        if rng.random() < 0.55:
            # ---- reshuffle a random subset (size 3 … 12) ----
            k_sub = rng.integers(3, 13)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = p[idx].sum()
            if total <= 0.0:
                continue
            # Dirichlet redistribution of the same total mass
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            delta = np.zeros_like(p)
            delta[idx] = new_alloc - p[idx]
        else:
            # ---- tiny transfer between two positions ----
            i, j = rng.choice(n, size=2, replace=False)
            # Limit the transferred amount to avoid negativity and keep it modest
            eps = rng.random() * min(p[i], 0.2)   # up‑to 20 % of the source mass
            if eps <= 0.0:
                continue
            delta = np.zeros_like(p)
            delta[i] = -eps
            delta[j] = eps

        # --------------------------------------------------------------
        # Apply the delta to the convolution (in‑place)
        # --------------------------------------------------------------
        nz = np.where(delta != 0.0)[0]   # indices where delta ≠ 0
        if nz.size == 0:
            continue

        _apply_delta_to_conv(conv, p, delta, nz)

        new_max = float(conv.max())
        delta_obj = new_max - cur_max

        # --------------------------------------------------------------
        # Acceptance test (standard Metropolis criterion)
        # --------------------------------------------------------------
        accept = False
        if delta_obj < -1e-15:                  # strict improvement
            accept = True
        else:
            if T > 0.0:
                prob = math.exp(-delta_obj / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            # ---- accept the move ----
            p += delta                     # update the distribution
            cur_max = new_max
            # keep track of the best ever seen
            if cur_max < best_max:
                best_max = cur_max
                best_p = p.copy()
                best_conv = conv.copy()
                # occasional reporting
                if it % report_every == 0:
                    elapsed = time.time() - start_time
                    obj = 2.0 * n * best_max
                    print(f"[SA] it={it:,d}  best_obj={obj:.6f}  time={elapsed:.1f}s")
        else:
            # ---- reject: revert the convolution update ----
            _revert_delta_from_conv(conv, p, delta, nz)

        # --------------------------------------------------------------
        # Temperature update
        # --------------------------------------------------------------
        T = max(T * cooling, min_temp)

        # Periodic progress report (lightweight)
        if it % report_every == 0:
            elapsed = time.time() - start_time
            cur_obj = 2.0 * n * cur_max
            best_obj = 2.0 * n * best_max
            print(f"[SA] it={it:,d}  cur_obj={cur_obj:.6f}  best_obj={best_obj:.6f}  T={T:.2e}  time={elapsed:.1f}s")

    # ------------------------------------------------------------------
    # 4️⃣ Final evaluation & bookkeeping
    # ------------------------------------------------------------------
    final_obj = 2.0 * n * best_max
    try:
        true_val = evaluate_sequence(best_p.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {final_obj:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse by the next run
    globals()["height_sequence_1"] = best_p.tolist()
    return best_p.tolist()
