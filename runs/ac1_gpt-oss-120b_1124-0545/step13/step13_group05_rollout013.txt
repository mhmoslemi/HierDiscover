"""
Optimisation of a non‑negative step function for the objective

    2·n·max(conv(a,a)) / (sum(a)²)      (evaluate_sequence)

The program follows a focused three‑phase strategy:

1. **Seed discovery** – a large pool of analytically‑motivated families is
   generated (uniform, triangular, power, beta, Gaussian, sine‑window,
   pure and offset exponentials).  An exhaustive grid over the exponential
   ratio  r ≈ 1−2/n (theoretically optimal for a pure exponential) is also
   performed with very fine resolution.  The best few seeds are kept.

2. **Intensive Adam optimisation** – the single best seed is refined for
   many (≈ 8·10⁵) iterations using a log‑softmax parametrisation.
   The smooth max is approximated by a temperature‑controlled
   log‑sum‑exp (β grows from 0.5 to 5000).  Adam’s learning‑rate schedule is
   aggressive at the start and decays with 1/√t.

3. **Fine‑grained polishing** – a long sub‑gradient descent on the true
   max‑convolution, followed by exponentiated‑gradient descent and a
   heavy simulated‑annealing sweep.  All stages respect the global 1000 s
   deadline; the best sequence found is returned as a plain list of floats.

The whole search runs in pure NumPy (FFT based convolutions) and uses only
standard Python libraries, so it complies with the execution limits.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector
    (∑seq = 1).  The true ``evaluate_sequence`` value equals this surrogate
    because the denominator is 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exponential_shape(n: int, r: float) -> np.ndarray:
    """Monotone exponential a_i ∝ r**i, ℓ₁‑normalised."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    return a / a.sum()


def _exponential_offset_shape(n: int, offset: int, r: float) -> np.ndarray:
    """Zero prefix of length *offset* followed by an exponential tail."""
    if offset >= n:
        return _uniform(n)
    tail_len = n - offset
    if r <= 0.0:
        tail = np.full(tail_len, 1.0 / tail_len, dtype=np.float64)
    else:
        i = np.arange(tail_len, dtype=np.float64)
        tail = np.power(r, i)
        tail /= tail.sum()
    a = np.concatenate([np.zeros(offset, dtype=np.float64), tail])
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()

# Optimisation primitives (log‑softmax Adam, sub‑gradient, exp‑grad, SA)
# ----------------------------------------------------------------------
def _logspace_adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam on the log‑softmax parametrisation with a log‑sum‑exp smoothing
    of the max.  Returns the best vector seen and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)            # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5                # start temperature for smoothing
    beta_inc = 0.5
    beta_max = 5000.0         # higher than the baseline

    lr0 = 1.0                 # initial learning rate in log‑space

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # --- smoothed max via log‑sum‑exp ---
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1: n - 1 + n]

        # Jacobian of log‑softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / math.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule (smooth max becomes sharper)
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Euclidean projection onto the simplex after each step.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # active sub‑gradient (only the overlapping part)
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Works directly on the simplex via the log‑softmax representation.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    init_temp_factor: float = 0.08,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Aggressive annealing that reshuffles mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.65:          # reshuffle a random subset
            k = rng.integers(3, 15)        # size 3 … 14
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                           # pairwise transfer (up to 4 % of source)
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.04)   # ≤4 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _collect_rich_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian(n, sigma_frac=0.22))
    seeds.append(_power(n, p=0.6))
    seeds.append(_power(n, p=1.2))
    seeds.append(_beta_shape(n, alpha=0.6, beta=1.4))
    seeds.append(_beta_shape(n, alpha=1.4, beta=0.6))

    # Fine exponential sweep (r = 1 - eps, eps ≈ 2/n)
    base_eps = 2.0 / n
    eps_factors = np.linspace(0.3, 2.0, 120)   # many points around the optimum
    for fac in eps_factors:
        eps = fac * base_eps
        r = max(0.0, 1.0 - eps)
        seeds.append(_exponential_shape(n, r))

    # Offset exponentials (we also try larger offsets)
    offsets = [0, int(0.01 * n), int(0.04 * n), int(0.08 * n), int(0.12 * n)]
    for off in offsets:
        L = n - off
        if L <= 0:
            continue
        base_eps_off = 2.0 / L
        for fac in np.linspace(0.5, 2.0, 20):
            eps = fac * base_eps_off
            r = max(0.0, 1.0 - eps)
            seeds.append(_exponential_offset_shape(n, off, r))

    # Random Dirichlet draws – provide high‑dimensional diversity
    for conc in (0.05, 0.2, 0.5, 1.0):
        for _ in range(3):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # Random exponential draws (discrete analogue of geometric)
    for _ in range(5):
        a = rng.exponential(scale=1.0, size=n)
        a /= a.sum()
        seeds.append(a)

    # Previous solution, if any
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Normalise everything explicitly (protect against tiny rounding errors)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted budget.  Returns the best
    sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    overall_deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose a length – 8000 is a good compromise between resolution
    # and per‑iteration cost; we also try a slightly larger length
    # because a longer vector can approximate the infinite‑support optimum
    # more closely.
    # ------------------------------------------------------------------
    n = 8000
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_rich_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} candidate seeds.")

    # Cheap surrogate ranking
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 10          # keep a small elite set
    elite = [vec for vec, _ in evals[:top_k]]

    # initialise global best
    global_best_seq = elite[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – focused Adam optimisation (all time budget goes to the
    # best elite seed, with a few quick “explorer” runs on the others)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 60.0:
        print("\n--- Adam optimisation (focused) ---")
        # reserve a fraction for the elite run
        elite_iters = int(min(800_000, remaining * 5500))
        # a few short runs on the remaining elites (to guard against a bad elite)
        short_iters = int(max(10_000, elite_iters // (5 * len(elite))))

        for idx, seed_vec in enumerate(elite[1:]):   # quick probes
            _logspace_adam_optimize(
                seed_vec, n, overall_deadline, max_iters=short_iters
            )   # result discarded – used only for side‑effects (none)

        # now the main long run
        seq_adam, val_adam = _logspace_adam_optimize(
            elite[0], n, overall_deadline, max_iters=elite_iters
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 30.0:
        max_iters_sub = int(min(2_500_000, remaining * 2500))
        print(f"\n--- Sub‑gradient projection (max_iters={max_iters_sub}) ---")
        seq_sub, val_sub = _subgradient_proj(
            global_best_seq, n, overall_deadline, max_iters=max_iters_sub
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[SubProj] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 20.0:
        max_iters_exp = int(min(1_200_000, remaining * 1500))
        print(f"\n--- Exponentiated‑gradient refinement (max_iters={max_iters_exp}) ---")
        seq_exp, val_exp = _exp_gradient_optimize(
            global_best_seq, n, overall_deadline, max_iters=max_iters_exp
        )
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] new global best = {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – aggressive simulated‑annealing polishing
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 10.0:
        max_iters_sa = int(min(3_000_000, remaining * 2000))
        print(f"\n--- Simulated annealing polishing (max_iters={max_iters_sa}) ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, overall_deadline, max_iters=max_iters_sa
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
