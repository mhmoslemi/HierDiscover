def _refine_sparse(mask, n, deadline, max_iters=3000):
    k = len(mask)
    seq = np.zeros(n, dtype=np.float64)
    seq[mask] = 1.0 / k
    pos_map = np.full(n, -1, dtype=np.int32)
    pos_map[mask] = np.arange(k)
    log_w = np.full(k, -np.log(k), dtype=np.float64)  # log of uniform weights
    best_seq = seq.copy()
    best_val = _objective_surrogate(seq)  # 2 n max conv
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute conv and max_idx
        conv = _conv_fft(seq, seq)
        max_idx = int(np.argmax(conv))
        # compute gradient on support
        g = np.zeros(k, dtype=np.float64)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=np.int32)
        rev_idx = max_idx - idx_i
        # for each i, check if i is a support index
        mask_i = pos_map[idx_i]
        valid = mask_i >= 0
        # gather seq[rev_idx] values
        rev_vals = seq[rev_idx]
        g[mask_i[valid]] = 2.0 * rev_vals[valid]
        # step size
        eta = 0.5 / math.sqrt(it)
        log_w = log_w - eta * g
        # compute softmax
        # stable
        max_l = np.max(log_w)
        e = np.exp(log_w - max_l)
        w = e / e.sum()
        # assign
        seq[:] = 0.0
        seq[mask] = w
        # evaluate surrogate
        cur_val = _objective_surrogate(seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = seq.copy()
        # optionally break early if conv_max < target? Not needed.

    return best_seq, best_val

"""
Propose a step‑function (list of non‑negative heights) that minimizes

    2·n·max(conv(a, a)) / (sum(a)²)

where conv is the full linear convolution (numpy.convolve).  The objective
is scale‑invariant, so we work on the ℓ₁‑simplex (∑a = 1).  The search
strategy combines

* a rich pool of analytic and random *seeds*,
* many *sparse* binary masks (random subsets of the indices) whose
  non‑zero weights are refined by mirror‑descent on the exact max‑convolution,
* continuation L‑BFGS optimisation of a smooth log‑sum‑exp approximation
  of the max‑convolution (β is increased up to 5000),
* a final exponentiated‑gradient polishing step.

All stages respect a total time budget (default 1000 s).  The best vector
found is stored in ``height_sequence_1`` and returned as a plain Python
list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for the L‑BFGS stage)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numerical helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = exact objective for an ℓ₁‑normalised vector.
    (i.e. 2·n·max(conv) because sum(seq)=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Analytic families (used for diverse seeds)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # Modified Bessel (numpy approximates)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a Dirichlet draw and assign largest/smallest alternatingly."""
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Sparse‑mask generation + mirror‑descent refinement
# ----------------------------------------------------------------------
def _refine_sparse(mask: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 2500) -> tuple[np.ndarray, float]:
    """
    Start from a uniform distribution on the given mask and run a
    mirror‑descent (exponentiated‑gradient) optimisation of the *exact*
    max‑convolution objective.  Returns the best sequence and its surrogate
    (2·n·max(conv)).
    """
    k = len(mask)
    a = np.zeros(n, dtype=np.float64)
    a[mask] = 1.0 / k

    # maps from index → position in the support (‑1 for non‑support)
    pos_of = np.full(n, -1, dtype=np.int32)
    pos_of[mask] = np.arange(k, dtype=np.int32)

    # logits for the support (softmax → probability on the support)
    log_w = np.full(k, -np.log(k), dtype=np.float64)

    best_a = a.copy()
    best_val = _objective_surrogate(best_a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient wrt the active lag (only on the support)
        g = np.zeros(k, dtype=np.float64)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=np.int32)
        rev_idx = max_idx - idx_i

        sup_i = pos_of[idx_i]                # –1 if not in support
        valid = sup_i >= 0
        if np.any(valid):
            # a[rev_idx] is zero unless rev_idx ∈ mask, which is already true
            # because seq is zero elsewhere.
            g[sup_i[valid]] = 2.0 * a[rev_idx[valid]]

        # mirror‑descent step (log‑space)
        eta = 0.5 / math.sqrt(it)
        log_w -= eta * g

        # softmax on the support
        m = np.max(log_w)
        e = np.exp(log_w - m)
        w = e / e.sum()
        a[:] = 0.0
        a[mask] = w

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """Optimise the smoothed max‑convolution objective using L‑BFGS‑B."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser not available – cannot run L‑BFGS.")

    if beta_schedule is None:
        # Aggressive schedule (reaches a very sharp approximation of max)
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0]

    # initialise in log‑softmax space (logits)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        b = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stays on simplex
            conv = _conv_fft(a, a)
            m = np.max(conv)

            # log‑sum‑exp smoothing
            shift = m
            exp_b = np.exp(b * (conv - shift))
            s = exp_b.sum()
            obj = 2.0 * n * (shift + np.log(s) / b)

            # gradient of the smooth objective
            w = exp_b / s                               # weights (softmax over lag)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                  # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through the softmax (logits → probabilities)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline hit during β={b}")
            break
        except Exception as e:                                   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={b}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient polish (exact max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (projected via softmax)."""
    a = start_seq.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a, best_val

    global_best = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    # ----- a) Run L‑BFGS on each top seed (smooth max) -----------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break

        print(f"\n--- L‑BFGS run on seed #{idx+1} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                          n,
                                          deadline,
                                          max_iter_per_beta=2500)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

        # ----- b) Optional mirror‑descent polishing (short) ---------
        remaining = deadline - time.time()
        if remaining > 3.0:
            max_exp = int(min(200_000, remaining * 0.07 * 1e5))
            seq_exp, val_exp = _exp_gradient_opt(global_best,
                                                 n,
                                                 deadline,
                                                 max_iters=max_exp)
            if val_exp < global_best_val:
                global_best = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Dedicated refinement of sparse‑mask candidates (new stage)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sparse‑mask refinement stage ---")
        # choose a few promising sparse seeds (the ones with best surrogates)
        sparse_candidates = [(vec, _objective_surrogate(vec))
                             for vec in seeds if np.count_nonzero(vec) > 0 and np.count_nonzero(vec) < n/2]
        sparse_candidates.sort(key=lambda x: x[1])
        # we will work on the best 6 sparse seeds
        for i, (cand, cand_val) in enumerate(sparse_candidates[:6]):
            if time.time() > deadline:
                break
            mask = np.flatnonzero(cand)
            print(f"\n[Sparse #{i+1}] k={len(mask)} initial surrogate={cand_val:.6f}")
            seq_ref, val_ref = _refine_sparse(mask, n, deadline, max_iters=3000)
            print(f"[Sparse refine] surrogate = {val_ref:.6f}")
            if val_ref < global_best_val:
                global_best = seq_ref.copy()
                global_best_val = val_ref
                print(f"[Update] new global best (sparse) = {global_best_val:.6f}")

            # polish the refined vector with a brief exponentiated‑gradient run
            remaining = deadline - time.time()
            if remaining > 2.0:
                max_exp = int(min(150_000, remaining * 0.06 * 1e5))
                seq_exp, val_exp = _exp_gradient_opt(global_best,
                                                     n,
                                                     deadline,
                                                     max_iters=max_exp)
                if val_exp < global_best_val:
                    global_best = seq_exp.copy()
                    global_best_val = val_exp
                    print(f"[ExpGrad after sparse] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation by the supplied black‑box
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                                   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best.tolist()
    return global_best.tolist()
