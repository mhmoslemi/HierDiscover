"""
Improved search for a step‑function minimising

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in two phases:

1. **Fast random sampling** – generate many
   “half‑dense” 0/1‑vectors (support size ≈ n/2, equal weights 1/S)
   and keep the best by the cheap surrogate
   `2·n·max(conv)`.

2. **Smoothed gradient descent** – starting from the best
   candidate, minimise a smooth approximation of the max,
   namely the log‑sum‑exp of the convolution.  The gradient of the
   smoothed objective is computed by a single FFT‑based convolution,
   then an Adam‑style update is performed and the vector is projected
   onto the probability simplex (∑a_i = 1, a_i ≥ 0).  The soft‑max
   temperature (γ) is increased gradually so that the optimisation
   first smoothes the landscape and later focuses on the true maximum.

Both phases respect the user‑supplied time budget (default 1000 s).
The final sequence is stored in the global variable ``height_sequence_1``,
so that the next run can reuse it as a warm start.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (complexity O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of *v* onto the probability simplex
    (non‑negative entries that sum to one).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _surrogate(a: np.ndarray, n: int) -> float:
    """Fast surrogate 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))

        if sample_iter % 1000 == 0:
            elapsed = time.time() - start_time
            print(f"[Sample] {sample_iter} candidates, best={best_surrogate:.6f}, elapsed={elapsed:.1f}s")

    # -----------------------------------------------------------------
    # Stage 2 – smoothed gradient descent (log‑sum‑exp surrogate)
    # -----------------------------------------------------------------
    a = best_seq.copy()
    a = _proj_simplex(a)                        # guarantee simplex feasibility

    # Adam optimiser state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05                                   # base learning rate

    # Soft‑max temperature (γ); start low, increase gradually
    gamma = 5.0
    gamma_inc = 5.0
    epoch_len = 5000
    it = 0
    max_iters = 200_000                         # safety cap

    print("\n--- Smoothed gradient descent (log‑sum‑exp) ---")
    while time.time() < deadline - 1.0 and it < max_iters:
        # 1) current convolution and surrogate
        conv = _conv_fft(a, a)                  # length 2n‑1
        cur_surrogate = 2.0 * n * np.max(conv)

        # 2) keep the best ever seen
        if cur_surrogate < best_surrogate:
            best_surrogate = cur_surrogate
            best_seq = a.copy()
            print(f"[GD] iter={it:6d} new best surrogate = {cur_surrogate:.6f}")

        # 3) build the smooth max weights w_k = exp(γ·conv_k) / Σ exp(...)
        maxc = np.max(conv)
        exp_term = np.exp(gamma * (conv - maxc))   # stabilised exponentiation
        w = exp_term / np.sum(exp_term)           # w sums to 1

        # 4) gradient of Σ w_k·conv_k  :  g_i = 2·Σ_t a_t·w_{i+t}
        a_rev = a[::-1]                           # reversed a
        conv_rev = _conv_fft(a_rev, w)            # length 3n‑2
        start_idx = n - 1
        grad = 2.0 * conv_rev[start_idx:start_idx + n]

        # 5) Adam update + simplex projection
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _proj_simplex(a)

        # 6) temperature schedule
        if it % epoch_len == 0:
            gamma = min(gamma + gamma_inc, 200.0)
            # Reset Adam moments to avoid stale statistics after a sharp change
            m.fill(0.0)
            v.fill(0.0)
            print(f"[GD] iter={it} increased γ to {gamma:.1f}")

        # occasional progress report
        if it % 2000 == 0:
            elapsed = time.time() - start_time
            print(f"[GD] iter={it:6d} best={best_surrogate:.6f} γ={gamma:.1f} elapsed={elapsed:.1f}s")

        # stop early if we are about to exceed the budget
        if time.time() > deadline - 1.0:
            break

    # -----------------------------------------------------------------
    # Final (expensive) evaluation with the official scorer
    # -----------------------------------------------------------------
    final_seq = best_seq.tolist()
    true_val = None
    try:
        true_val = evaluate_sequence(final_seq)
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_surrogate:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = final_seq
    return final_seq
