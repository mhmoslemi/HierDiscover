def mirror_subgradient_optimize(start_seq, n, deadline, max_iters):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)
    lr0 = 0.5
    power = 0.25
    it = 0
    while it < max_iters and time.time() < deadline:
        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # gradient of max conv
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        lr = lr0 / ((it + 1) ** power)
        a = a * np.exp(-lr * g)
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = _uniform(n)
        it += 1
    return best_seq, best_val
```

"""
Search for a low‑max‑auto‑convolution step‑function.

The algorithm proceeds in four stages:

1. **Rich seed generation** – deterministic analytic families
   (uniform, triangular, cosine, beta, Gaussian, exponential,
   power‑law, sparse random, mixtures) are instantiated with a wide
   range of parameters and with random Dirichlet draws.  All vectors are
   ℓ₁‑normalised (∑aᵢ = 1).  The cheapest ``top_k`` seeds according to the
   cheap surrogate  ``2·n·max(conv(a,a))`` are kept.

2. **Mirror‑descent sub‑gradient optimiser** – on the exact
   max‑convolution we run an exponentiated‑gradient (mirror) descent.
   The update is ``a ← a·exp(−η·g)`` with ``g`` the sub‑gradient of the
   active convolution entry and a step‑size ``η = η₀/(t**p)``.
   Positivity is preserved and the vector is projected back onto the
   simplex after each step.

3. **Simulated‑annealing polishing** – a lightweight SA that moves mass
   between a few randomly chosen coordinates (swap, redistribution,
   pairwise transfer).  The temperature is initialised from the current
   surrogate value and cooled exponentially.  Every accepted move is
   projected onto the simplex.

4. **Final evaluation** – the resulting sequence is handed to the
   official ``evaluate_sequence`` function, stored globally as
   ``height_sequence_1`` and returned as a plain ``list`` of ``float``.

The whole search respects the global time budget (default 1000 s) and
makes no use of filesystem or network I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large n)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate_obj(seq: np.ndarray) -> float:
    """
    Fast surrogate for the official objective (∑a = 1):
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random vector supported on a small subset of coordinates.
    Number of support points is chosen uniformly in [2, n//10].
    """
    k = rng.integers(2, max(3, n // 10) + 1)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k) + 0.01
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_two_shapes(a: np.ndarray, b: np.ndarray, mix: float) -> np.ndarray:
    """Linear mixture of two normalised shapes."""
    mixed = mix * a + (1.0 - mix) * b
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 2_000_000,
                                 lr0: float = 0.5,
                                 power: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the exact max.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / ((it + 1) ** power)
        a = a * np.exp(-lr * g)

        # re‑normalise onto the simplex
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = _uniform(n)

        it += 1

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 3_000_000,
                                 init_temp_factor: float = 0.07,
                                 min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Lightweight SA that moves mass between a few random coordinates.
    Keeps the best sequence seen.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate_obj(cur)
    best_seq = cur.copy()
    best_val = cur_val

    # start temperature proportional to the current objective
    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        # exponential cooling
        T = max(T * 0.9996, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:
            # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            # redistribute mass on a small random subset
            k = rng.integers(2, 15)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total == 0.0:
                it += 1
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            # pairwise transfer of mass
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total == 0.0:
                it += 1
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _surrogate_obj(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 250_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))

    # Gaussian windows
    for sigma in np.linspace(0.04, 0.30, 7):
        seeds.append(_gaussian(n, sigma))

    # Power‑law tails
    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power_law(n, p))

    # Exponential decay
    for lam in np.linspace(0.0002, 0.012, 9):
        seeds.append(_exponential_decay(n, lam))

    # Beta shapes – extremely stretched parameters
    alphas = np.logspace(np.log10(0.01), np.log10(3.0), 12)
    betas = np.logspace(np.log10(0.01), np.log10(3.0), 12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Sparse random vectors
    for _ in range(12):
        seeds.append(_sparse_random(n, rng))

    # Dirichlet draws with a very wide concentration range
    concs = np.logspace(-6, 0.5, 15)   # 1e‑6 … ~3.2
    for conc in concs:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random mixtures of two beta shapes (can produce very irregular patterns)
    for _ in range(10):
        a1, b1 = rng.uniform(0.01, 3.0), rng.uniform(0.01, 3.0)
        a2, b2 = rng.uniform(0.01, 3.0), rng.uniform(0.01, 3.0)
        beta1 = _beta_shape(n, a1, b1)
        beta2 = _beta_shape(n, a2, b2)
        mix = rng.random()
        seeds.append(_mix_two_shapes(beta1, beta2, mix))

    # Previous solution, if any (helps continuity across runs)
    if previous is not None and previous.shape == (n,):
        avg = previous.astype(np.float64)
        if avg.sum() > 0.0:
            seeds.append(avg / avg.sum())

    return seeds

    """
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # Problem size – 4000 gives a good trade‑off between richness and speed
    n = 4000

    # Load previous solution (if any) and adapt it to the current length
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")

        try:

            if previous.size != n:
                if previous.size > n:
                    previous = previous[:n]
                else:
                    previous = np.pad(previous,
                                      (0, n - previous.size),

        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length={n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _surrogate_obj(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 30
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # Initialise the current global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _surrogate_obj(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Mirror‑descent optimisation (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # Allocate roughly 60 % of the remaining budget to mirror descent
        mirror_deadline = time.time() + 0.60 * (deadline - time.time())
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(

            max_iters=2_000_000, lr0=0.5, power=0.25
        )
        if val_mirror < global_best_val:
            global_best_seq, global_best_val = seq_mirror, val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Simulated annealing polishing (the part omitted in the
    #           previous run because of time pressure)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline,
            max_iters=3_000_000,
            init_temp_factor=0.07
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (for information only)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:   # pragma: no cover

    )

    return global_best_seq.tolist()
