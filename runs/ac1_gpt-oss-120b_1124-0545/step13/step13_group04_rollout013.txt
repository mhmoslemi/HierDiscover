def grad_softmax(a, gamma):
    conv = _conv_fft(a,a) # length 2n-1

    # compute softmax weights
    max_c = conv.max()
    exp_term = np.exp(gamma * (conv - max_c))
    w = exp_term / exp_term.sum()

    # convolution of w (length 2n-1) with a_rev (length n)
    a_rev = a[::-1]
    N = _next_pow2(len(w) + len(a_rev) - 1)
    fw = np.fft.rfft(w, n=N)
    fa_rev = np.fft.rfft(a_rev, n=N)
    prod = fw * fa_rev
    conv_wa = np.fft.irfft(prod, n=N)[:len(w)+len(a_rev)-1]
    # slice results
    g = 2 * conv_wa[n-1: n-1 + len(a)]
    return g

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    n = 5000   # choose length maybe.

    best_seq = None
    best_val = np.inf

    # try previous solution if length matches
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list):
        arr = np.asarray(prev, dtype=np.float64)
        if arr.size == n and np.sum(arr) > 0:
            val = _surrogate(arr / np.sum(arr), n)
            best_seq = arr / np.sum(arr)
            best_val = val
            print(f"[Init] using previous solution surrogate={val:.6f}")

    # baseline uniform
    uniform = np.full(n, 1.0/n)
    val_uni = _surrogate(uniform, n)
    if val_uni < best_val:
        best_seq = uniform
        best_val = val_uni
        print(f"[Init] uniform surrogate={val_uni:.6f}")

    # Stage 1: random support candidates (binary with random weights)
    sample_iter = 0
    while time.time() < deadline - 10.0:
        # sample support size
        # Choose fraction uniformly between 0.2 and 0.8
        frac = rng.uniform(0.2, 0.8)
        S = max(1, int(frac * n))
        indices = rng.choice(n, size=S, replace=False)
        # assign random positive weights using exponential (scale 1)
        vals = rng.exponential(scale=1.0, size=S)
        a = np.zeros(n, dtype=np.float64)
        a[indices] = vals
        a /= a.sum()   # ℓ1 normalised
        val = _surrogate(a, n)
        sample_iter += 1
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sample] iter={sample_iter} new best surrogate={val:.6f}")
        if sample_iter % 1000 == 0:
            print(f"[Sample] {sample_iter} candidates, best={best_val:.6f}, elapsed={time.time()-start:.1f}s")

    # Stage 2: projected gradient descent (softmax) from best_seq
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 2.0:
        max_iters = int(min(200_000, remaining * 400))   # roughly 400 iter per second
        print(f"\n--- Gradient descent start: max_iters={max_iters} ---")
        a = best_seq.copy()
        # initialize Adam moments
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        best_local_seq = a.copy()
        best_local_val = best_val

        for it in range(1, max_iters+1):
            # compute convolution and gradient
            conv = _conv_fft(a, a)
            cur_val = 2.0 * n * np.max(conv)
            if cur_val < best_local_val:
                best_local_val = cur_val
                best_local_seq = a.copy()
                if it % 5000 == 0:
                    print(f"[GD] iter={it} improved surrogate={cur_val:.6f}")

            # softmax weight
            gamma = min(15.0, 0.001 * it)   # increase gradually
            max_c = conv.max()
            # using softmax to spread gradient; if gamma very low treat as max gradient to save compute
            if gamma < 0.5:
                # use true max subgradient for speed
                k = int(np.argmax(conv))
                g = np.zeros_like(a)
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                idx_i = np.arange(i_start, i_end + 1, dtype=int)
                rev_idx = k - idx_i
                g[idx_i] = 2.0 * a[rev_idx]
            else:

                w = exp_term / exp_term.sum()

                a_rev = a[::-1]
                L = len(w) + len(a_rev) - 1

                grad_full = np.fft.irfft(fw * fa_rev, n=N)[:L]
                g = 2.0 * grad_full[n-1 : n-1 + n]

            # Adam update
            m = beta1 * m + (1 - beta1) * g

        print("evaluate_sequence raised:", e)

"""
Improved search for a step‑function minimizing

    2·n·max(conv(a,a)) / (sum(a)²)

Approach
--------
1. Initialise a candidate:
   * reuse a previously stored sequence if it matches the current length,
   * otherwise start from the uniform distribution.
2. Randomised exploration:
   * draw many sparse supports, give the support a random (exponential) weight
     vector and normalise it to ℓ₁‑norm = 1.
   * keep the best candidate according to the cheap surrogate
     ``2·n·max(conv)``.
3. Long projected sub‑gradient descent (no external libraries):
   * at each iteration compute the full self‑convolution,
     locate the index where the maximum occurs and take the exact sub‑gradient
     of that entry (g_i = 2·a_{k‑i} for the active indices).
   * an Adam‑style adaptive step‑size and Euclidean projection onto the
     probability simplex are used.
   * the loop runs for as many iterations as the remaining time budget permits.
4. Final call to ``evaluate_sequence`` stores the best sequence globally for
   later runs.

The routine respects the supplied time budget (default 1000 s) and never
exceeds it.  All arithmetic is performed in double precision; negative or
NaN values are eliminated by ``evaluate_sequence`` itself.

"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, surrogate)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (∑v = 1, v ≥ 0).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries equal – the simplex point is the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _surrogate(a: np.ndarray, n: int) -> float:
    """Fast surrogate 2·n·max(conv(a,a)) for an ℓ₁‑normalised vector."""
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the official ``evaluate_sequence`` objective within the allotted time.
    Returns the best sequence as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – a compromise between solution quality and speed
    # ------------------------------------------------------------------
    n = 8000                          # length of the vector

    best_seq = None
    best_surrogate = float("inf")

    # ------------------------------------------------------------------
    # 0) Re‑use a previously stored solution if it matches the current n
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list):
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and np.sum(arr) > 0.0:
                val = _surrogate(arr / np.sum(arr), n)
                best_seq = arr / np.sum(arr)
                best_surrogate = val
                print(f"[Init] previous solution surrogate = {val:.6f}")
        except Exception:
            pass  # just ignore if something goes wrong

    # ------------------------------------------------------------------
    # 1) Uniform distribution (baseline)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    val_uni = _surrogate(uniform, n)
    if val_uni < best_surrogate:

        print(f"[Init] uniform surrogate = {val_uni:.6f}")

    # ------------------------------------------------------------------
    # 2) Random sparse candidates with random (exponential) weights
    # ------------------------------------------------------------------
    sample_iter = 0
    # run this phase until a few seconds are left for the refinement stage
    while time.time() < deadline - 30.0:
        # support size between 20 % and 80 % of n (inclusive)
        frac = rng.uniform(0.20, 0.80)
        S = max(1, int(frac * n))

        # give each selected index a random positive weight (exponential)

        a = np.zeros(n, dtype=np.float64)

        a /= a.sum()                     # ℓ₁‑normalise by construction
        val = _surrogate(a, n)
        sample_iter += 1
        if val < best_surrogate:
            best_surrogate = val

        # occasional progress report
        if sample_iter % 2000 == 0:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 3.0:

            # Start with the exact sub‑gradient of the maximal entry.
            # After a few thousand iterations we soften the max using a
            # soft‑max (log‑sum‑exp) which provides a smoother direction.
            if it < 5000:
                # exact sub‑gradient (cheapest)
                k = int(np.argmax(conv))
                g = np.zeros_like(a)
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                idx_i = np.arange(i_start, i_end + 1, dtype=int)
                rev_idx = k - idx_i
                g[idx_i] = 2.0 * a[rev_idx]
            else:

                gamma = min(12.0, 0.001 * it)
                max_c = conv.max()
                # stable soft‑max

                w = exp_term / exp_term.sum()
                # gradient = 2·(w ⋆ a) where ⋆ is the correlation defined above

                L = len(w) + len(a_rev) - 1

                corr_full = np.fft.irfft(fw * fa_rev, n=N)[:L]

            # ------- Adam update -------
            lr = 0.5 / math.sqrt(it)          # base learning‑rate schedule

            m_hat = m / (1 - beta1 ** it)
            v_hat = v_vec / (1 - beta2 ** it)

                break

        print("Warning: evaluate_sequence raised an exception:", exc)

    return best_seq.tolist()
