def _parametric_optimization(n, deadline, rng):
    K = 12 maybe (so 22 parameters)
    # Precompute basis
    i = np.arange(n)
    freqs = np.arange(1, K)  # exclude zero freq (constant)
    cos_mat = np.cos(2*np.pi*freqs[:, None]*i / n)
    sin_mat = np.sin(2*np.pi*freqs[:, None]*i / n)
    # Initialize coefficients: small random (normal 0, std=0.01)
    coeffs = rng.normal(scale=0.01, size=2*(K-1))
    # Adam parameters
    m = np.zeros_like(coeffs)
    v = np.zeros_like(coeffs)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t = 0
    best_val = np.inf
    best_a = None

    temperature_schedule = [5,10,20,40,80,160,320,640,1280,2560,5120,10240,20480,40960]

    for temp in temperature_schedule:
        # for each temperature, run until time left small or fixed # its
        max_iters_for_temp = 2000 # may adjust based on n and time
        for it in range(max_iters_for_temp):
            if time.time() > deadline:
                return best_a, best_val
            # compute a
            # compute s = coeffs[0:K-1] @ cos_mat + coeffs[K-1:] @ sin_mat
            # shape: coeff_cos shape = (K-1,)
            c = coeffs[:K-1]
            d = coeffs[K-1:]
            s = c @ cos_mat + d @ sin_mat   # shape (n,)
            # subtract max for stability possibly
            # Not needed because exponentiate with shift is okay
            s = s - np.max(s)  # shift
            a_raw = np.exp(s)
            a = a_raw / a_raw.sum()
            # compute smooth max and gradient
            smooth, grad_a = _smooth_max_and_grad(a, temp)
            loss = 2.0 * n * smooth
            # compute gradient w.r.t params
            dot = np.dot(a, grad_a)
            ds = a * (grad_a - dot)  # shape (n,)
            # gradient w.r.t coeffs
            grad_c = ds @ cos_mat.T  # shape (K-1,)
            grad_d = ds @ sin_mat.T
            grad = np.concatenate([grad_c, grad_d])
            # Adam update (minimisation)
            t += 1
            m = beta1*m + (1-beta1)*grad
            v = beta2*v + (1-beta2)*(grad*grad)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            lr = 0.05 / (np.sqrt(v_hat) + eps_adam)  # maybe adjust
            coeffs = coeffs - lr * m_hat
            # maintain possible bound on coefficients? maybe clip to reasonable range, e.g., [-5,5]
            # but let them grow gradually.
            # evaluate actual objective (hard max)
            conv = _conv_fft(a, a)
            cur_val = 2.0 * n * np.max(conv)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                # print progress
                print(f"[Param] temp={temp} it={it} new best {best_val:.6f}")
    return best_a, best_val

   a = start_seq.copy()
   best_seq = a.copy()
   best_val = _objective(a)
   it = 1
   while it <= max_iters:
        if time.time() > deadline:
            break
        # compute conv
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:,d} best={cur_val:.6f}")
        # compute subgradient g
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        # g_i = 2 * a_{rev_idx}
        g[idx_i] = 2.0 * a[rev_idx]
        # multiplicative update
        lr = lr0 / (it ** power)
        a = a * np.exp(-lr * g)
        # symmetry enforcement
        a = (a + a[::-1]) / 2.0
        # renorm
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s
        it += 1
   return best_seq, best_val

def _simulated_annealing_optimize(start_seq, n, deadline, max_iters=2_000_000, init_temp_factor=0.04, min_temp=1e-12):
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    T = max(init_temp_factor * cur_val, 1e-12)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.9995, min_temp)
        # pick move type
        move_type = rng.integers(0, 3)
        if move_type == 0:
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]
        # enforce symmetry
        new = (new + new[::-1]) / 2.0
        # re-normalize
        s = new.sum()
        if s > 0:
            new = new / s
        else:
            new = _uniform(n)
        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 200_000 == 0:
                    print(f"[SA] it={it:,d} best={new_val:.6f}")
    return best_seq, best_val

"""
Improved optimiser for the step‑function challenge.

The algorithm combines several ideas that were not all present in the
previous solution:

1. **Rich seed pool** – in addition to the original analytic families we
   add Hann / Hamming / Blackman windows and a Gaussian bump.

2. **Symmetric low‑frequency Fourier parameterisation** – a vector is
   represented by a small set of cosine basis functions.
   The coefficients are optimised with Adam on the *smooth‑max*
   surrogate (log‑sum‑exp) while staying on the simplex via the
   soft‑max.  Symmetry is enforced automatically because only cosine
   terms are used.

3. **Mirror‑descent (exponentiated‑gradient) with symmetry projection** –
   a fast sub‑gradient method that reduces the maximal self‑convolution.

4. **Light simulated‑annealing polishing** – again with a symmetry step.

All stages respect the global time budget (default 1000 s) and the best
sequence found is stored in the global variable ``height_sequence_1``.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# FFT‑based convolution (real‑valued, full length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Soft‑max (used to stay on the simplex)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Surrogate objective (fast, identical ordering to official one)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – works on ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient w.r.t. the vector a
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """Return (smooth_max, grad_a) for the log‑sum‑exp surrogate."""
    conv = _conv_fft(a, a)
    shift = np.max(conv)                     # stable shift
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum
    smooth = shift + np.log(w_sum) / beta

    # gradient of Σ w_k·(a*a)_k w.r.t a  →  2·(rev(a) ⊛ w)
    rev_a = a[::-1]
    grad = 2.0 * _conv_fft(rev_a, weights)
    grad = grad[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = n - i
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _gaussian_window(n: int, sigma: float | None = None) -> np.ndarray:
    if sigma is None:
        sigma = n / 6.0
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed collection (includes the extra windows)
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_linear_decreasing(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_gaussian_window(n))

    # power‑law family
    for p in np.linspace(0.0, 4.0, 17):
        seeds.append(_power(n, p))

    # β‑shape family
    alphas = np.concatenate((np.linspace(0.01, 0.2, 6), np.linspace(0.2, 1.0, 6)))
    betas = np.concatenate((np.linspace(1.0, 5.0, 5), np.linspace(5.0, 10.0, 5)))
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # exponential decay
    for lam in np.linspace(0.001, 0.020, 10):
        seeds.append(_exponential_decay(n, lam))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Dirichlet draws (spiky candidates)
    for conc in (0.01, 0.05, 0.1):
        for _ in range(3):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)
            seeds.append(np.sort(raw)[::-1])  # descending
            seeds.append(np.sort(raw))        # ascending

    # previous solution if it matches the current length
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# L‑BFGS refinement (re‑used from the previous solution)
# ----------------------------------------------------------------------
def _l_bfgs_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iter: int = 300,
    temperature: float = 100.0,
) -> tuple[np.ndarray, float]:
    eps = 1e-30
    v = np.log(start_seq + eps)  # logits → simplex via softmax

    def fun(vv):
        a = _softmax(vv)
        smooth, grad_a = _smooth_max_and_grad(a, temperature)
        loss = 2.0 * n * smooth
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)  # chain rule through soft‑max
        return loss, (2.0 * n * grad_v).astype(np.float64)

    try:
        res = scipy.optimize.minimize(
            fun,
            v,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
    except Exception:
        return start_seq, _objective(start_seq)

    best_a = _softmax(res.x)
    return best_a, _objective(best_a)


# ----------------------------------------------------------------------
# Fourier‑cosine parametrisation optimiser (low‑frequency, symmetric)
# ----------------------------------------------------------------------
def _fourier_optimize(
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters_per_temp: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Optimises a symmetric vector using a small cosine basis.
    The representation is a = softmax(B @ p) where B contains
    a constant column and a few cosine terms (so the resulting a
    is automatically symmetric).  Adam is used on the smooth‑max
    surrogate with an increasingly sharp temperature schedule.
    """
    # number of cosine frequencies (including constant = 0)
    K = 12  # feel free to enlarge; 12 gives 0..11 frequencies
    i = np.arange(n, dtype=np.float64)

    # Build basis matrix B (n × K)
    B = np.empty((n, K), dtype=np.float64)
    B[:, 0] = 1.0  # constant term (has no effect after softmax)
    for col, k in enumerate(range(1, K), start=1):
        B[:, col] = np.cos(2.0 * np.pi * k * i / n)

    # Initialise parameters (small random values)
    p = rng.normal(scale=0.01, size=K)

    # Adam state
    m = np.zeros_like(p)
    v_vec = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = None
    best_val = np.inf

    # Temperature schedule – slowly sharpen the surrogate
    temperature_schedule = [
        5,
        10,
        20,
        40,
        80,
        160,
        320,
        640,
        1280,
        2560,
        5120,
        10240,
        20480,
        40960,
    ]

    for temp in temperature_schedule:
        # allocate a modest amount of iterations per temperature,
        # always respecting the global deadline
        remaining = deadline - time.time()
        est_iter_possible = int(max(1, remaining / 0.001 / len(temperature_schedule)))
        max_iters = min(max_iters_per_temp, est_iter_possible)

        for _ in range(max_iters):
            if time.time() > deadline:
                return best_a if best_a is not None else _uniform(n), best_val

            # logits → probabilities
            logits = B @ p
            a = _softmax(logits)

            # smooth‑max and gradient w.r.t. a
            smooth, grad_a = _smooth_max_and_grad(a, temp)
            loss = 2.0 * n * smooth

            # gradient w.r.t. logits (softmax chain rule)
            dot = np.dot(a, grad_a)
            grad_v = a * (grad_a - dot)  # shape (n,)

            # gradient w.r.t. parameters
            grad_p = B.T @ grad_v

            # Adam update (minimise)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_p
            v_vec = beta2 * v_vec + (1.0 - beta2) * (grad_p * grad_p)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v_vec / (1.0 - beta2 ** t)

            lr = 0.02 / (np.sqrt(v_hat) + eps)  # modest learning rate
            p -= lr * m_hat

            # evaluate the *exact* objective (hard max)
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Fourier] temp={temp} new best {cur_val:.6f}")

    if best_a is None:
        best_a = _uniform(n)
        best_val = _objective(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) with symmetry projection
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
    lr0: float = 0.18,
    power: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated‑gradient) updates on the simplex.
    After each update the vector is symmetrised (a ← (a + a[::-1]) / 2)
    and re‑normalised.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 1
    while it <= max_iters:
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:,d} best={cur_val:.6f}")

        # sub‑gradient for the active anti‑diagonal
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = lr0 / (it ** power)
        a = a * np.exp(-lr * g)

        # enforce symmetry and re‑normalise onto the simplex
        a = (a + a[::-1]) / 2.0
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light Simulated Annealing with symmetry projection
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:  # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:  # redistribute a random subset
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:  # split‑transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        # symmetry projection + normalisation
        new = (new + new[::-1]) / 2.0
        s = new.sum()
        if s > 0.0:
            new = new / s
        else:
            new = _uniform(n)

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 200_000 == 0:
                    print(f"[SA] it={it:,d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main optimiser entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the supplied time budget.
    Returns the best sequence (list of floats) and stores it as the global
    name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 5000

    # --------------------------------------------------------------
    # Load any previous solution for seeding.
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # Stage 0 – create a big and diverse seed pool.
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 30
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – L‑BFGS refinement on the smooth‑max surrogate.
    # --------------------------------------------------------------
    temperature_schedule = [5, 10, 20, 40, 80, 150, 300, 600, 1200, 2400, 4800]
    for idx, seed_vec in enumerate(best_seeds[:10], 1):  # focus on the very best seeds
        if time.time() > deadline:
            break
        seq = seed_vec.copy()
        for temp in temperature_schedule:
            if time.time() > deadline:
                break
            seq_opt, val_opt = _l_bfgs_optimize(seq, n, deadline,
                                                max_iter=300,
                                                temperature=temp)
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[L‑BFGS] seed #{idx}, temp={temp} → new best {val_opt:.6f}")
            seq = seq_opt   # warm‑start the next temperature

    # --------------------------------------------------------------
    # Stage 2 – Fourier‑cosine parametric optimisation (symmetric).
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n--- Fourier‑cosine parametric optimisation ---")
        # try a few random restarts
        best_param_seq = None
        best_param_val = np.inf
        for trial in range(5):
            if time.time() > deadline:
                break
            a_seq, val = _fourier_optimize(n, deadline, rng)
            if val < best_param_val:
                best_param_val = val
                best_param_seq = a_seq
                print(f"[Fourier] trial {trial + 1} → new best {val:.6f}")
        if best_param_seq is not None and best_param_val < global_best_val:
            global_best_seq = best_param_seq.copy()
            global_best_val = best_param_val
            print(f"[Fourier] updated global best = {best_param_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – Mirror‑descent refinement (still symmetric).
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        print("\n--- Mirror‑descent refinement ---")
        seq_m, val_m = _mirror_subgradient_optimize(
            global_best_seq, n, deadline,
            max_iters=1_200_000,
            lr0=0.2,
            power=0.30,
        )
        if val_m < global_best_val:
            global_best_seq = seq_m.copy()
            global_best_val = val_m
            print(f"[Mirror] new global best = {val_m:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – Simulated‑annealing polishing (still symmetric).
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=2_000_000,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (only for information).
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Eval] evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for possible reuse by a later run.
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
