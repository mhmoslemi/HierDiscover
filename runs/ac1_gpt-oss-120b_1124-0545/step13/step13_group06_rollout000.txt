Implementation of projection onto simplex:

def _simplex_projection(v):
    """Project vector v onto the probability simplex (sum to 1, nonnegative). Return projected vector."""
    # Implementation from Duchi et al.
    if v.sum() == 1 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)

"""
Multi‑stage optimiser for the step‑function attack.

Algorithm
---------
1. Build a diverse pool of ℓ₁‑normalised seeds:
   - analytic families (uniform, linear, power‑law, beta, exponential,
     low‑frequency sinusoidal curves).
   - random Dirichlet draws with a wide range of concentration.
   - random sparse “spiky’’ vectors.
   - the previously best solution (if any).

2. Keep the ~30 best seeds according to the fast surrogate
   `2·n·max(conv(a,a))` (identical ordering to the official objective).

3. Refine each top seed with L‑BFGS on a smooth‑max (log‑sum‑exp) surrogate.
   The temperature is increased gradually (5 → … → 1920) and each
   temperature is given up to 300 L‑BFGS iterations.

4. Fine‑tune the current global best with a projected sub‑gradient
   descent on the *exact* max‑convolution objective.
   The update is `a ← Π_simplex(a - η·g)` where `g` is the sub‑gradient
   and `Π_simplex` projects onto the probability simplex.
   The learning‑rate decays as `η₀ / t**0.5`.

5. Polish the result with lightweight simulated‑annealing (swap,
   Dirichlet‑redistribute, split‑transfer) while staying on the simplex.

All stages respect the global time budget (default 1000 s).  The final
vector is stored as the global name ``height_sequence_1`` and returned
as a plain list of floats.
"""

import time
import operator
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Basic numerical helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real‑valued inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = n - i
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_sparse_seed(n: int, rng: np.random.Generator, k: int = 12) -> np.ndarray:
    """k non‑zero entries at random positions, Dirichlet‑distributed masses."""
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


# ----------------------------------------------------------------------
# Surrogate objective (identical ordering to official one)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """Return (smooth_max, grad_a) for the log‑sum‑exp surrogate."""
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum
    smooth = shift + np.log(w_sum) / beta

    # gradient = 2 * (rev(a) ⊛ weights)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad_a = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad_a


def _lbfgs_fun(v: np.ndarray, n: int, temperature: float) -> tuple[float, np.ndarray]:
    """Objective and gradient for the L‑BFGS optimisation (log‑sum‑exp)."""
    a = _softmax(v)
    smooth, grad_a = _smooth_max_and_grad(a, temperature)
    loss = 2.0 * n * smooth
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)          # chain rule through softmax
    return loss, (2.0 * n * grad_v).astype(np.float64)


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iter: int = 300,
                     temperature: float = 100.0) -> tuple[np.ndarray, float]:
    """L‑BFGS on the smooth‑max surrogate."""
    eps = 1e-30
    v = np.log(start_seq + eps)               # logits → simplex via softmax
    try:
        res = scipy.optimize.minimize(
            _lbfgs_fun,
            v,
            args=(n, temperature),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
    except Exception:
        return start_seq, _objective(start_seq)

    best_a = _softmax(res.x)
    return best_a, _objective(best_a)


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _projected_subgradient_optimize(start_seq: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   max_iters: int = 1_500_000,
                                   lr0: float = 0.5,
                                   power: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Multiplicative projected sub‑gradient descent.
    a ← Π_simplex(a - η·g)   with g the sub‑gradient of max(conv(a,a)).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters:
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        max_idx = int(np.argmax(conv))

        # active set (indices achieving the max, with a tiny tolerance)
        tol = 1e-14
        active = np.where(np.abs(conv - max_val) <= tol)[0]

        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]

        it += 1
        eta = lr0 / (it ** power)

        a_tmp = a - eta * g
        a = _simplex_projection(a_tmp)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[PGD] it={it:,d} best={cur_val:.6f}")

        # occasional random jitter to help escaping flat regions
        if it % 250_000 == 0:
            jitter = np.random.default_rng().normal(scale=1e-4, size=n)
            a = _simplex_projection(a + jitter)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Lightweight simulated annealing for final polishing
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 1_000_000,
                                 init_temp_factor: float = 0.02,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a random subset
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                  # split‑transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 200_000 == 0:
                    print(f"[SA] it={it:,d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_linear_decreasing(n))

    # power‑law grid
    for p in np.linspace(0.0, 4.0, 17):
        seeds.append(_power(n, p))

    # beta‑shape grid
    alphas = np.concatenate((np.linspace(0.01, 0.2, 6), np.linspace(0.2, 1.0, 6)))
    betas = np.concatenate((np.linspace(1.0, 5.0, 5), np.linspace(5.0, 10.0, 5)))
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # exponential decay grid
    for lam in np.linspace(0.001, 0.020, 10):
        seeds.append(_exponential_decay(n, lam))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Dirichlet draws – different concentrations
    for conc in (0.01, 0.05, 0.1):
        for _ in range(3):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw order
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending

    # Random sparse vectors
    for k in (8, 12, 20):
        for _ in range(3):
            seeds.append(_random_sparse_seed(n, rng, k=k))

    # previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The best sequence found is stored globally as
    ``height_sequence_1`` and also returned as a plain list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Load any previous solution for seeding.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – create a big and diverse seed pool.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # Fast surrogate ranking – keep the most promising ones.
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=operator.itemgetter(1))
    top_k = 30
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # Initialise the global best.
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – L‑BFGS refinement on the smooth‑max surrogate.
    # ------------------------------------------------------------------
    temperature_schedule = [5, 15, 30, 60, 120, 240, 480, 960, 1920]
    for idx, seed_vec in enumerate(best_seeds, start=1):
        if time.time() > deadline:
            break
        seq = seed_vec.copy()
        for temp in temperature_schedule:
            if time.time() > deadline:
                break
            seq_opt, val_opt = _l_bfgs_optimize(
                seq, n, deadline=deadline, max_iter=300, temperature=temp
            )
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[L‑BFGS] seed #{idx}, temp={temp} → new best {val_opt:.6f}")
            seq = seq_opt   # warm‑start next temperature

    # ------------------------------------------------------------------
    # Stage 2 – Projected sub‑gradient descent on the exact max‑conv objective.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Projected sub‑gradient descent ---")
        seq_pgd, val_pgd = _projected_subgradient_optimize(
            global_best_seq,
            n,
            deadline=deadline,
            max_iters=1_500_000,
            lr0=0.5,
            power=0.5,
        )
        if val_pgd < global_best_val:
            global_best_seq = seq_pgd.copy()
            global_best_val = val_pgd
            print(f"[PGD] new global best = {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Simulated‑annealing polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline=deadline,
            max_iters=1_000_000,
            init_temp_factor=0.02,
            min_temp=1e-12,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (purely for information).
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Eval] evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for possible reuse by a later run.
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
