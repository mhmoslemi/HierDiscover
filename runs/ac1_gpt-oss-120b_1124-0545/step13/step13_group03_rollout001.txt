"""
Improved search for a low‑max‑convolution step‑function.

Algorithm
---------
1. Choose a power‑of‑two length `n` (8192) – this gives fast FFTs and
   enough degrees of freedom for a very flat convolution.
2. Build a large and diverse seed pool:
   • analytic windows (uniform, triangular, sine, Hann, Blackman,
     Tukey, Gaussian, power, beta, exponential)
   • random Dirichlet draws with many concentration values (including
     very sparse draws)
   • low‑frequency cosine mixtures.
3. Keep the `top_k` seeds with the smallest surrogate
   `2·n·max(conv)` (the true objective when the vector is
   ℓ₁‑normalised).
4. **Quick EG** – a few short exponentiated‑gradient runs on the top
   seeds to get a decent starting point.
5. **Adam on a soft‑max smoothing of the max‑convolution**.
   The smooth surrogate is  

        F_p(a) = (1/p)·log Σ_k exp(p·conv_k(a))

   Its gradient can be written as a convolution, which we compute with
   FFTs.  The temperature `p` starts small (smooth) and is increased
   slowly (factor ≈ 1.7) so the optimisation first explores the space
   and then focuses on the true `max`.
   The Adam step‑size decays as `lr / sqrt(t)`.
6. **Tiny final EG** – a very cheap mirror‑descent polishing step.
7. Return the best vector found (projected onto the simplex).  The
   vector is stored in ``height_sequence_1`` for possible reuse.

All stages stop before the global deadline (`budget_s`, default 1000 s)
and print intermediate surrogate values.  The code uses only NumPy
and runs on a single core (well within the 2‑CPU limit).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via a real FFT (O(n log n)).
    The result has length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate (exact when Σseq = 1):
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic window families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                         (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float,
                      rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int,
                   rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection (more diverse than the previous version)
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hann_window(n))
    seeds.append(_blackman_window(n))
    for gamma in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.15, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_exponential_decay(n, 0.9998))
    for p in (0.2, 0.5, 1.0, 2.0):
        seeds.append(_power(n, p))
    for a in (0.3, 0.7, 1.5):
        for b in (0.3, 0.7, 1.5):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.1, 0.3, 0.6, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws (wide range of concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 4, 6):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform (bias towards flatness)
    for mix_alpha in (0.1, 0.3, 0.6):
        for nf in (2, 3):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # keep previous solution (if any) as a seed
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # ensure everything is a proper probability vector
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 200_000,
                      eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative updates)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # stepsize schedule (√t)
        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # avoid overflow
        x = x * np.exp(delta)

        # simplex projection
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# Adam optimiser on soft‑max‑smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_smooth_optimize(start: np.ndarray, n: int, deadline: float,
                          init_lr: float = 0.07,
                          beta1: float = 0.9, beta2: float = 0.999,
                          eps: float = 1e-8,
                          p0: float = 2.0,
                          p_factor: float = 1.7,
                          p_step: int = 20_000,
                          max_total_iters: int = 500_000,
                          target_surrogate: float | None = None) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth surrogate
        F_p(a) = (1/p)·log Σ exp(p·conv_k(a)).
    `p` starts at `p0` (very smooth) and is multiplied by `p_factor`
    every `p_step` iterations.
    """
    a = start.copy()
    best_a = a.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    p = p0

    while time.time() < deadline and t < max_total_iters:
        # 1) convolution and exact surrogate
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam] it={t:7d} best={best_val:.6f}  p={p:.2f}")

        # early exit if we already passed the required bound
        if target_surrogate is not None and best_val <= target_surrogate:
            break

        # 2) soft‑max weights for the current p
        shifted = conv - max_conv                # ≤ 0
        exp_arg = p * shifted
        exp_arg = np.clip(exp_arg, -700, 0)      # safety
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:                         # under‑flow protection
            mask = (conv == max_conv)
            w = mask.astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # 3) gradient of the smooth surrogate:
        #    ∇F_p = 2·conv( w , a_rev )
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)          # length = 3n‑2
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # 4) Adam update (learning‑rate decay built‑in)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr / math.sqrt(t) * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)                   # keep non‑negative
        a = _proj_simplex(a)                     # ℓ₁‑projection

        # 5) schedule for p
        if t % p_step == 0:
            p *= p_factor
            print(f"[Adam] increased p to {p:.2f}")

    final_val = 2.0 * n * np.max(_conv_fft(best_a, best_a))
    return best_a, final_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Choose a power‑of‑two length – large enough for a flat spectrum.
    # --------------------------------------------------------------
    n = 8192                                   # 2^13

    # --------------------------------------------------------------
    # Stage 0 – generate a diverse pool of seeds.
    # --------------------------------------------------------------
    print(f"\n=== Seed generation (n={n}) ===")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # evaluate surrogate and keep the best few
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda tup: tup[1])
    top_k = 20
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f" #{i+1:2d}: {val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – quick Exponential‑Gradient runs on the elite seeds.
    # --------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Quick EG refinement on elite seeds ---")
        for idx, seed_vec in enumerate(best_seeds[:8]):   # a handful
            if time.time() > deadline:
                break
            # allocate a modest number of iterations for each seed
            remaining = deadline - time.time()
            iters = min(120_000, int(remaining * 1500))
            cand, val = _expgrad_optimize(seed_vec, n, deadline,
                                          max_iters=iters,
                                          eta0=0.30)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[EG] seed #{idx+1} improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – long Adam optimisation on the smooth max‑convolution.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam smooth optimisation (soft‑max) ---")
        remaining = deadline - time.time()
        max_adam_iters = min(800_000, int(remaining * 1000))
        # target surrogate a little below the current best to allow early stop
        target = 1.503 * 2.0  # (objective = surrogate; we keep a guard)
        best_seq, best_val = _adam_smooth_optimize(
            best_seq, n, deadline,
            init_lr=0.07,
            p0=2.0,
            p_factor=1.7,
            p_step=20_000,
            max_total_iters=max_adam_iters,
            target_surrogate=target
        )
        print(f"[Adam] best surrogate after polishing = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – final tiny EG polishing (very cheap, stabilises).
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = min(40_000, int(remaining * 2000))
        if max_iters > 0:
            best_seq, best_val = _expgrad_optimize(best_seq, n, deadline,
                                                   max_iters=max_iters,
                                                   eta0=0.20)
            print(f"[Final EG] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse in a later run (helps continuation)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
