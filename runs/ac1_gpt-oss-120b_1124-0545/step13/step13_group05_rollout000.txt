     is used for ranking – it is cheap because it uses an FFT
     convolution.

2. **Local refinement**  
   Starting from the best sequence found by the scan we run a
   long sub‑gradient descent on the true max‑convolution objective.
   The iterate is projected onto the probability simplex after each
   step.  A decreasing learning‑rate schedule and occasional tiny
   random perturbations are employed.

3. **Mixture polishing**  
   The refined vector is linearly mixed with the uniform distribution;
   a fine grid of mixing coefficients is tested and the best mixture
   is kept.

4. **Final evaluation**  
   The sequence is passed to the supplied ``evaluate_sequence`` to obtain
   the true score (which is exactly the surrogate because we keep Σa=1).
   The result is stored in ``globals()["height_sequence_1"]`` for possible
   reuse.

All heavy work uses NumPy (no file or network I/O).  The code respects a
hard deadline (budget – 2 s) and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq, seq)) for an ℓ₁‑normalised
    vector (Σ seq = 1).  This equals the true evaluate_sequence value.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exponential_shape(n: int, r: float) -> np.ndarray:
    """Monotone exponential a_i ∝ r**i."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s


def _exponential_offset_shape(n: int, offset: int, r: float) -> np.ndarray:
    """Zero prefix of length ``offset`` followed by exponential tail."""
    if offset >= n:
        return _uniform(n)
    tail_len = n - offset
    if r <= 0.0:
        tail = np.full(tail_len, 1.0 / tail_len, dtype=np.float64)
    else:
        i = np.arange(tail_len, dtype=np.float64)
        tail = np.power(r, i)
        tail /= tail.sum()
    a = np.concatenate([np.zeros(offset, dtype=np.float64), tail])
    a /= a.sum()
    return a


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shaped vector on the discrete grid."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the vector."""
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(
    start_seq: np.ndarray,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # compute convolution and locate the maximal index
        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        cur_val = 2.0 * n * float(conv[idx])

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active max term
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # learning rate schedule
        lr = 0.30 / math.sqrt(it)

        # apply step and project back to simplex
        a = a - lr * g
        a = _proj_simplex(a)

        # occasional tiny random jitter to escape shallow plateaus
        if it % 5000 == 0:
            eps = 1e-4 * rng.random()
            if eps > 0.0:
                # move a tiny random amount of mass between two random coordinates
                i, j = rng.choice(n, size=2, replace=False)
                delta = eps * min(a[i], 1.0)
                a[i] -= delta
                a[j] += delta
                a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    overall_deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – exhaustive scan of simple analytic families
    # ------------------------------------------------------------------
    # Lengths to probe – a modest set covering small to fairly large vectors
    length_candidates = [2000, 4000, 6000, 8000, 10000, 12000, 16000, 18000]

    # keep the best seen across the whole scan
    global_best_seq = None
    global_best_val = float("inf")
    global_best_n = None

    # If a previous solution exists, evaluate it first
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0:
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0.0:
                    prev_arr /= s
                    val = _objective_surrogate(prev_arr)
                    global_best_seq = prev_arr.copy()
                    global_best_val = val
                    global_best_n = len(prev_arr)
                    print(f"[Seed] previous solution (n={global_best_n}) surrogate = {val:.6f}")
        except Exception:
            pass

    print("\n--- Exhaustive analytic scan ---")
    for n in length_candidates:
        if time.time() > overall_deadline:
            break
        print(f"\nScanning length n = {n}")

        # ---- pure exponential (no offset) ----
        base_eps = 2.0 / n
        for factor in np.linspace(0.2, 5.0, 30):
            eps = factor * base_eps
            r = max(0.0, 1.0 - eps)
            seq = _exponential_shape(n, r)
            val = _objective_surrogate(seq)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                global_best_n = n

        # ---- exponential with zero‑prefix offsets ----
        offsets = [
            0,
            int(0.03 * n),
            int(0.07 * n),
            int(0.12 * n),
            int(0.20 * n),
            int(0.30 * n),
            int(0.45 * n),
        ]
        for offset in offsets:
            tail_len = n - offset
            if tail_len <= 2:
                continue
            base_eps_off = 2.0 / tail_len
            for factor in np.linspace(0.2, 5.0, 25):
                eps = factor * base_eps_off
                r = max(0.0, 1.0 - eps)
                seq = _exponential_offset_shape(n, offset, r)
                val = _objective_surrogate(seq)
                if val < global_best_val:
                    global_best_val = val
                    global_best_seq = seq.copy()
                    global_best_n = n

        # ---- power‑law family ----
        for p in np.linspace(0.2, 4.0, 30):
            seq = _power(n, p)
            val = _objective_surrogate(seq)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                global_best_n = n

        # ---- beta‑distribution shapes (symmetric & skewed) ----
        betas = [0.5, 1.0, 2.0, 3.0, 5.0]
        alphas = [0.5, 1.0, 2.0, 3.0, 5.0]
        for alpha in alphas:
            for beta in betas:
                seq = _beta_shape(n, alpha, beta)
                val = _objective_surrogate(seq)
                if val < global_best_val:
                    global_best_val = val
                    global_best_seq = seq.copy()
                    global_best_n = n

        # ---- narrow Gaussian windows ----
        for sigma_frac in (0.07, 0.10, 0.13, 0.16, 0.20):
            seq = _gaussian(n, sigma_frac)
            val = _objective_surrogate(seq)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                global_best_n = n

    print("\n--- Scan finished ---")
    if global_best_seq is None:
        # fallback – uniform (should never happen)
        n = 8000
        global_best_seq = _uniform(n)
        global_best_n = n
        global_best_val = _objective_surrogate(global_best_seq)

    print(f"Best surrogate after scan: n={global_best_n}, value={global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long sub‑gradient refinement on the best candidate
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 30.0:
        max_iters = int(min(2_500_000, remaining * 2500))
        print(f"\n--- Sub‑gradient refinement (max_iters={max_iters}) ---")
        seq_refined, val_refined = _subgradient_descent(
            global_best_seq, overall_deadline, max_iters=max_iters
        )
        if val_refined < global_best_val:
            global_best_seq = seq_refined.copy()
            global_best_val = val_refined
            print(f"[Refine] new best surrogate = {val_refined:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – mixture with the uniform distribution (fine polishing)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 5.0:
        print("\n--- Uniform mixture polishing ---")
        uniform = _uniform(global_best_n)
        best_mix_seq = global_best_seq
        best_mix_val = global_best_val
        for alpha in np.linspace(0.0, 1.0, 21):
            mixed = (1.0 - alpha) * global_best_seq + alpha * uniform
            mixed /= mixed.sum()
            val = _objective_surrogate(mixed)
            if val < best_mix_val:
                best_mix_val = val
                best_mix_seq = mixed.copy()
        if best_mix_val < global_best_val:
            global_best_seq = best_mix_seq
            global_best_val = best_mix_val
            print(f"[Mix] improved surrogate = {best_mix_val:.6f}")

    # ------------------------------------------------------------------
    # Final true evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:               # pragma: no cover
        print("Warning: evaluate_sequence raised:", exc)
        true_val = None

    print("\n=== Result ===")
    print(f"Length = {global_best_n}")
    print(f"Surrogate (ℓ₁‑norm 1) = {global_best_val:.6f}")
    if true_val is not None:
        print(f"True evaluate_sequence = {true_val:.6f}")

    # expose for possible reuse in later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
