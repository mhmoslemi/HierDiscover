"""
Propose a step‑function (non‑negative vector) that minimises

        evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a))²  .

The objective is scale‑invariant, therefore we may normalise the vector to
sum 1 and minimise the maximal entry of the linear convolution.
A continuous result states that if X has the arcsine law on [0,1] then
X+X is uniformly distributed on [0,2]; discretising this law gives a
distribution whose sum‑distribution is almost flat, i.e. the maximal
convolution entry is close to its average 1/(2n‑1).  Starting from this
beta(½,½) (arcsine) shape we run a few optimisation phases:

1. **L2‑flattening** – minimise ∑(conv−avg)² by Adam (the gradient is
   a convolution of the residual with the reversed vector).  This pushes
   the whole convolution towards a flat shape.

2. **Exponential‑gradient polishing** – a cheap mirror‑descent on the
   true objective (max‑convolution) to squeeze out the last bits.

3. **Optional simulated‑annealing** – a tiny stochastic perturbation to
   escape tiny plateaus.

All stages respect a 1000 s global time‑budget; the code works for any
length (here we use n≈2000 for speed, but the method scales).  The best
vector found is stored in the global name ``height_sequence_1`` for
possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate objective: 2·n·max(conv(seq,seq)) (valid when Σseq = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on {1,…,n} (α,β>0)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector and ℓ₁‑normalise (already normalised)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Gradient of the L2‑flattening loss  L(a)=∑(conv(a,a)−avg)²
# ----------------------------------------------------------------------
def _grad_l2_loss(a: np.ndarray, avg: float) -> np.ndarray:
    """
    Returns the gradient of the loss L(a)=∑_k (conv_k(a)-avg)^2.
    Gradient formula: 4 * ( (conv(a,a)-avg)  *  a_rev )  (central slice).
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)                     # length 2n‑1
    diff = conv - avg                          # residual
    a_rev = a[::-1]
    grad_full = _conv_fft(diff, a_rev)        # length 3n‑2
    grad = 4.0 * grad_full[n - 1 : n - 1 + n]  # central n entries
    return grad


# ----------------------------------------------------------------------
# Adam optimiser for the L2 loss
# ----------------------------------------------------------------------
def _adam_l2_optimize(start: np.ndarray,
                      n: int,
                      deadline: float,
                      init_lr: float = 0.02,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Adam minimisation of the L2 loss that flattens the convolution.
    Returns the best vector (according to the true surrogate).
    """
    a = start.copy()
    best_a = a.copy()
    avg = 1.0 / (2 * n - 1)                     # target uniform probability

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    best_val = _objective_surrogate(a)

    while time.time() < deadline and t < max_iters:
        # 1) compute loss gradient
        grad = _grad_l2_loss(a, avg)

        # 2) Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # non‑negativity + simplex projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = _uniform(n)

        # keep best according to the true objective
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # occasional progress report
            if t % 10_000 == 0:
                print(f"[Adam‑L2] it={t:6d} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser for the true max‑convolution objective
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 200_000,
                      eta0: float = 0.30) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative updates)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # safety
        x = x * np.exp(delta)

        # simplex projection
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# Simulated annealing (tiny polishing, optional)
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 150_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.6:
            k = rng.integers(3, 13)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
    return best, best_val

    # --------------------------------------------------------------
    if best_seq is None:

    print(f"[Phase 2] surrogate after EG = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – optional simulated‑annealing fine‑tuning
    # --------------------------------------------------------------

        best_seq, best_val = _simulated_annealing(best_seq, n, deadline,

    return best_seq.tolist()
