# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate_obj(seq: np.ndarray) -> float:
    """
    Fast surrogate objective (the denominator Σa_i = 1 is enforced by the
    caller).  Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Basic probability‑vector helpers (always ℓ1‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a symmetric Dirichlet with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Three‑segment random piecewise‑linear shape."""
    breaks = np.sort(rng.choice(n, size=3, replace=False))
    heights = rng.random(3) + 0.01
    x = np.arange(n, dtype=np.float64)
    a = np.interp(x, breaks, heights)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _shuffle_descending(values: np.ndarray) -> np.ndarray:
    """
    Place the sorted values alternating from the left and right ends.
    This heuristic tends to keep large values far apart (useful for reducing
    large products in the convolution).
    """
    order = np.argsort(-values)            # descending order
    sorted_vals = values[order]
    out = np.empty_like(values)
    left, right = 0, len(values) - 1
    turn = True
    for v in sorted_vals:
        if turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        turn = not turn
    return out

def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the *exact* max.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    lr0 = 0.25
    power = 0.28

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # build sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s > 0.0:
            a = a / s

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 init_temp_factor: float = 0.12,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:

    Lightweight SA with three move types and periodic “mix‑and‑renorm’’
    recombination of the current elite list.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate_obj(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while time.time() < deadline:
        it += 1
        # geometric cooling
        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 4)
        if move_type == 0:

            new = cur.copy()

            idx = rng.choice(n, size=k, replace=False)

            new = cur.copy()
            new[idx] = new_alloc
        elif move_type == 2:
            # pairwise transfer (mass exchange between two bins)
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total == 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]
        else:
            # global shuffle – re‑order the whole vector using the
            # “alternating ends’’ heuristic (keeps the distribution but
            # often reduces large products)
            new = _shuffle_descending(cur)

        new_val = _surrogate_obj(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()

        # every 200 000 steps run a few exact mirror steps to polish
        if it % 200_000 == 0 and time.time() < deadline:
            seq_m, val_m = _mirror_subgradient_optimize(cur, n,
                                                       deadline,
                                                       max_iters=30_000)
            if val_m < best_val:
                best_seq = seq_m.copy()
                best_val = val_m
                cur = seq_m.copy()
                cur_val = val_m
                print(f"[SA‑polish] it={it:7d} improved to {best_val:.6f}")

        if it % 250_000 == 0:
            print(f"[SA] it={it:7d} current best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation (rich and varied)
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))

    # Random Dirichlet draws with a spread of concentrations
    concs = np.logspace(-4, 0.5, 16)   # 1e‑4 … ~3.2
    for conc in concs:
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Exponential decay curves (different rates)
    for lam in np.linspace(0.0005, 0.015, 12):
        seeds.append(_exponential_decay(n, lam))

    # Random piece‑wise‑linear shapes
    for _ in range(8):
        seeds.append(_piecewise_linear(n, rng))

    # Shuffle‑ordered versions of the Dirichlet draws (to break symmetry)
    for conc in [0.01, 0.1, 1.0]:
        for _ in range(3):
            base = _random_dirichlet(n, conc, rng)
            seeds.append(_shuffle_descending(base))

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective.  Returns the best sequence as a plain
    list of Python ``float`` values and registers it globally as
    ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # Fixed problem size (the original challenge used 4000)
    n = 4000

    # ------------------------------------------------------------------
    # Stage 0 – generate a big, diverse seed pool
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng)
    print(f"Generated {len(seeds)} raw seeds.")

    # Normalise all seeds to ℓ₁ = 1 (the surrogate assumes this)
    seeds = [s / s.sum() if s.sum() > 0 else _uniform(n) for s in seeds]

    # Evaluate cheaply with the surrogate (ignores denominator)
    evaluated = [(vec, _surrogate_obj(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 30
    elite = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – a quick mirror‑subgradient polish of each elite candidate
    # ------------------------------------------------------------------
    global_best_seq = elite[0].copy()
    global_best_val = _surrogate_obj(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(elite, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        max_iters = min(120_000, int(remaining * 120))   # ~120 iters / sec budget
        print(f"\n--- Mirror polishing run {idx}/{len(elite)} (max_iters={max_iters}) ---")
        seq_m, val_m = _mirror_subgradient_optimize(seed_vec, n, deadline,
                                                    max_iters=max_iters)
        if val_m < global_best_val:
            global_best_seq = seq_m.copy()
            global_best_val = val_m
            print(f"[overall‑Mirror] new global best = {val_m:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long Simulated‑Annealing run starting from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑Annealing (global) ---")
        seq_sa, val_sa = _simulated_annealing_optimize(global_best_seq, n,
                                                       deadline,
                                                       init_temp_factor=0.06)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Optional tiny LP polishing (very fast – runs only if we still have time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        try:
            # Build the convolution matrix for the current best – this is cheap
            # because the matrix is extremely sparse (only n² non‑zeros).
            f = global_best_seq.copy()
            M = _conv_fft(f, f).max()

            rows = np.empty(n * n, dtype=np.int32)
            cols = np.empty(n * n, dtype=np.int32)
            data = np.empty(n * n, dtype=np.float64)

            idx = 0
            for k in range(n):
                start = k
                end = k + n
                rows[idx:idx + n] = np.arange(start, end, dtype=np.int32)
                cols[idx:idx + n] = k
                data[idx:idx + n] = f
                idx += n

            A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
            b = np.full(A.shape[0], M, dtype=np.float64)
            bounds = [(0.0, None)] * n
            c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i

            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs',
                          options={'presolve': True, 'time_limit': 10})
            if res.success:
                g = res.x
                f_new = 0.85 * f + 0.15 * g
                f_new = f_new / f_new.sum()
                global_best_seq = f_new.copy()
                global_best_val = _surrogate_obj(f_new)
                print(f"[LP] polishing succeeded – new best = {global_best_val:.6f}")
        except Exception as exc:   # pragma: no cover
            print("LP polishing failed:", exc)

    # ------------------------------------------------------------------
    # Official evaluation (for information only – the platform will also call it)
    # ------------------------------------------------------------------
    try:
        from __main__ import evaluate_sequence   # the function is already imported in the environment
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
