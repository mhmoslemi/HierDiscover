def _mirror_update(a, n, max_idx):
    g = np.zeros_like(a)
    i_start = max(0, max_idx - (n-1))
    i_end = min(n-1, max_idx)
    idx_i = np.arange(i_start, i_end+1)
    rev_idx = max_idx - idx_i
    g[idx_i] = 2.0 * a[rev_idx]
    return g
```

"""Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a))²

The algorithm expands on the previous multi‑stage optimiser:

1. **Rich seed pool** – analytic families, many Dirichlet draws,
   random piece‑wise constant shapes and the incumbent solution.

2. **Fast surrogate ranking** – keep the best ~30 seeds according to
   2·n·max(conv) (identical ordering to the official objective).

3. **Smooth‑max L‑BFGS** – a long schedule of temperatures up to
   1e5, many iterations and warm‑starts, to approach the true max‑convolution.

4. **Heavy‑weight mirror‑descent** (exponentiated‑gradient) on the exact
   max‑convolution objective.  The loop runs until the deadline, using a
   decreasing learning‑rate (η₀=0.22, exponent = 0.35) and a cheap O(n) gradient.

5. **Final polishing** – lightweight simulated‑annealing with swap / Dirichlet /
   split‑transfer moves.

All stages stop before the global deadline (default 1000 s).  The best
sequence found is stored in the global name ``height_sequence_1`` and
returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.optimize


# --------------------------------------------------------------
# FFT‑based convolution (real‑valued, full length)
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real‑valued inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# --------------------------------------------------------------
# Softmax (maps logits→simplex)
# --------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# --------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# --------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = n - i
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _piecewise_constant(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random K‑segment piecewise‑constant shape."""
    K = rng.integers(2, 11)                # 2 … 10 segments
    # random break points (ensure each segment length ≥ 1)
    cuts = np.sort(rng.choice(np.arange(1, n), size=K - 1, replace=False))
    lengths = np.diff(np.concatenate(([0], cuts, [n])))
    heights = rng.dirichlet(np.full(K, 1.0))
    a = np.repeat(heights, lengths)
    return a / a.sum()


# --------------------------------------------------------------
# Surrogate objective (fast, identical ordering to official one)
# --------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – works on ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# --------------------------------------------------------------
# Smooth‑max+gradient (log‑sum‑exp surrogate)
# --------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """Return (smooth_max, gradient w.r.t. a) for the log‑sum‑exp surrogate."""
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    # stable soft‑max over the convolution entries
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum

    # gradient of Σ w_k·(a*a)_k  w.r.t. a  ->  2·(rev(a) ⊛ w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    smooth = shift + np.log(w_sum) / beta
    return smooth, grad


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iter: int = 300,
                     temperature: float = 100.0):
    """L‑BFGS on the smooth‑max surrogate (soft‑max of the convolution)."""
    eps = 1e-30
    v = np.log(start_seq + eps)               # logits → simplex via softmax

    def fun(v):
        a = _softmax(v)
        smooth, grad_a = _smooth_max_and_grad(a, temperature)
        loss = 2.0 * n * smooth
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)           # chain‑rule through softmax
        return loss, (2.0 * n * grad_v).astype(np.float64)

    # honour the remaining budget – stop after max_iter or when time runs out
    try:
        res = scipy.optimize.minimize(
            fun,
            v,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
    except Exception:   # pragma: no cover
        return start_seq, _objective(start_seq)

    best_a = _softmax(res.x)
    return best_a, _objective(best_a)


# --------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑conv objective
# --------------------------------------------------------------
def _mirror_descent(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    lr0: float = 0.22,
                    power: float = 0.35,
                    max_iters: int = 10_000_000):
    """Multiplicative update on the simplex using the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 1
    while it <= max_iters and time.time() < deadline:
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:,d} best={best_val:.6f}")

        # gradient of the active convolution entry (sub‑gradient)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = lr0 / (it ** power)
        a = a * np.exp(-lr * g)

        # re‑normalise onto the simplex, guard against total collapse
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        it += 1

    return best_seq, best_val


# --------------------------------------------------------------
# Light Simulated‑Annealing for final polishing
# --------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 2_000_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-14):
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        T = max(T * 0.9995, min_temp)

        move = rng.integers(0, 4)
        if move == 0:                                 # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move == 1:                               # redistribute a random subset
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new = cur.copy()
            new[idx] = rng.dirichlet(np.full(k, 1.0)) * total
        elif move == 2:                               # split‑transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]
        else:                                         # small Gaussian perturb on a subset
            k = rng.integers(5, 15)
            idx = rng.choice(n, size=k, replace=False)
            new = cur.copy()
            noise = rng.normal(scale=0.005, size=k)
            new[idx] = np.clip(new[idx] + noise, 0.0, None)
            new_sum = new.sum()
            if new_sum <= 0.0:
                continue
            new = new / new_sum

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 250_000 == 0:
                    print(f"[SA] it={it:,d} best={best_val:.6f}")

    return best_seq, best_val


# --------------------------------------------------------------
# Seed generation (rich and diverse)
# --------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_linear_decreasing(n))

    # power‑law
    for p in np.linspace(0.0, 4.0, 17):
        seeds.append(_power(n, p))

    # β‑shape – extensive asymmetric grid
    alphas = np.concatenate((np.linspace(0.01, 0.3, 8), np.linspace(0.3, 2.0, 7)))
    betas = np.concatenate((np.linspace(0.5, 2.0, 8), np.linspace(2.0, 12.0, 7)))
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # exponential decay – fine grid
    for lam in np.linspace(0.001, 0.05, 12):
        seeds.append(_exponential_decay(n, lam))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Dirichlet draws – wide range of concentrations
    for conc in (0.001, 0.005, 0.01, 0.05, 0.1):
        for _ in range(3):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw ordering
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending

    # Random piecewise‑constant shapes
    for _ in range(6):
        seeds.append(_piecewise_constant(n, rng))

    # previous solution, if available
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0.0:
            seeds.append(avg / s)

    return seeds

    """

    rng = np.random.default_rng(seed)

    start_time = time.time()

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------

    previous = globals().get("height_sequence_1")
