def _lp_step(f, M, t, deadline):
    n = len(f)
    # build sparse A
    offsets = np.arange(n)
    data = [np.full(n, f[d]) for d in range(n)]
    A = sp.diags(data, offsets, shape=(2*n-1, n), format='csc')
    c = -np.ones(n)  # maximize sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = _opt.linprog(c=c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
    except Exception:
        return f, np.inf
    if not res.success:
        return f, np.inf
    g = res.x
    s = g.sum()
    if s <= 1e-12:
        return f, np.inf
    g_norm = g / s
    # mix
    new_f = (1 - t) * f + t * g_norm
    # ensure non‑negative sum=1
    new_f = np.maximum(new_f, 0.0)
    total = new_f.sum()
    if total == 0:
        new_f = f
    else:
        new_f = new_f / total
    return new_f, _objective_surrogate(new_f)

        2·n·max(conv(a, a)) / (sum(a)²)

where conv is the full linear convolution.
The optimiser works in four stages:

1️⃣  **Seed generation** – a rich pool of ℓ₁‑normalised vectors
    (analytic windows, Dirichlet draws, mixtures, …) is built and the
    best few are kept using the exact surrogate 2·n·max(conv) (valid for
    ℓ₁‑normalised vectors).

2️⃣  **Continuation L‑BFGS** – the smooth log‑sum‑exp approximation of the
    max‑convolution is minimised starting from each top seed.  The β‑schedule
    reaches 5 000 to get very close to the true max.

3️⃣  **Linear‑programming mixing step** – for the current vector *f* we solve

        maximise   Σ g_j

    down.

4️⃣  **Mirror‑/sub‑gradient polishing** – a long exponentiated‑gradient pass
    (mirror descent on the true max‑convolution) followed by a final

sequence found is stored in ``height_sequence_1`` and returned as a plain

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² for a (possibly un‑normalised) vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = exact objective for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # modified Bessel, NumPy provides approximation
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a Dirichlet draw and assign largest/smallest alternatingly."""
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _collect_seeds_extended(n: int,

    """Assemble a large, diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families

    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12):

    seeds.append(_double_gaussian(n, sigma_frac=0.10, shift_frac=0.30))
    seeds.append(_double_gaussian(n, sigma_frac=0.15, shift_frac=0.25))

    # Power‑law
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta‑family grid
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws (heavy‑tailed)
    for _ in range(4):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Special alternating‑extrema Dirichlet seed
    seeds.append(_alternating_dirichlet(n, rng))

    # Previous solution if available
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    cleaned = []
    for v in seeds:

        s = v.sum()
        if s > 0:

        else:

# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 5000.0]

    # initialise in log‑softmax space
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)          # stays on the simplex
            conv = _conv_fft(a, a)          # linear convolution
            max_c = np.max(conv)
            # smoothed version (log‑sum‑exp)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # softmax weights of the (smoothed) max
            w = exp_beta / sum_exp
            # gradient wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)         # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient wrt logits via Jacobian of softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a

# Exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (projected via softmax)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

# Sub‑gradient descent (projected on simplex)

def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val

# Linear‑programming mixing step (active‑set version)
# ----------------------------------------------------------------------
def _build_active_matrix(f: np.ndarray, rows: list[int]) -> _sp.csr_matrix:
    """
    Build a sparse matrix A such that (A @ g)[r] = (f * g)[rows[r]]
    for the given active rows.  Only the constraints for those rows are
    included – this keeps the LP tiny.
    """
    n = len(f)
    data = []
    col_ix = []
    row_ix = []
    for r, k in enumerate(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            data.append(f[i])
            col_ix.append(j)
            row_ix.append(r)
    A = _sp.csr_matrix((data, (row_ix, col_ix)), shape=(len(rows), n))
    return A


def _lp_mix_step(f: np.ndarray,
                 max_f: float,
                 t: float,
                 deadline: float,
                 max_cuts: int = 12) -> tuple[np.ndarray, float]:
    """
    Perform one LP‑based mixing step.
    Returns the new vector and its surrogate value.
    """
    n = len(f)
    # initial active set = indices where f*f is (almost) maximal
    conv_ff = _conv_fft(f, f)
    tol = max_f * 1e-10
    active = set(np.where(conv_ff >= max_f - tol)[0].tolist())

    # Cutting‑plane loop
    for _ in range(max_cuts):
        if time.time() > deadline:
            break
        rows = sorted(active)
        A = _build_active_matrix(f, rows)
        b = np.full(len(rows), max_f, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise Σ g_j  ⇒ minimise -Σ g_j
        bounds = [(0.0, 1e6)] * n                    # safeguard against unboundedness

        try:
            res = _opt.linprog(c=c, A_ub=A, b_ub=b,
                               bounds=bounds, method='highs',
                               options={'presolve': True})
        except Exception:
            # LP failed – give up this mixing step
            break

        if not res.success:
            # Could be unbounded – abort this step
            break

        g = res.x
        # verify feasibility on *all* shifts
        conv_fg = _conv_fft(f, g)
        viol = np.where(conv_fg > max_f + 1e-12)[0]
        if viol.size == 0:
            # feasible – stop cutting
            break
        # add violating rows to the active set
        for v in viol:
            active.add(int(v))
    else:
        # reached max_cuts without full feasibility – proceed anyway
        pass

    s = g.sum()
    if s <= 1e-12:
        return f, _objective_surrogate(f)

    g_norm = g / s
    new_f = (1.0 - t) * f + t * g_norm
    # maintain positivity and unit ℓ₁ norm
    new_f = np.maximum(new_f, 0.0)
    total = new_f.sum()
    if total > 0:
        new_f = new_f / total
    else:
        new_f = f.copy()
    return new_f, _objective_surrogate(new_f)

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # initialise RNG and timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # leave a small safety buffer

    # ------------------------------------------------------------------
    # problem size – a modest 2000 gives a good speed/quality trade‑off
    # ------------------------------------------------------------------
    n = 2000
    print(f"\n[Info] problem size set to n={n}")

    # ------------------------------------------------------------------
    # 0. seed generation
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seed vectors.")

    # cheap surrogate evaluation (exact for ℓ₁‑normalised vectors)
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 10
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best surrogate after seeding (top {top_k}):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 1. continuation L‑BFGS on the most promising seeds
    # ------------------------------------------------------------------
    global_best = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline before finishing L‑BFGS runs.")
            break
        sec_left = deadline - time.time()
        max_iter_beta = min(5000, int(sec_left * 0.07 * 1e5))
        max_iter_beta = max(800, max_iter_beta)
        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter/beta={max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         max_iter_per_beta=max_iter_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Linear‑programming mixing steps (the main novelty)
    # ------------------------------------------------------------------
    print("\n--- LP mixing phase ---")
    mix_iters = 7
    t_schedule = [0.5, 0.4, 0.3, 0.3, 0.2, 0.2, 0.15]   # decreasing step sizes
    for it in range(mix_iters):
        if time.time() > deadline:
            print("[Time] deadline reached during LP mixing.")
            break
        M = np.max(_conv_fft(global_best, global_best))
        t = t_schedule[it] if it < len(t_schedule) else 0.1
        remaining = deadline - time.time()
        # each LP step is cheap; we give a tiny slice of the budget
        sub_deadline = time.time() + min(remaining * 0.07, 30.0)
        new_seq, new_val = _lp_mix_step(global_best, M, t, sub_deadline)
        print(f"[LP mix {it+1}] surrogate = {new_val:.6f}")
        if new_val < global_best_val:
            global_best = new_seq.copy()
            global_best_val = new_val
            print(f"[LP Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. long exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp = int(min(600_000, remaining * 0.10 * 1e5))
        if max_exp > 0:
            print("\n--- Exponentiated‑gradient polishing ---")
            seq_exp, val_exp = _exp_gradient_opt(global_best,
                                                 n,
                                                 deadline,
                                                 max_iters=max_exp)
            if val_exp < global_best_val:
                global_best = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. final sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(600_000, remaining * 0.10 * 1e5))
        if max_sub > 0:
            print("\n--- Sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best,
                                                n,
                                                deadline,
                                                max_iters=max_sub)
            if val_sub < global_best_val:
                global_best = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best.tolist()
    return global_best.tolist()
