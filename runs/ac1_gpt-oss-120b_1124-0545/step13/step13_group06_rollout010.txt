All operations stay on the simplex via a softmax (logits) parametrisation
or multiplicative updates, guaranteeing non‑negativity and unit ℓ₁‑norm.
The routine respects the 1000 s deadline; if the budget runs out it simply
returns the best solution discovered so far.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Basic FFT‑based convolution (real, full length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Softmax (logits → simplex) – keeps the vector non‑negative, sum=1
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = n - i
    s = a.sum()
    return a / s


def _poly_decrease(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p (p>0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    s = a.sum()
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    s = a.sum()
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth random curve formed by a few low‑frequency sinusoids."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Fast surrogate objective (identical ordering to the official one)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – works on ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient w.r.t. a
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """Return (smooth_max, gradient) of the log‑sum‑exp surrogate."""
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    # stable soft‑max over the convolution entries
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum
    smooth = shift + np.log(w_sum) / beta

    # grad wrt a : 2 * (rev(a) ⊛ weights)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad_a = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad_a


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iter: int = 400,
                     temperature: float = 100.0) -> tuple[np.ndarray, float]:
    """
    L‑BFGS on the smooth‑max surrogate.
    The optimisation is performed on logits (log‑softmax) to keep the iterate on the simplex.
    """
    eps = 1e-30
    v = np.log(start_seq + eps)               # logits → simplex via softmax

    def fun(v):
        a = _softmax(v)
        smooth, grad_a = _smooth_max_and_grad(a, temperature)
        loss = 2.0 * n * smooth                 # matches the official objective
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)             # chain rule through softmax
        return loss, (2.0 * n * grad_v).astype(np.float64)

    # The optimiser may be stopped early if the deadline is hit;
    # we rely on the fact that SciPy's ``minimize`` respects the iteration limit.
    try:
        res = _opt.minimize(
            fun,
            v,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
    except Exception:  # safety net – fall back to the start vector
        return start_seq, _objective(start_seq)

    best_a = _softmax(res.x)
    return best_a, _objective(best_a)


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 2_000_000,
                                 lr0: float = 0.12,
                                 power: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated‑gradient) updates on the simplex
    using the exact max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 1
    while it <= max_iters:
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:,d} best={cur_val:.6f}")

        # sub‑gradient for the active anti‑diagonal
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light simulated‑annealing for final polishing
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 1_500_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a random subset
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                  # split‑transfer between a pair
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 200_000 == 0:
                    print(f"[SA] it={it:,d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation (rich and diverse)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_linear_decreasing(n))
    seeds.append(_poly_decrease(n, 1.5))
    seeds.append(_poly_decrease(n, 2.0))

    # power‑law family (dense grid)
    for p in np.linspace(0.0, 5.0, 51):
        seeds.append(_power(n, p))

    # exponential decay – explore very small rates as well
    for lam in np.logspace(-4, -1, 30):  # 1e‑4 … 1e‑1
        seeds.append(_exponential_decay(n, lam))

    # β‑shape – moderate grid covering both <1 and >1
    alphas = np.linspace(0.2, 2.0, 9)
    betas = np.linspace(0.2, 2.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Dirichlet draws – a wide range of concentration values
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw ordering
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0.0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the supplied time budget.
    Returns the best sequence (list of floats) and stores it as the global
    variable ``height_sequence_1``.
    """
    # ------------------------------------------------------------------
    # basic set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a small safety margin

    # problem size – 4000 offers a good trade‑off between flexibility and speed
    n = 4000

    # ------------------------------------------------------------------
    # Stage 0 – generate a large, diverse seed pool
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    previous = globals().get("height_sequence_1")
    try:
        previous = np.asarray(previous, dtype=np.float64) if previous is not None else None
    except Exception:
        previous = None

    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # fast surrogate ranking – keep the most promising ones
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 40
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – L‑BFGS refinement on a sequence of increasing temperatures
    # ------------------------------------------------------------------
    temperature_schedule = [5, 10, 20, 40, 80, 150, 300, 600, 1000, 2000, 5000]
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        seq = seed_vec.copy()
        for temp in temperature_schedule:
            if time.time() > deadline:
                break
            seq_opt, val_opt = _l_bfgs_optimize(
                seq, n, deadline, max_iter=400, temperature=temp
            )
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[L‑BFGS] seed #{idx}, temp={temp} → new best {val_opt:.6f}")
            seq = seq_opt   # warm‑start next temperature

    # ------------------------------------------------------------------
    # Stage 2 – Mirror‑descent (exponentiated‑gradient) fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=2_000_000,
            lr0=0.12,
            power=0.30,
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=1_500_000,
            init_temp_factor=0.04,
            min_temp=1e-12,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (informational only)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Eval] evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # store for possible reuse by a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
