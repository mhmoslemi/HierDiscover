"""
Hybrid optimiser for the step‑function bound

1. Generate a very diverse pool of ℓ₁‑normalised seeds.
   - Analytic families (uniform, linear, power‑law with fine p‑grid,
     shifted power‑law, exponential, β‑shape, low‑frequency sinusoids).
   - Dirichlet draws with several concentration parameters.
   - The previous best solution (if any).

2. Keep the ~10 best seeds according to the fast surrogate
   `2·n·max(conv(a,a))` (identical ordering to the official objective).

3. Refine each top seed with L‑BFGS on a log‑sum‑exp smooth‑max of the
   convolution.  The temperature is increased geometrically
   `5,10,20,…,5120` and each stage is allowed up to 400 iterations.
   This yields a high‑quality, smooth approximation of the exact optimum.

4. Starting from the best refined seed run a long exponentiated‑gradient
   (mirror‑descent) on the *exact* max‑convolution objective.
   Learning‑rate = `lr0 / t**power` with `lr0=0.4` and `power=0.33`
   (a slightly more aggressive schedule than the reference code).

5. Polish the result with a simulated‑annealing phase that uses three
   move types (swap, Dirichlet‑redistribute, split‑transfer) and a
   very slow exponential cooling.

6. If any time remains, perform a final short L‑BFGS pass with a very
   high temperature (≈ 8000) to tighten the smooth‑max around the true
   maximum.

The best sequence found is stored as the global name
``height_sequence_1`` and returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.optimize

# --------------------------------------------------------------
# FFT‑based full convolution (real, length = 2·n‑1)
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# --------------------------------------------------------------
# Softmax (used to stay on the simplex)
# --------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# --------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# --------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = n - i
    return a / a.sum()


def _power_law(n: int, p: float, shift: float = 0.0) -> np.ndarray:
    """a_i ∝ 1/(i+shift)^{p}, i = 0,…,n‑1."""
    i = np.arange(n, dtype=np.float64) + shift
    a = 1.0 / (i ** p + 1e-30)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# --------------------------------------------------------------
# Surrogate objective (fast, identical ordering to official one)
# --------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – works on ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# --------------------------------------------------------------
# Smooth‑max and its gradient (log‑sum‑exp surrogate)
# --------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """Return (smooth_max, grad) for the log‑sum‑exp surrogate."""
    conv = _conv_fft(a, a)
    shift = np.max(conv)                     # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum
    smooth = shift + np.log(w_sum) / beta

    # gradient of Σ w_k·(a∗a)_k w.r.t. a  →  2·rev(a) ⊛ w
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad_a = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad_a


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iter: int = 400,
                     temperature: float = 100.0) -> tuple[np.ndarray, float]:
    """L‑BFGS on the smooth‑max surrogate."""
    eps = 1e-30
    v = np.log(start_seq + eps)               # logits → simplex via softmax

    def fun(v):
        a = _softmax(v)
        smooth, grad_a = _smooth_max_and_grad(a, temperature)
        loss = 2.0 * n * smooth
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)           # chain‑rule through softmax
        return loss, (2.0 * n * grad_v).astype(np.float64)

    # honour the deadline – minimise for at most max_iter iterations
    try:
        res = scipy.optimize.minimize(
            fun,
            v,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
    except Exception:  # pragma: no cover
        return start_seq, _objective(start_seq)

    best_a = _softmax(res.x)
    return best_a, _objective(best_a)


# --------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑conv objective
# --------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 5_000_000,
                                 lr0: float = 0.4,
                                 power: float = 0.33) -> tuple[np.ndarray, float]:
    """Multiplicative update on the simplex using the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 1
    while it <= max_iters:
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:,d} best={cur_val:.6f}")

        # sub‑gradient for the active anti‑diagonal
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # re‑normalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        it += 1

    return best_seq, best_val


# --------------------------------------------------------------
# Light simulated annealing for final polishing
# --------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 2_000_000,
                                 init_temp_factor: float = 0.06,
                                 min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # very slow geometric cooling
        T = max(T * 0.9999, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # Dirichlet redistribution on a random subset
            k = rng.integers(2, 14)            # subset size
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                   # split‑transfer between a pair
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 250_000 == 0:
                    print(f"[SA] it={it:,d} best={new_val:.6f}")

    return best_seq, best_val

def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_linear_decreasing(n))

    # fine grid of power‑law shapes (including shifted versions)
    for p in np.linspace(0.45, 0.65, 41):
        seeds.append(_power_law(n, p, shift=0.0))
        for sh in (0.5, 1.0, 2.0, 5.0):
            seeds.append(_power_law(n, p, shift=sh))

    # exponential family (fine grid)
    for lam in np.linspace(0.001, 0.020, 20):
        seeds.append(_exponential_decay(n, lam))

    # β‑shape – moderate grid around the region that often works well
    alphas = np.concatenate((np.linspace(0.05, 0.3, 6), np.linspace(0.3, 1.0, 6)))
    betas = np.concatenate((np.linspace(1.0, 4.0, 5), np.linspace(4.0, 12.0, 5)))
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # low‑frequency random curves (few samples)
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Dirichlet draws – low concentrations (spiky candidates)
    for conc in (0.01, 0.03, 0.07):
        for _ in range(2):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw order
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending

    # Include the previous solution if it exists
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0.0:
            seeds.append(avg / avg.sum())

    return seeds

    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # Initialise the global best.
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – L‑BFGS refinement on the smooth‑max surrogate.
    # ------------------------------------------------------------------
    temperature_schedule = [5, 10, 20, 40, 80, 160, 320, 640, 1280, 2560, 5120]
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        seq = seed_vec.copy()
        for temp in temperature_schedule:
            if time.time() > deadline:
                break
            seq_opt, val_opt = _l_bfgs_optimize(seq, n, deadline,
                                                max_iter=400,
                                                temperature=temp)
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[L‑BFGS] seed #{idx}, temp={temp} → new best {val_opt:.6f}")
            seq = seq_opt   # warm‑start the next temperature

    # ------------------------------------------------------------------
    # Stage 2 – Mirror‑descent sub‑gradient optimisation (exact max).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        # Rough estimate: one sub‑gradient iteration costs ≈0.00013 s
        # (FFT of length ≈8192).  Use ~70 % of the remaining budget.
        est_iters = int(min(8_000_000, max(1_000_000, time_left / 0.00013 * 0.7)))
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        print(f"Planned iterations: {est_iters:,d}")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=est_iters,
            lr0=0.4, power=0.33)
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Simulated‑annealing polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline,
            max_iters=2_000_000,
            init_temp_factor=0.06,
            min_temp=1e-14)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Final high‑temperature L‑BFGS fine‑tuning (optional).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final high‑temperature L‑BFGS pass ---")
        seq_final, val_final = _l_bfgs_optimize(
            global_best_seq, n, deadline,
            max_iter=200, temperature=8000.0)
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Final L‑BFGS] new best = {val_final:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (information only).
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Eval] evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for later runs and return.
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
