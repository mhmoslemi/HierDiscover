"""
Improved optimiser for the non‑negative step‑function problem.

The algorithm builds a rich pool of ℓ₁‑normalised seeds (analytic families,
random Dirichlet draws, a greedy Sidon‑set construction and many random
permutations).  The best seeds are refined with a long L‑BFGS schedule on a
smooth‑max (log‑sum‑exp) surrogate, followed by a high‑precision Adam
optimiser on the same surrogate.  The result is then polished with a
multiplicative mirror‑descent (exponentiated‑gradient) on the exact
max‑convolution objective and, finally, a light simulated‑annealing phase.

All stages respect a global time‑budget (default 1000 s) and the best
vector found is stored in the global name ``height_sequence_1`` and returned
as a plain list of floats.
"""

import time
import numpy as np
import scipy.optimize

# --------------------------------------------------------------
# FFT‑based full linear convolution (real‑valued)
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# --------------------------------------------------------------
# Softmax (simple re‑parameterisation onto the simplex)
# --------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# --------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# --------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = n - i
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth random curve built from a few low‑frequency sinusoids."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sidon_seed(n: int, target_m: int = 70) -> np.ndarray:
    """
    Greedy construction of a B₂ (Sidon) set: distinct pairwise sums.
    The returned vector puts equal mass on the selected positions.
    """
    positions = []
    sums = set()
    for i in range(n):
        conflict = False
        for p in positions:
            if (i + p) in sums:
                conflict = True
                break
        if conflict or (i + i) in sums:
            continue
        # accept i
        for p in positions:
            sums.add(i + p)
        sums.add(i + i)
        positions.append(i)
        if len(positions) >= target_m:
            break
    if not positions:
        return _uniform(n)
    a = np.zeros(n, dtype=np.float64)
    w = 1.0 / len(positions)
    a[positions] = w
    return a


# --------------------------------------------------------------
# Fast surrogate objective (identical ordering to official one)
# --------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – works on ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# --------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient w.r.t. a
# --------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Returns (smooth_max, grad_a) for the log‑sum‑exp surrogate.
    grad_a = 2·(rev(a) ⊛ weights) where weights are the softmax over
    beta·(conv - max(conv)).
    """
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum
    smooth = shift + np.log(w_sum) / beta
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad_a = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad_a


# --------------------------------------------------------------
# L‑BFGS optimiser on the smooth‑max surrogate
# --------------------------------------------------------------
def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iter: int = 500,
                     temperature: float = 100.0):
    """L‑BFGS on the log‑sum‑exp surrogate."""
    eps = 1e-30
    v = np.log(start_seq + eps)               # logits → simplex

    def fun(v):
        a = _softmax(v)
        smooth, grad_a = _smooth_max_and_grad(a, temperature)
        loss = 2.0 * n * smooth
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)            # chain rule
        return loss, (2.0 * n * grad_v).astype(np.float64)

    # honour remaining time – if the optimiser does not finish in time,
    # we simply return the starting point.
    try:
        res = scipy.optimize.minimize(
            fun,
            v,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
    except Exception:
        return start_seq, _objective(start_seq)

    best_a = _softmax(res.x)
    return best_a, _objective(best_a)


# --------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# --------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 40000,
                   beta_smooth: float = 2000.0,
                   lr0: float = 0.15):
    """
    Adam‑style updates on the smooth‑max surrogate.
    Operates in the log‑space (logits → softmax) so the ℓ₁‑constraint
    is always satisfied.
    """
    eps = 1e-30
    v = np.log(start_seq + eps)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)          # second moment
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8

    best_seq = _softmax(v)
    best_val = _objective(best_seq)

    t = 0
    while t < max_iters:
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        smooth, grad_a = _smooth_max_and_grad(a, beta_smooth)
        dot = np.dot(grad_a, a)
        grad_v = (2.0 * n) * (a * (grad_a - dot))

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad_v
        v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)

        m_hat = m / (1 - beta1 ** t)
        v2_hat = v2 / (1 - beta2 ** t)

        lr = lr0 / np.sqrt(t)                     # simple decay
        v = v - lr * m_hat / (np.sqrt(v2_hat) + epsilon)

        # periodic monitoring (every 2000 steps)
        if (t % 2000) == 0:
            a_cur = _softmax(v)
            cur_val = _objective(a_cur)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a_cur.copy()
                print(f"[Adam] it={t:,d} best={cur_val:.6f}")

    return best_seq, best_val


# --------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on exact max‑conv objective
# --------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 8_000_000,
                                 lr0: float = 0.6,
                                 power: float = 0.30):
    """Multiplicative update on the simplex using the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 1
    while it <= max_iters:
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = float(2.0 * n * conv[max_idx])

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if (it % 250_000) == 0:
                print(f"[Mirror] it={it:,d} best={cur_val:.6f}")

        # sub‑gradient for the active anti‑diagonal
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # projection back to the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        it += 1

    return best_seq, best_val


# --------------------------------------------------------------
# Light simulated‑annealing for final polishing
# --------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 3_000_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12):
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 4)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a random subset
            k = rng.integers(2, 14)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        elif move_type == 2:                   # split‑transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]
        else:                                  # rotate a small block
            blk = rng.integers(3, 7)
            start = rng.integers(0, n - blk)
            seg = cur[start:start + blk]
            new = cur.copy()
            new[start:start + blk] = np.roll(seg, 1)

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if (it % 250_000) == 0:
                    print(f"[SA] it={it:,d} best={new_val:.6f}")

    return best_seq, best_val


# --------------------------------------------------------------
# Seed pool construction (rich and diverse)
# --------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a pool of ℓ₁‑normalised candidate vectors."""
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_linear_decreasing(n))

    # power‑law family (wide grid)
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # β‑shape – broad asymmetric grid
    alphas = np.concatenate((np.linspace(0.01, 0.25, 6),
                             np.linspace(0.25, 1.2, 6)))
    betas = np.concatenate((np.linspace(1.0, 5.0, 5),
                            np.linspace(5.0, 12.0, 5)))
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # exponential decay (fine grid)
    for lam in np.linspace(0.001, 0.030, 12):
        seeds.append(_exponential_decay(n, lam))

    # low‑frequency random curves – more diversity
    for nf in (2, 3, 4, 5):
        for _ in range(5):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Dirichlet draws with a variety of concentration parameters
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(4):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                               # raw order
            seeds.append(np.sort(raw)[::-1])                # descending
            seeds.append(np.sort(raw))                      # ascending

    # Random permutations of a few promising families
    perm_base = [_power(n, 0.6), _beta_shape(n, 0.3, 3.0), _exponential_decay(n, 0.008)]
    for base in perm_base:
        for _ in range(10):
            perm = base.copy()
            rng.shuffle(perm)
            seeds.append(perm)

    # Greedy Sidon set seed (spreads pairwise sums)
    seeds.append(_sidon_seed(n, target_m=int(np.sqrt(2 * n))))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

# Main entry point – orchestrates the whole search
# --------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **_):
    """
    Search for a non‑negative step‑function that minimizes the official
    evaluate_sequence objective within the given time budget.
    The best sequence is stored as ``height_sequence_1`` and also returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safeguard margin

    # ----------------------------------------------------------
    # Problem size – a few thousand items give a good trade‑off.
    # ----------------------------------------------------------
    n = 4000

    # ----------------------------------------------------------
    # Load any previously found solution (if the notebook re‑runs).
    # ----------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ----------------------------------------------------------
    # Stage 0 – generate a large and diverse pool of seeds.
    # ----------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # Fast surrogate ranking – keep the most promising ones.
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 30
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # Initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ----------------------------------------------------------
    # Stage 1 – L‑BFGS refinement on a smooth‑max surrogate.
    # ----------------------------------------------------------
    temperature_schedule = [5, 10, 20, 40, 80, 150, 300, 600,
                            1000, 2000, 5000, 10000]
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        seq = seed_vec.copy()
        for temp in temperature_schedule:
            if time.time() > deadline:
                break
            seq_opt, val_opt = _l_bfgs_optimize(
                seq, n, deadline, max_iter=500, temperature=temp
            )
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[L‑BFGS] seed #{idx}, temp={temp} → new best {val_opt:.6f}")
            seq = seq_opt   # warm‑start next temperature

    # ----------------------------------------------------------
    # Stage 2 – Adam optimisation on the same surrogate (high precision).
    # ----------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam optimisation on smooth‑max surrogate ---")
        seq_adam, val_adam = _adam_optimize(
            global_best_seq, n, deadline,
            max_iters=40000, beta_smooth=2000.0, lr0=0.15
        )
        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam] new global best = {val_adam:.6f}")

    # ----------------------------------------------------------
    # Stage 3 – Mirror‑descent (exact max) fine‑tuning.
    # ----------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=8_000_000,
            lr0=0.6,
            power=0.30,
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # ----------------------------------------------------------
    # Stage 4 – Simulated‑annealing polishing.
    # ----------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=3_000_000,
            init_temp_factor=0.04,
            min_temp=1e-12,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ----------------------------------------------------------
    # Final official evaluation (just for reporting).
    # ----------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[Eval] evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for potential reuse.
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
