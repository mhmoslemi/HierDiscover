"""
Propose a step‑function (non‑negative vector) that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a))²

The implementation builds a diversified pool of analytic and random
“seed” windows, performs a short exponentiated‑gradient (EG) refinement,
followed by a long Adam optimisation on a soft‑max‑smoothed max‑convolution.
Finally a prolonged simulated‑annealing (SA) polish and a tiny EG sweep
are applied.  The whole search respects a 1000 s budget and returns the
best sequence found (stored globally as ``height_sequence_1``).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities ---------------------------------------------------
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate (2·n·max(conv(seq,seq))) when Σseq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors) -----------------
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _half_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Mass only on the first half of the vector."""
    half = n // 2
    vec = np.zeros(n, dtype=np.float64)
    vec[:half] = rng.dirichlet(np.full(half, conc))
    s = vec.sum()
    return vec / s if s > 0 else _uniform(n)


def _random_sparse_binary(n: int, density: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse binary‑like vector (few non‑zero entries)."""
    a = np.zeros(n, dtype=np.float64)
    k = max(1, int(density * n))
    idx = rng.choice(n, size=k, replace=False)
    a[idx] = rng.random(k) + 1e-6  # ensure positivity
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    for sigma_frac in (0.2, 0.3, 0.4):
        seeds.append(_gaussian(n, sigma_frac))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # hybrid/mixture ideas
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # half‑Dirichlet (mass only on left half)
    for conc in (0.5, 1.0):
        seeds.append(_half_dirichlet(n, conc, rng))

    # sparse binary seeds
    for density in (0.02, 0.05, 0.1):
        seeds.append(_random_sparse_binary(n, density, rng))

    # keep previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final sanity pass
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent on the simplex) -----
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Multiplicative mirror‑descent (KL‑divergence) on the simplex."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # safety for the exponent
        x = x * np.exp(delta)

        # simplex projection
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val

                          beta1: float = 0.9, beta2: float = 0.999,
                          eps: float = 1e-8,
                          p0: float = 5.0, p_max: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed surrogate
        F_p(a) = (1/p)·log Σ exp(p·conv_k(a))
    where p grows linearly from p0 to p_max.
    Returns the best vector seen (according to the exact surrogate).
    """
    a = start.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    for t in range(1, total_iters + 1):
        if time.time() > deadline:
            break

        # ----- current convolution & exact max -----
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # occasional progress report
            if t % 50000 == 0:
                print(f"[Adam] it={t:7d} best={best_val:.6f}")

        # ----- soft‑max weights (p schedules linearly) -----
        p = p0 + (p_max - p0) * (t / total_iters)
        shifted = conv - max_conv                     # ≤ 0
        exp_arg = p * shifted
        exp_arg = np.clip(exp_arg, -700, 0)          # guard overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:                              # under‑flow fallback
            mask = conv == max_conv
            w = mask.astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of the smooth surrogate -----
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)               # length 3n‑2
        grad = 2.0 * grad_full[n - 1: n - 1 + n]       # central slice

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = _uniform(n)

    # final exact evaluation of best vector
    final_val = 2.0 * n * np.max(_conv_fft(best_a, best_a))
    return best_a, final_val


# ----------------------------------------------------------------------
# Simulated‑annealing polish (longer) ---------------------------------
# ----------------------------------------------------------------------
def _simulated_annealing_long(start: np.ndarray, n: int, deadline: float,
                             max_iters: int = 2_000_000,
                             init_temp_factor: float = 0.08,
                             min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # exponential cooling, but very gentle
        T = max(T * 0.99999, min_temp)

        if rng.random() < 0.55:
            # small mass transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            if cur[i] <= 0.0:
                continue
            eps = rng.random() * min(cur[i], 0.05)
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps
        else:
            # reshuffle a tiny random subset
            k = rng.integers(4, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            cand = cur.copy()
            cand[idx] = rng.dirichlet(np.full(k, 1.0)) * total

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 250_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns the best sequence (list of floats) and stores it in the global
    variable ``height_sequence_1`` for possible reuse.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Length of the window – 4096 gives a power‑of‑two footprint.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Stage 0 – generate a diverse pool of seeds.
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # keep the best few according to the simple surrogate
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – quick EG refinement on the best seeds.
    # --------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Quick EG refinement (first pass) ---")
        for idx, seed_vec in enumerate(best_seeds[:5]):   # a few fast runs
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            max_it = int(min(80_000, remaining * 2500))
            cand, val = _expgrad_optimize(seed_vec, n, deadline,
                                          max_iters=max_it, eta0=0.30)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Quick EG #{idx+1}] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – Adam optimisation with a gentle p‑schedule.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam smooth optimisation (soft‑max) ---")
        remaining = deadline - time.time()
        max_adam_iters = int(min(1_200_000, remaining * 1500))
        # run a few Adam restarts with slightly different settings
        restarts = 2
        for r in range(restarts):
            iters = int(max_adam_iters / restarts)
            cand, val = _adam_smooth_optimize(best_seq, n, deadline,
                                              total_iters=iters,
                                              init_lr=0.05,
                                              p0=5.0, p_max=200.0)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Adam restart {r+1}] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – prolonged simulated‑annealing polishing.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long simulated‑annealing polishing ---")
        remaining = deadline - time.time()
        max_sa_iters = int(min(2_000_000, remaining * 1800))
        cand, val = _simulated_annealing_long(best_seq, n, deadline,
                                              max_iters=max_sa_iters)
        if val < best_val:
            best_val, best_seq = val, cand
            print(f"[SA] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – tiny EG sweep for final stabilisation.
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_it = int(min(50_000, remaining * 2000))
        if max_it > 0:
            cand, val = _expgrad_optimize(best_seq, n, deadline,
                                          max_iters=max_it,
                                          eta0=0.25)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Final EG] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
