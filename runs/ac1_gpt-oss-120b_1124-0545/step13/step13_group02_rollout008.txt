def solve_lp(f, M, active_idxs):
    n = len(f)
    rows = []
    for k in active_idxs:
        row = np.zeros(n)
        # coefficient for g_j is f_{k - j} if within bounds
        lo = max(0, k - (n-1))
        hi = min(k, n-1)
        # For j in lo..hi: coeff = f_{k - j}

        #    row[j] = f[k - j]
        # Using slices:
        # we can compute: idx = np.arange(lo, hi+1)
        # coeffs = f[k - idx]
        # row[idx] = coeffs
        rows.append(row)
    A_ub = np.vstack(rows) if rows else None
    b_ub = np.full(len(active_idxs), M)
    c = -np.ones(n)
    bounds = [(0, None)]*n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None
    g = res.x
    return g

"""
Propose a non‑negative step function (list of floats) that minimises the
official ``evaluate_sequence`` objective.

The search consists of four stages:

0️⃣  Seed generation – a large, diverse library of analytic windows,
    random Dirichlet draws, low‑frequency cosine sums and the previous
    best solution (if any).  All seeds are ℓ₁‑normalised (∑ = 1) and the
    ten best according to the true surrogate 2·n·max(conv) are kept.

1️⃣  Linear‑programming “mirror‑step’’ – for the current best vector *f* we
    compute the active convolution constraints (those attaining the
    current maximum).  A tiny LP maximising the ℓ₁‑mass of a new vector
    *g* while respecting the same constraints (f⋆g ≤ M) is solved.
    The extreme point *g* is mixed into *f* with a small step size
    (convex combination).  Repeating this quickly reduces the peak of
    the self‑convolution.

2️⃣  Smooth optimisation – a soft‑max Adam optimiser (smooth
    approximation of the max) refines the best vector obtained from the
    LP stage.

3️⃣  Polish – a long‑run exponential‑gradient (mirror‑descent) run and a
    short simulated‑annealing pass finish the search.

All stages run until the global deadline (≈ 1000 s) is reached.
The final sequence is stored in ``height_sequence_1`` for reuse.
"""

import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(p,p)) for a distribution p (∑p=1)."""
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse pool of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    seeds.append(_gaussian(n, 0.2))
    seeds.append(_gaussian(n, 0.3))
    seeds.append(_gaussian(n, 0.4))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final normalisation safety‑check
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

# Linear‑programming mirror step

def _active_constraints(conv: np.ndarray, M: float,
                        max_keep: int = 12,
                        rel_tol: float = 1e-6) -> np.ndarray:
    """
    Return indices of convolution entries that are “active’’
    (within `rel_tol` of the current maximum).  At most `max_keep`
    indices are returned, preferring the largest values.
    """
    idx = np.where(conv >= M * (1.0 - rel_tol))[0]
    if idx.size > max_keep:
        # keep the top‑max_keep entries by value
        top = np.argsort(-conv[idx])[:max_keep]
        idx = idx[top]
    return idx


def _solve_lp_for_g(f: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   1ᵀ g
        subject to f ⋆ g ≤ M   (only for active_idx)
                    g ≥ 0
    The coefficients are taken from a *slightly* regularised version of f
    (very tiny value added so that the LP is always bounded).
    Returns the optimal g (still ℓ₁‑unnormalised) or ``None`` on failure.
    """
    n = f.shape[0]
    # avoid exact zeros – they would make the LP unbounded
    f_pos = np.maximum(f, 1e-12)

    rows = []
    for k in active_idx:
        row = np.zeros(n, dtype=np.float64)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # set row[lo:hi+1] = f_pos[k - lo : k - hi - 1 : -1]
        j = np.arange(lo, hi + 1, dtype=np.int64)
        row[j] = f_pos[k - j]
        rows.append(row)

    if not rows:
        return None

    A_ub = np.vstack(rows)
    b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise sum(g)  <=> minimise -sum(g)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15.0})
    except Exception:               # pragma: no cover – safety net
        return None

    if res.success:
        g = res.x
        # If the LP was technically unbounded but a feasible point was found,
        # we fall back to a uniform direction.
        if np.all(g == 0):
            g = _uniform(n)
        return g
    else:
        return None


def _lp_mirror_step(start: np.ndarray,
                    n: int,
                    deadline: float,
                    max_outer_iters: int = 30,
                    step_size: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Repeatedly improve ``start`` by the LP‑mirror procedure described in the
    paper.  Returns the best vector found and its surrogate value.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_outer_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        M = np.max(conv)

        # identify the (few) active constraints
        active = _active_constraints(conv, M, max_keep=12, rel_tol=1e-6)

        g = _solve_lp_for_g(x, M, active)
        if g is None:
            # LP failed – fall back to a tiny random perturbation
            rng = np.random.default_rng()
            g = _random_dirichlet(n, 1.0, rng)

        # Normalise g to sum = 1 (scaling down never violates the constraints)
        s = g.sum()
        if s <= 0:
            g = _uniform(n)
        else:
            g = g / s

        # convex combination (mirrored update)
        t = step_size
        x_new = (1.0 - t) * x + t * g
        # project back onto the simplex (numerical safety)
        x_new = _proj_simplex(x_new)

        new_val = _objective_surrogate(x_new)
        if new_val < best_val:
            best_val, best_x = new_val, x_new.copy()
            print(f"[LP] iter={it:3d} surrogate={best_val:.6f}")

        # accept if we got an improvement, otherwise reduce step size
        if new_val < _objective_surrogate(x):
            x = x_new
        else:
            t *= 0.5
            if t < 1e-4:
                break

    return best_x, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent on simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative update)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # avoid overflow/underflow
        x = x * np.exp(delta)

        # project onto the simplex
        s = x.sum()
        if s <= 0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val

# Soft‑max + Adam optimiser (smooth approximation of max)
# ----------------------------------------------------------------------
def _adam_softmax_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    alpha_start: float = 20.0,
    alpha_end: float = 2000.0,
    lr: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        L(p) = (1/α)·log Σₖ exp(α·(p*p)[k]).
    α grows linearly from ``alpha_start`` to ``alpha_end``.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    N = _next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- convolution p * p (full) -----
        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        # ----- soft‑max over the convolution -----
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        # ----- correlation p ⋆ soft (gradient of L) -----
        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]      # = Σ_j p_j·soft[i+j]
        grad = 2.0 * corr                                  # ∂L/∂p

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # ----- projection onto simplex -----
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if t % 20000 == 0:
                print(f"[Adam] it={t:7d} best={best_val:.6f} α={alpha:.1f}")

    return best_x, best_val

# Simulated‑annealing polish (unchanged from previous solution)
# ----------------------------------------------------------------------
def _simulated_annealing(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        # two move types
        if rng.random() < 0.6:
            # reshuffle a random subset (size 3‑12)
            k_sub = rng.integers(3, 13)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    the official ``evaluate_sequence`` objective within ``budget_s`` seconds.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0     # safety margin

    # ------------------------------------------------------------------
    # Length selection – larger n gives a lower theoretical bound.
    # n = 4000 works well; feel free to increase if time permits.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 0 – diverse seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # evaluate seeds with the fast surrogate and keep the best few
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Initialise from the very best seed
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Linear‑programming mirror step
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- LP mirror improvement (multiple passes) ---")
        # we give the LP stage a generous slice of the remaining budget
        remaining = deadline - time.time()
        # approx 0.2 s per outer iteration → about 150 iterations possible
        max_outer = max(5, int(min(30, remaining * 10)))
        best_seq, best_val = _lp_mirror_step(best_seq, n, deadline,
                                            max_outer_iters=max_outer,
                                            step_size=0.25)
        print(f"[LP] best surrogate after mirror steps = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Soft‑max Adam optimisation (multiple starts)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Soft‑max Adam optimisation (multiple starts) ---")
        for idx, seed_vec in enumerate(best_seeds[:4]):
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            # allocate a fraction of the remaining time per start
            max_iters = int(min(120_000, remaining * 120))
            cand, val = _adam_softmax_optimize(
                seed_vec, n, deadline, max_iters=max_iters,
                alpha_start=20.0, alpha_end=1800.0, lr=0.12
            )
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Adam] seed #{idx+1} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Long‑run Exponential‑Gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long‑run Exponential‑Gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 1500))
        best_seq, best_val = _expgrad_optimize(
            best_seq, n, deadline, max_iters=max_iters, eta0=0.25
        )
        print(f"[Long EG] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(600_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(
            best_seq, n, deadline, max_iters=max_iters
        )
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse by the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
