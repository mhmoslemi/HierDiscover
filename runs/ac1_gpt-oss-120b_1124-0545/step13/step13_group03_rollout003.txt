def _adam_optimize(start, deadline, init_lr=0.05, max_iters=2000000):
    a = start.copy()
    best_a = a.copy()
    best_val = _objective_fft(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t=0
    p = 5.0
    p_factor = 2.0
    p_step = 50000
    beta1=0.9
    beta2=0.999
    eps=1e-8
    while time.time() < deadline and t < max_iters:
        # compute conv and max
        conv = _conv_fft(a, a)
        max_c = conv.max()
        cur_val = 2.0 * len(a) * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # compute softmax weights
        shifted = conv - max_c
        exp_arg = p * shifted
        exp_arg = np.clip(exp_arg, -700, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_c).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum
        # gradient
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)
        grad = 2.0 * grad_full[len(a)-1:len(a)-1+len(a)] 
        # Adam update
        t += 1
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad*grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / len(a))
        if t % p_step == 0:
            p *= p_factor
        # possibly adjust learning rate schedule: reduce after many steps
    return best_a, best_val
```

"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

within a 1000 s budget.

Algorithm
---------
1. **Shape library** – deterministic windows (uniform, triangular, sine,
   Gaussian, power, beta, Tukey, exponential) and several random draws
   (Dirichlet, exponential, low‑frequency cosine sums).  All are
   ℓ₁‑normalised.

2. **Initial refinement** – a short exponentiated‑gradient (mirror‑descent)
   on the simplex for a handful of the best seeds.

3. **Heavy Adam optimisation** – minimise the smooth surrogate
   `F_p(a) = (1/p)·log Σ exp(p·conv_k(a))` using Adam with a
   time‑adaptive schedule for the smoothness parameter `p`.  This drives
   the max‑convolution down whilst keeping the vector on the simplex.

4. **Local hill‑climbing** – random mass‑transfer moves accepted if they
   improve the exact objective (or with a tiny Metropolis probability).
   This often finds a few last percent‑points of improvement.

5. **Final evaluation** – the best vector is fed to the official
   `evaluate_sequence` routine, printed and stored globally as
   ``height_sequence_1`` for possible reuse.

The implementation uses FFT‑based convolutions (O(n log n)) and
periodically checks the deadline, guaranteeing termination before the
budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities (FFT‑based convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (fast for large n)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_fft(a: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant) for a ℓ₁‑normalised vector:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection (diverse, ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.2, 0.3, 0.4):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # keep previous solution (if any)
    if previous is not None and previous.size == n:
        arr = previous.astype(np.float64)
        if arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final safety – guarantee ℓ₁‑normalisation
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        clean.append(v / s if s > 0 else _uniform(n))
    return clean


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror‑descent on simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, deadline: float,
                      max_iters: int = 250_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    n = start.shape[0]
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_fft(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift k
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # diminishing stepsize (√t schedule)
        eta = eta0 / math.sqrt(it)

        # multiplicative KL‑update
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # simplex projection
        x = _proj_simplex(x)

        cur_val = _objective_fft(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# Adam optimiser on log‑sum‑exp smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray, deadline: float,
                   init_lr: float = 0.05,
                   max_total_iters: int = 2_000_000,
                   p0: float = 5.0,
                   p_factor: float = 2.0,
                   p_step: int = 70_000) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth surrogate
        F_p(a) = (1/p)·log Σ exp(p·conv_k(a))
    The surrogate becomes the true max as p → ∞.
    """
    a = start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective_fft(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    p = p0

    while time.time() < deadline and t < max_total_iters:
        # ---- current surrogate (exact max) ---------------------------------
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # ---- soft‑max weights ---------------------------------------------
        shifted = conv - max_conv          # ≤ 0
        exp_arg = p * shifted
        exp_arg = np.clip(exp_arg, -700, 0)  # avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:                 # extreme under‑flow – fallback
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ---- gradient of the smooth surrogate ------------------------------
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)   # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # ---- Adam update ---------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        a = _proj_simplex(a)

        # ---- schedule for p ------------------------------------------------
        if t % p_step == 0:
            p *= p_factor

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing polishing (random mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start: np.ndarray, deadline: float,
                max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    n = start.shape[0]
    a = start.copy()
    best = a.copy()
    best_val = _objective_fft(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iters:
        i, j = rng.integers(0, n, size=2, replace=False)
        if a[i] <= 1e-12:
            it += 1
            continue
        # transfer a small fraction of the mass at i to j
        delta = rng.random() * min(a[i], 0.015)   # ≤ 1.5 % of the source mass
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        val = _objective_fft(cand)
        if val < best_val:
            best_val = val
            best = cand.copy()
            a = cand
        else:
            # occasional Metropolis move to escape shallow local minima
            T = max(0.02 * best_val, 1e-6)
            if rng.random() < math.exp(-(val - best_val) / T):
                a = cand
        it += 1
    return best, best_val


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the allotted time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # small safety margin to avoid overrunning the judge
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Length of the candidate – a power of two makes the FFT cheap.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # 0. Seed generation (large, diverse pool)
    # --------------------------------------------------------------
    print(f"\n[Stage 0] generating seeds (n={n})")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None
    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"generated {len(seeds)} seeds")

    # keep the best few according to the exact surrogate
    scored = [(s, _objective_fft(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # --------------------------------------------------------------
    # 1. Quick Exponentiated‑Gradient refinement on the top seeds
    # --------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_fft(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n[Stage 1] quick EG refinement")
        for idx, seed_vec in enumerate(best_seeds[:5]):
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            # allocate a modest number of iterations per seed
            max_iters = int(min(120_000, remaining * 1400))
            cand, val = _expgrad_optimize(seed_vec, deadline,
                                          max_iters=max_iters,
                                          eta0=0.30)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Quick EG] seed #{idx+1} improved → {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Heavy Adam optimisation (log‑sum‑exp smoothing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n[Stage 2] Adam smooth optimisation")
        remaining = deadline - time.time()
        max_adam_iters = int(min(2_000_000, remaining * 1800))
        best_seq, best_val = _adam_optimize(best_seq, deadline,
                                            init_lr=0.05,
                                            max_total_iters=max_adam_iters,
                                            p0=5.0,
                                            p_factor=2.0,
                                            p_step=70_000)
        print(f"[Adam] best surrogate after polishing = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Local hill‑climbing polishing (random mass transfers)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n[Stage 3] hill‑climbing polishing")
        remaining = deadline - time.time()
        max_hc_iters = int(min(400_000, remaining * 2000))
        best_seq, best_val = _hill_climb(best_seq, deadline,
                                         max_iters=max_hc_iters)
        print(f"[Hill‑climb] final surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final official evaluation
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("warning: evaluate_sequence raised", exc)
        true_val = None

    # store for possible reuse in later runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
