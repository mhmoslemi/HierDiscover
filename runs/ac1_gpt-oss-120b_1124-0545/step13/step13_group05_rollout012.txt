"""
Better step‑function optimizer

The objective is               2·n·max(conv(a,a)) / (sum(a)**2)

Because the ratio is scale‑invariant we work with ℓ₁‑normalised vectors
(∑a = 1) and minimise the surrogate 2·n·max(conv(a,a)).  The optimiser
consists of three stages:

0.  A very rich pool of analytic “seed’’ families (uniform, triangular,
    sine‑window, beta, exponential, exponential with offset, random
    Dirichlet / exponential draws).  The best few seeds are kept.

1.  Long Adam optimisation on the *exact* max‑convolution (the true
    sub‑gradient of the max‐operator) – this converges faster than the
    log‑sum‑exp surrogate used before.

2.  An aggressive simulated‑annealing polishing stage that performs large
    random mass‑reshuffles together with a cool‑down schedule.  This
    explores the neighbourhood of the best point found by Adam.

All phases respect the global 1000 s deadline; they continuously check the
remaining time and stop early if necessary.  The final vector is returned
as a plain list of floats and stored in the global name ``height_sequence_1``
for possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_shape(n: int, r: float) -> np.ndarray:
    """Monotone exponential a_i ∝ r**i, ℓ₁‑normalised."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_offset_shape(n: int, offset: int, r: float) -> np.ndarray:
    """Zero prefix of length *offset* then an exponential tail."""
    if offset >= n:
        return _uniform(n)
    tail_len = n - offset
    if r <= 0.0:
        tail = np.full(tail_len, 1.0 / tail_len, dtype=np.float64)
    else:
        i = np.arange(tail_len, dtype=np.float64)
        tail = np.power(r, i)
        tail /= tail.sum()
    a = np.concatenate([np.zeros(offset, dtype=np.float64), tail])
    a /= a.sum()
    return a


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _adam_subgrad_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑softmax parametrisation using the *exact* max‑convolution
    sub‑gradient.  Returns the best vector seen and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits (stay on simplex via softmax)
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 1.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # simplex point
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[AdamSub] it={it:8d} best={cur_val:.6f}")

        # --- true sub‑gradient of max(conv) w.r.t. a -----------------
        # gradient of max(conv) is 2 * a_rev on the overlapping part
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # transform gradient to log‑softmax space
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam update
        lr = lr0 / math.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray,
                                  n: int,
                                  deadline: float,
                                  max_iters: int = 1_500_000,
                                  init_temp_factor: float = 0.07,
                                  cooling: float = 0.99995,
                                  min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Aggressive annealing that repeatedly reshuffles mass on a random sub‑set.
    Returns the best vector and its surrogate value.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * cooling, min_temp)

        # choose a move
        if rng.random() < 0.65:                     # block reshuffle (size 3‑12)
            k = rng.integers(3, 13)
            idx = rng.choice(n, size=k, replace=False)
            tot = cur[idx].sum()
            if tot > 0.0:
                new_alloc = rng.dirichlet(np.full(k, 1.0)) * tot
                new = cur.copy()
                new[idx] = new_alloc
            else:
                continue
        else:                                        # small pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.random() * min(cur[i], 0.03)      # ≤3 % of source mass
            if delta <= 0.0:
                continue
            new = cur.copy()
            new[i] -= delta
            new[j] += delta

        new_val = _objective_surrogate(new)
        delta_val = new_val - cur_val
        if delta_val < 0.0 or rng.random() < math.exp(-delta_val / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # classic deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_power(n, p=0.8))
    seeds.append(_power(n, p=1.2))

    # beta‑grid (α,β) – coarse but captures a wide family
    alphas = np.geomspace(0.2, 5.0, 7)
    betas = np.geomspace(0.2, 5.0, 7)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, alpha=a, beta=b))

    # exponential sweep (fine around theoretical optimum)
    base_eps = 2.0 / n
    eps_factors = np.linspace(0.4, 3.2, 30)           # a bit wider search
    for fac in eps_factors:
        eps = fac * base_eps
        r = max(0.0, 1.0 - eps)
        seeds.append(_exponential_shape(n, r))

    # offset exponential families (larger offset range)
    offsets = [int(p * n) for p in (0.00, 0.03, 0.06, 0.10, 0.15, 0.20, 0.30, 0.40)]
    for off in offsets:
        L = n - off
        if L <= 0:
            continue
        base_eps_off = 2.0 / L
        for fac in np.linspace(0.5, 3.0, 12):
            eps = fac * base_eps_off
            r = max(0.0, 1.0 - eps)
            seeds.append(_exponential_offset_shape(n, off, r))

    # random draws – Dirichlet with several concentrations
    for conc in (0.05, 0.2, 0.6, 1.2, 3.0):
        for _ in range(4):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # random exponentials (different scales)
    for scale in (0.5, 1.0, 2.0):
        for _ in range(3):
            a = rng.exponential(scale=scale, size=n)
            a /= a.sum()
            seeds.append(a)

    # include the previously discovered solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final clean‑up – guarantee ℓ₁‑normalisation and non‑negativity
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

        max_total_iters = int(min(1_500_000, remaining * 5000))
        it_per_seed = max_total_iters // len(best_seeds)
        print(f"\n--- Adam‑subgradient optimisation (total it≈{max_total_iters}) ---")
        for idx, seed_vec in enumerate(best_seeds):
            seq_adam, val_adam = _adam_subgrad_optimize(
                seed_vec, n, overall_deadline, max_iters=it_per_seed
            )
            if seq_adam is not None and val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[Adam‑seed{idx+1}] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Simulated annealing polishing (longer than baseline)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 15.0:
        max_iters_sa = int(min(2_500_000, remaining * 2500))
        print(f"\n--- Simulated annealing polishing (max_iters={max_iters_sa}) ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, overall_deadline, max_iters=max_iters_sa,
            init_temp_factor=0.07, cooling=0.99995
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
