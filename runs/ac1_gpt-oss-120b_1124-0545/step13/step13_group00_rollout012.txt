"""
Propose a step‑function (list of non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where conv is the full linear convolution (numpy.convolve).

Algorithm
---------

1. **Rich seed pool** – Analytic windows (sine, Kaiser, Gaussian, …),
   a family of geometric‑decay vectors, many random Dirichlet/exponential
   draws, plus the previous best sequence (if any).
2. **Cheap surrogate ranking** – All seeds are ℓ₁‑normalised, therefore the
   objective reduces to ``2·n·max(conv)``.  The best few seeds are kept.
3. **Continuation L‑BFGS** – Optimise a smooth log‑sum‑exp approximation
   of the max‑convolution.  The inverse temperature β is increased
   gradually (up to 10 000) and each β‑stage runs a few thousand L‑BFGS‑B
   iterations.
4. **Long exponentiated‑gradient (mirror descent)** – Exact max‑convolution
   gradient with the simplex enforced via the softmax (log‑space) map.
   Millions of iterations are performed while respecting the remaining
   time budget.
5. **Optional sub‑gradient polishing** – Projected sub‑gradient descent
   for a final fine‑tune.
6. **Final evaluation** – The true ``evaluate_sequence`` function is called
   once on the best candidate found.

All steps respect the global time limit (default 1000 s).  The best
sequence is stored in the global variable ``height_sequence_1`` and also
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact and surrogate objectives (vectors are assumed ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """True objective 2·n·max(conv) / sum² (any scaling is irrelevant)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _surrogate(seq: np.ndarray) -> float:
    """Surrogate = exact objective for an ℓ₁‑normalised vector (sum=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families of ℓ₁‑normalised vectors (used as seeds)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    if beta <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)               # modified Bessel function of the first kind
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geometric(n: int, r: float) -> np.ndarray:
    """Monotonically decreasing geometric sequence r**i, ℓ₁‑normalised."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed collection (richer than the previous version)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                  rng: np.random.Generator,
                  previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    for beta in (2, 4, 6, 8, 10, 12, 16):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.10, 0.45, 9):
        seeds.append(_gaussian(n, sigma))

    # Geometric family – dense sweep
    for r in np.linspace(0.80, 0.9999, 30):
        seeds.append(_geometric(n, r))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(10):
        seeds.append(_random_exponential(n, rng))

    # Include the previous solution if available
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise robustly (guard against numerical issues)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [0.5, 2, 10, 40, 150, 500, 2000, 5000, 10000]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on simplex
            conv = _conv_fft(a, a)                     # full convolution
            max_c = np.max(conv)

            # log‑sum‑exp smoothing
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp                     # softmax over conv entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient w.r.t. logits via Jacobian of softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached at β={beta_val:.0f}")
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val:.0f}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (softmax projection)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # learning rate schedule (standard 1/√t decay)
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[ExpGrad] it={it:,d}  best_surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent (projected on simplex)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[SubGrad] it={it:,d}  best_surrogate={best_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step size
        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing style refinement (log‑space perturbations)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Simple SA on log‑softmax parameters."""
    rng = np.random.default_rng(12345)
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # linear cooling
        progress = it / max_iters
        T = 0.10 * (1.0 - progress) + 1e-6
        sigma = 0.30 * (1.0 - progress) + 1e-5

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _surrogate(proposal_seq)

        # Metropolis acceptance
        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 200_000 == 0:
            print(f"[SA] it={it:,d}  best_surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence`` within the allotted time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # safety margin

    # --------------------------------------------------------------
    # 0. Problem size
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}, budget={budget_s}s) ===")

    # --------------------------------------------------------------
    # 1. Seed generation
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seed vectors.")

    # cheap surrogate evaluation (exact for ℓ₁‑normalised vectors)
    evaluated = [(vec, _surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 15
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best surrogate among seeds (top {top_k}):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # --------------------------------------------------------------
    # 2. Continuation L‑BFGS on the most promising seeds
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS runs.")
            break

        print(f"\n--- L‑BFGS run on seed #{idx+1} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         max_iter_per_beta=6000)
        val_opt = _surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Long exponentiated‑gradient polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp = int(min(1_200_000, remaining * 0.70))
        if max_exp > 0:
            print(f"\n--- Exponentiated‑gradient polishing (max_iters={max_exp}) ---")
            seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_exp)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Sub‑gradient polishing (optional)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(600_000, remaining * 0.20))
        if max_sub > 0:
            print(f"\n--- Sub‑gradient polishing (max_iters={max_sub}) ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Simulated‑annealing style refinement (optional)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(500_000, remaining * 0.20))
        if max_sa > 0:
            print(f"\n--- Simulated‑annealing refinement (max_iters={max_sa}) ---")
            seq_sa, val_sa = _simulated_annealing(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_sa)
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[SA] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
