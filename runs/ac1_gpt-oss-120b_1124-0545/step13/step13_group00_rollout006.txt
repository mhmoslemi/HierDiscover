"""
Propose a candidate step‑function (non‑negative heights) that minimizes

    2·n·max(conv(a, a)) / (sum(a)²)

using a multi‑stage optimisation pipeline:

1️⃣  Build a large, diverse pool of ℓ₁‑normalised seed vectors (analytic windows,
    random Dirichlet draws, tiled short patterns, etc.).
2️⃣  Run a continuation L‑BFGS optimiser on a smooth log‑sum‑exp approximation
    of the max‑convolution.  A coarse β‑schedule is used first, followed by a
    high‑β fine‑tuning stage.
3️⃣  Apply exponentiated‑gradient (mirror descent) on the exact max‑convolution.
4️⃣  Refine with projected sub‑gradient descent.
5️⃣  Perform a simulated‑annealing style random walk in log‑it space with a
    slowly decreasing temperature and step‑size.
6️⃣  Return the best sequence found (as a plain list of floats).

The routine respects a global time budget (default 1000 s) and works for any
sequence length.  By default it uses n=2000 (smaller than the previous
4000‑element run) to allow many more optimisation iterations within the time
budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (needed for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector on the simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluate_sequence objective (no clamping effects for our ranges).
    Returns 2·n·max(conv(seq,seq)) / (sum(seq)²).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate used during optimisation: assumes sum(seq)=1, so returns
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families of ℓ₁‑normalised vectors (used as seeds)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # modified Bessel (approx)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _tile_pattern(n: int, pattern: np.ndarray) -> np.ndarray:
    """Repeat a short pattern to length n (truncated) and normalise."""
    repeats = (n + len(pattern) - 1) // len(pattern)
    tiled = np.tile(pattern, repeats)[:n].astype(np.float64)
    s = tiled.sum()
    return tiled / s if s > 0 else _uniform(n)


def _downsample(vec: np.ndarray, target_len: int) -> np.ndarray:
    """Simple block‑averaging down‑sample to target_len elements."""
    src_len = len(vec)
    if src_len == target_len:
        return vec.copy()
    factor = src_len / target_len
    out = np.empty(target_len, dtype=np.float64)
    for i in range(target_len):
        start = int(i * factor)
        end = int((i + 1) * factor)
        out[i] = vec[start:end].mean() if end > start else vec[start]
    s = out.sum()
    return out / s if s > 0 else _uniform(target_len)


# ----------------------------------------------------------------------
# Seed collection (rich pool)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    seeds.append(_double_gaussian(n, sigma_frac=0.10, shift_frac=0.30))
    seeds.append(_double_gaussian(n, sigma_frac=0.15, shift_frac=0.25))
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet / exponential draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(4):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Tiled short patterns (periodic seeds)
    for block in (5, 8, 12, 20):
        pat = _random_dirichlet(block, 0.3, rng)
        seeds.append(_tile_pattern(n, pat))

    # Mixtures of uniform with a low‑freq curve
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (down‑sample / up‑sample if needed)
    if previous is not None:
        if previous.size != n:
            # try to adapt size by averaging or repeating
            if previous.size > n:
                prev_adj = _downsample(previous, n)
            else:
                # repeat the short vector to reach length n
                repeats = (n + previous.size - 1) // previous.size
                prev_adj = np.tile(previous, repeats)[:n]
            seeds.append(prev_adj / prev_adj.sum())
        else:
            seeds.append(previous / previous.sum())

    # Normalise everything robustly
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    The variable is a log‑softmax (logits) vector; the returned vector lies
    on the probability simplex.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser not available.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]

    # initialise logits
    logits = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(logv):
            a = _softmax(logv)                     # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smoothed objective (log‑sum‑exp)
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # weights = softmax(beta * conv)
            w = exp_beta / sum_exp

            # gradient w.r.t a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient w.r.t logits via Jacobian of softmax
            dot = np.dot(grad_a, a)
            grad_logv = a * (grad_a - dot)
            return obj, grad_logv

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=logits,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            logits = res.x
        except KeyboardInterrupt:
            # ran out of time inside L‑BFGS
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(logits)
    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (softmax‑parameterised)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # learning rate schedule (standard 1/√t)
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Simple sub‑gradient descent on the true max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = _objective_surrogate(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step size
        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing style refinement (logit space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Logit‑space Metropolis‑Hastings walk with linear cooling."""
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    rng = np.random.default_rng()

    # schedules
    init_T = 0.08
    final_T = 1e-6
    init_sigma = 0.20
    final_sigma = 1e-4

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = init_T * (1.0 - progress) + final_T
        sigma = init_sigma * (1.0 - progress) + final_sigma

        # occasional large jump (1 % probability)
        if rng.random() < 0.01:
            proposal_seq = _random_dirichlet(n, 1.0, rng)
        else:
            proposal_logits = logits + sigma * rng.normal(size=n)
            proposal_seq = _softmax(proposal_logits)

        proposal_val = _objective_surrogate(proposal_seq)
        delta = proposal_val - cur_val

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            logits = np.log(proposal_seq + 1e-30)
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    evaluation function within the allotted time budget.
    Returns the best sequence as a list of Python floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # safety margin

    # Problem size – we deliberately use a smaller n to afford many iterations.
    n = 2000
    print(f"\n--- Problem size: n = {n} ---")

    # ------------------------------------------------------------------
    # 1. Seed generation
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            prev_vec = prev_arr
        except Exception:
            prev_vec = None

    seeds = _collect_seeds(n, rng, prev_vec)
    print(f"Collected {len(seeds)} seed vectors.")

    # cheap surrogate evaluation (exact for ℓ₁‑normalised vectors)
    evaluated = [(v, _objective_surrogate(v)) for v in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("Best surrogate after seeding (top 12):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 2. Continuation L‑BFGS on the best seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    # modest L‑BFGS runs on the top seeds
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before L‑BFGS on all seeds.")
            break

        print(f"\n--- L‑BFGS run on seed #{idx+1} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=[1, 5, 20, 80, 200, 500, 1000],
                                         max_iter_per_beta=1500)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

        # brief exponentiated‑gradient polishing after each run
        if time.time() < deadline - 10.0:
            exp_seq, exp_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=80_000)
            if exp_val < global_best_val:
                global_best_seq = exp_seq.copy()
                global_best_val = exp_val
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Extended high‑β L‑BFGS refinement on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        print("\n--- High‑β L‑BFGS refinement ---")
        high_beta_schedule = [2000, 5000, 10000, 20000]
        seq_refined = _lbfgs_smooth_optimize(global_best_seq,
                                             n,
                                             deadline,
                                             beta_schedule=high_beta_schedule,
                                             max_iter_per_beta=3000)
        val_refined = _objective_surrogate(seq_refined)
        print(f"[High‑β L‑BFGS] surrogate = {val_refined:.6f}")
        if val_refined < global_best_val:
            global_best_seq = seq_refined.copy()
            global_best_val = val_refined
            print(f"[Update] after high‑β refinement = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient descent (more iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n--- Sub‑gradient descent ---")
        sub_seq, sub_val = _subgradient_opt(global_best_seq,
                                            n,
                                            deadline,
                                            max_iters=250_000)
        if sub_val < global_best_val:
            global_best_seq = sub_seq.copy()
            global_best_val = sub_val
            print(f"[SubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Simulated‑annealing style refinement (long run)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        # allocate roughly 0.5 × remaining iterations (capped)
        max_sa = int(min(800_000, remaining * 700))
        print(f"\n--- Simulated‑annealing refinement (max_iters={max_sa}) ---")
        sa_seq, sa_val = _simulated_annealing_opt(global_best_seq,
                                                  n,
                                                  deadline,
                                                  max_iters=max_sa)
        if sa_val < global_best_val:
            global_best_seq = sa_seq.copy()
            global_best_val = sa_val
            print(f"[SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
