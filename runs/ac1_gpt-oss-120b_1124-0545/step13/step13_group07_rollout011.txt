def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate_obj(seq: np.ndarray) -> float:
    """Fast surrogate - 2·n·max(conv(seq,seq))  (seq is ℓ₁‑normalised)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_triangular(n: int) -> np.ndarray:
    """U‑shaped: high at the edges, low in the centre."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine_density(n: int) -> np.ndarray:
    """Discrete version of the beta(½,½) – arcsine shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i * (n - i + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random 3‑segment piecewise linear shape."""
    breaks = np.sort(rng.choice(n, size=3, replace=False))
    heights = rng.random(3) + 0.01
    x = np.arange(n, dtype=np.float64)
    a = np.interp(x, breaks, heights)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few randomly positioned Gaussians."""
    a = np.zeros(n, dtype=np.float64)
    num = rng.integers(2, 6)
    for _ in range(num):
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.02 * n, 0.15 * n)
        x = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, nf: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth curve built from a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, nf + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

        # ---- smooth max via log‑sum‑exp ----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 2n‑1
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 2_000_000,
                                 active_frac: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) descent on the *exact* max of the
    convolution, but the sub‑gradient is built from *all* indices whose
    value lies within ``active_frac`` of the maximum (default 1 %).
    This spreads the corrective pressure across the whole plateau.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    lr0 = 0.25
    power = 0.28

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # Identify plateau indices
        thresh = cur_max * (1.0 - active_frac)
        active = np.where(conv >= thresh)[0]

        # Build sub‑gradient as sum of contributions from all active indices
        g = np.zeros_like(a)
        rev_a = a[::-1]
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            # each pair contributes 2*a_i*a_j to the derivative w.r.t a_i
            g[idx_i] += 2.0 * a[rev_idx]

        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _lp_refine(seq: np.ndarray, deadline: float,
               max_iters: int = 30, tau_start: float = 0.25,
               tau_end: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Perform several linear‑programming polishing steps.
    The step size τ is linearly decreased from ``tau_start`` to ``tau_end``.
    """
    f = seq.copy()
    n = len(f)

    # pre‑compute τ schedule
    taus = np.linspace(tau_start, tau_end, max_iters)

    for it, tau in enumerate(taus, 1):
        if time.time() > deadline:
            break

        M = _conv_fft(f, f).max()

        # Build convolution matrix for current f (sparse)
        # The matrix maps g → conv(f,g) (size (2n‑1) × n)
        rows = np.empty(n * n, dtype=np.int32)
        cols = np.empty(n * n, dtype=np.int32)
        data = np.empty(n * n, dtype=np.float64)

        idx = 0
        for k in range(n):
            rows[idx:idx + n] = np.arange(k, k + n, dtype=np.int32)
            cols[idx:idx + n] = k
            data[idx:idx + n] = f
            idx += n

        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i

        # Solve LP; a time limit of 15 s per call keeps us inside the global budget
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 15})

        if not res.success:
            # If the LP fails we stop the polishing stage
            break

        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g <= sum_f + 1e-12:
            # No ℓ₁‑gain – further LP steps will not improve anything
            break

        # Convex combination and renormalisation
        f_new = (1.0 - tau) * f + tau * g
        f_new = f_new / f_new.sum()
        f = f_new

    return f, _surrogate_obj(f)


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 500_000,
                                 init_temp_factor: float = 0.06,
                                 min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """Lightweight SA with three simple move types."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate_obj(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            k = rng.integers(2, 14)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total == 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:  # move_type == 2
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total == 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _surrogate_obj(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Construct a diverse pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_inverse_triangular(n))
    seeds.append(_cosine_window(n))
    seeds.append(_arcsine_density(n))

    # Gaussian sweeps
    for sigma in np.linspace(0.04, 0.30, 9):
        seeds.append(_gaussian(n, sigma))

    # Power‑law sweeps
    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power_law(n, p))

    # Beta shape – include several extreme parameter pairs
    alphas = np.array([0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 3.0, 5.0])
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Exponential decay
    for lam in np.linspace(0.0002, 0.02, 12):
        seeds.append(_exponential_decay(n, lam))

    # Random Dirichlet draws with a broad range of concentrations
    concs = np.concatenate([np.logspace(-5, -1, 8), np.logspace(-1, 0.5, 6)])
    for conc in concs:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Mixtures of random Gaussians
    for _ in range(8):
        seeds.append(_gaussian_mixture(n, rng))

    # Piecewise linear random shapes
    for _ in range(8):
        seeds.append(_piecewise_linear(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 5, 8, 13):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mix uniform with low‑frequency curves
    for mix_alpha in (0.25, 0.5, 0.75):
        for nf in (3, 6, 9):
            curve = _lowfreq_curve(n, nf, rng)
            mixed = (1.0 - mix_alpha) * _uniform(n) + mix_alpha * curve
            mixed = mixed / mixed.sum()
            seeds.append(mixed)

    # Previous solution (if supplied)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0.0:
            seeds.append(avg / avg.sum())

    return seeds

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the official
    ``evaluate_sequence`` objective.  The best sequence is stored globally as
    ``height_sequence_1`` and also returned as a plain ``list[float]``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Choose a length – 4000 gives a good compromise between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load any previous solution (if the runner supplied one)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size != n:
                if previous.size > n:
                    previous = previous[:n]
                else:
                    previous = np.pad(previous, (0, n - previous.size), constant_values=0.0)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – generate a rich pool of seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _surrogate_obj(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"\nTop {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _surrogate_obj(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation on the smooth surrogate (short runs)
    # ------------------------------------------------------------------
    print("\n--- Adam optimisation (short runs) ---")
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate up to 0.12·remaining seconds for each Adam run (capped)
        max_iters = min(150_000, int(remaining * 0.12 / 0.1))
        print(f"[Adam] seed #{idx}, max_iters={max_iters}")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Long Adam run on the current best (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 0.5 / 0.1))
        print("\n--- Extended Adam refinement on global best ---")
        seq_adam2, val_adam2 = _adam_optimize(global_best_seq, n,
                                              deadline, max_iters=max_iters)
        if seq_adam2 is not None and val_adam2 < global_best_val:
            global_best_seq = seq_adam2.copy()
            global_best_val = val_adam2
            print(f"[Extended‑Adam] new global best = {val_adam2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Mirror‑descent with multi‑peak sub‑gradient
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent (multi‑peak) optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=2_000_000, active_frac=0.01
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Iterated LP polishing (more iterations than before)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Linear‑programming polishing (extended) ---")
        seq_lp, val_lp = _lp_refine(global_best_seq, deadline,
                                    max_iters=30, tau_start=0.25, tau_end=0.05)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[overall‑LP] new global best = {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – Final Adam run (short) to soak up remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 0.8 / 0.1))
        print("\n--- Final Adam polishing ---")
        seq_final_adam, val_final_adam = _adam_optimize(global_best_seq, n,
                                                        deadline, max_iters=max_iters)
        if seq_final_adam is not None and val_final_adam < global_best_val:
            global_best_seq = seq_final_adam.copy()
            global_best_val = val_final_adam
            print(f"[Final‑Adam] new global best = {val_final_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – Light SA polishing (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline, max_iters=500_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (only for reporting)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
