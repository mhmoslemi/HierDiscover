def _frank_wolfe_optimize(start_seq, n, deadline, max_iters=10, mix_coeff=0.5):
    a = start_seq.copy()
    best = a.copy()
    best_val = _objective(a)
    it = 0
    while it < max_iters and time.time() < deadline:
        conv = _conv_fft(a, a)
        M = np.max(conv)
        # construct linear constraints matrix
        A = _conv_matrix(a)  # shape (2*n-1, n)
        b = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        # solve LP: maximize sum(g) => minimization with negative c
        res = scipy.optimize.linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds,
                                    method='highs', options={'presolve': True})
        if not res.success:
            print("[FW] LP failed at iteration", it, "status:", res.message)
            break
        g = res.x
        s = g.sum()
        if s <= 0:
            break
        g_norm = g / s
        # mixing
        a = (1 - mix_coeff) * a + mix_coeff * g_norm
        a = a / a.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[FW] iter {it+1} improved best to {val:.6f}")
        it += 1
    return best, best_val

Propose a step‑function (non‑negative vector) that minimises
    2·n·max(conv(a,a)) / (sum(a))²

The optimiser works in four stages:

1.  **Seed generation** – a rich pool of ℓ₁‑normalised vectors:
    * analytic families (uniform, linear‑decrease, power‑law,
      β‑shape, exponential, low‑frequency sinusoids)
    * Dirichlet draws with several very small concentration parameters
    * the previously best solution (if any).

2.  **Smooth‑max L‑BFGS refinement** – each top seed is optimised
    with L‑BFGS on a log‑sum‑exp surrogate for the max‑convolution.
    The temperature is raised progressively up to 5000.

3.  **Frank‑Wolfe–style LP move** – given a current vector `a`,
    solve the linear programme
        maximise   Σ g_i
        s.t.       conv(a,g) ≤ M   (M = max(conv(a,a)))
                   g ≥ 0
    The solution `g` is ℓ₁‑normalised and mixed with `a`
    (mixing coefficient 0.5, then renormalised).  A few such
    iterations dramatically lower the maximal anti‑diagonal.

4.  **Fine‑tuning** – a short exponentiated‑gradient (mirror‑descent)
    run followed by lightweight simulated‑annealing.

The best sequence found before the global deadline is stored in the
global name ``height_sequence_1`` and returned as a plain list of
floats.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# FFT‑based full linear convolution
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Softmax (projection onto the simplex)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = n - i
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Surrogate objective (fast, identical ordering to official one)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – works on ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum
    smooth = shift + np.log(w_sum) / beta

    # grad of Σ w_k * (a*a)_k  w.r.t a  = 2 * (rev(a) ⊛ w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad_a = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad_a


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iter: int = 600,
                     temperature: float = 100.0) -> tuple[np.ndarray, float]:
    """L‑BFGS on the smooth‑max surrogate."""
    eps = 1e-30
    v = np.log(start_seq + eps)               # logits → simplex via softmax

    def fun(v):
        a = _softmax(v)
        smooth, grad_a = _smooth_max_and_grad(a, temperature)
        loss = 2.0 * n * smooth
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)           # chain rule through softmax
        return loss, (2.0 * n * grad_v).astype(np.float64)

    # honour the remaining budget – stop after max_iter or when time runs out
    try:
        res = scipy.optimize.minimize(
            fun,
            v,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
    except Exception:  # pragma: no cover
        return start_seq, _objective(start_seq)

    best_a = _softmax(res.x)
    return best_a, _objective(best_a)


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑conv objective
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 2_000_000,
                                 lr0: float = 0.3,
                                 power: float = 0.30) -> tuple[np.ndarray, float]:
    """Multiplicative update on the simplex using the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 1
    while it <= max_iters:
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:,d} best={cur_val:.6f}")

        # sub‑gradient for the active anti‑diagonal
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated Annealing – final polishing
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 2_000_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a random subset
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                  # split‑transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 200_000 == 0:
                    print(f"[SA] it={it:,d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming (Frank‑Wolfe) auxiliary
# ----------------------------------------------------------------------
def _conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Return a sparse (2n‑1) × n matrix A such that A @ g = conv(a, g)
    for any vector g (full linear convolution).
    The matrix consists of n constant diagonals whose values are the
    entries of a (reversed order on the anti‑diagonals).
    """
    n = len(a)
    m = 2 * n - 1
    # offsets from -(n-1) … 0 (inclusive)
    offsets = np.arange(-(n - 1), 1, dtype=int)
    # each diagonal receives the same constant value a[-offset]
    data = np.empty((len(offsets), n), dtype=np.float64)
    for idx, d in enumerate(offsets):
        data[idx, :] = a[-d]            # -d is in 0..n‑1
    A = sp.diags(data, offsets, shape=(m, n), format="csr")
    return A


def _lp_step(a: np.ndarray, M: float, deadline: float) -> np.ndarray | None:
    """
    Solve  maximise Σ g_i   s.t.   conv(a, g) ≤ M,   g ≥ 0 .
    Returns the ℓ₁‑normalised solution g (or None on failure).
    """
    n = len(a)
    A = _conv_matrix(a)                     # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    c = -np.ones(n, dtype=np.float64)       # minimise –Σg_i  == maximise Σg_i

    # Use the high‑performance HiGHS solver; abort if time runs out.
    try:
        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"time_limit": max(0.0, deadline - time.time())},
        )
    except Exception:                       # pragma: no cover
        return None

    if not res.success:
        return None
    g = res.x
    s = g.sum()
    if s <= 0.0:
        return None
    return g / s


def _frank_wolfe_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 8,
                          mix_coeff: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe style improvement: repeatedly solve the LP that
    maximises the ℓ₁‑norm of a feasible partner g and mix it into the
    current iterate.  The mixing coefficient is constant (0.5) – this
    yields a strong contraction of the maximal anti‑diagonal.
    """
    a = start_seq.copy()
    best = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = np.max(conv)

        g = _lp_step(a, M, deadline)
        if g is None:
            # LP failed – abort the FW phase.
            print("[FW] LP failed at iteration", it)
            break

        # convex combination and renormalise (still ℓ₁‑normalised)
        a = (1.0 - mix_coeff) * a + mix_coeff * g
        a = a / a.sum()

        val = _objective(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[FW] iter {it} improved best to {val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Seed generation (rich and diverse)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_linear_decreasing(n))

    # power‑law family (dense grid)
    for p in np.linspace(0.0, 4.0, 17):
        seeds.append(_power(n, p))

    # β‑shape – wide asymmetric grid (both ends can be heavy)
    alphas = np.concatenate((np.linspace(0.01, 0.2, 6), np.linspace(0.2, 1.0, 6)))
    betas = np.concatenate((np.linspace(0.01, 0.2, 6), np.linspace(0.2, 1.0, 6)))
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # exponential decay (fine grid)
    for lam in np.linspace(0.001, 0.020, 10):
        seeds.append(_exponential_decay(n, lam))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Dirichlet draws – very low concentration → spiky candidates
    for conc in (0.001, 0.005, 0.01, 0.05):
        for _ in range(4):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw ordering
            seeds.append(np.sort(raw)[::-1])      # descending
            seeds.append(np.sort(raw))            # ascending

    # a simple "U‑shaped" family (high at both ends, low in the centre)
    t = np.arange(n, dtype=np.float64)
    u_shape = (t + 1) ** (-0.7) * (n - t) ** (-0.7)
    u_shape = np.maximum(u_shape, 0.0)
    u_shape = u_shape / u_shape.sum()
    seeds.append(u_shape)

    # previous solution, if available
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Main optimiser entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the given
    `evaluate_sequence` objective within the supplied time budget.
    Returns the best sequence (list of floats) and stores it as the
    global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – a few thousand items give a good trade‑off.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load any previous solution for seeding.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – create a big and diverse seed pool.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 30
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # Initialise the global best.
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – L‑BFGS refinement on the smooth‑max surrogate.
    # ------------------------------------------------------------------
    temperature_schedule = [5, 10, 20, 40, 80, 200, 500, 1000, 2000, 5000]
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        seq = seed_vec.copy()
        for temp in temperature_schedule:
            if time.time() > deadline:
                break
            seq_opt, val_opt = _l_bfgs_optimize(
                seq, n, deadline, max_iter=600, temperature=temp
            )
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[L‑BFGS] seed #{idx}, temp={temp} → new best {val_opt:.6f}")
            seq = seq_opt   # warm‑start the next temperature

    # ------------------------------------------------------------------
    # Stage 2 – Frank‑Wolfe LP improvement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Frank‑Wolfe LP improvement ---")
        seq_fw, val_fw = _frank_wolfe_optimize(
            global_best_seq, n, deadline, max_iters=8, mix_coeff=0.5
        )
        if val_fw < global_best_val:
            global_best_seq = seq_fw.copy()
            global_best_val = val_fw
            print(f"[FW] new global best = {val_fw:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Mirror‑descent sub‑gradient (exact max) fine‑tuning.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=2_000_000,
            lr0=0.3,
            power=0.30,
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=2_000_000,
            init_temp_factor=0.04,
            min_temp=1e-12,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (for information only).
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Eval] evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for possible reuse by a later run.
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
