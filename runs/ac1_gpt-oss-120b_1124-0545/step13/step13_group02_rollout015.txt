1️⃣ Seed generation – a diverse set of analytic and random shapes,
   all ℓ₁‑normalised to sum = 1.  The best few (according to the

   drive the distribution toward a flat self‑convolution.

3️⃣ Exponential‑gradient (mirror‑descent) – a sub‑gradient descent on the

def _lp_refine(f, n, deadline, max_iters=30, mix_fixed=0.5):
    """Perform up to max_iters LP refinements on f."""
    f = np.asarray(f, dtype=np.float64)
    best_f = f.copy()
    conv = _conv_fft(f, f)
    best_M = np.max(conv)
    best_val = 2.0 * n * best_M
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # build convolution matrix A with current f
        if not _HAS_SCIPY:
            print("SciPy not available – skipping LP refinement.")
            break
        # factor: use diags to create sparse matrix
        offsets = np.arange(n, dtype=np.int32)
        A = sp.diags(f, offsets=offsets, shape=(2 * n - 1, n), format='csr')
        M = best_M
        b = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)  # maximise sum(g)
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if res.status != 0:
            # LP failed – abort refinement
            print(f"[LP] iteration {it}: linprog status {res.status}, message: {res.message}")
            break
        g = res.x
        s = g.sum()
        if s <= 0:
            # fallback: uniform
            g_n = _uniform(n)
        else:
            g_n = g / s
        # try a few mixing coefficients and keep the best improvement
        best_local_f = f
        best_local_M = best_M
        best_lambda = 0.0
        for lam in (0.1, 0.25, 0.5, 0.75):
            cand = (1.0 - lam) * f + lam * g_n
            # projection onto simplex (to guard against rounding errors)
            cand = _proj_simplex(cand)
            conv_cand = _conv_fft(cand, cand)
            M_cand = np.max(conv_cand)
            if M_cand < best_local_M:
                best_local_M = M_cand
                best_local_f = cand
                best_lambda = lam
        if best_lambda == 0.0:  # no improvement found
            print(f"[LP] iteration {it}: no improvement, stopping.")
            break
        f = best_local_f
        best_M = best_local_M
        best_val = 2.0 * n * best_M
        print(f"[LP] iteration {it}: mixing λ={best_lambda:.2f} → surrogate {best_val:.6f}")
        # clean up sparse matrix to free memory
        del A
    return best_f, best_val

Optimisation of a non‑negative step function to minimise the official
``evaluate_sequence`` objective.

The algorithm proceeds in five stages (all respect the global time budget):

1️⃣ **Seed generation** – a large, analytic + random library of ℓ₁‑normalised
   vectors (uniform, triangular, sine windows, Gaussians, power laws,
   Beta‑shapes, Tukey windows, Dirichlet draws, low‑frequency cosine
   sums, …).  The ten best seeds w.r.t. the surrogate
   ``2·n·max(conv)`` are kept.

2️⃣ **Soft‑max Adam** – a smooth approximation of the max (log‑sum‑exp)
   is optimised with Adam, projected onto the probability simplex.
   Several short runs from the top seeds give a good starting point.

3️⃣ **Exponential‑gradient (mirror‑descent)** – the true
   ∞‑norm is tackled directly via sub‑gradients of the active shift.
   A very large number of iterations drives the maximum convolution down.

4️⃣ **LP‑refinement (alternating linear programme)** – given the current
   vector ``f`` and its worst convolution value ``M = max(f*f)``, we solve

       maximise   Σ g_j
       subject to   f * g ≤ M            (linear convolution constraints)
                    g ≥ 0.

   The optimal ``g`` concentrates mass where ``f`` is smallest.
   After normalising ``g`` we perform a line‑search over mixing factors
   λ∈[0,1] and replace ``f ← (1‑λ)·f + λ·g`` if the surrogate improves.
   Repeating this step (a few dozen times) rapidly shrinks the peak.

5️⃣ **Simulated‑annealing polish** – a brief SA pass makes tiny mass
   transfers and reshuffles small subsets; any improvement in the
   surrogate is accepted.

All stages stop as soon as the deadline is reached.  The best sequence
found is stored in the global name ``height_sequence_1`` for possible
reuse by a later run and is returned as a plain list of floats.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(p,p)) for a distribution p (∑p=1)."""
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    """Monotonically decreasing exponential, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse pool of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    seeds.append(_gaussian(n, 0.2))
    seeds.append(_gaussian(n, 0.3))
    seeds.append(_gaussian(n, 0.4))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final normalisation safety‑check
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

    alpha_start: float = 20.0,
    alpha_end: float = 2000.0,
    lr: float = 0.12,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth surrogate L(p)=1/α·log Σ exp(α·(p*p))."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    N = _next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution p*p (full)
        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        # soft‑max over the convolution
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        # gradient = 2·correlation(p, soft)
        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # projection onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if t % 20000 == 0:
                print(f"[Adam] it={t:7d} best={best_val:.6f} α={alpha:.1f}")

    return best_x, best_val


def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent (multiplicative update) with sub‑gradient of the active shift."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # avoid overflow/underflow
        x = x * np.exp(delta)

        # projection onto the simplex
        s = x.sum()
        if s <= 0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _simulated_annealing(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Short SA polishing on the surrogate objective."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.6:   # reshuffle a random subset
            k_sub = rng.integers(3, 13)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:                     # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# LP refinement (alternating linear programme)
# ----------------------------------------------------------------------
def _lp_refine(
    f: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 30,
    max_active: int = 20,
) -> tuple[np.ndarray, float]:
    """
    Perform up to ``max_iters`` alternating LP refinements.
    Only constraints that are (near) active are kept to keep the LP
    small.
    """
    if not _HAS_SCIPY:                     # pragma: no cover
        print("[LP] SciPy not available – skipping LP refinement.")
        return f, 2.0 * n * np.max(_conv_fft(f, f))

    f = np.asarray(f, dtype=np.float64)
    best_f = f.copy()
    best_M = np.max(_conv_fft(f, f))
    best_val = 2.0 * n * best_M

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # --------------------------------------------------------------
        # Identify (near) active convolution constraints
        # --------------------------------------------------------------
        conv_f = _conv_fft(f, f)
        M = np.max(conv_f)
        # tolerance – 1e‑8 relative to M is enough to pick the true peak(s)
        rel_tol = 1e-8
        active_idxs = np.where(conv_f >= M * (1 - rel_tol))[0]
        if active_idxs.size == 0:
            break
        # limit the number of active constraints for speed
        if active_idxs.size > max_active:
            # keep the indices with the largest values (they are already near M)
            sorted_idx = np.argsort(-conv_f[active_idxs])
            active_idxs = active_idxs[sorted_idx[:max_active]]

        # --------------------------------------------------------------
        # Build the sparse constraint matrix A (|active| × n)
        # --------------------------------------------------------------
        rows = []
        cols = []
        data = []
        for row_id, k in enumerate(active_idxs):
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1)
            j_vals = np.arange(j_start, j_end + 1, dtype=int)
            i_vals = k - j_vals                      # i = k - j
            coeff = f[i_vals]                        # coefficient = f_i
            rows.extend([row_id] * len(j_vals))
            cols.extend(j_vals.tolist())
            data.extend(coeff.tolist())
        A = sp.csr_matrix((data, (rows, cols)), shape=(len(active_idxs), n))

        # --------------------------------------------------------------
        # Linear programme: maximise Σ g_j  subject to A·g ≤ M, g ≥ 0
        # --------------------------------------------------------------
        c = -np.ones(n, dtype=np.float64)            # maximise sum(g)
        b = np.full(len(active_idxs), M, dtype=np.float64)
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
        )
        if res.status != 0:                          # LP failed
            print(f"[LP] iteration {it}: linprog failed (status={res.status}) – aborting.")
            break
        g = res.x
        s = g.sum()
        if s <= 0.0:
            g_n = _uniform(n)
        else:
            g_n = g / s                              # normalise to ℓ₁

        # --------------------------------------------------------------
        # Mixing – try a few λ values and keep the best improvement
        # --------------------------------------------------------------
        best_local_f = f
        best_local_M = M
        best_lambda = 0.0
        for lam in (0.1, 0.25, 0.5, 0.75):
            cand = (1.0 - lam) * f + lam * g_n
            cand = _proj_simplex(cand)               # re‑project to simplex
            M_cand = np.max(_conv_fft(cand, cand))
            if M_cand < best_local_M:
                best_local_M = M_cand
                best_local_f = cand
                best_lambda = lam
        if best_lambda == 0.0:                       # no improvement
            print(f"[LP] iteration {it}: no improvement – stopping refinements.")
            break

        # accept the improvement
        f = best_local_f
        best_f = f.copy()
        best_M = best_local_M
        best_val = 2.0 * n * best_M
        print(f"[LP] iter={it:2d} λ={best_lambda:.2f} → surrogate {best_val:.6f}")

        # free memory of the temporary sparse matrix
        del A

    return best_f, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    the official ``evaluate_sequence`` objective within ``budget_s`` seconds.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Length selection – 4000 is a good compromise.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 0 – diverse seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"\nTop {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation (soft‑max) from the best seeds
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Soft‑max Adam optimisation (multiple starts) ---")
        for idx, seed_vec in enumerate(best_seeds[:4]):
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            # allocate roughly 10 % of remaining budget per seed
            max_iters = int(min(120_000, remaining * 130))
            cand, val = _adam_softmax_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=max_iters,
                alpha_start=20.0,
                alpha_end=1500.0,
                lr=0.12,
            )
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Adam] seed #{idx+1} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Exponential‑gradient (mirror‑descent) polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long‑run Exponential‑Gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(1_200_000, remaining * 1500))  # more iterations than before
        best_seq, best_val = _expgrad_optimize(
            best_seq, n, deadline, max_iters=max_iters, eta0=0.25
        )
        print(f"[EG] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Linear‑programming refinement (alternating LP)
    # ------------------------------------------------------------------
    if time.time() < deadline and _HAS_SCIPY:
        print("\n--- Linear‑programming refinement ---")
        remaining = deadline - time.time()
        # reserve a few seconds for the final SA pass
        lp_deadline = time.time() + remaining - 30.0
        best_seq, best_val = _lp_refine(best_seq, n, lp_deadline, max_iters=30)
        print(f"[LP] best surrogate after refinement = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(600_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(
            best_seq, n, deadline, max_iters=max_iters
        )
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
