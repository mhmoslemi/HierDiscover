def _simulated_annealing(initial_seq, n, deadline, max_iters):
    seq = initial_seq.copy()
    best_seq = seq.copy()
    best_val = _objective_surrogate(seq)
    cur_val = best_val
    rng = np.random.default_rng()
    # Define temperature schedule
    T0 = 0.1
    Tf = 1e-6
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        # Choose two distinct positions i,j
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # Choose amount to move
        # Choose epsilon as a fraction of seq[i] or a small uniform value.
        max_move = seq[i]
        if max_move <= 0:
            continue
        epsilon = min(max_move, rng.exponential(scale=0.01))
        # Propose new sequence
        seq_new = seq.copy()
        seq_new[i] -= epsilon
        seq_new[j] += epsilon
        # ensure non-negative (we already ensured epsilon <= seq[i])
        # Compute new surrogate

"""
Propose a non‑negative step‑function (ℓ₁‑normalised) that minimises
``evaluate_sequence``.  The algorithm consists of:

1.  A rich seed pool (analytic windows, random Dirichlet/exp, low‑freq
    curves, Chebyshev and DPSS windows, exponential decay, mixtures).
2.  Continuation L‑BFGS on a smooth log‑sum‑exp surrogate of the max
    convolution.  The β‑schedule is extended (up to 5 000) and many
    iterations per β are allowed.
3.  Gradual up‑sampling (1024 → 2048 → 4096) with re‑optimisation at each
    resolution.
4.  A fast simulated‑annealing (mass‑transfer) refinement that can escape
    local minima after the gradient step.
5.  A final short L‑BFGS polish.

The whole search respects the 1000 s budget and always returns a valid
sequence (stored as ``height_sequence_1``)."""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.signal as _sig
import scipy.signal.windows as _win

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _chebwin(n: int, attn: float) -> np.ndarray:
    """Chebyshev window (attenuation in dB)."""
    try:
        a = _win.chebwin(n, attn)
        return a / a.sum()
    except Exception:       # pragma: no cover
        return _uniform(n)


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """DPSS (Slepian) window."""
    try:
        a = _win.dpss(n, nw, sym=False)
        return a / a.sum()
    except Exception:       # pragma: no cover
        return _uniform(n)


def _exp_decay_window(n: int, lambd: float) -> np.ndarray:
    """Symmetric exponential decay around the centre."""
    if lambd <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.exp(-lambd * np.abs(i - c))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture (α∈[0,1]) of two normalised vectors."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

    seeds: list[np.ndarray] = []

    # Deterministic analytic families

    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0):

    for sigma in np.linspace(0.10, 0.40, 7):

    # Power‑law
    for p in np.linspace(0.0, 4.0, 21):
        seeds.append(_power(n, p))

    # Beta family grid
    alphas = np.linspace(0.5, 4.0, 7)
    betas = np.linspace(0.5, 4.0, 7)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Chebyshev windows (various attenuations)
    for attn in (40, 60, 80, 100, 120):
        seeds.append(_chebwin(n, attn))

    # DPSS windows (different time‑bandwidth products)
    for nw in (2.5, 3.5, 4.5):
        seeds.append(_dpss_window(n, nw))

    # Exponential decay shapes
    for lam in (0.001, 0.002, 0.005, 0.01, 0.02):

    # Random Dirichlet draws (different concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):

    # Random exponential draws (heavy‑tailed)
    for _ in range(5):

    # Low‑frequency random curves

    # Previous solution (if any)

# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector (softmax of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0]

    # Initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # probability vector
            conv = _conv_fft(a, a)                     # full autocorrelation
            shift = np.max(conv)                       # max(conv)
            # smooth max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # weights for the smooth max
            w = exp_beta / sum_exp

            # gradient w.r.t. a : 2·conv(rev_a, w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule: d obj / d logits = Jacobian(softmax)·grad_a
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # abort if we out‑of‑time
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            # time ran out – keep the current v
            break
        except Exception:      # pragma: no cover
            # any other failure – stop this β level
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple SA: repeatedly move a tiny amount of mass from one index to another.
    Accepts worsening moves with probability exp(-Δ/T).  Returns the best
    sequence seen and its surrogate value.
    """
    rng = np.random.default_rng()
    seq = start_seq.copy()
    cur_val = _objective_surrogate(seq)
    best_seq = seq.copy()
    best_val = cur_val

    # temperature schedule (exponential cooling)
    T0 = 0.08
    Tf = 1e-6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # pick source (i) that actually has mass to give
        i = rng.integers(n)
        attempts = 0
        while seq[i] <= 0 and attempts < 3:
            i = rng.integers(n)
            attempts += 1
        if seq[i] <= 0:
            continue

        # pick a different destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # amount to move – exponential with scale 0.01, capped by source mass
        eps = min(seq[i], rng.exponential(scale=0.01))
        if eps <= 0:
            continue

        # propose new sequence
        seq_new = seq.copy()
        seq_new[i] -= eps
        seq_new[j] += eps

        # surrogate for the proposal
        new_val = _objective_surrogate(seq_new)
        delta = new_val - cur_val

        # temperature at this iteration
        tau = it / max_iters
        T = T0 * (Tf / T0) ** tau

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True

        if accept:
            seq = seq_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = seq_new.copy()
                if it % 50000 == 0:
                    print(f"[SA] iter={it:,d} best surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Upsampling utility
# ----------------------------------------------------------------------
def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = new_seq / new_seq.sum()
    return new_seq

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence as a plain Python list.
    """
    # ------------------------------------------------------------------
    # Initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Stage 0 – start with a modest resolution to keep the early search cheap
    # ------------------------------------------------------------------
    n = 1024
    print(f"\n=== Starting search (n={n}) ===")

    # ------------------------------------------------------------------
    # Stage 1 – seed generation & cheap surrogate ranking
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"Best surrogate values among the top {top_k} seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – continuation L‑BFGS on the best seeds (coarse grid)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] current best surrogate = {global_best_val:.6f}")

    beta_sched = [1, 5, 20, 80, 200, 500, 1000, 2000, 5000]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all coarse seeds.")
            break

        remaining = deadline - time.time()
        # give each run a modest iteration budget, but never exceed the remaining budget
        max_iters_beta = max(500, int(min(2500, remaining * 0.05 * 1e5)))

        print(f"\n--- L‑BFGS run on coarse seed #{idx+1} (max_iter/beta={max_iters_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=max_iters_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – up‑sampling to a finer grid and re‑optimising
    # ------------------------------------------------------------------
    while n < 4096 and time.time() < deadline:
        # double the resolution
        new_n = n * 2
        print(f"\n--- Upsampling to n={new_n} ---")
        up_seq = _upsample_sequence(global_best_seq, new_n)

        # quick L‑BFGS polish on the finer grid
        remaining = deadline - time.time()
        max_iters_beta = max(400, int(min(2000, remaining * 0.05 * 1e5)))
        up_opt = _lbfgs_smooth_optimize(up_seq,
                                        new_n,
                                        deadline,
                                        beta_schedule=beta_sched,
                                        max_iter_per_beta=max_iters_beta)
        up_val = _objective_surrogate(up_opt)
        print(f"[Upsample‑L‑BFGS] surrogate = {up_val:.6f}")

        if up_val < global_best_val:
            global_best_seq = up_opt.copy()
            global_best_val = up_val
            print(f"[Update] New best after up‑sampling = {global_best_val:.6f}")

        n = new_n
        # continue up‑sampling while we still have room and time
        if n >= 4096:
            break

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing refinement (mass‑transfer moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa_iters = int(min(300_000, remaining * 0.5 * 1e5))
        if max_sa_iters > 0:
            print("\n--- Simulated‑annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_sa_iters)
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[SA] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – final short L‑BFGS polish (if time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_beta = max(200, int(min(800, remaining * 0.25 * 1e5)))
        if max_iters_beta > 0:
            print("\n--- Final L‑BFGS polish ---")
            final_seq = _lbfgs_smooth_optimize(global_best_seq,
                                               n,
                                               deadline,
                                               beta_schedule=[2000, 5000],
                                               max_iter_per_beta=max_iters_beta)
            final_val = _objective_surrogate(final_seq)
            if final_val < global_best_val:
                global_best_seq = final_seq.copy()
                global_best_val = final_val
                print(f"[Final‑L‑BFGS] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
