def _lbfgsb_optimize(start_a, n, deadline, p, max_iter):
    y0 = np.log(np.maximum(start_a, 1e-15))
    import scipy.optimize as opt

    def func_grad(y):
        # compute a
        y_max = y.max()
        exp_y = np.exp(y - y_max)
        a = exp_y / exp_y.sum()
        # compute objective and grad w.r.t a
        conv = _conv_fft(a, a)
        max_c = conv.max()
        shifted = conv - max_c
        exp_arg = p * shifted
        # Underflow safe
        exp_arg = np.clip(exp_arg, -700, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            mask = conv == max_c
            w = mask.astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)
        grad_a = 2.0 * grad_full[n-1:n-1+n]  # shape n

        # gradient w.r.t y
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)

        # objective (smooth)
        obj = (np.log(w_sum) + p * max_c) / p
        return obj, grad_y

    # Use callback for progress, but not needed.
    # Set options: maxiter = max_iter
    # bound can be set to None
    res = opt.minimize(func_grad, y0, method='L-BFGS-B', jac=True,
                       options={'maxiter': max_iter, 'disp': False})
    # Get final a
    y_opt = res.x
    a_opt = np.exp(y_opt - np.max(y_opt))
    a_opt = a_opt / a_opt.sum()
    # Compute exact objective
    max_conv = _conv_fft(a_opt, a_opt).max()
    val = 2.0 * n * max_conv
    return a_opt, val

"""
Propose a non‑negative step function that minimises the
``evaluate_sequence`` objective.

Algorithm overview
------------------
1. **Seed pool** – generate a diverse set of ℓ₁‑normalised vectors using a
   collection of analytic windows (uniform, triangular, sine, Gaussian,
   power, beta, Tukey, exponential) together with random Dirichlet draws.
2. **Quick Exponentiated‑Gradient (EG)** – a few fast EG runs on the
   best seeds give decent starting points.
3. **L‑BFGS‑B on a smoothed max‑convolution** – minimise the smooth
   surrogate ``Fₚ(a) = (1/p)·log Σₖ exp(p·convₖ(a))`` with the L‑BFGS‑B
   optimiser (variables are logits of a soft‑max, guaranteeing the simplex).
   The smoothing parameter *p* is increased geometrically (5 → 15 → 45 → 135)
   and a short random‑restart is performed for each *p*.
4. **Final EG polishing** – a brief EG pass from the best vector.
5. **Official evaluation** – the final vector is handed to
   ``evaluate_sequence`` and stored globally as ``height_sequence_1``.
The whole pipeline respects a 1000 s deadline and spends the majority of
the time on the L‑BFGS‑B refinement, which empirically finds a lower
max‑convolution than the previous baseline.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    m = y.max()
    e = np.exp(y - m)
    return e / e.sum()


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate when Σseq = 1 : 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0, 5.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.15, 0.2, 0.3, 0.4, 0.6):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform (adds slight flat bias)
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # keep previous solution (if any) as a seed
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final sanity normalisation
    out = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        out.append(v / s if s > 0 else _uniform(n))
    return out

# Exponentiated‑gradient optimiser (mirror descent on simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000,
                      eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative updates)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # stepsize (√t schedule)
        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # projection onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# L‑BFGS‑B optimisation of the smooth surrogate
# ----------------------------------------------------------------------
def _lbfgsb_func(y: np.ndarray, n: int, p: float):
    """Smooth surrogate and its gradient w.r.t. logits y."""
    a = _softmax(y)                     # a lies on the simplex
    conv = _conv_fft(a, a)              # length 2n‑1
    max_c = conv.max()

    shifted = conv - max_c               # ≤ 0
    exp_arg = p * shifted
    exp_arg = np.clip(exp_arg, -700, 0)  # safety
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                    # fallback to exact max locations
        mask = conv == max_c
        w = mask.astype(np.float64)
        w_sum = w.sum()
    w /= w_sum

    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)     # length 3n‑2
    grad_a = 2.0 * grad_full[n - 1: n - 1 + n]

    # chain rule: ∂F/∂y = diag(a)·(∂F/∂a – (a·∂F/∂a)·1)
    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    # smooth surrogate value: (1/p)·log Σ exp(p·cₖ)
    obj = (np.log(w_sum) + p * max_c) / p
    return obj, grad_y


def _lbfgsb_optimize(start_a: np.ndarray, n: int, deadline: float,
                     p: float = 20.0, max_iter: int = 250) -> tuple[np.ndarray, float]:
    """
    L‑BFGS‑B on the smooth surrogate with logits parametrisation.
    Returns the best vector found and its *exact* surrogate value
    (2·n·max(conv)).
    """
    y0 = np.log(np.maximum(start_a, 1e-15))
    # SciPy's minimize will call our function with (y, n, p) via args
    res = _opt.minimize(_lbfgsb_func, y0,
                        args=(n, p),
                        method='L-BFGS-B',
                        jac=True,
                        options={'maxiter': max_iter, 'disp': False})

    y_opt = res.x
    a_opt = _softmax(y_opt)

    max_conv = _conv_fft(a_opt, a_opt).max()
    val = 2.0 * n * max_conv
    return a_opt, val

    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # --------------------------------------------------------------
    # Choose a length that balances expressiveness and speed.
    # --------------------------------------------------------------
    n = 4000
    print(f"\n--- Problem size -------------------------------------------------")
    print(f"Length of the sequence: n = {n}")

    # --------------------------------------------------------------
    # Stage 0 – generate a diverse pool of seeds.
    # --------------------------------------------------------------
    print("\n--- Seed generation ------------------------------------------------")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} candidate seeds.")

    # keep the best few according to the simple surrogate
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – quick Exponentiated‑Gradient refinement.
    # --------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Quick EG refinement ------------------------------------------")
        for idx, seed_vec in enumerate(best_seeds[:5]):   # a handful of seeds
            if time.time() >= deadline:
                break
            max_iters = 150_000
            cand, val = _expgrad_optimize(seed_vec, n, deadline,
                                          max_iters=max_iters, eta0=0.30)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Quick EG] seed #{idx+1} improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – L‑BFGS‑B on a smoothed max‑convolution.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- L‑BFGS‑B optimisation (smooth max) ---------------------------")
        # geometric schedule for the smoothing parameter p
        p_schedule = [5.0, 15.0, 45.0, 135.0]
        for p in p_schedule:
            if time.time() >= deadline:
                break
            # a modest number of L‑BFGS‑B iterations – each iteration is cheap
            max_iter = 250
            cand, val = _lbfgsb_optimize(best_seq, n, deadline, p=p,
                                         max_iter=max_iter)
            print(f"[L‑BFGS] p={p:5.1f} → surrogate = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"    ** improved global best to {best_val:.6f}")

            # one random‑restart per p value
            if time.time() < deadline:
                y_noise = np.log(np.maximum(best_seq, 1e-15)) + rng.normal(scale=0.2, size=n)
                perturbed = _softmax(y_noise)
                cand2, val2 = _lbfgsb_optimize(perturbed, n, deadline, p=p,
                                               max_iter=max_iter)
                if val2 < best_val:
                    best_val, best_seq = val2, cand2
                    print(f"[Restart] p={p:.1f} → improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – final EG polishing (very cheap, helps stabilise)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing --------------------------------------------")
        remaining = deadline - time.time()
        max_iters = int(min(40_000, remaining * 2000))
        if max_iters > 0:
            cand, val = _expgrad_optimize(best_seq, n, deadline,
                                          max_iters=max_iters, eta0=0.25)
            print(f"[Final EG] surrogate = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand

    # ------------------------------------------------------------------
    # Official evaluation.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse in a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
