Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

within a fixed time budget.

Algorithm
---------
0.  Initialise a modest resolution (n≈2000) and a starting shape
    (previous best if available, otherwise a rich collection of
    analytic beta‑family shapes, including very “U‑shaped’’ parameters).

1.  **LP refinement** – iterate a linear program that, for the current
    shape f, finds a vector g maximising the ℓ₁‑norm under the linear
    constraints  f∗g ≤ M  (M = max(f∗f)).  Mixing f←(1‑t)f + t g with a
    decreasing step‑size pushes mass away from the positions that most
    contribute to the maximal convolution.

2.  **Smooth L‑BFGS** – run a continuation L‑BFGS on a log‑sum‑exp
    surrogate of the max‑convolution (a smooth approximation whose
    gradient is cheap via FFT).  A long β‑schedule and many iterations
    bring the solution close to a local optimum.

3.  **Mirror‑descent (exponentiated‑gradient)** – fine‑tune the exact
    max‑convolution objective using a sub‑gradient update on the simplex
    (step‑size ηₜ=0.5/√t).  This is cheap and improves the bound further.

4.  **Upsampling** – if time permits, double the resolution by linear
    interpolation, repeat steps 2–3 on the finer grid and keep the best
    sequence.

All vectors are kept ℓ₁‑normalised (sum = 1); the evaluation function

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _downsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a smaller grid, then ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    return new_seq / s if s > 0 else _uniform(new_n)


def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid, then ℓ₁‑renormalise."""
    return _downsample_sequence(seq, new_n)

                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Continuated L‑BFGS on the smooth max‑convolution surrogate."""
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]

    # logarithmic parameters of the softmax (log‑probabilities)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] time limit reached (beta={beta_val})")
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent with diminishing step‑size."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming step (maximise ℓ₁‑norm under f∗g ≤ M)
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, cap: float = 1000.0) -> np.ndarray:
    """
    Solve   max 1ᵀg  s.t.  f∗g ≤ M , g ≥ 0.
    Returns a normalised vector (sum=1) ready for mixing.
    """
    n = len(f)
    L = 2 * n - 1

    # Build sparse constraint matrix A (size L×n) where
    #   A[k, j] = f[k‑j]   if 0≤k‑j<n else 0
    rows = []
    cols = []
    data = []

    for i in range(n):
        fi = f[i]
        row_start = i
        for j in range(n):
            rows.append(row_start + j)
            cols.append(j)
            data.append(fi)

    A = _sp.csr_matrix((data, (rows, cols)), shape=(L, n))
    b = np.full(L, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)             # minimise –sum(g)  ⇒ maximise sum(g)
    bounds = [(0.0, None)] * n

    res = _lp_opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs',
                          options={'presolve': True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    g0 = res.x
    # Clip to stay within evaluator’s caps, then renormalise
    g0 = np.clip(g0, 0.0, cap)
    s = g0.sum()
    if s <= 0.0:
        return _uniform(n)
    g = g0 / s
    return g

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence() within the given time budget.
    Returns the best sequence found (as a plain list of floats).
    """
    # --------------------------------------------------------------
    # Initialisation
    # --------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0           # small safety margin

    # --------------------------------------------------------------
    # Choose a working resolution
    # --------------------------------------------------------------
    n = 2000
    print(f"\n=== Starting optimisation (n={n}) ===")

    # --------------------------------------------------------------
    # Retrieve previous solution if any and adjust its size
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0:
                prev_vec = _downsample_sequence(prev_arr, n)
        except Exception:
            prev_vec = None

    # --------------------------------------------------------------
    # Seed collection (rich family, emphasising U‑shaped Beta)
    # --------------------------------------------------------------
    seeds = [_uniform(n)]
    # enriched beta grid (including very small parameters)
    alphas = np.concatenate((np.linspace(0.05, 0.3, 6), np.linspace(0.4, 2.0, 5)))
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, float(a), float(b)))
    if prev_vec is not None:
        seeds.append(prev_vec / prev_vec.sum())

    # --------------------------------------------------------------
    # Evaluate seeds (surrogate) and keep the best few
    # --------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    best_val = evaluated[0][1]
    best_seq = top_seeds[0].copy()
    print(f"Initial surrogate best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – LP‑based iterative improvement
    # --------------------------------------------------------------
    f = best_seq.copy()
    # ensure strictly positive entries for LP stability
    eps = 1e-12
    f = np.maximum(f, eps)
    f = f / f.sum()
    t_mix = 0.5
    LP_MAX_ITER = 12

    for it in range(LP_MAX_ITER):
        if time.time() > deadline:
            break

        conv_f = _conv_fft(f, f)
        M = np.max(conv_f)

        try:
            g = _lp_step(f, M)
        except Exception as e:
            print(f"[LP] iteration {it+1} failed: {e}")
            break

        # convex combination
        f_new = (1.0 - t_mix) * f + t_mix * g
        f_new = np.clip(f_new, 0.0, 1000.0)
        f_new = f_new / f_new.sum()

        cur_val = _objective_surrogate(f_new)
        print(f"[LP {it+1:02d}] surrogate = {cur_val:.6f}")

        if cur_val < best_val:
            best_val = cur_val
            best_seq = f_new.copy()

        f = f_new
        # slowly shrink the mixing coefficient
        t_mix = max(0.1, t_mix * 0.85)

    # --------------------------------------------------------------
    # Stage 2 – Smooth L‑BFGS refinement (log‑sum‑exp surrogate)
    # --------------------------------------------------------------
    if time.time() < deadline:
        beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0]
        max_iter_beta = 5000
        try:
            seq_opt = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                             beta_schedule=beta_sched,
                                             max_iter_per_beta=max_iter_beta)
            val_opt = _objective_surrogate(seq_opt)
            print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
            if val_opt < best_val:
                best_val = val_opt
                best_seq = seq_opt.copy()
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception: {e}")

    # --------------------------------------------------------------
    # Stage 3 – Exponentiated‑gradient fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(400_000, remaining * 180))  # ≈180 iters per second
        if max_iters > 0:
            seq_fg, val_fg = _exp_gradient_optimize(best_seq, n, deadline,
                                                    max_iters=max_iters)
            print(f"[ExpGrad] surrogate = {val_fg:.6f}")
            if val_fg < best_val:
                best_val = val_fg
                best_seq = seq_fg.copy()

    # --------------------------------------------------------------
    # Stage 4 – Upsample to a finer grid (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 30.0:
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} ---")
            up_seq = _upsample_sequence(best_seq, new_n)

            # L‑BFGS on finer grid
            seq_up = _lbfgs_smooth_optimize(up_seq, new_n, deadline,
                                            beta_schedule=beta_sched,
                                            max_iter_per_beta=3000)
            val_up = _objective_surrogate(seq_up)
            print(f"[Upsample L‑BFGS] surrogate = {val_up:.6f}")
            if val_up < best_val:
                best_seq = seq_up.copy()
                best_val = val_up
                n = new_n

            # Final exponentiated‑gradient sweep on the finer grid
            if time.time() < deadline:
                remaining = deadline - time.time()
                max_iters2 = int(min(250_000, remaining * 150))
                if max_iters2 > 0:
                    seq_fg2, val_fg2 = _exp_gradient_optimize(best_seq,
                                                             n,
                                                             deadline,
                                                             max_iters=max_iters2)
                    print(f"[Final ExpGrad] surrogate = {val_fg2:.6f}")
                    if val_fg2 < best_val:
                        best_seq = seq_fg2.copy()
                        best_val = val_fg2

    # --------------------------------------------------------------
    # Final true evaluation (outside the timer is fine – we already
    # accounted for a safety margin above)
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for possible warm‑starts in subsequent runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
