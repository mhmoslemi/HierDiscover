def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate (exact when Σseq = 1):
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:

# Exponentiated‑gradient optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative updates)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # stepsize (√t schedule)
        eta = eta0 / math.sqrt(it)

        # multiplicative update (clip exponent for safety)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # project onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val

# Adam optimiser on soft‑max‑smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_smooth_optimize(start: np.ndarray, n: int, deadline: float,
                          init_lr: float = 0.05,
                          beta1: float = 0.9, beta2: float = 0.999,
                          eps: float = 1e-8,
                          p0: float = 5.0,
                          p_factor: float = 3.0,
                          p_step: int = 50000,
                          max_total_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth surrogate
        F_p(a) = (1/p)·log Σ exp(p·conv_k(a))
    with p increased geometrically.
    Returns best vector seen and its *exact* surrogate value.
    """
    a = start.copy()
    best_a = a.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    p = p0

    while time.time() < deadline and t < max_total_iters:
        # --------------------------------------------------------------
        # 1) current convolution and surrogate (exact)
        # --------------------------------------------------------------
        conv = _conv_fft(a, a)                     # length 2n‑1
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam] it={t:7d} best={best_val:.6f}  p={p:.1f}")

        # --------------------------------------------------------------
        # 2) soft‑max weights
        # --------------------------------------------------------------
        shifted = conv - max_conv                     # ≤ 0
        exp_arg = p * shifted
        # clip exponent to a safe range (avoid overflow in the rare case)
        exp_arg = np.clip(exp_arg, -700, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:  # extreme under‑flow → fall back to uniform over max locations
            mask = conv == max_conv
            w = mask.astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # --------------------------------------------------------------
        # 3) gradient of the smooth surrogate
        #    ∇F_p = 2·conv(w , a_rev)  (central slice)
        # --------------------------------------------------------------
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1: n - 1 + n]   # shape (n,)

        # --------------------------------------------------------------
        # 4) Adam update
        # --------------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # enforce non‑negativity and renormalise (simple ℓ₁ projection)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = _uniform(n)

        # --------------------------------------------------------------
        # 5) schedule for p (smoothness parameter)
        # --------------------------------------------------------------
        if t % p_step == 0:
            p *= p_factor
            print(f"[Adam] increased p to {p:.1f}")

    # final exact evaluation of the best vector found
    final_val = 2.0 * n * np.max(_conv_fft(best_a, best_a))
    return best_a, final_val

def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.6:
            # reshuffle a random small subset
            k = rng.integers(3, 13)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny mass transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val
