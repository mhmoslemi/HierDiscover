def _project_onto_simplex(v):
    # v is a numpy array (may sum not 1)
    # return projection onto nonnegative simplex (sum=1)
    # from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, len(v)+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Improved search for a non‑negative step function that minimises

        2·n·max(conv(a, a)) / (sum(a))²

The algorithm expands on the previous multi‑stage optimiser with the
following additions:

* a richer pool of analytic seeds (Hann, Hamming, Blackman windows,
  finer grids for the power‑law and exponential families, and many
  low‑frequency random curves);
* a quick 1‑D golden‑section search for the optimal power‑law exponent
  and exponential decay rate;
* an extended L‑BFGS continuation schedule that reaches temperatures
  up to 10 000 and allows many more iterations on the final (high‑β)
  stages;
* a simple projected‑gradient descent on the exact max‑convolution
  objective (sub‑gradient of the active anti‑diagonal) to polish the
  result after L‑BFGS;
* a final short L‑BFGS refinement at the highest temperature;
* all stages respect a global deadline (default 1000 s) and constantly
  update the global best sequence.

The final sequence is stored in the global name ``height_sequence_1``

"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# FFT‑based convolution (real‑valued, full length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Softmax helper (used to stay on the simplex)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = n - i
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    s = a.sum()
    return a / s


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * t / (n - 1)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * t / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * t / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * t / (n - 1))
    )
    return a / a.sum()


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Random smooth curve made of a few low‑frequency sinusoids."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Fast surrogate objective (identical ordering to the official one)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – works on ℓ₁‑normalised vectors."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and gradient
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """Return (smooth_max, grad_a) for the log‑sum‑exp surrogate."""
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        weights = np.zeros_like(conv)
    else:
        weights = exp_term / w_sum
    smooth = shift + np.log(w_sum) / beta

    # Gradient of Σ w_k * (a * a)_k  w.r.t a  ->  2 * (rev(a) ⊛ w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    grad_a = 2.0 * full[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad_a


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iter: int = 500,
                     temperature: float = 100.0) -> tuple[np.ndarray, float]:
    """Run L‑BFGS on the log‑sum‑exp surrogate for a given temperature."""
    eps = 1e-30
    v = np.log(start_seq + eps)  # logits → simplex via softmax

    def fun(v):
        a = _softmax(v)
        smooth, grad_a = _smooth_max_and_grad(a, temperature)
        loss = 2.0 * n * smooth                     # equals the official objective for ℓ₁‑norm
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)                 # chain rule through softmax
        return loss, (2.0 * n * grad_v).astype(np.float64)

    # honour the remaining budget – stop after max_iter or deadline
    opts = {"maxiter": max_iter, "disp": False}
    try:
        res = scipy.optimize.minimize(
            fun, v, method="L-BFGS-B", jac=True, options=opts
        )
    except Exception:  # pragma: no cover
        return start_seq, _objective(start_seq)

    best_a = _softmax(res.x)
    return best_a, _objective(best_a)


# ----------------------------------------------------------------------
# Projected gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, len(v) + 1) > (cssv - 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _projected_subgradient_optimize(start_seq: np.ndarray,
                                    n: int,
                                    deadline: float,
                                    max_iters: int = 500_000,
                                    lr0: float = 0.15,
                                    power: float = 0.35) -> tuple[np.ndarray, float]:
    """
    Simple projected (additive) sub‑gradient descent on the true max‑conv objective.
    Uses the exact active anti‑diagonal as sub‑gradient.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 1
    while it <= max_iters:
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ProjGrad] it={it:,d} best={cur_val:.6f}")

        # sub‑gradient for the active anti‑diagonal
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / ((it) ** power)
        a = a - lr * g
        a = _project_simplex(a)

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light simulated annealing (kept as a final polishing stage)
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 500_000,
                                 init_temp_factor: float = 0.04,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a random subset
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                  # split‑transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 200_000 == 0:
                    print(f"[SA] it={it:,d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Helper – golden‑section search for a 1‑D continuous parameter
# ----------------------------------------------------------------------
def _golden_section_optimize(func, a, b, tol=1e-3, max_iter=30):
    """Return the point x in [a,b] minimising func(x) using golden section."""
    phi = (1 + np.sqrt(5)) / 2.0
    invphi = 1 / phi
    c = b - (b - a) * invphi
    d = a + (b - a) * invphi
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - (b - a) * invphi
            fc = func(c)
        else:
            a, fc = c, fd
            c = d
            d = a + (b - a) * invphi
            fd = func(d)
    return (a + b) / 2.0


# ----------------------------------------------------------------------
# Seed generation (richer and finer‑grained)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_linear_decreasing(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    # Power‑law – fine grid and a golden‑section refinement later
    for p in np.linspace(0.0, 3.0, 61):
        seeds.append(_power(n, p))

    # Exponential decay – fine grid
    for lam in np.linspace(0.001, 0.050, 30):
        seeds.append(_exponential_decay(n, lam))

    # β‑shape – moderately fine grid
    alphas = np.concatenate((np.linspace(0.01, 0.30, 10), np.linspace(0.30, 2.0, 10)))
    betas = np.concatenate((np.linspace(0.5, 5.0, 10), np.linspace(5.0, 20.0, 10)))
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Low‑frequency random curves (more frequencies)
    for nf in (2, 3, 4, 5, 6):
        for _ in range(5):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Dirichlet draws (several concentration levels)
    for conc in (0.01, 0.05, 0.1, 0.5, 1.0):
        for _ in range(4):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # natural ordering (random)
            seeds.append(np.sort(raw)[::-1])      # descending
            # also a shuffled version of the sorted vector
            shuffled = np.sort(raw)[::-1].copy()
            rng.shuffle(shuffled)
            seeds.append(shuffled)

    # previous solution, if available
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Main optimiser entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the supplied time budget.
    The best sequence (list of floats) is stored in the global name
    ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # problem size – a few thousand items give a good trade‑off
    n = 4000

    # load any previous solution for seeding
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – generate a rich pool of seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # quick surrogate ranking
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 25
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – fine‑tune power‑law exponent via golden‑section search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        def power_obj(p):
            a = _power(n, p)
            return _objective(a)

        # search only if we have time left
        best_p = _golden_section_optimize(power_obj, 0.0, 3.0, tol=1e-3)
        a = _power(n, best_p)
        val = _objective(a)
        print(f"[Power] best p ≈ {best_p:.4f} → val = {val:.6f}")
        if val < global_best_val:
            global_best_seq = a.copy()
            global_best_val = val

    # ------------------------------------------------------------------
    # Stage 2 – L‑BFGS refinement on the smooth‑max surrogate
    # ------------------------------------------------------------------
    temperature_schedule = [5, 10, 20, 40, 80, 150, 300,
                            600, 1200, 2400, 4800, 9600]
    print("\n--- L‑BFGS refinement ---")
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        seq = seed_vec.copy()
        for temp in temperature_schedule:
            if time.time() > deadline:
                break
            # allocate more iterations for the later (high‑β) stages
            iters = 400 if temp < 1200 else 800
            seq_opt, val_opt = _l_bfgs_optimize(
                seq, n, deadline, max_iter=iters, temperature=temp
            )
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[L‑BFGS] seed #{idx}, temp={temp} → new best {val_opt:.6f}")
            seq = seq_opt   # warm‑start next temperature

    # ------------------------------------------------------------------
    # Stage 3 – Projected sub‑gradient descent on the exact objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Projected sub‑gradient descent (exact max) ---")
        seq_proj, val_proj = _projected_subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=500_000,
            lr0=0.18,
            power=0.33,
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[ProjGrad] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Mirror‑descent (exponentiated‑gradient) fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        from math import log2  # local import to avoid unnecessary import earlier
        # reuse the same routine as in the reference solution but with a
        # slightly larger iteration budget (if time permits)
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=2_500_000,
            lr0=0.18,
            power=0.30,
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – Simulated‑annealing polishing (shortened but kept)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq,
            n,
            deadline,
            max_iters=400_000,
            init_temp_factor=0.04,
            min_temp=1e-12,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – Final high‑temperature L‑BFGS polish (few iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS polish (β=12000) ---")
        seq_final, val_final = _l_bfgs_optimize(
            global_best_seq, n, deadline, max_iter=300, temperature=12000.0
        )
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[FinalL‑BFGS] new best = {val_final:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (for information only)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Eval] evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for possible reuse by a later run.
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
