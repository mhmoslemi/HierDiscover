"""
Propose a step‑function (non‑negative vector) that minimises the official
``evaluate_sequence`` objective.

Algorithm (≈ 1000 s budget)
--------------------------------
0.  Problem size: n = 4000 (the same as the reference solution).
1.  **Seed generation** – a large diversified pool of ℓ₁‑normalised
    candidates is built from many analytic families (uniform, triangular,
    cosine windows, Gaussians, beta‑shapes, power‑laws, edge‑heavy
    reciprocals, random mixtures, low‑frequency cosine sums, Dirichlet
    draws) together with the previous best solution (if any).  The pool
    contains a few hundred vectors.
2.  **Surrogate ranking** – each seed is scored with the cheap surrogate

        s(a) = 2·n·max(conv(a,a))

    (identical to the true objective for Σa = 1).  The `top_k`
    best seeds are kept.
3.  **Adam‐style optimisation** – each top seed is refined with Adam on
    the log‑softmax parametrisation of the simplex.  The surrogate is a
    temperature‑controlled log‑sum‑exp approximation of the max.
    Temperature (β) is increased up to 5000, and the learning‑rate is
    scheduled to decay slowly.  The best Adam run is stored.
4.  **Exact mirror‑descent** – the current best vector is further improved
    with exponentiated‑gradient sub‑gradient steps on the true max (no
    smoothing).  A more aggressive step‑size schedule (power 0.25) is used.
5.  **Iterated LP polishing** – a handful (≤ 12) of linear‑programming
    refinements are performed.  In each iteration we maximise the ℓ₁‑mass
    of a new vector `g` under the linear constraints
    `conv(f,g) ≤ M` (where `M` is the present peak of `conv(f,f)`);
    the solution is mixed back with weight τ (starting at 0.25 and

6.  **Second Adam pass** – after LP polishing we run a final long Adam

7.  **Simulated‑annealing polishing** – a lightweight SA with three move

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    # Numerical noise can produce tiny negative values; clip.
    return np.clip(conv, 0.0, None)


def _surrogate_obj(seq: np.ndarray) -> float:
    """
    Fast surrogate objective for a sequence normalised to Σseq = 1:
        2·n·max(conv(seq,seq))

    The surrogate differs from the official objective only by a constant
    factor (the denominator is 1), therefore ordering of candidates is
    unchanged.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _edge_heavy(n: int, p: float) -> np.ndarray:
    """Reciprocal‑law heavy on the two borders (symmetrised)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (i + 1) ** p + 1.0 / (n - i) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random 3‑segment piece‑wise linear shape."""
    # pick three break points (sorted) in the index range
    breaks = np.sort(rng.choice(n, size=3, replace=False))
    heights = rng.random(3) + 0.01  # avoid zeros
    x = np.arange(n, dtype=np.float64)
    a = np.interp(x, breaks, heights)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few random Gaussians (different centres / widths)."""
    a = np.zeros(n, dtype=np.float64)
    num = rng.integers(2, 6)
    for _ in range(num):
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.02 * n, 0.15 * n)
        x = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, nf: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth curve built from a few random low‑frequency cosine terms."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, nf + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int,
                   deadline: float, max_iters: int = 300_000) -> tuple[np.ndarray | None, float]:
    """
    Adam on log‑softmax parameters with a temperature‑controlled log‑sum‑exp
    surrogate of the max.  Temperature β is increased up to 5000, making the
    surrogate approach the true max.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 1.0
    beta = 0.5
    beta_inc = 1.0
    beta_max = 5000.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # smooth max via log‑sum‑exp with temperature β
        shift = np.max(conv)
        # numerically stable exponentials; clip to avoid overflow
        exparg = beta * (conv - shift)
        exparg = np.clip(exparg, -700, 700)
        w = np.exp(exparg)
        w_sum = w.sum()
        if w_sum == 0.0:
            # unlikely, but keep going
            continue
        w = w / w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 2n‑1
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 4_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) descent on the exact max (no smoothing).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    lr0 = 0.3
    power = 0.25  # slightly more aggressive than 0.28

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _lp_refine(seq: np.ndarray, deadline: float,
               max_iters: int = 12, tau0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Perform several LP refinements.

    Each iteration solves:
        maximise   Σ g_i
        subject to conv(f, g) ≤ M   (where M = max conv(f, f))
                  g_i ≥ 0

    The solution g is mixed back into the current sequence with weight τ.
    τ is geometrically reduced to keep the iteration stable.
    """
    f = seq.copy()
    n = len(f)

    tau = tau0
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        M = _conv_fft(f, f).max()

        # Build the convolution matrix A (sparse, shape (2n-1) × n)
        rows = np.empty(n * n, dtype=np.int32)
        cols = np.empty(n * n, dtype=np.int32)
        data = np.empty(n * n, dtype=np.float64)

        idx = 0
        for k in range(n):
            start = k
            end = k + n
            rows[idx:idx + n] = np.arange(start, end, dtype=np.int32)
            cols[idx:idx + n] = k
            data[idx:idx + n] = f
            idx += n

        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i → minimise -Σ g_i

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if not res.success:
            print(f"[LP] iteration {it} failed – stopping LP stage.")
            break

        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g <= sum_f + 1e-12:
            print(f"[LP] iteration {it} – no ℓ₁ gain.")
            break

        f_new = (1.0 - tau) * f + tau * g
        f_new = f_new / f_new.sum()
        f = f_new

        cur_obj = _surrogate_obj(f)
        print(f"[LP] iteration {it} – objective = {cur_obj:.6f}")

        # Reduce τ for stability
        tau = max(tau * 0.6, 0.05)

    return f, _surrogate_obj(f)


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 2_500_000,
                                 init_temp_factor: float = 0.07,
                                 min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Light SA that combines three move types.
    In the final stage (when temperature is tiny) only improving moves
    are accepted – effectively a hill‑climber.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate_obj(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9998, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:
            # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            # redistribute mass on a random small subset
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total == 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            # pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total == 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _surrogate_obj(new)
        delta = new_val - cur_val
        # Accept if improvement or according to Metropolis rule
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))

    # Gaussian width sweep
    for sigma in np.linspace(0.05, 0.35, 9):
        seeds.append(_gaussian(n, sigma))

    # Power‑law sweep
    for p in np.linspace(0.0, 5.0, 12):
        seeds.append(_power_law(n, p))

    # Edge‑heavy reciprocals
    for p in np.linspace(0.1, 2.5, 8):
        seeds.append(_edge_heavy(n, p))

    # Beta shape grid
    alphas = np.linspace(0.2, 3.0, 9)
    betas = np.linspace(0.2, 3.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Exponential decay sweep
    for lam in np.linspace(0.0005, 0.020, 9):
        seeds.append(_exponential_decay(n, lam))

    # Random Dirichlet draws (varied concentration)
    concs = np.logspace(-4, 0.5, 14)
    for conc in concs:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Gaussian mixtures
    for _ in range(6):
        seeds.append(_gaussian_mixture(n, rng))

    # Piecewise linear shapes
    for _ in range(6):
        seeds.append(_piecewise_linear(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 5, 7, 9):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mix uniform with low‑frequency curves (different mixing ratios)
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4, 6):
            curve = _lowfreq_curve(n, nf, rng)
            mixed = (1.0 - mix_alpha) * _uniform(n) + mix_alpha * curve
            mixed = mixed / mixed.sum()
            seeds.append(mixed)

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0.0:
            seeds.append(avg / avg.sum())

    return seeds

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective.  The best found sequence is stored
    globally as ``height_sequence_1`` and returned as a plain ``list``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # problem size (tuned to the reference solution)
    n = 4000

    # ------------------------------------------------------------------
    # Try to load a previous solution (if the notebook/session provides one)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size != n:
                if previous.size > n:
                    previous = previous[:n]
                else:
                    previous = np.pad(previous, (0, n - previous.size), constant_values=0.0)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – rich seed pool
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _surrogate_obj(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 30
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _surrogate_obj(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation on smooth surrogate (few top seeds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # allocate roughly half the remaining budget to Adam runs
        remaining = deadline - time.time()
        max_total_adam_iters = int(remaining * 20)   # heuristic
        iters_per_seed = max(30_000, min(250_000, max_total_adam_iters // len(best_seeds[:10])))
        print(f"\n--- Adam stage (max_iters per seed = {iters_per_seed}) ---")
        for idx, seed_vec in enumerate(best_seeds[:10], 1):
            if time.time() > deadline:
                break
            seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline,
                                                max_iters=iters_per_seed)
            if seq_adam is not None and val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Mirror‑descent (exact max) – longer run
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent (exact max) ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=4_000_000
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – LP polishing (iterated, decreasing τ)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- LP polishing (iterated) ---")
        seq_lp, val_lp = _lp_refine(global_best_seq, deadline,
                                    max_iters=12, tau0=0.25)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[overall‑LP] new global best = {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Second Adam pass (fine tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Second Adam fine‑tuning ---")
        # allocate remaining time for a single long Adam run
        remaining = deadline - time.time()
        max_iters = max(100_000, int(remaining * 60))   # very aggressive
        seq_adam2, val_adam2 = _adam_optimize(global_best_seq, n, deadline,
                                              max_iters=max_iters)
        if seq_adam2 is not None and val_adam2 < global_best_val:
            global_best_seq = seq_adam2.copy()
            global_best_val = val_adam2
            print(f"[overall‑Adam2] new global best = {val_adam2:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – Simulated‑annealing polishing (final)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline,
            max_iters=2_500_000, init_temp_factor=0.07
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (information only)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
