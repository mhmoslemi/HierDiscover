Propose a step‑function (non‑negative sequence) minimising

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **LP‑mass‑inflation** – Inspired by the classic linear‑programming
   update described in the problem statement.  For a normalised
   vector `f` we compute `M = max(conv(f,f))` and solve the LP

        maximise   1ᵀ g
        subject to A·g ≤ M·1 ,   g ≥ 0

   where `A` is the (2n‑1) × n convolution matrix of `f`.  
   The optimum yields `sum(g) > 1`; after scaling `g̃ = g / sum(g)` the
   bound `max(conv(g̃,g̃))` is reduced roughly by the factor `1 / sum(g)`.
   We repeat this step a few times, accepting either the scaled `g̃`
   or a convex combination with the previous iterate.

2. **Adam refinement** – On the log‑softmax parameterisation we minimise
   the smooth surrogate `2·n·max(conv)` using Adam with a slowly increasing
   temperature (log‑sum‑exp approximation of the max).  This sharpens the
   solution obtained by the LP stage.

3. **Final evaluation** – The best vector found is evaluated with the
   provided `evaluate_sequence`.  The result is stored in the global
   variable `height_sequence_1` for possible reuse.

All stages respect the global 1000 s deadline.  The implementation uses
only NumPy, SciPy (sparse matrix and `linprog`) and pure Python.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (a is assumed ℓ₁‑normalised)."""
    return np.max(_conv_fft(a, a))


def _surrogate(a: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(a,a)) for ℓ₁‑normalised a."""
    n = len(a)
    return 2.0 * n * _max_conv(a)


# ----------------------------------------------------------------------
# Analytic initial family
# ----------------------------------------------------------------------
def _exponential_shape(n: int, r: float) -> np.ndarray:
    """Monotone exponential a_i ∝ r**i, ℓ₁‑normalised (r∈[0,1])."""
    if r <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Build the convolution matrix for the LP constraints
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray):
    """Return the (2n‑1)×n sparse matrix A with A·g = conv(f,g)."""
    n = len(f)
    # each diagonal k (negative offset) contains the constant f[-k]
    diagonals = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    offsets = -np.arange(n)          # 0, -1, -2, …, -(n‑1)
    A = diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr')
    return A


# ----------------------------------------------------------------------
# One LP iteration – maximise ℓ₁‑norm under the same max‑conv bound
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float):
    """
    Solve   maximise 1ᵀ g   s.t.   A·g ≤ M·1 , g ≥ 0,
    where A is the convolution matrix of f.
    Returns (g, sum(g)) or (None, None) on failure.
    """
    n = len(f)
    A = _conv_matrix(f)

    # objective (minimise –sum(g) == maximise sum(g))
    c = -np.ones(n, dtype=np.float64)

    # right‑hand side of the inequality constraints
    b = np.full(2 * n - 1, M, dtype=np.float64)

    bounds = [(0.0, None)] * n

    # use HiGHS interior‑point solver (fast for large sparse LPs)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return None, None
    g = res.x
    return g, g.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate (log‑sum‑exp of the max)
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 300_000):
    """Adam on the log‑softmax parametrisation with a growing temperature."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits = log‑softmax arguments
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5               # start temperature for log‑sum‑exp smoothing
    beta_inc = 0.5
    beta_max = 2000.0

    best_seq = start_vec.copy()
    best_val = _surrogate(best_seq)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        a = np.exp(v - np.max(v))      # softmax (numerically stable)
        a = a / a.sum()

        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # smooth max via log‑sum‑exp with temperature β
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1: n - 1 + n]

        # Jacobian of the softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = 1.0 / math.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # increase temperature
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats).  The function respects the
    overall time budget (default 1000 s) and stores the final sequence in the
    global variable ``height_sequence_1`` for possible reuse.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a small safety margin

    # ------------------------------------------------------------------
    # Choose a tractable problem size for the LP stage.
    # ------------------------------------------------------------------
    n = 2000   # a compromise between quality and LP solving time

    # ------------------------------------------------------------------
    # Initial sequence – try to reuse a previous result if it exists.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and len(previous) >= n:
        f = np.asarray(previous[:n], dtype=np.float64)
        if f.sum() <= 0.0:
            f = _exponential_shape(n, max(0.0, 1.0 - 2.0 / n))
        else:
            f = f / f.sum()
    else:
        # analytic exponential seed (theoretical sweet spot r ≈ 1‑2/n)
        r0 = max(0.0, 1.0 - 2.0 / n)
        f = _exponential_shape(n, r0)

    best_f = f.copy()
    best_val = _surrogate(best_f)
    print(f"\n[Init] n={n}, surrogate={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP‑based mass inflation (alternating improvement)
    # ------------------------------------------------------------------
    lp_iter = 0
    while time.time() < deadline - 5.0 and lp_iter < 15:
        lp_iter += 1
        M = _max_conv(f)
        cur_val = 2.0 * n * M
        print(f"[LP {lp_iter}] cur_val={cur_val:.6f}, M={M:.6e}")

        g, sum_g = _lp_step(f, M)
        if g is None:
            print(f"[LP {lp_iter}] LP failed – exiting LP stage.")
            break

        print(f"[LP {lp_iter}] sum(g)={sum_g:.6f}")

        if sum_g <= 1.0 + 1e-9:
            print(f"[LP {lp_iter}] No improvement (sum(g)≈1). Stopping LP loop.")
            break

        # Scale g back to unit ℓ₁‑norm
        g_scaled = g / sum_g

        # Evaluate the scaled candidate
        cand_val = _surrogate(g_scaled)
        print(f"[LP {lp_iter}] candidate (scaled g) surrogate={cand_val:.6f}")

        if cand_val < cur_val:
            f_new = g_scaled
            print(f"[LP {lp_iter}] Accepting scaled g as new iterate.")
        else:
            # convex combination to keep feasibility
            t = 0.5
            f_new = (1.0 - t) * f + t * g_scaled
            f_new = f_new / f_new.sum()   # guard against rounding drift

        f = f_new
        new_val = _surrogate(f)
        print(f"[LP {lp_iter}] new surrogate after update = {new_val:.6f}")

        if new_val < best_val:
            best_val = new_val
            best_f = f.copy()
            print(f"[LP {lp_iter}] *** New global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinement on the smooth surrogate
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        print(f"\n[Adam] Remaining time {remaining:.1f}s – start refinement.")
        seq_adam, val_adam = _adam_optimize(best_f, n, deadline,
                                            max_iters=int(5e5))
        if val_adam < best_val:
            best_f = seq_adam.copy()
            best_val = val_adam
            print(f"[Adam] Improved global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final exact evaluation and export
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_f.tolist())
        print(f"\n=== FINAL === surrogate={best_val:.6f}, "
              f"true evaluate_sequence={true_val:.6f}")
    except Exception as exc:            # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for reuse
    globals()["height_sequence_1"] = best_f.tolist()
    return best_f.tolist()
