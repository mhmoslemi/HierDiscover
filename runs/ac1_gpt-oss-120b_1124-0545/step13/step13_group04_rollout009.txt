"""
Propose a step‑function (non‑negative vector) that minimizes

        2·n·max(conv(a,a)) / (sum(a)²)

where `conv` is the linear convolution (np.convolve).  The objective is
scale‑invariant, therefore we keep the ℓ₁‑norm equal to one and minimise
`2·n·max(conv)`.  The algorithm consists of three phases

1. **Random sparse sampling** – generate many vectors of the form
   *a_i = 1/S on a random support of size `S`* and keep the best
   (fast surrogate evaluation via FFT).

2. **Smooth‑max gradient descent** – from the best candidate we minimise
   the smooth approximation of the maximum,
       L(a)= (1/β)·log Σ_k exp(β·conv_k(a)),
   whose gradient can be obtained by one additional FFT.  Adam‑style
   updates with Euclidean projection onto the probability simplex keep
   the vector feasible.

3. **Local random perturbations** – cheap moves that transfer a small
   amount of mass between two coordinates are tried while time permits.

All heavy work uses FFT‑based convolution (O(n log n)), allowing many
thousands of iterations within the 1000 s budget.  The best sequence is
stored in the global variable ``height_sequence_1`` for possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray) -> float:
    """Return max(conv(a,a)) for an ℓ₁‑normalised vector."""
    conv = _conv_fft(a, a)
    return float(np.max(conv))

        # 6) slowly raise β to sharpen the approximation
        if it % 5000 == 0:
            beta = min(beta + beta_inc, 200_000.0)
            print(f"[GD] iter={it} beta increased to {beta:.0f}")

    # --------------------------------------------------------------
    # Phase 3 – cheap random perturbations (optional) ---------------
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0:
        # try a few thousand tiny moves while we have time
        moves = int(remaining * 800)            # rough estimate of cheap moves
        for _ in range(moves):
            i, j = rng.choice(n, size=2, replace=False)
            # move at most 10 % of the mass from i to j
            delta = rng.uniform(0.0, a[i] * 0.1)
            if delta <= 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            cur_max = _max_convolution(a_new)
            cur_sur = 2.0 * n * cur_max
            if cur_sur < best_surrogate:
                best_surrogate = cur_sur
                best_a = a_new.copy()
                best_max = cur_max
                a = a_new                         # keep the improved point
                # occasional report (does not affect timing much)
                if _ := (it % 1000):
                    pass

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    result = best_a.tolist()
    true_val = None
    try:
        true_val = evaluate_sequence(result)
        print("\n=== Finished ===")
        print(f"Best surrogate ≈ {best_surrogate:.6f}")
        print(f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = result
    return result
```
