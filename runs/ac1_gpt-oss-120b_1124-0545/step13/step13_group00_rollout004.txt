"""
Propose a candidate step-function (list of non‑negative heights) that
minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is the full linear convolution.  The algorithm consists of:

1. **Rich seed generation** – analytical families (beta, power‑law,
   Gaussian, Tukey, windows), a few random Dirichlet draws and the
   previous best solution (if any).  All seeds are ℓ₁‑normalised.

2. **Coarse‑to‑fine parametric search** – a fast grid evaluation of the
   beta‑distribution, power‑law and Gaussian families; the best few
   seeds are then refined with a low‑dimensional L‑BFGS optimiser.

3. **Continuation L‑BFGS** – smooth max‑convolution via a
   log‑sum‑exp surrogate, with an aggressive β‑schedule that reaches
   β≈10 000, allowing the optimisation to approach the true
   ℓ∞‑objective.

4. **Mirror‑descent (exponentiated‑gradient)** and **projected
   sub‑gradient descent** for fine polishing on the exact max‑convolution.

5. **Simulated‑annealing** in the softmax‑logit space for a final
   local search.

The routine respects a 1000 s wall‑clock budget, checks the time
regularly and stops gracefully.  The best sequence found is stored in
the global variable ``height_sequence_1`` and also returned as a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS and local param optimisation)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (no normalisation assumed)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = exact objective for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # modified Bessel (numpy)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a Dirichlet draw and assign largest/smallest alternatingly."""
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector (softmax of the final logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        # Aggressive schedule extending to very large β.
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0, 10000.0]

    # Initialise logits (log‑softmax parametrisation)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)          # stays on the simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # log‑sum‑exp smoothing
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (derived from the soft‑max representation)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # early stop if we run out of time
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            # deadline reached during this β‑stage
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (softmax parametrisation)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # Sub‑gradient of the active entry (same pattern as in EG)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing style refinement (logit + softmax)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Simple SA in logit space.  Uses a linear cooling schedule and a
    decreasing perturbation scale.
    """
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    rng = np.random.default_rng(12345)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.06 * (1.0 - progress) + 1e-7          # temperature
        sigma = 0.20 * (1.0 - progress) + 1e-5    # proposal width

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def _grid_search_beta(n: int, deadline: float):
    """Return a list of (vector, score) for a coarse beta‑shape grid."""
    vals = []
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                return vals
            vec = _beta_shape(n, a, b)
            score = _objective_surrogate(vec)
            vals.append((vec, score, ('beta', a, b)))
    return vals


def _grid_search_power(n: int, deadline: float):
    """Return a list of (vector, score) for a power‑law grid."""
    vals = []
    for p in np.linspace(0.0, 5.0, 21):
        if time.time() > deadline:
            return vals
        vec = _power(n, p)
        score = _objective_surrogate(vec)
        vals.append((vec, score, ('power', p)))
    return vals


def _grid_search_gaussian(n: int, deadline: float):
    """Return a list of (vector, score) for a Gaussian‑sigma grid."""

    for sigma in np.linspace(0.10, 0.45, 7):

def _grid_search_tukey(n: int, deadline: float):

def _refine_params_beta(start_params, n, deadline):
    """Local refinement of a beta‑shape using low‑dimensional L‑BFGS."""
    if _opt is None:
        return None
    def obj(params):
        a, b = params
        if a <= 0 or b <= 0:
            return 1e9
        vec = _beta_shape(n, a, b)
        return _objective_surrogate(vec)

    bounds = [(1e-3, 10.0), (1e-3, 10.0)]
    try:
        res = _opt.minimize(fun=obj,
                            x0=np.array(start_params, dtype=float),
                            method='L-BFGS-B',
                            bounds=bounds,
                            options={'maxfun': 200, 'ftol': 1e-12})
        if not res.success:
            return None
        a_opt, b_opt = res.x
        return _beta_shape(n, a_opt, b_opt)
    except Exception:  # pragma: no cover
        return None


def _refine_params_power(start_p, n, deadline):
    """Local refinement of a power‑law exponent."""
    if _opt is None:
        return None
    def obj(p):
        p = float(p[0])
        if p < -0.5 or p > 10.0:
            return 1e9
        vec = _power(n, p)
        return _objective_surrogate(vec)

    try:
        res = _opt.minimize(fun=obj,
                            x0=np.array([start_p]),
                            method='L-BFGS-B',
                            bounds=[(-0.5, 10.0)],
                            options={'maxfun': 200, 'ftol': 1e-12})
        if not res.success:
            return None
        p_opt = res.x[0]
        return _power(n, p_opt)
    except Exception:  # pragma: no cover
        return None


def _refine_params_gaussian(start_sigma, n, deadline):
    """Local refinement of a Gaussian sigma‑fraction."""
    if _opt is None:
        return None
    def obj(s):
        s = float(s[0])
        if s <= 0.01 or s >= 0.9:
            return 1e9
        vec = _gaussian(n, s)
        return _objective_surrogate(vec)

    try:
        res = _opt.minimize(fun=obj,
                            x0=np.array([start_sigma]),
                            method='L-BFGS-B',
                            bounds=[(0.01, 0.9)],
                            options={'maxfun': 200, 'ftol': 1e-12})
        if not res.success:
            return None
        sigma_opt = res.x[0]
        return _gaussian(n, sigma_opt)
    except Exception:  # pragma: no cover
        return None

# Final candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Previous solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0:
                prev_arr = prev_arr / prev_arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Coarse parametric grid search (time‑aware)
    # ------------------------------------------------------------------
    print("\n--- Parametric grid search ---")
    candidate_pool = []

    # Uniform baseline
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # Add previous best if available
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr), ('previous',)))

    # Beta‑shape grid
    candidate_pool.extend(_grid_search_beta(n, deadline))
    # Power‑law grid
    candidate_pool.extend(_grid_search_power(n, deadline))
    # Gaussian grid
    candidate_pool.extend(_grid_search_gaussian(n, deadline))
    # Tukey window grid
    candidate_pool.extend(_grid_search_tukey(n, deadline))

    # Random Dirichlet seeds (few)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            score = _objective_surrogate(vec)
            candidate_pool.append((vec, score, ('dirichlet', conc)))

    # ------------------------------------------------------------------
    # 2. Rank seeds and keep a modest top‑k for deeper optimisation
    # ------------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Collected {len(candidate_pool)} seeds, keeping top {top_k}.")

    # ------------------------------------------------------------------
    # 3. Local parametric refinements (optional, cheap)
    # ------------------------------------------------------------------
    refined = []
    for vec, score, meta in candidate_pool[:5]:
        if time.time() > deadline:
            break
        if meta[0] == 'beta':
            a0, b0 = meta[1], meta[2]
            refined_vec = _refine_params_beta((a0, b0), n, deadline)
            if refined_vec is not None:
                refined.append((refined_vec, _objective_surrogate(refined_vec)))
        elif meta[0] == 'power':
            p0 = meta[1]
            refined_vec = _refine_params_power(p0, n, deadline)
            if refined_vec is not None:
                refined.append((refined_vec, _objective_surrogate(refined_vec)))
        elif meta[0] == 'gauss':
            s0 = meta[1]
            refined_vec = _refine_params_gaussian(s0, n, deadline)
            if refined_vec is not None:
                refined.append((refined_vec, _objective_surrogate(refined_vec)))
    # Add the refined vectors to the pool and re‑select top seeds
    for vec, val in refined:
        candidate_pool.append((vec, val, ('refined',)))
    candidate_pool.sort(key=lambda x: x[1])
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]

    # ------------------------------------------------------------------
    # 4. Continuation L‑BFGS on the best seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break

        remaining = deadline - time.time()
        # allocate a modest share of the remaining budget
        max_iter_beta = int(min(3500, remaining * 0.07 * 1e5))
        max_iter_beta = max(800, max_iter_beta)

        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter/beta={max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         max_iter_per_beta=max_iter_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

        # --------------------------------------------------------------
        # Short exponentiated‑gradient polishing
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 2.0:
            max_exp = int(min(250_000, remaining * 0.06 * 1e5))
            seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_exp)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Long sub‑gradient polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(600_000, remaining * 0.07 * 1e5))
        if max_sub > 0:
            print("\n--- Long sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing refinement (final push)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(500_000, remaining * 0.06 * 1e5))
        if max_sa > 0:
            print("\n--- Simulated‑annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing(global_best_seq,
                                                  n,
                                                  deadline,
                                                  max_iters=max_sa)
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[SA] final best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
