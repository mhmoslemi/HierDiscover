"""
Propose a low‑value step‑function sequence.

The algorithm proceeds in three phases:
1.  A very large and diverse pool of *seed* vectors is generated
    (analytic windows, random Dirichlet / exponential draws,
    low‑frequency cosine mixtures, etc.).  Each seed is ℓ₁‑normalised
    (sum=1) and its exact objective 2·n·max(conv(a,a)) is computed.
    The best few seeds are kept.

2.  From the best seed we run a long *exponentiated‑gradient* (mirror‑descent)
    optimisation on the true max‑convolution objective.  The step size
    ηₜ = 0.5/√t is used and the iteration count is limited only by the
    remaining time budget.

3.  A very cheap stochastic *local‑search* fine‑tuner takes the current
    best sequence and repeatedly moves a tiny amount of mass from one
    index to another (keeping the ℓ₁‑norm fixed).  Moves that improve the
    objective are kept; a simple simulated‑annealing acceptance rule
    allows occasional uphill moves to escape shallow local minima.

The final sequence is returned as a plain list of floats and stored in the
global variable ``height_sequence_1``.  All computation respects the
global 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper routines
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) for a normalised (ℓ₁=1) vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    # the function is homogeneous; we assume sum(seq)=1 (or close)
    return float(2.0 * n * max_c)


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # modified Bessel function (numpy)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed collection (larger and more diverse than before)
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds = []

    # Analytic families (dense parameter sweeps)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in np.linspace(0.3, 2.0, 9):
        seeds.append(_sine_window(n, gamma=gamma))

    for sigma in np.linspace(0.08, 0.45, 13):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 5.0, 26):
        seeds.append(_power(n, p))

    # Beta‑shape random draws (instead of a grid)
    for _ in range(200):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        seeds.append(_beta_shape(n, a, b))

    # Kaiser windows with many β values
    for beta in np.linspace(2, 12, 11):
        seeds.append(_kaiser_window(n, beta))

    # Low‑frequency random curves
    for _ in range(200):
        nf = rng.integers(2, 7)
        seeds.append(_lowfreq_curve(n, nf, rng))

    # Random mixtures of uniform + low‑freq curves
    for _ in range(100):
        curve = _lowfreq_curve(n, rng.integers(2, 6), rng)
        alpha = rng.uniform(0.1, 0.9)
        seeds.append(_mix_shapes(_uniform(n), curve, alpha))

    # Random Dirichlet draws (wide range of concentrations)
    for _ in range(600):
        conc = 10 ** rng.uniform(-2.5, 1.0)  # from ~0.003 to 10
        seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(200):
        seeds.append(_random_exponential(n, rng))

    # Include the previous solution if it matches the size
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise and guard against pathological vectors
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser on the exact max‑conv.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # work in log‑space to stay on the simplex after softmax
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (as in the original code)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size schedule (larger initial steps)
        eta = 0.7 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic local search (mass‑transfer moves)
# ----------------------------------------------------------------------
def _local_search(seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_moves: int = 300_000) -> tuple[np.ndarray, float]:
    """Tiny random mass‑transfer moves, accepting only improving moves."""
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rng = np.random.default_rng()
    move_cnt = 0
    while time.time() < deadline and move_cnt < max_moves:
        move_cnt += 1

        # pick two distinct positions
        i = rng.integers(n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        # transfer a modest amount of mass
        eps = min(1e-4, a[i] * 0.05)   # up to 5 % of the source weight
        if eps <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        cur_val = _objective(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()
            a = a_new
            if move_cnt % 25000 == 0:
                print(f"[LocalSearch] moves={move_cnt} best={best_val:.6f}")

    return best_seq, best_val
