start = time.time()
deadline = start + budget_s - 5

# Try baseline from previous:
if previous exists:
   evaluate and keep

# Add uniform baseline similar to prior

# Loop while time remains:
for restart in range(restarts):
   allocate fraction of time for this restart based on remaining time.
   generate random initial a: random support size S = rng.integers(int(0.3*n), int(0.9*n)+1)
   a = np.zeros(n)
   support = rng.choice(n, size=S, replace=False)
   a[support] = 1.0 / S
   a = a / np.sum(a)  # sum=1 already
   
   # compute conv and best locally.
   best_local_val = 2 * n * np.max(_conv_fft(a,a))
   best_local_seq = a.copy()
   
   # gradient descent with smoothing.
   it = 0
   alpha = 1.0
   while time.time() < deadline - 0.5 and it < max_iters_per_restart:
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_local_val:
            best_local_val = cur_val
            best_local_seq = a.copy()
        # compute softmax weights
        max_c = np.max(conv)
        exp_vals = np.exp(alpha * (conv - max_c))
        w = exp_vals / np.sum(exp_vals)
        # gradient
        w_rev = w[::-1]
        conv_w_rev_a = _conv_fft(w_rev, a)
        Lw = len(w)
        g = 2.0 * np.flip(conv_w_rev_a[Lw - n : Lw])
        # step size schedule
        lr = 0.4 / math.sqrt(it+1)
        a = a - lr * g
        a = _proj_simplex(a)
        # increase alpha gradually
        if (it+1) % 100 == 0:
            alpha = min(alpha * 1.2, max_alpha)   # cap at e.g., 50
        it += 1
        # could break if no progress.
   # after loops update global best if improved.
   if best_local_val < best_val:
        best_val = best_local_val
        best_seq = best_local_seq.copy()
        print(f"[Restart {restart}] new best val {best_val:.6f}")

"""
Propose a step‑function (non‑negative sequence) that minimizes
evaluate_sequence(sequence) = 2·n·max(conv(a,a)) / (sum(a)^2).

The algorithm uses several complementary ideas:

1. **Warm‑starts** – re‑use a previously found sequence if it matches the
   current length, otherwise start from the uniform distribution.

2. **Random sparse initialisations** – for each restart we pick a random
   support of size S∈[0.2 n, 0.9 n] and give the support equal mass.
   This already spreads the ℓ₁‑mass over many entries.

3. **Smooth max‑descent** – we minimise a smooth approximation of the
   “max‑convolution’’:
        f_α(a) = (1/α)·log Σ_k exp(α·c_k) ,  c = a*a .
   Its gradient is
        ∇f_α = 2·Σ_k w_k·a_{k‑i}  with  w = softmax(α·c).
   The gradient is evaluated via a cross‑correlation implemented with FFTs.
   The temperature α is increased gradually so that the optimisation
   first smoothes the landscape and later focuses on the true max.

4. **ℓ₂‑flattening descent** – as a secondary driver we also minimise the
   squared deviation of the convolution from the perfectly flat target
   μ = 1/(2·n‑1).  The gradient of
        J(a) = Σ_k (c_k‑μ)²
   is  ∇J = 2·cross_corr(c‑μ, a) and is again computed with FFTs.

   Both gradients are blended (weight γ) and a Euclidean projection onto the
   probability simplex (∑a=1, a≥0) is applied after each step.

5. **Final sub‑gradient polishing** – a few iterations of the classic
   sub‑gradient (max‑only) step are performed on the current best vector.

All heavy work (two FFT convolutions per iteration) is cheap compared to the
official scorer, so many restarts and iterations fit easily into the 1000 s
budget.  The best sequence found is stored in the global variable
``height_sequence_1`` for possible reuse.

The function prints timing and improvement information and finally returns
the best list of heights.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n·log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    return np.maximum(v - theta, 0.0)


def _surrogate(a: np.ndarray, n: int) -> float:
    """Fast surrogate 2·n·max(conv(a,a)) for an ℓ₁‑normalised vector."""
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final expensive call
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------
    # Problem dimension (a compromise between quality and speed)
    # --------------------------------------------------------
    n = 8000                     # length of the vector

    # --------------------------------------------------------
    # Keep the best solution found so far
    # --------------------------------------------------------
    best_seq = None
    best_surrogate = float("inf")

    # --------------------------------------------------------
    # Warm‑start from a previously stored solution (if any)
    # --------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size == n and np.sum(prev_arr) > 0:
                prev_norm = prev_arr / np.sum(prev_arr)
                val = _surrogate(prev_norm, n)
                best_seq, best_surrogate = prev_norm, val
                print(f"[Warm‑start] previous solution surrogate = {val:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------
    # Baseline: the uniform distribution (surrogate = 2)
    # --------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    val_uni = _surrogate(uniform, n)
    if val_uni < best_surrogate:
        best_seq, best_surrogate = uniform, val_uni
        print(f"[Init] uniform surrogate = {val_uni:.6f}")

    # --------------------------------------------------------
    # Helper: compute the gradient of the smooth‑max objective
    # --------------------------------------------------------
    def _grad_smooth_max(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
        """∇f_α where f_α is the log‑sum‑exp smoothing of max(conv)."""
        # soft‑max weights
        max_c = np.max(conv)
        exp_vals = np.exp(alpha * (conv - max_c))
        w = exp_vals / np.sum(exp_vals)                # length 2n‑1
        # cross‑correlation w ⋆ a  (g_i = 2 Σ_j w_{i+j} a_j)
        w_rev = w[::-1]
        tmp = _conv_fft(w_rev, a)                      # length 3n‑2
        Lw = len(w)                                    # = 2n‑1
        # slice indices Lw‑n … Lw‑1  (size n) and reverse
        g = 2.0 * np.flip(tmp[Lw - n : Lw])
        return g

    # --------------------------------------------------------
    # Helper: gradient of the L2 flattening term J(a) = Σ (conv‑μ)²
    # --------------------------------------------------------
    mu_const = 1.0 / (2 * n - 1)                       # target flat value

    def _grad_flatten(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
        """Gradient of Σ (conv‑μ)² w.r.t. a."""
        d = conv - mu_const                           # can be negative
        d_rev = d[::-1]
        tmp = _conv_fft(d_rev, a)                     # length 3n‑2
        Ld = len(d)                                   # = 2n‑1
        g = 2.0 * np.flip(tmp[Ld - n : Ld])
        return g

    # --------------------------------------------------------
    # Main optimisation loop – many short restarts
    # --------------------------------------------------------
    restart_id = 0
    while time.time() < deadline - 1.0:
        # -----------------------------------------------------------------
        # Random sparse initialisation
        # -----------------------------------------------------------------
        frac = rng.uniform(0.20, 0.90)
        S = max(2, int(frac * n))
        support = rng.choice(n, size=S, replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[support] = 1.0 / S
        a = a / np.sum(a)                               # ℓ₁‑normalised

        # quick surrogate evaluation for the start point
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_surrogate:
            best_surrogate, best_seq = cur_val, a.copy()
            print(f"[Restart {restart_id}] init surrogate = {cur_val:.6f}")

        # -----------------------------------------------------------------
        # Gradient descent with blended objective
        # -----------------------------------------------------------------
        # We blend the smooth‑max gradient (weight γ) and the flattening gradient.
        gamma = 0.6                      # more focus on flattening initially
        alpha = 0.5                      # temperature for soft‑max
        max_alpha = 60.0
        it = 0
        # allocate a reasonable number of iterations for this restart
        # (empirically ~1500 iters per second)
        remaining = deadline - time.time()
        max_iters = int(min(8000, remaining * 1400))
        while it < max_iters and time.time() < deadline - 0.5:
            it += 1
            # compute current convolution
            conv = _conv_fft(a, a)
            max_conv = np.max(conv)
            cur_val = 2.0 * n * max_conv

            # keep global best
            if cur_val < best_surrogate:
                best_surrogate, best_seq = cur_val, a.copy()
                print(f"[Restart {restart_id}] new global best = {cur_val:.6f} (it {it})")

            # gradients
            g_smooth = _grad_smooth_max(a, conv, alpha)
            g_flat = _grad_flatten(a, conv)
            g = gamma * g_smooth + (1.0 - gamma) * g_flat

            # diminishing step size
            lr = 0.6 / math.sqrt(it)
            a = a - lr * g
            a = _proj_simplex(a)

            # slowly raise the temperature (sharpen the max)
            if it % 150 == 0:
                alpha = min(alpha * 1.2, max_alpha)

        restart_id += 1

    # ------------------------------------------------------------
    # Short sub‑gradient polishing on the best sequence found
    # ------------------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
        # compute a small number of classic sub‑gradient steps
        max_iters = 3000
        it = 0
        while it < max_iters and time.time() < deadline - 0.5:
            it += 1
            conv = _conv_fft(a, a)
            k = int(np.argmax(conv))
            max_val = conv[k]
            surrogate = 2.0 * n * max_val
            # update global best if improved
            if surrogate < best_surrogate:
                best_surrogate, best_seq = surrogate, a.copy()
                if it % 500 == 0:
                    print(f"[Sub‑grad] it {it} improved best = {surrogate:.6f}")

            # sub‑gradient for the active index
            g = np.zeros_like(a)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx
            g[idx] = 2.0 * a[rev]

            lr = 0.30 / math.sqrt(it)
            a = a - lr * g
            a = _proj_simplex(a)

    # ------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print("\n=== Finished ===")
        print(f"Best surrogate ≈ {best_surrogate:.6f}")
        print(f"Official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
