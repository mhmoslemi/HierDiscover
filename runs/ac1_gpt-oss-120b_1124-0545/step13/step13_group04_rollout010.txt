"""
Propose a step‑function (non‑negative sequence) with a low value of

    2·n·max(conv(a,a)) / (sum(a)²)

The strategy consists of three phases:

1. **Fast random sampling** – generate many uniform‑weight supports of
   size `S ≈ n/2`.  A random low‑discrepancy (golden‑ratio) support is also
   evaluated because it often yields a flatter self‑convolution.
2. **Hill‑climbing on the support** – starting from the best candidate,
   repeatedly swap one index inside the support with one outside it.  The
   surrogate `2·n·max(conv)` is recomputed (FFT‑based, O(n log n)) and the
   swap is kept if it improves the surrogate.  Simple simulated‑annealing
   is used to occasionally accept worsening moves.
3. **Smooth gradient descent** – from the best support we run a few
   iterations of projected gradient descent on a smoothed version of the
   max‑convolution (soft‑max with a large temperature).  The update uses the
   exact gradient `∇ conv = 2·(w * a_rev)` and Euclidean projection onto the
   probability simplex.

All operations respect the 1000 s time budget; the final sequence is
validated with `evaluate_sequence` and stored globally as
`height_sequence_1` for possible reuse.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _surrogate(a: np.ndarray, n: int) -> float:
    """Fast surrogate 2·n·max(conv(a,a)) for an ℓ₁‑normalised vector."""
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))

            best_surrogate = val
            best_seq = a.copy()
            print(f"[Sample] iter={sample_iter:6d} new best surrogate = {val:.6f}")

        if sample_iter % 1000 == 0:
            elapsed = time.time() - start_time
            print(f"[Sample] {sample_iter} candidates, best={best_surrogate:.6f}, elapsed={elapsed:.1f}s")

    # --------------------------------------------------------
    # Stage 2 – hill‑climbing on the support (swap one index)
    # --------------------------------------------------------
    if best_seq is not None:
        cur_a = best_seq.copy()
        cur_val = best_surrogate
        support = set(np.flatnonzero(cur_a > 0))
        S = len(support)                     # should stay constant
        weight = 1.0 / S
        swap_iter = 0
        temperature = 0.0005                # tiny annealing temperature
        print("\n--- Hill‑climbing on support ---")
        while time.time() < deadline - 2.0:
            # pick a random element to remove
            i_out = rng.choice(list(support))
            # pick a random element not currently in the support
            i_in = rng.integers(0, n)
            while i_in in support:
                i_in = rng.integers(0, n)

            # propose the swap (weights stay identical)
            trial_a = cur_a.copy()
            trial_a[i_out] = 0.0
            trial_a[i_in] = weight

            val = _surrogate(trial_a, n)
            swap_iter += 1
            # accept if better, otherwise with a tiny probability to escape plateaus
            if val < cur_val or rng.random() < math.exp(-(val - cur_val) / temperature):
                cur_a = trial_a
                cur_val = val
                support.remove(i_out)
                support.add(i_in)
                if cur_val < best_surrogate:
                    best_surrogate = cur_val
                    best_seq = cur_a.copy()
                    print(f"[Hill] iter={swap_iter:6d} new best surrogate = {cur_val:.6f}")

            # very gentle cooling (not really needed but harmless)
            temperature *= 0.99999

            # occasional progress report
            if swap_iter % 2000 == 0:
                elapsed = time.time() - start_time
                print(f"[Hill] iter={swap_iter}, cur={cur_val:.6f}, best={best_surrogate:.6f}, elapsed={elapsed:.1f}s")

    # --------------------------------------------------------
    # Stage 3 – short smoothed gradient descent (optional)
    # --------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 4.0 and best_seq is not None:
        a = best_seq.copy()
        a = a / np.sum(a)                      # enforce ℓ₁‑norm = 1
        best_local_seq = a.copy()
        best_local_val = best_surrogate
        alpha = 5e5                             # smoothing parameter (≈ max α·conv ≈ 30)
        lr = 0.25                               # initial learning rate
        it = 0
        print("\n--- Smoothed gradient descent ---")
        while time.time() < deadline - 1.0:
            conv = _conv_fft(a, a)
            cur_max = np.max(conv)
            cur_sur = 2.0 * n * cur_max

            # soft‑max weights for the “max”
            shift = np.max(alpha * conv)        # avoid overflow
            w = np.exp(alpha * conv - shift)
            w = w / np.sum(w)                   # shape = (2n‑1,)

            # gradient = 2 * (w * a_rev) (full convolution, then slice)
            a_rev = a[::-1]
            conv_wr = _conv_fft(w, a_rev)       # length 3n‑2
            grad = 2.0 * conv_wr[n - 1 : n - 1 + n]

            # projected gradient step
            a = a - lr * grad
            a = _proj_simplex(a)

            # keep the best seen
            if cur_sur < best_local_val:
                best_local_val = cur_sur
                best_local_seq = a.copy()
                print(f"[Grad] iter={it:5d} new best surrogate = {cur_sur:.6f}")

            # simple schedule
            lr *= 0.9995
            it += 1
            if it >= 5000:          # safety cap
                break

        if best_local_val < best_surrogate:
            best_surrogate = best_local_val
            best_seq = best_local_seq.copy()
            print(f"[Grad] final best surrogate after refinement = {best_local_val:.6f}")

    # --------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_surrogate:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                      # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
