def adam_optimize(start_seq, n, deadline, max_iters=200000):
    # start_seq ℓ1 normalized
    # logit initialization: z = np.log(start_seq + epsilon)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1=0.9; beta2=0.999; eps=1e-8; lr=0.1 (works in logit space)
    best_seq = start_seq.copy()
    best_obj = _objective_surrogate(start_seq) # or evaluate exact later
    for t in range(1, max_iters+1):
        if time.time()>deadline: break
        a = _softmax(z)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # compute subgradient:
        g = np.zeros_like(a)
        i_start = max(0, max_idx-(n-1))
        i_end = min(n-1, max_idx)
        idx = np.arange(i_start, i_end+1, dtype=int)
        rev_idx = max_idx-idx
        # each pair appears once per ordered direction: but conv computed as sum_i a[i]*a[k-i] includes both orders when i != k-i. So gradient w.r.t a = 2 * (conv of a_rev with indicator w?), earlier exponentiated gradient used that exact formula: gradient of conv[k] w.r.t a_i = a_{k-i} + a_{k-i}? Actually derivative of conv[k] = sum_i a_i a_{k-i} w.r.t a_j is a_{k-j} + a_{k-j} (maybe due to symmetry). However since conv uses sum_i a_i a_{k-i} (ordered pairs), derivative w.r.t a_j = a_{k-j} + a_{k-j} (since term i=j yields a_j*a_{k-j}, and term i=k-j yields a_{k-j}*a_j) but that double-counts when i=j i.e., when k=2j. But we can approximate with 2*a_{k-j} for simplicity (as earlier exponentiated gradient used g[idx] = 2*a[rev_idx]), ignoring diagonal special case; that's fine.

        g[idx] = 2.0 * a[rev_idx]   # same as exponentiated gradient code
        # But we need gradient normalized for logit: grad_z = a * (g - np.dot(g, a))
        dot = np.dot(g, a)
        grad_z = a * (g - dot)

        # Adam update
        m = beta1*m + (1-beta1)*grad_z
        v = beta2*v + (1-beta2)*(grad_z**2)
        m_hat = m / (1-beta1**t)
        v_hat = v / (1-beta2**t)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # compute objective:
        obj = _objective_surrogate(a)  # surrogate
        if obj < best_obj:
            best_obj = obj
            best_seq = a.copy()
    return best_seq, best_obj

"""
Adaptive Adam‑based mirror‑descent optimizer for the max‑convolution
ratio

The evaluation function is scale‑invariant, therefore the search is
performed on the probability simplex (ℓ₁‑normalised vectors) using a
soft‑max parametrisation.  For a vector a = softmax(z) the exact
non‑smooth objective is

    F(a) = 2·n·max(conv(a,a))                     (∑a = 1)

Its sub‑gradient at the active convolution index k* is

    g_i = 2·a_{k*‑i}   (if 0 ≤ i ≤ n‑1 and 0 ≤ k*‑i ≤ n‑1)
    g_i = 0           otherwise

The gradient on the log‑it space is  ∇_z F = a·(g – ⟨g,a⟩).  We
optimise this using Adam (adaptive moment estimation) and project
implicitly onto the simplex with the soft‑max.  The algorithm works in
three phases:

1.  Load the previous best sequence (if any) and use it as the initial
    point; otherwise start from the uniform distribution.
2.  Run a long Adam mirror‑descent on the exact objective.  The step‑size
    is constant, Adam’s momentum provides fast convergence, and a tiny
    Gaussian perturbation is added every 50 000 iterations to escape
    shallow local minima.
3.  If time permits, up‑sample the current best vector (linear
    interpolation) to a finer grid (2× points) and run a second Adam
    phase on the finer resolution.

The whole routine stops before the allotted budget (default 1000 s) and
returns the best sequence found.  All helper functions are defined at
module level, no closures or lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta‑like shape: (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a larger grid, ℓ₁‑renormalised."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = new_seq / new_seq.sum()
    return new_seq


def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector (sum==1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Adam mirror‑descent optimiser (exact max‑convolution)
# ----------------------------------------------------------------------


def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Adam‑based mirror descent on the exact objective.
    Returns the best sequence seen and its surrogate value.
    """
    # initialise logits (log‑softmax parameters)
    z = np.log(start_seq + 1e-30)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.3                     # learning rate in log‑it space
    rng = np.random.default_rng(0)

    best_seq = _softmax(z)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # probability vector
        a = _softmax(z)

        # full convolution
        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k_star - idx
            g[idx] = 2.0 * a[rev_idx]          # see analysis notes

        # gradient on the logits (soft‑max Jacobian)
        dot = np.dot(g, a)
        grad_z = a * (g - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional random perturbation to encourage escape
        if t % 50000 == 0:
            z += 0.01 * rng.standard_normal(size=n)

        # keep the best point seen so far (using the cheap surrogate)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if t % 100000 == 0:
                print(f"[Adam] iter={t:,} best surrogate={best_val:.6f}")

    return best_seq, best_val

    ``evaluate_sequence`` within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # Initialisation
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Fixed resolution (start with 4000 points – a good trade‑off)
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimisation start (n={n}) ===")

    # ------------------------------------------------------------------
    # Load previous solution (if any) and normalise it to ℓ₁ = 1
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape[0] != n:
                prev_arr = _upsample_sequence(prev_arr, n)
            prev_arr = prev_arr / prev_arr.sum()
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # ------------------------------------------------------------------
    # Baseline (fallback) – uniform distribution
    # ------------------------------------------------------------------
    baseline = _uniform(n) if prev_arr is None else prev_arr.copy()

    try:
        baseline_val = evaluate_sequence(baseline.tolist())
    except Exception:   # pragma: no cover
        baseline_val = float("inf")
    best_seq = baseline.copy()
    best_val = baseline_val
    print(f"[Baseline] evaluate_sequence = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Seed collection – a modest yet diverse pool
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = [baseline]

    # a few asymmetric Beta shapes
    alphas = np.linspace(0.2, 5.0, 6)
    betas = np.linspace(0.2, 5.0, 6)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (different concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves (smooth, non‑symmetric)
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # simple linear ramps (highly asymmetric)
    ramp = np.linspace(0.0, 1.0, n, dtype=np.float64)
    ramp = ramp / ramp.sum()
    seeds.append(ramp)
    seeds.append(ramp[::-1])

    # ------------------------------------------------------------------
    # Normalise all seeds (protect against accidental zeros)
    # ------------------------------------------------------------------
    clean_seeds = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            clean_seeds.append(v / s)

    # ------------------------------------------------------------------
    # Primary optimisation pass – Adam from each seed (time‑aware)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(clean_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached – stopping seed loop")
            break

        # allocate a modest number of iterations; the loop itself will stop
        # when the global deadline expires.
        remaining = deadline - time.time()
        # heuristic: ~200 000 iterations per minute of remaining time
        max_iters = int(min(300_000, remaining * 200.0))
        if max_iters < 10_000:
            max_iters = int(remaining * 200.0)   # at least a few thousand

        print(f"\n--- Adam optimisation from seed #{idx + 1} "
              f"(max_iters={max_iters:,}) ---")
        seq_opt, surrogate_opt = _adam_optimize(seed_vec, n, deadline,
                                                max_iters=max_iters)

        # evaluate the exact objective (cheap because sum=1)
        try:
            true_val = evaluate_sequence(seq_opt.tolist())
        except Exception:
            true_val = float("inf")

        print(f"[Result] true evaluate_sequence = {true_val:.6f}")

        if true_val < best_val:
            best_val = true_val
            best_seq = seq_opt.copy()
            print(f"[Update] New global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling to a finer grid (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 30.0:
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} for final refinement ---")
            up_seq = _upsample_sequence(best_seq, new_n)

            # a longer Adam run on the finer grid
            remaining = deadline - time.time()
            max_iters = int(min(300_000, remaining * 200.0))
            seq_up, _ = _adam_optimize(up_seq, new_n, deadline,
                                       max_iters=max_iters)

            try:
                up_val = evaluate_sequence(seq_up.tolist())
            except Exception:
                up_val = float("inf")

            print(f"[Upsample] evaluate_sequence = {up_val:.6f}")

            if up_val < best_val:
                best_val = up_val
                best_seq = seq_up.copy()
                n = new_n
                print(f"[Update] New best after up‑sampling = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final reporting
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest evaluate_sequence = {final_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # make the result globally available for the next round
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
