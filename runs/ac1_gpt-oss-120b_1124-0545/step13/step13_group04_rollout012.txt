"""search for a low‑value step‑function

The algorithm works on the integer “pair‑count’’ representation of a binary
support.  For a support `A⊂{0,…,n‑1}` of size `S` the ordered‑pair
convolution `c = conv(1_A,1_A)` satisfies

    c[k] = #{(i,j)∈A×A : i+j = k} .

Since the objective for a uniform weight on the support is

    2·n·max(c) / S² ,

minimising `max(c)` directly minimises the evaluation function.
Starting from a random support we repeatedly *swap* an element of the
support with an element outside it.  The swap updates the count array in
O(S) time, and we only accept a swap if the maximal count does not
increase (strict improvement is preferred, equal‑count swaps are accepted
with a small probability to escape plateaux).  The search is bounded by a
time budget (default 1000 s).  After the combinatorial phase a short
projected‑gradient refinement on the original continuous objective is
applied.

The final sequence uses the support found (uniform weight `1/S`) and is
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _surrogate(a: np.ndarray, n: int) -> float:
    """Fast surrogate 2·n·max(conv(a,a)) for an ℓ₁‑normalised vector."""
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))

# ----------------------------------------------------------------------

    n = 8000                         # length of the vector

            if prev_arr.size == n and np.sum(prev_arr) > 0:
                val = _surrogate(prev_arr / np.sum(prev_arr), n)
                best_seq = prev_arr / np.sum(prev_arr)
                best_surrogate = val
                best_mask = prev_arr > 0
                best_S = int(np.count_nonzero(best_mask))

    while time.time() < deadline - 5.0:
        # pick a support size around ½·n (the exact fraction is random)
        frac = rng.uniform(0.35, 0.65)
        S = max(2, int(frac * n))

        # start from a fresh random support (or reuse the current best)
        if best_mask is not None and rng.random() < 0.15:
            mask = best_mask.copy()
            # ensure the mask has size S; if sizes differ, adjust randomly
            cur_S = mask.sum()
            if cur_S > S:
                # drop random elements
                drop_idx = rng.choice(np.flatnonzero(mask), size=cur_S - S, replace=False)
                mask[drop_idx] = False
            elif cur_S < S:
                # add random elements
                add_candidates = np.flatnonzero(~mask)
                add_idx = rng.choice(add_candidates, size=S - cur_S, replace=False)
                mask[add_idx] = True
            # now S matches the desired size
        else:
            idx = rng.choice(n, size=S, replace=False)
            mask = np.zeros(n, dtype=bool)
            mask[idx] = True

        indices = np.flatnonzero(mask)                 # current support (size S)
        # ordered‑pair count via linear convolution of the indicator
        C = np.convolve(mask.astype(np.int64), mask.astype(np.int64))
        cur_max = int(C.max())

        # compute a threshold count that would already meet the target surrogate
        target_Cmax = int(math.floor(target_surrogate * S * S / (2.0 * n)))

        # ------------------------------------------------------------------
        # local combinatorial refinement – swaps
        # ------------------------------------------------------------------
        it_local = 0
        while time.time() < deadline - 2.0:
            # locate a sum with maximal count
            k_max = int(C.argmax())

            # contributors: i ∈ A such that (k_max‑i) ∈ A
            partner = k_max - indices
            valid = (partner >= 0) & (partner < n) & mask[partner]
            contributors = indices[valid]

            if contributors.size > 0:
                i_remove = rng.choice(contributors)
            else:
                i_remove = rng.choice(indices)

            # pick a new index outside the support
            q = rng.integers(0, n)
            while mask[q]:
                q = rng.integers(0, n)

            # ----------------------------------------------------------------
            # compute the new count array after the swap
            # ----------------------------------------------------------------
            C_new = C.copy()

            # remove contributions of i_remove
            sums_i = i_remove + indices                 # size S
            C_new[sums_i] -= 2
            C_new[2 * i_remove] += 1                     # fix self‑pair (net –1)

            # new support after removal
            new_indices = indices[indices != i_remove]   # size S‑1

            # add contributions of q
            sums_q = q + new_indices                     # size S‑1
            C_new[sums_q] += 2
            C_new[2 * q] += 1                            # self‑pair for q

            new_max = int(C_new.max())

            # accept the swap if it does not increase the maximal count
            accept = False
            if new_max < cur_max:
                accept = True
            elif new_max == cur_max and rng.random() < 0.03:
                # occasional equal‑count moves to escape plateaux
                accept = True

            if accept:
                # update data structures
                mask[i_remove] = False
                mask[q] = True
                indices = np.append(new_indices, q)
                C = C_new
                cur_max = new_max

                # keep the best mask seen so far (global)
                surrogate_val = 2.0 * n * cur_max / (S * S)
                if surrogate_val < best_surrogate:
                    best_surrogate = surrogate_val
                    best_mask = mask.copy()
                    best_S = S
                    print(f"[Global] new best surrogate = {surrogate_val:.6f} (S={S})")

                if cur_max <= target_Cmax:
                    # already good enough for this support size
                    break

            it_local += 1

            # safety break if we have exhausted the time budget
            if time.time() > deadline - 5.0:
                break

        # early exit if the global target has been reached
        if best_surrogate <= target_surrogate:
            break

    # ------------------------------------------------------------------
    # construct the best sequence found so far (uniform weight on support)
    # ------------------------------------------------------------------
    if best_mask is None:
        # fallback – uniform sequence
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_S = n
    else:
        best_seq = np.zeros(n, dtype=np.float64)
        best_seq[best_mask] = 1.0 / best_S   # ℓ₁‑normalised by construction

    # ------------------------------------------------------------------
    # optional short projected sub‑gradient refinement on the continuous
    # objective (helps shave the last few 1e‑4)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 4.0:
        max_iters = int(min(15000, remaining * 1400))
        print(f"\n--- Sub‑gradient refinement (max_iters={max_iters}) ---")
        a = best_seq.copy()
        best_local_val = best_surrogate
        best_local_seq = a.copy()
        for it in range(1, max_iters + 1):
            conv = _conv_fft(a, a)
            max_idx = int(conv.argmax())
            cur_val = 2.0 * n * conv[max_idx]
            if cur_val < best_local_val:
                best_local_val = cur_val
                best_local_seq = a.copy()
            # sub‑gradient of the active entry
            g = np.zeros_like(a)
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            g[idx_i] = 2.0 * a[rev_idx]
            # step‑size schedule + projection onto simplex
            lr = 0.30 / math.sqrt(it)
            a = a - lr * g
            a = _proj_simplex(a)
            if time.time() > deadline - 1.0:
                break
        if best_local_val < best_surrogate:
            best_surrogate = best_local_val
            best_seq = best_local_seq.copy()
            print(f"[Sub] refined surrogate = {best_surrogate:.6f}")

    # ------------------------------------------------------------------
    # final call to the official scorer
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_surrogate:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible reuse in a following run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
