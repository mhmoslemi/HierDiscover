"""
Propose a candidate step‑function using a fast targeted annealing (mass‑transfer)
meta‑heuristic followed by a final L‑BFGS refinement.
The annealer repeatedly moves a small amount of mass from a high‑contributing
position (with respect to the current maximal convolution entry) to a random
position, updating the full convolution exactly in O(n) time per move.
The procedure preserves the ℓ₁‑norm (sum = 1) and non‑negativity.
After the annealing phase a continuation L‑BFGS optimisation on a
log‑sum‑exp smooth approximation of the max‑convolution is applied to polish
the result.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for an ℓ₁‑normalised vector (2·n·max(conv))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic / random seeds (lightweight)
# ----------------------------------------------------------------------
def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with alternating large / small entries."""
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """A plain Dirichlet draw."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    L‑BFGS‑B on the log‑sum‑exp smoothing of max(conv(a,a)).
    Returns an ℓ₁‑normalised vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy is required for the L‑BFGS stage.")

    if beta_schedule is None:
        beta_schedule = [1.0, 10.0, 100.0, 1000.0, 10000.0, 50000.0, 100000.0]

    # logits for the simplex (softmax parametrisation)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # log‑sum‑exp approximation of max
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                     # softmax weights
            rev_a = a[::-1]
            grad_a = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print("[L‑BFGS] time limit reached during β={}".format(beta_val))
            break
        except Exception as e:  # pragma: no cover
            print("[L‑BFGS] exception:", e)
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Fast targeted annealing (mass‑transfer) optimiser
# ----------------------------------------------------------------------
def _anneal_opt(initial_seq: np.ndarray,
                deadline: float,
                direct_prob: float = 0.75,
                start_step: float = 0.5,
                min_step: float = 0.001,
                report_every: int = 50000) -> np.ndarray:
    """
    Mass‑transfer simulated annealing.
    Keeps sum = 1 and non‑negativity.
    Returns the best sequence found.
    """
    n = len(initial_seq)
    a = initial_seq.copy()                     # current vector (sum = 1)
    conv = _conv_fft(a, a)                     # exact convolution
    cur_peak = float(conv.max())
    best_seq = a.copy()
    best_peak = cur_peak
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        it += 1

        # fraction of elapsed time for schedule
        elapsed_frac = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - (deadline - time.time())) + 1e-12)
        # linear decay of step size
        step_factor = start_step * (1.0 - elapsed_frac) + min_step * elapsed_frac

        # ----- choose donor / recipient -----
        if rng.random() < direct_prob:
            # directed move: pick donor among indices that contribute most to current max
            k_star = int(np.argmax(conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            rev = k_star - idx
            # scores = a[i] * a[rev]
            scores = a[idx] * a[rev]
            if scores.size == 0:
                continue
            donor = idx[int(np.argmax(scores))]
            if a[donor] <= 0.0:
                continue
            # recipient chosen uniformly from the whole domain, avoiding donor
            recip = rng.integers(n - 1)
            if recip >= donor:
                recip += 1
        else:
            donor = rng.integers(n)
            if a[donor] <= 0.0:
                continue
            recip = rng.integers(n - 1)
            if recip >= donor:
                recip += 1

        # maximum transferable mass from donor
        max_transfer = a[donor] * step_factor
        if max_transfer <= 0.0:
            continue
        delta = max_transfer * rng.random()
        if delta <= 0.0:
            continue

        # ----- apply in‑place update (exact) -----
        # store slice views for undo
        slice_i = slice(donor, donor + n)
        slice_j = slice(recip, recip + n)

        # remove mass from donor
        conv[slice_i] -= 2.0 * delta * a
        # add mass to recipient
        conv[slice_j] += 2.0 * delta * a
        # quadratic corrections
        conv[2 * donor] += delta * delta
        conv[2 * recip] += delta * delta
        conv[donor + recip] -= 2.0 * delta * delta

        new_peak = float(conv.max())

        if new_peak < cur_peak:                # monotone descent (no uphill)
            # commit the move
            a[donor] -= delta
            a[recip] += delta
            cur_peak = new_peak
            if new_peak < best_peak:
                best_peak = new_peak
                best_seq = a.copy()
        else:
            # undo everything using the same old a (still unchanged)
            conv[slice_i] += 2.0 * delta * a
            conv[slice_j] -= 2.0 * delta * a
            conv[2 * donor] -= delta * delta
            conv[2 * recip] -= delta * delta
            conv[donor + recip] += 2.0 * delta * delta
            # a unchanged

        if it % report_every == 0:
            print(f"[Anneal] it={it:7d} best obj≈{2.0 * n * best_peak:.6f} cur obj≈{2.0 * n * cur_peak:.6f}")

    print(f"[Anneal] finished after {it} iterations, best surrogate = {2.0 * n * best_peak:.6f}")
    return best_seq
