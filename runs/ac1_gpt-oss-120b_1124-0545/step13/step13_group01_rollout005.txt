def _proj_gradient_optimize(start_seq, n, deadline, max_iters=500000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a) # compute with our own

    step0 = 0.5
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # compute subgradient g
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = step0 / np.sqrt(it)
        a_tmp = a - eta * g
        a = _proj_simplex(a_tmp)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[ProjGrad] it={it} best={best_val:.6f}")
    return best_seq, best_val

"""
Propose a non‑negative step‑function (a list of floats) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm consists of several stages:

1.  Build a rich pool of ℓ₁‑normalised seeds (uniform, triangular,
    sine‑windows, Kaiser, Gaussian, power‑law, exponential‑decay and
    random Dirichlet draws).  The previous best sequence (global variable
    ``height_sequence_1``) is also added.

2.  Rank the seeds with a fast surrogate (2·n·max‑convolution, since all
    seeds are normalised to sum = 1) and keep the best few.

3.  Refine every top seed with a continuation L‑BFGS optimiser that
    smooths the “max’’ via a log‑sum‑exp surrogate.  A β‑schedule ending
    at 1000 provides a very close approximation to the true max.

4.  Perform a long projected‑gradient (mirror‑descent) refinement
    directly on the exact max‑convolution objective.  The sub‑gradient
    of the active shift is computed analytically, the step‑size follows
    ηₜ = c/√t and the iterate is projected onto the probability simplex
    (Euclidean projection) after each step.

5.  If time permits, up‑sample the current best vector (linear
    interpolation), repeat the L‑BFGS + projected‑gradient steps on the
    finer grid and finish with a short random local‑search.

All work is bounded by the supplied ``budget_s`` (≈ 1000 s).  The final
vector is stored in the global name ``height_sequence_1`` and also
returned as a plain list.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors, sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _exponential_decay(n: int, rate: float) -> np.ndarray:
    """a_i = exp(-rate * i)   (normalised)"""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_value(seq: np.ndarray) -> float:
    """Fast exact objective for a sequence with sum(seq)=1:
       2·n·max(conv(seq,seq))."""
    n = len(seq)
    max_b = np.max(_conv_fft(seq, seq))
    return 2.0 * n * max_b


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Optimise the smoothed max‑convolution objective.
    Returns a ℓ₁‑normalised vector (softmax of the logits)."""
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]

    # initialise logits = log‑softmax parameters
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # on the simplex
            conv = _conv_fft(a, a)                     # full autocorrelation
            shift = np.max(conv)                       # max value (certainty shift)
            # smooth‑max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                     # weights for the smooth max
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient w.r.t. logits (softmax Jacobian)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] time limit reached (beta={beta_val})")
            break
        except Exception as e:               # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Projected‑gradient (mirror‑descent) refinement on the exact objective
# ----------------------------------------------------------------------
def _proj_gradient_optimize(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 500_000,
                            step0: float = 0.5) -> tuple[np.ndarray, float]:
    """Projected (Euclidean) gradient descent on the true max‑convolution.
    Returns the best sequence seen and its objective value."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_value(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = step0 / np.sqrt(it)
        a_tmp = a - eta * g
        a = _proj_simplex(a_tmp)      # back to the probability simplex

        cur_val = _objective_value(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ProjGrad] it={it} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Upsampling utilities
# ----------------------------------------------------------------------
def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid (preserves sum)."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()

    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
    seeds = cleaned
    print(f"Generated {len(seeds)} initial seeds.")

    # ------------------------------------------------------------------
    # Quick surrogate ranking (2·n·max(conv) – exact because sum=1)
    # ------------------------------------------------------------------
    scored = [(v, _objective_value(v)) for v in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = min(8, len(scored))
    top_seeds = [v for v, _ in scored[:top_k]]

    print("Top seed surrogate objectives:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Global incumbent
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_value(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on the best seeds
    # ------------------------------------------------------------------
    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all L‑BFGS seeds.")
            break

        remaining = deadline - time.time()
        # allocate a reasonable iteration budget per β based on remaining time
        max_iter_beta = max(500, int(min(4000, remaining * 0.03 * 1e5)))
        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter_per_beta={max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=max_iter_beta)
        val_opt = _objective_value(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print("[Update] New global best after L‑BFGS")

    # ------------------------------------------------------------------
    # Stage 2 – projected‑gradient refinement (exact max‑convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_pg = int(min(600_000, remaining * 0.25 * 1e5))
        print(f"\n--- Projected‑gradient refinement (max_iters={max_iters_pg}) ---")
        seq_pg, val_pg = _proj_gradient_optimize(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_iters_pg,
                                                 step0=0.5)
        if val_pg < global_best_val:
            global_best_seq = seq_pg.copy()
            global_best_val = val_pg
            print(f"[ProjGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – up‑sampling to a finer grid (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 30.0:
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} ---")
            up_seq = _upsample_sequence(global_best_seq, new_n)

            # L‑BFGS on the higher resolution
            max_iter_up = int(min(3000, remaining * 0.07 * 1e5))
            up_opt = _lbfgs_smooth_optimize(up_seq,
                                            new_n,
                                            deadline,
                                            beta_schedule=beta_sched,
                                            max_iter_per_beta=max_iter_up)
            up_val = _objective_value(up_opt)
            print(f"[Upsample‑L‑BFGS] surrogate = {up_val:.6f}")

            if up_val < global_best_val:
                global_best_seq = up_opt.copy()
                n = new_n
                global_best_val = up_val
                print(f"[Update] new best after up‑sampling = {global_best_val:.6f}")

            # final projected‑gradient pass on finer grid
            if time.time() < deadline:
                remaining2 = deadline - time.time()
                max_iters_final = int(min(300_000, remaining2 * 0.15 * 1e5))
                print(f"\n--- Final projected‑gradient on n={n} (max_iters={max_iters_final}) ---")
                seq_final, val_final = _proj_gradient_optimize(global_best_seq,
                                                               n,
                                                               deadline,
                                                               max_iters=max_iters_final,
                                                               step0=0.5)
                if val_final < global_best_val:
                    global_best_seq = seq_final.copy()
                    global_best_val = val_final
                    print(f"[Final‑ProjGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional tiny random local search (if a sliver of time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        attempts = int(min(2000, remaining * 1000))   # very cheap
        if attempts > 0:
            print(f"\n--- Random local search (attempts={attempts}) ---")
            best_seq = global_best_seq.copy()
            best_val = global_best_val
            for _ in range(attempts):
                i, j = rng.choice(n, size=2, replace=False)
                eps = 0.01 * best_seq[i]          # move 1 % of mass from i to j
                if eps <= 0:
                    continue
                candidate = best_seq.copy()
                candidate[i] -= eps
                candidate[j] += eps
                # projection not needed – sum unchanged and all entries stay ≥0
                val = _objective_value(candidate)
                if val < best_val:
                    best_val = val
                    best_seq = candidate
            if best_val < global_best_val:
                global_best_seq = best_seq
                global_best_val = best_val
                print(f"[RandomSearch] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (exact, with clamping)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nFast surrogate ≈ {global_best_val:.6f}, "
          f"true evaluate_sequence = {true_val:.6f}")

    # store for the next round (if any)
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
