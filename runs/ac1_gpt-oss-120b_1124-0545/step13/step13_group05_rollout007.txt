Define function _exp_offset_shape_params(params, n):

    offset_frac, eps_scale = params
    offset = int(np.round(offset_frac * n))
    base_eps = 2.0 / n
    eps = max(0.0, eps_scale * base_eps)
    r = max(0.0, 1.0 - eps)
    tail_len = n - offset
    if tail_len <= 0:
        # fallback uniform
        return _uniform(n)
    i = np.arange(tail_len, dtype=np.float64)
    tail = np.power(r, i)
    tail_sum = tail.sum()
    if tail_sum == 0:
        tail = np.full(tail_len, 1.0 / tail_len)
    else:
        tail = tail / tail_sum
    a = np.concatenate([np.zeros(offset, dtype=np.float64), tail])
    a /= a.sum()
    return a

Define _exp_offset_plateau_shape_params(params, n):

    offset_frac, plateau_frac, eps_scale, plateau_factor = params
    offset = int(np.round(offset_frac * n))
    plateau_len = int(np.round(plateau_frac * n))
    total_len = n - offset - plateau_len
    if total_len <= 0:
        return _uniform(n)
    base_eps = 2.0 / n
    eps = max(0.0, eps_scale * base_eps)
    r = max(0.0, 1.0 - eps)
    # tail
    i = np.arange(total_len, dtype=np.float64)
    tail = np.power(r, i)
    tail_sum = tail.sum()
    if tail_sum == 0:
        tail = np.full(total_len, 1.0 / total_len)
    else:
        tail = tail / tail_sum

    # Combine zeros prefix, plateau, tail
    a = np.concatenate([np.zeros(offset, dtype=np.float64), plateau, tail])

"""
Hybrid parametric + gradient search for the step‑function minimisation problem.

The algorithm follows three stages:

0. **Rich seeding** – Analytic families (uniform, triangular, sine‑window,
   Gaussian, power, beta) plus a large random sweep over low‑dimensional
   parametric families (exponential with offset, exponential+plateau,
   power‑law, beta‑distribution).  Fast surrogate evaluation
   `2·n·max(conv(a,a))` (the true objective for ℓ₁‑normalised vectors) is
   used to rank seeds.

1. **Long Adam optimisation** – Starting from the best few seeds we run a
   log‑softmax Adam optimiser on a smoothed version of the max‑convolution.
   The learning‑rate and temperature schedule are tuned for the larger
   problem size (n≈18000).  The total number of Adam iterations is limited
   by the remaining wall‑clock time.

2. **Fine polishing** – A short sub‑gradient descent, an exponentiated‑gradient
   mirror descent and a quick simulated‑annealing walk are performed on the
   current global best.

The whole search respects a 1000 s wall‑clock budget; the best sequence is
returned as a plain list of floats and stored in the global variable
``height_sequence_1`` for possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised
    vector (∑seq = 1).  The true ``evaluate_sequence`` value equals this
    surrogate because the denominator is 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_offset_shape_params(n: int, offset_frac: float, eps_scale: float) -> np.ndarray:
    """
    Zero prefix of length floor(offset_frac·n) followed by a normalised
    exponential tail a_i ∝ r**i with r = 1 - eps, eps = eps_scale·(2/n).
    """
    offset = int(np.round(offset_frac * n))
    base_eps = 2.0 / n
    eps = max(0.0, eps_scale * base_eps)
    r = max(0.0, 1.0 - eps)
    tail_len = max(0, n - offset)
    if tail_len == 0:
        return _uniform(n)
    i = np.arange(tail_len, dtype=np.float64)
    tail = np.power(r, i)
    tail_sum = tail.sum()
    if tail_sum == 0:
        tail = np.full(tail_len, 1.0 / tail_len)
    else:
        tail = tail / tail_sum
    a = np.concatenate([np.zeros(offset, dtype=np.float64), tail])
    a /= a.sum()
    return a


def _exp_offset_plateau_shape_params(
    n: int,
    offset_frac: float,
    plateau_frac: float,
    eps_scale: float,
    plateau_factor: float,
) -> np.ndarray:
    """
    Zero prefix → constant plateau → exponential tail.
    The plateau consists of plateau_len = floor(plateau_frac·n) entries,
    each equal to plateau_factor (relative to the raw tail before normalisation).
    """
    offset = int(np.round(offset_frac * n))
    plateau_len = int(np.round(plateau_frac * n))
    tail_len = n - offset - plateau_len
    if tail_len <= 0:
        return _uniform(n)

    base_eps = 2.0 / n
    eps = max(0.0, eps_scale * base_eps)
    r = max(0.0, 1.0 - eps)

    i = np.arange(tail_len, dtype=np.float64)
    tail = np.power(r, i)

    # raw plateau (same height for all plateau entries)
    plateau = np.full(plateau_len, plateau_factor, dtype=np.float64)

    a = np.concatenate([np.zeros(offset, dtype=np.float64), plateau, tail])
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

# Optimisation primitives (log‑space Adam, sub‑gradient, EG, SA)

def _logspace_adam_optimize_long(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray | None, float]:
    """Adam on the log‑softmax parametrisation with a temperature‑annealed
    log‑sum‑exp smooth of the max.  Returns the best vector seen."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5               # start temperature
    beta_inc = 0.5
    beta_max = 2000.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * float(conv.max())
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[AdamLong] it={it:7d} best={cur_val:.6f}")

        # smoothed max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2 (trimmed later)
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        # Jacobian of log‑softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = 1.0 / math.sqrt(it)       # learning‑rate decays ~1/√it
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        cur_val = 2.0 * n * float(conv[idx])

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50_000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active lag
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror descent) on the exact max‑conv objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
    init_temp_factor: float = 0.06,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Aggressive simulated‑annealing polish on the surrogate."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9999, min_temp)

        if rng.random() < 0.7:
            # reshuffle a random subset (size 2 … 12)
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            # simple pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _resample_to_n(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a previous solution to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        a = old_seq.copy()
    else:
        x_old = np.arange(old_n, dtype=np.float64)
        x_new = np.linspace(0, old_n - 1, new_n)
        a = np.interp(x_new, x_old, old_seq)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(new_n)
    return a / s

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a safety margin for final evaluation
    overall_deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – a larger n improves the asymptotic constant.
    # ------------------------------------------------------------------
    n = 18000  # chosen to stay within the 1000 s budget while getting close to the limit

    # ------------------------------------------------------------------
    # Phase 0 – generate a diverse pool of ℓ₁‑normalised candidate vectors
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds: list[np.ndarray] = []

    # deterministic classic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_power(n, p=1.0))
    seeds.append(_power(n, p=2.0))
    seeds.append(_beta_shape(n, alpha=2.0, beta=2.0))

    # previous best (if available) – resample to current n
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            seeds.append(_resample_to_n(prev_arr, n))
        except Exception:
            pass

    # ----- parameteric families via Differential Evolution -----
    # (few dozen cheap evaluations each – parallel‑free, deterministic)
    try:
        from scipy.optimize import differential_evolution
    except Exception:  # pragma: no cover
        differential_evolution = None

    # Helper wrappers for DE – they return the surrogate value
    def _de_obj_exp_offset(params):
        seq = _exp_offset_shape_params(n, offset_frac=params[0], eps_scale=params[1])
        return _objective_surrogate(seq)

    def _de_obj_exp_offset_plateau(params):
        seq = _exp_offset_plateau_shape_params(
            n,
            offset_frac=params[0],
            plateau_frac=params[1],
            eps_scale=params[2],
            plateau_factor=params[3],
        )
        return _objective_surrogate(seq)

    def _de_obj_power(params):
        p = max(0.05, params[0])  # enforce positivity
        seq = _power(n, p)
        return _objective_surrogate(seq)

    def _de_obj_beta(params):
        alpha = max(0.05, params[0])
        beta = max(0.05, params[1])
        seq = _beta_shape(n, alpha, beta)
        return _objective_surrogate(seq)

    if differential_evolution is not None:
        # exponential offset (2 parameters)
        bounds_exp = [(0.0, 0.5), (0.3, 5.0)]
        res = differential_evolution(_de_obj_exp_offset, bounds_exp,
                                     maxiter=30, popsize=12, seed=seed, polish=False)
        seeds.append(_exp_offset_shape_params(n, res.x[0], res.x[1]))
        print(f"[DE] exp‑offset best ≈ {_de_obj_exp_offset(res.x):.6f}")

        # exponential offset + plateau (4 parameters)
        bounds_plateau = [(0.0, 0.5),   # offset_frac
                          (0.0, 0.5),   # plateau_frac
                          (0.3, 5.0),   # eps_scale
                          (0.0, 5.0)]   # plateau_factor
        res = differential_evolution(_de_obj_exp_offset_plateau, bounds_plateau,
                                     maxiter=30, popsize=12, seed=seed + 1, polish=False)
        seeds.append(_exp_offset_plateau_shape_params(n, *res.x))
        print(f"[DE] exp‑plateau best ≈ {_de_obj_exp_offset_plateau(res.x):.6f}")

        # power‑law exponent (1 parameter)
        bounds_power = [(0.05, 5.0)]
        res = differential_evolution(_de_obj_power, bounds_power,
                                     maxiter=30, popsize=12, seed=seed + 2, polish=False)
        seeds.append(_power(n, max(0.05, res.x[0])))
        print(f"[DE] power‑law best ≈ {_de_obj_power(res.x):.6f}")

        # beta‑distribution (2 parameters)
        bounds_beta = [(0.05, 5.0), (0.05, 5.0)]
        res = differential_evolution(_de_obj_beta, bounds_beta,
                                     maxiter=30, popsize=12, seed=seed + 3, polish=False)
        seeds.append(_beta_shape(n, max(0.05, res.x[0]), max(0.05, res.x[1])))
        print(f"[DE] beta best ≈ {_de_obj_beta(res.x):.6f}")

    # ----- random draws from the same families (adds diversity) -----
    for _ in range(80):
        # random exponential offset
        off = rng.random() * 0.5
        eps_scale = rng.random() * 5.0 + 0.3
        seeds.append(_exp_offset_shape_params(n, off, eps_scale))

    for _ in range(60):
        # random exponential offset + plateau
        off = rng.random() * 0.5
        plat = rng.random() * 0.5
        eps_scale = rng.random() * 5.0 + 0.3
        plat_factor = rng.random() * 5.0
        seeds.append(
            _exp_offset_plateau_shape_params(n, off, plat, eps_scale, plat_factor)
        )

    for _ in range(60):
        # random power law
        p = rng.random() * 4.5 + 0.05
        seeds.append(_power(n, p))

    for _ in range(60):
        alpha = rng.random() * 4.9 + 0.05
        beta = rng.random() * 4.9 + 0.05
        seeds.append(_beta_shape(n, alpha, beta))

    for _ in range(120):
        # Dirichlet draws with varying concentration
        conc = rng.choice([0.05, 0.2, 0.5, 1.0, 2.0])
        a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
        seeds.append(a)

    # ------------------------------------------------------------------
    # Cheap surrogate ranking
    # ------------------------------------------------------------------
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 30
    best_seeds = [vec for vec, _ in evals[:top_k]]

    print(f"\nTop {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation (distributed over the top seeds)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 60.0:
        # allocate roughly half of the remaining time to Adam
        max_total_iters = int(min(300_000, remaining * 1800))
        it_per_seed = max_total_iters // top_k
        it_per_seed = max(1000, it_per_seed)  # ensure a few thousand iterations at least
        print(f"\n--- Adam optimisation (total it≈{max_total_iters}, it/seed≈{it_per_seed}) ---")
        for idx, seed_vec in enumerate(best_seeds):
            seq_adam, val_adam = _logspace_adam_optimize_long(
                seed_vec, n, overall_deadline, max_iters=it_per_seed
            )
            if seq_adam is not None and val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[Adam‑seed{idx+1}] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient projection
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 30.0:
        max_iters_sub = int(min(800_000, remaining * 2500))
        print(f"\n--- Sub‑gradient projection (max_iters={max_iters_sub}) ---")
        seq_sub, val_sub = _subgradient_proj(
            global_best_seq, n, overall_deadline, max_iters=max_iters_sub
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[SubProj] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 20.0:
        max_iters_exp = int(min(400_000, remaining * 1500))
        print(f"\n--- Exponentiated‑gradient refinement (max_iters={max_iters_exp}) ---")
        seq_exp, val_exp = _exp_gradient_optimize(
            global_best_seq, n, overall_deadline, max_iters=max_iters_exp
        )
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] new global best = {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated‑annealing polishing
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 10.0:
        max_iters_sa = int(min(800_000, remaining * 2000))
        print(f"\n--- Simulated annealing polishing (max_iters={max_iters_sa}) ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, overall_deadline, max_iters=max_iters_sa
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
