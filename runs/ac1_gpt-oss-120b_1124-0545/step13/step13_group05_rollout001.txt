"""
Hybrid analytic seeding + multi‑stage L‑BFGS + sub‑gradient + SA refinement.

The algorithm works as follows:

1. Choose a modest problem size `n` (8000) and generate a diverse pool of
   ℓ₁‑normalised candidate step‑functions using simple analytic families
   (uniform, triangular, sine‑window, exponential, offset‑exponential,
   random Dirichlet and exponential draws).  The pool is intentionally
   small (≈30 vectors) so that we spend almost all of the 1000 s budget
   on optimisation.

2. Evaluate the cheap surrogate `2·n·max(conv(a,a))` for each seed,
   keep the best `top_k` seeds (default 5).

3. For every top seed run a short L‑BFGS optimisation on the
   log‑softmax parametrisation `a = softmax(v)`.  The objective is a
   smoothed approximation of the true max‑convolution:

        smax_β(conv) = (1/β)·log Σ_i exp(β·conv_i)

   A schedule of increasing temperatures `[10, 30, 100, 300, 1000]`
   is used – the optimiser first sees a very smooth landscape and then
   tightens to the true max.  Only a few hundred L‑BFGS iterations are
   performed per temperature, which is enough to drive the surrogate
   down while staying well within the time budget.

4. The best result from the L‑BFGS stage is polished with a Euclidean
   projection sub‑gradient descent (the same routine used in the
   reference code) for a limited number of iterations.

5. Finally a lightweight simulated‑annealing sweep makes random
   local mass‑reallocations; it is fast and often finds a small
   improvement that the gradient methods miss.

All phases are bounded by the global deadline (the function stops as
soon as the remaining time drops below a small safety margin).  The

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT (O(n log n)).
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector.
    This equals the value returned by ``evaluate_sequence`` when the
    sequence sums to exactly 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_shape(n: int, r: float) -> np.ndarray:
    """Monotone exponential a_i ∝ r**i, ℓ₁‑normalised."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_offset_shape(n: int, offset: int, r: float) -> np.ndarray:
    """Zero prefix of length `offset` followed by an exponential tail."""
    if offset >= n:
        return _uniform(n)
    tail_len = n - offset
    if r <= 0.0:
        tail = np.full(tail_len, 1.0 / tail_len, dtype=np.float64)
    else:
        i = np.arange(tail_len, dtype=np.float64)
        tail = np.power(r, i)
        tail /= tail.sum()
    a = np.concatenate([np.zeros(offset, dtype=np.float64), tail])
    a /= a.sum()
    return a

    """
    try:
        import scipy.optimize  # pylint: disable=import-outside-toplevel
    except Exception as exc:  # pragma: no cover
        raise RuntimeError("SciPy is required for L‑BFGS optimisation") from exc

    # initialise logits
    v = np.log(start_seq + 1e-30)

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        # if we are already out of time, break
        if time.time() > deadline:
            break

        # ---------- objective & gradient for a given β ----------
        def fun(v_flat: np.ndarray) -> Tuple[float, np.ndarray]:
            a = _softmax(v_flat)
            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            shift = np.max(conv)
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            # avoid division by zero (practically never happens)
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
                w_sum = 1.0
            w = w / w_sum
            smax = shift + np.log(w_sum) / beta  # smooth approximation of max(conv)

            # gradient of smooth max w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1 : n - 1 + n]    # length n

            # chain rule through softmax (log‑softmax Jacobian)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            return smax, grad_v.astype(np.float64)

        # ---------- run L‑BFGS ----------
        res = scipy.optimize.minimize(
            fun=fun,
            x0=v,
            jac=True,
            method="L-BFGS-B",
            options={"maxiter": iters_per_beta, "disp": False},
        )
        v = res.x  # use the final point as start for next β

        # evaluate surrogate after this temperature
        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # debug output
        print(
            f"[L‑BFGS β={beta:.0f}] it≈{iters_per_beta}  best={cur_val:.6f}"
        )
    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Euclidean‑projection sub‑gradient descent on the true max‑convolution
    objective.  The routine stops early if the global deadline is reached.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    init_temp_factor: float = 0.08,
    min_temp: float = 1e-12,
) -> Tuple[np.ndarray, float]:
    """A light‑weight annealing phase – random mass reshuffles on
    small subsets of coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        if rng.random() < 0.6:  # reshuffle a random subset (size 3‑9)
            k = rng.integers(3, 10)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.ones(k)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:  # simple pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.04)  # up to 4 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 25_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    top_k: int = 30,
) -> List[np.ndarray]:
    """
    Produce a modest but diverse pool of ℓ₁‑normalised vectors.
    Returns the `top_k` seeds according to the cheap surrogate.
    """
    seeds: List[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0):
        seeds.append(_sine_window(n, gamma=g))

    # exponential family (no offset)
    base_eps = 2.0 / n
    for fac in np.linspace(0.5, 2.5, 12):
        eps = fac * base_eps
        r = max(0.0, 1.0 - eps)
        seeds.append(_exponential_shape(n, r))

    # offset‑exponential families (two offsets only)
    for off_frac in (0.02, 0.07):
        offset = int(off_frac * n)
        L = n - offset
        if L <= 0:
            continue
        base_eps_off = 2.0 / L
        for fac in np.linspace(0.5, 2.0, 8):
            eps = fac * base_eps_off
            r = max(0.0, 1.0 - eps)
            seeds.append(_exponential_offset_shape(n, offset, r))

    # random Dirichlet draws with different concentrations
    for conc in (0.05, 0.2, 0.7, 1.5):
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # exponential random draws
    for _ in range(3):
        a = rng.exponential(scale=1.0, size=n)
        a /= a.sum()
        seeds.append(a)

    # previous solution, if any
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # evaluate surrogate and keep the best `top_k`
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_seeds = [vec for vec, _ in evals[:top_k]]
    print(f"Generated {len(seeds)} raw seeds, selecting the best {top_k}.")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  seed #{i+1:2d}: surrogate = {val:.6f}")
    return top_seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final exact evaluation
    overall_deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – 8000 yields a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 8000

    # try to reuse a previously found sequence
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, list):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # Phase 0 – generate a compact set of analytic/draw seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_seeds(n, rng, prev_arr, top_k=30)

    # initialise global best with the best seed
    global_best_seq = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS refinement on the top few seeds
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 30.0:
        # allocate roughly half of the remaining budget to L‑BFGS
        beta_schedule = [10, 30, 100, 300, 1000]
        iters_per_beta = max(50, int(min(300, remaining * 0.9 / (len(seeds) * len(beta_schedule)))))
        print(f"\n--- L‑BFGS optimisation (iters/β≈{iters_per_beta}) ---")
        for idx, seed_vec in enumerate(seeds[:5]):  # restrict to the 5 most promising seeds
            seq_opt, val_opt = _lbfgs_optimize(
                seed_vec, n, overall_deadline, beta_schedule, iters_per_beta=iters_per_beta
            )
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[L‑BFGS‑seed{idx+1}] new global best = {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient projection polishing
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 10.0:
        max_iters_sub = int(min(300_000, remaining * 15_000))
        print(f"\n--- Sub‑gradient projection (max_iters={max_iters_sub}) ---")
        seq_sub, val_sub = _subgradient_proj(
            global_best_seq, n, overall_deadline, max_iters=max_iters_sub
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[SubProj] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 5.0:
        max_iters_sa = int(min(200_000, remaining * 30_000))
        print(f"\n--- Simulated annealing polishing (max_iters={max_iters_sa}) ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, overall_deadline, max_iters=max_iters_sa
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation (scale‑invariant, sum≈1)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
