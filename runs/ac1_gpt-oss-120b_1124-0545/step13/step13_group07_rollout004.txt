def propose_candidate(seed = 123, budget_s = 1000):

    evaluated.sort(key=lambda x: x[1])
    # select top_k seeds, e.g., 10
    top_k = 10
    best_seeds = [vec for vec,val in evaluated[:top_k]]
    global_best_seq = best_seeds[0].copy()
    global_best_val = _surrogate_obj(global_best_seq)

    # Stage Adam: iterate over best_seeds, allocate some time
    for i, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        # compute max_iters derived from remaining time: allocate maybe (deadline - now)/ (remaining seeds + 2)
        remaining = deadline - time.time()
        # We allocate approx 0.5 sec per 1000 iterations? Hard.
        max_iters = min(200_000, int(remaining * 200))
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam

    # Stage mirror descent: allocate time
    if time.time() < deadline:
        remaining_mirror = deadline - time.time()
        # allocate maybe 60% of remaining time
        time_for_mirror = remaining_mirror * 0.6
        # compute approximate iters: one iteration takes conv O(n log n) ~ 0.0001? Let's assume each iteration ~0.0005 sec.
        # We cap at 2 million or time_for_mirror/0.0005.
        max_iters_mirror = int(time_for_mirror / 0.0005)
        max_iters_mirror = min(max_iters_mirror, 2_000_000)
        seq_mirror, val_mirror = _mirror_subgradient_optimize(global_best_seq, n, deadline, max_iters=max_iters_mirror)
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror

    # Stage LP polishing: allocate time
    if time.time() < deadline:
        # allocate at most maybe 10 iterations or until time runs out
        max_lp_iters = 30
        seq_lp, val_lp = _lp_refine(global_best_seq, deadline, max_iters=max_lp_iters, tau=0.12)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp

    # Stage SA: remaining time
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing_optimize(global_best_seq, n, deadline, max_iters=5_000_000)

   low‑frequency cosine curves) plus a few *extreme* β‑shapes and
   edge‑concentrated shapes.  All seeds are evaluated with a cheap surrogate
   ``obj = 2·n·max(conv(a,a))`` (with ``∑a = 1``) and the best ``top_k`` are
   kept.

2. **Adam on a smoothed max** – each top seed is refined by Adam on the
   log‑softmax parametrisation of the simplex, using a temperature‑controlled
   log‑sum‑exp surrogate of the max.  The number of iterations is capped by the
   remaining time budget.

3. **Exact mirror‑descent** – a sub‑gradient / exponentiated‑gradient
   optimisation on the *exact* max of the convolution is performed.  The
   iteration limit is chosen dynamically so that a fraction of the total time
   is reserved for the later polishing stages.

4. **Iterated LP polishing** – repeatedly solve the linear program

        maximise   Σ g_i
        subject to  conv(f,g) ≤ M   (M = max conv(f,f))
                    g_i ≥ 0

   and mix the solution back into ``f`` with a small weight ``τ``.
   This strictly decreases the official objective after normalisation.
   Up to 30 LP iterations are attempted (each with a modest 5 s time limit).

5. **Simulated‑annealing polishing** – a lightweight SA that swaps,
   redistributes or transfers mass between random positions is run for the
   remaining time.  It can escape shallow local minima left by the previous
   deterministic steps.

All stages respect a single 1000 s wall‑clock budget.  The best vector found
is stored globally as ``height_sequence_1`` and also returned as a plain list.

The implementation is self‑contained, uses only ``numpy`` and ``scipy`` and
contains no closures or file I/O.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate_obj(seq: np.ndarray) -> float:
    """
    Fast surrogate objective (∑seq = 1):
        2·n·max(conv(seq,seq))
    The factor 2·n is the only part that survives the official evaluation
    once the vector is normalised.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random 3‑segment piece‑wise linear shape."""
    breaks = np.sort(rng.choice(n, size=3, replace=False))
    heights = rng.random(3) + 0.01
    x = np.arange(n, dtype=np.float64)
    a = np.interp(x, breaks, heights)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few randomly positioned Gaussians."""
    a = np.zeros(n, dtype=np.float64)
    num = rng.integers(2, 6)
    for _ in range(num):
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.02 * n, 0.15 * n)
        x = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, nf: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth curve built from a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, nf + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _edge_blocks(n: int) -> np.ndarray:
    """Uniform mass on the two outer blocks (≈20 % each)."""
    block = max(1, n // 5)
    a = np.zeros(n, dtype=np.float64)
    a[:block] = 1.0
    a[-block:] = 1.0
    return a / a.sum()


def _two_gaussian_edges(n: int, rng: np.random.Generator) -> np.ndarray:
    """Two narrow Gaussians placed at the two ends."""
    sigma = 0.02 * n
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-((x) ** 2) / (2 * sigma ** 2))
    a += np.exp(-((x - (n - 1)) ** 2) / (2 * sigma ** 2))
    return a / a.sum()


def _beta_extremes(n: int) -> list[np.ndarray]:
    """A collection of very U‑shaped beta distributions (α,β ≤ 0.5)."""
    seeds = []
    alphas = np.linspace(0.05, 0.5, 10)
    betas = np.linspace(0.05, 0.5, 10)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))
    return seeds


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray | None, float]:
    """Adam on log‑softmax parameters with a log‑sum‑exp surrogate of max."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.9
    beta = 0.5
    beta_inc = 0.5
    beta_max = 3000.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # smooth max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                max_iters: int = 4_000_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror) sub‑gradient descent on the exact max."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    lr0 = 0.25
    power = 0.28

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _lp_refine(seq: np.ndarray, deadline: float,
               max_iters: int = 12, tau: float = 0.15,
               lp_time_limit: int = 5) -> tuple[np.ndarray, float]:
    """
    Perform several LP refinements.

    Each iteration solves:
        maximise   Σ g_i
        subject to conv(seq, g) ≤ M   (where M = max conv(seq, seq))
                  g_i ≥ 0
    The solution g is mixed back into the current sequence with weight τ
    and renormalised.
    """
    f = seq.copy()
    n = len(f)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        M = _conv_fft(f, f).max()

        # Build convolution matrix A for current f (sparse, exactly n² non‑zeros)
        rows = np.empty(n * n, dtype=np.int32)
        cols = np.empty(n * n, dtype=np.int32)
        data = np.empty(n * n, dtype=np.float64)

        idx = 0
        for k in range(n):
            start = k
            end = k + n
            rows[idx:idx + n] = np.arange(start, end, dtype=np.int32)
            cols[idx:idx + n] = k
            data[idx:idx + n] = f
            idx += n

        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': lp_time_limit})
        if not res.success:
            print(f"[LP] iteration {it} failed – stopping LP stage.")
            break

        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g <= sum_f + 1e-12:
            print(f"[LP] iteration {it} – no L1 gain.")
            break

        f_new = (1.0 - tau) * f + tau * g
        f_new = f_new / f_new.sum()
        f = f_new

        cur_obj = _surrogate_obj(f)
        print(f"[LP] iteration {it} – surrogate = {cur_obj:.6f}")

    return f, _surrogate_obj(f)


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                 max_iters: int = 2_500_000,
                                 init_temp_factor: float = 0.06,
                                 min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """Lightweight SA with three move types."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate_obj(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # gentle exponential cooling
        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:
            # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            # redistribute mass on a small random subset
            k = rng.integers(2, 14)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total == 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            # pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total == 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _surrogate_obj(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))

    # parameter sweeps
    for sigma in np.linspace(0.05, 0.35, 8):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 4.5, 10):
        seeds.append(_power_law(n, p))

    alphas = np.linspace(0.2, 3.0, 9)
    betas = np.linspace(0.2, 3.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for lam in np.linspace(0.0005, 0.015, 10):
        seeds.append(_exponential_decay(n, lam))

    # random Dirichlet draws with a broad range of concentrations
    concs = np.logspace(-4, 0.5, 12)   # 1e‑4 … ~3.2
    for conc in concs:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixtures of random Gaussians
    for _ in range(5):
        seeds.append(_gaussian_mixture(n, rng))

    # piece‑wise‑linear random shapes
    for _ in range(5):
        seeds.append(_piecewise_linear(n, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 5, 7):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mix uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4, 6):
            curve = _lowfreq_curve(n, nf, rng)
            mixed = (1.0 - mix_alpha) * _uniform(n) + mix_alpha * curve
            mixed = mixed / mixed.sum()
            seeds.append(mixed)

    # **extra families for extra U‑shape**
    seeds.append(_edge_blocks(n))
    seeds.append(_two_gaussian_edges(n, rng))
    seeds.extend(_beta_extremes(n))

    # previous solution, if any (for continuity across runs)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0.0:
            seeds.append(avg / avg.sum())

    # keep only unique vectors (by hashing their rounded representation)
    uniq = {}
    for v in seeds:
        key = tuple(np.round(v, 12))
        uniq[key] = v
    return list(uniq.values())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective.  The best sequence is stored globally as
    ``height_sequence_1`` and also returned as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # problem size – 4000 works well with the LP stage (memory‑friendly)
    n = 4000

    # ------------------------------------------------------------------
    # Stage 0 – rich seed pool
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size != n:
                if previous.size > n:
                    previous = previous[:n]
                else:
                    previous = np.pad(previous, (0, n - previous.size), constant_values=0.0)
        except Exception:
            previous = None
    else:
        previous = None

    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _surrogate_obj(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 15                     # a modest pool for the subsequent stages
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _surrogate_obj(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation on smooth surrogate
    # ------------------------------------------------------------------
    print("\n--- Adam optimisation on top seeds ---")
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        # allocate a few thousand iterations per seed, respecting remaining budget
        remaining = deadline - time.time()
        max_iters = min(150_000, int(remaining * 80))
        if max_iters < 1:
            break
        print(f"\n[Adam] seed #{idx}/{len(best_seeds)} – max_iters={max_iters}")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Mirror‑descent (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allow up to ~60 % of the remaining time for mirror descent
        target_time = remaining * 0.6
        # assume ~0.0005 s per iteration (empirical)
        max_iters = int(target_time / 0.0005)
        max_iters = max(100_000, min(max_iters, 1_500_000))
        print(f"\n--- Mirror‑descent (max_iters={max_iters}) ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=max_iters
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Iterated LP polishing (more iterations than before)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # give the LP stage up to 30 iterations (or until time runs out)
        print("\n--- Linear‑programming polishing (up to 30 iters) ---")
        seq_lp, val_lp = _lp_refine(global_best_seq, deadline,
                                    max_iters=30, tau=0.12, lp_time_limit=5)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[overall‑LP] new global best = {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing polishing (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa_iters = int(remaining * 500)   # rough knob: 500 iters per second
        max_sa_iters = max(200_000, min(max_sa_iters, 5_000_000))
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline, max_iters=max_sa_iters
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (for information only)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
