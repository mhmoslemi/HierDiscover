        conv = _conv_fft(a, a)   # length 2n-1
        cur_sur = 2.0 * n * conv.max()
        # tracking best
        if cur_sur < best_sur:

            best_seq = a.copy()
            if it % 500 == 0:
                elapsed = time.time() - start
                print(f"[Iter] it={it:7d}, new best surrogate={cur_sur:.6f}, elapsed={elapsed:.1f}s")
        # compute softmax weights
        maxc = conv.max()
        # increase temperature slowly
        beta_temp = min(beta_temp * 1.005, 2000.0)
        # use shifted exponent to avoid overflow
        w = np.exp(beta_temp * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # gradient

        conv2 = _conv_fft(a, revw)   # length 3n-2
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * conv2[idx]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        # bias-corrected
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr0 / math.sqrt(it) * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project onto simplex (simple)
        a = np.maximum(a, 0.0)
        sum_a = a.sum()
        if sum_a > 0:
            a = a / sum_a
        else:
            a = np.full(n, 1.0 / n)

            print(f"[Iter] it={it:7d}, cur surrogate={cur_sur:.6f}, elapsed={elapsed:.1f}s")

"""
Optimization of a non‑negative step function for the objective

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three stages:

1. **Initialisation** – reuse the previously found best sequence if it exists,
   otherwise start from a random sparse uniform support (the same idea as the
   reference code).

2. **Projected gradient descent on a smooth surrogate** – we minimise a
   temperature‑scaled soft‑max of the convolution entries

        F(a) = (1/τ)·log Σ_k exp(τ·(conv(a,a))_k)

   whose gradient can be written in closed form and is evaluated with two
   FFT‑based convolutions per iteration.  After each step the vector is
   projected back onto the probability simplex (non‑negative and ℓ₁‑norm = 1)
   by a cheap renormalisation.

   The temperature τ is increased slowly so that the optimisation starts
   with a smooth landscape and gradually concentrates on the true max‑norm.
   An Adam‑style momentum scheme with a diminishing learning‑rate
   (∝ 1/√t) is used to accelerate convergence.

3. **Final evaluation** – the best candidate found is finally scored with the
   official ``evaluate_sequence`` function and stored globally for reuse.

The routine respects the supplied time budget (default 1000 s) and prints
periodic progress information.  The returned list contains the heights of the
step function (already ℓ₁‑normalised, all entries ≥ 0).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _simplex_normalize(v: np.ndarray) -> np.ndarray:
    """Fast (non‑optimal) projection onto the simplex: clip negatives and renormalise."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        return v / s
    else:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])


def _surrogate(a: np.ndarray, n: int) -> float:
    """Fast surrogate 2·n·max(conv(a,a)) for an ℓ₁‑normalised vector."""
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))

# ----------------------------------------------------------------------
