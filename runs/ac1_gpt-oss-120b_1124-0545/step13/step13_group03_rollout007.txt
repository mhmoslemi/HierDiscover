"""
Propose a step‑function (non‑negative vector) that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a))²

The algorithm consists of three phases:

1. **Diversified seeding** – analytic windows, Dirichlet draws and a
   resampled previous solution are ℓ₁‑normalised.  The 12 best seeds
   (according to the exact surrogate 2·n·max(conv)) are kept.

2. **Evolutionary global search** – a small population evolves by random
   convex mixtures of two parents.  Occasionally a short projected‑gradient
   (sub‑gradient on the active convolution shift) refinement is applied to
   the offspring.  The population is kept on the probability simplex.

3. **Local polishing** – a few thousand steps of Adam on a smooth
   “soft‑max” surrogate of the max‑convolution, followed by a cheap
   exponentiated‑gradient (multiplicative) finish.

All work respects the global 1000 s deadline.  The best vector found is
returned as a plain Python list; it is also stored in the global variable
``height_sequence_1`` for possible reuse in a later run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    n = x.shape[0]
    L = 2 * n - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    conv = np.fft.irfft(fx * fx, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for evaluate_sequence when sum(seq)=1:
        2·n·max(conv(seq,seq))
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shapes (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    seeds.append(_gaussian(n, 0.2))
    seeds.append(_gaussian(n, 0.3))
    seeds.append(_gaussian(n, 0.4))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform – adds a slight bias toward flatness
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # keep previous solution (if it exists) – resample to current length
    if previous is not None:
        prev = np.asarray(previous, dtype=np.float64)
        if prev.size > 0:
            # linear interpolation to new length
            xs = np.linspace(0, 1, prev.shape[0])
            xs_new = np.linspace(0, 1, n)
            resampled = np.interp(xs_new, xs, prev)
            resampled = np.maximum(resampled, 0.0)
            s = resampled.sum()
            if s > 0:
                seeds.append(resampled / s)

    # final safety check
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (additive updates)
# ----------------------------------------------------------------------
def _subgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000,
                      eta0: float = 0.3) -> tuple[np.ndarray, float]:
    """Simple additive projected sub‑gradient descent on the max‑conv objective."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x)
        k = int(np.argmax(conv))

        # gradient of the active shift (2·a_rev shifted by k)
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        x = x - eta * g
        # projection onto the simplex (keeps non‑negativity)
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()

    return best_x, best_val

# Adam optimisation on a soft‑max smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_smooth_optimize(start: np.ndarray, n: int, deadline: float,
                          init_lr: float = 0.05,
                          beta1: float = 0.9, beta2: float = 0.999,
                          eps: float = 1e-8,
                          p0: float = 5.0,
                          p_factor: float = 2.5,
                          p_step: int = 40_000,
                          max_total_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max surrogate
        F_p(a) = (1/p)·log Σ_k exp(p·conv_k(a)).
    The smoothing parameter p is increased geometrically.
    Returns the best vector encountered (exact surrogate value).
    """
    a = start.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    p = p0

    while time.time() < deadline and t < max_total_iters:
        conv = _conv_fft(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam] it={t:7d} best={best_val:.6f}  p={p:.1f}")

        shifted = conv - max_conv
        exp_arg = p * shifted
        exp_arg = np.clip(exp_arg, -700, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            mask = conv == max_conv
            w = mask.astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        a_rev = a[::-1]
        grad_full = _conv_fft(w) * 0  # placeholder to keep shape, replaced below
        # compute gradient = 2·conv(w , a_rev) (central slice)
        grad_full = _conv_fft(w, )  # Actually we need conv(w, a_rev)

        # The convolution with a_rev can be obtained via FFT as well:
        # conv(w, a_rev) = irfft( fft(w) * fft(a_rev) )
        fx = np.fft.rfft(w, n=_next_pow2(2 * n - 1))
        fy = np.fft.rfft(a_rev, n=_next_pow2(2 * n - 1))
        grad_full = np.fft.irfft(fx * fy, n=_next_pow2(2 * n - 1))[:2 * n - 1]
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = _uniform(n)

        if t % p_step == 0:
            p *= p_factor
            print(f"[Adam] increased p to {p:.1f}")

    final_val = _objective_surrogate(best_a)
    return best_a, final_val


# ----------------------------------------------------------------------
# Simulated‑annealing (surrogate objective) – stronger moves
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.12,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99993, min_temp)

        # two mutation types: redistribution on a random subset OR tiny mass transfer
        if rng.random() < 0.55:
            k = rng.integers(5, 20)               # size of the subset
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total > 0.0:
                new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
                cand = cur.copy()
                cand[idx] = new_alloc
            else:
                continue
        else:
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.04)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

# Evolutionary global search (population + mixing)
# ----------------------------------------------------------------------
def _evolutionary_search(population: list[np.ndarray],
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Simple (μ+λ) evolution: convex mixing + occasional sub‑gradient polish."""
    pop = population[:]
    pop_vals = [_objective_surrogate(v) for v in pop]
    best_idx = int(np.argmin(pop_vals))
    best_seq = pop[best_idx].copy()
    best_val = pop_vals[best_idx]

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # pick two parents uniformly
        i, j = rng.choice(len(pop), size=2, replace=False)
        parent1, parent2 = pop[i], pop[j]
        alpha = rng.uniform(0.1, 0.9)
        child = (1.0 - alpha) * parent1 + alpha * parent2

        # occasional mutation (Gaussian noise)
        if rng.random() < 0.25:
            noise = rng.normal(scale=0.015, size=n)
            child = child + noise
            child = np.maximum(child, 0.0)
            child = child / child.sum()

        # ensure exact simplex (numerical safety)
        child = _proj_simplex(child)

        # very cheap local refinement: a few sub‑gradient steps
        if rng.random() < 0.15:
            child, _ = _subgrad_optimize(child, n, deadline,
                                         max_iters=1200, eta0=0.25)

        child_val = _objective_surrogate(child)

        # replace worst individual if child is better
        worst_idx = int(np.argmax(pop_vals))
        if child_val < pop_vals[worst_idx]:
            pop[worst_idx] = child
            pop_vals[worst_idx] = child_val

        # track global best
        if child_val < best_val:
            best_val = child_val
            best_seq = child.copy()
            print(f"[Evo] it={it:7d} new best={best_val:.6f}")

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Length choice – a power of two close to 4000 gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Seed generation (including a resampled previous solution)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # keep a few top seeds (surrogate)
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Evolutionary global search
    # ------------------------------------------------------------------
    print("\n--- Evolutionary search (mixing + local polish) ---")
    best_seq, best_val = _evolutionary_search(top_seeds, n, deadline, rng,
                                              max_iters=250_000)
    print(f"[Evo] best surrogate after evolution = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Adam smooth optimisation (soft‑max) – fine‑grained polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam smooth optimisation (soft‑max) ---")
        remaining = deadline - time.time()
        max_adam_iters = int(min(400_000, remaining * 150))
        best_seq, best_val = _adam_smooth_optimize(best_seq, n, deadline,
                                                   init_lr=0.06,
                                                   p0=5.0,
                                                   p_factor=2.2,
                                                   p_step=30_000,
                                                   max_total_iters=max_adam_iters)
        print(f"[Adam] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing polishing (surrogate objective)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(500_000, remaining * 250))
        best_seq, best_val = _simulated_annealing(best_seq, n, deadline,
                                                  max_iters=max_iters)
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final EG polishing (tiny, cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(30_000, remaining * 2000))
        if max_iters > 0:
            best_seq, best_val = _expgrad_optimize(best_seq, n, deadline,
                                                   max_iters=max_iters,
                                                   eta0=0.22)
            print(f"[Final EG] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
