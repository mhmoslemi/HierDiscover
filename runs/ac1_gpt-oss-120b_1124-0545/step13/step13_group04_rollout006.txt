"""
Propose a coefficient sequence that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

using a smooth‑max gradient descent on the self‑convolution.  The algorithm
runs an adaptive ADAM optimizer directly on the probability simplex
(projecting after each step).  It starts from the previously found
solution (if any) or from the uniform distribution, then iteratively
updates the vector while the time budget allows.  The final sequence is
returned as a list of floats.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Low‑level utilities
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all entries equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _surrogate(a: np.ndarray, n: int) -> float:
    """Fast surrogate 2·n·max(conv(a,a)) for an ℓ₁‑normalised vector."""
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    `evaluate_sequence` objective within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Keep a small safety margin for the final evaluation call.
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------
    # Problem size (the previous solution used n = 8000)
    # --------------------------------------------------------
    n = 8000

    # --------------------------------------------------------
    # Initialise the vector a
    # --------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            a0 = np.asarray(previous, dtype=np.float64)
            if a0.size == n and np.sum(a0) > 0.0:
                a = a0 / np.sum(a0)          # put on the simplex
                print("[Init] Re‑using previous solution.")
            else:
                raise ValueError
        except Exception:
            a = np.full(n, 1.0 / n, dtype=np.float64)
            print("[Init] Failed to reuse previous solution – using uniform.")
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)
        print("[Init] No previous solution – using uniform.")

    best_seq = a.copy()
    best_val = _surrogate(a, n)
    print(f"[Init] initial surrogate = {best_val:.6f}")

    # --------------------------------------------------------
    # Quick random perturbations (optional – cheap)
    # --------------------------------------------------------
    for _ in range(30):
        perturb = a + rng.normal(scale=0.02, size=n)
        perturb = _proj_simplex(perturb)
        val = _surrogate(perturb, n)
        if val < best_val:
            best_val, best_seq = val, perturb.copy()
            print(f"[Perturb] improved surrogate = {val:.6f}")

    # --------------------------------------------------------
    # ADAM optimisation with a smooth‑max surrogate
    # --------------------------------------------------------
    # ADAM hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # FFT length for linear convolution / correlation (≥ 3n)
    N_fft = _next_pow2(3 * n - 2)

    # Soft‑max temperature (γ) schedule
    gamma_start = 2.0e4            # relatively smooth at the beginning
    gamma_end = 1.0e5              # sharper towards the end
    gamma_inc_iters = 250_000      # after this many iterations we hold γ = gamma_end

    it = 0

        A_hat = np.fft.rfft(a, n=N_fft)

        conv = np.fft.irfft(A_hat * A_hat, n=N_fft)[:2 * n - 1]

        max_conv = conv.max()

        # to avoid overflow we shift by the maximum value

        lr = 0.6 / math.sqrt(it)

            if cur_val < best_val:

                elapsed = time.time() - start_time
                print(f"[Iter] it={it:6d}, new best surrogate = {cur_val:.6f}, elapsed={elapsed:.1f}s")
            # early‑exit safety
            if time.time() >= deadline:
                break

    # --------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
