import time, math, numpy as np

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v):
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    return e / s

def _proj_simplex(v):
    # Euclidean projection onto simplex
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    return np.maximum(v - theta, 0.0)

def _uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _beta_shape(n, alpha, beta):
    if alpha <= 0 or beta <= 0:
        return _uniform(n)
    i = np.arange(1, n+1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s

def _mix_shapes(a, b, alpha):
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    if s <= 0:
        return _uniform(len(a))
    return mixed / s

def _objective_surrogate(seq):
    # surrogate for L1‑normalized seq: 2*n*max(conv)
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _subgradient_opt(start_seq, n, deadline, max_iters=500_000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # compute sub‑gradient of max entry
        g = np.zeros_like(a)
        i0 = max(0, max_idx - (n - 1))
        i1 = min(n - 1, max_idx)
        idx_i = np.arange(i0, i1 + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # simple diminishing step size
        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)
    return best_seq, best_val

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s

    """Convex mixture of two ℓ₁‑normalised vectors."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    if s <= 0.0:
        return _uniform(len(a))
    return mixed / s

    return 2.0 * n * np.max(conv)

        a = _proj_simplex(a)
