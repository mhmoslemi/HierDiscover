    n = _global_n
    alpha = _global_alpha
    # compute p from z
    p = np.exp(z)
    s = p.sum()

    conv = _conv_fft(p, p)
    max_c = np.max(conv)
    # log-sum-exp stable
    exp_vals = np.exp(alpha * (conv - max_c))
    sum_exp = np.sum(exp_vals)
    L = max_c + (1.0/alpha) * np.log(sum_exp)

    grad_p = 2.0 * corr
    # gradient w.r.t z
    dot = np.dot(p, grad_p)
    grad_z = p * (grad_p - dot)
    return L, grad_z.astype(np.float64)

"""
Propose a non‑negative step function (list of heights) that minimises the
official ``evaluate_sequence`` objective.

Algorithm
---------
* Choose a fairly large length (power of two) – here n = 8192.
* Build a diverse portfolio of analytic and random “seed” windows.
* Keep the best few seeds according to the exact surrogate
  2·n·max(conv(p,p))  (the surrogate equals the true objective for
  L1‑normalised p).
* Optimise each seed with a **limited‑memory BFGS** method on a smooth
  approximation of the max‑convolution:
        Lα(p) = (1/α)·log Σ exp(α·(p∗p)[k]) .
  The optimisation works on the log‑domain (z = log p) so that
  the simplex constraints (p ≥ 0, Σp = 1) are automatically satisfied.
  Three α‑stages (200, 800, 2000) are performed – the first stages shape
  the solution, the last stage focuses on the true peak.
* The best vector from the BFGS runs is polished with a short
  simulated‑annealing (SA) pass that reshuffles tiny mass fractions.
* Finally the vector is returned (as a plain Python list); the caller
  will invoke ``evaluate_sequence`` to obtain the true score.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

_lbfgs_alpha: float = 1.0

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    n = len(x) + len(y) - 1
    N = _next_pow2(n)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:n]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(p,p)) for a probability vector p."""
    n = p.shape[0]
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)

# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final safety normalisation (avoid exact zeros for log‑transform)
    clean = []
    eps = 1e-12
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, eps)          # guarantee positivity
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the smooth surrogate
# ----------------------------------------------------------------------
def _lbfgs_objective(z: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns (value, gradient) of the smoothed surrogate
        Lα(p) = (1/α)·log Σ exp(α·(p∗p)[k])
    where p = softmax(z) = exp(z) / Σ exp(z).
    Global variables ``_lbfgs_alpha``, ``_lbfgs_n`` and ``_lbfgs_fft_size``
    must be set before calling.
    """
    n = _lbfgs_n
    alpha = _lbfgs_alpha
    N = _lbfgs_fft_size

    # p = exp(z) / sum(exp(z))
    expz = np.exp(z)
    s = expz.sum()
    if s == 0.0:
        p = np.full_like(expz, 1.0 / n)
    else:
        p = expz / s

    # full convolution p * p
    conv = _conv_fft(p, p)                     # length 2n‑1
    max_c = np.max(conv)

    # log‑sum‑exp (stable)
    shifted = conv - max_c                     # ≤ 0
    exp_vals = np.exp(alpha * shifted)
    sum_exp = np.sum(exp_vals)
    L = max_c + (1.0 / alpha) * np.log(sum_exp)

    # softmax weights over the convolution entries
    w = exp_vals / sum_exp                     # length 2n‑1

    # correlation p ⋆ w (gradient of L w.r.t p)
    fx = np.fft.rfft(p, n=N)
    fw = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(np.conj(fx) * fw, n=N)[:n]   # length n
    grad_p = 2.0 * corr

    # chain rule: dz = Jᵀ·grad_p,  J = diag(p) - p pᵀ
    dot = np.dot(p, grad_p)
    grad_z = p * (grad_p - dot)

    return float(L), grad_z.astype(np.float64)


def _run_lbfgs(start: np.ndarray, alpha: float, max_iter: int, deadline: float) -> np.ndarray:
    """
    Runs L‑BFGS from ``start`` (already L1‑normalised) with a given
    smoothing parameter ``alpha``. Returns the best probability vector found
    (L1‑normalised). The routine respects the global ``deadline``.
    """
    global _lbfgs_alpha, _lbfgs_n, _lbfgs_fft_size

    n = start.shape[0]
    _lbfgs_alpha = alpha
    _lbfgs_n = n
    _lbfgs_fft_size = _next_pow2(3 * n - 2)

    # transform to log‑domain (add a tiny offset to avoid log(0))
    eps = 1e-15
    z0 = np.log(np.maximum(start, eps))

    # SciPy L‑BFGS‑B with analytic gradient
    options = {"maxiter": max_iter, "disp": False}
    # ``minimize`` will internally call the objective repeatedly; we break
    # out early if we run out of time by raising a custom exception.
    class _TimeOut(RuntimeError):
        pass

    def _callback(xk):
        if time.time() > deadline:
            raise _TimeOut()

    try:
        res = minimize(_lbfgs_objective, z0, method="L-BFGS-B",
                       jac=True, callback=_callback, options=options)
    except _TimeOut:
        # Time ran out – we still have the last point in ``z0`` (updated in‑place)
        pass

    # Recover p from the (potentially unfinished) solution
    expz = np.exp(z0) if 'z0' in locals() else np.exp(res.x)
    p = expz / expz.sum()
    # Enforce exact simplex projection (numerical safety)
    p = _proj_simplex(p)
    return p


# ----------------------------------------------------------------------
# Simple simulated‑annealing polish (identical to the previous implementation)
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> np.ndarray:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    n = cur.shape[0]

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.6:
            k_sub = rng.integers(3, 13)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    the official ``evaluate_sequence`` objective within ``budget_s`` seconds.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Length selection – a power of two for fast FFTs.
    # Larger n gives more flexibility; 8192 is still cheap for the chosen
    # optimisation budget.
    # ------------------------------------------------------------------
    n = 8192
    print(f"\n=== Starting search with n = {n} ===")

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} diverse seeds.")

    # Evaluate seeds (exact surrogate) and keep the best few
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS with a modest smoothing parameter
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    alphas = [200.0, 800.0, 2000.0]   # increasing focus on the true max
    max_iter_per_phase = 180          # L‑BFGS will internally do many gradient evals

    for phase, alpha in enumerate(alphas, start=1):
        if time.time() > deadline:
            break
        print(f"\n--- L‑BFGS Phase {phase} (α = {alpha}) ---")
        for idx, seed_vec in enumerate(best_seeds[:8]):   # a handful of promising seeds
            if time.time() > deadline:
                break
            cand = _run_lbfgs(seed_vec, alpha=alpha, max_iter=max_iter_per_phase, deadline=deadline)
            cand_val = _objective_surrogate(cand)
            if cand_val < best_val:
                best_val = cand_val
                best_seq = cand.copy()
                print(f"[Phase {phase}] seed #{idx+1} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – brief simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        remaining = deadline - time.time()
        max_sa_iters = int(min(300_000, remaining * 1800))
        best_seq = _simulated_annealing(best_seq, deadline, max_iters=max_sa_iters)
        best_val = _objective_surrogate(best_seq)
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible reuse by later runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
