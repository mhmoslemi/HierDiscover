def _smooth_obj_and_grad(v, n, beta):
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1:n-1+n]
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v
```

def _random_blocks(n, num_blocks, rng):
    block_size = n // num_blocks
    weights = rng.random(num_blocks) + 0.01  # >0
    a = np.repeat(weights, block_size)
    remainder = n - len(a)

    max‑operator.
3.  Polish the resulting vector with a long exponentiated‑gradient (mirror‑descent)
    on the exact max‑convolution objective; finally up‑sample (linear
    interpolation) to a finer grid and repeat the Adam + EG steps if time
    permits.

All vectors are kept ℓ₁‑normalised (scale‑invariant objective) and non‑negative.
The final best vector is stored in the global name ``height_sequence_1``.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate objective (smooth max) and its gradient w.r.t. logits
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (surrogate objective, gradient w.r.t. logits v) for a given β.
    The surrogate is 2·n·(shift + log∑exp(β·(conv‑shift))/β),
    where shift = max(conv).
    """
    a = _softmax(v)                         # probability simplex
    conv = _conv_fft(a, a)                  # full auto‑convolution
    shift = np.max(conv)

    # log‑sum‑exp of β·(conv‑shift)
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_beta / sum_exp                  # convex weights
    rev_a = a[::-1]
    # convolution rev_a * w  -> length 2n‑1 ; extract the n centre entries
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)             # chain rule through softmax
    return obj, grad_v


def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   betas=None,
                   iters_per_beta: int = 800,
                   lr: float = 0.08) -> np.ndarray:
    """
    Adam optimiser on the smooth surrogate.
    Parameters:
        start_vec      – initial ℓ₁‑normalised probability vector
        n              – length of the vector
        deadline       – unix timestamp after which the optimisation stops
        betas          – list of β values (increasing) used for continuation
        iters_per_beta – iterations allocated for each β
        lr             – base learning rate for Adam
    Returns a ℓ₁‑normalised vector (softmax of the final logits).
    """
    if betas is None:
        betas = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]

    v = np.log(start_vec + 1e-30)            # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    best_v = v.copy()
    best_obj = float("inf")

    for beta in betas:
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                return _softmax(best_v)

            t += 1
            obj, grad = _smooth_obj_and_grad(v, n, beta)
            # keep best seen (surrogate objective)
            if obj < best_obj:
                best_obj = obj
                best_v = v.copy()

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v2 = beta2 * v2 + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)
            v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return _softmax(best_v)


# ----------------------------------------------------------------------
# Exact max‑convolution optimisation (exponentiated gradient)
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 500_000,
                          eta_scale: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # Sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta_scale / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geometric(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, ℓ₁‑normalised."""
    if not (0.0 < r < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture of two ℓ₁‑normalised vectors."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_dirichlet_sorted(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet sample sorted in descending order (strongly skewed)."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    a.sort()
    a = a[::-1]   # descending
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_blocks(n: int, block_cnt: int, rng: np.random.Generator) -> np.ndarray:
    """Piecewise‑constant random blocks (non‑negative)."""
    if block_cnt <= 0:
        return _uniform(n)
    # assign a random weight to each block
    weights = rng.random(block_cnt) + 1e-3
    # repeat each weight to fill the vector
    repeats = (n + block_cnt - 1) // block_cnt
    a = np.repeat(weights, repeats)[:n]
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = new_seq / new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
# Simple surrogate for quick ranking (exact max‑convolution)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed generation (much richer than the previous version)
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        # reuse sine‑window as a simple smooth family
        i = np.arange(n, dtype=np.float64)
        a = np.sin(np.pi * (i + 1) / (n + 1)) ** g
        a = a / a.sum()
        seeds.append(a)

    # Beta family – much wider parameter grid than before
    alphas = np.geomspace(0.05, 10.0, num=12)
    betas = np.geomspace(0.05, 10.0, num=12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Geometric decays + mixtures
    for r in (0.70, 0.80, 0.85, 0.90, 0.93, 0.95, 0.97, 0.98, 0.99):
        seeds.append(_geometric(n, r))
        # mixtures with uniform
        seeds.append(_mix_shapes(_uniform(n), _geometric(n, r), 0.3))
        seeds.append(_mix_shapes(_uniform(n), _geometric(n, r), 0.6))

    # Mixture of two geometric families
    for r1, r2, a in [(0.8, 0.95, 0.4), (0.85, 0.99, 0.5), (0.7, 0.99, 0.3)]:
        g1 = _geometric(n, r1)
        g2 = _geometric(n, r2)
        seeds.append(_mix_shapes(g1, g2, a))

    # Random Dirichlet draws – many concentrations, sorted versions too
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
            seeds.append(_random_dirichlet_sorted(n, conc, rng))

    # Random block‑structured vectors
    for blk in (3, 5, 7, 11):
        for _ in range(3):
            seeds.append(_random_blocks(n, blk, rng))

    # Previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise all seeds robustly
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found (list of floats).  The global variable
    ``height_sequence_1`` is updated with the result.
    """
    # ------------------------------------------------------------------
    # Initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – coarse resolution (fast exploration)
    # ------------------------------------------------------------------
    n0 = 2000
    print(f"\n=== Starting search (coarse length n={n0}) ===")

    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    # generate a large pool of seeds
    seeds = _collect_seeds_extended(n0, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors (coarse).")

    # cheap surrogate ranking
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (coarse):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on the smooth surrogate (continuation)
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")

    # continuation schedule (β values) – more stages than before
    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]
    iters_per_beta = 800  # modest but fast

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all seeds (Adam).")
            break

        print(f"\n--- Adam optimisation on seed #{idx+1} ---")
        seq_opt = _adam_optimize(seed_vec,
                                 n0,
                                 deadline,
                                 betas=beta_sched,
                                 iters_per_beta=iters_per_beta,
                                 lr=0.08)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[Adam] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_seq = seq_opt.copy()
            print(f"[Update] New global best (coarse) = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient refinement on exact max‑conv.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n0).copy()
        global_best_val = _objective_surrogate(global_best_seq)

    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate a good chunk for EG (roughly 30‑40% of remaining time)
        max_exp_iters = int(min(2_500_000, remaining * 0.35 * 1e5))
        print(f"\n--- Exponentiated‑gradient refinement (coarse) "
              f"max_iters={max_exp_iters} ---")
        seq_exp, val_exp = _exp_gradient_optimize(global_best_seq,
                                                  n0,
                                                  deadline,
                                                  max_iters=max_exp_iters,
                                                  eta_scale=0.5)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – up‑sampling to a finer grid (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 30.0:
            n1 = n0 * 4            # 8000 points – still comfortable for FFTs
            print(f"\n=== Upsampling to finer length n={n1} ===")
            up_seq = _upsample_sequence(global_best_seq, n1)

            # Adam on the finer grid (shorter schedule, more iterations)
            finer_beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]
            finer_iters_per_beta = 1200
            seq_fine = _adam_optimize(up_seq,
                                      n1,
                                      deadline,
                                      betas=finer_beta_sched,
                                      iters_per_beta=finer_iters_per_beta,
                                      lr=0.06)
            fine_val = _objective_surrogate(seq_fine)
            print(f"[Fine‑Adam] surrogate = {fine_val:.6f}")

            if fine_val < global_best_val:
                global_best_seq = seq_fine.copy()
                global_best_val = fine_val
                n0 = n1
                print(f"[Update] new best after up‑sampling = {global_best_val:.6f}")

            # final EG polishing on the finer grid
            if time.time() < deadline:
                rem = deadline - time.time()
                max_exp2 = int(min(800_000, rem * 0.25 * 1e5))
                print(f"\n--- Final exponentiated‑gradient (fine) "
                      f"max_iters={max_exp2} ---")
                seq_final, val_final = _exp_gradient_optimize(global_best_seq,
                                                              n0,
                                                              deadline,
                                                              max_iters=max_exp2,
                                                              eta_scale=0.5)
                if val_final < global_best_val:
                    global_best_seq = seq_final.copy()
                    global_best_val = val_final
                    print(f"[Final‑ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
