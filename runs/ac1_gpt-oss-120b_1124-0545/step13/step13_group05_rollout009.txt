"""
Improved optimisation for the “max‑convolution” step–function problem.

The code follows three clear stages

1️⃣  **Rich seed generation** – a dense sweep over analytically
    motivated families (uniform, triangular, sine‑window,
    Gaussian, power‑law, Beta, pure & offset exponentials) and a few
    random Dirichlet draws.  All seeds are ℓ₁‑normalised (∑a = 1).

2️⃣  **Long Adam optimisation** – start from the best seed and run a
    projected Adam optimiser on the log‑softmax parameters.  The
    max‑convolution is smoothed by a temperature‑controlled
    log‑sum‑exp; the temperature is increased exponentially up to a
    large value (≈ 5000) so the algorithm finally follows the true
    ℓ∞‑objective.  The loop runs for as many iterations as the
    remaining time budget allows (typically several hundred‑thousand
    updates for the 8000‑dimensional problem).

3️⃣  **Exact sub‑gradient polishing** – after the Adam phase we switch
    to a true sub‑gradient (the gradient of the active max‑convolution
    entry) and project onto the probability simplex after each step.
    This stage refines the solution further and also uses the full
    remaining time budget.

Both optimisation phases keep track of the best sequence seen so far.
At the end we evaluate the final candidate with ``evaluate_sequence``,
store it in a global variable for possible reuse and return it as a
plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised
    vector (∑seq = 1).  This equals the real ``evaluate_sequence`` value
    because the denominator cancels (∑seq = 1 → denominator = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_shape(n: int, r: float) -> np.ndarray:
    """Monotone exponential a_i ∝ r**i, ℓ₁‑normalised."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_offset_shape(n: int, offset: int, r: float) -> np.ndarray:
    """Zero prefix of length *offset* followed by an exponential tail."""
    if offset >= n:
        return _uniform(n)
    tail_len = n - offset
    if r <= 0.0:
        tail = np.full(tail_len, 1.0 / tail_len, dtype=np.float64)
    else:
        i = np.arange(tail_len, dtype=np.float64)
        tail = np.power(r, i)
        tail /= tail.sum()
    a = np.concatenate([np.zeros(offset, dtype=np.float64), tail])
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   deadline: float,
                   max_iters: int | None = None,
                   beta_max: float = 5000.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parametrisation with a
    temperature‑controlled log‑sum‑exp smoothing of the max.
    Returns the best sequence seen and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits – unconstrained
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    # start temperature (smoothness) and its multiplicative schedule
    beta = 0.5
    beta_mult = 1.001  # exponential growth
    lr0 = 1.0

    best_seq = start_vec.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1

        a = _softmax(v)                     # stays on simplex, ∑a = 1
        conv = _conv_fft(a, a)
        cur_val = 2.0 * len(a) * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # --- smoothed max (log‑sum‑exp) ---
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            # numerical safety – skip this iteration
            continue
        w = w / w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (see derivation in the paper)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3·n‑2 (mid‑section used)
        grad_a = 2.0 * full[len(a) - 1:len(a) - 1 + len(a)]

        # Jacobian of log‑softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam step
        lr = lr0 / math.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule
        if beta < beta_max:
            beta = min(beta * beta_mult, beta_max)

    return best_seq, best_val


def _subgradient_optimize(start_vec: np.ndarray,
                          deadline: float,
                          max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on the true ℓ∞‑objective
    (max convolution).  After each step we project onto the simplex.
    """
    a = start_vec.copy()
    n = len(a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step size
        lr = 0.20 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation (expanded families + random draws)
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_power(n, p=1.0))
    seeds.append(_power(n, p=2.0))

    # Beta shapes – grid on (α,β) excluding the trivial (1,1)
    alphas = np.linspace(0.3, 3.0, 7)
    betas = np.linspace(0.3, 3.0, 7)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Exponential family – dense sweep around the theoretical optimum
    base_eps = 2.0 / n
    eps_vals = np.linspace(0.5 * base_eps, 3.0 * base_eps, 80)
    for eps in eps_vals:
        r = max(0.0, 1.0 - eps)
        seeds.append(_exponential_shape(n, r))

    # Offset exponentials – zero prefix + exponential tail
    offsets = [0,
               int(0.01 * n), int(0.03 * n), int(0.06 * n),
               int(0.10 * n), int(0.15 * n), int(0.20 * n)]
    for off in offsets:
        L = n - off
        if L <= 0:
            continue
        base_eps_off = 2.0 / L
        eps_off_vals = np.linspace(0.5 * base_eps_off,
                                   3.0 * base_eps_off,
                                   30)
        for eps in eps_off_vals:
            r = max(0.0, 1.0 - eps)
            seeds.append(_exponential_offset_shape(n, off, r))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # Random exponential draws
    for _ in range(6):
        a = rng.exponential(scale=1.0, size=n)
        a /= a.sum()
        seeds.append(a)

    # Previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final normalisation guard
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted time budget.
    Returns the best sequence found as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # initialise RNG and timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a small safety margin for the final validation call
    overall_deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # problem size – 8000 proved to be a good trade‑off
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    print(f"\n--- Generating seeds (n={n}) ---")
    seed_pool = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seed_pool)} candidate seeds.")

    # cheap surrogate evaluation to pick the most promising ones
    scored = [(vec, _objective_surrogate(vec)) for vec in seed_pool]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    promising = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # initialise global best
    global_best_seq = promising[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – intensive Adam optimisation on the single best seed
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 20.0:
        max_adam_iters = int(min(3_000_000, remaining * 2500))
        print(f"\n--- Adam optimisation (max_iters≈{max_adam_iters}) ---")
        seq_adam, val_adam = _adam_optimize(
            global_best_seq,
            deadline=overall_deadline,
            max_iters=max_adam_iters,
            beta_max=5000.0,
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient polishing (uses the true max)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 10.0:
        max_sub_iters = int(min(5_000_000, remaining * 3000))
        print(f"\n--- Sub‑gradient polishing (max_iters≈{max_sub_iters}) ---")
        seq_sub, val_sub = _subgradient_optimize(
            global_best_seq,
            deadline=overall_deadline,
            max_iters=max_sub_iters,
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[Sub‑grad] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation (the user‑visible score)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
