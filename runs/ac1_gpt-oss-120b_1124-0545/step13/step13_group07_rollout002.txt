"""
Improved step‑function search for minimise ``evaluate_sequence``.

Algorithm
---------
0️⃣  Initialise a large pool of analytically‑inspired and random seeds,
    all ℓ₁‑normalised (Σ = 1).  Keep the best *top_k* by a fast FFT‑based
    surrogate ``2·n·max(conv(a,a))``.

1️⃣  **Adam on a smoothed max** – each top seed is refined by Adam on the
    log‑softmax parametrisation of the simplex.  The temperature of the
    log‑sum‑exp surrogate is increased much more slowly (up to 1 e4) so
    the optimiser sees a near‑exact max.

2️⃣  **Mirror‑descent (exponentiated sub‑gradient)** – start from the best
    Adam result and run a long exponentiated‑gradient descent on the exact
    max (no smoothing).  The learning‑rate schedule is made more aggressive
    (larger initial step, slower decay) to reach a finer local optimum.

3️⃣  **Iterated LP polishing** – up to 20 linear‑programming refinements.
    Each LP solves ``max Σ g_i`` subject to ``conv(f,g) ≤ M`` where
    ``M = max conv(f,f)``.  The mixing weight τ decays as ``τ = 0.2/√it``
    giving very fine adjustments near convergence.

4️⃣  **Targeted simulated‑annealing** – a lightweight SA that
    *focuses* on the active convolution index (the current maximum) and
    moves a small fraction of mass from “high‑contribution’’ positions
    to “low‑contribution’’ positions.  Random moves are also interleaved
    (swap, Dirichlet‑redistribute, pairwise transfer).  The temperature
    schedule is exponential and the number of iterations is limited only
    by the remaining time budget.

All stages respect a global deadline (default 1000 s).  The best sequence
found is stored globally as ``height_sequence_1`` and also returned.

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate_obj(seq: np.ndarray) -> float:
    """
    Cheap surrogate (Σseq=1) used for ranking:
        2·n·max(conv(seq,seq))
    The denominator Σseq² is omitted because it is constant for normalised vectors.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random 3‑segment piece‑wise linear shape."""
    breaks = np.sort(rng.choice(n, size=3, replace=False))
    heights = rng.random(3) + 0.01
    x = np.arange(n, dtype=np.float64)
    a = np.interp(x, breaks, heights)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few randomly positioned Gaussians."""
    a = np.zeros(n, dtype=np.float64)
    num = rng.integers(2, 6)
    for _ in range(num):
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.02 * n, 0.15 * n)
        x = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, nf: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth curve built from a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, nf + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 400_000) -> tuple[np.ndarray | None, float]:
    """
    Adam on log‑softmax parameters with a log‑sum‑exp surrogate of the max.
    Temperature β is increased slowly (up to 1e4) for a tighter approximation.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.9
    beta = 0.5               # start temperature
    beta_inc = 0.3
    beta_max = 10000.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # smooth max via log‑sum‑exp with temperature β
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                max_iters: int = 8_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the **exact** max.
    Uses a slower decay of the learning‑rate for finer adjustments.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    lr0 = 0.6    # a bit larger than before
    power = 0.25

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # construct sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _lp_refine(seq: np.ndarray, deadline: float,
               max_iters: int = 20, init_tau: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Perform several LP refinements.

    Each iteration solves:
        maximise   Σ g_i
        subject to conv(seq, g) ≤ M   (where M = max conv(seq, seq))
                  g_i ≥ 0
    The solution g is mixed back with a decreasing weight τ (τ_i = init_tau / sqrt(i))
    and renormalised, guaranteeing a non‑increasing objective.
    """
    f = seq.copy()
    n = len(f)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        M = _conv_fft(f, f).max()

        # Build convolution matrix A (size (2n‑1) × n) in CSR format.
        rows = np.empty(n * n, dtype=np.int32)
        cols = np.empty(n * n, dtype=np.int32)
        data = np.empty(n * n, dtype=np.float64)

        idx = 0
        for k in range(n):
            start = k
            end = k + n
            rows[idx:idx + n] = np.arange(start, end, dtype=np.int32)
            cols[idx:idx + n] = k
            data[idx:idx + n] = f
            idx += n

        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if not res.success:
            print(f"[LP] iteration {it} failed – stopping LP stage.")
            break

        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g <= sum_f + 1e-12:
            print(f"[LP] iteration {it} – no L1 gain, stopping.")
            break

        tau = init_tau / np.sqrt(it)   # decreasing step size
        f_new = (1.0 - tau) * f + tau * g
        f_new = f_new / f_new.sum()
        f = f_new

        cur_obj = _surrogate_obj(f)
        print(f"[LP] iteration {it} – τ={tau:.5f}  obj={cur_obj:.6f}")

    return f, _surrogate_obj(f)


def _targeted_move(cur: np.ndarray, max_idx: int, rng: np.random.Generator,
                   frac: float = 0.05) -> np.ndarray:
    """
    Move a small fraction of mass from a high‑contribution element
    (i with large a_i·a_{max_idx-i}) to a low‑contribution element.
    Returns a copy with the modification applied.
    """
    n = len(cur)

    # Indices that actually contribute to the active convolution position.
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = max_idx - idx_i
    contrib = cur[idx_i] * cur[rev_idx]

    if contrib.sum() == 0.0:
        # fallback to a generic small random move
        new = cur.copy()
        i, j = rng.choice(n, size=2, replace=False)
        eps = cur[i] * frac
        new[i] = max(0.0, cur[i] - eps)
        new[j] = cur[j] + eps
        return new

    # Pick a source index (high contribution) proportional to its contribution.
    prob_src = contrib / contrib.sum()
    src = rng.choice(idx_i, p=prob_src)

    # Choose a destination index outside the active window (or any low‑contribution).
    # To keep it simple we pick uniformly from the whole set excluding src.
    dst_candidates = np.delete(np.arange(n), src)
    dst = rng.choice(dst_candidates)

    eps = cur[src] * frac
    new = cur.copy()
    new[src] = max(0.0, cur[src] - eps)
    new[dst] = cur[dst] + eps
    return new


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                 max_iters: int = 6_000_000,
                                 init_temp_factor: float = 0.07,
                                 min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """Targeted lightweight SA with many cheap moves."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate_obj(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 4)   # four move types now
        if move_type == 0:
            # swap two entries (simple permutation)
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            # random Dirichlet redistribution on a small subset
            k = rng.integers(2, 16)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total == 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        elif move_type == 2:
            # pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total == 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]
        else:
            # targeted move using the current max‑index
            conv = _conv_fft(cur, cur)
            max_idx = int(np.argmax(conv))
            new = _targeted_move(cur, max_idx, rng, frac=0.07)

        new_val = _surrogate_obj(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 200_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))

    # parameter sweeps
    for sigma in np.linspace(0.05, 0.35, 10):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 5.5, 12):
        seeds.append(_power_law(n, p))

    alphas = np.linspace(0.2, 3.5, 12)
    betas = np.linspace(0.2, 3.5, 12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for lam in np.linspace(0.0003, 0.02, 12):
        seeds.append(_exponential_decay(n, lam))

    # random Dirichlet draws with a broad range of concentrations
    concs = np.logspace(-5, 0.6, 14)   # 1e‑5 … ~4.0
    for conc in concs:
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixtures of random Gaussians
    for _ in range(8):
        seeds.append(_gaussian_mixture(n, rng))

    # piece‑wise‑linear random shapes
    for _ in range(8):
        seeds.append(_piecewise_linear(n, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 5, 7, 9):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mix uniform with low‑frequency curves
    for mix_alpha in (0.15, 0.3, 0.6):
        for nf in (2, 4, 6):
            curve = _lowfreq_curve(n, nf, rng)
            mixed = (1.0 - mix_alpha) * _uniform(n) + mix_alpha * curve
            mixed = mixed / mixed.sum()
            seeds.append(mixed)

    # previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0.0:
            seeds.append(avg / avg.sum())

    return seeds

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective.  Returns the best sequence as a plain
    list of floats and registers it globally as ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # problem size – we keep the historically used value (4000)
    n = 4000

    # try to load a previous solution (if this script is run iteratively)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size != n:
                if previous.size > n:
                    previous = previous[:n]
                else:
                    previous = np.pad(previous, (0, n - previous.size), constant_values=0.0)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – generate a rich seed pool
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _surrogate_obj(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _surrogate_obj(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation on smooth surrogate (few strong seeds)
    # ------------------------------------------------------------------
    print("\n--- Adam optimisation (strong seeds) ---")
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate a fraction of remaining time to each Adam run
        max_iters = min(500_000, int(remaining * 60))
        print(f"\n[Adam] run {idx}/{len(best_seeds)} – max_iters={max_iters}")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline,
                                            max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Mirror‑descent (exact max) – longest phase
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=8_000_000
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Iterated LP polishing (more iterations, decreasing τ)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Linear‑programming polishing (extended) ---")
        seq_lp, val_lp = _lp_refine(global_best_seq, deadline,
                                    max_iters=20, init_tau=0.2)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[overall‑LP] new global best = {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Targeted simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Targeted simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline,
            max_iters=6_000_000, init_temp_factor=0.07
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (for information only)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose the result for the next iteration (if any)
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
