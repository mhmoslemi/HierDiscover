"""
enhanced search for a non‑negative step function
------------------------------------------------

The algorithm builds a very diverse pool of analytic and random
distributions (uniform, triangular, sine, Gaussian, beta, Tukey,
exponential‑decay, random Dirichlet, low‑frequency cosine sums …).
All vectors are ℓ₁‑normalised and projected onto the probability simplex.

1. **Seeding** – a few thousand candidates are generated and scored with a
   fast FFT‑based surrogate  `2·n·max(conv(p,p))`.  The best `top_k` vectors
   are kept.

2. **Soft‑max Adam** – each top seed receives a short Adam optimisation with
   a smooth approximation of the max (log‑sum‑exp).  The temperature `α`
   grows from a modest value to a very large one, turning the smooth loss
   into the true `max`.  The best Adam result becomes the new incumbent.

3. **Long‑run Adam** – the incumbent is refined with a much longer Adam run,
   a larger final temperature (`α≈8000`) and a fine‑tuned learning‑rate.
   This stage alone often pushes the objective well below the previous record.

4. **Mirror‑descent (exponential‑gradient)** – a sub‑gradient step on the
   active convolution shift (`k = argmax(conv)`) is applied many times,
   always projecting back onto the simplex.  This removes the remaining
   “spikes” that the smooth Adam cannot see.

5. **Simulated‑annealing polishing** – a classic SA pass performs tiny
   transfers and random reshuffles on small subsets, accepting any
   improvement (or a Metropolis move).  This finishes the search with a
   final local optimum.

All stages respect a global deadline (`budget_s` seconds) – the function
returns a valid list before the timeout, otherwise the competition scores
−∞.  The final sequence is stored in the global name ``height_sequence_1``
for reuse in later runs.

"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(p,p)) for a distribution with ∑p=1."""
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection – huge, very diverse library
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a large, diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families – a handful of parameter sweeps
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))

    for sigma in (0.12, 0.16, 0.20, 0.24, 0.28):
        seeds.append(_gaussian(n, sigma))

    for rate in (0.99, 0.995, 0.998, 0.999, 0.9995):
        seeds.append(_exponential_decay(n, rate))

    for p in (0.3, 0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))

    alphas = (0.3, 0.5, 0.8, 1.2, 2.0)
    betas = (0.3, 0.5, 0.8, 1.2, 2.0)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for alpha in (0.1, 0.2, 0.35, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws – many concentrations
    for conc in (0.01, 0.03, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves – different numbers of frequencies
    for nf in (2, 3, 5, 7):
        for _ in range(5):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures of uniform with a low‑freq curve
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4, 6):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # mixture of two random Dirichlet draws
    for _ in range(8):
        a = _random_dirichlet(n, rng.uniform(0.01, 5.0), rng)
        b = _random_dirichlet(n, rng.uniform(0.01, 5.0), rng)
        lam = rng.random()
        seeds.append(_mix_shapes(a, b, lam))

    # previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final safety‑check – ensure every seed is a proper probability vector
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0:
            v = _uniform(n)
            s = v.sum()
        clean.append(v / s)
    return clean


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000,
                   alpha_start: float = 20.0, alpha_end: float = 2000.0,
                   lr: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        L(p) = (1/α)·log Σₖ exp(α·(p*p)[k]).
    α is increased linearly from ``alpha_start`` to ``alpha_end``.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # FFT size – large enough for both convolution and correlation
    N = _next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- convolution p * p (full) -----
        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        # ----- soft‑max over the convolution -----
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)
        mc = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - mc))   # stability shift
        soft = exp_vals / np.sum(exp_vals)

        # ----- correlation p ⋆ soft (gradient of L) -----
        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # ----- projection onto simplex -----
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            # occasional status output
            if t % 50000 == 0:
                print(f"[Adam] iter={t:7d} best={best_val:.6f} α={alpha:.1f}")

    return best_x, best_val


def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000,
                      eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative update)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # avoid overflow/underflow
        x = x * np.exp(delta)

        # project onto the simplex
        s = x.sum()
        if s <= 0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Short SA polish – works with the fast surrogate."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.6:
            # reshuffle a random subset (size 3‑12)
            k_sub = rng.integers(3, 13)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the official
    ``evaluate_sequence`` objective within ``budget_s`` seconds.
    Returns the best sequence (list of floats) found.
    """
    # ------------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # tiny safety margin

    # Length – 4000 gives a good trade‑off between quality and speed
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution (if any)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # Stage 0 – Very diverse seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} raw seeds.")

    # Score all seeds with the fast surrogate
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 30
    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Best surrogate after seeding: {scored[0][1]:.6f}")

    # ------------------------------------------------------------------
    # Initial incumbent
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] incumbent surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Short Adam runs on the top few seeds
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Short Adam optimisation on top seeds ---")
        seeds_for_adam = top_seeds[:5]      # try a handful of promising starts
        for idx, seed_vec in enumerate(seeds_for_adam, 1):
            if time.time() >= deadline:
                break
            remaining = deadline - time.time()
            max_iters = int(min(120_000, remaining * 130))   # adapt to time left
            if max_iters <= 0:
                break
            print(f"[Adam‑short] seed #{idx}, max_iters={max_iters}")
            cand, val = _adam_optimize(seed_vec, n, deadline,
                                       max_iters=max_iters,
                                       alpha_start=20.0,
                                       alpha_end=1500.0,
                                       lr=0.12)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"   New incumbent surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Long‑run Adam on the incumbent (high final temperature)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long‑run Adam refinement (high α) ---")
        remaining = deadline - time.time()
        max_iters = int(min(400_000, remaining * 1400))
        if max_iters > 0:
            best_seq, best_val = _adam_optimize(best_seq, n, deadline,
                                                max_iters=max_iters,
                                                alpha_start=500.0,
                                                alpha_end=8000.0,
                                                lr=0.10)
            print(f"[Adam‑long] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Exponential‑gradient polishing (sub‑gradient on active shift)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponential‑gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(600_000, remaining * 1500))
        if max_iters > 0:
            best_seq, best_val = _expgrad_optimize(best_seq, n, deadline,
                                                   max_iters=max_iters,
                                                   eta0=0.25)
            print(f"[EG] surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(500_000, remaining * 1800))
        if max_iters > 0:
            best_seq, best_val = _simulated_annealing(best_seq, n, deadline,
                                                      max_iters=max_iters)
            print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – Quick greedy hill‑climb on the surrogate (very cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(30_000, remaining * 3000))
        if iters > 0:
            print("\n--- Final greedy surrogate hill‑climb ---")
            cur = best_seq.copy()
            cur_val = best_val
            for it in range(iters):
                i, j = rng.integers(n, size=2, replace=False)
                eps = rng.random() * min(cur[i], 0.03)
                if eps <= 0.0:
                    continue
                cand = cur.copy()
                cand[i] -= eps
                cand[j] += eps
                # sum is unchanged and all entries stay ≥0
                cand_val = _objective_surrogate(cand)
                if cand_val < cur_val:
                    cur = cand
                    cur_val = cand_val
            best_seq, best_val = cur, cur_val
            print(f"[Hill‑climb] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (exact objective)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible reuse by the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
