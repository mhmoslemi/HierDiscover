def _kaiser_window(n, beta):
    if beta <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    arg = 1 - ((i - alpha) / alpha) ** 2
    w = np.where(arg > 0, np.i0(np.pi * beta * np.sqrt(arg)), 0.0)
    return w / w.sum()

def _generate_parametric_seeds(n, rng):
    seeds = []
    # exponential
    for alpha in np.logspace(-5, -1, 30):
        r = 1.0 - alpha
        if r <= 0.0:
            continue
        seeds.append(_exponential_decay(n, rate=r))
    # power
    for p in np.linspace(0.1, 4.0, 30):
        seeds.append(_power(n, p))
    # gaussian
    for sigma in np.linspace(0.05, 0.5, 20):
        seeds.append(_gaussian(n, sigma))
    # beta
    alphas = np.linspace(0.5, 4.0, 5)
    betas = np.linspace(0.5, 4.0, 5)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))
    # tukey
    for alpha in np.linspace(0.1, 0.9, 15):
        seeds.append(_tukey_window(n, alpha))
    # kaiser
    for beta in np.linspace(0.0, 12.0, 20):
        seeds.append(_kaiser_window(n, beta))
    # random Dirichlet low conc
    for conc in [0.05, 0.1, 0.2]:
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    return seeds

"""
Propose a step‑function (non‑negative vector) that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a))²

The algorithm works in four stages:

1. **Rich seed generation** – deterministic windows (uniform, triangular,
   sine, Gaussian, beta, Tukey, Kaiser, exponential, power) plus a dense
   parametric sweep (exponential rate, power‐law exponent, Gaussian sigma,
   beta α/β, Tukey α, Kaiser β) and a few random Dirichlet draws.  All seeds
   are ℓ₁‑normalised.

2. **Surrogate ranking** – each seed is evaluated with a fast FFT‑based
   surrogate ``_objective_surrogate`` (exact when the ℓ₁‑sum is 1).  The
   best 20 seeds are retained.

3. **Local refinement** – a quick Exponentiated‑Gradient (multiplicative)
   mirror‑descent is run on the top few seeds (≈30 k iterations each) to
   escape poor local minima.  The best result then undergoes a longer
   Adam optimisation of the smooth soft‑max surrogate (parameter p
   increased gradually).  Finally a short additive projected‑gradient
   polish cleans up the remaining slack.

4. **Official evaluation** – the resulting vector is fed to
   ``evaluate_sequence`` and stored globally as ``height_sequence_1``.

All stages respect the global deadline (default 1000 s).  The code uses
only NumPy and (optionally) SciPy for the Kaiser window; it contains no
closures or lambdas and can be called repeatedly.
"""

import time
import math
import numpy as np
from scipy import special  # for the Kaiser window (i0)

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Fast surrogate for the official objective (exact when Σseq = 1):
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    """Geometric series a_i = rate**i, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser‑window (β≥0).  Uses the modified Bessel function I₀."""
    if beta <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    # argument of the sqrt term, clipped to non‑negative
    tmp = 1.0 - ((i - alpha) / alpha) ** 2
    tmp = np.clip(tmp, 0.0, 1.0)
    w = special.i0(np.pi * beta * np.sqrt(tmp))
    s = w.sum()
    return w / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Parametric seed generation
# ----------------------------------------------------------------------
def _generate_parametric_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse list of analytic windows with finely sampled parameters."""
    seeds: list[np.ndarray] = []

    # --- exponential decay (rate = 1‑α) --------------------------------
    for alpha in np.logspace(-5, -1, 30):          # α = 1‑rate
        rate = max(0.0, 1.0 - alpha)
        seeds.append(_exponential_decay(n, rate))

    # --- power law -------------------------------------------------------
    for p in np.linspace(0.1, 4.0, 30):
        seeds.append(_power(n, p))

    # --- Gaussian --------------------------------------------------------
    for sigma in np.linspace(0.05, 0.5, 20):
        seeds.append(_gaussian(n, sigma))

    # --- beta shape -------------------------------------------------------
    alphas = np.linspace(0.5, 4.0, 5)
    betas = np.linspace(0.5, 4.0, 5)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # --- Tukey window -----------------------------------------------------
    for a in np.linspace(0.1, 0.9, 15):
        seeds.append(_tukey_window(n, a))

    # --- Kaiser window ----------------------------------------------------
    for b in np.linspace(0.0, 12.0, 20):
        seeds.append(_kaiser_window(n, b))

    # --- random Dirichlet draws (low concentration) ----------------------
    for conc in (0.05, 0.1, 0.2):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    return seeds


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (multiplicative updates on the simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL (multiplicative) updates."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)   # keep exponent bounded
        x = x * np.exp(delta)

        # simplex projection
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()

    return best_x, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max‑smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_smooth_optimize(start: np.ndarray, n: int, deadline: float,
                          init_lr: float = 0.04,
                          beta1: float = 0.9, beta2: float = 0.999,
                          eps: float = 1e-8,
                          p0: float = 5.0,
                          p_factor: float = 2.0,
                          p_step: int = 20_000,
                          max_total_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth surrogate
        F_p(a) = (1/p)·log Σ_k exp(p·conv_k(a))
    The smoothing parameter p grows geometrically.
    Returns the best exact surrogate value seen.
    """
    a = start.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    p = p0

    while time.time() < deadline and t < max_total_iters:
        # ---- current convolution & exact surrogate ----
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # ---- soft‑max weights ----
        shifted = conv - max_conv               # ≤ 0
        exp_arg = p * shifted
        exp_arg = np.clip(exp_arg, -700, 0)     # prevent overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:                        # fall back to hard max
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ---- gradient of the smooth surrogate ----
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)        # length 3n‑2
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # ---- Adam update ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = _uniform(n)

        # ---- schedule p ----
        if t % p_step == 0:
            p *= p_factor
    # final exact evaluation of the best vector
    final_val = 2.0 * n * np.max(_conv_fft(best_a, best_a))
    return best_a, final_val


# ----------------------------------------------------------------------
# Additive projected sub‑gradient polish (tiny steps, very cheap)
# ----------------------------------------------------------------------
def _additive_subgrad_optimize(start: np.ndarray, n: int, deadline: float,
                              max_iters: int = 80_000, eta0: float = 0.15) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        x = x - eta * g
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s > 0.0:
            x = x / s
        else:
            x = _uniform(n)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val

        except Exception:
            prev_arr = None

        if val < best_val:

    # Stage 3 – long Adam optimisation on the best vector.
    # --------------------------------------------------------------
    if time.time() < deadline:

        # allocate most of the remaining budget to Adam
        max_adam_iters = int(min(1_500_000, remaining * 1800))
        print("\n--- Adam smooth optimisation (soft‑max) ---")
        best_seq, best_val = _adam_smooth_optimize(best_seq, n, deadline,
                                                   init_lr=0.04,
                                                   p0=5.0,
                                                   p_factor=2.0,
                                                   p_step=20000,
                                                   max_total_iters=max_adam_iters)
        print(f"[Adam] best surrogate after polishing = {best_val:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(80_000, remaining * 2000))
        if max_iters > 0:
            print("\n--- Additive sub‑gradient polishing ---")
            best_seq, best_val = _additive_subgrad_optimize(best_seq, n, deadline,
                                                            max_iters=max_iters,
                                                            eta0=0.15)
            print(f"[Additive] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse in a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
