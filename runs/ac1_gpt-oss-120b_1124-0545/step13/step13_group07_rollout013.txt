def _project_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _l2_flatten_optimize(start_vec: np.ndarray, n: int, deadline: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Gradient descent (projected onto simplex) to flatten the convolution
    distribution towards uniform.   Returns best sequence seen and its surrogate.
    """
    a = start_vec.copy()
    eps = 1e-12
    best_seq = a.copy()
    best_val = _surrogate_obj(a)   # evaluate? we track max conv.
    avg = 1.0 / (2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        # compute gradient of L2 loss
        diff = conv - avg
        # convolution with reversed a
        grad = 2.0 * _conv_fft(diff, a[::-1])[n-1:n-1+n]

        # step size schedule
        lr = 0.1 / np.sqrt(it)   # initial step small
        a_new = a - lr * grad
        # ensure nonnegative and project onto simplex
        a_new = np.maximum(a_new, 0.0)
        # project onto simplex
        if a_new.sum() > 0:
            a = _project_simplex(a_new)
        else:
            a = _uniform(n)

        # evaluate surrogate
        cur_val = _surrogate_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # optionally print sometimes
            if it % 50000 == 0:
                print(f"[L2] iter {it:7d} best={cur_val:.6f}")

    return best_seq, best_val

Now define _project_simplex:

def _project_simplex(v):
    """Projects a vector onto the probability simplex."""
    # reference algorithm
    if v.ndim != 1:
        raise ValueError("v must be 1-D")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    # avoid division by zero at boundaries
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

"""
Propose a step‑function (non‑negative sequence) that minimizes the official
``evaluate_sequence`` objective.

Algorithm
---------
0️⃣  Create a diverse pool of ℓ₁‑normalised candidate vectors.
    In addition to the families used before we add an *arcsine* shape
    (the discrete analogue of the Beta(½,½) density) which is known to
    produce a nearly flat self‑convolution.

1️⃣  Keep the *top_k* seeds with the smallest surrogate objective
    2·n·max(conv(a,a))   (the denominator Σa is fixed to 1).

2️⃣  For each seed we run two optimisation phases:

   • **L2‑flattening** – minimise the squared error
        ‖a∗a – 1/(2n‑1)·1‖₂²
     using projected gradient descent on the probability simplex.
     This drives the convolution distribution towards uniformity,
     which also lowers its maximum entry.

   • **Maximum‑surrogate optimisation** – Adam optimisation on a
     log‑sum‑exp (smooth max) surrogate of max(conv).  Temperature β is
     increased gradually up to 3000, making the surrogate arbitrarily
     close to the true max.

   After each phase we keep the best vector seen (lowest max‑convolution).

3️⃣  **Mirror‑descent (exact sub‑gradient)** – a few passes of exponentiated‑gradient
    descent on the exact max (as in the reference code) are applied to the
    current best vector.

4️⃣  **Simulated‑annealing polishing** – a lightweight SA that swaps,
    redistributes or transfers mass between random positions for the
    remaining time.

5️⃣  Finally we evaluate the best sequence with ``evaluate_sequence``,
    store it globally as ``height_sequence_1`` and return it as a plain list.

All heavy work is performed with FFT‑based convolutions (O(n log n)).
The length of the sequence is set to 8000 – large enough to give a
fine‑grained search while remaining comfortably within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are the same → uniform vector
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety
    w_sum = w.sum()
    if w_sum <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / w_sum


# ----------------------------------------------------------------------
# Surrogate objective (fast, assumes Σa = 1)
# ----------------------------------------------------------------------
def _surrogate_obj(seq: np.ndarray) -> float:
    """Fast surrogate: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random 3‑segment piece‑wise linear shape."""
    breaks = np.sort(rng.choice(n, size=3, replace=False))
    heights = rng.random(3) + 0.01
    x = np.arange(n, dtype=np.float64)
    a = np.interp(x, breaks, heights)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few randomly positioned Gaussians."""
    a = np.zeros(n, dtype=np.float64)
    num = rng.integers(2, 6)
    for _ in range(num):
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.02 * n, 0.15 * n)
        x = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, nf: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth curve built from a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, nf + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _arcsine_shape(n: int) -> np.ndarray:
    """
    Discrete analogue of the Beta(½,½) density:
        a_i ∝ 1 / sqrt( (i+½)·(n‑½‑i) )
    """
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))
    seeds.append(_arcsine_shape(n))

    # parameter sweeps
    for sigma in np.linspace(0.05, 0.35, 8):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 4.5, 10):
        seeds.append(_power_law(n, p))

    alphas = np.linspace(0.2, 3.0, 9)
    betas = np.linspace(0.2, 3.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for lam in np.linspace(0.0005, 0.015, 10):
        seeds.append(_exponential_decay(n, lam))

    # random Dirichlet draws
    concs = np.logspace(-4, 0.5, 12)   # 1e‑4 … ~3.2
    for conc in concs:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixtures of random Gaussians
    for _ in range(5):
        seeds.append(_gaussian_mixture(n, rng))

    # piece‑wise‑linear random shapes
    for _ in range(5):
        seeds.append(_piecewise_linear(n, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 5, 7):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0.0:
            seeds.append(avg / avg.sum())

    return seeds

# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray | None, float]:
    """Adam on log‑softmax parameters with a log‑sum‑exp surrogate of max."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.9
    beta = 0.5          # initial temperature
    beta_inc = 0.5
    beta_max = 3000.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = np.exp(v - np.max(v))          # unnormalised softmax
        a = a / a.sum()

        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # smooth max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # raise temperature
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _l2_flatten_optimize(start_vec: np.ndarray, n: int, deadline: float,
                         max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the L2 loss
        ‖conv(a,a) – 1/(2n‑1)·1‖₂² .
    Returns the best sequence seen (lowest max‑convolution).
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)
    avg = 1.0 / (2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        diff = conv - avg                           # length 2n‑1
        # gradient of ½‖diff‖₂² w.r.t a: 2 * (diff ∗ a_rev) on the overlap region
        grad = 2.0 * _conv_fft(diff, a[::-1])[n - 1:n - 1 + n]

        # simple diminishing step size
        lr = 0.2 / np.sqrt(it)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a = _project_simplex(a_new)

        cur_val = _surrogate_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[L2] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror) sub‑gradient descent on the exact max."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    lr0 = 0.25
    power = 0.28

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / ((it) ** power)
        a = a * np.exp(-lr * g)
        a = _project_simplex(a)

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int, deadline: float,
                                 max_iters: int = 2_500_000,
                                 init_temp_factor: float = 0.06,
                                 min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """Lightweight SA with three move types."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate_obj(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9997, min_temp)

        move_type = rng.integers(0, 3)
        if move_type == 0:
            # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            # redistribute mass on a small random subset
            k = rng.integers(2, 14)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total == 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            # pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total == 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _surrogate_obj(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 200_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the official
    ``evaluate_sequence`` objective.  Returns the best sequence as a list of
    floats and registers it globally as ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # problem size – larger n gives a finer search space
    n = 8000

    # load any previous solution (if present)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size != n:
                if previous.size > n:
                    previous = previous[:n]
                else:
                    previous = np.pad(previous,
                                      (0, n - previous.size),
                                      constant_values=0.0)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – broad seeding
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _surrogate_obj(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _surrogate_obj(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – L2 flattening + max‑surrogate Adam + mirror descent
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        print(f"\n--- Optimising seed {idx}/{len(best_seeds)} ---")
        # L2 flattening
        seq_l2, val_l2 = _l2_flatten_optimize(
            seed_vec, n, deadline, max_iters=120_000
        )
        print(f"[Seed {idx}] after L2 flattening: surrogate = {val_l2:.6f}")

        # Adam on smooth max
        seq_adam, val_adam = _adam_optimize(
            seq_l2, n, deadline, max_iters=120_000
        )
        print(f"[Seed {idx}] after Adam: surrogate = {val_adam:.6f}")

        # Mirror‑descent refinement
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            seq_adam, n, deadline, max_iters=800_000
        )
        print(f"[Seed {idx}] after Mirror: surrogate = {val_mirror:.6f}")

        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall] New global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Simulated‑annealing polishing (uses remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing (final) ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline, max_iters=3_000_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall-SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (for information only)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
