"""
A compact alternative to the previous optimisation pipeline.

Idea
----
* Start from the best known solution (global variable ``height_sequence_1``) or,
  if unavailable, fall back to a simple uniform distribution.
* Repeatedly improve the sequence by solving a small linear programme:
    maximise Σ g_i  subject to   (f * g)_ℓ ≤ M  for all ℓ,
  where ``M = max(f * f)``.  The LP yields a direction ``g`` whose
  convolution with ``f`` stays below the current worst peak.
* Mix the current sequence ``f`` with ``g`` (several step‑sizes are tried)
  and keep the mixture that most reduces the exact objective
  ``2·n·max(conv(f,f))`` (the same value returned by ``evaluate_sequence``).
  Because the objective is scale‑invariant we always renormalise to unit‑ℓ₁.
* After a few LP‑steps we run a cheap stochastic “mass‑transfer” hill‑climb
  to polish the result.
* The whole routine respects the 1000 s budget and stores the best
  vector globally as ``height_sequence_1`` for warm‑starts.

The LP is tiny: only the lags that are currently active (starting with the
lag achieving the maximal convolution) are kept as constraints.  When the
candidate ``g`` violates any previously ignored lag we add the offending lag
to the active set and resolve – a classic cutting‑plane scheme.
"""

import time
import numpy as np
import scipy.optimize as opt
from scipy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low level convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective_exact(a: np.ndarray) -> float:
    """Exact (scale‑invariant) objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Linear‑programming step (active‑set cutting‑plane)
# ----------------------------------------------------------------------
def _lp_improve(f: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    One LP‑based improvement step.
    Returns a new vector (renormalised to sum 1) and its objective value.
    """
    n = len(f)
    # current worst convolution height
    conv_ff = _conv_fft(f, f)
    M = np.max(conv_ff)
    lag_max = int(np.argmax(conv_ff))

    # active set initially contains the lag that actually attains the max
    active_set = [lag_max]

    # --------------------------------------------------------------
    # Solve the LP with a growing active set until the solution is feasible
    # --------------------------------------------------------------
    while True:
        K = len(active_set)
        # Build a dense (K × n) matrix A where A[row, col] = f_i for the
        # appropriate coefficient in (f * g)_lag.
        A = np.zeros((K, n), dtype=np.float64)
        for r, lag in enumerate(active_set):
            i_start = max(0, lag - (n - 1))
            i_end = min(n - 1, lag)
            # slice of f that participates
            f_sub = f[i_start:i_end + 1]
            # corresponding column range for g
            j_start = lag - i_end
            A[r, j_start:j_start + f_sub.size] = f_sub[::-1]

        # Linear programme: maximise Σ g_i  <=> minimise -Σ g_i
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = opt.linprog(
            c=c,
            A_ub=A,
            b_ub=np.full(K, M, dtype=np.float64),
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # should be very rare; exit the LP loop
            break

        g = res.x

        # Check whether (f * g) respects *all* lag constraints
        conv_fg = _conv_fft(f, g)
        viol = np.where(conv_fg > M * (1.0 + 1e-9))[0]
        if viol.size == 0:
            # feasible – we can stop adding constraints
            break

        # Add a few of the worst violating lags to the active set
        for v in viol[:3]:         # limit growth per iteration
            if v not in active_set:
                active_set.append(v)

        # If we are running out of time, abort early
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # Mix the current f with the LP solution g.
    # Try several mixing parameters and keep the best.
    # --------------------------------------------------------------
    best_vec = f
    best_obj = _objective_exact(f)
    # candidate mixing coefficients
    for t in (0.1, 0.3, 0.5, 0.7, 1.0):
        cand = (1.0 - t) * f + t * g
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s
        obj = _objective_exact(cand)
        if obj < best_obj:
            best_obj = obj
            best_vec = cand
    return best_vec, best_obj


# ----------------------------------------------------------------------
# Simple stochastic local search (mass‑transfer hill climbing)
# ----------------------------------------------------------------------
def _mass_transfer_local_search(
    start: np.ndarray,
    deadline: float,
    iters: int = 150_000,
    frac_start: float = 0.15,
    frac_end: float = 0.01,
) -> tuple[np.ndarray, float]:
    """
    Classical hill‑climbing: repeatedly move a small fraction of mass from
    one index to another.  The step size decays linearly.
    """
    n = len(start)
    a = start.copy()
    best = a.copy()
    best_val = _objective_exact(a)
    rng = np.random.default_rng(314159)

    for it in range(iters):
        if time.time() > deadline:
            break
        # linearly decreasing fraction of mass to move
        frac = frac_start + (frac_end - frac_start) * (it / iters)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best = cand.copy()
            a = cand  # keep the improvement for the next step
    return best, best_val


# ----------------------------------------------------------------------
# Main entry point -----------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Uses a cutting‑plane linear programme followed by a stochastic local search.
    The best vector found is stored globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a tiny safety margin

    # --------------------------------------------------------------
    # 1. Load previous best (if any) or initialise a uniform seed.
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            a0 = np.asarray(previous, dtype=np.float64)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 0.0:
                a0 = a0 / s
            else:
                a0 = None
        except Exception:
            a0 = None
    else:
        a0 = None

    # Problem dimension – keep the legacy size (good trade‑off)
    n = 4000

    if a0 is None or a0.size != n:
        # Uniform distribution is a safe fall‑back
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # Ensure a proper probability vector
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()

    # Keep the globally best result
    best_vec = a0.copy()
    best_val = _objective_exact(best_vec)
    print(f"[Init] objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Repeated LP improvement steps (cutting‑plane)
    # --------------------------------------------------------------
    max_lp_iters = 12
    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break
        best_vec, new_val = _lp_improve(best_vec, deadline)
        if new_val < best_val - 1e-9:      # genuine improvement
            best_val = new_val
            print(f"[LP {lp_it+1}] improved objective = {best_val:.6f}")
        else:
            print(f"[LP {lp_it+1}] no improvement – stopping LP phase")
            break

    # --------------------------------------------------------------
    # 3. Fine‑grained stochastic local search (mass transfers)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate a fraction of the remaining budget to hill‑climbing
        iters = int(min(250_000, remaining * 180))
        print(f"[LocalSearch] running {iters} hill‑climb iterations")
        best_vec, best_val = _mass_transfer_local_search(
            best_vec, deadline, iters=iters, frac_start=0.12, frac_end=0.008
        )
        print(f"[LocalSearch] best objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final official evaluation (for logging only)
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_vec.tolist())
        print(f"Final evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("evaluate_sequence raised an exception:", exc)

    # Store the result for warm‑starts in later runs
    globals()["height_sequence_1"] = best_vec.tolist()
    return best_vec.tolist()
