def exp_gradient_opt(start_seq, n, deadline, max_iters):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = objective_surrogate(a)
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute conv
        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))
        # compute subgradient
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        eta = 0.5 / math.sqrt(t)
        log_a -= eta * g
        # softmax to stay on simplex
        a = _softmax(log_a)
        cur_val = objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        if t % 200000 == 0:
            print(f"[ExpGrad] iter {t:,}, best {best_val:.6f}")
    return best_seq, best_val

"""
Improved low‑autocorrelation step‑function search.

The algorithm is a cheap random‑restart sub‑gradient (exponentiated‑gradient)
optimizer with occasional smoothing (L‑BFGS on a log‑sum‑exp surrogate) and a
final swap‑based hill‑climber.  It works on a moderate problem size
(`n≈2000`) which allows many optimisation iterations within the 1000 s budget.
The optimisation loop repeatedly draws a new random Dirichlet seed (or a
perturbed copy of the current best), runs a short L‑BFGS warm‑start followed
by a long exact sub‑gradient descent, and keeps the best sequence seen.
A final permutation‑swap local search is applied to polish the result.

All helper functions are top‑level; no closures or lambdas are used.
"""

import time
import math
import numpy as np

# optional SciPy – only needed for the L‑BFGS warm‑start
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
    2·n·max(conv(seq,seq)).
    The sequence is assumed to be ℓ₁‑normalised (sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Desc‑sorted Dirichlet values placed alternately left/right."""
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(α,…,α)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# L‑BFGS smooth optimisation (optional)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """Minimise the log‑sum‑exp smoothed max‑convolution objective."""
    if _opt is None:
        # SciPy not available – fall back to the start vector unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6
        ]

    # work in log‑probability space (softmax enforces the simplex)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # ℓ₁‑normalised
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smoothed max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # interrupt L‑BFGS when we exceed the deadline
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser (exponentiated gradient)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient for the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size 0.5 / sqrt(t)  (standard diminishing schedule)
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 100_000 == 0:
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑swap hill‑climber
# ----------------------------------------------------------------------
def _perm_swap_opt(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_swaps: int = 300_000) -> tuple[np.ndarray, float]:
    """Randomly swap two positions; keep swap only if it improves the surrogate."""
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # perform swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_val = _objective_surrogate(cur_seq)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if it % 100_000 == 0:
            print(f"[PermSwap] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_initial_vectors(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a modest diverse pool of ℓ₁‑normalised candidates.
    """
    seeds = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.2))
    seeds.append(_beta_shape(n, 0.5, 0.5))
    seeds.append(_beta_shape(n, 2.0, 2.0))
    seeds.append(_alternating_dirichlet(n, rng))

    # Random Dirichlet draws with several concentrations
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Heavy‑tailed perturbations of the previous best (if any)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n:
                seeds.append(prev_arr / prev_arr.sum())
        except Exception:
            pass

    # Ensure all seeds are proper probability vectors
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence found as a plain Python list.
    """
    # ------------------------------------------------------------------
    # 0. Setup RNG and timing
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # a small safety margin so we stay inside the time budget
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Problem size – 2000 is a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 2000
    print(f"\n=== Starting optimisation (n={n}) ===")

    # ------------------------------------------------------------------
    # 2. Build an initial pool of seeds
    # ------------------------------------------------------------------
    seeds = _collect_initial_vectors(n, rng)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # 3. Quick evaluation – keep the best few
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    best_seed, best_val = evaluated[0]
    print(f"Best seed surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop (random restarts + EG)
    # ------------------------------------------------------------------
    global_best_seq = best_seed.copy()
    global_best_val = best_val

    # a short L‑BFGS warm‑start for the initial seed
    if time.time() < deadline:
        print("\n[Phase] L‑BFGS warm‑start on the best seed")
        try:
            seq_lbfgs = _lbfgs_smooth_optimize(global_best_seq,
                                               n,
                                               deadline,
                                               max_iter_per_beta=8000)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
            print(f"[L‑BFGS] surrogate = {val_lbfgs:.6f}")
            if val_lbfgs < global_best_val:
                global_best_seq = seq_lbfgs
                global_best_val = val_lbfgs
        except Exception as e:  # pragma: no cover
            print("[L‑BFGS] failed:", e)

    # --------------------------------------------------------------
    # 4b. Repeated random‑restart sub‑gradient descent
    # --------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 30.0:   # keep ~30 s for the final swap phase
        remaining = deadline - time.time()
        restart += 1
        print(f"\n--- Restart #{restart} (remaining {remaining:.1f}s) ---")

        # pick a seed: either a fresh Dirichlet draw or a perturbed copy of
        # the current best (to escape possible local minima)
        if rng.random() < 0.6 or global_best_seq is None:
            conc = rng.choice([0.05, 0.2, 0.5, 1.0, 2.0])
            seed_vec = _random_dirichlet(n, conc, rng)
        else:
            # perturb the current best in log‑space
            log_best = np.log(global_best_seq + 1e-30)
            noise = rng.normal(scale=0.12, size=n)
            seed_vec = _softmax(log_best + noise)

        # optional quick L‑BFGS warm‑start (few iterations) to get into a
        # decent basin before the exact sub‑gradient phase
        if _opt is not None and time.time() < deadline - 10.0:
            seq_pre = _lbfgs_smooth_optimize(seed_vec,
                                             n,
                                             deadline,
                                             beta_schedule=[1.0, 5.0, 20.0, 80.0, 200.0],
                                             max_iter_per_beta=3000)
        else:
            seq_pre = seed_vec

        # exact exponentiated‑gradient descent
        max_iters = int(min(300_000, (deadline - time.time()) * 0.55))
        max_iters = max(150_000, max_iters)   # guarantee a decent amount of work
        seq_opt, val_opt = _exp_gradient_opt(seq_pre,
                                             n,
                                             deadline,
                                             max_iters=max_iters)

        print(f"[Restart #{restart}] EG best surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_seq = seq_opt.copy()
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final permutation‑swap local search
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        max_swaps = int(min(400_000, remaining * 400))
        max_swaps = max(150_000, max_swaps)
        print("\n=== Final permutation‑swap refinement ===")
        seq_final, val_final = _perm_swap_opt(global_best_seq,
                                              n,
                                              deadline,
                                              max_swaps=max_swaps)
        if val_final < global_best_val:
            global_best_val = val_final
            global_best_seq = seq_final
            print(f"[Final] Improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n*** Finished ***\nSurrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {official_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    # store for possible future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
