"""
Low‑autocorrelation step‑function search.

The algorithm works in four stages (all respecting a 1000 s global budget):

1. **Seed generation** – a very diverse pool of ℓ¹‑normalised vectors is built.
   Analytic families (Gaussian, sine‑window, power‑law, beta‑shape, etc.) are
   scanned on a fine grid, and many random Dirichlet / exponential draws are
   added.  The cheap surrogate  2·n·max(conv)  is evaluated for every seed and
   the best few are kept.

2. **Continuation L‑BFGS** – each top seed is refined with L‑BFGS‑B on a
   smooth‑max surrogate (log‑sum‑exp with an increasing β schedule).  The
   optimisation works in log‑probability space and always stays on the
   probability simplex via a softmax.

3. **Exact max‑gradient polishing** – the best vector from step 2 is then
   polished with a long exponentiated‑gradient (mirror‑descent) on the *exact*
   max‑convolution objective.  The step‑size follows a 1/√t schedule and the
   loop runs for as many iterations as the remaining budget permits.

4. **Simple re‑ordering post‑process** – the multiset of weights is kept but
   the order may be changed.  A small collection of deterministic permutations
   (sorted descending, reversed, a bit‑reversal order) is tested and the one
   with the smallest surrogate is returned.

The final sequence is stored as ``height_sequence_1`` and also returned as a
list of Python floats.  All numbers are non‑negative, the sum is 1.0, and the
search never exceeds the allotted time.
"""

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that maps ℝⁿ → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate for ℓ¹‑normalised vectors:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ¹‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s


# ----------------------------------------------------------------------
# Seed collection (fine‑gridded analytic families + random draws)
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # --- Analytic families ------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Sine window – dense grid on gamma
    for gamma in np.linspace(0.5, 4.0, 15):
        seeds.append(_sine_window(n, gamma=gamma))

    # Gaussian – fine grid on sigma_frac
    for sigma in np.linspace(0.10, 0.45, 20):
        seeds.append(_gaussian(n, sigma))

    # Power law – dense grid on p
    for p in np.linspace(0.0, 5.0, 26):
        seeds.append(_power(n, p))

    # Beta‑shape – moderate grid on alpha/beta
    alphas = np.linspace(0.2, 5.0, 9)
    betas = np.linspace(0.2, 5.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sparse random vectors
    for k_frac in (0.005, 0.01, 0.02):
        k = max(1, int(round(k_frac * n)))
        for _ in range(2):
            seeds.append(_sparse_random(n, k, rng))

    # Block‑uniform vectors
    for bf in (0.03, 0.07, 0.15, 0.25):
        for _ in range(2):
            seeds.append(_block_uniform(n, bf, rng))

    # Previous solution (if any) – normalised to ℓ¹
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Ensure all seeds are proper ℓ¹‑vectors
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _lbfgs_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule=None,
                    max_iter_per_beta: int = 6000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Works in log‑probability space and returns an ℓ¹‑normalised vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 2e3, 1e4,
            5e4, 2e5, 1e6, 2e6
        ]

    # start in log‑space (avoid log(0))
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # ℓ¹‑normalised
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # log‑sum‑exp smooth max
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (see derivation in original code)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # stop early if we run out of time
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at beta={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact max‑gradient polishing (exponentiated‑gradient / mirror descent)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size schedule (slightly more aggressive than 0.5/√t)
        eta = 0.7 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 200_000 == 0:
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

def _bitrev_permutation(n: int) -> np.ndarray:
    """Return a bit‑reversal order for the first n indices (n assumed ≤2^k)."""
    k = (n - 1).bit_length()
    rev = np.arange(n, dtype=int)
    for i in range(k):
        rev = ((rev & 1) << (k - 1)) | (rev >> 1)
    return rev


def _best_permutation_variant(base_seq: np.ndarray,
                              n: int,
                              deadline: float) -> np.ndarray:
    """
    Try a handful of deterministic permutations of the same multiset
    (sorted descending, reversed, bit‑reversal) and keep the one with the
    smallest surrogate value.
    """
    variants = []

    # 1. original ordering
    variants.append(base_seq)

    # 2. descending order of weights
    desc = np.sort(base_seq)[::-1]
    variants.append(desc)

    # 3. ascending order (the reverse of descending)
    asc = np.sort(base_seq)
    variants.append(asc)

    # 4. bit‑reversal order – fill via the sorted descending values
    br = _bitrev_permutation(n)
    br_seq = np.empty_like(base_seq)
    br_seq[br] = desc
    variants.append(br_seq)

    # 5. random shuffle (single attempt, deterministic seed)
    rng = np.random.default_rng(42)
    rnd = base_seq.copy()
    rng.shuffle(rnd)
    variants.append(rnd)

    best_seq = base_seq
    best_val = _objective_surrogate(base_seq)

    for cand in variants:
        if time.time() > deadline:
            break
        val = _objective_surrogate(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    return best_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence as a list of floats.
    """
    # --------------------------------------------------------------
    # 0. Initialise RNG and timing
    # --------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0            # safety margin

    # --------------------------------------------------------------
    # 1. Problem size – 4000 is a good trade‑off, 4096 (power‑of‑2) is fast
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # 2. Gather seeds
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seed vectors.")

    # --------------------------------------------------------------
    # 3. Quick surrogate evaluation, keep the elite
    # --------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print("\nTop seed surrogate values (pre‑opt):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # --------------------------------------------------------------
    # 4. Continuation L‑BFGS on elite seeds
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
        2e3, 1e4, 5e4, 2e5, 1e6, 2e6
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break

        # allocate a few seconds per seed (adaptively)
        remaining = deadline - time.time()
        max_iter_per_beta = int(min(8000, remaining * 0.03))
        max_iter_per_beta = max(5000, max_iter_per_beta)

        print(f"\n--- L‑BFGS on seed #{idx+1} (max_iter_per_beta={max_iter_per_beta}) ---")
        seq_opt = _lbfgs_optimize(seed_vec,
                                   n,
                                   deadline,
                                   beta_schedule=beta_schedule,
                                   max_iter_per_beta=max_iter_per_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Exact max‑gradient polishing (exponentiated gradient)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp = int(min(1_200_000, remaining * 0.70))
        max_exp = max(600_000, max_exp)
        print(f"\n--- Exponentiated‑gradient polishing (max_iters={max_exp}) ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                             n,
                                             deadline,
                                             max_iters=max_exp)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Simple permutation post‑processing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation post‑processing ---")
        seq_perm = _best_permutation_variant(global_best_seq, n, deadline)
        val_perm = _objective_surrogate(seq_perm)
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[Perm] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Final official evaluation
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
