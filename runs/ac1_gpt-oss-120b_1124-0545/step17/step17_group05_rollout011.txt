    best_seq = seq.copy()

    while time.time() < deadline:

        i = rng.integers(0,n)

        # propose transfer amount

"""
Low‑autocorrelation step‑function search.

1. Build a large, diverse pool of ℓ₁‑normalised candidates:
   analytic windows, power‑law, Gaussian, double‑Gaussian,
   Beta‑shapes, exponential decays, random Dirichlet / exponential draws,
   low‑frequency cosines, sparse / block‑uniform vectors and mixtures.
2. Evaluate the cheap surrogate 2·n·max(conv) for all seeds,
   keep the best few (≈15).
3. For each kept seed run a mirror‑descent (exponentiated‑gradient)
   on the *exact* max‑convolution objective for up to a few‑hundred‑k
   iterations, keeping the best result seen.
4. Polish the current global best with a longer mirror‑descent run.
5. Perform a cheap mass‑transfer hill‑climber (move a tiny amount of
   mass from one index to another) for a while.
6. Finish with a random‑swap permutation hill‑climber.
All steps respect the global 1000 s budget and use only NumPy
(‑ no external binaries).  The returned list is a non‑negative
step‑function whose official score `evaluate_sequence` is expected
to be < 1.503.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Softmax (maps ℝⁿ → probability simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors of length n)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block of length block_frac·n."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a / a.sum()

def _exp_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)"""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_rev(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·(n‑1‑i))  (mirror of _exp_decay)"""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed construction
# ----------------------------------------------------------------------
def _collect_all_seeds(n: int, rng: np.random.Generator,
                       previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a huge, diverse list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families ----------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for g in (0.3, 0.5, 1.0, 2.0, 4.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))

    for beta in (2, 4, 8, 12, 16, 20):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    # more fine‑grained Gaussian sweep
    for sigma in np.linspace(0.08, 0.20, 5):
        seeds.append(_gaussian(n, sigma))

    # Double Gaussian grid (σ, shift)
    for sigma in np.linspace(0.08, 0.20, 5):
        for shift in np.linspace(0.15, 0.35, 5):
            seeds.append(_double_gaussian(n, sigma, shift))

    # Power‑law sweep
    for p in np.linspace(0.0, 10.0, 21):
        seeds.append(_power(n, p))

    # Beta‑shape finer grid
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 17):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sparse random vectors (different sparsities)
    for k in (max(1, int(0.005 * n)), max(1, int(0.02 * n)), max(1, int(0.05 * n))):
        for _ in range(2):
            seeds.append(_sparse_random(n, k, rng))

    # Block‑uniform vectors (different block fractions)
    for bf in (0.03, 0.07, 0.15, 0.25):
        for _ in range(2):
            seeds.append(_block_uniform(n, bf, rng))

    # Exponential decay families
    for lam in (0.0005, 0.001, 0.002, 0.005, 0.01, 0.02):
        seeds.append(_exp_decay(n, lam))
        seeds.append(_exp_decay_rev(n, lam))

    # Alternating Dirichlet
    seeds.append(_alternating_dirichlet(n, rng))

    # Mixtures of random pairs (adds extra diversity)
    base_for_mix = seeds.copy()
    for _ in range(30):
        a, b = rng.choice(base_for_mix, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(a, b, alpha))

    # Previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise & sanity‑check
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # compute full convolution and locate active lag
        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient for the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size schedule (1/√t style, scaled a bit larger)
        eta = 0.6 / math.sqrt(it)

        # mirror‑descent update in log‑space (KL‑projection = softmax)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 100_000 == 0:
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Swap‑based permutation hill‑climber
# ----------------------------------------------------------------------
def _perm_swap_opt(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_swaps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Randomly swap two positions; keep the swap if the surrogate improves.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _objective_surrogate(cur_seq)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if it % 75_000 == 0:
            print(f"[PermSwap] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Mass‑transfer hill‑climber (move a tiny fraction of mass)
# ----------------------------------------------------------------------
def _mass_transfer_opt(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 300_000,
                       max_eps: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Pick two indices i→j and move a small fraction of a_i to a_j.
    Accept only if the surrogate improves.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i = rng.integers(0, n)
        if cur_seq[i] <= 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1  # ensure i != j without extra check
        eps = rng.random() * max_eps
        delta = eps * cur_seq[i]

        # perform transfer
        cur_seq[i] -= delta
        cur_seq[j] += delta

        new_val = _objective_surrogate(cur_seq)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert
            cur_seq[i] += delta
            cur_seq[j] -= delta

        if it % 75_000 == 0:
            print(f"[MassTransfer] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Build a huge, diverse seed pool
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_arr = arr
        except Exception:
            prev_arr = None

    seeds = _collect_all_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} candidate seeds.")

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation, keep the best few
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 15
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Mirror‑descent (exact max) on each top seed
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] budget exhausted before finishing seed refinements.")
            break

        remaining = deadline - time.time()
        # Allocate a modest iteration budget per seed (≈150 k if time permits)
        max_iters = int(min(150_000, max(20_000, remaining * 40)))
        print(f"\n--- Exp‑Gradient on seed #{idx+1} (max_iters={max_iters}) ---")
        seq_opt, val_opt = _exp_gradient_opt(seed_vec, n, deadline, max_iters=max_iters)
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Extended refinement on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        extra_iters = int(min(600_000, max(200_000, remaining * 50)))
        print(f"\n--- Extra long Exp‑Gradient on global best (iters={extra_iters}) ---")
        seq_opt, val_opt = _exp_gradient_opt(global_best_seq, n, deadline, max_iters=extra_iters)
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Extra] improved global best to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Mass‑transfer hill‑climbing (tiny local moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        mt_iters = int(min(400_000, max(100_000, remaining * 40)))
        print(f"\n--- Mass‑Transfer hill climb (iters={mt_iters}) ---")
        seq_mt, val_mt = _mass_transfer_opt(global_best_seq, n, deadline,
                                            max_iters=mt_iters, max_eps=0.015)
        if val_mt < global_best_val:
            global_best_seq = seq_mt.copy()
            global_best_val = val_mt
            print(f"[MassTransfer] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final permutation‑swap local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        swaps = int(min(500_000, max(150_000, remaining * 400)))
        print(f"\n--- Final permutation‑swap search (swaps={swaps}) ---")
        seq_perm, val_perm = _perm_swap_opt(global_best_seq, n, deadline,
                                            max_swaps=swaps)
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[PermSwap] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation and return
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
