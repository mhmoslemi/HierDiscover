"""
Propose a step‑function (non‑negative heights) minimising

    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------

1. **Rich seed generation** – generate many analytic windows (including a
   Chebyshev window not used before), dense sweeps of their parameters,
   random Dirichlet draws, exponential draws and low‑frequency cosine
   curves.  The best `top_k` candidates (according to the exact surrogate
   2·n·max(conv) for ℓ₁‑normalised vectors) are kept in a heap.

2. **Mirror‑descent polishing** – start from the best seed and run
   exponentiated‑gradient mirror descent on the exact max‑convolution
   objective (sub‑gradient of the active entry).  This phase receives the
   bulk of the time budget.

3. **Continuation L‑BFGS** – smooth the max with a log‑sum‑exp
   surrogate (temperature β) and optimise the smooth objective with
   L‑BFGS‑B, increasing β on a schedule up to 5·10⁵.  The optimisation lives
   in log‑softmax space, guaranteeing non‑negativity and the simplex
   constraint.

4. **Order‑optimisation** – the objective depends on the order of the
   coefficients.  A fast greedy “swap‑search” repeatedly swaps two
   positions and keeps the change if it reduces the surrogate.  This can
   lower the peak of the convolution further.

5. **Local mass‑transfer hill‑climb** – final tiny perturbations that move
   a small fraction of mass from one coordinate to another, accepting only
   improving moves.

The best sequence is stored globally as ``height_sequence_1`` for reuse.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N).real[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a vector normalised to sum = 1:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Modified Bessel function I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cheb_window(n: int, attenuation: float) -> np.ndarray:
    if chebwin is None:
        return _uniform(n)
    a = chebwin(n, at=attenuation)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random generators (return ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet with all concentration parameters = conc."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # learning rate schedule η = η₀ / sqrt(t)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 15_000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    The optimiser works in log‑softmax space (v), guaranteeing a simplex
    solution.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0, 5_000.0,
                         10_000.0, 20_000.0, 50_000.0,
                         100_000.0, 200_000.0, 500_000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # on simplex
            conv = _conv_fft(a, a)            # full convolution
            max_c = np.max(conv)
            # smooth approximation of max via log‑sum‑exp
            shift = max_c                     # avoid overflow
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp            # soft‑max weights

            # gradient wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Order optimisation – greedy swapping of positions
# ----------------------------------------------------------------------
def _reorder_greedy(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy search that swaps two positions if it reduces the surrogate.
    Returns the best ordering found.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        new_val = _objective_surrogate(new)
        if new_val < cur_val:
            cur, cur_val = new, new_val
            if new_val < best_val:
                best, best_val = new.copy(), new_val
    return best, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing / simulated‑annealing local refinement
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer moves on the simplex.  Each step picks two
    indices i,j and moves a random fraction of a[i] to a[j].
    Accept only improving moves.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        eps = cur[i] * rng.uniform(0.001, 0.01)   # move 0.1‑1 % of mass
        if eps <= 0.0 or cur[i] - eps < 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective_surrogate(new)
        if new_val < cur_val:
            cur, cur_val = new, new_val
            if new_val < best_val:
                best, best_val = new, new_val
    return best, best_val

# Main candidate proposer

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The algorithm respects the time budget and
    stores the best found sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # reserve a few seconds for the final call to evaluate_sequence
    deadline = start_time + budget_s - 3.0

    # problem size – 4000 gives a good trade‑off between quality and speed
    n = 4000

    # ------------------------------------------------------------------
    # helper for the heap of best seeds
    # ------------------------------------------------------------------
    top_k = 20
    best_heap = []   # stores (-value, vector)

    def _push_candidate(vec: np.ndarray):
        """Insert a candidate into the heap maintaining the best ``top_k``."""
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if val < -best_heap[0][0]:
                heapq.heappushpop(best_heap, (-val, vec))

    # ------------------------------------------------------------------
    # deterministic analytic seeds (dense parameter sweeps)
    # ------------------------------------------------------------------
    _push_candidate(_uniform(n))
    _push_candidate(_triangular(n))

    # sine window – finer grid
    for gamma in np.linspace(0.1, 6.0, 36):
        _push_candidate(_sine_window(n, gamma))

    # Gaussian – sigma fraction
    for sigma in np.linspace(0.04, 0.30, 27):
        _push_candidate(_gaussian(n, sigma))

    # Power law
    for p in np.linspace(0.0, 5.0, 26):
        _push_candidate(_power(n, p))

    # Beta shape – broader range
    alphas = np.geomspace(0.05, 20.0, 10)
    betas = np.geomspace(0.05, 20.0, 10)
    for a in alphas:
        for b in betas:
            _push_candidate(_beta_shape(n, a, b))

    # Tukey window
    for alpha in np.linspace(0.0, 1.0, 12):
        _push_candidate(_tukey_window(n, alpha))

    # Kaiser window – many β values
    for beta in np.linspace(0.5, 20.0, 30):
        _push_candidate(_kaiser_window(n, beta))

    # Chebyshev window – attenuation dB
    if chebwin is not None:
        for att in [30, 45, 60, 80, 100, 120, 140]:
            _push_candidate(_cheb_window(n, att))

    # Exponential decay shapes
    for lam in np.linspace(0.0005, 0.02, 25):
        _push_candidate(_exponential_decay(n, lam))

    # Double Gaussian (few combos)
    for sigma in (0.08, 0.12, 0.16):
        for shift in (0.18, 0.25, 0.32):
            _push_candidate(_double_gaussian(n, sigma, shift))

    # ------------------------------------------------------------------
    # Massive random seeding (Dirichlet, exponential, low‑freq curves)
    # ------------------------------------------------------------------
    conc_choices = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    random_seed_budget = 4000
    for _ in range(random_seed_budget):
        if time.time() > deadline - 60.0:
            break
        conc = rng.choice(conc_choices)
        vec = _random_dirichlet(n, conc, rng)
        # random cyclic shift adds diversity
        shift = rng.integers(n)
        if shift:
            vec = np.roll(vec, shift)
        _push_candidate(vec)

    # exponential draws
    for _ in range(500):
        if time.time() > deadline - 60.0:
            break
        vec = _random_exponential(n, rng)
        shift = rng.integers(n)
        if shift:
            vec = np.roll(vec, shift)
        _push_candidate(vec)

    # low‑frequency curves
    for nf in (2, 3, 5, 7):
        for _ in range(200):
            if time.time() > deadline - 60.0:
                break
            vec = _lowfreq_curve(n, nf, rng)
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
            _push_candidate(vec)

    # ------------------------------------------------------------------
    # Re‑use previous solution if available
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0:
                prev_norm = prev_arr / prev_arr.sum()
                _push_candidate(prev_norm)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Extract the best seeds
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seed phase] kept {len(seeds)} candidates, best surrogate = {_objective_surrogate(seeds[0]):.6f}")

    # ------------------------------------------------------------------
    # Global best initialization
    # ------------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Exponentiated‑gradient polishing (major part of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        eg_deadline = time.time() + (deadline - time.time()) * 0.55
        seq_eg, val_eg = _exp_gradient_opt(global_best_seq,
                                           n,
                                           eg_deadline,
                                           max_iters=5_000_000,
                                           eta0=5.0)
        if val_eg < global_best_val:
            global_best_seq, global_best_val = seq_eg, val_eg
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        print("\n--- L‑BFGS fine‑tuning ---")
        lbfgs_deadline = deadline - 20.0
        seq_lbfgs = _lbfgs_smooth_optimize(global_best_seq,
                                           n,
                                           lbfgs_deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                          200.0, 500.0, 1_000.0,
                                                          2_000.0, 5_000.0,
                                                          10_000.0, 20_000.0,
                                                          50_000.0, 100_000.0,
                                                          200_000.0, 500_000.0],
                                           max_iter_per_beta=20_000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < global_best_val:
            global_best_seq, global_best_val = seq_lbfgs, val_lbfgs
            print(f"[L‑BFGS] refined surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Order optimisation – greedy swapping
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        print("\n--- Order optimisation (swap search) ---")
        order_deadline = deadline - 12.0
        seq_ord, val_ord = _reorder_greedy(global_best_seq,
                                           n,
                                           order_deadline,
                                           rng)
        if val_ord < global_best_val:
            global_best_seq, global_best_val = seq_ord, val_ord
            print(f"[Swap] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final hill‑climb (mass transfer)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print("\n--- Final hill‑climb ---")
        seq_hc, val_hc = _hill_climb(global_best_seq,
                                     n,
                                     deadline,
                                     rng)
        if val_hc < global_best_val:
            global_best_seq, global_best_val = seq_hc, val_hc
            print(f"[Hill‑climb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
