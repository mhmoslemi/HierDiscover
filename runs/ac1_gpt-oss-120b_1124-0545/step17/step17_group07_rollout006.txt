def propose_candidate(seed=123, budget_s=1000, **kwargs):

    deadline = start_time + budget_s - 5.0

    # base seeds

    for rate in np.logspace(-5, -1, 200):
        seeds.append(_exponential(n, rate))

    # power exponents
    for p in np.linspace(0.0, 5.0, 200):
        seeds.append(_power(n, p))

    # beta asymmetric grid
    alpha_vals = np.logspace(-0.3, 0.8, 30)  # approx 0.5-6
    beta_vals = np.logspace(-0.3, 0.8, 30)
    for a in alpha_vals:
        for b in beta_vals:
            seeds.append(_beta_distribution(n, a, b))

    # random Dirichlet seeds: add few
    rng = np.random.default_rng(seed)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0):

            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # include previous solution if exists
    previous = globals().get('height_sequence_1')

        prev_arr = np.asarray(previous, dtype=np.float64)

"""
Improved step‑function construction.

The algorithm proceeds in three phases:

1️⃣  Parameter sweep.
    Exhaustively evaluates several analytic families (exponential,
    power‑law and *asymmetric* Beta distributions) on a fine grid.
    The best few candidates are kept.

2️⃣  Smooth L‑BFGS refinement.
    Starting from the top candidates we minimise a log‑sum‑exp
    surrogate for the max‑convolution while staying on the simplex
    (variables are the logits of a soft‑max).

3️⃣  Long exponentiated‑gradient polishing.
    Mirror‑descent on the exact objective (max‑convolution) inside the
    probability simplex, with a step‑size 0.5/√t, runs until the time
    budget is exhausted.

The final sequence (ℓ₁‑normalised, i.e. sum = 1) is stored in the
global variable ``height_sequence_1`` and returned as a plain list.
All intermediate progress is printed.
"""

# ----------------------------------------------------------------------
# Imports & optional SciPy
# ----------------------------------------------------------------------
import time
import math
import numpy as np

try:                       # SciPy is optional – L‑BFGS is used only if present.
    import scipy.optimize as _opt
except Exception:         # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective helpers (surrogate works on ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) for a sequence that sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised (possibly asymmetric) Beta(α,β) on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 0.5) ** (alpha - 1.0)) * ((n - i - 0.5) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500) -> np.ndarray:
    """Refine a ℓ₁‑normalised vector with L‑BFGS‑B on a smoothed max‑conv."""
    if _opt is None:                     # SciPy not available – return unchanged.
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 3.0, 7.0, 15.0, 30.0, 60.0,
            120.0, 250.0, 500.0, 1000.0, 2500.0,
            5000.0, 10000.0, 25000.0, 50000.0
        ]

    # we optimise the logits – soft‑max always stays on the simplex.
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on simplex
            conv = _conv_fft(a, a)
            m = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = m
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth max w.r.t. a
            w = exp_beta / sum_exp                    # soft weights (sum to 1)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # convolution of rev(a) with w
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={
                                    'maxiter': max_iter_per_beta,
                                    'disp': False,
                                    'ftol': 1e-12
                                })
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:                 # pragma: no cover
            print(f"[L‑BFGS] optimisation exception: {e}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser (exact surrogate)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent polish on the exact max‑convolution surrogate."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (pairwise terms)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Small random perturbation (used after a good solution)
# ----------------------------------------------------------------------
def _random_perturb(seq: np.ndarray,
                    n: int,
                    rng: np.random.Generator,
                    magnitude: float = 0.02) -> np.ndarray:
    """Add a tiny random logit perturbation and re‑softmax."""
    logit = np.log(seq + 1e-30)
    perturb = magnitude * rng.normal(size=n)
    new_log = logit + perturb
    return _softmax(new_log)


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and time budget
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a few seconds spare for the final evaluation
    deadline = start_time + budget_s - 4.0

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    n = 4000                     # length of the step function

    # ------------------------------------------------------------------
    # 2. Exhaustive parameter sweep over analytic families
    # ------------------------------------------------------------------
    print(f"\n--- Parameter sweep (n={n}) ---")
    candidates: list[np.ndarray] = []

    # uniform baseline
    candidates.append(_uniform(n))

    # ---- Exponential family (rate on a log‑scale) --------------------
    exp_rates = np.logspace(-5, -1, 150)          # 1e‑5 … 0.1
    for rate in exp_rates:
        candidates.append(_exponential(n, float(rate)))

    # ---- Power‑law family --------------------------------------------
    powers = np.linspace(0.0, 5.0, 150)            # p ∈ [0,5]
    for p in powers:
        candidates.append(_power(n, float(p)))

    # ---- Asymmetric Beta family ---------------------------------------
    # α,β ∈ [0.3, 6] on a log‑scale, ~400 combos
    alphas = np.logspace(-0.5, 0.78, 30)          # ≈0.3 … 6
    betas = np.logspace(-0.5, 0.78, 30)
    for a in alphas:
        for b in betas:
            candidates.append(_beta_distribution(n, float(a), float(b)))

    # ---- Random Dirichlet draws (different concentrations) ----------
    concs = (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    for conc in concs:
        for _ in range(3):
            raw = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            candidates.append(raw)

    # ---- Previous solution (if any) -----------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0.0:
                candidates.append(prev_arr / prev_arr.sum())
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3. Evaluate all candidates (surrogate = 2·n·max(conv) for sum=1)
    # ------------------------------------------------------------------
    best_val = float('inf')
    best_vec = None
    for idx, vec in enumerate(candidates):
        if time.time() > deadline:
            break
        vec = np.asarray(vec, dtype=np.float64)
        # protect against accidental negatives or NaNs
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s                               # ℓ₁‑normalise
        val = _objective_surrogate(vec)
        if val < best_val:
            best_val = val
            best_vec = vec.copy()
        if (idx + 1) % 200 == 0:
            print(f"  evaluated {idx + 1:4d} candidates, current best = {best_val:.6f}")

    print(f"\n[Sweep] Best surrogate after sweep = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth L‑BFGS refinement on a few top seeds
    # ------------------------------------------------------------------
    if _opt is not None and time.time() < deadline:
        # pick the 8 best seeds from the sweep for a cheap L‑BFGS run
        evals = [(vec, _objective_surrogate(vec)) for vec in candidates]
        evals.sort(key=lambda x: x[1])
        top_seeds = [vec for vec, _ in evals[:8]]

        beta_schedule = [
            1.0, 3.0, 7.0, 15.0, 30.0, 60.0,
            120.0, 250.0, 500.0, 1000.0, 2500.0,
            5000.0, 10000.0, 25000.0, 50000.0
        ]

        for i, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            # modest per‑seed budget (few thousand L‑BFGS iterations)
            seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                             n,
                                             deadline,
                                             beta_schedule=beta_schedule,
                                             max_iter_per_beta=4000)
            val_opt = _objective_surrogate(seq_opt)
            print(f"[L‑BFGS #{i + 1}] surrogate = {val_opt:.6f}")
            if val_opt < best_val:
                best_val = val_opt
                best_vec = seq_opt.copy()
                print(f"   → new global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Long exponentiated‑gradient polishing (exact surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate the bulk of the remaining time to this stage
        max_iters = int(min(2_000_000, remaining * 2500))
        max_iters = max(300_000, max_iters)
        print(f"\n--- Exponentiated‑gradient polishing (≈ {max_iters} iters) ---")
        seq_exp, val_exp = _exp_gradient_opt(best_vec,
                                             n,
                                             deadline,
                                             max_iters=max_iters)
        print(f"[ExpGrad] surrogate = {val_exp:.6f}")
        if val_exp < best_val:
            best_val = val_exp
            best_vec = seq_exp.copy()
            print(f"   → improved global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Light random‑perturbation + quick L‑BFGS restarts (if time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        for rep in range(2):
            if time.time() > deadline:
                break
            pert = _random_perturb(best_vec, n, rng, magnitude=0.07)
            seq_restart = _lbfgs_smooth_optimize(pert,
                                                 n,
                                                 deadline,
                                                 beta_schedule=beta_schedule[:6],
                                                 max_iter_per_beta=2500)
            val_restart = _objective_surrogate(seq_restart)
            if val_restart < best_val:
                best_val = val_restart
                best_vec = seq_restart.copy()
                print(f"[Restart {rep + 1}] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_vec.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible later runs
    globals()["height_sequence_1"] = best_vec.tolist()
    return best_vec.tolist()
