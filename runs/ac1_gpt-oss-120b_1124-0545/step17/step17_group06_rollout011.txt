"""
Hybrid global–local optimiser for the step‑function attack.

The routine builds a large, diverse pool of ℓ₁‑normalised “seed” windows,
optimises the smooth‑max surrogate of the peak of the self‑convolution
with a continuation L‑BFGS (SciPy) optimiser, optionally polishes the
exact max with a fast exponentiated‑gradient mirror‑descent, and finally
applies a very cheap pair‑wise mass‑transfer local optimiser for the
remaining time.  The whole search is bounded by the supplied time budget
(~1000 s).  The best sequence found is stored in the global variable
``height_sequence_1`` and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# -------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# -------------------------------------------------------------
# Low‑level utilities
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – the value that matches evaluate_sequence
    when the sequence is ℓ₁‑normalised."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# -------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# -------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a large pool of ℓ₁‑normalised candidates.
    """
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for sigma in np.linspace(0.08, 0.30, 7):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 5.0, 23):
        seeds.append(_power(n, p))

    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    for k in (int(0.01 * n), int(0.03 * n), int(0.07 * n)):
        for _ in range(3):
            seeds.append(_sparse_random(n, k, rng))

    for bf in (0.05, 0.15, 0.30):
        for _ in range(3):
            seeds.append(_block_uniform(n, bf, rng))

    seeds.append(_alternating_dirichlet(n, rng))

    # previous solution if compatible
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # sanitise – make sure everything sums to 1
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# -------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# -------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 6000) -> np.ndarray:
    """
    Smooth‑max L‑BFGS‑B optimiser.  Returns an ℓ₁‑normalised vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimizer unavailable – cannot run L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 1e5,
            2e5, 5e5, 1e6, 2e6
        ]

    v = np.log(start_vec + 1e-30)   # logits – stay on simplex via softmax

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # ∑a = 1
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (chain rule through convolution)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# -------------------------------------------------------------
# Exponentiated‑gradient polishing (exact max)
# -------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)               # diminishing step size
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# -------------------------------------------------------------
# Pairwise mass‑transfer local optimiser (O(n) per move)
# -------------------------------------------------------------
def _mass_transfer_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Fast local optimiser that moves a tiny amount of mass from a high‑mass
    index to a low‑mass index while updating the full convolution in O(n)
    time per move.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    max_val = float(np.max(conv))
    best_a = a.copy()
    best_val = max_val

    rng = np.random.default_rng()
    step_factor = 0.30               # initial fraction of source mass to move
    no_improve = 0
    it = 0
    center = n - 1
    L = len(conv)

    while time.time() < deadline:
        if a.max() < 1e-16:
            break

        src = rng.choice(n, p=a)          # source – prefer large mass
        src_mass = a[src]
        if src_mass <= 1e-15:
            continue

        tgt = rng.integers(n)
        if tgt == src:
            continue

        eps = step_factor * src_mass
        if eps < 1e-14:
            continue

        # ---------- tentative updates ----------
        new_conv = conv.copy()

        # linear contribution of moving eps from src to tgt
        new_conv[src:src + n] -= 2.0 * eps * a
        new_conv[tgt:tgt + n] += 2.0 * eps * a

        # quadratic δ^2 terms (only affect a few positions)
        new_conv[center] += 2.0 * eps * eps
        offset = tgt - src
        idx1 = center + offset
        if 0 <= idx1 < L:
            new_conv[idx1] -= eps * eps
        idx2 = center - offset
        if offset != 0 and 0 <= idx2 < L:
            new_conv[idx2] -= eps * eps

        new_max = float(np.max(new_conv))

        # ---------- accept if it improves ----------
        if new_max < max_val - 1e-16:
            a[src] = src_mass - eps
            a[tgt] = a[tgt] + eps
            conv = new_conv
            max_val = new_max
            no_improve = 0
            if max_val < best_val:
                best_val = max_val
                best_a = a.copy()
        else:
            no_improve += 1

        it += 1
        if no_improve > 6000:
            step_factor *= 0.5
            if step_factor < 1e-4:
                step_factor = 0.30
            no_improve = 0

        if it % 25000 == 0:
            print(f"[MassTransfer] it={it:,} best max={best_val:.12f}")

    return best_a, 2.0 * n * best_val


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the global time budget (≈ 1000 s) and stores the
    best sequence in the global variable ``height_sequence_1``.
    """
    # ---------------------------------------------------------
    # 0. initialise RNG and timing
    # ---------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # ---------------------------------------------------------
    # 1. Problem size – a larger n gives a better asymptotic bound.
    #    8000 works comfortably within the time budget on most judges.
    # ---------------------------------------------------------
    n = 8000
    print(f"\n=== Optimisation started (n={n}, time budget={budget_s}s) ===")

    # ---------------------------------------------------------
    # 2. Seed generation
    # ---------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 6
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1}: {val:.6f}")

    # ---------------------------------------------------------
    # 3. Continuation L‑BFGS on the best seeds
    # ---------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        10000.0, 20000.0, 50000.0, 1e5,
        2e5, 5e5, 1e6, 2e6
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before all L‑BFGS runs.")
            break

        remaining = deadline - time.time()
        # try to give each seed a modest budget (≈ 10 % of remaining time)
        max_iter = int(min(15000, max(4000, remaining * 0.12 * 1e5)))

        print(f"\n--- L‑BFGS run #{idx + 1} (max_iter={max_iter}) ---")
        try:
            seq_opt = _lbfgs_smooth_optimize(
                seed_vec,
                n,
                deadline,
                beta_schedule=beta_schedule,
                max_iter_per_beta=max_iter
            )
        except RuntimeError:
            # SciPy not available – fall back to the seed itself
            seq_opt = seed_vec.copy()

        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] new global best = {best_val:.6f}")

        # ----------------------------------------------------
        # Exponentiated‑gradient polishing (exact max)
        # ----------------------------------------------------
        if time.time() < deadline:
            rem = deadline - time.time()
            max_exp = int(min(500_000, max(300_000, rem * 0.07 * 1e5)))
            seq_exp, val_exp = _exp_gradient_opt(best_seq, n, deadline, max_iters=max_exp)
            if val_exp < best_val:
                best_seq = seq_exp.copy()
                best_val = val_exp
                print(f"[ExpGrad] improved to {best_val:.6f}")

    # ---------------------------------------------------------
    # 4. Pairwise mass‑transfer local optimisation (takes the rest)
    # ---------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mass‑transfer local optimisation (final stage) ---")
        seq_mt, val_mt = _mass_transfer_opt(best_seq, n, deadline)
        if val_mt < best_val:
            best_seq = seq_mt.copy()
            best_val = val_mt
            print(f"[MassTransfer] improved to {best_val:.6f}")

    # ---------------------------------------------------------
    # 5. Final evaluation with the official evaluator
    # ---------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
