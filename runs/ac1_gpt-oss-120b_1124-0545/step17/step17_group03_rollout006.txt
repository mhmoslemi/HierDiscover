def _flatten_optimize(start_a, n, deadline, max_steps=10000):
    v = np.log(start_a + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1=0.9; beta2=0.999; eps=1e-8
    t=0
    for step in range(max_steps):
        if time.time()>deadline:
            break
        # compute a
        a = _softmax(v)
        # conv a*a
        conv = _conv_fft(a, a)
        target_val = 1.0/(2*n - 1)  # since sum a = 1
        diff = conv - target_val
        # gradient w.r.t a
        rev_a = a[::-1]
        grad_full = _conv_fft(diff, rev_a)   # length = 3n-2
        grad_a = 2.0 * grad_full[n-1:n-1+n]   # slice length n
        # chain rule to v
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1*m + (1-beta1)*grad_v
        v2 = beta2*v2 + (1-beta2)*(grad_v**2)
        m_hat = m/(1-beta1**t)
        v2_hat = v2/(1-beta2**t)
        lr = 0.2 / np.sqrt(t)   # maybe 0.2 base
        v -= lr * m_hat / (np.sqrt(v2_hat)+eps)

        if step % 2000 == 0:
            # evaluate actual objective for progress
            obj = evaluate_sequence(a.tolist())
            print(f"[Flatten] step {step} obj {obj:.6f}")
    a = _softmax(v)
    return a

def _adam_flatten_optimize(start_a, n, deadline, max_steps=20000, lr_initial=0.3):
    v = np.log(start_a + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    target_val = 1.0 / (2 * n - 1)
    best_a = start_a.copy()
    best_obj = _objective_exact(best_a)

    for step in range(1, max_steps+1):
        if time.time() > deadline:
            break
        a = _softmax(v)
        conv = _conv_fft(a, a)
        diff = conv - target_val
        rev_a = a[::-1]
        grad_full = _conv_fft(diff, rev_a)
        grad_a = 2.0 * grad_full[n-1:n-1+n]
        # chain rule to logits
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)
        lr = lr_initial / np.sqrt(t)
        v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

        # periodic evaluation
        if step % 2000 == 0:
            cur_obj = _objective_exact(a)
            if cur_obj < best_obj:

"""
search for a step‑function that minimizes
    2·n·max(conv(a,a)) / (sum(a)²)

The method consists of three stages:

1.  Seed creation – a handful of analytic and random ℓ₁‑normalised vectors.
2.  “Flat‑convolution” optimisation – Adam minimises the squared error
    between a*a and the flat target 1/(2n‑1).  This drives the full
    convolution towards a constant, which is (asymptotically) the optimal
    situation and gives an objective close to 1.
3.  A short continuation L‑BFGS on a smooth log‑sum‑exp surrogate of the
    max‑convolution (high β) to clean up the remaining peaks,
    followed by an optional tiny stochastic mass‑transfer local search.

All steps respect the 1000 s time‑budget and the best vector is stored in
the global name ``height_sequence_1`` for warm‑starting the next run.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Exact objective (max‑convolution) for a ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))  (assumes sum(a)=1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Adam optimiser that flattens the convolution
# ----------------------------------------------------------------------
def _adam_flatten_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_steps: int = 15000,
                           lr_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Adam minimises ½·‖conv(a,a) – 1/(2n‑1)·1‖².
    Returns the best vector found (still ℓ₁‑normalised) and its exact objective.
    """
    v = np.log(start_a + 1e-30)          # logits – guarantees positivity & unit ℓ₁‑norm
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    target_val = 1.0 / (2 * n - 1)

    best_a = start_a.copy()
    best_val = _objective_exact(best_a)

    for step in range(1, max_steps + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # current probability vector
        conv = _conv_fft(a, a)               # full convolution (length 2n‑1)
        diff = conv - target_val             # deviation from flat target

        # gradient w.r.t. a : 2·conv(diff, rev(a)) slice
        rev_a = a[::-1]
        grad_full = _conv_fft(diff, rev_a)   # length 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on logits
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr_initial / np.sqrt(t)
        v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

        # occasional progress report & best‑so‑far update
        if step % 3000 == 0:
            cur_val = _objective_exact(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
            print(f"[Flatten] step {step:,}  objective = {cur_val:.6f}")

    # final evaluation
    a = _softmax(v)
    cur_val = _objective_exact(a)
    if cur_val < best_val:
        best_val, best_a = cur_val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth‑max surrogate and continuation L‑BFGS (taken from the reference)
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Log‑sum‑exp smooth approximation of max(conv(a,a)).  Returns (obj, grad_v)."""
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = np.max(conv)                     # numeric stabiliser
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_term / sum_exp                     # soft‑max weights, length 2n‑1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 1200) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)      # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_loc):
            return _smooth_objective(v_loc, n, beta_val)

        def callback(v_loc):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method="L-BFGS-B",
                                jac=True,
                                callback=callback,
                                options={"maxiter": max_iter_per_beta,
                                         "ftol": 1e-12,
                                         "disp": False})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during β={beta_val}")
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} (β={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Very cheap stochastic mass‑transfer hill climber
# ----------------------------------------------------------------------
def _mass_transfer_local_search(start_a: np.ndarray,
                               n: int,
                               deadline: float,
                               total_iters: int = 80_000,
                               step_frac_start: float = 0.12,
                               step_frac_end: float = 0.008) -> tuple[np.ndarray, float]:
    """Hill‑climbing: move a fraction of mass from i → j, linearly decaying step size."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(12345)

    for it in range(total_iters):
        if time.time() > deadline:
            break

        frac = step_frac_start + (step_frac_end - step_frac_start) * (it / total_iters)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue

        delta = frac * a[i]
        if delta <= 0.0:
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        cur_val = _objective_exact(a_candidate)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_candidate.copy()
            a = a_candidate

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    The best sequence is stored globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # problem size – 6000 gives a nice trade‑off between expressivity and speed
    n = 6000
    print(f"\n=== Optimisation start (n={n}, budget≈{budget_s}s) ===")

    # ------------------------------------------------------------------
    # 1. Seed generation (few diverse ℓ₁‑normalised candidates)
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    for lam in (0.001, 0.002, 0.005, 0.01):
        seeds.append(_exponential_decay(n, lam))
    for sigma in (0.10, 0.20, 0.30):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet / sparse draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))
    for frac in (0.05, 0.1, 0.2, 0.3):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # warm‑start from previous run if available
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0.0:
                seeds.append(prev_arr / prev_arr.sum())
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Quick exact evaluation, keep the best few
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)          # 2·n·max(conv)  (sum(v)=1)
        if val < best_val:
            best_val, best_seq = val, v.copy()
    print(f"[Init] best exact objective among raw seeds = {best_val:.6f}")

    # retain top‑k seeds for the intensive flattening stage
    seeds.sort(key=_objective_exact)
    top_k = min(5, len(seeds))
    top_seeds = seeds[:top_k]

    # ------------------------------------------------------------------
    # 3. Flatten‑convolution optimisation (Adam on squared‑error surrogate)
    # ------------------------------------------------------------------
    global_best_a = best_seq.copy()
    global_best_obj = best_val

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate up to ~0.15 s per seed for Adam flattening (adjustable)
        max_iters = int(min(15000, remaining * 0.15 * 1e5))
        print(f"\n--- Flatten run #{idx+1} (max_iters={max_iters}) ---")
        a_flat, obj_flat = _adam_flatten_optimize(seed_vec,
                                                  n,
                                                  deadline,
                                                  max_steps=max_iters,
                                                  lr_initial=0.3)
        print(f"[Flatten] result objective = {obj_flat:.6f}")

        if obj_flat < global_best_obj:
            global_best_a, global_best_obj = a_flat.copy(), obj_flat
            print(f"[Update] New global best after flatten = {global_best_obj:.6f}")

    # ------------------------------------------------------------------
    # 4. Refinement with continuation L‑BFGS on the smooth max‑surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iter_per_beta = int(min(2000, remaining * 0.08 * 1e5))
        print("\n--- L‑BFGS refinement (smooth max) ---")
        a_refined = _lbfgs_smooth_optimize(global_best_a,
                                           n,
                                           deadline,
                                           max_iter_per_beta=max_iter_per_beta)
        obj_refined = _objective_exact(a_refined)
        print(f"[L‑BFGS] refined objective = {obj_refined:.6f}")

        if obj_refined < global_best_obj:
            global_best_a, global_best_obj = a_refined.copy(), obj_refined
            print(f"[Update] New global best after L‑BFGS = {global_best_obj:.6f}")

    # ------------------------------------------------------------------
    # 5. Tiny stochastic mass‑transfer local search (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(80_000, remaining * 100))
        if iters > 0:
            print("\n--- Local mass‑transfer search ---")
            a_loc, obj_loc = _mass_transfer_local_search(global_best_a,
                                                         n,
                                                         deadline,
                                                         total_iters=iters)
            print(f"[Local] objective = {obj_loc:.6f}")
            if obj_loc < global_best_obj:
                global_best_a, global_best_obj = a_loc.copy(), obj_loc
                print(f"[Update] New global best after local search = {global_best_obj:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official scorer
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_a.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_obj:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for warm‑starts in a subsequent call
    globals()["height_sequence_1"] = global_best_a.tolist()
    return global_best_a.tolist()
