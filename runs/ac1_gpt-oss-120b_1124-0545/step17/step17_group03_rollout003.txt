"""
Hybrid search for a low‑max‑autocorrelation (step‑function) sequence.

The algorithm proceeds in the following stages (all respect the 1000 s
budget):
1.  Generate a very large pool of ℓ₁‑normalised candidate vectors.
    The pool mixes analytical windows (uniform, triangular, Gaussian,
    β‑shapes, Welch, Hann, sinusoidal, exponential, power‑law, Kaiser
    etc.), random Dirichlet draws, sparse patterns and low‑frequency
    cosine sums.  Parameter sweeps are performed densely.
2.  Compute the exact objective 2·n·max(conv(a,a)) for all candidates and
    keep the best *k* seeds (k≈30).
3.  For each of those seeds run a continuation L‑BFGS on a smooth
    log‑sum‑exp surrogate of the max‑convolution.  The β‑schedule is
    extended up to 10⁶ and each β is allowed a few thousand L‑BFGS
    iterations, yielding high‑quality local minima.
4.  Starting from the best vector obtained above, perform a long
    exponentiated‑gradient (mirror‑descent) optimisation using a
    multi‑subgradient of the exact max‑convolution.  The step‑size is
    η₀/√t and we run up to ~6·10⁵ iterations (adjusted to the time left).
5.  Several short EG restarts from random Dirichlet seeds diversify the
    search.
6.  Finally a stochastic “mass‑transfer’’ local search with a simple
    simulated‑annealing acceptance rule is applied to escape shallow
    minima.

All intermediate progress is printed.  The best sequence found is stored
in the global name ``height_sequence_1`` and returned as a plain list.
The returned sequence is non‑negative; its ℓ₁‑norm is 1 (the evaluation
function is scale‑invariant).
"""

# ----------------------------------------------------------------------
# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (lies on the simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic window families (all return ℓ₁‑normalised vectors, sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _welch(n: int) -> np.ndarray:
    """Quadratic (Welch) window: 1 - ((i-(n-1)/2)/((n-1)/2))**2."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - ((i - (n - 1) / 2.0) / ((n - 1) / 2.0)) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                # Bessel I0
    s = a.sum()
    return a / s


def _linear_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    s = a.sum()
    return a / s


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """σ = sigma_frac·n, centred."""
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on the discrete set {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i = 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s


def _logistic_shape(n: int, scale: float) -> np.ndarray:
    """a_i ∝ 1/(1+exp((i-μ)/scale))."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp((i - mu) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a vector from Dirichlet(conc,…,conc)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    """
    Choose a random subset of size m = max(2, int(frac * n)),
    put equal mass 1/m on each chosen index, zero elsewhere.
    """
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α) + b·α, then renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    """Cyclic shift."""
    return np.roll(v, shift)

def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a that sums to 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv


def _subgradient_maxconv_multi(a: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Multi‑subgradient: average over all lags whose value is within eps of the
    maximum.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    idxs = np.where(conv >= max_c - eps)[0]
    g = np.zeros_like(a)
    for idx in idxs:
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - i_range
        g[i_range] += 2.0 * a[rev_idx]
    if len(idxs):
        g /= len(idxs)
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (used by L‑BFGS / Adam)
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Log‑sum‑exp smooth approximation of max(conv(a,a)).
    Returns (obj, grad) where grad is wrt the logits v.
    """
    a = _softmax(v)                     # simplex point
    conv = _conv_fft(a, a)               # full convolution, length 2n‑1
    shift = np.max(conv)                 # numerical shift for stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # derivative of the smoothed max‑convolution w.r.t. a
    w = exp_term / sum_exp               # weights, length 2n‑1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice that aligns with a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6,
        ]
    v = np.log(start_vec + 1e-30)   # logits (ensures positivity)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during β={beta_val}")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} (β={beta_val})")
            break

    return _softmax(v)   # back to a probability vector


def _adam_smooth_optimize(start_vec: np.ndarray,
                          n: int,
                          deadline: float,
                          beta: float,
                          max_steps: int,
                          lr_initial: float = 0.5) -> np.ndarray:
    """Adam optimiser on the smooth surrogate (single β)."""
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta_val = float(beta)

    for _ in range(max_steps):
        if time.time() > deadline:
            break
        loss, grad = _smooth_objective(v, n, beta_val)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)
        lr = lr_initial / np.sqrt(t)
        v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize_multi(start_a: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 400_000,
                                 eta0: float = 2.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using a multi‑subgradient of the exact max‑convolution.
    Returns (best_a, best_obj).
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_maxconv_multi(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 150_000 == 0:
            cur = _objective_exact(a)
            print(f"[ExpGrad] it={it:,} obj={cur:.6f}")
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Stochastic mass‑transfer / simulated‑annealing local search
# ----------------------------------------------------------------------
def _simulated_annealing_local_search(start_a: np.ndarray,
                                      n: int,
                                      deadline: float,
                                      total_iters: int = 150_000,
                                      step_frac_start: float = 0.12,
                                      step_frac_end: float = 0.008) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing with a simple simulated‑annealing acceptance rule.
    """
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456789)

    T0 = 1e-3          # initial temperature (relative to objective scale)
    Tf = 1e-6          # final temperature

    for it in range(total_iters):
        if time.time() > deadline:
            break

        # linear schedule for step size and temperature
        frac = step_frac_start + (step_frac_end - step_frac_start) * (it / total_iters)
        T = T0 + (Tf - T0) * (it / total_iters)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue

        delta = frac * a[i]
        if delta <= 0.0:
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        val = _objective_exact(a_candidate)
        accept = False
        if val < best_val:
            accept = True
        else:
            # accept with probability exp(-(val-best)/T)
            prob = math.exp(-(val - best_val) / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if val < best_val:
                best_val = val
                best_a = a_candidate.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Seed‑pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Generate a diversified pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families (cover a wide range of shapes)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_welch(n))
    seeds.append(_hann(n))
    seeds.append(_linear_decrease(n))
    for lam in (0.0003, 0.0007, 0.0015, 0.003, 0.006, 0.012):
        seeds.append(_exponential_decay(n, lam))
    for g in (0.2, 0.5, 0.8, 1.2, 1.5, 2.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 12, 16):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian family (different spreads)
    for sigma in np.linspace(0.07, 0.45, 13):
        seeds.append(_gaussian(n, sigma))

    # Power‑law
    for p in np.linspace(0.0, 4.0, 25):
        seeds.append(_power_law(n, p))

    # Logistic family (different steepness)
    for scale in np.linspace(0.05, 0.30, 12):
        seeds.append(_logistic_shape(n, scale))

    # Beta family (grid)
    for a in np.linspace(0.5, 3.0, 9):
        for b in np.linspace(0.5, 3.0, 9):
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.02, 0.05, 0.1, 0.3, 0.7, 1.5, 3.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse patterns (different sparsities)
    for frac in (0.03, 0.07, 0.15, 0.25, 0.35, 0.45):
        for _ in range(5):
            seeds.append(_random_sparse(n, frac, rng))

    # Low‑frequency cosine curves (different numbers of frequencies)
    for nf in (2, 3, 4, 5, 6):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Pairwise mixtures of already‑generated seeds
    base = np.arange(len(seeds))
    for _ in range(250):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Rotations of promising seeds
    for _ in range(40):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # Include the previous solution (if any) for warm start
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Clean & normalise
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 314159, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser for the step‑function bound.
    Returns the best sequence found (list of non‑negative floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a few seconds safety margin

    # ---------- Problem size ----------
    n = 4000                      # keeps per‑iteration cost modest
    print(f"\n=== Optimisation start (n={n}), time budget ≈{budget_s}s ===")

    # ---------- Warm‑start ----------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    # ---------- 1. Seed generation ----------
    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} raw seed vectors.")

    # ---------- 2. Quick exact evaluation, keep top‑k ----------
    best_val = np.inf
    best_seq = None
    for vec in seeds:
        val = _objective_exact(vec)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best raw objective = {best_val:.6f}")

    # Sort and keep the best few for heavy optimisation
    seeds.sort(key=_objective_exact)
    top_k = min(30, len(seeds))
    top_seeds = seeds[:top_k]
    print(f"Keeping top {top_k} seeds for L‑BFGS refinement.")

    # ---------- 3. Continuation L‑BFGS ----------
    global_best_seq = best_seq.copy()
    global_best_val = best_val

    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        1e4, 2e4, 5e4, 1e5,
        2e5, 5e5, 1e6,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline before finishing L‑BFGS runs.")
            break

        remaining = deadline - time.time()
        per_beta = max(800, int(min(5000, remaining * 0.045)))  # a few thousand max‑iters
        print(f"\n--- L‑BFGS run #{idx+1}/{top_k} (max_iter/beta={per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] exact objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ---------- 4. Exponentiated‑gradient (multi‑subgradient) restarts ----------
    remaining = deadline - time.time()
    if remaining > 0.0:
        restarts = 6
        iters_per_restart = int(min(300_000, remaining * 0.10 / restarts))
        print("\n--- EG restarts (multi‑subgradient) ---")
        for r in range(restarts):
            if time.time() > deadline:
                break
            conc = rng.uniform(0.02, 3.0)
            start_vec = _random_dirichlet(n, conc, rng)
            seq_exp, val_exp = _exp_gradient_optimize_multi(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_restart,
                eta0=2.8,
            )
            print(f"[EG restart {r+1}] best = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] New best after EG restart = {global_best_val:.6f}")

    # ---------- 5. Long EG run from current best ----------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(1_200_000, remaining * 0.20))
        if max_iters > 0:
            print("\n--- Long EG optimisation (starting from current best) ---")
            seq_exp, val_exp = _exp_gradient_optimize_multi(
                global_best_seq,
                n,
                deadline,
                max_iters=max_iters,
                eta0=3.2,
            )
            print(f"[Long EG] best = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] New best after long EG = {global_best_val:.6f}")

    # ---------- 6. Optional Adam refinement (high β) ----------
    # Empirically this step sometimes harms the objective, so we apply it
    # only if a noticeable time margin remains.
    if time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 20.0:
            print("\n--- Adam refinement (high β) ---")
            a = global_best_seq.copy()
            for beta in [5e5, 1e6, 2e6]:
                if time.time() > deadline:
                    break
                steps = int(min(2000, (deadline - time.time()) * 0.08))
                if steps <= 0:
                    break
                a = _adam_smooth_optimize(a, n, deadline, beta, max_steps=steps, lr_initial=0.6)
                cur_val = _objective_exact(a)
                print(f"[Adam β={beta:.0f}] objective = {cur_val:.6f}")
                if cur_val < global_best_val:
                    global_best_seq = a.copy()
                    global_best_val = cur_val
                    print(f"[Update] New best after Adam = {global_best_val:.6f}")

    # ---------- 7. Simulated‑annealing mass‑transfer local search ----------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(250_000, remaining * 120))
        if iters > 0:
            print("\n--- Simulated‑annealing local search ---")
            seq_loc, val_loc = _simulated_annealing_local_search(
                global_best_seq,
                n,
                deadline,
                total_iters=iters,
                step_frac_start=0.10,
                step_frac_end=0.006,
            )
            print(f"[SA LocalSearch] best = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] New best after SA = {global_best_val:.6f}")

    # ---------- Final evaluation (official) ----------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
