"""
Propose a step‑function (non‑negative heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three phases:

1. **Fast low‑resolution search** – using n_small=1024 we generate tens of
   thousands of diverse probability vectors (Dirichlet draws with many
   concentration parameters, random beta shapes, low‑frequency cosine
   superpositions, etc.).  The best few are kept.

2. **Upscaling + heavy continuous optimisation** – the selected candidates
   are up‑sampled to the final size n_large (default 4000) and refined with
   exponentiated‑gradient mirror descent, followed by a continuation L‑BFGS
   optimiser that smoothes the max‑convolution via a log‑sum‑exp surrogate.
   The optimisation runs until the time budget is almost exhausted.

3. **Simulated‑annealing hill‑climb** – a lightweight SA that moves a tiny
   fraction of mass between two coordinates, accepting occasional uphill
   moves.  This squeezes a few extra digits out of the best candidate.

The best sequence found is stored globally as ``height_sequence_1`` for
future runs.
"""

import time
import math
import heapq
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Numeric utilities (FFT‑based convolution, softmax, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a vector normalised to sum = 1:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Random generators (return ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet with all concentration parameters == conc."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random beta‑shape (α,β) with α,β drawn uniformly in log‑space."""
    # draw logs uniformly to cover several orders of magnitude
    log_alpha = rng.uniform(-2.0, 1.5)   # α in [1e‑2, ~30]
    log_beta = rng.uniform(-2.0, 1.5)
    alpha = 10 ** log_alpha
    beta = 10 ** log_beta
    return _beta_shape(n, alpha, beta)


def _smooth(vec: np.ndarray, width: int = 5) -> np.ndarray:
    """Simple moving‑average smoothing (width must be odd)."""
    if width <= 1:
        return vec
    pad = width // 2
    v = np.pad(vec, (pad, pad), mode='reflect')
    kernel = np.ones(width) / width
    sm = np.convolve(v, kernel, mode='valid')
    s = sm.sum()
    return sm / s if s > 0 else _uniform(len(vec))


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 6.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 20000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    The variable lives in log‑softmax space (v), guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         1e4, 2e4, 5e4, 1e5, 2e5,
                         5e5, 1e6]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # stays on simplex
            conv = _conv_fft(a, a)            # full convolution
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c                     # for numerical stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp            # soft‑max weights over conv entries

            # gradient wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _annealing_hill_climb(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search.
    Moves a small fraction of mass between two coordinates.
    Accepts uphill moves with probability exp(-(Δ)/T).
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    # start temperature ~ 1 % of current objective
    T = max(1e-4, 0.01 * cur_val)
    decay = 0.9995  # geometric decay per step

    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # Fraction of a[i] to move (0.1% – 1% of its current mass)
        eps = cur[i] * rng.uniform(0.001, 0.01)
        if eps <= 0 or cur[i] - eps < 0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective_surrogate(new)

        delta = new_val - cur_val
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = math.exp(-delta / T) if T > 1e-12 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new.copy()
                best_val = new_val

        # temperature schedule
        T = max(1e-12, T * decay)

    return best, best_val

                                                          5e5, 1e6],
                                           max_iter_per_beta=25000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < global_best_val:
            global_best_seq = seq_lbfgs
            global_best_val = val_lbfgs
            print(f"[L‑BFGS] refined surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 10. Simulated‑annealing hill‑climb (use leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Simulated‑annealing hill‑climb ---")
        seq_hc, val_hc = _annealing_hill_climb(global_best_seq,
                                                n_large,
                                                deadline,
                                                rng)
        if val_hc < global_best_val:
            global_best_seq = seq_hc
            global_best_val = val_hc
            print(f"[Hill‑climb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 11. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
