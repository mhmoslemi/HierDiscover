def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that projects onto the simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    s = a.sum()
    return a / s


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape on a discrete grid."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a vector from Dirichlet(conc,…,conc)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    """Pick a random subset of size m = max(2, int(frac*n)) and split mass uniformly."""
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α) + b·α, then renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    """Cyclic shift."""
    return np.roll(v, shift)

def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Log‑sum‑exp smoothing of max(conv(a,a)).
    Returns (objective, grad_wrt_logits).
    """
    a = _softmax(v)                     # simplex point
    conv = _conv_fft(a, a)              # full convolution (len = 2n‑1)
    shift = np.max(conv)                # numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # soft‑max weights over all lags
    w = exp_term / sum_exp               # length = 2n‑1

    # gradient of the smoothed max‑convolution w.r.t. a
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the soft‑max parametrisation
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 1500,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 100000.0,
            200000.0, 500000.0, 1_000_000.0, 2_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline hit during β={beta_val}")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} (β={beta_val})")
            break

    return _softmax(v)


def _adam_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta: float,
    max_steps: int,
    lr_initial: float = 0.5,
) -> np.ndarray:
    """Adam optimiser on the smooth surrogate for a fixed β."""
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta_val = float(beta)

    for _ in range(max_steps):

            break
        loss, grad = _smooth_objective(v, n, beta_val)
        t += 1

        v2_hat = v2 / (1.0 - beta2 ** t)
        lr = lr_initial / np.sqrt(t)
        v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

    return _softmax(v)

def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (sum(a)=1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient corresponding to a single maximising lag."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = idx - i_range
    g[i_range] = 2.0 * a[rev_idx]
    return g


def _subgradient_maxconv_multi(a: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Multi‑subgradient: average over all lags whose value is within eps of the max.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    idxs = np.where(conv >= max_c - eps)[0]
    g = np.zeros_like(a)
    for idx in idxs:
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - i_range
        g[i_range] += 2.0 * a[rev_idx]
    if len(idxs) > 0:
        g /= len(idxs)
    return g


def _exp_gradient_optimize_multi(

# Targeted hill‑climb (mass transfer guided by the exact sub‑gradient)

def _targeted_hill_climb(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    total_iters: int = 300_000,
    step_start: float = 0.02,
    step_end: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climbing: repeatedly move a fraction of probability mass
    from the index with the largest sub‑gradient (worst‑case lag) to the index
    with the smallest sub‑gradient.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)
    rng = np.random.default_rng(123456)

    for it in range(total_iters):
        if time.time() > deadline:
            break

        # linear decay of the transferred fraction
        frac = step_start + (step_end - step_start) * (it / total_iters)

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))

        # compute the exact sub‑gradient for this lag
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev = idx - i_range
        g = 2.0 * a[rev]                     # gradient values on the active range

        # pick worst and best indices inside the active range
        i_high = i_range[np.argmax(g)]
        i_low = i_range[np.argmin(g)]

        # sanity: ensure we actually have mass to move
        if a[i_high] <= 0.0 or i_high == i_low:
            # fall back to a random move to keep the search alive
            i_high = rng.integers(0, n)
            i_low = rng.integers(0, n)
            if i_high == i_low or a[i_high] <= 0.0:
                continue

        delta = frac * a[i_high]
        if delta <= 0.0:
            continue

        a[i_high] -= delta
        a[i_low] += delta

        # re‑normalise to compensate floating‑point drift
        a_sum = a.sum()
        if a_sum != 1.0:
            a /= a_sum

        cur_val = _objective_exact(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a, best_val

                max_iters=iters_per_run,
                eta0=3.0,
            )
            print(f"[EG run {r+1}] best = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] New best after EG run = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Targeted hill‑climb (mass‑transfer guided by the exact sub‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(600_000, remaining * 150))   # roughly 150 iters per second
        print("\n--- Targeted hill‑climb (gradient‑guided mass transfer) ---")
        seq_hc, val_hc = _targeted_hill_climb(
            global_best_seq,
            n,
            deadline,
            total_iters=iters,
            step_start=0.015,
            step_end=0.001,
        )
        print(f"[Hill‑climb] best = {val_hc:.6f}")
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[Update] New best after hill‑climb = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Adam refinement with very high β (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print("\n--- Adam refinement (final polishing) ---")
        a = global_best_seq.copy()
        for beta in [500_000.0, 1_000_000.0, 2_000_000.0, 5_000_000.0]:
            if time.time() > deadline:
                break
            steps = int(min(6000, (deadline - time.time()) * 0.06 * 1e5))
            if steps <= 0:
                break
            a = _adam_smooth_optimize(a, n, deadline, beta, max_steps=steps, lr_initial=0.5)
            cur_val = _objective_exact(a)
            print(f"[Adam β={beta:.0f}] objective = {cur_val:.6f}")
            if cur_val < global_best_val:
                global_best_seq = a.copy()
                global_best_val = cur_val
                print(f"[Update] New best after Adam = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final official evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
