def _conv_fft(a, b):
    # using real FFT as defined earlier
    L = len(a) + len(b) - 1
    N = 1 << ((L - 1).bit_length())
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv

def _objective_surrogate(seq):
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# analytic families
def _uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _triangular(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        return _uniform(n)
    return a / s

def _beta_distribution(n, alpha, beta):
    i = np.arange(n, dtype=np.float64)
    a = ((i + 0.5) ** (alpha - 1.0)) * ((n - i - 0.5) ** (beta - 1.0))
    s = a.sum()
    if s == 0:
        return _uniform(n)
    return a / s

# other families for enrichment
def _gaussian(n, sigma_frac):
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s == 0:
        return _uniform(n)
    return a / s

def _power(n, p):
    idx = np.arange(1, n+1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s == 0:
        return _uniform(n)
    return a / s

    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0:
        return _uniform(n)
    return a / s

def _subgradient_opt(start_seq, n, deadline, max_iters=200_000):
    # simple subgradient descent
    a = start_seq.copy()
    best_seq = a.copy()

        if time.time() >= deadline:

        max_idx = int(np.argmax(conv))

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def _proj_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1

    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

"""
LP‑guided step‑function optimisation.

* Build a modest pool of analytic seeds (uniform, triangular, Gaussians,
  power‑law, exponential, symmetric Beta, random Dirichlet).
* Pick the best seed w.r.t. the exact surrogate
        2·n·max(conv(a,a))       (a is ℓ₁‑normalised)
* Repeatedly solve a linear programme

        maximise   Σ g_j
        subject to  (a ∗ g)_k ≤ M   for all k,
                    g_j ≥ 0,

  where M = max(conv(a,a)).
  The LP tries to pack as much total mass as possible into positions that
  are “quiet’’ for the current a.  After normalising g back to sum 1 the
  peak of its self‑convolution often drops.

* If an LP step improves the surrogate, replace a by the normalised g
  and continue.  When the LP stops improving, finish with a short
  projected sub‑gradient polish.

The whole pipeline respects the 1000 s budget and stores the best
sequence in the global variable ``height_sequence_1`` for the next run.

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = 1 << ((L - 1).bit_length())
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) for a ℓ₁‑normalised vector."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic families (already ℓ₁‑normalised)
# ------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_distribution(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised symmetric Beta(α,β) on {0,…,n‑1} (half‑offset avoids zeros)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 0.5) ** (alpha - 1.0)) * ((n - i - 0.5) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Seed pool construction

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.05, 0.12, 0.20, 0.30, 0.45):
        seeds.append(_gaussian(n, sigma))
    for p in (0.0, 0.25, 0.5, 0.75, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for rate in (0.001, 0.005, 0.01, 0.02):
        seeds.append(_exp_distribution(n, rate))
    for a in (0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        seeds.append(_beta_distribution(n, a, a))

    # Random Dirichlet draws
    for conc in (0.01, 0.05, 0.1, 0.5, 1.0):
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)))

    # Previous solution if it matches the current size
    if previous is not None and previous.shape == (n,) and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Normalise everything (guards against accidental zero‑sum vectors)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ------------------------------------------------------------
# Linear‑program refinement
# ------------------------------------------------------------
def _lp_refine(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve
        maximise Σ g
        s.t.   (start ∗ g) ≤ M,
               g ≥ 0,
    where M = max(start∗start).
    After each successful LP step the solution is normalised to sum 1 and
    becomes the new start vector.
    """
    best_seq = start_vec.copy()
    best_val = _objective_surrogate(best_seq)

    iteration = 0
    while time.time() < deadline:
        M = float(np.max(_conv_fft(best_seq, best_seq)))

        # Build the (2n‑1)×n constraint matrix A where
        #   A[k, j] = best_seq[k‑j]   (with zero padding)
        # This is a Toeplitz–like matrix with constant diagonals.
        diags_data = [np.full(n, val, dtype=np.float64) for val in best_seq]
        A = _sp.diags(diags_data,
                      offsets=np.arange(n),
                      shape=(2 * n - 1, n),
                      format="csr")

        b = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)         # minimise –Σg  →  maximise Σg
        bounds = [(0.0, None)] * n

        try:
            res = _opt.linprog(
                c,
                A_ub=A,
                b_ub=b,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
        except Exception as exc:  # pragma: no cover
            print(f"[LP] exception at iteration {iteration}: {exc}")
            break

        if not res.success:
            print(f"[LP] failed at iteration {iteration}: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            print(f"[LP] zero sum at iteration {iteration}")
            break

        g_norm = g / sum_g
        new_val = _objective_surrogate(g_norm)

        print(
            f"[LP] iter {iteration:2d} – M={M:.6e}, Σg={sum_g:.6f}, "
            f"surrogate={new_val:.6f}"
        )

        if new_val < best_val - 1e-9:
            best_seq, best_val = g_norm, new_val
            # continue refining from the improved sequence
        else:
            print("[LP] No improvement, terminating LP phase.")
            break

        iteration += 1
        del A, diags_data  # free memory for the next round

    return best_seq, best_val


# ------------------------------------------------------------
# Short sub‑gradient polishing (exact surrogate)
# ------------------------------------------------------------
def _subgradient_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.3 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()

    return best_seq, best_val

    deadline = start + budget_s - 3.0

    n = 4000  # problem size (large enough to push the lower bound close to 1)

    # -----------------------------------------------------------------
    # Initialise from the previous run if possible
    # -----------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.shape != (n,) or prev_arr.sum() == 0.0:
            prev_arr = None

    # -----------------------------------------------------------------
    # Build a diverse seed pool and pick the best raw seed
    # -----------------------------------------------------------------
    seeds = _collect_seeds(n, rng, prev_arr)
    evaluated = [(s, _objective_surrogate(s)) for s in seeds]
    evaluated.sort(key=lambda x: x[1])
    best_seed, best_val = evaluated[0]
    print(f"Best raw seed surrogate = {best_val:.6f}")

    # -----------------------------------------------------------------
    # LP‑guided refinement (if SciPy is available)
    # -----------------------------------------------------------------
    if _opt is not None and _sp is not None:
        print("\n--- LP refinement phase ---")
        # allocate roughly 70 % of the remaining budget to the LP loop
        lp_deadline = time.time() + (deadline - time.time()) * 0.70
        best_seq, best_val = _lp_refine(best_seed, n, lp_deadline)
        print(f"LP phase finished – best surrogate = {best_val:.6f}")

        # -----------------------------------------------------------------
        # Short sub‑gradient polishing (if time permits)
        # -----------------------------------------------------------------
        if time.time() < deadline:
            print("\n--- Sub‑gradient polishing ---")
            best_seq, best_val = _subgradient_opt(
                best_seq, n, deadline, max_iters=400_000
            )
            print(f"Polishing finished – best surrogate = {best_val:.6f}")
    else:
        # -----------------------------------------------------------------
        # Fallback: only the sub‑gradient polish (no LP)
        # -----------------------------------------------------------------
        print("SciPy not available – falling back to pure sub‑gradient method.")
        best_seq, best_val = _subgradient_opt(best_seed, n, deadline, max_iters=800_000)
        print(f"Fallback polishing finished – best surrogate = {best_val:.6f}")

    # -----------------------------------------------------------------
    # Final official evaluation
    # -----------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next iteration (if any)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
