def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Objective helpers (surrogate works on ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate value = 2·n·max(conv) for a vector that sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True (scale‑invariant) objective used by evaluate_sequence."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta(α,β) on {0,…,n‑1} with half‑offset to avoid zeros."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 0.5) ** (alpha - 1.0)) * ((n - i - 0.5) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p (symmetric, zero at the ends)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_uniform_block(n: int, block_size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of `block_size` indices."""
    idx = rng.choice(n, size=block_size, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / block_size
    return a

        seeds.append(_sin_power(n, p))

    # Random Dirichlet draws (different concentrations)
    conc_list = (0.01, 0.03, 0.07, 0.15, 0.3, 0.6, 1.0, 2.0)
    for conc in conc_list:
        for _ in range(3):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw
            seeds.append(np.sort(raw)[::-1])     # sorted descending
            # a simple random permutation
            seeds.append(raw[rng.permutation(n)])

    # Sparse uniform blocks ------------------------------------------------
    for blk in (120, 200, 300, 400):
        seeds.append(_sparse_uniform_block(n, blk, rng))

    # Include previous solution (if size matches)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Normalise all seeds to ℓ₁‑norm = 1
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500) -> np.ndarray:
    """
    Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smoothed
    max‑convolution.  The optimisation variables are logits; the softmax
    guarantees the simplex constraint.
    """
    if _opt is None:        # SciPy not present – fall back to simple gradient steps
        return _gradient_descent_smooth(start_vec, n, deadline,
                                        beta_schedule, max_iter_per_beta)

    if beta_schedule is None:
        beta_schedule = [
            1.0, 3.0, 7.0, 15.0, 30.0, 60.0,
            120.0, 250.0, 500.0, 1000.0, 2500.0,
            5000.0, 10000.0, 25000.0, 50000.0, 100000.0
        ]

    # Work in logit space – softmax maps back onto the simplex.
    v = np.log(start_vec + 1e-30)

    eps = 1e-30
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            # smooth max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = np.sum(exp_beta)
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a
            w = exp_beta / sum_exp                     # soft weights (sum to 1)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # convolution length = 2n-1
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # extract the n components

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={
                                    'maxiter': max_iter_per_beta,
                                    'disp': False,
                                    'ftol': 1e-12
                                })
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


def _gradient_descent_smooth(start_vec: np.ndarray,
                             n: int,
                             deadline: float,
                             beta_schedule,
                             max_iter_per_beta: int) -> np.ndarray:
    """Very simple (and slow) fallback gradient descent on the smooth objective."""
    a = start_vec.copy()
    logit = np.log(a + 1e-30)
    eps = 1e-30
    for beta in beta_schedule:
        for it in range(max_iter_per_beta):
            if time.time() > deadline:
                return _softmax(logit)
            a = _softmax(logit)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            grad_a = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]
            dot = np.dot(grad_a, a)
            grad_logit = a * (grad_a - dot)
            # simple step‑size schedule
            eta = 0.5 / math.sqrt(it + 1)
            logit -= eta * grad_logit
    return _softmax(logit)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑conv)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Long sub‑gradient descent on the exact max‑conv objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Random perturbation (logit space) – used for restarts
# ----------------------------------------------------------------------
def _random_perturb(seq: np.ndarray,
                    n: int,
                    rng: np.random.Generator,
                    magnitude: float = 0.08) -> np.ndarray:
    """Add a tiny random logit perturbation and re‑softmax."""
    logit = np.log(seq + 1e-30)
    perturb = magnitude * rng.normal(size=n)
    new_log = logit + perturb
    return _softmax(new_log)


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and timing
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 3.0

    # ------------------------------------------------------------------
    # 1. Problem size – a larger n gives a finer discretisation of the
    #    continuous problem and usually yields a slightly smaller bound.
    # ------------------------------------------------------------------
    n = 8192          # power‑of‑two for efficient FFTs

    # ------------------------------------------------------------------
    # 2. Seed pool (rich & diverse)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                # resample previous solution to the new length by linear interpolation
                x_old = np.linspace(0.0, 1.0, prev_arr.size)
                x_new = np.linspace(0.0, 1.0, n)
                prev_arr = np.interp(x_new, x_old, prev_arr)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # ------------------------------------------------------------------
    # 3. Quick surrogate evaluation & keep the best few
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 20                       # limit to a handful of promising starts
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nBest surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 4. Continuation L‑BFGS on the best seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    # β‑schedule – we go up to 1e5 to approximate the true max very closely.
    beta_schedule = [
        1.0, 3.0, 7.0, 15.0, 30.0, 60.0,
        120.0, 250.0, 500.0, 1000.0, 2500.0,
        5000.0, 10000.0, 25000.0, 50000.0, 100000.0
    ]

    # Allocate a moderate time slice per seed (the first few seeds get more)
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] Deadline reached before finishing L‑BFGS on all seeds.")
            break

        remaining = deadline - time.time()
        # we allow up to 0.08·remaining seconds for this seed
        max_iter_beta = max(2500, int(min(8000, remaining * 0.08 * 1e5)))
        print(f"\n--- L‑BFGS on seed #{idx + 1} (max_iter/β={max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq, global_best_val = seq_opt.copy(), val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(2_000_000, remaining * 0.06 * 1e5))
        if max_sub > 0:
            print("\n--- Sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq, global_best_val = seq_sub.copy(), val_sub
                print(f"[SubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Random perturbation + short L‑BFGS re‑starts
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random perturbations + L‑BFGS restarts ---")
        for rep in range(3):
            if time.time() > deadline:
                break
            perturbed = _random_perturb(global_best_seq, n, rng, magnitude=0.07)
            remaining = deadline - time.time()
            # a short, tight run – we keep it very cheap
            max_iter_beta = max(1500,
                                int(min(4000, remaining * 0.05 * 1e5)))
            seq_restart = _lbfgs_smooth_optimize(perturbed,
                                                 n,
                                                 deadline,
                                                 beta_schedule=beta_schedule[:10],
                                                 max_iter_per_beta=max_iter_beta)
            val_restart = _objective_surrogate(seq_restart)
            if val_restart < global_best_val:
                global_best_seq, global_best_val = seq_restart.copy(), val_restart
                print(f"[Restart] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
