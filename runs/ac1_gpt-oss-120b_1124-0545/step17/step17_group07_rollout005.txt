    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three phases:

1. **Diverse seed pool** – analytic families (Beta, power‑law,
   sin‑power, exponential, sparse blocks) and many random Dirichlet
   draws.  All seeds are ℓ₁‑normalised (∑a = 1).

2. **Continuation L‑BFGS on a smoothed max‑convolution** – the non‑smooth
   ``max`` is replaced by a log‑sum‑exp surrogate.
   The smoothing parameter β is increased geometrically from 1 up to

3. **Fine‑tuning** – a long exponentiated‑gradient (mirror‑descent)
   optimisation on the exact max‑convolution, followed by a cheap
   pair‑wise logit‑swap hill‑climber.  The final candidate is evaluated
   with the official ``evaluate_sequence`` routine and stored in the
   global ``height_sequence_1`` variable for possible reuse.

The implementation is deliberately lightweight (no closures, no lambdas)

the budget on a modern two‑core CPU.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (∑ = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact surrogate (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv) for a sequence that already sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta(α,β) on {0,…,n‑1} (half‑offset avoids zeros)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 0.5) ** (alpha - 1.0)) * ((n - i - 0.5) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p – symmetric, zero at the ends."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_uniform_block(n: int, block_size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of `block_size` indices."""
    idx = rng.choice(n, size=block_size, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / block_size
    return a


# ----------------------------------------------------------------------
# Permutation helpers (for richer seeds)
# ----------------------------------------------------------------------
def _alternating_order_from_vec(vec: np.ndarray) -> np.ndarray:
    """Place sorted values alternating from the left and right ends."""
    sorted_desc = np.sort(vec)[::-1]
    n = len(vec)
    out = np.empty_like(vec)
    left, right = 0, n - 1
    for val in sorted_desc:
        if (left + right) % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out


# ----------------------------------------------------------------------
# Seed pool construction (expanded)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse collection of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families ----------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for sigma in (0.04, 0.08, 0.12, 0.2, 0.3):
        seeds.append(_gaussian(n, sigma))

    for p in (0.1, 0.3, 0.5, 0.8, 1.2, 1.5, 2.0):
        seeds.append(_power(n, p))

    for rate in (0.0005, 0.001, 0.003, 0.006, 0.015):
        seeds.append(_exponential(n, rate))
        seeds.append(_double_exponential(n, rate))

    # Beta families – now also very U‑shaped (α<1) and mildly peaked (α>2)
    beta_params = (0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0)
    for a in beta_params:
        seeds.append(_beta_distribution(n, a, a))           # symmetric
        seeds.append(_beta_distribution(n, a, 5.0))         # asymmetric
        seeds.append(_beta_distribution(n, 5.0, a))

    # Sin‑power families – explore higher powers
    for p in (2.0, 4.0, 6.0, 8.0, 12.0):
        seeds.append(_sin_power(n, p))

    # Random Dirichlet draws (dense & sparse)
    conc_list = (0.001, 0.01, 0.05, 0.1, 0.3, 0.7, 1.0, 2.0)
    for conc in conc_list:
        for _ in range(5):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                                 # as‑is
            seeds.append(np.sort(raw)[::-1])                 # descending
            seeds.append(_alternating_order_from_vec(raw))    # spread large values
            # sparsify very low entries
            th = np.quantile(raw, 0.9)
            sparse = np.where(raw > th, raw, 0.0)
            if sparse.sum() > 0.0:
                seeds.append(sparse / sparse.sum())

    # Sparse uniform blocks – many different block sizes
    for blk in (30, 50, 80, 120, 200):
        for _ in range(3):
            seeds.append(_sparse_uniform_block(n, blk, rng))

    # Previous solution ----------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Normalise all seeds (guard against zero‑sum) -------------------------
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float],
                           max_iter_per_beta: int = 500) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) using L‑BFGS‑B on a smoothed max‑conv."""
    if _opt is None:
        # SciPy not available – fall back to a few gradient steps.
        return start_vec.copy()

    # Optimisation variables are logits; softmax always stays on simplex.
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stay on simplex
            conv = _conv_fft(a, a)
            m = np.max(conv)                          # shift for stability
            shift = m
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # Gradient of smoothed max w.r.t. a
            w = exp_beta / sum_exp                    # soft weights, sum to 1
            # derivative of conv = a * a  →  d(conv)/da = 2 * (a * reversed w)
            rev_a = a[::-1]
            grad_a = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]

            # Chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={
                                    'maxiter': max_iter_per_beta,
                                    'disp': False,
                                    'ftol': 1e-12
                                })
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser – exact surrogate
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent polish on the exact max‑conv surrogate."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Cheap pair‑wise hill‑climbing in logit space
# ----------------------------------------------------------------------
def _hill_climbing_opt(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Very cheap pair‑wise logit swap hill‑climbing."""
    rng = np.random.default_rng()
    logit = np.log(start_seq + 1e-30)

    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        sigma = 0.015 / math.sqrt(it)
        delta = rng.normal(scale=sigma)
        new_log = logit.copy()
        new_log[i] += delta
        new_log[j] -= delta
        new_seq = _softmax(new_log)
        new_val = _objective_surrogate(new_seq)

        if new_val < cur_val:
            logit, cur_seq, cur_val = new_log, new_seq, new_val
            if new_val < best_val:
                best_seq, best_val = new_seq.copy(), new_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Small random perturbation (used after a good solution)
# ----------------------------------------------------------------------
def _random_perturb(seq: np.ndarray,
                    n: int,
                    rng: np.random.Generator,
                    magnitude: float = 0.02) -> np.ndarray:
    """Add a tiny random logit perturbation and re‑softmax."""
    logit = np.log(seq + 1e-30)
    new_log = logit + magnitude * rng.normal(size=n)
    return _softmax(new_log)


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and timing
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 3.0

    # ------------------------------------------------------------------
    # 1. Problem size (a compromise between expressiveness and speed)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Build a rich seed pool (including the previous best if present)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # Quick surrogate evaluation (exact for ℓ₁‑normalised vectors)
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    # keep only the very best few – the rest would waste time
    top_k = 20
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print("\nBest surrogate values among raw seeds (top 5 shown):")
    for i, (_, val) in enumerate(evaluated[:5]):
        print(f"  #{i + 1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS with an aggressive β schedule
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    # Very aggressive geometric schedule (ends at 1e6)
    beta_schedule = [1.0, 3.0, 10.0, 30.0, 100.0, 300.0,
                     1_000.0, 3_000.0, 10_000.0, 30_000.0,
                     100_000.0, 300_000.0, 1_000_000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] Deadline reached before finishing L‑BFGS.")
            break

        remaining = deadline - time.time()
        # allocate a modest slice of time per seed (≈5 % of what remains)
        max_iter_beta = max(300, int(0.05 * remaining * 500))  # a few hundred per β
        print(f"\n--- L‑BFGS on seed #{idx + 1} (max_iter/β≈{max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq, global_best_val = seq_opt.copy(), val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Exponentiated‑gradient polish (exact surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp = max(200_000, int(remaining * 3e5))  # many iterations if time permits
        print("\n--- Exponentiated‑gradient polishing ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                             n,
                                             deadline,
                                             max_iters=max_exp)
        if val_exp < global_best_val:
            global_best_seq, global_best_val = seq_exp.copy(), val_exp
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Random perturbation + short L‑BFGS restart (helps escape local minima)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random perturbation + short L‑BFGS restart ---")
        for rep in range(2):
            if time.time() > deadline:
                break
            perturbed = _random_perturb(global_best_seq, n, rng, magnitude=0.07)
            # very short L‑BFGS just to climb out of a shallow basin
            seq_restart = _lbfgs_smooth_optimize(perturbed,
                                                 n,
                                                 deadline,
                                                 beta_schedule=beta_schedule[:6],
                                                 max_iter_per_beta=250)
            val_restart = _objective_surrogate(seq_restart)
            if val_restart < global_best_val:
                global_best_seq, global_best_val = seq_restart.copy(), val_restart
                print(f"[Restart] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Pair‑wise hill‑climbing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc = max(50_000, int(remaining * 2e5))
        print("\n--- Pair‑wise hill‑climbing fine‑tuning ---")
        seq_hc, val_hc = _hill_climbing_opt(global_best_seq,
                                            n,
                                            deadline,
                                            max_iters=max_hc)
        if val_hc < global_best_val:
            global_best_seq, global_best_val = seq_hc.copy(), val_hc
            print(f"[HillClimb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next iteration (if any)
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
