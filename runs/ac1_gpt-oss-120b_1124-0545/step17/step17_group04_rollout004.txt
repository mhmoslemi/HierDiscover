"""
Implementation of iterative LP-based refinement followed by Adam
gradient descent to minimize max auto-convolution of a non‑negative
step‑function.  The algorithm starts from the best known sequence
(`height_sequence_1`) or a high‑entropy Dirichlet seed, performs a few
linear‑programming updates that provably decrease the current
upper‑bound, and finishes with a fast Adam optimisation of the exact
max‑convolution objective.
"""

import time, math, numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

def _softmax(v): ...
def _proj_simplex(v): ...

def _conv_fft(x, y): ...
def _objective(seq): ...

def _build_conv_matrix(f):
    n = len(f)
    L = 2*n - 1
    # each diagonal d (0..n-1) has constant f[d]
    data = [np.full(n, f[d]) for d in range(n)]
    A = diags(data, offsets=range(n), shape=(L, n), format='csr')
    return A

def _lp_update(seq, M):
    A = _build_conv_matrix(seq)
    n = len(seq)
    L = 2*n-1
    c = -np.ones(n)
    b = np.full(L, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

def _adam_optimize(start_seq, deadline, max_iters=20000):
    n = len(start_seq)
    logits = np.log(start_seq + 1e-30)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    best_seq = start_seq.copy()
    best_val = _objective(best_seq)
    t = 0
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(logits)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # compute subgradient for the max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # chain rule through softmax
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v = beta2 * v + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep best
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # optional printing
            if it % 2000 == 0:
                print(f"[Adam] it={it} best={best_val:.6f}")
    return best_seq, best_val

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # margin
    # problem size: choose n
    n = 3000   # trade‑off for LP memory / speed
    # load previous sequence if available, otherwise generate a high‑entropy seed
    prev = globals().get("height_sequence_1")
    if prev is not None and len(prev) == n:
        seq0 = np.array(prev, dtype=np.float64)
        seq0 = np.maximum(seq0, 0.0)
        s = seq0.sum()
        if s > 0:
            seq0 = seq0 / s
        else:
            seq0 = np.full(n, 1.0 / n)
    else:
        # generate a Dirichlet seed with moderate concentration
        conc = 0.3
        seq0 = rng.dirichlet(np.full(n, conc))
    # store best
    best_seq = seq0.copy()
    best_val = _objective(best_seq)
    print(f"[Init] n={n} initial surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. LP based monotone improvement loop
    # ------------------------------------------------------------------
    iter_lp = 0
    while time.time() < deadline and iter_lp < 8:
        conv = _conv_fft(best_seq, best_seq)
        M = float(conv.max())
        print(f"[LP] iteration {iter_lp+1}, current max = {M:.6f}")

        alpha = 0.5
        improved = False
        while alpha > 1e-4:
            cand = (1 - alpha) * best_seq + alpha * g_norm
            cand = cand / cand.sum()   # renormalise to 1
            val = _objective(cand)

            break

"""
Algorithm
---------

1.  Initialise a probability vector `a` (sum = 1) of length *n*.
    The previous best vector (`height_sequence_1`) is used when it has the
    same length, otherwise a high‑entropy Dirichlet draw is taken.

2.  **LP improvement step** – For the current vector `a` we compute the
    maximal auto‑convolution value `M = max(conv(a,a))`.  The linear program

        maximise   Σ g_i
        subject to  conv(a,g) ≤ M          (component‑wise)
                     g_i ≥ 0

    returns a vector `g` that can carry more total mass without increasing the
    current bound.  A line‑search blends `a` and the normalised `g` and keeps
    the improvement if the surrogate value `2·n·max(conv)` decreases.
    A few repetitions drive the bound down monotonically.

3.  **Adam fine‑tuning** – With the sequence on the simplex we optimise the
    exact (non‑smooth) objective `J(a)=2·n·max(conv(a,a))`.  The sub‑gradient
    at the active convolution index is computed analytically and Adam is
    applied in log‑softmax (log‑it) space.  This quickly descends the final
    non‑smooth landscape.

4.  **Local stochastic hill‑climbing** – A short random walk transfers a
    tiny amount of mass between two coordinates; moves that improve the
    surrogate are kept.  This helps to escape tiny plateaus left by Adam.

5.  The best vector found is stored in the global variable
    `height_sequence_1` and returned as a plain list of floats.  All steps
    respect the 1000 s budget.

The code is completely self‑contained, uses only NumPy and SciPy (when
available) and follows the required interface.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (fallbacks are provided)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy.sparse import diags
    _SCIPY = True
except Exception:               # pragma: no cover
    linprog = None
    diags = None
    _SCIPY = False


# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (complexity O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)).
    The sequence is assumed to be normalised (∑seq = 1),
    which makes it identical to the true ``evaluate_sequence`` value.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Linear‑programming utilities
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray):
    """
    Build a sparse matrix A (shape (2n‑1, n)) such that
    A @ g  =  conv(f, g)   (full linear convolution).
    Each diagonal d (0 ≤ d < n) of A contains the constant value f[d].
    """
    n = f.shape[0]
    L = 2 * n - 1
    # data for each diagonal: a constant vector of length n
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=int)          # 0,1,…,n‑1
    A = diags(data, offsets, shape=(L, n), format='csr')
    return A


def _lp_step(current: np.ndarray, M: float):
    """
    Solve the LP described in the algorithm overview.
    Returns the optimal vector g (may be None if the LP fails).
    """
    if not _SCIPY:
        return None
    A = _build_conv_matrix(current)
    n = current.shape[0]
    L = 2 * n - 1
    c = -np.ones(n)                     # minimise -∑g_i  → maximise ∑g_i
    b = np.full(L, M, dtype=np.float64)  # conv(f,g) ≤ M
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Adam optimisation on the exact non‑smooth objective
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray, deadline: float, max_iters: int = 20000):
    """
    Adam optimiser in log‑softmax space for the true max‑convolution objective.
    Returns the best sequence and its surrogate value.
    """
    n = start_seq.shape[0]
    logits = np.log(start_seq + 1e-30)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.1
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1

        a = _softmax(logits)                     # current probability vector
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry of the max‑convolution
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # chain rule through the softmax
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam update in logit space
        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep the best iterate seen so far
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny stochastic hill‑climbing (optional polishing)
# ----------------------------------------------------------------------
def _local_search(seq: np.ndarray, deadline: float, moves: int = 2000):
    """
    Randomly transfer a very small amount of mass between two coordinates.
    Accept only improving moves.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_val = _objective(best_seq)
    rng = np.random.default_rng()
    step = 1e-4                         # absolute amount of mass to move

    for _ in range(moves):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        if best_seq[i] < step:      # cannot take mass from an almost‑empty bin
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        cand = best_seq.copy()
        cand[i] -= step
        cand[j] += step
        # renormalise (should stay ≈1, but numerical drift may appear)
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    The returned list is a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # --------------------------------------------------------------
    # Problem size – a compromise between LP tractability and quality
    # --------------------------------------------------------------
    n = 3000

    # --------------------------------------------------------------
    # Initialise a feasible probability vector
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and len(prev) == n:
        a0 = np.asarray(prev, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 0.0:
            a0 = a0 / s
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        # a wide‑range Dirichlet draw gives a good generic start
        a0 = rng.dirichlet(np.full(n, 0.3))

    best_seq = a0.copy()
    best_val = _objective(best_seq)
    print(f"[Init] n={n}  initial surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. LP‑based monotone improvement loop
    # --------------------------------------------------------------
    if _SCIPY:
        lp_iters = 0
        while time.time() < deadline and lp_iters < 8:
            conv = _conv_fft(best_seq, best_seq)
            M = float(conv.max())
            print(f"[LP] iteration {lp_iters + 1}, current max = {M:.6f}")

            g = _lp_step(best_seq, M)
            if g is None:
                print("[LP] LP failed – abandoning further LP steps")
                break

            sum_g = g.sum()
            if sum_g <= 1.0 + 1e-12:
                print("[LP] No mass increase possible – stop LP loop")
                break

            g_norm = g / sum_g

            # simple back‑tracking line‑search on the mixing coefficient
            alpha = 0.5
            improved = False
            while alpha > 1e-4:
                cand = (1.0 - alpha) * best_seq + alpha * g_norm
                cand /= cand.sum()
                cand_val = _objective(cand)
                if cand_val < best_val:
                    best_seq = cand
                    best_val = cand_val
                    improved = True
                    print(f"[LP] accepted α={alpha:.4f}, new surrogate = {best_val:.6f}")
                    break
                alpha *= 0.5

            if not improved:
                print("[LP] line‑search failed – terminating LP phase")
                break

            lp_iters += 1

    else:
        print("[LP] SciPy not available – skipping LP refinement")

    # --------------------------------------------------------------
    # 2. Adam fine‑tuning on the exact max‑convolution surrogate
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # roughly 5000 iterations per second on a modern CPU (convolution dominates)
        max_iters = int(min(150_000, remaining * 5000))
        max_iters = max(5_000, max_iters)
        print(f"\n[Adam] start fine‑tuning, max_iters={max_iters}")
        best_seq, best_val = _adam_optimize(best_seq, deadline, max_iters=max_iters)
        print(f"[Adam] finished, best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Very cheap local stochastic polishing (optional)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n[Local] start stochastic hill‑climbing")
        best_seq, best_val = _local_search(best_seq, deadline, moves=3000)
        print(f"[Local] finished, best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final evaluation with the official evaluator
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:               # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
