def _lbfgs_obj(v, n, beta):
    a = _softmax(v)
    conv = _conv_fft(a, a)
    m = np.max(conv)
    # stable log-sum-exp
    shift = m
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_beta / sum_exp  # weights
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1:n-1+n]   # extract relevant part length n
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v

def _lbfgs_obj(v, n, beta):
    a = _softmax(v)
    conv = _conv_fft(a, a)
    m = np.max(conv)
    shift = m
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1:n-1+len(a)]
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v
```

def _lbfgs_smooth_optimize(start_vec, n, deadline, beta_schedule=None, max_iter_per_beta=2500):
    if _opt is None:

    if beta_schedule is None:

                                         'ftol': 1e-12})

"""
Propose a non‑negative step‑function that (almost) minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The search pipeline:

1️⃣  Build a rich pool of ℓ₁‑normalised seeds (analytic families,
    random Dirichlet draws and the previous best solution).
2️⃣  Run a continuation L‑BFGS optimiser on a smooth max‑convolution
    surrogate for each top seed.
3️⃣  Polish the current best vector with a long exponentiated‑gradient
    (mirror‑descent) run.
4️⃣  Perform a fine‑grained L‑BFGS run that focuses on the large‑β
    regime (tight approximation of the true max).
5️⃣  Finish with a long projected sub‑gradient descent for exact
    max‑convolution polishing.
6️⃣  Evaluate the final candidate with the official ``evaluate_sequence``.

All optimisation steps respect the 1000 s budget and keep the vector
on the probability simplex (sum = 1, non‑negative).  The best sequence
is stored in the global variable ``height_sequence_1`` for possible
reuse by subsequent calls.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np
import functools

# SciPy is optional – L‑BFGS is used only when available.
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact and surrogate objectives (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate value = 2·n·max(conv) for a sequence with sum = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by the official evaluator (no normalisation)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta(α,β) on {0,…,n‑1} with a half‑offset to avoid zeros."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 0.5) ** (alpha - 1.0)) * ((n - i - 0.5) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p  (symmetric, zero at the ends)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_uniform_block(n: int, block_size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of `block_size` indices."""
    idx = rng.choice(n, size=block_size, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / block_size
    return a


# ----------------------------------------------------------------------
# Permutation helpers (for richer seed pool)
# ----------------------------------------------------------------------
def _alternating_order_from_vec(vec: np.ndarray) -> np.ndarray:
    """Place sorted values alternating from the left and right ends."""
    sorted_desc = np.sort(vec)[::-1]
    n = len(vec)
    out = np.empty_like(vec)
    left, right = 0, n - 1
    for val in sorted_desc:
        if (left + right) % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out


def _shuffle_by_step(sorted_desc: np.ndarray, step: int) -> np.ndarray:
    """Distribute the sorted values over the index set using a coprime step."""
    n = len(sorted_desc)
    out = np.empty(n, dtype=np.float64)
    pos = 0
    for val in sorted_desc:
        out[pos] = val
        pos = (pos + step) % n
    return out


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse collection of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for sigma in (0.05, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))

    for p in (0.0, 0.25, 0.5, 0.75, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))

    for rate in (0.001, 0.003, 0.006, 0.01, 0.015):
        seeds.append(_exponential(n, rate))
        seeds.append(_double_exponential(n, rate))

    # Symmetric Beta families
    for a in (0.3, 0.4, 0.5, 0.6, 0.8, 1.0, 1.2, 1.5, 2.0):
        seeds.append(_beta_distribution(n, a, a))

    # Sin‑power families
    for p in (1.0, 2.0, 4.0, 6.0, 8.0):
        seeds.append(_sin_power(n, p))

    # Random Dirichlet draws – several concentrations
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                             # as‑is
            seeds.append(np.sort(raw)[::-1])              # sorted descending
            seeds.append(_alternating_order_from_vec(raw))
            seeds.append(raw[rng.permutation(n)])         # random permutation
            # Sparse thresholded version
            th = np.quantile(raw, 0.7)
            sparse = np.where(raw > th, raw, 0.0)
            if sparse.sum() > 0.0:
                seeds.append(sparse / sparse.sum())

    # Sparse uniform blocks
    for blk in (80, 100, 150, 200):
        for _ in range(2):
            seeds.append(_sparse_uniform_block(n, blk, rng))

    # Include the previous solution (if compatible)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Normalise everything (guard against accidental zero‑sum vectors)
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# L‑BFGS on a smooth max‑convolution surrogate
# ----------------------------------------------------------------------
def _lbfgs_obj(v: np.ndarray, n: int, beta: float):
    """Return (objective, gradient) for the smooth‑max surrogate."""
    a = _softmax(v)
    conv = _conv_fft(a, a)
    m = np.max(conv)
    shift = m
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_beta / sum_exp                     # soft weights (sum to 1)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length = 3·n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]       # extract the part that matches a
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)                # chain rule through softmax
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smoothed max‑conv."""
    if _opt is None:
        # SciPy not available – skip L‑BFGS entirely.
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 3.0, 7.0, 15.0, 30.0, 60.0,
            120.0, 250.0, 500.0, 1000.0, 2500.0,
            5000.0, 10000.0, 25000.0, 50000.0
        ]

    v = np.log(start_vec + 1e-30)  # work in log‑space (softmax keeps us on simplex)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        try:
            res = _opt.minimize(fun=functools.partial(_lbfgs_obj, n=n, beta=beta),
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser (exact surrogate)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent polish on the exact max‑conv surrogate."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # Sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Long sub‑gradient descent on 2·n·max(conv) (exact surrogate)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # Sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # --------------------------------------------------------------
    # 0. Initialise random generator and time budget
    # --------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 1. Problem size (trade‑off between quality and speed)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 2. Seed pool (rich and diverse)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n:
                prev_arr = arr
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"\n--- Seed generation (n={n}) ---")
    print(f"Collected {len(seeds)} raw seeds.")

    # quick surrogate evaluation (exact for ℓ₁‑normalised vectors)
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nBest surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1:2d}: {val:.6f}")

    # --------------------------------------------------------------
    # 3. Continuation L‑BFGS on top seeds
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    beta_schedule = [
        1.0, 3.0, 7.0, 15.0, 30.0, 60.0,
        120.0, 250.0, 500.0, 1000.0, 2500.0,
        5000.0, 10000.0, 25000.0, 50000.0
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] Deadline reached before finishing all L‑BFGS runs.")
            break

        print(f"\n--- L‑BFGS on seed #{idx + 1} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=2500)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq, global_best_val = seq_opt.copy(), val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

        # ---- short exponentiated‑gradient polish on the current best ----
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_exp = int(min(800_000, remaining * 200))
            max_exp = max(200_000, max_exp)
            seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_exp)
            if val_exp < global_best_val:
                global_best_seq, global_best_val = seq_exp.copy(), val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Random perturbation + short L‑BFGS restart (if time allows)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random perturbation + restart L‑BFGS ---")
        for rep in range(3):
            if time.time() > deadline:
                break
            # tiny random perturbation in log‑space
            logit = np.log(global_best_seq + 1e-30)
            perturbed_logit = logit + 0.08 * rng.normal(size=n)
            perturbed = _softmax(perturbed_logit)

            remaining = deadline - time.time()
            max_iter_beta = int(min(4000, remaining * 0.04 * 200))
            max_iter_beta = max(1500, max_iter_beta)

            seq_restart = _lbfgs_smooth_optimize(perturbed,
                                                 n,
                                                 deadline,
                                                 beta_schedule=beta_schedule[:8],
                                                 max_iter_per_beta=max_iter_beta)
            val_restart = _objective_surrogate(seq_restart)
            if val_restart < global_best_val:
                global_best_seq, global_best_val = seq_restart.copy(), val_restart
                print(f"[Restart] new best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Long projected sub‑gradient polishing (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(2_000_000, remaining * 200))
        max_sub = max(400_000, max_sub)
        print("\n--- Long sub‑gradient polishing ---")
        seq_sub, val_sub = _subgradient_opt(global_best_seq,
                                            n,
                                            deadline,
                                            max_iters=max_sub)
        if val_sub < global_best_val:
            global_best_seq, global_best_val = seq_sub.copy(), val_sub
            print(f"[SubGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next iteration (if any)
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
