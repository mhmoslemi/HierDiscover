"""
Propose a step‑function (list of non‑negative heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is a two‑stage global‑local optimiser:

1. **Coarse global search** – a low‑dimensional representation of the
   sequence (piecewise‑constant on *m* intervals) is optimised with
   SciPy's Differential Evolution (DE).  The parameters are the logits
   of a soft‑max vector that defines the total mass of each interval.
   DE quickly discovers a promising shape for the distribution.

2. **Fine local refinement** – the full‑resolution sequence obtained from
   DE is polished with a mirror‑descent (exponentiated‑gradient) optimiser
   that works directly on the exact max‑convolution objective.
   The optimiser runs until the time budget is exhausted, yielding a
   near‑flat self‑convolution and a low evaluation score.

The code respects the 1000 s time budget, prints progress, stores the
best sequence globally as ``height_sequence_1`` and returns it as a list
of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for the global DE stage)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact when the sequence sums to 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq, seq)) – the quantity that evaluate_sequence
    returns when sum(seq) == 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Build a full‑resolution sequence from low‑dimensional logits
# ----------------------------------------------------------------------
def _build_piecewise_constant(v: np.ndarray, n: int, lengths: np.ndarray) -> np.ndarray:
    """
    Convert a log‑softmax vector `v` of size m into a piecewise‑constant
    probability vector of length `n`.  The `lengths` array contains the
    sizes of the m intervals (they sum to n).  Each interval receives a
    uniform density equal to its softmax weight divided by its length.
    """
    w = _softmax(v)                     # interval masses, sum = 1
    a = np.empty(n, dtype=np.float64)   # final sequence
    pos = 0
    for weight, L in zip(w, lengths):
        a[pos:pos + L] = weight / float(L)
        pos += L
    # In the unlikely case of rounding errors, we clip to the simplex.
    s = a.sum()
    if s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution
    objective.  Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # Sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in the log‑softmax space (v) guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        # SciPy not available – fall back to the input vector unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0, 5_000.0,
                         20_000.0, 50_000.0, 100_000.0, 200_000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max using log‑sum‑exp (shifted for numerical stability)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (as derived from the smoothed max)
            w = exp_beta / sum_exp               # soft‑max on the conv values
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # slice the middle n entries

            # transform gradient back to log‑softmax space
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The search combines a global DE stage over a
    low‑dimensional piecewise‑constant representation with a long
    exponentiated‑gradient polish on the full‑resolution vector.
    The best sequence is stored globally as ``height_sequence_1``.
    """
    # ------------------------------------------------------------------
    # 0. Initialisation
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    n = 4000

    # ------------------------------------------------------------------
    # 1. Load previous solution (if any) for warm‑start
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 2. Low‑dimensional parameterisation (piecewise‑constant)
    # ------------------------------------------------------------------
    m = 80                                    # number of intervals (adjustable)
    base_len = n // m
    remainder = n % m
    # interval lengths – first `remainder` intervals are one element longer
    lengths = np.full(m, base_len, dtype=int)
    lengths[:remainder] += 1
    assert lengths.sum() == n

    # ------------------------------------------------------------------
    # 3. Global optimisation with Differential Evolution (if SciPy available)
    # ------------------------------------------------------------------
    best_sequence = None
    best_surrogate = float('inf')
    de_success = False

    if _opt is not None:
        print("\n--- Global DE optimisation (piecewise‑constant) ---")
        bounds = [(-12.0, 12.0)] * m

        # Callback to respect the time budget
        def de_callback(x, convergence):
            if time.time() > deadline:
                return True   # stop DE
            return False

        # DE parameters – modest values give a quick coarse search
        de_maxiter = 250          # generations
        de_popsize = 20           # individuals per generation

        try:
            result = _opt.differential_evolution(
                func=lambda v, n=n, lengths=lengths: _objective_surrogate(
                    _build_piecewise_constant(v, n, lengths)),
                bounds=bounds,
                args=(),
                strategy='best1bin',
                maxiter=de_maxiter,
                popsize=de_popsize,
                init='latinhypercube',
                polish=False,
                callback=de_callback,
                disp=False,
                workers=1)   # single‑threaded for determinism
            de_success = True
        except Exception as e:  # pragma: no cover
            print("[DE] exception:", e)
            de_success = False

        if de_success:
            v_best = result.x
            seq_de = _build_piecewise_constant(v_best, n, lengths)
            surrogate_de = _objective_surrogate(seq_de)
            best_sequence = seq_de
            best_surrogate = surrogate_de
            print(f"[DE] best surrogate = {best_surrogate:.6f}")

    # ------------------------------------------------------------------
    # 4. Fallback seed generation (if DE failed or for extra diversity)
    # ------------------------------------------------------------------
    # We keep a small pool of hand‑crafted seeds (uniform, arcsine, etc.)
    seeds = []

    # Uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Arcsine / Beta(0.5, 0.5)
    arcsine = _beta_shape(n, 0.5, 0.5) if ' _beta_shape' in globals() else None
    if arcsine is not None:
        seeds.append(arcsine)

    # Triangular
    seeds.append(_beta_shape(n, 2.0, 2.0))  # roughly triangular

    # If we have a previous solution, add a perturbed version
    if prev_arr is not None:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(3):
            noisy = prev_arr + 0.005 * rng.normal(size=n)
            noisy = np.maximum(noisy, 0.0)
            s = noisy.sum()
            if s > 0:
                seeds.append(noisy / s)

    # Evaluate these cheap seeds and keep the best one
    for s in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(s)
        if val < best_surrogate:
            best_surrogate = val
            best_sequence = s.copy()
    if best_sequence is None:
        # As a safety, fall back to uniform
        best_sequence = np.full(n, 1.0 / n, dtype=np.float64)
        best_surrogate = _objective_surrogate(best_sequence)

    print(f"\n[Init] current best surrogate after seeding = {best_surrogate:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

                                          n,

        if val_eg < best_surrogate:
            best_sequence = seq_eg
            best_surrogate = val_eg
            print(f"[EG] improved surrogate = {best_surrogate:.6f}")

    # ------------------------------------------------------------------
    # 6. Final smooth optimisation with L‑BFGS (optional, time‑aware)

    return best_sequence.tolist()
