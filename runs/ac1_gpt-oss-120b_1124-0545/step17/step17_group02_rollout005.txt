"""
propose_candidate – hybrid meta‑heuristic for the max‑convolution bound

The routine builds a diverse pool of ℓ₁‑normalized “seed’’ vectors (uniform,
triangular, exponential, Gaussian, Beta, random Dirichlet, sparse,
low‑frequency cosine mixtures and sorted Dirichlet variants).  The best
few seeds are polished with a continuation L‑BFGS run on a smooth
log‑sum‑exp surrogate of the objective.  From the best polished vector
a temperature‑controlled simulated‑annealing phase performs random swaps
and mass‑transfer moves, accepting uphill steps with probability
exp(−Δ/T).  Finally a short high‑β L‑BFGS polishing is applied and the
result is stored globally as ``height_sequence_1``.

All steps respect the global 1000 s budget; the function returns the
best sequence found (as a plain list of non‑negative floats).
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Exact objective & helpers
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² (sum may be ≠1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth surrogate of the max‑convolution objective.
    Returns (obj, grad) where the gradient is w.r.t. the logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)               # full convolution
    shift = np.max(conv)                 # numeric stabiliser
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # Gradient w.r.t. a
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # Back‑prop through the softmax (log‑sum‑exp ↔ simplex)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 1500) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0,
            100000.0, 200000.0, 500000.0,
            1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun,
                                x0=v,
                                method="L-BFGS-B",
                                jac=True,
                                callback=callback,
                                options={"maxiter": max_iter_per_beta,
                                         "disp": False,
                                         "ftol": 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)

# Simulated annealing on the exact objective
# ----------------------------------------------------------------------
def _simulated_annealing(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         initial_T: float = 0.04,
                         min_T: float = 1e-4,
                         step_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled random walk that performs random swaps and
    mass‑transfer moves.  Accepts uphill moves with probability exp(−Δ/T).
    """
    rng = np.random.default_rng(12345)
    cur = start_a.copy()
    cur_val = _objective_exact(cur)
    best = cur.copy()
    best_val = cur_val

    start = time.time()
    total = deadline - start

    it = 0
    while time.time() < deadline:
        it += 1
        # linear temperature schedule
        elapsed = time.time() - start
        T = max(min_T, initial_T * (1.0 - elapsed / total))

        # choose move type
        if rng.random() < 0.5:
            # ---- swap two entries ----
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # ---- transfer a small fraction of mass from i to j ----
            i = rng.integers(0, n)
            if cur[i] <= 0.0:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            # fraction up to step_frac of the source mass
            eps = rng.random() * step_frac
            delta = eps * cur[i]
            if delta <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        cand_val = _objective_exact(cand)
        d = cand_val - cur_val

        accept = False
        if d < 0.0:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-d / T):
                accept = True

        if accept:
            cur = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
                # occasional progress report
                if it % 5000 == 0:
                    print(f"[Anneal] iter={it} cur={cur_val:.6f} best={best_val:.6f}")

    return best, best_val

def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse pool of ℓ₁‑normalised vectors (length n)."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_beta_shape(n, 2.0, 2.0))

    # exponential grid
    for lam in np.logspace(-4, -0.6, 20):
        seeds.append(_exponential_decay(n, lam))

    # Gaussian grid
    for sigma in np.linspace(0.05, 0.4, 10):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet (different concentrations) + sorted variants
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            v = _random_dirichlet(n, conc, rng)
            seeds.append(v)
            # a descending‑sorted version (often useful)
            seeds.append(np.sort(v)[::-1])

    # random sparse vectors
    for frac in (0.05, 0.1, 0.2):
        for _ in range(3):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures of random pairs
    base = np.arange(len(seeds))
    for _ in range(150):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of a few promising seeds
    for _ in range(15):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best if available
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up – non‑negativity + ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

            print("[Time] deadline before finishing L‑BFGS runs.")
            break
        # allocate a modest per‑β iteration budget (depends on remaining time)
        remaining = deadline - time.time()
        per_beta = max(800, int(min(2500, remaining * 0.06 * 1e5)))
        print(f"\n--- L‑BFGS run #{idx + 1}/{len(elite)} (max_iter/β={per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=per_beta)
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – simulated annealing from the current best
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing (mass transfers & swaps) ---")
        seq_anneal, val_anneal = _simulated_annealing(global_best_seq,
                                                     n,
                                                     deadline,
                                                     initial_T=0.05,
                                                     min_T=1e-4,
                                                     step_frac=0.025)
        print(f"[Anneal] best = {val_anneal:.6f}")
        if val_anneal < global_best_val:
            global_best_seq = seq_anneal.copy()
            global_best_val = val_anneal
            print(f"[Update] New best after annealing = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – final high‑β polishing L‑BFGS (very few iterations)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(600, int(min(1800, remaining * 0.04 * 1e5)))
        print("\n--- Final polishing L‑BFGS (very high β) ---")
        seq_final = _lbfgs_smooth_optimize(global_best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=[500_000.0,
                                                          1_000_000.0,
                                                          2_000_000.0,
                                                          5_000_000.0],
                                           max_iter_per_beta=per_beta)
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging)
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for warm‑starts in future invocations
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
