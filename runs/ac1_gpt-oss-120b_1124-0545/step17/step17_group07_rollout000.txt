# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (single FFT per call)
# ----------------------------------------------------------------------
def _make_conv_fn(n: int):
    """Factory returning a fast linear convolution a*a → np.ndarray."""
    L = 2 * n - 1
    N = _next_pow2(L)

    def conv(a: np.ndarray) -> np.ndarray:
        # a is assumed to be length n, sum(a)=1
        fa = np.fft.rfft(a, n=N)
        # autocorrelation = IFFT(|FA|^2)
        res = np.fft.irfft(fa * fa, n=N)[:L]
        return res

    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _surrogate_val(a: np.ndarray, conv_fn) -> float:
    """Return 2·n·max(conv) for a probability vector a."""
    return float(2.0 * a.shape[0] * np.max(conv_fn(a)))


# ----------------------------------------------------------------------
# Analytic families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    a /= a.sum()
    return a


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    a /= a.sum()
    return a


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta(α,β) on {0,…,n‑1} with half‑offset to avoid zeros."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 0.5) ** (alpha - 1.0)) * ((n - i - 0.5) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _sin_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p (symmetric, zero at the ends)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n) ** p
    a /= a.sum()
    return a

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector and return it (already ℓ₁‑normalised)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_uniform_block(n: int, block_size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of `block_size` indices."""
    idx = rng.choice(n, size=block_size, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / block_size
    return a

# Seed pool construction

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a large and diverse set of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # Analytic families ----------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.05, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))
    for p in (0.0, 0.25, 0.5, 0.75, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for rate in (0.001, 0.003, 0.006, 0.01, 0.015):
        seeds.append(_exponential(n, rate))
    # symmetric beta, many parameters
    beta_params = np.geomspace(0.1, 3.0, 12)  # 12 log‑spaced values
    for a in beta_params:
        seeds.append(_beta_distribution(n, a, a))

    # Sin‑power ------------------------------------------------------------
    for p in (1.0, 2.0, 4.0, 6.0, 8.0):
        seeds.append(_sin_power(n, p))

    # Random Dirichlet draws (huge pool) ---------------------------------
    # concentration drawn log‑uniformly between 0.005 and 5.0
    conc_samples = rng.uniform(np.log(0.005), np.log(5.0), size=2000)
    for logc in conc_samples:
        conc = float(np.exp(logc))
        raw = _random_dirichlet(n, conc, rng)
        seeds.append(raw)                          # as‑is
        seeds.append(np.sort(raw)[::-1])          # sorted descending
        # a random permutation (adds diversity)
        seeds.append(raw[rng.permutation(n)])

    # Sparse blocks ---------------------------------------------------------
    for blk in (80, 120, 200):
        for _ in range(3):
            seeds.append(_sparse_uniform_block(n, blk, rng))

    # Previous solution -----------------------------------------------------
    if previous is not None and previous.size == n and np.isfinite(previous).all():
        # normalise just in case
        prev = previous.copy()
        total = prev.sum()
        if total > 0:
            prev /= total
            seeds.append(prev)

    # Ensure every seed is a proper probability vector --------------------
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimizer (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    conv_fn,
    beta_schedule=None,
    max_iter_per_beta: int = 4000,
) -> np.ndarray:
    """
    Refine ``start_vec`` (ℓ₁‑normalised) using L‑BFGS‑B on a smoothed version
    of max(conv).  The variables are logits so the simplex constraint is
    handled by a soft‑max.
    """
    if _opt is None:
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 3.0, 7.0, 15.0, 30.0, 60.0,
            120.0, 250.0, 500.0, 1000.0, 2500.0,
            5000.0, 10000.0, 25000.0, 50000.0, 100000.0,
        ]

    # initialise logits (log‑softmax)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = conv_fn(a)
            m = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = m
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of smooth max w.r.t. a
            w = exp_beta / sum_exp  # soft weights, sum to 1
            # convolution of rev(a) with w
            rev_a = a[::-1]
            # we can reuse the FFT routine (conv of rev_a with w)
            fa = np.fft.rfft(rev_a, n=_next_pow2(2 * n - 1))
            fw = np.fft.rfft(w, n=_next_pow2(2 * n - 1))
            full = np.fft.irfft(fa * fw, n=_next_pow2(2 * n - 1))
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Fast projected sub‑gradient descent with Nesterov momentum
# ----------------------------------------------------------------------
def _subgrad_momentum_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    conv_fn,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective 2·n·max(conv)
    using Nesterov momentum (mu≈0.9) and a decreasing step‑size
    lr = 0.6 / sqrt(t).  The routine stops as soon as time runs out.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_val(a, conv_fn)

    # momentum buffer
    v = np.zeros_like(a)
    mu = 0.9

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and max index
        conv = conv_fn(a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]  # derivative of max(conv) w.r.t a

        # step size schedule
        lr = 0.6 / math.sqrt(it)

        # Nesterov update: v = mu*v - lr*g ; a = proj(a + v)
        v = mu * v - lr * g
        a = _proj_simplex(a + v)

        cur_val = _surrogate_val(a, conv_fn)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

) -> tuple[np.ndarray, float]:
    """
    Very cheap pair‑wise logit‑swap hill‑climbing.  Works in logit space,
    because the softmax already guarantees feasibility.
    """
    rng = np.random.default_rng()
    logit = np.log(start_seq + 1e-30)

    cur_seq = start_seq.copy()
    cur_val = _surrogate_val(cur_seq, conv_fn)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        sigma = 0.015 / math.sqrt(it)
        delta = rng.normal(scale=sigma)
        new_log = logit.copy()
        new_log[i] += delta
        new_log[j] -= delta
        new_seq = _softmax(new_log)
        new_val = _surrogate_val(new_seq, conv_fn)
        if new_val < cur_val:
            logit, cur_seq, cur_val = new_log, new_seq, new_val
            if new_val < best_val:
                best_seq, best_val = new_seq.copy(), new_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 123,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the allocated time budget (default 1000 s) and stores
    the best sequence found in the global variable ``height_sequence_1``.
    """
    # ------------------------------------------------------------------
    # 0. RNG and time budget
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 3.0

    # ------------------------------------------------------------------
    # 1. Problem size – a compromise between resolution and speed
    # ------------------------------------------------------------------
    n = 4000  # same size as the previous best solution

    # ------------------------------------------------------------------
    # 2. Prepare a fast convolution helper for this ``n``
    # ------------------------------------------------------------------
    conv_fn = _make_conv_fn(n)

    # ------------------------------------------------------------------
    # 3. Seed pool (very large, diverse)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n:
                prev_arr = arr.copy()
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # quick surrogate evaluation (exact for ℓ₁‑normalised vectors)
    evaluated = [(vec, _surrogate_val(vec, conv_fn)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 30
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nBest surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 4. Short L‑BFGS refinement on the very best seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _surrogate_val(global_best_seq, conv_fn)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    beta_schedule = [
        1.0, 3.0, 7.0, 15.0, 30.0, 60.0,
        120.0, 250.0, 500.0, 1000.0, 2500.0,
        5000.0, 10000.0, 25000.0, 50000.0, 100000.0,
    ]

    # limit the expensive L‑BFGS to only the top few seeds
    for idx, seed_vec in enumerate(top_seeds[:5]):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate a modest slice of remaining time
        max_iter_beta = int(min(6000, remaining * 0.03 * 1e5))
        max_iter_beta = max(2500, max_iter_beta)

        print(f"\n--- L‑BFGS on seed #{idx+1} (max_iter/β={max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            conv_fn,
            beta_schedule=beta_schedule,
            max_iter_per_beta=max_iter_beta,
        )
        val_opt = _surrogate_val(seq_opt, conv_fn)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq, global_best_val = seq_opt.copy(), val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Long projected sub‑gradient descent with momentum (main phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allow up to ~80 % of the remaining budget for this phase
        max_iters = int(min(5_000_000, remaining * 0.8 * 1e3))
        print("\n--- Momentum sub‑gradient polishing ---")
        seq_sub, val_sub = _subgrad_momentum_opt(
            global_best_seq, n, deadline, conv_fn, max_iters=max_iters
        )
        if val_sub < global_best_val:
            global_best_seq, global_best_val = seq_sub.copy(), val_sub
            print(f"[SubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Very cheap hill‑climbing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_hc = int(min(800_000, remaining * 0.2 * 1e3))
        if max_iters_hc > 0:
            print("\n--- Logit hill‑climbing fine‑tuning ---")
            seq_hc, val_hc = _hill_climb_logit_opt(
                global_best_seq, n, deadline, conv_fn, max_iters=max_iters_hc
            )
            if val_hc < global_best_val:
                global_best_seq, global_best_val = seq_hc.copy(), val_hc
                print(f"[HillClimb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
