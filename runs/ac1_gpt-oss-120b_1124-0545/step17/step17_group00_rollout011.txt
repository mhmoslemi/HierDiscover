"""
Propose a step‑function (non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The routine combines three ideas:

1. **Starting point** – reuse the best sequence discovered in a previous run
   (global variable ``height_sequence_1``).  If none is available a uniform
   distribution is used.

2. **Simulated‑annealing on the probability simplex** – repeatedly perform a
   cheap random move (small mass transfer, a swap of two coordinates or a
   cyclic shift) and accept it if it improves the surrogate objective
   ``2·n·max(conv)`` or with a probability that decreases with a linear
   temperature schedule.  The convolution is evaluated with an FFT (O(n log n))
   and the current best solution is kept.

3. **Final polishing** – a short exponentiated‑gradient (mirror‑descent) run
   and an optional L‑BFGS smooth optimisation are applied on the remaining
   time budget.

All numbers are kept non‑negative and normalised to sum 1, which makes the
surrogate objective exactly equal to the true ``evaluate_sequence`` result
(up to the clipping performed by the evaluator).  The function respects a
1000 s time limit and stores the best found sequence in ``height_sequence_1``
for reuse by later calls.
"""

import time
import math
import heapq
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Numeric utilities (FFT‑based convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact evaluation for a probability vector (sum == 1):
        2·n·max(conv(seq, seq))
    where n = len(seq).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = np.exp(log_a)
        a = a / a.sum()          # stay on the simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    The optimisation works in log‑softmax space, guaranteeing a simplex‑
    feasible point.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0]

    v = np.log(start_vec + 1e-30)   # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = np.exp(v_local)
            a = a / a.sum()
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                        # soft‑max weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                    # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = np.exp(v)
    best_a = best_a / best_a.sum()
    return best_a

# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine follows the scheme described in the module
    docstring and returns the best sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # a small safety margin – the final call to evaluate_sequence also costs time
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Initialise from the stored solution (if any), otherwise uniform
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) == n:
        a = np.asarray(prev, dtype=np.float64)
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total > 0:
            a = a / total
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    cur_seq = a.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    print(f"[Init] start surrogate = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing loop (mass transfer, swaps, cyclic shifts)
    # ------------------------------------------------------------------
    # Temperature starts as a few per‑cent of the current objective
    T0 = cur_val * 0.02
    iteration = 0
    no_improve_counter = 0
    while time.time() < deadline:
        iteration += 1

        # linear temperature decay
        remaining = deadline - time.time()
        fraction = remaining / (deadline - start_time)
        temperature = T0 * fraction

        move_type = rng.choice(["transfer", "swap", "shift"], p=[0.6, 0.3, 0.1])

        if move_type == "transfer":
            i, j = rng.integers(n, size=2)
            if i == j or cur_seq[i] <= 0.0:
                continue
            # Early in the search allow larger moves, later shrink them
            max_frac = 0.5 if temperature > T0 * 0.1 else 0.05
            frac = rng.uniform(1e-6, max_frac)
            eps = cur_seq[i] * frac
            if eps <= 0.0 or cur_seq[i] - eps < 0.0:
                continue
            new_seq = cur_seq.copy()
            new_seq[i] -= eps
            new_seq[j] += eps
        elif move_type == "swap":
            i, j = rng.integers(n, size=2)
            if i == j:
                continue
            new_seq = cur_seq.copy()
            new_seq[i], new_seq[j] = cur_seq[j], cur_seq[i]
        else:   # shift
            shift = rng.integers(1, n)  # non‑zero shift
            new_seq = np.roll(cur_seq, shift)

        # Guard against drifting away from the simplex because of floating‑point
        new_seq = np.maximum(new_seq, 0.0)
        s = new_seq.sum()
        if s == 0.0:
            continue
        new_seq = new_seq / s

        # evaluate surrogate (fast)
        val_new = _objective_surrogate(new_seq)

        delta = val_new - cur_val
        accept = False
        if delta < 0.0:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-delta / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_seq = new_seq
            cur_val = val_new
            no_improve_counter = 0
            if val_new < best_val:
                best_val = val_new
                best_seq = new_seq.copy()
                elapsed = time.time() - start_time
                print(f"[SA] new best = {best_val:.6f} (iter {iteration}, time {elapsed:.1f}s)")
        else:
            no_improve_counter += 1

        # occasional random restart of the *current* point to explore fresh basins
        if no_improve_counter > 50000:
            # perturb the current best a little and continue from there
            eps_vec = rng.dirichlet(np.full(n, 0.01))
            new_seq = best_seq * 0.9 + eps_vec * 0.1
            new_seq = new_seq / new_seq.sum()
            cur_seq = new_seq
            cur_val = _objective_surrogate(cur_seq)
            no_improve_counter = 0

    # ------------------------------------------------------------------
    # Final polishing – short exponentiated‑gradient run
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        eg_deadline = deadline - 4.0
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, eg_deadline,
                                           max_iters=800_000,
                                           eta0=5.0)
        if val_eg < best_val:
            best_seq = seq_eg
            best_val = val_eg
            print(f"[ExpGrad‑final] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional L‑BFGS refinement if time still remains
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0 and _opt is not None:
        try:
            seq_lb = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                            beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                           200.0, 500.0, 1000.0,
                                                           2000.0, 5000.0,
                                                           10000.0, 20000.0],
                                            max_iter_per_beta=15000)
            val_lb = _objective_surrogate(seq_lb)
            if val_lb < best_val:
                best_seq = seq_lb
                best_val = val_lb
                print(f"[L‑BFGS‑final] refined to {best_val:.6f}")
        except Exception as e:  # pragma: no cover
            print("[L‑BFGS‑final] exception:", e)

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised:", exc)

    # store for future calls
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
