"""
Optimisation of a non‑negative step‑function (height sequence) for the
``evaluate_sequence`` benchmark.

The algorithm proceeds in several stages, all respecting the 1000 s
budget:

1. **Extensive seed generation** – analytic families (uniform, triangular,
   exponential, Gaussian, beta, power‑law, sine‑window, …) are swept over
   a dense grid of parameters and a large collection of random Dirichlet
   / sparse vectors is added.  All seeds are ℓ₁‑normalised (sum = 1).

2. **Quick exact evaluation** – the best k seeds (by the true objective)
   are kept for heavy optimisation.

3. **Continuation L‑BFGS** on a log‑sum‑exp smooth surrogate of
   ``max(conv(a,a))``.  The schedule of the smoothing parameter β is
   aggressive and the inner optimiser is allowed many iterations.

4. **Exponentiated‑gradient (mirror descent)** – multiple restarts with a
   large number of steps (the sub‑gradient of the exact max‑convolution
   is used).  An adaptive step‑size η₀/√t is employed with η₀ ≈ 5.

5. **Adam‑type fine‑tuning** on the same smooth surrogate with very large β.

6. **Stochastic simulated‑annealing local search** – hill‑climbing moves
   that transfer a random fraction of mass between two positions; worse
   moves are accepted with a temperature schedule to escape shallow minima.

All stages constantly track the best sequence found (global_best_seq)
and finally return it as a plain list.  The sequence is also stored in
the global name ``height_sequence_1`` for warm‐starts on the next run.

The code is written without closures or lambdas and uses only NumPy,
SciPy and the standard library.
"""

# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families – all return ℓ₁‑normalised vectors (sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    s = a.sum()
    return a / s


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a vector from Dirichlet(conc,…,conc)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random subset of size m = max(2, int(frac*n))."""
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)

def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors of length n."""
    seeds: list[np.ndarray] = []

    # Deterministic families with richer parameter grids
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_linear_decrease(n))

    # Exponential decay – fine log‑grid
    for lam in np.logspace(-5, -1, num=20):
        seeds.append(_exponential_decay(n, float(lam)))

    # Gaussian – fine linear grid on sigma fraction
    for sigma in np.linspace(0.08, 0.45, 15):
        seeds.append(_gaussian(n, float(sigma)))

    # Sine window – a wide range of exponents
    for g in np.linspace(0.2, 4.0, 20):
        seeds.append(_sine_window(n, float(g)))

    # Beta shapes – coarse grid, then refined around promising regions
    alphas = np.linspace(0.5, 8.0, 9)
    betas = np.linspace(0.5, 8.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, float(a), float(b)))

    # Power‑law – up to fairly steep decay
    for p in np.linspace(0.0, 6.0, 21):
        seeds.append(_power_law(n, float(p)))

    # Random Dirichlet draws – different concentrations
    for conc in (0.02, 0.05, 0.1, 0.3, 0.6, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, float(conc), rng))

    # Random sparse vectors – various sparsities
    for frac in (0.02, 0.05, 0.1, 0.2, 0.35, 0.5):
        for _ in range(4):
            seeds.append(_random_sparse(n, float(frac), rng))

    # Low‑frequency curves – multiple frequencies
    for nf in (2, 3, 5, 8):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Pairwise mixtures of generated seeds
    base = np.arange(len(seeds))
    for _ in range(250):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], float(alpha)))

    # Rotations of promising seeds (adds diversity)
    for _ in range(40):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], int(shift)))

    # Include previous solution if supplied
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Clean: force non‑negative, ℓ₁‑normalise
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

# Smooth surrogate (log‑sum‑exp) of max(conv) and its gradient
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Log‑sum‑exp surrogate for max(conv(a,a)).
    Returns (objective, gradient w.r.t. logits v).
    """
    a = _softmax(v)                     # probability vector on the simplex
    conv = _conv_fft(a, a)               # full convolution, length 2n‑1
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # weights of the smooth‑max
    w = exp_term / sum_exp               # length 2n‑1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 100000.0,
            200000.0, 500000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits (strictly positive)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method="L-BFGS-B",
                                jac=True,
                                callback=callback,
                                options={"maxiter": max_iter_per_beta,
                                         "disp": False,
                                         "ftol": 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during β={beta_val}")
            break
        except Exception as e:      # pragma: no cover
            print(f"[L‑BFGS] exception {e} (β={beta_val})")
            break

    return _softmax(v)


def _adam_smooth_optimize(start_vec: np.ndarray,
                          n: int,
                          deadline: float,
                          beta: float,
                          max_steps: int,
                          lr_initial: float = 0.5) -> np.ndarray:
    """Adam optimiser on the smooth surrogate (single β)."""
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta_val = float(beta)

    for _ in range(max_steps):
        if time.time() > deadline:
            break
        loss, grad = _smooth_objective(v, n, beta_val)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)
        lr = lr_initial / np.sqrt(t)
        v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact objective and sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (sum(a)=1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv


def _subgradient_maxconv_multi(a: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient averaging over all lags whose convolution value is
    within ``eps`` of the current maximum.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    idxs = np.where(conv >= max_c - eps)[0]
    g = np.zeros_like(a)
    for idx in idxs:
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - i_range
        g[i_range] += 2.0 * a[rev_idx]
    if len(idxs) > 0:
        g /= len(idxs)
    return g


def _exp_gradient_optimize_multi(start_a: np.ndarray,
                                n: int,
                                deadline: float,
                                max_iters: int = 500_000,
                                eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Returns the best vector seen and its objective value.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_maxconv_multi(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 150_000 == 0:
            cur = _objective_exact(a)
            print(f"[ExpGrad] it={it:,} obj={cur:.6f}")
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing style stochastic local search
# ----------------------------------------------------------------------
def _simulated_annealing(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         total_iters: int = 150_000,
                         temp_start: float = 1e-2,
                         temp_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing with Metropolis acceptance.  Randomly transfers a
    fraction of mass from one index to another.  Temperature drops linearly.
    """
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456)

    for it in range(total_iters):
        if time.time() > deadline:
            break
        # linear temperature schedule
        T = temp_start + (temp_end - temp_start) * (it / total_iters)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue

        # move a random fraction (up to 15 %) of the source mass
        frac = rng.random() * 0.15
        delta = frac * a[i]
        if delta <= 0.0:
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        cur_val = _objective_exact(a_candidate)
        if cur_val < best_val or rng.random() < math.exp(-(cur_val - best_val) / T):
            # accept
            a = a_candidate
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function minimising ``evaluate_sequence`` within the
    allotted time budget.  Returns the best sequence (list of floats) and
    stores it globally as ``height_sequence_1`` for warm‑starts.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a slightly larger n gives a finer search space.
    # ------------------------------------------------------------------
    n = 5000
    print(f"\n=== Optimisation started, n={n}, budget≈{budget_s}s ===")

    # previous solution (if any)
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation
    # ------------------------------------------------------------------
    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"[Seeds] generated {len(seeds)} candidates.")

    # ------------------------------------------------------------------
    # 2. Quick exact evaluation – keep the top‑k seeds
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for vec in seeds:
        val = _objective_exact(vec)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best raw objective = {best_val:.6f}")

    # sort and retain the very best seeds for heavy optimisation
    seeds.sort(key=_objective_exact)
    top_k = min(25, len(seeds))
    top_seeds = seeds[:top_k]

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS on the smooth surrogate
    # ------------------------------------------------------------------
    global_best_seq = best_seq.copy()
    global_best_val = best_val

    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0, 100_000.0,
        200_000.0, 500_000.0, 1_000_000.0,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline before finishing L‑BFGS.")
            break
        # allocate a modest but generous number of max‑iter steps per β
        per_beta = 1500
        print(f"\n--- L‑BFGS run #{idx+1} (max_iter/β={per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=per_beta)
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multiple exponentiated‑gradient restarts (exact max‑conv)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        runs = 12
        iters_per_run = int(max(200_000, min(800_000, remaining * 0.12 / runs)))
        print("\n--- Exponentiated‑gradient restarts (exact max‑conv) ---")
        for r in range(runs):
            if time.time() > deadline:
                break
            # random Dirichlet start
            conc = rng.uniform(0.02, 2.5)
            start_vec = _random_dirichlet(n, float(conc), rng)
            seq_exp, val_exp = _exp_gradient_optimize_multi(start_vec,
                                                            n,
                                                            deadline,
                                                            max_iters=iters_per_run,
                                                            eta0=5.0)
            print(f"[EG restart {r+1}] best = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] new best after EG restart = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Long EG optimisation from current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(5_000_000, remaining * 0.25))
        if max_iters > 0:
            print("\n--- Long EG optimisation (starting from current best) ---")
            seq_exp, val_exp = _exp_gradient_optimize_multi(global_best_seq,
                                                            n,
                                                            deadline,
                                                            max_iters=max_iters,
                                                            eta0=6.0)
            print(f"[Long EG] best = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] new best after long EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Adam refinement with very high β
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print("\n--- Adam refinement (very high β) ---")
        a = global_best_seq.copy()
        for beta in [500_000.0, 1_000_000.0, 2_000_000.0, 5_000_000.0]:
            if time.time() > deadline:
                break
            steps = int(min(5000, (deadline - time.time()) * 0.04))
            if steps <= 0:
                break
            a = _adam_smooth_optimize(a, n, deadline, beta, max_steps=steps, lr_initial=0.5)
            cur_val = _objective_exact(a)
            print(f"[Adam β={beta:.0f}] objective = {cur_val:.6f}")
            if cur_val < global_best_val:
                global_best_seq = a.copy()
                global_best_val = cur_val
                print(f"[Update] new best after Adam = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Simulated‑annealing stochastic local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(300_000, remaining * 120))
        if iters > 0:
            print("\n--- Simulated‑annealing local search ---")
            seq_loc, val_loc = _simulated_annealing(global_best_seq,
                                                    n,
                                                    deadline,
                                                    total_iters=iters,
                                                    temp_start=1e-2,
                                                    temp_end=1e-6)
            print(f"[SA local] best = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] new best after SA = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (purely for logging)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate best ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
