   (up to 5·10⁶) and each stage is allowed many LBFGS iterations,

The procedure never exceeds the provided time budget and always returns

# Numerical utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a vector normalised to sum = 1:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, NW: float = 2.5) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window, if SciPy is available."""
    if _sigwin is None:
        raise RuntimeError("SciPy signal windows not available.")
    w = _sigwin.dpss(n, NW, Kmax=1, sym=True)[0]
    s = w.sum()
    return w / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random generators (return ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet with all concentration parameters = conc."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mixed_with_uniform(base: np.ndarray, rng: np.random.Generator,
                        mix_factor: float = None) -> np.ndarray:
    """Return λ·base + (1‑λ)·uniform with random λ∈[0,1] (or given)."""
    if mix_factor is None:
        lam = rng.random()
    else:
        lam = float(mix_factor)
    uniform = np.full_like(base, 1.0 / len(base))
    mixed = lam * base + (1.0 - lam) * uniform
    s = mixed.sum()
    return mixed / s if s > 0 else uniform


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 20_000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variable lives in log‑softmax space (v), guaranteeing non‑negativity
    and simplex constraint.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        # extended schedule – final β pushes the smooth approximation close to the true max
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0, 2_000_000.0, 5_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # stays on simplex
            conv = _conv_fft(a, a)            # full convolution
            max_c = np.max(conv)
            shift = max_c                     # stabilise exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # soft‑max weights over convolution entries
            w = exp_beta / sum_exp

            # gradient wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-14})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Simple hill‑climbing / simulated‑annealing local refinement
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer moves on the simplex.  Each step picks two indices i,j
    and moves a random fraction of a[i] to a[j]; accept only improving moves.
    Returns the best sequence found and its surrogate value.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        eps = cur[i] * rng.uniform(0.001, 0.01)   # 0.1 %–1 % of a[i]
        if eps <= 0.0 or cur[i] - eps < 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective_surrogate(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val

# Main candidate proposer

def propose_candidate(seed: int = 123,

                _push_candidate(prev_norm)
        except Exception:

    # ------------------------------------------------------------------
    # 4. Extract the best seeds (ordered from best to worst)

    # ------------------------------------------------------------------

            break
        # small random perturbation of the current best
        noise = rng.normal(scale=0.02, size=n)
        pert = global_best_seq + noise
        pert = np.clip(pert, 0.0, None)
        pert /= pert.sum()
        per_deadline = time.time() + max(3.0, (deadline - time.time()) * 0.15)
        try:
            seq_opt = _lbfgs_smooth_optimize(pert,
                                             n,
                                             per_deadline,
                                             beta_schedule=None,
                                             max_iter_per_beta=20_000)
            val_opt = _objective_surrogate(seq_opt)
            if val_opt < global_best_val:
                global_best_seq = seq_opt
                global_best_val = val_opt
                print(f"[Perturb pass {p}] improved to {global_best_val:.6f}")
        except Exception as e:  # pragma: no cover
            print(f"[Perturb pass {p}] exception: {e}")

    # ------------------------------------------------------------------
    # 8. Final greedy hill‑climb (use any leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline, rng)
        if val_hc < global_best_val:
            global_best_seq = seq_hc
            global_best_val = val_hc
            print(f"[Hill‑climb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
