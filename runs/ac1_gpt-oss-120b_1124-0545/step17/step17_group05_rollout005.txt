def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, math, random
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2

    n = 4000

    # load previous solution if exists
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(prev, list) and len(prev) == n:
        prev_arr = np.asarray(prev, dtype=np.float64)

    # generate seeds
    seeds = []
    # add analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    # random Dirichlet with various concs
    for conc in [0.01, 0.05, 0.1, 0.2, 0.5, 1.0]:
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    # add heavy-tailed anti-pair
    heavy = _random_dirichlet(n, 0.02, rng)
    for _ in range(5):
        perm = heavy[rng.permutation(n)]
        seeds.append(perm)

    # add previous if not None
    if prev_arr is not None:
        seeds.append(prev_arr/prev_arr.sum())

    # evaluate cheap surrogate and keep top few
    sur_values = [_objective_surrogate(v) for v in seeds]
    top_idx = np.argsort(sur_values)[:8]
    top_seeds = [seeds[i] for i in top_idx]

    # keep best found
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)

    # iterate through top seeds
    for seed_idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline: break
        # short L-BFGS smoothing
        seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline,
                                         beta_schedule=[1,5,20,80,200,500,1000],
                                         max_iter_per_beta=2000)
        val_opt = _objective_surrogate(seq_opt)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt.copy()
        # exponentiated gradient for a while
        remaining = deadline - time.time()
        max_iters = int(min(2_000_000, remaining * 0.15))
        seq_opt, val_opt = _exp_gradient_improved(best_seq, n, deadline,
                                                  max_iters=max_iters,
                                                  rng=rng)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt.copy()
        # optional permutation swaps if time left after loops

    # final permutation hill climbing
    if time.time() < deadline:
        max_swaps = int(min(800_000, (deadline - time.time()) * 0.2))
        best_seq, best_val = _perm_swap_opt(best_seq, n, deadline, max_swaps)

    # final evaluation
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as e:
        true_val = None
        print("Evaluation error:", e)

    globals()["height_sequence_1"] = best_seq.tolist()
    print(f"Best surrogate {best_val:.6f}, official {true_val}")
    return best_seq.tolist()

def _random_transfer_opt(seq, n, deadline, max_moves):
    cur = seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    for t in range(max_moves):
        if time.time() > deadline: break
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # pick amount epsilon up to a[i] * 0.5
        eps = rng.random() * cur[i] * 0.5
        if eps == 0: continue
        cur[i] -= eps
        cur[j] += eps
        new_val = _objective_surrogate(cur)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            # revert
            cur[i] += eps
            cur[j] -= eps
        if (t+1) % 50000 == 0:
            print(f"[Transfer] moves={t+1} best={best_val:.6f}")
    return best, best_val

"""
propose_candidate:  Heuristic search for a non‑negative step‑function with a low
max‑autocorrelation value (the objective of ``evaluate_sequence``).

Algorithm
---------

1. **Seed generation** – a diverse pool of ℓ₁‑normalised vectors:
   analytic windows, heavy‑tailed Dirichlet draws, an “alternating” ordering
   (large values opposite small values) and the previous solution (if any).

2. **Cheap pre‑screening** – evaluate the cheap surrogate
   ``2·n·max(conv)`` for every seed and keep the best few.

3. **Exponentiated‑gradient (mirror descent) on the exact max‑convolution**.
   This is the core optimiser.  The update uses the sub‑gradient of the
   active lag (the lag that attains the maximum).  A decreasing step‑size
   `ηₜ = 0.5 / √t` is employed, and every 50 000 iterations a small Gaussian
   perturbation is added to the log‑probabilities to escape shallow local
   minima.  The loop runs as long as time permits (up to a few million
   iterations).

4. **Random mass‑transfer local search** – repeatedly move a tiny random
   amount of probability mass from one index to another; keep the move only
   if it improves the surrogate.  This explores directions that the pure
   gradient step cannot reach.

5. **Swap‑based permutation hill‑climber** – as a final polishing step,
   randomly swap two positions and accept the swap if the surrogate improves.

6. **Return** the best sequence found (stored globally as ``height_sequence_1``).

All steps respect a global 1000 s budget and use only NumPy (SciPy is optional
for an L‑BFGS warm‑start, but the algorithm works without it).
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Optional SciPy import – used only for a quick L‑BFGS warm‑start
# --------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# --------------------------------------------------------------
# Low‑level utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate (ℓ₁‑normalised vectors):
    2·n·max(conv(seq, seq)).
    The input is assumed to already sum to 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# --------------------------------------------------------------
# Analytic families – they always return ℓ₁‑normalised vectors
# --------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Largest values on the left, smallest on the right (anti‑paired)."""
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a modest but diverse set of ℓ₁‑normalised seeds."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 1.0, 2.0, 4.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))

    # Random Dirichlet – a few concentration values
    for conc in (0.02, 0.05, 0.1, 0.3, 0.8, 1.5):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Heavy‑tailed alternating Dirichlet
    seeds.append(_alternating_dirichlet(n, rng))

    # Previous solution (if it matches the current size)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise sanity‑check (just in case)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

# L‑BFGS smooth optimisation (optional – warm‑start)
# --------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimise the log‑sum‑exp smoothed max‑convolution using L‑BFGS‑B.
    Returns an ℓ₁‑normalised probability vector.
    """
    if _opt is None:
        # SciPy not available – skip the L‑BFGS stage.
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1e3, 2e3, 5e3,
                         1e4, 2e4, 5e4, 1e5,
                         2e5, 5e5, 1e6]

    # work in log‑probability space (ensure we stay on the simplex)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # ℓ₁‑normalised
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (exact for the smoothed max)
            w = exp_beta / sum_exp                     # weights over each lag
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # slice the part w.r.t. a

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        # simple callback that aborts if we run out of time
        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:          # pragma: no cover
            print("[L‑BFGS] exception:", e)
            break

    return _softmax(v)


# --------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑convolution
# --------------------------------------------------------------
def _exp_gradient_annealed(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent optimisation of the exact max‑convolution.
    Periodically injects a small Gaussian perturbation to escape plateaus.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    epoch = 50000                # how often to inject noise / print progress
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient for the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # classic 1/√t step size
        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % epoch == 0:
            # small Gaussian noise in log‑space (magnitude slowly decays)
            noise_scale = 0.1 / math.sqrt(it)
            log_a += rng.normal(scale=noise_scale, size=n)
            a = _softmax(log_a)
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# --------------------------------------------------------------
# Random mass‑transfer local search (tiny moves)
# --------------------------------------------------------------
def _random_transfer_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_moves: int,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small random amount of probability mass from one index
    to another; keep the move only if the surrogate improves.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    report_every = max_moves // 10 if max_moves >= 10 else 1

    for mv in range(1, max_moves + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if cur[i] == 0.0:
            continue

        # move a random fraction (up to 50 %) of the mass at i towards j
        eps = rng.random() * cur[i] * 0.5
        cur[i] -= eps
        cur[j] += eps

        new_val = _objective_surrogate(cur)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            # revert
            cur[i] += eps
            cur[j] -= eps

        if mv % report_every == 0:
            print(f"[Transfer] moves={mv:,} best={best_val:.6f}")

    return best, best_val

# Swap‑based permutation hill‑climber (final polishing)
# --------------------------------------------------------------
def _perm_swap_opt(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_swaps: int) -> tuple[np.ndarray, float]:
    """
    Randomly swap two positions; keep the swap only if it reduces the exact
    max‑convolution. Returns the best sequence and its surrogate value.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    report_every = max_swaps // 10 if max_swaps >= 10 else 1

    for sw in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _objective_surrogate(cur_seq)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if sw % report_every == 0:
            print(f"[PermSwap] swaps={sw:,} best={best_val:.6f}")

    return best_seq, best_val
