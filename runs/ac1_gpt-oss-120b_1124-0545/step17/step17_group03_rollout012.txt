"""
Propose a step‑function (non‑negative “heights”) that minimizes
``evaluate_sequence```.  The algorithm works in three stages:

1. **Diverse seed pool** – a large collection of analytically‑generated
   probability vectors (uniform, triangular, beta, Gaussian, power‑law,
   low‑frequency cosine sums, random Dirichlet / sparse draws, mixtures,
   rotations).  All vectors are ℓ₁‑normalised (sum = 1).

2. **Continuation L‑BFGS on a smooth log‑sum‑exp surrogate** – for the
   best few seeds we run a deterministic quasi‑Newton optimizer on the
   smoothed objective  ``log‑sum‑exp(β·conv)``.  The smoothing parameter
   ``β`` is increased gradually (continuation) so that the final iterate
   approximates the true max‑convolution objective.

3. **Mirrored exponentiated‑gradient (mirror descent) + stochastic
   local search** – starting from the best L‑BFGS result we perform a
   long exact sub‑gradient descent (mirror descent on the simplex) and,
   finally, a hill‑climbing “mass‑transfer” refinement.

All computations use fast FFT‑based convolutions (O(n log n)).  The
routine respects the 1000 s budget, periodically checks the deadline
and stops cleanly.  The best sequence found is stored globally as
``height_sequence_1`` and returned as a plain list.
"""

import time

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for moderately sized n)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _linear_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-9) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)

# ----------------------------------------------------------------------
# Exact objective & sub‑gradients (max‑convolution)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact value 2·n·max(conv(a,a)) for a with ℓ₁‑norm = 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _subgradient_maxconv_multi(a: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient averaged over all (almost) max‑convolution indices.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    idxs = np.where(conv >= max_c - eps)[0]
    g = np.zeros_like(a)
    for idx in idxs:
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - i_range
        g[i_range] += 2.0 * a[rev_idx]
    if len(idxs):
        g /= len(idxs)
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) for L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Returns (obj, grad) for the log‑sum‑exp smooth approximation of
    max(conv(a,a)), where a = softmax(v).
    """
    a = _softmax(v)                     # probability vector
    conv = _conv_fft(a, a)               # length 2n‑1
    shift = np.max(conv)                 # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_term / sum_exp               # soft‑max weights, sum=1, len=2n‑1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 3000,
) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    ``start_vec`` must be ℓ₁‑normalised.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 100000.0,
            200000.0, 500000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits (log‑domain)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(x):
            return _smooth_objective(x, n, beta_val)

        def callback(x):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during β={beta_val}")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} (β={beta_val})")
            break

    return _softmax(v)   # probability vector (ℓ₁‑norm = 1)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize_multi(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 2.5,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective using the
    multi‑subgradient.  Returns the best vector encountered and its
    exact objective value.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_maxconv_multi(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        # occasional check for improvement
        if it % 200_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val, best_a = cur, a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val, best_a = cur, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Stochastic mass‑transfer local search
# ----------------------------------------------------------------------
def _mass_transfer_local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    total_iters: int = 200_000,
    step_frac_start: float = 0.12,
    step_frac_end: float = 0.008,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: move a fraction of mass from a random source
    index to a random destination index.  The transferred fraction decays
    linearly from ``step_frac_start`` to ``step_frac_end``.
    """
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(987654321)

    for it in range(total_iters):
        if time.time() > deadline:
            break

        frac = step_frac_start + (step_frac_end - step_frac_start) * (it / total_iters)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue

        delta = frac * a[i]
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        val = _objective_exact(cand)
        if val < best_val:
            best_val, best_a, a = val, cand.copy(), cand

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function minimising ``evaluate_sequence`` within the
    allotted time budget.  Returns the best sequence as a plain list and
    stores it globally as ``height_sequence_1`` for warm‑starts.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a small safety margin

    # Problem size – 4000 balances expressiveness and speed
    n = 4000
    print(f"\n=== Optimisation started (n={n}, budget≈{budget_s}s) ===")

    # ------------------------------------------------------------------
    # 1. Seed pool
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} raw seeds")

    # ------------------------------------------------------------------
    # 2. Quick exact evaluation of seeds – keep the best few
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for vec in seeds:
        val = _objective_exact(vec)
        if val < best_val:
            best_val, best_seq = val, vec.copy()
    print(f"[Init] best exact objective among raw seeds = {best_val:.6f}")

    # Pick a small set of promising seeds for L‑BFGS
    seeds.sort(key=_objective_exact)
    top_k = min(30, len(seeds))
    top_seeds = seeds[:top_k]

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS (more generous iteration budget)
    # ------------------------------------------------------------------
    global_best_seq = best_seq.copy()
    global_best_val = best_val

    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        10000.0, 20000.0, 50000.0, 100000.0,
        200000.0, 500000.0, 1_000_000.0, 2_000_000.0,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS runs.")
            break

        print(f"\n--- L‑BFGS run #{idx+1} (seed {idx+1}/{top_k}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_schedule,
            max_iter_per_beta=4000,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] exact objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq, global_best_val = seq_opt.copy(), val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long exponentiated‑gradient refinement (exact objective)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate roughly 60 % of the remaining budget to EG
        max_iters = int(min(8_000_000, remaining * 0.6 * 1e5))
        print("\n--- Exponentiated‑gradient refinement (exact max‑conv) ---")
        seq_exp, val_exp = _exp_gradient_optimize_multi(
            global_best_seq,
            n,
            deadline,
            max_iters=max_iters,
            eta0=3.0,
        )
        print(f"[EG] best objective after refinement = {val_exp:.6f}")
        if val_exp < global_best_val:
            global_best_seq, global_best_val = seq_exp.copy(), val_exp
            print("[Update] New global best after EG = {:.6f}".format(global_best_val))

    # ------------------------------------------------------------------
    # 5. Intensive stochastic local search (mass transfer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(400_000, remaining * 300))
        print("\n--- Stochastic local search (mass‑transfer) ---")
        seq_loc, val_loc = _mass_transfer_local_search(
            global_best_seq,
            n,
            deadline,
            total_iters=iters,
            step_frac_start=0.12,
            step_frac_end=0.008,
        )
        print(f"[LocalSearch] best = {val_loc:.6f}")
        if val_loc < global_best_val:
            global_best_seq, global_best_val = seq_loc.copy(), val_loc
            print("[Update] New global best after local search = {:.6f}".format(global_best_val))

    # ------------------------------------------------------------------
    # Final evaluation with the official API (for logging only)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
