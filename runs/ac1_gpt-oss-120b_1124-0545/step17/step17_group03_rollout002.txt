def _simulated_annealing(start_a, n, deadline, max_iters, init_temp, final_temp):
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)
    rng = np.random.default_rng()
    for it in range(1, max_iters+1):
        if time.time() > deadline: break
        # temperature schedule linear to it/max_iters
        t = init_temp + (final_temp - init_temp) * (it / max_iters)
        # propose move: choose two indices i, j, move some fraction of mass delta from i to j
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or a[i] <= 0: continue
        # move fraction up to some fraction allowed
        frac = rng.uniform(0.0, 0.2)  # fraction of a[i] to move
        delta = frac * a[i]
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        # evaluate
        val = _objective_exact(a_candidate)
        delta_val = val - best_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / max(t, 1e-12)):
            # accept
            a = a_candidate
            if val < best_val:
                best_val = val
                best_a = a_candidate.copy()
    return best_a, best_val

"""
Propose a step‑function (probability vector) that improves the previous best bound
≈ 1.5085. The search pipeline:

1. Generate a very diverse pool of analytically‑parameterised and random seeds.
   The parameter ranges are intentionally broader than before.
2. Keep the top‑k seeds (by exact objective) and run a continuation L‑BFGS on a
   log‑sum‑exp smooth surrogate of the max‑convolution.  A long β‑schedule is
   used and the number of L‑BFGS iterations per β is adjusted to the remaining
   time budget.
3. Run several restarts of exponentiated‑gradient (mirror‑descent) on the exact
   max‑convolution using a multi‑subgradient.  The step size decays as 1/√t.
4. Refine the current best solution with a simulated‑annealing “mass‑transfer”
   local search that sometimes accepts worsening moves, which helps escape
   shallow local minima.
5. A final Adam‑type optimisation with a very large β further smoothes the
   surrogate.
6. Return the best vector found and store it globally as ``height_sequence_1``.
   All vectors are kept on the probability simplex (non‑negative, sum = 1).

The routine is time‑limited (default 1000 s) and automatically adapts the
workload to the remaining budget.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a point on the simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (each returns ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    s = a.sum()
    return a / s


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = a0 - a1 * np.cos(2 * np.pi * i / (n - 1)) + a2 * np.cos(4 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                # Modified Bessel I0
    s = a.sum()
    return a / s


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    """Choose a random subset of size m = max(2, int(frac*n)) and put equal mass."""
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


def _random_mixture_two_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random convex mixture of two independent Dirichlet draws."""
    a = _random_dirichlet(n, conc=rng.uniform(0.05, 2.0), rng=rng)
    b = _random_dirichlet(n, conc=rng.uniform(0.05, 2.0), rng=rng)
    alpha = rng.random()
    return _mix_shapes(a, b, alpha)

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a list of diverse ℓ₁‑normalised vectors of length n."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_linear_decrease(n))
    for lam in (0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05):
        seeds.append(_exponential_decay(n, lam))
    for g in (0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12, 15, 20):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian family (wider sigma range)
    for sigma in np.linspace(0.05, 0.55, 9):
        seeds.append(_gaussian(n, sigma))

    # Power‑law family
    for p in np.linspace(0.0, 5.0, 26):
        seeds.append(_power_law(n, p))

    # Beta family – broader grid
    for a in np.linspace(0.1, 8.0, 13):
        for b in np.linspace(0.1, 8.0, 13):
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(10):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse patterns
    for frac in (0.02, 0.05, 0.1, 0.2, 0.3, 0.5):
        for _ in range(6):
            seeds.append(_random_sparse(n, frac, rng))

    # Low‑frequency curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Random mixtures of two Dirichlets
    for _ in range(150):
        seeds.append(_random_mixture_two_dirichlet(n, rng))

    # Rotations of a few random seeds
    for _ in range(30):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # Include previous solution if supplied
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Clean and ℓ₁‑normalise
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

# Surrogate objective (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Log‑sum‑exp smooth surrogate of the exact objective.
    Returns (obj, grad) where grad is w.r.t. the logits v."""
    a = _softmax(v)                     # point on the simplex
    conv = _conv_fft(a, a)               # length = 2n‑1
    shift = np.max(conv)                 # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # Smooth‑max weights
    w = exp_term / sum_exp               # length = 2n‑1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice length n

    # Chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 1500,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 100000.0,
            200000.0, 500000.0, 1_000_000.0,
        ]
    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline during β={beta_val}")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} (β={beta_val})")
            break

    return _softmax(v)

def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (assumes sum(a)=1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _subgradient_maxconv_multi(a: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient over all (almost) maximal lags."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    idxs = np.where(conv >= max_c - eps)[0]
    g = np.zeros_like(a)
    for idx in idxs:
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - i_range
        g[i_range] += 2.0 * a[rev_idx]
    if len(idxs):
        g /= len(idxs)
    return g


def _exp_gradient_optimize_multi(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        g = _subgradient_maxconv_multi(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 100_000 == 0:
            cur = _objective_exact(a)
            print(f"[ExpGrad] it={it:,} obj={cur:.6f}")
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------
def _adam_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta: float,
    max_steps: int,
    lr_initial: float = 0.5,
) -> np.ndarray:
    """Adam‑type optimiser on the smooth surrogate (single β)."""
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta_val = float(beta)

    for _ in range(max_steps):
        if time.time() > deadline:
            break
        loss, grad = _smooth_objective(v, n, beta_val)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)
        lr = lr_initial / np.sqrt(t)
        v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing “mass‑transfer” local search
# ----------------------------------------------------------------------
def _simulated_annealing(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    temp_start: float = 0.2,
    temp_end: float = 0.001,
) -> tuple[np.ndarray, float]:
    """Hill‑climbing with occasional uphill moves (temperature schedule)."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)
    rng = np.random.default_rng(123456789)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # linear temperature schedule
        T = temp_start + (temp_end - temp_start) * (it / max_iters)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        # move a random fraction of mass from i to j
        frac = rng.uniform(0.0, 0.2)
        delta = frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        val = _objective_exact(cand)
        delta_val = val - best_val
        if delta_val < 0.0 or rng.random() < np.exp(-delta_val / max(T, 1e-12)):
            a = cand
            if val < best_val:
                best_val = val
                best_a = cand.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function minimising ``evaluate_sequence`` within the
    allocated time budget.  Returns the best sequence (list of floats) and
    stores it globally as ``height_sequence_1`` for warm‑starting.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # Choose problem size.  8000 points gives a finer discretisation; if we are
    # already low on time we fall back to 4000.
    n = 8000 if (deadline - time.time()) > 200.0 else 4000
    print(f"\n=== Optimisation start: n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # 1. Seed generation
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} raw seeds.")

    # ------------------------------------------------------------------
    # 2. Quick exact evaluation of all seeds – keep the top‑k
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for vec in seeds:
        val = _objective_exact(vec)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best raw objective = {best_val:.6f}")

    # Keep a modest top‑k for the expensive L‑BFGS stage
    top_k = min(30, len(seeds))
    seeds.sort(key=_objective_exact)
    top_seeds = seeds[:top_k]

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS on a smooth surrogate
    # ------------------------------------------------------------------
    global_best_seq = best_seq.copy()
    global_best_val = best_val

    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        10000.0, 20000.0, 50000.0, 100000.0,
        200000.0, 500000.0, 1_000_000.0,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] budget exhausted before finishing L‑BFGS.")
            break
        remaining = deadline - time.time()
        # allocate a budget proportional to remaining time but bounded
        per_beta = max(500, int(min(3000, remaining * 0.04 * 1e5)))
        print(f"\n--- L‑BFGS run #{idx+1} (max_iter/beta={per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_seq = seq_opt.copy()
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Exponentiated‑gradient (mirror descent) restarts
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        restarts = 10
        iters_per_restart = int(min(300_000, remaining * 0.10 / restarts))
        print("\n--- Exponentiated‑gradient restarts ---")
        for r in range(restarts):
            if time.time() > deadline:
                break
            conc = rng.uniform(0.05, 2.0)
            start_vec = _random_dirichlet(n, conc, rng)
            seq_eg, val_eg = _exp_gradient_optimize_multi(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_restart,
                eta0=2.5,
            )
            print(f"[EG {r+1:02d}] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_val = val_eg
                global_best_seq = seq_eg.copy()
                print(f"[Update] New best after EG restart = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Long EG run starting from current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(2_000_000, remaining * 0.20 * 1e5))
        if max_iters > 0:
            print("\n--- Long EG optimisation (starting from current best) ---")
            seq_eg, val_eg = _exp_gradient_optimize_multi(
                global_best_seq,
                n,
                deadline,
                max_iters=max_iters,
                eta0=3.0,
            )
            print(f"[Long EG] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_val = val_eg
                global_best_seq = seq_eg.copy()
                print(f"[Update] New best after long EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Adam refinement with very high β values
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam refinement with increasing β ---")
        a = global_best_seq.copy()
        for beta in [200_000.0, 500_000.0, 1_000_000.0, 2_000_000.0]:
            if time.time() > deadline:
                break
            steps = int(min(4000, (deadline - time.time()) * 0.04 * 1e5))
            if steps <= 0:
                break
            a = _adam_smooth_optimize(a, n, deadline, beta, max_steps=steps, lr_initial=0.5)
            cur_val = _objective_exact(a)
            print(f"[Adam β={beta:.0f}] objective = {cur_val:.6f}")
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
                print(f"[Update] New best after Adam = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Simulated‑annealing local search (allows uphill moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(400_000, remaining * 150))
        if iters > 0:
            print("\n--- Simulated‑annealing local search ---")
            seq_sa, val_sa = _simulated_annealing(
                global_best_seq,
                n,
                deadline,
                max_iters=iters,
                temp_start=0.15,
                temp_end=0.001,
            )
            print(f"[SA] best = {val_sa:.6f}")
            if val_sa < global_best_val:
                global_best_val = val_sa
                global_best_seq = seq_sa.copy()
                print(f"[Update] New best after SA = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final evaluation with the official evaluator (for reporting)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
