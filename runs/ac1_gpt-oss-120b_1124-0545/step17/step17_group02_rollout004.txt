    # evaluate seeds, keep top

    elite = [v for v,_ in seed_vals[:15]]
    best_seq = elite[0]
    best_val = _objective_exact(best_seq)
    print(f"Initial best objective: {best_val}")

    # L-BFGS runs
    beta_sched = [1.0, 10.0, 50.0, 200.0, 800.0, 3000.0, 10000.0, 40000.0, 200000.0, 1e6, 5e6]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:

            break
        # compute remaining time and per_beta iteration count
        remaining = deadline - time.time()
        # allocate up to e.g., 3000 iterations per beta
        per_beta = max(500, int(min(3000, remaining*0.03*1e5)))  # adjust

        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    if time.time() < deadline:

            # run small L-BFGS

            per_beta = max(400, int(min(2000, remaining_local*0.02*1e5)))

            val = _objective_exact(seq)

            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print("Improved after perturb.")
    # Local search mass transfer
    if time.time() < deadline:

def _collect_seeds_new(n, rng):

        seeds.append(_gaussian(n, sigma_frac))
    # beta shape scanning
    for alpha in [0.5, 1.0, 1.5,2.0,3.0]:
        for beta in [0.5, 1.0, 1.5,2.0,3.0]:
            seeds.append(_beta_shape(n, alpha, beta))
    # power law scanning
    for p in [0.5,0.8,1.0,1.2,1.5]:
        seeds.append(_power_law(n, p))
    # raised cosine
    seeds.append(_raised_cosine(n))

    # random Dirichlet conc scanning
    for conc in [0.05,0.1,0.2,0.5,1.0,2.0]:
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse
    for frac in [0.05,0.1,0.2,0.3]:
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # lowfreq curves
    for nf in [2,3,4]:
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixture of random pairs
    base = list(range(len(seeds)))
    for _ in range(200):
        i,j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of random seeds
    for _ in range(20):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # maybe include previous sequence
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())
        else:
            # upsample or truncate using interpolation
            # if length diff, interpolate to length n:
            x_old = np.linspace(0,1,len(arr))
            x_new = np.linspace(0,1,n)
            arr_interp = np.interp(x_new, x_old, arr)
            seeds.append(arr_interp/arr_interp.sum())
    # ensure non-negative and normalized
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0)
        s = v.sum()

"""
Propose a step‑function (non‑negative sequence) that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm:
1. Build a large and diverse pool of ℓ₁‑normalised “seed’’ vectors
   (analytic families, random Dirichlet, sparse, low‑frequency,
   mixtures, rotations, plus an up‑sampled previous best if available).
2. Evaluate all seeds exactly (fast FFT) and keep the elite few.
3. For each elite seed run a continuation L‑BFGS on a smooth
   log‑sum‑exp surrogate of the max‑convolution.  The softmax
   parametrisation guarantees the ℓ₁‑constraint.
4. Perturb the current best (add Gaussian noise in log‑space) and
   re‑optimise a few times to escape local minima.
5. If time permits run a few mirror‑descent (exponentiated‑gradient)
   steps on the exact objective.
6. Finish with an aggressive stochastic mass‑transfer local search
   and a short high‑β polishing L‑BFGS run.
All stages respect the global 1000 s deadline.  The best sequence is
stored in the global ``height_sequence_1`` for warm‑starts.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (each returns an ℓ₁‑normalised vector)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (i - (n - 1) / 2.0) / (n - 1)) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_squared(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Exact objective (used for evaluation and ranking)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (sum(a))² (sum may be ≠1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) for the active lag (full convolution)."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate for continuation L‑BFGS
# ----------------------------------------------------------------------
class _SmoothObj:
    """Callable wrapper required by scipy.optimize (no closures)."""
    __slots__ = ("n", "beta")

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, v: np.ndarray):
        a = _softmax(v)                     # ℓ₁‑normalised
        conv = _conv_fft(a, a)
        shift = np.max(conv)                # stability shift
        exp_term = np.exp(self.beta * (conv - shift))
        sum_exp = exp_term.sum()
        obj = 2.0 * self.n * (shift + np.log(sum_exp) / self.beta)

        # gradient w.r.t. a
        w = exp_term / sum_exp               # soft‑max over lags
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)           # length = 3n‑2
        grad_a = 2.0 * full[self.n - 1: self.n - 1 + self.n]

        # back‑prop through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return obj, grad_v


class _DeadlineCallback:
    """Stops L‑BFGS when the global deadline is reached."""
    __slots__ = ("deadline",)

    def __init__(self, deadline: float):
        self.deadline = deadline

    def __call__(self, xk):
        if time.time() > self.deadline:
            raise KeyboardInterrupt()


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 1500,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0,
                         1_000_000.0, 2_000_000.0]

    v = np.log(start_vec + 1e-30)   # logits (unconstrained)
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        obj = _SmoothObj(n, float(beta))
        cb = _DeadlineCallback(deadline)
        try:
            res = _opt.minimize(
                fun=obj,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=cb,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break
    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 100_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val

# Stochastic mass‑transfer local search
# ----------------------------------------------------------------------
def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 30_000,
    step_frac: float = 0.015,
) -> tuple[np.ndarray, float]:
    """Hill‑climbing: move a small fraction of mass from i to j."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(987654321)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors (length n)."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_exponential_decay(n, 0.001))
    seeds.append(_exponential_decay(n, 0.003))
    seeds.append(_gaussian(n, 0.15))
    seeds.append(_gaussian(n, 0.30))
    seeds.append(_beta_shape(n, 0.5, 0.5))
    seeds.append(_beta_shape(n, 1.0, 2.0))
    seeds.append(_beta_shape(n, 2.0, 2.0))
    seeds.append(_power_law(n, 0.7))
    seeds.append(_power_law(n, 1.2))
    seeds.append(_raised_cosine(n))
    seeds.append(_sin_squared(n))

    # Random Dirichlet families
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse families
    for frac in (0.05, 0.1, 0.2, 0.4):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # Low‑frequency curves
    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Pairwise mixtures of random seeds
    base = np.arange(len(seeds))
    for _ in range(250):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Rotations of promising seeds
    for _ in range(30):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # Up‑sampled previous best, if it exists
    if previous is not None and previous.size > 0:
        if previous.size == n:
            prev = previous.copy()
        else:
            # interpolate linearly to required length
            x_old = np.linspace(0.0, 1.0, previous.size)
            x_new = np.linspace(0.0, 1.0, n)
            prev = np.interp(x_new, x_old, previous)
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0.0:
            seeds.append(prev / s)

    # Final clean‑up – ensure non‑negativity + ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

        if val_final < best_val:
            best_val = val_final
            best_seq = seq_final.copy()
            print(f"[Update] New best after final polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (for logging)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally for warm‑starts in future invocations
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
