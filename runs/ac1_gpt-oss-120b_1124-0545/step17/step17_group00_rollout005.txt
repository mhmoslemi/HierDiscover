"""
Improved step‑function search for minimise

    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------

1. **Seeding** – generate a diverse set of ℓ₁‑normalised candidates:
   analytic windows (uniform, sine, Kaiser, Gaussian, β‑shapes, etc.),
   Dirichlet draws with several concentrations, exponential draws and a few
   low‑frequency cosine mixtures.  The ten best (by the exact surrogate
   2·n·max(conv)) are kept.

2. **Exponentiated‑gradient polishing** – starting from the current best
   candidate we run mirror‑descent (log‑space updates) on the exact
   max‑convolution.  The learning‑rate schedule ηₜ = η₀/√t and a large number of
   iterations quickly drives the maximum convolution down.

3. **Active‑set linear‑program refinement** – with the current vector `f`
   we compute `M = max(conv(f,f))`.  The active set consists of those indices
   where the convolution attains (or is very close to) `M`.  We then solve

        maximise    Σ g_j
        subject to  g ≥ 0
                    conv(f, g)[i] ≤ M   for all i in the active set

   (a tiny linear programme).  The optimiser returns a `g` with a larger ℓ₁
   mass while respecting the bound.  After normalising `g` we move a fraction
   `t` towards it:

        f ← (1‑t)·f + t·ĝ      (ĝ = g / Σg)

   and repeat the active‑set update until no improvement or time runs out.

4. **Continuation L‑BFGS** – a smooth surrogate based on the log‑sum‑exp
   approximation of the max is minimised with L‑BFGS‑B in the log‑softmax
   parametrisation.  A long β‑schedule (1 → 5·10⁵) makes the surrogate converge
   to the true max.

5. **Local hill‑climbing** – greedy mass‑transfer moves between two random
   coordinates finish the optimisation.

All stages respect the overall time budget (default 1000 s).  The final vector
is stored in the global variable ``height_sequence_1`` for reuse.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a vector already ℓ₁‑normalised (sum=1):
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Analytic families (return ℓ₁‑normalised vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)           # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with all concentration parameters = conc."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superpose a few low‑frequency cosines, clip to ≥0 and renormalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation on exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Works in log‑space (soft‑max guarantees simplex constraints).
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # initialise log‑probabilities
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (see comment in original code)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 15000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    The optimisation variable lives in log‑softmax space guaranteeing the simplex
    and non‑negativity.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0, 5_000.0,
                         10_000.0, 20_000.0, 50_000.0,
                         100_000.0, 200_000.0, 500_000.0]

    v = np.log(start_vec + 1e-30)      # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # numerical stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # soft‑max weights over convolution entries
            w = exp_beta / sum_exp

            # gradient w.r.t. a: 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Active‑set Linear‑Program refinement
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               step: float = 0.2,
               max_outer: int = 30,
               tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Perform active‑set LP refinement as described in the paper.
    Returns a refined vector and its surrogate value.
    """
    if _opt is None or _sp is None:
        # Linprog not available – skip this stage.
        return f, _objective_surrogate(f)

    best_f = f.copy()
    best_val = _objective_surrogate(f)

    for outer in range(max_outer):
        if time.time() > deadline:
            break

        conv_ff = _conv_fft(best_f, best_f)
        M = float(conv_ff.max())

        # active set: indices where conv is within a tiny epsilon of M
        eps = max(tol, M * 1e-12)
        active_idx = np.where(conv_ff >= M - eps)[0]

        # Build sparse matrix A_ub for constraints conv(best_f, g) ≤ M on active set
        rows = []
        cols = []
        data = []
        for r, i in enumerate(active_idx):
            j_start = max(0, i - (n - 1))
            j_end = min(i, n - 1)
            # variable g[k] where k = i - j
            for j in range(j_start, j_end + 1):
                col = i - j
                rows.append(r)
                cols.append(col)
                data.append(best_f[j])   # coefficient = f[j]

        A_ub = _sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
        b_ub = np.full(len(active_idx), M, dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)   # maximise sum(g) → minimise -sum(g)

        # Solve LP
        lp_res = _opt.linprog(c=c_obj,
                              A_ub=A_ub,
                              b_ub=b_ub,
                              bounds=[(0, None)] * n,
                              method='highs',
                              options={'presolve': True, 'time_limit': max(5.0, deadline - time.time())})
        if not lp_res.success:
            # LP failed – abort refinement
            break

        g = lp_res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        g_hat = g / sum_g

        # convex combination
        f_new = (1.0 - step) * best_f + step * g_hat
        f_new /= f_new.sum()   # re‑normalise (numerical safety)

        new_val = _objective_surrogate(f_new)

        if new_val + 1e-12 < best_val:
            best_f = f_new
            best_val = new_val
            # continue with the new vector
        else:
            # No improvement – stop early
            break
    return best_f, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer hill‑climbing on the simplex.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # move a small fraction of mass from i to j
        eps = cur[i] * rng.uniform(0.001, 0.01)   # 0.1 % – 1 % of a[i]
        if eps <= 0.0 or cur[i] - eps < 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective_surrogate(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val

def propose_candidate(seed: int = 123,

        """Insert a candidate into the heap of size ≤ top_k."""
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:

    # ------------------------------------------------------------------
    # Deterministic analytic seeds

    _push_candidate(_uniform(n))
    _push_candidate(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 9):
        _push_candidate(_sine_window(n, gamma))
    _push_candidate(_hamming_window(n))
    _push_candidate(_blackman_window(n))
    _push_candidate(_nuttall_window(n))
    for beta in np.linspace(1.0, 15.0, 7):
        _push_candidate(_kaiser_window(n, beta))
    for sigma in np.linspace(0.09, 0.26, 10):
        _push_candidate(_gaussian(n, sigma))
    for p in np.linspace(0.0, 2.0, 7):
        _push_candidate(_power(n, p))
    alphas = np.linspace(0.2, 5.0, 4)
    betas = np.linspace(0.2, 5.0, 4)

            elif prev_arr.size < n:
                # up‑sample by linear interpolation
                x_old = np.linspace(0, 1, prev_arr.size)
                x_new = np.linspace(0, 1, n)
                up = np.interp(x_new, x_old, prev_arr)
                up = up / up.sum()
                _push_candidate(up)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Extract top seeds (ordered best → worst)
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seed phase] kept {len(seeds)} candidates, best surrogate = {_objective_surrogate(seeds[0]):.6f}")

    # ------------------------------------------------------------------
    # Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 1) Exponentiated‑gradient polishing (major portion of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        remaining = deadline - time.time()

                                           n,
                                           eg_deadline,
                                           max_iters=2_500_000,
                                           eta0=5.0)
        if val_eg < global_best_val:
            global_best_seq, global_best_val = seq_eg, val_eg

    # ------------------------------------------------------------------
    # 2) Active‑set LP refinement
    # ------------------------------------------------------------------

                                    step=0.25,

    # ------------------------------------------------------------------

                                                          100_000.0, 200_000.0,
                                                          500_000.0],
                                           max_iter_per_beta=20000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < global_best_val:
            global_best_seq, global_best_val = seq_lbfgs, val_lbfgs
            print(f"[L‑BFGS] refined surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Final hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Final hill‑climb ---")
        seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline, rng)
        if val_hc < global_best_val:
            global_best_seq, global_best_val = seq_hc, val_hc
            print(f"[Hill‑climb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Final evaluation with official evaluator (just for reporting)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation (allows incremental improvement across runs)
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
