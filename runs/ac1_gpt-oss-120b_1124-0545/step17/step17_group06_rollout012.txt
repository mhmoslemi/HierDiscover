def hill_climb_reorder(a, deadline):
    a_arr = np.array(a, dtype=np.float64)
    best_val = evaluate_sequence(a_arr.tolist())
    best_arr = a_arr.copy()
    n = len(a_arr)
    rng = np.random.default_rng()
    T0 = 1.0  # starting temperature in terms of objective difference.
    T = T0
    # schedule: exponential decay
    decay = 0.9995
    iter = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        # swap
        cand = best_arr.copy()
        cand[i], cand[j] = cand[j], cand[i]
        val = evaluate_sequence(cand.tolist())
        if val < best_val or rng.random() < np.exp(-(val - best_val)/T):
            best_val = val
            best_arr = cand
        # temperature update
        T *= decay
        iter += 1
        if iter % 1000 == 0:
            print(...)
    return best_arr, best_val

def anneal_reorder(a, deadline):
    a = np.array(a)  # shape
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    cur_val = best_val
    cur_a = a.copy()
    n = len(a)
    rng = np.random.default_rng()
    T0 = 1e-2 * best_val # start small fractions?
    T = T0
    decay = 0.9995
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        cur_a[i], cur_a[j] = cur_a[j], cur_a[i]
        new_val = evaluate_sequence(cur_a.tolist())
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
        else:
            # revert
            cur_a[i], cur_a[j] = cur_a[j], cur_a[i]
        # update temperature
        it += 1
        if (it & 0xFFF) == 0:
            # periodic temperature adjustment
            T = max(T * decay, 1e-12)
            print(f"[Anneal] it={it} best_val={best_val:.6f}")
    return best_a, best_val
```

"""
Step‑function optimiser (SA + mass‑transfer polish).

* Build a small pool of analytic / random ℓ₁‑normalised seeds.
* Pick the best seed (by the same surrogate that the official evaluator uses).
* Run a simulated‑annealing hill‑climber that swaps two entries.
  Swaps preserve non‑negativity and total mass, and the objective
  (2·n·max(conv)) can be evaluated extremely fast via FFT.
* Finish with the cheap pairwise “mass‑transfer” local optimiser
  (identical to the one used in the reference implementation) to
  polish any remaining tiny peaks.
* Return the best sequence found before the deadline.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Low‑level convolution utilities (FFT, O(n log n))
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (exact, O(n log n))."""
    return float(np.max(_conv_fft(a, a)))


# -------------------------------------------------------------
# Surrogate objective for ℓ₁‑normalised vectors  (no division needed)
# -------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) – identical to evaluate_sequence for Σa = 1."""
    n = len(a)
    return 2.0 * n * _max_conv(a)


# -------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# -------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a[a < 0] = 0.0
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a ℓ₁‑normalised Dirichlet vector."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# -------------------------------------------------------------
# Simple deterministic permutations of a multiset of values
# -------------------------------------------------------------
def _high_low_pattern(vals: np.ndarray) -> np.ndarray:
    """Place the largest, then the smallest, alternating from the left."""
    n = len(vals)
    res = np.empty(n, dtype=np.float64)
    left, right = 0, n - 1
    for idx, v in enumerate(vals):
        if idx % 2 == 0:
            res[left] = v
            left += 1
        else:
            res[right] = v
            right -= 1
    return res / res.sum()


def _alternating_high_low(vals: np.ndarray) -> np.ndarray:
    """High values at even positions, low values at odd positions."""
    n = len(vals)
    half = (n + 1) // 2
    high = vals[:half]
    low = vals[half:][::-1]
    res = np.empty(n, dtype=np.float64)
    res[0::2] = high
    res[1::2] = low
    return res / res.sum()


def _front_back(vals: np.ndarray) -> np.ndarray:
    """Largest values on the left half, smallest on the right half."""
    n = len(vals)
    half = n // 2
    res = np.empty(n, dtype=np.float64)
    res[:half] = vals[:half]
    res[half:] = vals[half:]
    return res / res.sum()


# -------------------------------------------------------------
# Pairwise mass‑transfer optimiser (exact max‑conv update)
# -------------------------------------------------------------
def _mass_transfer_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Very fast local optimiser that moves a tiny amount of probability mass."""
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    max_val = float(np.max(conv))
    best_a = a.copy()
    best_val = max_val

    rng = np.random.default_rng()
    step_factor = 0.25
    no_improve = 0
    it = 0
    center = n - 1

    while time.time() < deadline:
        if a.max() < 1e-16:
            break
        src = rng.choice(n, p=a)          # source index weighted by current mass
        src_mass = a[src]
        if src_mass <= 1e-15:
            continue
        tgt = rng.integers(n)
        if tgt == src:
            continue

        eps = step_factor * src_mass
        if eps < 1e-14:
            continue

        # exact update of the convolution (see original code)
        new_conv = conv.copy()
        new_conv[src:src + n] -= 2.0 * eps * a
        new_conv[tgt:tgt + n] += 2.0 * eps * a
        new_conv[center] += 2.0 * eps * eps
        offset = tgt - src
        idx1 = center + offset
        new_conv[idx1] -= eps * eps
        if offset != 0:
            idx2 = center - offset
            new_conv[idx2] -= eps * eps

        new_max = float(np.max(new_conv))

        if new_max < max_val - 1e-16:
            a[src] = src_mass - eps
            a[tgt] = a[tgt] + eps
            conv = new_conv
            max_val = new_max
            no_improve = 0
            if max_val < best_val:
                best_val = max_val
                best_a = a.copy()
        else:
            no_improve += 1

        it += 1
        if no_improve > 5000:
            step_factor *= 0.5
            if step_factor < 1e-4:
                step_factor = 0.25
            no_improve = 0
        if it % 20000 == 0:
            print(f"[MassTransfer] it={it} best max={best_val:.12f}")

    return best_a, 2.0 * n * best_val

# Simulated‑annealing optimiser that swaps two entries
# -------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float) -> tuple[np.ndarray, float]:
    """Swap‑based simulated annealing using the exact max‑convolution objective."""
    n = len(start_seq)
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    # start temperature – a few percent of the current value
    T = max(cur_val * 0.01, 1e-12)
    decay = 0.9995
    rng = np.random.default_rng()
    it = 0
    stagnant = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose swap
        cur[i], cur[j] = cur[j], cur[i]
        new_val = _objective_surrogate(cur)
        delta = new_val - cur_val
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = math.exp(-delta / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]
            stagnant += 1

        it += 1
        if it % 1000 == 0:
            T *= decay
            if T < 1e-12:
                T = 1e-12
        if it % 20000 == 0:
            print(f"[Anneal] it={it} cur={cur_val:.6f} best={best_val:.6f}")

        # occasional random restart if stuck for a long time
        if stagnant > 15000:
            rng.shuffle(cur)
            cur_val = _objective_surrogate(cur)
            stagnant = 0

    return best, best_val


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (length ≈4000) that minimises
    the official ``evaluate_sequence`` objective.  The method builds a
    small seed pool, picks the best ℓ₁‑normalised seed, improves it by a
    simulated‑annealing swap optimiser and finishes with a fast
    pairwise mass‑transfer polish.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Load previous solution if available (already ℓ₁‑normalised)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            previous = np.asarray(prev, dtype=np.float64)
            if previous.size != n:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # 3. Assemble a diverse seed pool (ℓ₁‑normalised vectors)
    # ------------------------------------------------------------------
    seeds = []

    # analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # a few random beta‑shapes
    for _ in range(12):
        a = rng.uniform(0.2, 5.0)
        b = rng.uniform(0.2, 5.0)
        seeds.append(_beta_shape(n, a, b))

    # random Dirichlet with various concentrations
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution, if any
    if previous is not None:
        seeds.append(previous / previous.sum())

    # ------------------------------------------------------------------
    # 4. Evaluate the surrogate objective and keep a few top seeds
    # ------------------------------------------------------------------
    evaluated = [(s, _objective_surrogate(s)) for s in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    best_seq = evaluated[0][0].copy()
    best_val = evaluated[0][1]

    print(f"\n[Init] best surrogate among seeds = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Simulated‑annealing based re‑ordering / local optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print("\n--- Annealing optimisation ---")
        best_seq, best_val = _anneal_optimize(best_seq,
                                              deadline)

        print(f"[Anneal] best surrogate after anneal = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Pairwise mass‑transfer polish (very fast)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mass‑transfer polish ---")
        best_seq, best_val = _mass_transfer_opt(best_seq,
                                                n,
                                                deadline)
        print(f"[MassTransfer] best surrogate after polish = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for potential warm‑starts of the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
