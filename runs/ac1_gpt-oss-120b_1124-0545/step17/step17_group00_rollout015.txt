def _simulated_annealing(start_seq, n, deadline, rng):
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    best = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    # Parameters
    init_T = 0.01 * best_val   # or a fraction of best value
    final_T = 1e-6 * best_val
    total_time = deadline - time.time()
    # number of steps predetermined maybe 200k or until time.
    steps = int(1e6) # high bound but break when time.
    for step in range(steps):
        if time.time() > deadline:
            break
        # compute temperature schedule linearly decreasing with steps
        t_frac = step / steps
        T = init_T * (1 - t_frac) + final_T * t_frac
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # epsilon relative fraction
        # bound epsilon to maybe up to 0.1 of a[i]
        if a[i] <= 0:
            continue
        eps = a[i] * rng.uniform(0.001, 0.02)  # 0.1% - 2%
        if eps <= 0:
            continue
        # compute delta conv via slice operations
        delta = np.zeros_like(conv)
        start_i = i
        start_j = j
        delta[start_i:start_i+n] -= 2.0 * eps * a
        delta[start_j:start_j+n] += 2.0 * eps * a
        new_conv = conv + delta
        new_max = np.max(new_conv)
        new_val = 2.0 * n * new_max
        # acceptance
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            # accept
            a[i] -= eps
            a[j] += eps
            conv = new_conv
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = a.copy()
    return best, best_val

Search for a non‑negative step‑function minimizing

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm expands the previous approach:

1. **Broad seeding** – deterministic windows, many random Dirichlet / exponential
   draws and a dense random search over Beta‑shape parameters.  The best few
   (by the exact surrogate 2·n·max(conv) with sum = 1) are kept.

   followed by a continuation L‑BFGS optimisation of the smooth log‑sum‑exp
   surrogate (the “β‑schedule”).

3. **Simulated‑annealing hill‑climb** – fast O(n) updates of the full
   convolution when moving a tiny amount of mass between two coordinates.
   A temperature schedule lets the search escape shallow local minima and
   exploit the final remaining time budget.

All steps respect the simplex (non‑negative, ℓ₁‑normalised) constraint.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a vector normalised to sum = 1:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Decreasing power law a_i ∝ (i+1)^{-p} (already normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08

         - a1 * np.cos(2 * np.pi * i / (n - 1))

    a0 = 0.355768

def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geometric(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, 0 < r < 1."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _monotonic_power_desc(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power a_i ∝ ((n-i)/n)**p."""
    i = np.arange(n, dtype=np.float64)
    a = ((n - i) / n) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:

    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s

def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space, guaranteeing non‑negativity and the simplex.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Fast O(n) simulated‑annealing hill‑climb.  Moves a random fraction of mass
    from index i to j, updates the full convolution via slice operations and
    accepts moves according to a linear temperature schedule.
    """
    a = start_seq.copy()
    conv = _conv_fft(a, a)                     # length 2n‑1
    cur_val = _objective_surrogate(a)

    best = a.copy()
    best_val = cur_val

    # temperature schedule – from 10 % of current value down to 1e‑7 % of it
    T0 = max(1e-3, 0.1 * cur_val)
    T_end = max(1e-9, 1e-6 * cur_val)

    # pre‑allocate delta array to avoid repeated allocations
    delta = np.zeros_like(conv)

    steps = 0
    while time.time() < deadline:
        steps += 1
        # linear temperature decay
        t_frac = steps / 500_000  # hard limit on steps
        T = T0 * (1 - t_frac) + T_end * t_frac

        i, j = rng.integers(n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move a random fraction of a[i] (between 0.1 % and 2 % of its mass)
        eps = a[i] * rng.uniform(0.001, 0.02)
        if eps <= 0.0:
            continue

        # build delta for this move
        delta.fill(0.0)
        delta[i:i + n] -= 2.0 * eps * a
        delta[j:j + n] += 2.0 * eps * a

        new_conv = conv + delta
        new_max = np.max(new_conv)
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / (T + 1e-16))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= eps
            a[j] += eps
            conv = new_conv
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = a.copy()

        # safety break if enough steps have been performed
        if steps >= 500_000:
            break

    return best, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The function respects the time budget and stores the best result in the
    global variable ``height_sequence_1`` for reuse.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final ``evaluate_sequence`` call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Problem size (chosen as a compromise between quality and speed)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Helper for keeping the top‑K seeds (by exact surrogate)
    # ------------------------------------------------------------------
    top_k = 30
    best_heap = []   # stores (-value, vector)

    def _push(vec: np.ndarray):
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if val < -best_heap[0][0]:
                heapq.heappushpop(best_heap, (-val, vec))

    # ------------------------------------------------------------------
    # 3. Deterministic analytic families (including a few extra shapes)
    # ------------------------------------------------------------------
    _push(_uniform(n))
    _push(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 12):
        _push(_sine_window(n, gamma))
    _push(_hamming_window(n))
    _push(_blackman_window(n))
    _push(_nuttall_window(n))
    for beta in np.linspace(1.0, 15.0, 10):
        _push(_kaiser_window(n, beta))
    for sigma in np.linspace(0.09, 0.26, 12):
        _push(_gaussian(n, sigma))
    for p in np.linspace(0.0, 2.0, 9):
        _push(_power(n, p))
    # extra monotone power shapes
    for p in np.linspace(0.5, 8.0, 9):
        _push(_monotonic_power_desc(n, p))
    # geometric families with random r
    for r in np.linspace(0.90, 0.999, 8):
        _push(_geometric(n, r))

    # beta distributions – dense random sampling of (α,β)
    for _ in range(2000):
        if time.time() > deadline:
            break
        # log‑uniform sampling between 0.1 and 100
        a = 10 ** rng.uniform(-1, 2)
        b = 10 ** rng.uniform(-1, 2)
        _push(_beta_shape(n, a, b))

    # ------------------------------------------------------------------
    # 4. Massive random seeding
    # ------------------------------------------------------------------
    conc_choices = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for _ in range(3000):
        if time.time() > deadline:
            break
        conc = rng.choice(conc_choices)
        vec = _random_dirichlet(n, conc, rng)
        # random cyclic shift – yields many asymmetric versions
        shift = rng.integers(n)
        if shift:
            vec = np.roll(vec, shift)
        _push(vec)

    for _ in range(500):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        shift = rng.integers(n)
        if shift:
            vec = np.roll(vec, shift)
        _push(vec)

    for nf in (2, 3, 5):
        for _ in range(300):
            if time.time() > deadline:
                break
            vec = _lowfreq_curve(n, nf, rng)
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
            _push(vec)

    # ------------------------------------------------------------------
    # 5. Re‑use previous solution if it matches the current size
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0.0:
                prev_norm = prev_arr / prev_arr.sum()
                _push(prev_norm)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 6. Extract the best seeds (ordered from best to worst)
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seed phase] kept {len(seeds)} seeds, best surrogate = {_objective_surrogate(seeds[0]):.6f}")

    # ------------------------------------------------------------------
    # 7. Exponentiated‑gradient polishing (if time permits)
    # ------------------------------------------------------------------
    best_seq = seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    if time.time() < deadline - 30.0:
        eg_deadline = time.time() + (deadline - time.time()) * 0.45
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, eg_deadline,
                                           max_iters=2_500_000,
                                           eta0=5.0)
        if val_eg < best_val:
            best_seq, best_val = seq_eg, val_eg
            print(f"[ExpGrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Continuation L‑BFGS fine‑tuning (use most of the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0 and _opt is not None:
        print("\n--- L‑BFGS fine‑tuning ---")
        lbfgs_deadline = deadline - 5.0
        seq_lb = _lbfgs_smooth_optimize(best_seq, n, lbfgs_deadline,
                                         max_iter_per_beta=15000)
        val_lb = _objective_surrogate(seq_lb)
        if val_lb < best_val:
            best_seq, best_val = seq_lb, val_lb
            print(f"[L‑BFGS] refined surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Simulated‑annealing hill‑climb (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n--- Simulated‑Annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline, rng)
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print(f"[SA] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 10. Final true evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
