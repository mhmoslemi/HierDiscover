    a = ((i + 0.5) ** (alpha - 1)) * ((n - i - 0.5) ** (beta - 1))
    return a / a.sum()

    a = _softmax(v)
    conv = _conv_fft(a, a)
    m = np.max(conv)
    shift = m
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_beta)
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1:n-1+n]
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v

        seeds.append(_exponential(n, rate))

def _smooth_max_fun(v, beta, n):
    a = _softmax(v)
    conv = _conv_fft(a, a)
    m = np.max(conv)
    shift = m
    # avoid overflow
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1:n-1+n]
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v
```

def _subgradient_opt(start_seq, n, deadline, max_iters=2_500_000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # gradient of active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        lr = 0.3 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)
    return best_seq, best_val

def _hill_climbing_opt(start_seq, n, deadline, max_iters=500_000):
    rng = np.random.default_rng()
    logit = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()

    best_seq = cur_seq.copy()
    best_val = cur_val
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        sigma = 0.015 / math.sqrt(it)
        delta = rng.normal(scale=sigma)
        new_log = logit.copy()
        new_log[i] += delta
        new_log[j] -= delta
        new_seq = _softmax(new_log)
        new_val = _objective_surrogate(new_seq)

            logit, cur_seq, cur_val = new_log, new_seq, new_val
            if new_val < best_val:
                best_seq, best_val = new_seq.copy(), new_val
    return best_seq, best_val

    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_uniform_block(n: int, block_size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of `block_size` indices."""
    idx = rng.choice(n, size=block_size, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / block_size
    return a


# ----------------------------------------------------------------------
# Seed‑pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse collection of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families --------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Gaussian family – several bandwidths
    for sigma in (0.05, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))

    # Power‑law families
    for p in (0.0, 0.25, 0.5, 0.75, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))

    # Exponential
    for rate in (0.001, 0.003, 0.006, 0.01, 0.015):
        seeds.append(_exponential(n, rate))

    # Asymmetric Beta families (α≠β)
    for alpha in (0.2, 0.4, 0.6, 0.8, 1.2):
        for beta in (1.0, 2.0, 3.0):
            seeds.append(_beta_asymmetric(n, alpha, beta))
            seeds.append(_beta_asymmetric(n, beta, alpha))  # swapped

    # Sin‑power families -------------------------------------------------
    for p in (1.0, 2.0, 4.0, 6.0, 8.0):
        seeds.append(_sin_power(n, p))

    # Random Dirichlet draws (various concentrations) -------------------
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            d = _random_dirichlet(n, conc, rng)
            seeds.append(d)                                 # as‑is
            seeds.append(np.sort(d)[::-1])                 # descending
            seeds.append(d[rng.permutation(n)])            # random perm

    # Sparse uniform blocks -----------------------------------------------
    for blk in (80, 120, 200):
        for _ in range(2):
            seeds.append(_sparse_uniform_block(n, blk, rng))

    # Previous solution ---------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Normalisation -------------------------------------------------------
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Smooth‑max helper (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_max_fun(v: np.ndarray, beta: float, n: int):
    """Return (value, gradient) of the smoothed max‑convolution."""
    a = _softmax(v)                     # stays on the simplex
    conv = _conv_fft(a, a)
    m = np.max(conv)                    # true max
    shift = m
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient of the smooth max w.r.t. a
    w = exp_beta / sum_exp                    # soft weights, sum to 1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # conv(rev(a), w)
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 15000) -> np.ndarray:
    """Continuation L‑BFGS on the smooth max‑convolution."""
    if _opt is None:
        # SciPy not available – fall back to the start vector.
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 3.0, 7.0, 15.0, 30.0, 60.0,
            120.0, 250.0, 500.0, 1000.0, 2500.0,
            5000.0, 10000.0, 25000.0, 50000.0, 100000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits – optimisation variables

    global _CURRENT_DEADLINE
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        _CURRENT_DEADLINE = deadline
        beta_val = float(beta)

        # scipy expects a callable returning (f, grad)
        fun = functools.partial(_smooth_max_fun, beta=beta_val, n=n)

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=_deadline_callback,
                                options={
                                    'maxiter': max_iter_per_beta,
                                    'disp': False,
                                    'ftol': 1e-12,
                                })
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as exc:  # pragma: no cover
            print(f"[L‑BFGS] exception {exc} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_500_000) -> tuple[np.ndarray, float]:
    """Long sub‑gradient descent on 2·n·max(conv) (exact surrogate)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size ∝ 1/√t
        lr = 0.3 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Pair‑wise logit swap hill‑climbing (cheap final polishing)
# ----------------------------------------------------------------------
def _hill_climbing_opt(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Very cheap pair‑wise logit‑swap hill‑climbing."""
    rng = np.random.default_rng()
    logit = np.log(start_seq + 1e-30)

    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        sigma = 0.015 / math.sqrt(it)
        delta = rng.normal(scale=sigma)
        new_log = logit.copy()
        new_log[i] += delta
        new_log[j] -= delta
        new_seq = _softmax(new_log)
        new_val = _objective_surrogate(new_seq)
        if new_val < cur_val:
            logit, cur_seq, cur_val = new_log, new_seq, new_val
            if new_val < best_val:
                best_seq, best_val = new_seq.copy(), new_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Small random perturbation (used after a good solution)
# ----------------------------------------------------------------------
def _random_perturb(seq: np.ndarray,
                    n: int,
                    rng: np.random.Generator,
                    magnitude: float = 0.02) -> np.ndarray:
    """Add a tiny random logit perturbation and re‑softmax."""
    logit = np.log(seq + 1e-30)
    perturb = magnitude * rng.normal(size=n)
    new_log = logit + perturb
    return _softmax(new_log)


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Initialise random generator and time budget
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # reserve a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Problem size (trade‑off between quality and speed)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Seed pool (rich and diverse)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_arr = arr
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # quick surrogate evaluation (exact for ℓ₁‑normalised vectors)
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nBest surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS (allocated ≈30 % of the remaining budget)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    lbfgs_deadline = time.time() + time_left * 0.30

    # start from the best raw seed
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    seq_opt = _lbfgs_smooth_optimize(global_best_seq, n,
                                     lbfgs_deadline,
                                     beta_schedule=None,
                                     max_iter_per_beta=15000)
    val_opt = _objective_surrogate(seq_opt)
    print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

    if val_opt < global_best_val:
        global_best_seq, global_best_val = seq_opt, val_opt
        print("[Update] new global best after L‑BFGS")

    # ------------------------------------------------------------------
    # 4. Random perturbation + short L‑BFGS restart (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        perturbed = _random_perturb(global_best_seq, n, rng, magnitude=0.05)
        remaining = deadline - time.time()
        restart_deadline = time.time() + remaining * 0.10
        seq_restart = _lbfgs_smooth_optimize(perturbed, n,
                                             restart_deadline,
                                             beta_schedule=None,
                                             max_iter_per_beta=8000)
        val_restart = _objective_surrogate(seq_restart)
        if val_restart < global_best_val:
            global_best_seq, global_best_val = seq_restart, val_restart
            print(f"[Perturb+L‑BFGS] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Long projected sub‑gradient polishing (≈60 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        subgrad_deadline = time.time() + time_left * 0.60
        seq_sub, val_sub = _subgradient_opt(global_best_seq, n,
                                            subgrad_deadline,
                                            max_iters=5_000_000)
        if val_sub < global_best_val:
            global_best_seq, global_best_val = seq_sub, val_sub
            print(f"[SubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final cheap hill‑climbing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        seq_hc, val_hc = _hill_climbing_opt(global_best_seq, n,
                                            hill_deadline,
                                            max_iters=500_000)
        if val_hc < global_best_val:
            global_best_seq, global_best_val = seq_hc, val_hc
            print(f"[HillClimb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next iteration (if any)
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
