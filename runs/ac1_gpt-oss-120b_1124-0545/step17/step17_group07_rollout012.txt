"""
Propose a step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

by a three‑stage pipeline:

1. Build a rich pool of ℓ₁‑normalised seeds (analytic families,
   random Dirichlet draws, sparse blocks and the previous solution if any)
   for a relatively large problem size (n≈8000).  Keep the K best seeds
   according to the exact surrogate 2·n·max(conv) (sum(a)=1).

2. Refine each of the K seeds with a continuation L‑BFGS optimiser that
   minimises a smooth approximation of the max‑convolution (log‑sum‑exp).
   The schedule of β values is truncated for speed, and the number of
   L‑BFGS iterations per β is adjusted to the remaining time budget.

3. Polish the current best candidate with cheap first‑order methods
   (exponentiated‑gradient mirror descent, optional projected sub‑gradient,
   simulated‑annealing and a tiny hill‑climbing pass).  All steps respect
   the global deadline and stop early if the budget runs out.

The best sequence (as a plain list of floats) is stored in the global
variable ``height_sequence_1`` for reuse by a later run.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# SciPy is optional – L‑BFGS is only used when available.
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (surrogate works on ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv) for a sequence that sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by evaluate_sequence (no normalisation)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta(α,β) on {0,…,n‑1} using a half‑offset to avoid zeros."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 0.5) ** (alpha - 1.0)) * ((n - i - 0.5) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p  (symmetric, zero at the ends)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_uniform_block(n: int, block_size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of `block_size` indices."""
    idx = rng.choice(n, size=block_size, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / block_size
    return a


# ----------------------------------------------------------------------
# Permutation helpers (for richer seed pool)
# ----------------------------------------------------------------------
def _alternating_order_from_vec(vec: np.ndarray) -> np.ndarray:
    """Place sorted values alternating from the left and right ends."""
    sorted_desc = np.sort(vec)[::-1]
    n = len(vec)
    out = np.empty_like(vec)
    left, right = 0, n - 1
    for val in sorted_desc:
        if (left + right) % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out


def _shuffle_by_step(sorted_desc: np.ndarray, step: int) -> np.ndarray:
    """Distribute the sorted values over the index set using a coprime step."""
    n = len(sorted_desc)
    out = np.empty(n, dtype=np.float64)
    pos = 0
    for val in sorted_desc:
        out[pos] = val
        pos = (pos + step) % n
    return out


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse collection of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families ------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Gaussian family (different bandwidths)
    for sigma in (0.05, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))

    # Power‑law families
    for p in (0.0, 0.25, 0.5, 0.75, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))

    # Exponential / double‑exponential
    for rate in (0.001, 0.003, 0.006, 0.01, 0.015):
        seeds.append(_exponential(n, rate))
        seeds.append(_double_exponential(n, rate))

    # Beta‑shaped families (symmetric)
    for a in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        seeds.append(_beta_distribution(n, a, a))

    # Sin‑power families
    for p in (1.0, 2.0, 4.0, 6.0):
        seeds.append(_sin_power(n, p))

    # Random Dirichlet draws (different concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                                 # as‑is
            seeds.append(np.sort(raw)[::-1])                 # sorted
            seeds.append(_alternating_order_from_vec(raw))   # anti‑symmetric
            # sparse version (keep only the top 30 % mass)
            th = np.quantile(raw, 0.7)
            sparse = np.where(raw > th, raw, 0.0)
            if sparse.sum() > 0.0:
                seeds.append(sparse / sparse.sum())

    # Sparse uniform blocks (different sizes)
    for blk in (80, 150, 300):
        for _ in range(2):
            seeds.append(_sparse_uniform_block(n, blk, rng))

    # Previous solution (if it matches the current size)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Normalisation safeguard
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 1500) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smoothed max‑conv."""
    if _opt is None:
        # SciPy not available – skip L‑BFGS entirely.
        return start_vec.copy()

    if beta_schedule is None:
        # a compact schedule – enough for a decent smoothing progression
        beta_schedule = [1.0, 3.0, 7.0, 15.0, 30.0,
                         60.0, 120.0, 250.0, 500.0, 1000.0,
                         2500.0, 5000.0, 10000.0]

    # optimisation variables are logits (softmax keeps us on simplex)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on simplex
            conv = _conv_fft(a, a)
            m = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = m
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of smooth max w.r.t. a
            w = exp_beta / sum_exp                    # soft weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # conv(rev(a), w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={
                                    'maxiter': max_iter_per_beta,
                                    'disp': False,
                                    'ftol': 1e-12
                                })
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser (exact surrogate)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent polish on the exact max‑conv surrogate."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Long sub‑gradient descent on 2·n·max(conv) (exact surrogate)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing style refinement (logit‑space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """Random walk in log‑softmax space with a linear cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)

    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.08 * (1.0 - progress) + 1e-6
        sigma = 0.12 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Local hill‑climbing (pairwise logit swaps)
# ----------------------------------------------------------------------
def _hill_climbing_opt(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Very cheap pair‑wise logit swap hill‑climbing."""
    rng = np.random.default_rng()
    logit = np.log(start_seq + 1e-30)

    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        sigma = 0.015 / math.sqrt(it)
        delta = rng.normal(scale=sigma)
        new_log = logit.copy()
        new_log[i] += delta
        new_log[j] -= delta
        new_seq = _softmax(new_log)
        new_val = _objective_surrogate(new_seq)
        if new_val < cur_val:
            logit, cur_seq, cur_val = new_log, new_seq, new_val
            if new_val < best_val:
                best_seq, best_val = new_seq.copy(), new_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Small random perturbation (used after a good solution)
# ----------------------------------------------------------------------
def _random_perturb(seq: np.ndarray,
                    n: int,
                    rng: np.random.Generator,
                    magnitude: float = 0.02) -> np.ndarray:
    """Add a tiny random logit perturbation and re‑softmax."""
    logit = np.log(seq + 1e-30)
    perturb = magnitude * rng.normal(size=n)
    new_log = logit + perturb
    return _softmax(new_log)


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and time budget
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 3.0    # a small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a problem size (large enough to allow a flat autocorrelation)
    # ------------------------------------------------------------------
    n = 8000
    print(f"\n=== Starting optimisation with n={n} ===")

    # ------------------------------------------------------------------
    # 2. Assemble a diverse seed pool
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    print("[Seed] generating candidates …")
    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"[Seed] collected {len(seeds)} candidates")

    # Quick surrogate evaluation of all seeds
    surrogate_vals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    surrogate_vals.sort(key=lambda x: x[1])

    top_k = 30                     # keep a modest number for L‑BFGS
    top_seeds = [vec for vec, _ in surrogate_vals[:top_k]]

    print("\n[Seed] best surrogate values among raw seeds:")
    for i, (_, val) in enumerate(surrogate_vals[:top_k]):
        print(f"  #{i+1:02d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS on the top seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    # a compact β‑schedule – sufficient for a smooth convergence
    beta_schedule = [1.0, 3.0, 7.0, 15.0, 30.0,
                     60.0, 120.0, 250.0, 500.0, 1000.0,
                     2500.0, 5000.0, 10000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached during L‑BFGS stage")
            break

        # allocate a modest slice of remaining time per seed
        remaining = deadline - time.time()
        # aim for at most ~5 % of remaining time for a single seed
        max_iter_beta = int(min(1500, remaining * 0.05 * 1e4))
        max_iter_beta = max(600, max_iter_beta)

        print(f"\n--- L‑BFGS on seed #{idx+1} (max_iter/β={max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq, global_best_val = seq_opt.copy(), val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

        # short exponentiated‑gradient polish after each seed (if time permits)
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_exp = int(min(300_000, remaining * 0.04 * 1e4))
            max_exp = max(100_000, max_exp)
            seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_exp)
            if val_exp < global_best_val:
                global_best_seq, global_best_val = seq_exp.copy(), val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Random perturbations + occasional L‑BFGS restarts (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random perturbation + restart L‑BFGS ---")
        for rep in range(2):
            if time.time() > deadline:
                break
            perturbed = _random_perturb(global_best_seq, n, rng, magnitude=0.07)
            remaining = deadline - time.time()
            max_iter_beta = int(min(1200, remaining * 0.04 * 1e4))
            max_iter_beta = max(500, max_iter_beta)
            seq_restart = _lbfgs_smooth_optimize(perturbed,
                                                 n,
                                                 deadline,
                                                 beta_schedule=beta_schedule[:8],
                                                 max_iter_per_beta=max_iter_beta)
            val_restart = _objective_surrogate(seq_restart)
            if val_restart < global_best_val:
                global_best_seq, global_best_val = seq_restart.copy(), val_restart
                print(f"[Restart] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Long projected sub‑gradient polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(1_200_000, remaining * 0.05 * 1e4))
        if max_sub > 0:
            print("\n--- Long sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq, global_best_val = seq_sub.copy(), val_sub
                print(f"[SubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing refinement (last phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(800_000, remaining * 0.04 * 1e4))
        if max_sa > 0:
            print("\n--- Simulated‑annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing_opt(global_best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=max_sa)
            if val_sa < global_best_val:
                global_best_seq, global_best_val = seq_sa.copy(), val_sa
                print(f"[SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Local hill‑climbing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc = int(min(300_000, remaining * 0.03 * 1e4))
        if max_hc > 0:
            print("\n--- Hill‑climbing fine‑tuning ---")
            seq_hc, val_hc = _hill_climbing_opt(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_hc)
            if val_hc < global_best_val:
                global_best_seq, global_best_val = seq_hc.copy(), val_hc
                print(f"[HillClimb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next iteration (if any)
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
