def propose_candidate(seed=123, budget_s=1000):

    import needed libs.

    start_time = time.time()
    deadline = start_time + budget_s - 2.0

        try:
            prev_arr = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            if prev_arr.size != n:
                prev_arr=None
        except:
            prev_arr=None

    # seed generation
    seeds = _collect_seeds_extended(n, rng, prev_arr)
    # Add extra analytic families
    seeds.extend(_generate_extra_families(n, rng))

    # evaluate surrogate for each
    evals = [(s, _objective_surrogate(s)) for s in seeds]
    evals.sort(key=lambda x:x[1])
    top_k = 10
    top_seeds = [vec for vec,_ in evals[:top_k]]

    print initial best.

    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)

    # main loop: for each seed, local optimization
    for i, seed_vec in enumerate(top_seeds):
        if time.time()>deadline: break
        # LBFGS smooth optimize
        seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline, beta_schedule=_default_beta_schedule,
                                          max_iter_per_beta=4000)
        val_opt = _objective_surrogate(seq_opt)
        if val_opt < best_val:
            best_seq, best_val = seq_opt.copy(), val_opt
        # Exp gradient polish
        if time.time()<deadline:
            seq_polished, val_polished = _exp_gradient_opt(best_seq, n, deadline,
                                                          max_iters=200_000)
            if val_polished < best_val:
                best_seq, best_val = seq_polished.copy(), val_polished
        # Mass transfer quick pass
        if time.time()<deadline:
            seq_mt, val_mt = _mass_transfer_opt(best_seq, n, deadline)
            if val_mt < best_val:
                best_seq, best_val = seq_mt.copy(), val_mt

    # maybe additional passes: run mass_transfer again if time remains
    while time.time()<deadline:
        seq_mt, val_mt = _mass_transfer_opt(best_seq, n, deadline)
        if val_mt < best_val:
            best_seq, best_val = seq_mt.copy(), val_mt
        else:
            break

    # final evaluation
    true_val = evaluate_sequence(best_seq.tolist())
    print final surrogate and true_val.

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

ℓ₁‑norm is 1 (the evaluation function is scale‑invariant, so any scaling
is accepted).  The global variable ``height_sequence_1`` is updated so
that later runs can continue from the current best solution.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate for ℓ₁‑normalised vectors:
        2·n·max(conv(seq,seq))
    This equals the official ``evaluate_sequence`` value when the
    sequence sums to 1 (up to rounding).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_power(n: int, p: float) -> np.ndarray:
    """Decreasing power law (largest entry at the left side)."""
    idx = np.arange(1, n + 1, dtype=np.float64)               # 1 … n
    a = 1.0 / (np.flip(idx) ** p)                            # 1/n^p … 1^p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# New families (not present in the original code)
# ----------------------------------------------------------------------
def _exp_decay(n: int, tau: float) -> np.ndarray:
    """Monotone exponential decay a_i ∝ exp(-i/tau)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-i / tau)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inv_power_shifted(n: int, p: float, offset: float) -> np.ndarray:
    """Shifted inverse‑power a_i ∝ 1/(i+offset)^p."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / ((i + offset) ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_then_exp_decay(n: int, block_frac: float, tau: float) -> np.ndarray:
    """
    Uniform mass on a leading block, then exponential tail.
    block_frac ∈ (0,1) controls the length of the flat prefix.
    """
    block_len = max(1, int(round(block_frac * n)))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = 1.0
    tail_len = n - block_len
    if tail_len > 0:
        a[block_len:] = np.exp(-np.arange(tail_len, dtype=np.float64) / tau)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random families
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block of length block_frac·n."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection – combines the original pool with the new families
# ----------------------------------------------------------------------
def _collect_all_seeds(n: int,
                       rng: np.random.Generator,
                       previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a large and diverse list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # ---- Analytic families (original) --------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_gaussian(n, sigma_frac=0.12))

    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    for k in (int(0.01 * n), int(0.03 * n), int(0.07 * n)):
        for _ in range(2):
            seeds.append(_sparse_random(n, k, rng))

    for bf in (0.05, 0.15, 0.30):
        for _ in range(2):
            seeds.append(_block_uniform(n, bf, rng))

    # ---- New families -------------------------------------------------
    for tau in (200, 400, 800, 1600, 3200):
        seeds.append(_exp_decay(n, tau))
    for _ in range(8):
        tau = rng.uniform(150.0, 2500.0)
        seeds.append(_exp_decay(n, tau))

    for p in np.linspace(0.2, 4.0, 11):
        seeds.append(_rev_power(n, p))

    for _ in range(10):
        p = rng.uniform(0.1, 3.0)
        off = rng.uniform(0.1, 5.0)
        seeds.append(_inv_power_shifted(n, p, off))

    for bf in np.linspace(0.1, 0.9, 5):
        for tau in (300, 1000, 2000):
            seeds.append(_block_then_exp_decay(n, bf, tau))

    # ---- Mixtures -----------------------------------------------------
    # mix a few random pairs from the pool we already have
    basic_pool = seeds.copy()
    for _ in range(30):
        a, b = rng.choice(basic_pool, size=2, replace=False)
        alpha = rng.uniform(0.0, 1.0)
        seeds.append(_mix_shapes(a, b, alpha))

    # ---- Previous solution, if any ------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # ---- Sanity – ensure non‑negative, ℓ₁‑normalised --------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns a probability vector (ℓ₁‑normalised).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 1e5,
            2e5, 5e5, 1e6
        ]

    # work in log‑space to stay on the simplex automatically
    v = np.log(start_vec + 1e-30)   # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exact‑max exponentiated‑gradient optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # decreasing learning rate
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Pairwise mass‑transfer local optimiser (O(n) per move)
# ----------------------------------------------------------------------
def _mass_transfer_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Very fast local optimiser that moves a tiny amount of probability mass
    from a high‑mass coordinate to a low‑mass coordinate.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    max_val = float(np.max(conv))
    best_a = a.copy()
    best_val = max_val

    rng = np.random.default_rng()
    step_factor = 0.25          # initial fraction of source mass to move
    no_improve = 0
    it = 0
    center = n - 1
    L = len(conv)

    while time.time() < deadline:
        # source weighted by its mass (prefer large entries)
        if a.max() < 1e-16:
            break
        src = rng.choice(n, p=a)                # source index
        src_mass = a[src]
        if src_mass <= 1e-15:
            continue
        # target uniform
        tgt = rng.integers(n)
        while tgt == src:
            tgt = rng.integers(n)

        eps = step_factor * src_mass
        if eps < 1e-14:
            continue

        # tentative new convolution (linear + quadratic terms)
        new_conv = conv.copy()
        new_conv[src:src + n] -= 2.0 * eps * a
        new_conv[tgt:tgt + n] += 2.0 * eps * a

        new_conv[center] += 2.0 * eps * eps
        offset = tgt - src
        idx1 = center + offset
        new_conv[idx1] -= eps * eps
        if offset != 0:
            idx2 = center - offset
            new_conv[idx2] -= eps * eps

        new_max = float(np.max(new_conv))

        if new_max < max_val - 1e-16:
            a[src] = src_mass - eps
            a[tgt] += eps
            conv = new_conv
            max_val = new_max
            no_improve = 0
            if max_val < best_val:
                best_val = max_val
                best_a = a.copy()
        else:
            no_improve += 1

        it += 1
        if no_improve > 5000:
            step_factor *= 0.5
            if step_factor < 1e-4:
                step_factor = 0.25
            no_improve = 0

        if it % 20000 == 0:
            print(f"[MassTransfer] it={it} best max={best_val:.12f}")

    return best_a, 2.0 * n * best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the allotted time budget.
    The function returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Load previous solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, list):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 3. Generate a rich pool of seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_all_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # 4. Quick surrogate evaluation, keep the most promising ones
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 5. Initialise global best
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Local refinement on each top seed
    # ------------------------------------------------------------------
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        10000.0, 20000.0, 50000.0, 1e5,
        2e5, 5e5, 1e6
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing local refinements.")
            break

        remaining = deadline - time.time()
        # allocate a few seconds per seed; adapt if we are close to the end
        max_iter = int(min(12000, remaining * 0.04 * 1e5))
        max_iter = max(4000, max_iter)

        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter={max_iter}) ---")
        try:
            seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                             n,
                                             deadline,
                                             beta_schedule=beta_schedule,
                                             max_iter_per_beta=max_iter)
        except RuntimeError:
            # SciPy not available – fall back to the raw seed
            seq_opt = seed_vec.copy()

        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < best_val:
            best_seq, best_val = seq_opt.copy(), val_opt
            print(f"[Update] new global best = {best_val:.6f}")

        # --------------------------------------------------------
        # Exponentiated‑gradient polishing (exact max)
        # --------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_exp = int(min(600_000, remaining * 0.07 * 1e5))
            max_exp = max(300_000, max_exp)
            seq_exp, val_exp = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_exp)
            if val_exp < best_val:
                best_seq, best_val = seq_exp.copy(), val_exp
                print(f"[ExpGrad] improved to {best_val:.6f}")

        # --------------------------------------------------------
        # Brief mass‑transfer pass after polishing
        # --------------------------------------------------------
        if time.time() < deadline:
            seq_mt, val_mt = _mass_transfer_opt(best_seq, n, deadline)
            if val_mt < best_val:
                best_seq, best_val = seq_mt.copy(), val_mt
                print(f"[MassTransfer] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final intensive mass‑transfer optimisation (as many iterations
    #    as remain)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final mass‑transfer optimisation ---")
        seq_final, val_final = _mass_transfer_opt(best_seq, n, deadline)
        if val_final < best_val:
            best_seq, best_val = seq_final.copy(), val_final
            print(f"[FinalMassTransfer] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible later runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
