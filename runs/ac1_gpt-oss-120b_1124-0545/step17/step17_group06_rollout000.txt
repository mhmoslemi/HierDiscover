Optimise a non‑negative step‑function for the ``evaluate_sequence`` objective.

The optimiser combines several strategies in a time‑bounded pipeline:

1. **Diverse Analytic and Random Seeds** – a collection of ℓ₁‑normalised vectors
   (uniform, beta‑shapes, low‑frequency cosine curves, sparse Dirichlet,
   block‑uniform, etc.) plus the previous solution if available.

2. **Continuation L‑BFGS** – smooth‑max surrogate optimisation (log‑sum‑exp)
   using the softmax parametrisation.  A schedule of increasing smoothing
   parameters drives the iterate towards the true max‑convolution objective.

3. **Exact Mirror‑Descent** – exponentiated‑gradient (mirror descent) on the
   *exact* ℓ∞‑convolution.  All active maximal indices are used to form the
   sub‑gradient, yielding a tighter polishing step.

4. **Simulated‑Annealing Mass Transfer** – a fast O(n) local optimiser that moves
   a small amount of probability mass between two coordinates.  Moves are
   accepted according to a temperature schedule, allowing occasional uphill
   steps to escape local minima.

5. **Random Perturbation Refinement** – occasional Gaussian perturbations of
   the current best vector followed by projection onto the probability simplex.

The whole search stops when the global deadline (≈ budget_s seconds) is reached.
The best vector found is stored in ``height_sequence_1`` and returned as a list
of Python floats.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate for ℓ₁‑normalised vectors:
        2·n·max(conv(seq,seq)).
    For a ℓ₁‑normalised seq this equals the official ``evaluate_sequence``
    value (up to rounding).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic / random families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    for a in np.linspace(0.2, 5.0, 9):
        for b in np.linspace(0.2, 5.0, 9):
            seeds.append(_beta_shape(n, a, b))
    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))
    for frac in (0.05, 0.15, 0.30):
        seeds.append(_block_uniform(n, frac, rng))
    for k in (int(0.01 * n), int(0.03 * n)):
        for _ in range(2):
            seeds.append(_sparse_random(n, k, rng))
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Previous solution, if compatible
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Sanity‑check / normalise all seeds
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """Optimise the smoothed max‑convolution objective using L‑BFGS‑B."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser not available.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1e3, 2e3, 5e3,
                         1e4, 2e4, 5e4, 1e5,
                         2e5, 5e5, 1e6]

    # Initialise logits (softmax parametrisation)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp (stable)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp                     # soft‑max weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact mirror‑descent (exponentiated‑gradient) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the exact max‑convolution.
    Returns best sequence and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        # all indices attaining the maximum (within tolerance)
        active = np.where(np.abs(conv - max_val) < 1e-12)[0]

        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx
            g[idx] += 2.0 * a[rev]

        # Diminishing stepsize
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 50000 == 0:
                print(f"[ExpGrad] iter={it} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer optimiser
# ----------------------------------------------------------------------
def _mass_transfer_opt(start_seq: np.ndarray,
                       n: int,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Fast local optimiser that moves a tiny amount of probability mass
    between two coordinates, accepting probabilistically according to a
    temperature schedule.
    """
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_max = float(np.max(conv))
    best_a = a.copy()
    best_val = cur_max

    rng = np.random.default_rng()
    temperature = 0.02          # relative tolerance for uphill moves
    decay = 0.9995              # geometric decay per accepted move
    it = 0

    while time.time() < deadline:
        # choose source weighted by its mass (prefer large entries)
        if a.max() < 1e-16:
            break
        src = rng.choice(n, p=a)
        src_mass = a[src]
        if src_mass <= 1e-15:
            continue
        # choose target uniformly at random
        tgt = rng.integers(n)
        while tgt == src:
            tgt = rng.integers(n)

        # move a fraction of the source mass
        eps = 0.2 * src_mass               # 20 % move; can be tuned
        if eps < 1e-14:
            continue

        # propose new vector and recompute convolution (full FFT – still cheap)
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[tgt] += eps
        conv_prop = _conv_fft(a_prop, a_prop)
        new_max = float(np.max(conv_prop))

        # accept if improvement, or with probability exp(-(Δ)/temp)
        if new_max < cur_max:
            accept = True
        else:
            # Δ = new_max - cur_max  (positive)
            prob = math.exp(-(new_max - cur_max) / (temperature * cur_max + 1e-18))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            conv = conv_prop
            cur_max = new_max
            temperature *= decay   # cool down slowly
            if cur_max < best_val:
                best_val = cur_max
                best_a = a.copy()
        else:
            # occasional temperature increase after many rejections
            if it % 4000 == 0 and temperature < 0.03:
                temperature *= 1.05

        it += 1
        if it % 20000 == 0:
            print(f"[MassTransfer] it={it} best max={best_val:.12f}")

    return best_a, 2.0 * n * best_val


# ----------------------------------------------------------------------
# Random Gaussian perturbation optimiser
# ----------------------------------------------------------------------
def _random_perturb_opt(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 15000) -> tuple[np.ndarray, float]:
    """
    Occasionally add Gaussian noise to the current best solution,
    re‑project onto the simplex and keep the improvement if any.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    rng = np.random.default_rng()
    sigma = 0.05  # initial noise magnitude

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # adapt noise magnitude slowly
        sigma = max(0.01, sigma * 0.9999)

        noisy = a + rng.normal(scale=sigma, size=n)
        noisy = np.maximum(noisy, 0.0)
        s = noisy.sum()
        if s <= 0:
            continue
        noisy = noisy / s

        val = _objective_surrogate(noisy)
        if val < best_val:
            best_val = val
            best_seq = noisy.copy()
            a = noisy.copy()
            if it % 2000 == 0:
                print(f"[Perturb] iter={it} improved to {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective.  Returns the best sequence found as a
    plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and timing
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # 1. Determine problem size (use previous length if available)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    n = 4000  # default length
    prev_arr = None
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size > 0:
                n = int(arr.size)
                prev_arr = arr
        except Exception:
            prev_arr = None

    print(f"\n--- Optimisation start (n={n}) ---")

    # ------------------------------------------------------------------
    # 2. Seed generation
    # ------------------------------------------------------------------
    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # 3. Preliminary evaluation and selection of promising seeds
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 4. Global best initialisation
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Continuation L‑BFGS + exact mirror‑descent on each top seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline before finishing L‑BFGS runs.")
            break

        print(f"\n--- L‑BFGS run on seed #{idx+1} ---")
        try:
            seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                             n,
                                             deadline,
                                             max_iter_per_beta=5000)
        except RuntimeError:
            # SciPy not available – skip L‑BFGS
            seq_opt = seed_vec.copy()
            print("[Info] SciPy missing, skipping L‑BFGS.")

        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] new global best = {best_val:.6f}")

        # Exact mirror‑descent polishing
        if time.time() < deadline:
            print("[ExpGrad] polishing ...")
            seq_exp, val_exp = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=200_000)
            if val_exp < best_val:
                best_seq = seq_exp.copy()
                best_val = val_exp
                print(f"[ExpGrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing mass‑transfer local optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mass‑transfer local optimisation (SA) ---")
        seq_mt, val_mt = _mass_transfer_opt(best_seq, n, deadline)
        if val_mt < best_val:
            best_seq = seq_mt.copy()
            best_val = val_mt
            print(f"[MassTransfer] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Random perturbation refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random perturbation refinement ---")
        seq_pert, val_pert = _random_perturb_opt(best_seq, n, deadline)
        if val_pert < best_val:
            best_seq = seq_pert.copy()
            best_val = val_pert
            print(f"[Perturb] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final evaluation with official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally for potential warm‑starts in subsequent runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
