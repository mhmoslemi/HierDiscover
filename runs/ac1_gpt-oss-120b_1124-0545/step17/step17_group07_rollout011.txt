def _adam_exp_gradient(start_seq, n, deadline, max_iters=500000, lr=0.5):
    a = start_seq.copy()
    logit = np.log(a + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline: break
        # compute conv
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # compute subgrad
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # gradient w.r.t. logits
        dot = np.dot(g, a)
        grad_logit = a * (g - dot)   # chain rule
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_logit
        v = beta2 * v + (1 - beta2) * (grad_logit ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        # step size schedule
        eta = lr / (np.sqrt(it) + 1)
        logit -= eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(logit)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

"""
Hybrid optimization for the step‑function problem.

The pipeline:
1. Build a very diverse seed pool (analytic windows, beta‑skewed,
   Dirichlet draws, random uniform blocks, previous best, …).
2. Quick surrogate evaluation (2·n·max(conv) for ℓ₁‑normalised vectors)
   and keep the best few seeds.
3. Refine each seed with a continuation L‑BFGS run on a smoothed
   max‑convolution (log‑sum‑exp) – this quickly moves the iterate into a
   good basin.
4. Perform a long Adam‑accelerated exponentiated‑gradient descent on the
   exact max‑convolution (sub‑gradient of the active entry).  The iterate
   always stays on the probability simplex.
5. Finish with a cheap simulated‑annealing walk in logit space to hop out
   of possible local minima.
6. Return the best sequence (list of non‑negative floats) and store it in
   the global ``height_sequence_1`` for the next run.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# SciPy is optional – L‑BFGS is only used when available.
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (surrogate works on ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv) for a ℓ₁‑normalised sequence."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by evaluate_sequence (no normalisation)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _linear_decrease(n: int) -> np.ndarray:
    """Monotonically decreasing line (peak at left, zero at right)."""
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    return a / a.sum()


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _double_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta(α,β) on {0,…,n‑1} with half‑offset to avoid zeros."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 0.5) ** (alpha - 1.0)) * ((n - i - 0.5) ** (beta - 1.0))
    return a / a.sum()


def _sin_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p  (symmetric, zero at the ends)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n) ** p
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    # Normalised distance from centre
    x = (2.0 * i) / (n - 1) - 1.0
    a = np.i0(beta * np.sqrt(1 - x ** 2))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_uniform_block(n: int, block_size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of `block_size` indices."""
    idx = rng.choice(n, size=block_size, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / block_size
    return a / a.sum()


def _two_blocks(n: int, size1: int, size2: int, start1: int, start2: int) -> np.ndarray:
    """Two uniform blocks (non‑overlapping)."""
    a = np.zeros(n, dtype=np.float64)
    a[start1:start1 + size1] = 1.0 / size1
    a[start2:start2 + size2] = 1.0 / size2
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse collection of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families (core)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_linear_decrease(n))

    # Gaussian (different bandwidths)
    for sigma in (0.06, 0.12, 0.18, 0.30):
        seeds.append(_gaussian(n, sigma))

    # Power‑law families
    for p in (0.0, 0.25, 0.5, 0.75, 1.0, 2.0):
        seeds.append(_power(n, p))

    # Exponential / double‑exponential
    for rate in (0.001, 0.005, 0.01, 0.02):
        seeds.append(_exponential(n, rate))
        seeds.append(_double_exponential(n, rate))

    # Window functions
    seeds.append(_hann_window(n))
    for beta in (3.0, 5.0, 7.0, 9.0):
        seeds.append(_kaiser_window(n, beta))

    # Sin‑power families
    for p in (1.0, 2.0, 4.0, 8.0):
        seeds.append(_sin_power(n, p))

    # Beta‑shaped (both symmetric and skewed)
    betas = (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0)
    for a in betas:
        seeds.append(_beta_distribution(n, a, a))      # symmetric
        seeds.append(_beta_distribution(n, a, 3.0))    # left‑skewed
        seeds.append(_beta_distribution(n, 3.0, a))    # right‑skewed

    # Random Dirichlet draws (various concentrations)
    conc_list = (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    for conc in conc_list:
        for _ in range(3):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                             # as‑is
            seeds.append(np.sort(raw)[::-1])             # sorted
            # a shuffled version – helps break symmetry
            seeds.append(raw[rng.permutation(n)])

    # Sparse uniform blocks (different sizes)
    for blk in (50, 80, 120):
        for _ in range(2):
            seeds.append(_sparse_uniform_block(n, blk, rng))

    # Two‑block constructions (far apart)
    for sz in (30, 50):
        start1 = rng.integers(0, n // 3)
        start2 = rng.integers(2 * n // 3, n - sz)
        seeds.append(_two_blocks(n, sz, sz, start1, start2))

    # Previous solution, if provided
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Normalise everything and filter out any pathological case
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smoothed max‑conv."""
    if _opt is None:                     # pragma: no cover
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 3.0, 7.0, 15.0, 30.0, 60.0,
            120.0, 250.0, 500.0, 1000.0, 2500.0,
            5000.0, 10000.0, 25000.0, 50000.0, 100000.0
        ]

    # optimisation variables are logits (softmax stays on simplex)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on simplex
            conv = _conv_fft(a, a)
            m = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = m
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of smooth max w.r.t. a
            w = exp_beta / sum_exp                    # soft weights (sum to 1)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # convolution rev(a) with w
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={
                                    'maxiter': max_iter_per_beta,
                                    'disp': False,
                                    'ftol': 1e-12
                                })
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam‑accelerated exponentiated‑gradient on the exact max‑conv
# ----------------------------------------------------------------------
def _adam_exp_gradient_opt(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponential‑gradient) with Adam momentum on the *exact*
    max‑convolution objective.
    """
    a = start_seq.copy()
    logit = np.log(np.clip(a, 1e-30, None))

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # --- convolution & locate active index --------------------------------
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # --- sub‑gradient of the active entry ---------------------------------
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # --- gradient w.r.t. logits (softmax chain rule) ----------------------
        dot = np.dot(g, a)
        grad_logit = a * (g - dot)

        # --- Adam update -------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_logit
        v = beta2 * v + (1.0 - beta2) * (grad_logit ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (larger at the start, decay ~1/√t)
        eta = 0.6 / (np.sqrt(it) + 1.0)

        logit -= eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(logit)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (logit‑space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """Random walk in log‑softmax space with linear cooling."""
    rng = np.random.default_rng()
    logits = np.log(np.clip(start_seq, 1e-30, None))

    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.12 * (1.0 - progress) + 1e-6          # temperature
        sigma = 0.10 * (1.0 - progress) + 5e-5       # proposal width

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and time budget
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # reserve a few seconds for the final ``evaluate_sequence`` call
    deadline = start_time + budget_s - 3.0

    # ------------------------------------------------------------------
    # 1. Problem size – keep the same length as any previous solution,
    #    otherwise fall back to 4000.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and len(previous) > 0:
        n = len(previous)
        prev_arr = np.asarray(previous, dtype=np.float64)
    else:
        n = 4000
        prev_arr = None

    print(f"\n=== Optimisation start (n={n}) ===")

    # ------------------------------------------------------------------
    # 2. Seed pool construction
    # ------------------------------------------------------------------
    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} raw seeds.")

    # quick surrogate evaluation (exact for ℓ₁‑normalised vectors)
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 60   # a bit larger than before, still cheap
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nBest surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. L‑BFGS refinement on the top seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    # aggressive beta schedule – ends with a very sharp approximation
    beta_schedule = [
        1.0, 3.0, 7.0, 15.0, 30.0, 60.0,
        120.0, 250.0, 500.0, 1000.0, 2500.0,
        5000.0, 10000.0, 25000.0, 50000.0, 100000.0
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS runs.")
            break

        # allocate a modest slice of the remaining budget
        remaining = deadline - time.time()
        max_iter_beta = int(min(6000, remaining * 0.03 * 1e5))
        max_iter_beta = max(3000, max_iter_beta)

        print(f"\n--- L‑BFGS on seed #{idx + 1} (max_iter/β={max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq, global_best_val = seq_opt.copy(), val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long Adam‑accelerated exponentiated‑gradient optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam‑exp‑gradient polishing (long run) ---")
        # Determine how many iterations we can afford (rough estimate)
        # One iteration ≈ a single FFT + a few vector ops.
        # Empirically ~2 · 10⁻⁴ s for n≈4000 on the target hardware.
        # We'll simply run until the time budget expires.
        seq_exp, val_exp = _adam_exp_gradient_opt(global_best_seq,
                                                  n,
                                                  deadline,
                                                  max_iters=2_000_000)
        if val_exp < global_best_val:
            global_best_seq, global_best_val = seq_exp.copy(), val_exp
            print(f"[Adam] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Simulated‑annealing final refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing_opt(global_best_seq,
                                                  n,
                                                  deadline,
                                                  max_iters=500_000)
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa.copy(), val_sa
            print(f"[SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
          f"true evaluate_sequence = {true_val:.6f if true_val is not None else 'N/A'}")

    # store for the next iteration (if any)
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
