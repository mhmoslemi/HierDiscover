"""
Non‑negative step‑function optimiser – enhanced edition.

Main ideas
-----------
1. Build a rich pool of analytic / random seed vectors (ℓ₁‑normalised).
2. Run a continuation L‑BFGS optimiser on a log‑sum‑exp smooth‑max of the
   self‑convolution, using a longer β‑schedule (up to 1e7) than before.
3. Refine the best candidate with a *full‑set* exponentiated‑gradient
   mirror‑descent: the sub‑gradient is obtained from **all** active
   (maximal) convolution entries, not just a single arg‑max.
4. Apply a deterministic, gradient‑guided mass‑transfer local optimiser.
   The source of transferred mass is chosen from the coordinates with
   largest sub‑gradient (they increase the objective the most); the target
   is the coordinate with the smallest sub‑gradient.
5. If time permits, perform a final short random‑move mass‑transfer pass
   and a final L‑BFGS refinement.
6. Return the best sequence found (as a plain list of floats).  The whole
   pipeline respects the global deadline (≈ 1000 s) and never returns an
   empty / invalid sequence.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# -------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# -------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised seq.
    This coincides with the official evaluate_sequence value up to rounding.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Analytic families (all return ℓ₁‑normalised vectors)
# -------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block of length block_frac·n."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    seeds.append(_double_gaussian(n, sigma_frac=0.10, shift_frac=0.30))
    seeds.append(_double_gaussian(n, sigma_frac=0.15, shift_frac=0.25))

    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(10):
        seeds.append(_random_exponential(n, rng))

    for nf in (2, 3, 4, 5):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    for k in (int(0.01 * n), int(0.03 * n), int(0.07 * n)):
        for _ in range(4):
            seeds.append(_sparse_random(n, k, rng))

    for bf in (0.05, 0.15, 0.30):
        for _ in range(4):
            seeds.append(_block_uniform(n, bf, rng))

    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    seeds.append(_alternating_dirichlet(n, rng))

    # include previous solution (if compatible)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # sanitise
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# -------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# -------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 20000) -> np.ndarray:
    """
    Optimise the smoothed‑max convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised probability vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser required for L‑BFGS.")

    if beta_schedule is None:
        # extended schedule – final β much larger than before
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 1e5,
            2e5, 5e5, 1e6, 2e6, 5e6, 1e7
        ]

    v = np.log(start_vec + 1e-30)   # logits (stay on simplex via softmax)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)               # always on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (weighted sum of convolutions)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# -------------------------------------------------------------
# Exponentiated‑gradient optimiser (full active‑set sub‑gradient)
# -------------------------------------------------------------
def _exp_gradient_opt_full(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 500_000,
                          tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Sub‑gradient is built from *all* active (maximal) indices.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        # active set (indices within tolerance of the max)
        active = np.where(conv >= max_val - tol)[0]

        # build weight vector for the active set (equal weight)
        w = np.zeros(len(conv), dtype=np.float64)
        w[active] = 1.0

        rev_a = a[::-1]
        grad_full = _conv_fft(rev_a, w)
        g = 2.0 * grad_full[n - 1:n - 1 + n]         # sub‑gradient of max

        # decreasing step size (tuned a bit)
        eta = 0.4 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGradFull] it={it} best={best_val:.12f}")

    return best_seq, best_val


# -------------------------------------------------------------
# Gradient‑guided mass‑transfer local optimiser
# -------------------------------------------------------------
def _mass_transfer_grad_opt(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Fast O(n) local optimiser.
    The source of transferred mass is chosen from the index with the largest
    sub‑gradient (most responsible for the current max); the target is the
    index with the smallest sub‑gradient.
    """
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    max_val = float(conv.max())
    best_a = a.copy()
    best_val = max_val

    rng = np.random.default_rng()
    step_factor = 0.25         # initial fraction of source mass to move
    it = 0
    no_improve = 0

    while time.time() < deadline:
        # --- compute active set and sub‑gradient once per iteration ---
        max_val = float(conv.max())
        active = np.where(conv >= max_val - tol)[0]
        w = np.zeros(len(conv), dtype=np.float64)
        w[active] = 1.0
        rev_a = a[::-1]
        grad_full = _conv_fft(rev_a, w)
        g = 2.0 * grad_full[n - 1:n - 1 + n]

        # choose source (high contribution) and target (low contribution)
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]
        tgt = np.argmin(g)
        if tgt == src:
            # fall back to a random different index
            tgt = rng.integers(n - 1)
            while tgt == src:
                tgt = rng.integers(n - 1)

        src_mass = a[src]
        eps = step_factor * src_mass
        if eps < 1e-15:
            # shrink step size if we are moving too little
            step_factor *= 0.5
            continue

        # ---- tentative update of convolution (exact) ----
        new_conv = conv.copy()
        # linear part
        new_conv[src:src + n] -= 2.0 * eps * a
        new_conv[tgt:tgt + n] += 2.0 * eps * a
        # quadratic delta*delta part
        new_conv[n - 1] += 2.0 * eps * eps
        offset = tgt - src
        idx1 = n - 1 + offset
        new_conv[idx1] -= eps * eps
        if offset != 0:
            idx2 = n - 1 - offset
            new_conv[idx2] -= eps * eps

        new_max = float(np.max(new_conv))

        # accept if improvement
        if new_max < max_val - 1e-16:
            a[src] = src_mass - eps
            a[tgt] += eps
            conv = new_conv
            max_val = new_max
            no_improve = 0
            if new_max < best_val - 1e-16:
                best_val = new_max
                best_a = a.copy()
                # optionally tighten step factor when we make progress
                step_factor = min(0.5, step_factor * 1.1)
        else:
            no_improve += 1
            # slowly decay step factor if stuck
            if no_improve > 4000:
                step_factor *= 0.7
                if step_factor < 1e-4:
                    step_factor = 0.25
                no_improve = 0

        it += 1
        if it % 20000 == 0:
            print(f"[MassTransferGrad] it={it} best max={best_val:.12f}")

    return best_a, 2.0 * n * best_val


# -------------------------------------------------------------
# Random (uninformed) mass‑transfer local optimiser (fallback)
# -------------------------------------------------------------
def _mass_transfer_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Same as the original implementation – useful as a final clean‑up pass."""
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    max_val = float(np.max(conv))
    best_a = a.copy()
    best_val = max_val

    rng = np.random.default_rng()
    step_factor = 0.25
    no_improve = 0
    it = 0
    center = n - 1

    while time.time() < deadline:
        if a.max() < 1e-16:
            break
        src = rng.choice(n, p=a)                # source index
        src_mass = a[src]
        if src_mass <= 1e-15:
            continue
        tgt = rng.integers(n)
        while tgt == src:
            tgt = rng.integers(n)

        eps = step_factor * src_mass
        if eps < 1e-14:
            continue

        new_conv = conv.copy()
        new_conv[src:src + n] -= 2.0 * eps * a
        new_conv[tgt:tgt + n] += 2.0 * eps * a
        new_conv[center] += 2.0 * eps * eps
        offset = tgt - src
        idx1 = center + offset
        new_conv[idx1] -= eps * eps
        if offset != 0:
            idx2 = center - offset
            new_conv[idx2] -= eps * eps

        new_max = float(np.max(new_conv))

        if new_max < max_val - 1e-16:
            a[src] = src_mass - eps
            a[tgt] += eps
            conv = new_conv
            max_val = new_max
            no_improve = 0
            if new_max < best_val:
                best_val = new_max
                best_a = a.copy()
        else:
            no_improve += 1

        it += 1
        if no_improve > 5000:
            step_factor *= 0.5
            if step_factor < 1e-4:
                step_factor = 0.25
            no_improve = 0

        if it % 20000 == 0:
            print(f"[MassTransfer] it={it} best max={best_val:.12f}")

    return best_a, 2.0 * n * best_val


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and timing
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size – keep the historic size (4000)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # 3. Evaluate seeds (surrogate) and keep the best few
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print("\nTop seeds (surrogate values):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 4. Continuation L‑BFGS on the elite seeds
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # extended β‑schedule (final β = 1e7)
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        10000.0, 20000.0, 50000.0, 1e5,
        2e5, 5e5, 1e6, 2e6, 5e6, 1e7
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS runs.")
            break
        # allocate a modest amount of iterations per seed (adaptive)
        remaining = deadline - time.time()
        max_iter = int(min(20000, remaining * 0.02 * 1e5))
        max_iter = max(8000, max_iter)

        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter={max_iter}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] new global best = {best_val:.6f}")

        # --------------------------------------------------------
        # Exponentiated‑gradient polishing (full active set)
        # --------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_exp = int(min(600_000, remaining * 0.07 * 1e5))
            max_exp = max(300_000, max_exp)
            print(f"[ExpGradFull] polishing with up to {max_exp} iterations")
            seq_exp, val_exp = _exp_gradient_opt_full(best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=max_exp)
            if val_exp < best_val:
                best_seq = seq_exp.copy()
                best_val = val_exp
                print(f"[ExpGradFull] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Gradient‑guided mass‑transfer local optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Gradient‑guided mass‑transfer optimisation ---")
        seq_mt, val_mt = _mass_transfer_grad_opt(best_seq, n, deadline)
        if val_mt < best_val:
            best_seq = seq_mt.copy()
            best_val = val_mt
            print(f"[MassTransferGrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Fallback random mass‑transfer pass (short)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random mass‑transfer clean‑up ---")
        seq_rand, val_rand = _mass_transfer_opt(best_seq, n, deadline)
        if val_rand < best_val:
            best_seq = seq_rand.copy()
            best_val = val_rand
            print(f"[MassTransferRandom] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final L‑BFGS refinement (if time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS refinement ---")
        seq_final = _lbfgs_smooth_optimize(best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=beta_schedule,
                                           max_iter_per_beta=8000)
        val_final = _objective_surrogate(seq_final)
        if val_final < best_val:
            best_seq = seq_final.copy()
            best_val = val_final
            print(f"[Final L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
