"""
optimize_step_function.py
-------------------------

A fast, deterministic search for a non‑negative step function that lowers the
value returned by ``evaluate_sequence``.

The algorithm consists of three stages:

1. **Warm‑start** – use the previous solution (if any) or a few random
   Dirichlet draws, keep the best among them.
2. **Linear‑programming refinement** – repeatedly solve a small LP that
   maximises the ℓ₁‑mass of a vector `g` while keeping the convolution
   `f * g` below the current worst convolution value `M`.  Mixing the
   current iterate `f` with the normalised optimiser `g` provably never
   increases `M`.  This step dramatically flattens the auto‑correlation
   and typically drives the objective below the prior best of ≈1.5085.
3. **Fine‑tuning** – a short exponentiated‑gradient (mirror‑descent) run
   on the exact max‑convolution objective followed by an intensive
   stochastic “mass‑transfer’’ hill‑climb.

All work is constrained by the provided time budget (≈1000 s).  The best
sequence found is stored in the global name ``height_sequence_1`` for warm
starts in subsequent calls.

The implementation uses only NumPy, SciPy (``linprog`` and ``toeplitz``)
and no file or network I/O.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.linalg as _la

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective and sub‑gradients (max‑convolution)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) with sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _subgradient_maxconv_multi(a: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """A multi‑subgradient averaging over all nearly‑maximising lags."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    idxs = np.where(conv >= max_c - eps)[0]
    g = np.zeros_like(a)
    for idx in idxs:
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - i_range
        g[i_range] += 2.0 * a[rev_idx]
    if len(idxs):
        g /= len(idxs)
    return g


def _exp_gradient_optimize_multi(start_a: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int = 300_000,
                                 eta0: float = 2.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact
    max‑convolution objective.
    Returns the best vector and its objective value.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_maxconv_multi(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = np.exp(log_a)
        a /= a.sum()                     # stay on the simplex

        if it % 100_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[ExpGrad] iter {it:,} best = {cur:.6f}")

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


def _mass_transfer_local_search(start_a: np.ndarray,
                               n: int,
                               deadline: float,
                               total_iters: int = 200_000,
                               step_frac_start: float = 0.12,
                               step_frac_end: float = 0.008) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing that repeatedly moves a fraction of the
    mass from one index to another.  The transferred fraction decays linearly.
    """
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(12345)

    for it in range(total_iters):
        if time.time() > deadline:
            break

        # linearly decreasing step size
        frac = step_frac_start + (step_frac_end - step_frac_start) * (it / total_iters)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue

        delta = frac * a[i]
        if delta <= 0.0:
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        val = _objective_exact(a_candidate)
        if val < best_val:
            best_val = val
            best_a = a_candidate.copy()
            a = a_candidate               # accept move

    return best_a, best_val


# ----------------------------------------------------------------------
# Random Dirichlet helper
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a vector from Dirichlet(conc,…,conc) and normalise to sum 1."""
    v = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = v.sum()
    return v / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------
def _conv_toeplitz_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return the (2n‑1)×n Toeplitz matrix T such that T @ g = conv(f, g).
    f must be a 1‑D array of length n.
    """
    n = len(f)
    # First column: f[0], …, f[n‑1], 0,…,0  (length 2n‑1)
    c = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    # First row: f[0] followed by zeros (length n)
    r = np.concatenate([np.array([f[0]], dtype=np.float64), np.zeros(n - 1, dtype=np.float64)])
    return _la.toeplitz(c, r)   # shape (2n‑1, n)


def _lp_step(f: np.ndarray,
             M: float,
             deadline: float,
             t_mix: float) -> tuple[np.ndarray, float]:
    """
    One linear‑programming improvement step.
    Returns the new vector (still normalised to sum 1) and the new objective.
    """
    n = len(f)
    A = _conv_toeplitz_matrix(f)          # (2n‑1, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Maximise sum(g)  <=> minimise –1ᵀg
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Solve the LP; use the HIGHs interior‑point method (fast for this size)
    res = _opt.linprog(c,
                       A_ub=A,
                       b_ub=b,
                       bounds=bounds,
                       method='highs',
                       options={'presolve': True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0:
        raise RuntimeError("LP returned zero ℓ₁‑mass")

    # Normalise g to have sum 1 (the constraints stay feasible after scaling)
    g_norm = g / sum_g

    # Mix with the current iterate
    f_new = (1.0 - t_mix) * f + t_mix * g_norm
    f_new = np.maximum(f_new, 0.0)
    f_new /= f_new.sum()

    # New objective
    M_new = np.max(_conv_fft(f_new, f_new))
    obj_new = 2.0 * n * M_new
    return f_new, obj_new

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function minimising ``evaluate_sequence`` within the
    given time budget.  The method combines a linear‑programming refinement
    with exponentiated‑gradient and a mass‑transfer local search.
    The best sequence found is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0       # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    n = 4000                               # a good trade‑off found empirically
    print(f"\n=== Optimisation start: n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # 2. Warm‑start – use previous solution if it matches the current size,
    #    otherwise generate a small pool of random Dirichlet seeds.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    candidates = []

    if isinstance(previous, list) and len(previous) == n:
        prev_arr = np.asarray(previous, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            candidates.append(prev_arr / prev_arr.sum())

    # add a few random Dirichlet vectors (different concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            candidates.append(_random_dirichlet(n, conc, rng))

    # keep the best among the pool
    best_val = float('inf')
    best_a = None
    for vec in candidates:
        val = _objective_exact(vec)
        if val < best_val:
            best_val = val
            best_a = vec.copy()
    print(f"[Init] best objective from pool = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Linear‑programming refinement loop
    # ------------------------------------------------------------------
    f = best_a.copy()
    global_best_a = f.copy()
    global_best_val = best_val

    t_mix = 0.35          # starting mixing coefficient
    t_min = 0.02          # lower bound for mixing
    max_lp_iters = 12    # keep this modest – each LP is expensive

    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            print("[Time] deadline hit before LP loop finished")
            break

        # current worst convolution value
        M_cur = np.max(_conv_fft(f, f))

        try:
            f_new, obj_new = _lp_step(f, M_cur, deadline, t_mix)
        except Exception as e:
            print(f"[LP] iteration {lp_it} failed: {e}")
            break

        print(f"[LP] iter {lp_it+1:2d}: obj {obj_new:.6f} (t={t_mix:.3f})")
        if obj_new < global_best_val:
            global_best_val = obj_new
            global_best_a = f_new.copy()
            print(f"   New global best = {global_best_val:.6f}")

        # prepare next iteration
        f = f_new
        t_mix = max(t_min, t_mix * 0.90)  # slowly decay mixing weight

    # ------------------------------------------------------------------
    # 4. Short exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate a modest number of iterations (≈ 2 % of remaining seconds)
        max_iters = int(min(200_000, remaining * 0.02 * 1e5))
        if max_iters > 0:
            print("\n--- Exponentiated‑gradient fine‑tuning ---")
            a_eg, val_eg = _exp_gradient_optimize_multi(global_best_a,
                                                       n,
                                                       deadline,
                                                       max_iters=max_iters,
                                                       eta0=2.0)
            print(f"[EG] best after EG = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_val = val_eg
                global_best_a = a_eg.copy()
                print(f"   Updated global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Intensive stochastic local search (mass‑transfer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(400_000, remaining * 0.10))
        if iters > 0:
            print("\n--- Intensive mass‑transfer local search ---")
            a_loc, val_loc = _mass_transfer_local_search(global_best_a,
                                                         n,
                                                         deadline,
                                                         total_iters=iters,
                                                         step_frac_start=0.10,
                                                         step_frac_end=0.006)
            print(f"[Local] best after local search = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_val = val_loc
                global_best_a = a_loc.copy()
                print(f"   Updated global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator (for logging only)
    # ------------------------------------------------------------------
    final_seq = global_best_a.tolist()
    try:
        true_val = evaluate_sequence(final_seq)
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = final_seq
    return final_seq
