def _proj_simplex(v):
    # v arbitrary
    # compute projection onto simplex
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _subgradient_opt(start_seq, n, deadline, max_iters=500_000, eta0=0.5):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        conv = _conv_fft(a,a)
        max_val = conv.max()
        # find active indices
        active = (conv >= max_val - 1e-12)
        # create weights w equal to active.astype(float) (or normalized)
        w = active.astype(float)  # length 2n-1
        # optional normalize by sum(w)
        sum_w = w.sum()
        if sum_w > 0:
            w /= sum_w
        # compute subgradient
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)  # length 3n-2
        g = 2.0 * full[n-1:n-1+n]
        # step size

        a = _proj_simplex(a_new)
        # ensure nonnegative
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:

            best_seq = a.copy()

def _proj_simplex(v):
    """Projection onto simplex (nonnegative, sum=1)"""
    # using method by Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _adam_optimize(start_vec, n, deadline, beta_schedule, iters_per_beta=5000, lr=0.1):
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    t_iter=0
    for beta in beta_schedule:
        if time.time()>deadline: break
        # compile gradient function like defined earlier
        beta_val=float(beta)
        for _ in range(iters_per_beta):
            if time.time()>deadline: break
            t_iter+=1
            a = _softmax(v)
            conv = _conv_fft(a,a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp
            # grad w.r.t a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            # Adam update
            m = beta1 * m + (1-beta1) * grad_v
            v_hat = beta2 * v_hat + (1-beta2) * (grad_v * grad_v)
            m_hat = m / (1 - beta1**t_iter)
            v_hat_corr = v_hat / (1 - beta2**t_iter)
            v -= lr * m_hat / (np.sqrt(v_hat_corr) + eps)
        # after each beta, maybe evaluate surrogate and keep best.
        cur_a = _softmax(v)
        cur_val = _objective_surrogate(cur_a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_a.copy()
    return best_seq
```

"""
Propose a non‑negative step‑function (list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines:

1. **Diverse seeding** – analytic windows, random Dirichlet / exponential
   draws, low‑frequency cosine curves, plus the previous solution if
   available.  The best ``top_k`` candidates (by the exact surrogate
   2·n·max(conv) for ℓ₁‑normalised vectors) are retained.

2. **Mirror‑descent (exponentiated‑gradient)** – optimisation on the simplex
   using the exact sub‑gradient of the active convolution entry.  A large
   number of iterations are performed within the bulk of the time budget.

3. **Continuation L‑BFGS** – smooth the max with a log‑sum‑exp
   surrogate and run L‑BFGS‑B with a long schedule of β (inverse temperature)
   values and generous iteration limits, pushing the solution close to a
   true local minimum.

4. **Greedy hill‑climbing** – tiny mass‑transfer moves between coordinates
   (and occasional larger moves) that strictly improve the exact surrogate.

All steps respect the time limit (default 1000 s).  The best sequence found
is stored in the global variable ``height_sequence_1`` for reuse.
"""

import time
import math
import heapq
import numpy as np

# Numeric utilities (FFT‑based convolution, softmax, simplex projection)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a vector normalised to sum = 1:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:

    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:

def _gaussian(n: int, sigma_frac: float) -> np.ndarray:

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:

def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _centered_exponential(n: int, lam: float = 0.005) -> np.ndarray:

    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.exp(-lam * np.abs(i - c))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Updates ``log_a ← log_a – η·g`` with η = η₀ / sqrt(t).
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 25_000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    The variables live in log‑softmax space (v), guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0]

    # initialise logits
    v = np.log(start_vec + 1e-30)
    best_a = None
    best_obj = np.inf

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # stabilise exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # soft‑max weights over convolution entries
            w = exp_beta / sum_exp

            # gradient wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

        # keep best intermediate solution
        a_candidate = _softmax(v)
        cand_obj = _objective_surrogate(a_candidate)
        if cand_obj < best_obj:
            best_obj = cand_obj
            best_a = a_candidate.copy()

    if best_a is None:
        best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Simple hill‑climbing / simulated‑annealing local refinement
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny (and occasional medium) mass‑transfer moves on the simplex.
    Accept only strictly improving moves.  Returns the best sequence and
    its surrogate value.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # move a random fraction of mass from i to j (0.1 % – 5 %)
        frac = rng.uniform(0.001, 0.05)
        eps = cur[i] * frac
        if eps <= 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective_surrogate(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the time budget and stores the best found sequence in
    the global variable ``height_sequence_1``.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and timing
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Problem size (compromise between quality and speed)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Helper to maintain a heap of the best seeds
    # ------------------------------------------------------------------
    top_k = 15
    best_heap = []   # stores (-value, vector)

    def _push_candidate(vec: np.ndarray):
        """Insert a candidate into the heap maintaining the best ``top_k``."""
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if val < -best_heap[0][0]:
                heapq.heappushpop(best_heap, (-val, vec))

    # ------------------------------------------------------------------
    # 3. Deterministic analytic seeds (and a few extra windows)
    # ------------------------------------------------------------------
    _push_candidate(_uniform(n))
    _push_candidate(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 8):
        _push_candidate(_sine_window(n, gamma))
    _push_candidate(_hamming_window(n))
    _push_candidate(_blackman_window(n))
    _push_candidate(_nuttall_window(n))
    for beta in np.linspace(1.0, 15.0, 6):
        _push_candidate(_kaiser_window(n, beta))
    for sigma in np.linspace(0.09, 0.26, 9):
        _push_candidate(_gaussian(n, sigma))
    for p in np.linspace(0.0, 2.0, 7):
        _push_candidate(_power(n, p))
    alphas = np.linspace(0.2, 5.0, 4)
    betas = np.linspace(0.2, 5.0, 4)
    for a_ in alphas:
        for b_ in betas:
            _push_candidate(_beta_shape(n, a_, b_))
    for alpha in np.linspace(0.1, 0.9, 5):
        _push_candidate(_tukey_window(n, alpha))
    for sigma in (0.10, 0.12):
        for shift in (0.20, 0.25, 0.30):
            _push_candidate(_double_gaussian(n, sigma, shift))
    # extra centred exponential window
    _push_candidate(_centered_exponential(n, lam=0.003))

    # ------------------------------------------------------------------
    # 4. Massive random seeding (Dirichlet, exponential, low‑freq curves)
    # ------------------------------------------------------------------
    conc_choices = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    random_seed_budget = 5000
    for _ in range(random_seed_budget):
        if time.time() > deadline:
            break
        conc = rng.choice(conc_choices)
        vec = _random_dirichlet(n, conc, rng)
        # random cyclic shift to add diversity
        shift = rng.integers(n)
        if shift:
            vec = np.roll(vec, shift)
        _push_candidate(vec)

    for _ in range(800):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        shift = rng.integers(n)
        if shift:
            vec = np.roll(vec, shift)
        _push_candidate(vec)

    for nf in (2, 3, 5):
        for _ in range(300):
            if time.time() > deadline:
                break
            vec = _lowfreq_curve(n, nf, rng)
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
            _push_candidate(vec)

    # ------------------------------------------------------------------
    # 5. Re‑use previous solution if available and matches size
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0:
                prev_norm = prev_arr / prev_arr.sum()
                _push_candidate(prev_norm)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 6. Extract the best seeds (ordered from best to worst)
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seed phase] kept {len(seeds)} best candidates, best surrogate = {_objective_surrogate(seeds[0]):.6f}")

    # ------------------------------------------------------------------
    # 7. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Exponentiated‑gradient polishing (majority of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        eg_time = (deadline - time.time()) * 0.65   # allocate roughly 65 % of remaining time
        eg_deadline = time.time() + eg_time
        seq_eg, val_eg = _exp_gradient_opt(global_best_seq,
                                           n,
                                           eg_deadline,
                                           max_iters=5_000_000,
                                           eta0=6.0)
        if val_eg < global_best_val:
            global_best_seq = seq_eg
            global_best_val = val_eg
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Continuation L‑BFGS fine‑tuning (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n--- L‑BFGS fine‑tuning ---")
        lbfgs_deadline = deadline - 5.0
        seq_lbfgs = _lbfgs_smooth_optimize(global_best_seq,
                                           n,
                                           lbfgs_deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0, 200.0,
                                                          500.0, 1000.0, 2000.0, 5000.0,
                                                          10000.0, 20000.0, 50000.0,
                                                          100000.0, 200000.0, 500000.0],
                                           max_iter_per_beta=30_000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < global_best_val:
            global_best_seq = seq_lbfgs
            global_best_val = val_lbfgs
            print(f"[L‑BFGS] refined surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 10. Final greedy hill‑climb (use any leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Final hill‑climb ---")
        seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline, rng)
        if val_hc < global_best_val:
            global_best_seq = seq_hc
            global_best_val = val_hc
            print(f"[Hill‑climb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 11. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
