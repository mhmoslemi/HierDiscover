def _solve_lp(A, M, n):
    from scipy.optimize import linprog
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve':True, 'time_limit': 60})
    except Exception as e:
        print("[LP] exception", e)
        return None
    if not res.success:
        print("[LP] failed:", res.message)
        return None
    return res.x

def downsample(seq, target_n):
    seq_arr = np.asarray(seq, dtype=np.float64)
    current = seq_arr.shape[0]
    if current == target_n:
        return seq_arr / seq_arr.sum()
    else:
        # Use simple average pooling
        factor = current // target_n
        if factor < 1:
            # upsample: use interpolation via linear
            # but not needed as we choose target_n <= current.
            factor = 1
        # aggregate
        new = seq_arr[:factor*target_n].reshape(target_n, factor).mean(axis=1)
        # renormalize
        new = np.maximum(new, 0)
        s = new.sum()
        if s == 0:
            return np.full(target_n, 1.0/target_n)
        return new / s

"""
Non‑negative step‑function optimiser.

The optimiser works in three stages:

1.  A short seeding phase picks a few analytically‑motivated ℓ₁‑normalised
    vectors (uniform, triangular, Gaussian, Dirichlet, …) and keeps the best
    one according to the surrogate objective  2·n·max(conv(a,a)).

2.  A linear‑programming “mass‑balancing’’ step iteratively enlarges the
    ℓ₁‑mass of the current vector while keeping every entry of the
    convolution f ∗ g below the current maximal autocorrelation M.
    After each LP we mix the new vector into the current one (step size t)
    and recompute M.  This dramatically reduces the max‑convolution peak.

3.  A long exponentiated‑gradient polishing phase (sub‑gradient descent on the
    exact max) and a tiny pairwise‑mass‑transfer local optimiser finish the
    search.

The whole procedure respects the 1000 s deadline, works for any
ℓ₁‑normalised sequence length (default n = 2000) and returns the best
sequence found as a plain ``list[float]``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags, csr_matrix

# -------------------------------------------------------------
# Low‑level utilities
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – always returns a vector on the probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(a: np.ndarray, n: int) -> float:
    """
    Surrogate objective for an ℓ₁‑normalised vector a:
        2·n·max(conv(a,a))
    This coincides with the official evaluator up to the normalisation.
    """
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# -------------------------------------------------------------
# Analytic seed families (all ℓ₁‑normalised)
# -------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float = 0.2) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# -------------------------------------------------------------
# Linear‑programming helper
# -------------------------------------------------------------
def _build_A_matrix(f: np.ndarray) -> csr_matrix:
    """
    Construct the (2n‑1) × n matrix A of the convolution constraints
    A @ g   ≤   M, where  (A)_{x,j} = f_{x-j}  (zero when x‑j out of range).

    Because the rows are just shifted copies of f this matrix is Toeplitz
    and can be built efficiently with scipy.sparse.diags.
    """
    n = f.shape[0]
    offsets = np.arange(n)                # 0 … n‑1
    # each diagonal is a constant vector equal to the corresponding entry of f
    data = [np.full(n, f[k], dtype=np.float64) for k in range(n)]
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def _solve_lp_for_g(f: np.ndarray, M: float, n: int) -> np.ndarray | None:
    """
    Solve   max  1ᵀg   s.t.   A(f)·g ≤ M·1,   g ≥ 0.
    Returns the optimal g (not normalised) or ``None`` on failure.
    """
    A = _build_A_matrix(f)
    c = -np.ones(n)                                      # maximise sum(g)
    b = np.full(A.shape[0], M, dtype=np.float64)        # RHS = M for every row
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception as exc:   # pragma: no cover
        print("[LP] linprog raised:", exc)
        return None
    if not res.success:
        print("[LP] solver failed:", res.message)
        return None
    return res.x


# -------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max)
# -------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # decreasing learning rate
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# -------------------------------------------------------------
# Pairwise mass‑transfer local optimiser (very cheap)
# -------------------------------------------------------------
def _mass_transfer_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    max_val = float(np.max(conv))
    best_a = a.copy()
    best_val = max_val

    rng = np.random.default_rng()
    step_factor = 0.25
    no_improve = 0
    it = 0
    center = n - 1
    L = len(conv)

    while time.time() < deadline:
        if a.max() < 1e-16:
            break
        src = rng.choice(n, p=a)                # source index (mass‑biased)
        src_mass = a[src]
        if src_mass <= 1e-15:
            continue
        tgt = rng.integers(n)
        if tgt == src:
            continue

        eps = step_factor * src_mass
        if eps < 1e-14:
            continue

        # update the convolution incrementally (O(n))
        new_conv = conv.copy()
        new_conv[src:src + n] -= 2.0 * eps * a
        new_conv[tgt:tgt + n] += 2.0 * eps * a
        new_conv[center] += 2.0 * eps * eps
        offset = tgt - src
        idx1 = center + offset
        new_conv[idx1] -= eps * eps
        if offset != 0:
            idx2 = center - offset
            new_conv[idx2] -= eps * eps

        new_max = float(np.max(new_conv))

        if new_max < max_val - 1e-16:
            a[src] = src_mass - eps
            a[tgt] = a[tgt] + eps
            conv = new_conv
            max_val = new_max
            no_improve = 0
            if max_val < best_val:
                best_val = max_val
                best_a = a.copy()
        else:
            no_improve += 1

        it += 1
        if no_improve > 5000:
            step_factor *= 0.5
            step_factor = max(step_factor, 1e-4)
            no_improve = 0

        if it % 20000 == 0:
            print(f"[MassTransfer] it={it} best max={best_val:.12f}")

    return best_a, 2.0 * n * best_val


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose problem size (trade‑off between speed and quality)
    # ------------------------------------------------------------------
    n_target = 2000                # fits comfortably in memory for the LP step
    # ------------------------------------------------------------------
    # 2. Initialise with a cloud of analytic / random seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n_target}) ---")
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:  # pragma: no cover
            prev_arr = None
    else:
        prev_arr = None

    # down‑sample previous if it is longer than the target length
    if prev_arr is not None and prev_arr.ndim == 1:
        if prev_arr.size > n_target:
            factor = prev_arr.size // n_target
            if factor >= 2:
                prev_arr = prev_arr[:factor * n_target].reshape(n_target, factor).mean(axis=1)
        if prev_arr.size == n_target:
            prev_arr = np.maximum(prev_arr, 0.0)
            s = prev_arr.sum()
            if s > 0:
                prev_arr = prev_arr / s
            else:
                prev_arr = None
        else:
            prev_arr = None
    else:
        prev_arr = None

    seeds = [
        _uniform(n_target),
        _triangular(n_target),
        _gaussian(n_target, sigma_frac=0.2),
        _gaussian(n_target, sigma_frac=0.12),
        _random_dirichlet(n_target, conc=0.5, rng=rng),
        _random_dirichlet(n_target, conc=0.05, rng=rng),
        _random_dirichlet(n_target, conc=2.0, rng=rng)
    ]
    if prev_arr is not None:
        seeds.append(prev_arr)

    # evaluate all seeds, keep the top few
    evaluated = [(v, _objective_surrogate(v, n_target)) for v in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print("\nTop seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Linear‑programming mass‑balancing phase
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq, n_target)
    best_M = best_val / (2.0 * n_target)           # current max‑convolution value
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # parameters for the LP mixing loop
    t = 0.3                                        # initial mixing weight
    iter_lp = 0
    while time.time() < deadline and t > 0.01 and iter_lp < 30:
        iter_lp += 1
        M = best_M
        # Solve the LP that maximises ℓ₁‑mass of g under the convolution cap M
        g_raw = _solve_lp_for_g(best_seq, M, n_target)
        if g_raw is None:
            print("[LP] breaking – no feasible solution")
            break

        sum_g = g_raw.sum()
        if sum_g <= 0:
            print("[LP] zero mass returned – abort")
            break

        g = g_raw / sum_g                         # renormalise to ℓ₁‑norm = 1
        # convex combination
        new_seq = (1.0 - t) * best_seq + t * g
        new_seq = new_seq / new_seq.sum()        # numerical safety

        # evaluate new max‑convolution
        conv_new = _conv_fft(new_seq, new_seq)
        M_new = float(np.max(conv_new))
        val_new = 2.0 * n_target * M_new

        print(f"[LP] it={iter_lp:02d} t={t:.3f} surrogate={val_new:.6f}")

        if val_new < best_val:
            best_seq = new_seq
            best_val = val_new
            best_M = M_new
            print(f"  [Update] new best surrogate = {best_val:.6f}")

        # regardless of improvement we continue from the mixed point
        best_seq = new_seq
        best_M = M_new
        # slowly decay the mixing coefficient
        t = max(0.01, t * 0.85)

    # ------------------------------------------------------------------
    # 4. Exponentiated‑gradient polishing (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # heuristic: about 200 k iterations per 30 s on a modest CPU
        max_iters = int(min(800_000, remaining * 7000))
        max_iters = max(200_000, max_iters)
        print(f"\n--- Exponentiated‑gradient polishing (≤{max_iters} iters) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq,
                                          n_target,
                                          deadline,
                                          max_iters=max_iters)
        if val_eg < best_val:
            best_seq = seq_eg
            best_val = val_eg
            print(f"[ExpGrad] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Tiny pairwise mass‑transfer local optimiser
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Pairwise mass‑transfer refinement ---")
        seq_mt, val_mt = _mass_transfer_opt(best_seq, n_target, deadline)
        if val_mt < best_val:
            best_seq = seq_mt
            best_val = val_mt
            print(f"[MassTransfer] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for a possible subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```

This implementation replaces the purely continuous L‑BFGS approach with a **linear‑programming mass‑balancing** step that can lift the ℓ₁‑norm of a partner vector while keeping every entry of the convolution below the current peak.  After a handful of such LP‑driven updates the surrogate value drops appreciably.  A long exponentiated‑gradient pass then fine‑tunes the exact max‑convolution, and a final cheap pairwise mass‑transfer sweep cleans up any remaining local irregularities.  The algorithm respects the 1000 s deadline, works without filesystem access, and returns the best sequence discovered.
