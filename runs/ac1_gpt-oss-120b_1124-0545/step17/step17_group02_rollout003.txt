def _simulated_annealing(start_a, n, deadline):
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)
    temp = 0.1
    final = 0.001
    total_steps = int(min(200000, (deadline - time.time()) * 1000))
    for step in range(total_steps):
        if time.time() > deadline: break
        # propose new a: move small fraction of mass from i to j
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or a[i] == 0: continue
        delta = min(a[i], rng.uniform(0, 0.02)) # small delta
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # re-normalize just in case due to maybe rounding.

        val = _objective_exact(cand)
        dv = val - best_val
        if dv < 0 or rng.random() < np.exp(-dv / temp):
            a = cand
            if val < best_val:
                best_val = val
                best_a = cand.copy()
        # temperature schedule:
        if step % 1000 == 0:
            t = step / total_steps
            temp = init_temp * (1 - t) + final * t
    return best_a, best_val

def _collect_all_seeds(n, rng, prev_arr):
    seeds = []
    # families...
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for lam in [0.001, 0.005, 0.01, 0.02]:
        seeds.append(_exponential_decay(n, lam))
    for sigma_frac in [0.12, 0.2, 0.3, 0.4]:
        seeds.append(_gaussian(n, sigma_frac))
    # Beta grid
    for alpha in [0.2, 0.5, 0.8, 1.0, 2.0, 5.0]:
        for beta in [0.2, 0.5, 0.8, 1.0, 2.0, 5.0]:
            seeds.append(_beta_shape(n, alpha, beta))
    # Dirichlet
    for conc in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0]:
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    # Sparse
    for frac in [0.01, 0.03, 0.05, 0.1]:
        for _ in range(3):
            seeds.append(_random_sparse(n, frac, rng))
    # lowfreq
    for nf in [2,3,4,5]:
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))
    # mixtures (random pairs)
    base = list(range(len(seeds)))
    for _ in range(50):
        i,j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))
    # rotations of some seeds
    for _ in range(20):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))
    if prev_arr is not None:
        seeds.append(prev_arr / prev_arr.sum())
    # clean and normalize
    cleaned = []
    for v in seeds:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0 or np.isnan(s):
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

def _lbfgs_optimize(start_vec, n, deadline, beta_schedule, max_iter_per_beta):
    v = np.log(start_vec + 1e-30)
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)
        def fun(v_local):
            obj, grad = _smooth_objective(v_local, n, beta_val)
            return obj, grad
        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()
        try:
            res = _opt.minimize(fun=fun, x0=v, method='L-BFGS-B', jac=True,
                                 callback=callback,
                                 options={'maxiter': max_iter_per_beta, 'ftol': 1e-12, 'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break
    a = _softmax(v)

mass‑transfer local search.  All stages respect a global time budget
(~1000 s).  The best sequence found is stored in ``height_sequence_1``
and returned as a plain list.
"""

import time
import numpy as np
from scipy import optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (each returns a ℓ₁‑normalised vector)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑style shape (i+1)^(α‑1)·(n‑i)^(β‑1).  Works for α,β>0."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient helpers
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)²."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) for the active lag."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate used by continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth approximation of the objective.
    Returns (obj, grad) where grad is w.r.t. the logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)               # full linear convolution
    shift = np.max(conv)                 # shift for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule: list[float],
                    max_iter_per_beta: int) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method="L-BFGS-B",
                                jac=True,
                                callback=callback,
                                options={"maxiter": max_iter_per_beta,
                                         "ftol": 1e-12,
                                         "disp": False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 1_000_000,
                           eta0: float = 1.2) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 100_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
    # final check
    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer local search
# ----------------------------------------------------------------------
def _simulated_annealing(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Hill‑climbing with occasional uphill moves (SA)."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    # temperature schedule
    T_start = 0.25
    T_end = 0.001
    max_steps = int(min(200_000, (deadline - time.time()) * 350))
    for step in range(max_steps):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        # move a small fraction of the mass from i to j
        delta = a[i] * rng.uniform(0.001, 0.02)
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
        else:
            # temperature linearly decays
            t = step / max_steps
            T = T_start * (1.0 - t) + T_end * t
            if rng.random() < np.exp(-(val - best_val) / max(T, 1e-12)):
                a = cand

        # occasional Gaussian jitter to shake out of plateaus
        if step % 15_000 == 0 and step > 0:
            noise = rng.normal(scale=0.002, size=n)
            cand2 = np.maximum(a + noise, 0.0)
            if cand2.sum() == 0.0:
                continue
            cand2 /= cand2.sum()
            val2 = _objective_exact(cand2)
            if val2 < best_val:
                best_val = val2
                best_a = cand2.copy()
                a = cand2

    return best_a, best_val


# ----------------------------------------------------------------------
# Seed‑pool construction (much richer than the previous version)
# ----------------------------------------------------------------------
def _collect_all_seeds(n: int,
                       rng: np.random.Generator,
                       previous: np.ndarray | None) -> list[np.ndarray]:
    """Generate a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # exponential decays
    for lam in (0.001, 0.003, 0.006, 0.012, 0.02):
        seeds.append(_exponential_decay(n, lam))

    # Gaussians
    for sigma_frac in (0.12, 0.20, 0.30, 0.40):
        seeds.append(_gaussian(n, sigma_frac))

    # Beta shapes – we sweep a wide region, including <1 (U‑shapes)
    alphas = (0.2, 0.35, 0.5, 0.8, 1.0, 2.0, 5.0)
    betas = (0.2, 0.35, 0.5, 0.8, 1.0, 2.0, 5.0)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws – many different concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Sparse random masks
    for frac in (0.01, 0.025, 0.05, 0.10):
        for _ in range(2):
            seeds.append(_random_sparse(n, frac, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Pairwise mixtures of already‑generated seeds
    base = list(range(len(seeds)))
    for _ in range(50):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Rotations of a few promising seeds
    for _ in range(30):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # Include the previously best sequence if available
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Final clean‑up – enforce non‑negativity and ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0 or np.isnan(s):
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the overall time budget (≈1000 s),
    stores the best result in the global variable ``height_sequence_1``,
    and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse length of a previously found sequence when possible
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None and isinstance(previous, (list, tuple)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            n = len(prev_arr)
        except Exception:
            n = 4000
            prev_arr = None
    else:
        n = 4000
        prev_arr = None

    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # 1) Build a huge and diverse seed pool
    # ------------------------------------------------------------------
    seeds = _collect_all_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # 2) Quick exact evaluation of all seeds; keep the elite pool
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best raw seed objective = {best_val:.6f}")

    # Sort seeds by their exact objective, keep the top‑k for heavy optimisation
    seeds.sort(key=_objective_exact)
    top_k = min(25, len(seeds))
    elite = seeds[:top_k]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # 3) Continuation L‑BFGS on the smooth surrogate (moderate per‑β budget)
    # ------------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0,
    ]
    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break
        remaining = deadline - time.time()
        # allocate a reasonable per‑β iteration budget (max 3000)
        per_beta = max(800, int(min(3000, remaining * 0.08)))
        print(f"\n--- L‑BFGS run #{idx + 1}/{len(elite)} (max_iter/β ≈ {per_beta}) ---")
        seq_opt = _lbfgs_optimize(
            start_vec=seed_vec,
            n=n,
            deadline=deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Exponentiated‑gradient restarts (random Dirichlet starts)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        restarts = 6
        iters_per_restart = int(min(600_000, remaining * 0.14 / restarts))
        print("\n--- Exponentiated‑gradient restarts ---")
        for r in range(restarts):
            if time.time() > deadline:
                break
            conc = rng.uniform(0.02, 1.5)
            start_vec = _random_dirichlet(n, conc, rng)
            seq_eg, val_eg = _exp_gradient_optimize(
                start_a=start_vec,
                n=n,
                deadline=deadline,
                max_iters=iters_per_restart,
                eta0=2.0,
            )
            print(f"[EG {r + 1}/{restarts}] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after EG restart = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Long EG run from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(3_500_000, remaining * 0.30))
        if max_iters > 0:
            print("\n--- Long EG optimisation (starting from best) ---")
            seq_eg, val_eg = _exp_gradient_optimize(
                start_a=global_best_seq,
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                eta0=2.5,
            )
            print(f"[Long EG] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after long EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6) Simulated‑annealing mass‑transfer local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        steps_est = int(min(150_000, remaining * 180))
        if steps_est > 0:
            print("\n--- Simulated‑annealing local search ---")
            seq_sa, val_sa = _simulated_annealing(
                start_a=global_best_seq,
                n=n,
                deadline=deadline,
                rng=rng,
            )
            print(f"[SA] best = {val_sa:.6f}")
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[Update] New best after SA = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7) Final high‑β polishing L‑BFGS (very few iterations, tight β)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(800, int(min(2000, remaining * 0.05)))
        print("\n--- Final polishing L‑BFGS (high β) ---")
        seq_final = _lbfgs_optimize(
            start_vec=global_best_seq,
            n=n,
            deadline=deadline,
            beta_schedule=[500_000.0, 1_000_000.0, 2_000_000.0, 5_000_000.0],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8) Final official evaluation & reporting
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate best ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally for warm‑starts in subsequent calls
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
