    k = np.argmax(conv)
    n = len(a)

def smooth_obj_and_grad(v, n, beta):
    a = softmax(v)
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta*(conv - shift))
    sum_exp = exp_term.sum()
    loss = shift + np.log(sum_exp)/beta
    w = exp_term / sum_exp
    grad_a = 2.0 * _conv_fft(a[::-1], w)[n-1:n-1+n]  # shape (n,)
    # chain rule:
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return loss, grad_v
```

"""
Improved step‑function search.

The algorithm proceeds in four phases:

1. **Quick analytic/sweep search** – scan several one‑parameter families
   (Beta, Gaussian, exponential, power‑law, sin‑power) on a coarse grid
   and keep the best ℓ₁‑normalised candidate.

2. **Adam optimisation of a smooth max‑convolution surrogate** – the true
   objective 2·n·max(conv) is approximated by a log‑sum‑exp (smooth max)
   and optimised in the log‑softmax (simplex) space with the Adam
   optimiser.  The smoothing parameter β is increased gradually (continuation)
   so that the solution approaches the exact non‑smooth optimum.

3. **Long projected sub‑gradient descent** – after the Adam phase the exact
   objective is further polished by a sub‑gradient method on the true
   max‑convolution (no smoothing).  A diminishing step‑size and Euclidean
   projection onto the simplex are used.

4. **Final hill‑climbing on logits** – cheap pairwise logit swaps are performed
   to escape tiny local irregularities.

All phases respect the global time budget (default 1000 s).  The best sequence
found is stored in the global variable ``height_sequence_1`` for reuse in
subsequent runs and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact surrogate (sum of a is 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv) for a sum‑1 sequence."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _double_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta(α,β) on {0,…,n‑1} with a half‑offset."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 0.5) ** (alpha - 1.0)) * ((n - i - 0.5) ** (beta - 1.0))
    return a / a.sum()


def _sin_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n) ** p
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Quick family sweep

def _sweep_families(n: int, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Return the best ℓ₁‑normalised candidate from a cheap parameter sweep."""
    best_seq = _uniform(n)
    best_val = _objective_surrogate(best_seq)

    # – sin‑power ---------------------------------------------------------
    for p in np.linspace(0.5, 20.0, 40):
        seq = _sin_power(n, p)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    # – Beta (asymmetric) ------------------------------------------------
    alphas = np.linspace(0.2, 3.0, 30)
    betas = np.linspace(0.2, 3.0, 30)
    for a in alphas:
        for b in betas:
            seq = _beta_distribution(n, a, b)
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val, best_seq = val, seq

    # – exponential -------------------------------------------------------
    for rate in np.linspace(5e-4, 2e-2, 30):
        seq = _exponential(n, rate)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    # – double‑exponential ------------------------------------------------
    for rate in np.linspace(1e-3, 2e-2, 20):
        seq = _double_exponential(n, rate)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    # – Gaussian -----------------------------------------------------------
    for sigma_frac in np.linspace(0.05, 0.4, 15):
        seq = _gaussian(n, sigma_frac)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    # – power law ---------------------------------------------------------
    for p in np.linspace(0.0, 3.0, 25):
        seq = _power(n, p)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    # – a few random Dirichlet draws (just in case) -----------------------
    for _ in range(30):
        conc = rng.uniform(0.01, 2.0)
        seq = _random_dirichlet(n, conc, rng)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on smooth max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    Returns the best sequence (ℓ₁‑normalised) found and its exact surrogate value.
    """
    # Adam hyper‑parameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # logits (softmax will enforce simplex)
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    # continuation schedule for β (smoothness)
    beta_schedule = [1.0, 3.0, 7.0, 15.0, 30.0, 60.0,
                     120.0, 250.0, 500.0, 1000.0, 2500.0,
                     5000.0, 10000.0, 25000.0, 50000.0]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    for beta in beta_schedule:
        # allocate a chunk of iterations – bounded by remaining time
        while time.time() < deadline:
            # rough upper bound of 20 000 steps per β if time permits
            if it > 200_000:
                break
            it += 1
            a = _softmax(v)
            conv = _conv_fft(a, a)                     # length 2n‑1
            shift = float(np.max(conv))
            # smooth‑max weights
            exp_term = np.exp(beta * (conv - shift))
            sum_exp = float(np.sum(exp_term))
            w = exp_term / sum_exp                     # shape (2n‑1,)

            # gradient of smooth max w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a_smooth = 2.0 * full[n - 1:n - 1 + n]   # shape (n,)
            # scale to the true objective 2·n·max(conv)
            grad_a = (2.0 * n) * grad_a_smooth

            # chain rule through softmax
            dot = float(np.dot(grad_a, a))
            grad_v = a * (grad_a - dot)

            # Adam update (in logit space)
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat_corr = v_hat / (1.0 - beta2 ** it)
            v -= lr * m_hat / (np.sqrt(v_hat_corr) + eps)

            # track exact surrogate (no smoothing) for final comparison
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        # move to next β
        if time.time() >= deadline:
            break
    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Long projected sub‑gradient descent on the exact objective
    2·n·max(conv). Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_max - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step‑size
        lr = 0.25 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Cheap pairwise hill‑climbing on logits
# ----------------------------------------------------------------------
def _hill_climb_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing using pairwise logit swaps.
    Returns the best sequence encountered.
    """
    rng = np.random.default_rng()
    logit = np.log(start_seq + 1e-30)

    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        sigma = 0.015 / math.sqrt(it)
        delta = rng.normal(scale=sigma)
        new_log = logit.copy()
        new_log[i] += delta
        new_log[j] -= delta
        new_seq = _softmax(new_log)
        new_val = _objective_surrogate(new_seq)
        if new_val < cur_val:
            logit, cur_seq, cur_val = new_log, new_seq, new_val
            if new_val < best_val:
                best_seq, best_val = new_seq.copy(), new_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Random perturbation (tiny logit jitter)
# ----------------------------------------------------------------------
def _random_perturb(seq: np.ndarray,
                    n: int,
                    rng: np.random.Generator,
                    magnitude: float = 0.02) -> np.ndarray:
    logit = np.log(seq + 1e-30)
    pert = magnitude * rng.normal(size=n)
    new_log = logit + pert
    return _softmax(new_log)
