"""
Minimal‐max‑convolution search.

We search for a non‑negative step function (a probability vector)
that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

(the same quantity that ``evaluate_sequence`` returns after the internal
normalisation).  The programme combines a cheap parametric grid (uniform,
triangular, Gaussian + a flexible U‑shaped Beta family), a short
L‑BFGS continuation on a smooth log‑sum‑exp surrogate, and a long
exact sub‑gradient descent on the true max‑convolution.  The whole
pipeline adapts dynamically to the remaining time budget (≈1000 s) and
stores the best vector in the global ``height_sequence_1`` for reuse.

The added ingredients – an explicit Beta(α,α) family with
α∈(½,1) and a simple mixture with the uniform distribution – give the
optimiser new directions that were not explored by the previous
implementation and usually improve the bound (target ≈ 1.503) within the
time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import for L‑BFGS‑B (fallback to pure sub‑gradient)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:          # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Fast convolution (linear) via real FFT
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Building blocks
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum=1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_u_shape(n: int, alpha: float, eps: float = 1e-12) -> np.ndarray:
    """
    Symmetric U‑shaped density ∝ [(i+ε)(n‑1‑i+ε)]^{α‑1},
    i.e. Beta(α,α) on {0,…,n‑1} (α∈(0.5,1) yields a heavier edge weight).
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + eps) * (n - 1 - i + eps)
    a = prod ** (alpha - 1.0)          # α‑1 is negative for α<1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mixture_beta_uniform(n: int, alpha: float, lam: float) -> np.ndarray:
    """Convex combination (1‑λ)·Uniform + λ·BetaU(α). Normalised to sum‑1."""
    u = _uniform(n)
    b = _beta_u_shape(n, alpha)
    a = (1.0 - lam) * u + lam * b
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule=None,
                    max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Refine ``start_vec`` (ℓ₁‑normalised) using L‑BFGS‑B on the smooth
    surrogate 2·n·log‑sum‑exp_{β}(conv).
    """
    if _opt is None:                              # pragma: no cover
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0,
        ]

    # Initialise logits so that softmax(start_vec)=start_vec
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                 # stays on the simplex
            conv = _conv_fft(a, a)                 # full linear conv
            max_c = np.max(conv)
            shifted = conv - max_c                 # ≤0, safe for exp
            exp_beta = np.exp(beta_val * shifted)
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (max_c + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (see derivation in original script)
            w = exp_beta / sum_exp                # soft‑weights, sum=1
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:                     # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on 2·n·max(conv) for a probability vector.
    Projection onto the simplex keeps the iterates feasible.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        # Record improvement
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # Build sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Simple diminishing step size
        lr = 0.25 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin for the final ``evaluate_sequence`` call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – 5000 offers a slightly finer discretisation
    # ------------------------------------------------------------------
    n = 5000
    print(f"\n=== start optimisation, n={n} (budget≈{budget_s}s) ===")

    # ------------------------------------------------------------------
    # 1. Build a diverse seed pool (analytic families + random)
    # ------------------------------------------------------------------
    seeds = []

    # Classic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sin2(n))
    seeds.append(_hann_window(n))

    # Gaussian family (different bandwidths)
    for sigma in (0.07, 0.12, 0.20, 0.30):
        seeds.append(_gaussian(n, sigma))

    # Beta‑U family (U‑shaped) – sweep α∈(0.55,0.95)
    for alpha in np.linspace(0.55, 0.95,  nine := 9):
        seeds.append(_beta_u_shape(n, alpha))

    # Mixture of Uniform + BetaU – sweep λ
    for alpha in (0.60, 0.70, 0.80):
        for lam in (0.1, 0.2, 0.3, 0.4):
            seeds.append(_mixture_beta_uniform(n, alpha, lam))

    # Random Dirichlet draws (wide range of concentration)
    conc_params = np.logspace(-2, 0.5, 10)          # 0.01 … ~3.16
    for conc in conc_params:
        for _ in range(3):
            seeds.append(rng.dirichlet(np.full(n, conc)))

    # Incorporate the previous best if present
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                seeds.append(arr / arr.sum())
        except Exception:
            pass

    # Normalise all seeds onto the simplex (protect against 0‑sum)
    clean_seeds = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        clean_seeds.append(v / s if s > 0 else _uniform(n))
    seeds = clean_seeds

    # ------------------------------------------------------------------
    # 2. Quick surrogate ranking – keep only the most promising seeds
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nBest surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d} : {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Global best tracking
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] current best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. L‑BFGS refinement (smooth surrogate)
    # ------------------------------------------------------------------
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        20_000.0, 50_000.0, 100_000.0,
        200_000.0, 500_000.0, 1_000_000.0,
        2_000_000.0, 5_000_000.0,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS passes.")
            break

        # allocate a modest slice of remaining budget for each seed
        remaining = deadline - time.time()
        max_iter_beta = int(min(6000, remaining * 0.04 * 1e5))
        max_iter_beta = max(2500, max_iter_beta)

        print(f"\n--- L‑BFGS on seed #{idx+1} (max_iter/β={max_iter_beta}) ---")
        seq_opt = _lbfgs_optimize(seed_vec,
                                  n,
                                  deadline,
                                  beta_schedule=beta_schedule,
                                  max_iter_per_beta=max_iter_beta)

        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] new global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Prolonged sub‑gradient descent (exact max‑conv surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # we will keep a handful of restarts – the current best and

        restarts = [best_seq] + top_seeds[:4]

            if time.time() > deadline:

            max_sub = int(min(1_200_000, remaining * 0.045 * 1e5))

                                                n,
