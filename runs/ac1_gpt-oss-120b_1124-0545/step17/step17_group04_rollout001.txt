"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

The score is scale‑invariant, so we keep the vector on the probability
simplex (sum = 1).  The algorithm works in three phases:

1.   Build a diverse pool of analytic seeds (uniform, power‑law,
     exponential, Gaussian, Hann, arcsine, …) and keep the best one
     according to the fast surrogate 2·n·max(conv).

2.   Refine that seed with a **projected Adam** optimiser on the exact
     max‑convolution (sub‑gradient of the active convolution entry).
     The iterate is always projected onto the simplex.

3.   Perform a **simulated‑annealing permutation search**: the multiset
     of values is kept fixed while random swaps are accepted according
     to a temperature schedule.  This reduces the correlation between
     the vector and its reversal – the dominant term of the objective.

Both optimisation phases respect the time budget (≈ 1000 s).  The final
vector is stored in the global variable `height_sequence_1` and returned
as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i = r**i, 0<r<1."""
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Objective helper (fast surrogate)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) – sum(seq) is 1, so this equals the true
    objective up to the scaling‑invariant factor."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Projected Adam optimisation of the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_subgrad_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected sub‑gradient descent on max(conv).
    Returns the best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    it = 0
    # base learning‑rate (tuned empirically)
    lr0 = 0.3

    while True:
        if time.time() > deadline:
            break
        it += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = float(2.0 * n * conv[max_idx])

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the *active* convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]          # double counting is intentional

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr0 / (math.sqrt(it) + 1.0)      # slowly decaying stepsize
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep on simplex
        a = _proj_simplex(a)

        if it % 200_000 == 0:
            print(f"[Adam] iter={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing permutation search (values stay fixed)
# ----------------------------------------------------------------------
def _permute_sa(start_seq: np.ndarray,
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Randomly swap two positions; accept according to a simple cooling schedule.
    The multiset of values never changes – only the ordering.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    conv_cur = _conv_fft(cur_seq, cur_seq)
    cur_val = float(2.0 * n * np.max(conv_cur))

    best_seq = cur_seq.copy()
    best_val = cur_val

    # estimate a safe number of iterations from remaining time
    remaining = max(0.0, deadline - time.time())
    # each iteration costs ≈2 × FFT → ~0.0003 s on a 4000‑vector
    max_iters = int(min(500_000, remaining / 0.00035))
    if max_iters < 1:
        return best_seq, best_val

    T0 = 0.6   # initial temperature
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # propose a swap
        prop_seq = cur_seq.copy()
        prop_seq[i], prop_seq[j] = prop_seq[j], prop_seq[i]

        conv_prop = _conv_fft(prop_seq, prop_seq)
        prop_val = float(2.0 * n * np.max(conv_prop))

        delta = prop_val - cur_val
        # temperature linearly cools to near zero
        progress = it / max_iters
        T = T0 * (1.0 - progress) + 1e-9

        if delta < 0 or math.exp(-delta / T) > rng.random():
            cur_seq = prop_seq
            conv_cur = conv_prop
            cur_val = prop_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] iter={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # reserve a tiny safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – 4000 offers a good granularity / speed trade‑off
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Seed pool – diverse analytic families
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _triangular(n),
        _sin2(n),
        _hann_window(n),
        _arcsine(n),
    ]

    for sigma in (0.08, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))

    for p in (0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        seeds.append(_power(n, p))

    for r in (0.98, 0.95, 0.90, 0.85, 0.80):
        seeds.append(_exponential(n, r))

    # Previous best, if any, is also a useful start point
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            arr = np.maximum(arr, 0.0)
            s = arr.sum()
            if s > 0.0:
                seeds.append(arr / s)
        except Exception:
            pass

    # Normalise everything onto the simplex (sum = 1)
    clean_seeds = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0.0:
            clean_seeds.append(v / s)
        else:
            clean_seeds.append(_uniform(n))
    seeds = clean_seeds

    # ------------------------------------------------------------------
    # 2. Quick surrogate ranking – keep the most promising seed
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    for vec in seeds:
        val = _objective_surrogate(vec)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate among raw seeds = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected Adam sub‑gradient descent (exact surrogate)
    # ------------------------------------------------------------------
    subgrad_deadline = start_time + (budget_s - 2) * 0.65
    subgrad_deadline = min(subgrad_deadline, deadline)

    print("[Phase 1] Adam sub‑gradient optimisation")
    best_seq, best_val = _adam_subgrad_opt(best_seq, n, subgrad_deadline)
    print(f"[Phase 1] best surrogate after Adam = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Permutation simulated‑annealing (order optimisation)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Phase 2] Permutation simulated‑annealing")
        best_seq, best_val = _permute_sa(best_seq, n, deadline)
        print(f"[Phase 2] best surrogate after SA = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                       # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
