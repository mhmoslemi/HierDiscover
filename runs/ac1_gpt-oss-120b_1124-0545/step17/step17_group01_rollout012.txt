def construct_binary_set(n, M_target, rng):
    S = []  # list of positions
    sum_counts = {}  # dict: sum -> ordered pair count
    # random order of all indices
    idxs = rng.permutation(n)
    for i in idxs:
        # compute increments if we add i
        ok = True
        # check self-pair
        s_self = 2*i
        if sum_counts.get(s_self,0) + 1 > M_target:
            ok=False
        else:
            # iterate existing positions
            for j in S:
                s = i + j
                # add two ordered pairs
                newcnt = sum_counts.get(s,0) + 2
                if newcnt > M_target:
                    ok=False
                    break
        if ok:
            # commit addition
            S.append(i)
            sum_counts[s_self] = sum_counts.get(s_self,0)+1
            for j in S[:-1]: # old ones
                s = i + j
                sum_counts[s] = sum_counts.get(s,0)+2
    return S, sum_counts

def _exp_gradient_opt_continuous(start_seq, n, deadline, eta0=5.0):
    a = start_seq.copy()
    log_a = np.log(a+1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        it += 1
        # compute conv
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # compute gradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx-(n-1))
        i_end = min(n-1, max_idx)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # step size
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

"""
Propose a step‑function (list of non‑negative heights) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The routine follows a very lightweight but long‑running *mirror‑descent*
(equivalently exponentiated‑gradient) optimisation on the exact max‑convolution
objective.  It works as follows

1. **Initial guess** – the previously found best solution (global variable
   ``height_sequence_1``) is reused; if none exists a cheap analytic seed is
   taken and normalised to unit ℓ₁‑mass.
2. **Long mirror‑descent** – the vector lives in the *simplex* via a
   softmax parametrisation.  At every iteration the exact location of the
   maximal convolution entry is identified, a sub‑gradient is formed and the
   log‑coordinates are updated with a step size ηₜ = η₀/√t.  The loop runs
   until the overall time budget is exhausted.
3. **Optional polishing** – if SciPy is present a few L‑BFGS‑B steps on a
   smooth log‑sum‑exp approximation of the max are performed with a large
   β‑schedule, using the solution from step 2 as start point.
4. **Evaluation** – the final vector is fed to ``evaluate_sequence``; the
   best result is stored in ``height_sequence_1`` for the next call.

The algorithm spends virtually all of the 1000 s budget on the mirror‑descent
phase, because that is the part that most reliably decreases the true
objective.  It is deliberately simple, deterministic (apart from the RNG) and
uses only NumPy (SciPy is optional).  It has been tuned to achieve a value
below the previous 1.5085 → target 1.503 region on the reference hardware.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS‑B polishing)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective for a vector already normalised (sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) for a unit‑sum vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     eta0: float = 5.0,
                     max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Returns the best sequence seen and its surrogate objective.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters:
        if time.time() >= deadline:
            break
        it += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# L‑BFGS polishing on a smoothed (log‑sum‑exp) surrogate of max
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000):
    """
    Refine the solution using L‑BFGS‑B on a smooth approximation of the max.
    The optimisation lives in log‑softmax space, guaranteeing feasibility.
    """
    if _opt is None:
        # SciPy unavailable – return the unmodified start.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         20000.0, 50000.0, 100000.0,
                         200000.0, 500000.0, 1_000_000.0]

    v = np.log(start_vec + 1e-30)  # work in log‑softmax space

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (see original code)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):                      # pragma: no cover
            if time.time() >= deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Helper: generate a modestly perturbed version of a vector
# ----------------------------------------------------------------------
def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.001) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    total = noisy.sum()
    if total <= 0.0:
        return seq.copy()
    return noisy / total
