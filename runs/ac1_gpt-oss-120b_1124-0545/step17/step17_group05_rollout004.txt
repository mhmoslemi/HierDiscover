"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.  The search strategy is:

1.  Build a diverse pool of analytically‑parameterised candidates
    (power law, exponential decay, Beta‑shape, Dirichlet draws, …) and
    keep the best one according to the cheap surrogate
    ``2·n·max(conv)`` (scale‑invariant).

2.  Refine that best vector with a continuation L‑BFGS optimiser that
    smooths the max‑convolution via a log‑sum‑exp surrogate.  The
    optimisation works in log‑probability space and enforces the ℓ₁‑simplex
    with a soft‑max.

3.  Polish the result with many iterations of exponentiated‑gradient
    descent on the exact max‑convolution (sub‑gradient on the active
    lag).  The step size follows a classic 1/√t schedule.

4.  Perform a large random‑swap hill‑climber (reordering the values)
    and a lightweight mass‑transfer local search (move a small amount of
    mass between two positions) – both keep any improvement.

All phases respect a global 1000 s budget and use only NumPy / SciPy.
The final sequence is stored as ``height_sequence_1`` and returned as a
list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ → the probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Scale‑invariant surrogate: 2·n·max(conv(seq,seq)) / (∑seq)²."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv) / (np.sum(seq) ** 2))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors of length n)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p, decreasing."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), decreasing."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Heavy‑tailed Dirichlet values placed alternately high/low."""
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    L‑BFGS‑B optimisation of the smoothed max‑convolution objective.
    Operates in log‑probability space; the returned vector lies on the
    ℓ₁‑simplex.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6, 5e6
        ]

    # initialise log‑softmax variables
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)               # ℓ₁‑normalised probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (see derivation in the original code)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)           # classic 1/√t schedule
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 100_000 == 0:
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Swap‑based permutation hill‑climber
# ----------------------------------------------------------------------
def _perm_swap_opt(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_swaps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Randomly swap two positions; keep the swap only if it lowers the exact
    max‑convolution.  Returns the best sequence and its surrogate value.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_val = _objective_surrogate(cur_seq)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if it % 50_000 == 0:
            print(f"[PermSwap] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight mass‑transfer local search
# ----------------------------------------------------------------------
def _mass_transfer_opt(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Randomly move a small fraction of mass from one index to another,
    accepting the move only if it improves the surrogate.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass from i to j
        delta = cur_seq[i] * (rng.random() * 0.05)
        if delta <= 0.0:
            continue
        cur_seq[i] -= delta
        cur_seq[j] += delta

        new_val = _objective_surrogate(cur_seq)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert
            cur_seq[i] += delta
            cur_seq[j] -= delta

        if it % 50_000 == 0:
            print(f"[MassTransfer] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the allocated time.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and timing
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size (4000 is a good trade‑off between quality and speed)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Coarse parameter grid search
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    # ---- Power‑law family ------------------------------------------------
    print("\n--- Grid search: power law ---")
    for p in np.linspace(0.1, 4.0, 81):
        seq = _power(n, float(p))
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq
    # refine around the best p
    if best_seq is not None:
        p_center = float(p)
        for p in np.linspace(max(0.01, p_center - 0.2), p_center + 0.2, 81):
            seq = _power(n, float(p))
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val = val
                best_seq = seq

    # ---- Exponential decay family ----------------------------------------
    print("\n--- Grid search: exponential decay ---")
    for lam in np.linspace(5e-4, 2e-2, 60):
        seq = _exponential_decay(n, float(lam))
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq
    # local refinement for λ
    if best_seq is not None:
        lam_center = float(lam)
        for lam in np.linspace(max(1e-5, lam_center - 0.003),
                               lam_center + 0.003, 81):
            seq = _exponential_decay(n, float(lam))
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val = val
                best_seq = seq

    # ---- Beta‑shape family ------------------------------------------------
    print("\n--- Grid search: beta shape ---")
    alphas = [0.3, 0.5, 1.0, 2.0, 4.0]
    betas = [0.3, 0.5, 1.0, 2.0, 4.0]
    for a in alphas:
        for b in betas:
            seq = _beta_shape(n, float(a), float(b))
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val = val
                best_seq = seq

    # ---- Random Dirichlet draws -------------------------------------------
    print("\n--- Random Dirichlet seeds ---")
    for conc in (0.1, 0.2, 0.5, 1.0):
        for _ in range(8):
            seq = _random_dirichlet(n, float(conc), rng)
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val = val
                best_seq = seq

    # ---- Alternating Dirichlet (heavy‑tailed alternating pattern) ---------
    print("\n--- Alternating Dirichlet seed ---")
    seq = _alternating_dirichlet(n, rng)
    val = _objective_surrogate(seq)
    if val < best_val:
        best_val = val
        best_seq = seq

    print(f"\n[Grid] best surrogate after exploration = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS optimisation (smooth max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate a modest share of remaining time to L‑BFGS
        max_iter_per_beta = int(min(20000, remaining * 0.35))
        max_iter_per_beta = max(8000, max_iter_per_beta)

        beta_schedule = [

                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter_per_beta)
        val_opt = _objective_surrogate(seq_opt)

        if val_opt < best_val:

            print(f"[Update] new global best after L‑BFGS = {best_val:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if time.time() < deadline:

        max_iters = int(min(800_000, remaining * 0.40))

        seq_mt, val_mt = _mass_transfer_opt(best_seq, n, deadline,

    try:

    return best_seq.tolist()
