Project onto simplex: given vector v, we project onto set {x >= 0, sum x = 1}. The standard Euclidean projection algorithm is O(n log n). Implementation:

def _project_simplex(v):
    # ensure sum v = 1 after projection onto the simplex.
    # Sorted descending.

    u = sorted(v, reverse=True)
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # Find rho = max { j | u_j - cssv_j / j > 0 }
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Will handle negative entries.

def _projected_subgradient_descent(a0, n, deadline, max_iters, initial_step=1.0):
    a = a0.copy()
    step = initial_step
    best_val = _objective_exact(a)
    best_a = a.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # compute gradient of maxconv at current a
        # Identify active lag
        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        # compute subgradient of maxconv (g)
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev = idx - i_range
        g[i_range] = 2.0 * a[rev]
        # Move opposite direction
        a_new = a - step * g
        # Project onto simplex
        a_new = _project_simplex(a_new)
        # Compute new objective
        val = _objective_exact(a_new)
        # Simple step acceptance: if not improved, reduce step
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.5
        # Optionally adjust step size schedule based on iteration number
    return best_a, best_val

"""
Title:  Adaptive Adam‑Mirrored Gradient Search + Heavy Local Refinement

Summary
-------
The evaluator is scale‑invariant: 2·n·max(conv(a,a))/∑a² depends only on the *shape*
of the non‑negative step function a.  We therefore keep the ℓ₁‑norm fixed to 1
(by parametrising a via logits → softmax) and minimise the maximal auto‑convolution
value M = maxₖ(a∗a)[k].

Algorithm
---------
1. **Seed generation** – a diverse pool of ℓ₁‑normalised vectors:
   * deterministic families (uniform, triangular, beta‑shapes with α,β∈[0.2,2.5]),
   * Dirichlet draws with concentration both < 1 (sparse) and > 1,
   * random sparse masks, low‑frequency cosine sums,
   * random mixtures / rotations of the above,
   * the previous best vector (warm start).

2. **Elite selection** – evaluate all seeds with the exact objective
   (FFT‑based convolution, O(n log n)) and keep the best *k* (≈20) for heavy
   optimisation.

3. **Adam‑mirrored‑gradient optimisation** – each elite vector is used as the
   initial point for a mirror‑descent on the simplex with an **Adam**‑style
   adaptive step‑size on the logits.  The gradient for the exact objective
   is the sub‑gradient of the active convolution lag, lifted to the log‑space.
   The optimisation runs as long as time permits (a per‑seed deadline).

4. **Aggressive stochastic local search** – after the Adam phase we perform a
   hill‑climbing “mass‑transfer’’ routine that moves a small fraction of the
   probability mass from one index to another (up to 200 000 iterations).

5. **High‑β L‑BFGS polishing** – the current best point is finally refined with
   continuation L‑BFGS on a smooth log‑sum‑exp surrogate of the max‑convolution.
   Very large β values make the surrogate practically exact.

6. **Result** – the best sequence found is printed, stored globally as
   ``height_sequence_1`` and returned as a plain list.

The whole pipeline respects the 1000 s wall‑time budget, dynamically splitting
the remaining time among the stages.  The combination of a richer seed pool,
adaptive Adam updates and a longer local‑search phase yields a systematic

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ onto the ℓ₁‑simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback – uniform
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type density on {1,…,n}.  Allows α,β<1 (U‑shaped)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    """Pick `frac·n` indices and give them equal mass."""
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Few low‑frequency cosines, clipped to ≥0 and re‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)

def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² (sum may differ from 1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    Returns a vector g such that g·δ is a sub‑gradient for any δ.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))               # active lag
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth approximation of 2·n·max(conv)/ (∑)².
    Returns (obj, grad) – gradient w.r.t. logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)               # full convolution
    shift = np.max(conv)                 # numeric stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a (softmax‑back‑prop)
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 2000,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        b = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, b)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam‑mirrored‑gradient optimisation (direct on the exact objective)
# ----------------------------------------------------------------------
def _adam_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 3_000_000,
    eta0: float = 0.3,
    beta1: float = 0.9,
    beta2: float = 0.999,
    report_interval: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective with Adam‑style
    adaptive steps on the logits.  Returns the best vector seen and its
    exact objective value.
    """
    logits = np.log(start_a + 1e-30)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    best_a = None
    best_val = np.inf
    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1
        a = _softmax(logits)

        if t % report_interval == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[Adam] iter {t:,d} – best = {best_val:.6f}")

        # sub‑gradient of the max‑convolution term
        g = _subgradient_maxconv(a)

        # gradient w.r.t. logits (mirror‑descent direction)
        grad = a * (g - np.dot(g, a))

        # Adam updates on logits
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta0 / (np.sqrt(v_hat) + 1e-12)
        logits -= step * m_hat

    # final evaluation
    a_end = _softmax(logits)
    val_end = _objective_exact(a_end)
    if val_end < best_val:
        best_val = val_end
        best_a = a_end.copy()
        print(f"[Adam] final – best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Stochastic mass‑transfer local search
# ----------------------------------------------------------------------
def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 120_000,
    step_frac: float = 0.012,
) -> tuple[np.ndarray, float]:
    """Hill‑climbing: move a small fraction of mass from i to j."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(987654321)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Seed‑pool construction
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors (length n)."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_exponential_decay(n, 0.001))
    seeds.append(_gaussian(n, 0.25))
    # beta‑shapes – we sweep a richer grid (including U‑shaped)
    for a in (0.2, 0.4, 0.6, 1.0, 1.5, 2.0):
        for b in (0.2, 0.4, 0.6, 1.0, 1.5, 2.0):
            seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws (wide concentration range)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse masks
    for frac in (0.01, 0.03, 0.07, 0.15, 0.3):
        for _ in range(4):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 5, 7):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # pairwise mixtures of already generated seeds
    base = np.arange(len(seeds))
    for _ in range(200):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of a few random seeds
    for _ in range(30):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up – enforce non‑negativity and ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget, stores the best result in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # problem size – reuse previous length if possible
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        n = len(previous)
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None
    else:
        n = 4000
        prev_arr = None

    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Build a rich seed pool
    # ------------------------------------------------------------------
    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} raw seeds.")

    # ------------------------------------------------------------------
    # Quick exact evaluation of all seeds – keep the elite set
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # sort and keep top‑k
    seeds.sort(key=_objective_exact)
    elite_k = min(25, len(seeds))
    elite = seeds[:elite_k]

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # Stage 1 – Adam‑mirrored‑gradient on the exact objective
    # ------------------------------------------------------------------
    print("\n--- Stage 1: Adam optimisation on elite seeds ---")
    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing Adam runs.")
            break

        # allocate a per‑seed budget (≈ proportional share of remaining time)
        remaining = deadline - time.time()
        # keep a small safety reserve for later stages
        per_seed_budget = max(8.0, remaining * 0.45 / (elite_k - idx))
        seed_deadline = time.time() + per_seed_budget

        print(f"\n[Adam run #{idx + 1}/{elite_k}] allocated {per_seed_budget:.1f}s")
        seq_opt, val_opt = _adam_optimize(
            seed_vec,
            n,
            seed_deadline,
            max_iters=3_000_000,
            eta0=0.4,
        )
        print(f"[Adam] final objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Aggressive stochastic local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(250_000, remaining * 30))   # many cheap moves
        print("\n--- Stage 2: Stochastic mass‑transfer local search ---")
        seq_loc, val_loc = _local_search(
            global_best_seq,
            n,
            deadline,
            iters=iters,
            step_frac=0.009,
        )
        print(f"[LocalSearch] best = {val_loc:.6f}")
        if val_loc < global_best_val:
            global_best_seq = seq_loc.copy()
            global_best_val = val_loc
            print(f"[Update] New best after local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Final high‑β L‑BFGS polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(800, int(min(3_000, remaining * 0.07 * 1e5)))
        print("\n--- Stage 3: High‑β L‑BFGS polishing ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[500_000.0, 1_000_000.0, 2_000_000.0, 5_000_000.0],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final validation with the official evaluator (for reporting)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate best ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for warm‑starts in subsequent calls
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
