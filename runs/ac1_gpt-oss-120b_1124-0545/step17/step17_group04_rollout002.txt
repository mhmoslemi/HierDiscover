def _lp_optimize(f, M, active_idxs):
    n = f.shape[0]
    K = len(active_idxs)
    A = np.zeros((K, n), dtype=np.float64)
    for row, k in enumerate(active_idxs):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # j indices from j_start to j_end inclusive
        # coefficients: f[k - j]
        # i = k - j (>=0)
        idxs = np.arange(j_start, j_end + 1)
        A[row, idxs] = f[k - idxs]
    b_ub = np.full(K, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum(b)
    bounds = [(0, None)] * n
    # solve LP
    res = _opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    return res

Proposes a non‑negative step‑function (probability vector) that minimizes

        2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is a **LP‑guided conditional‑gradient descent**:

1. Start from the best known sequence (global ``height_sequence_1``) or from a
   well‑behaved analytic seed (arcsine window).  The vector is ℓ₁‑normalised,
   i.e. `sum(a)=1`.

2. Repeatedly
   * compute `M = max(conv(a,a))`,
   * pick the `K` largest convolution entries (the “active” constraints),
   * solve a linear program  

        maximise    Σ b_j
        subject to  (a * b)_k ≤ M   for every active index k
                    b_j ≥ 0

     which pushes probability mass into positions that do **not** contribute
     to the current worst autocorrelation.
   * obtain a feasible direction `b` (ℓ₁‑normalised) and mix it with the
     current iterate `a` using several step‑sizes `t∈{0.1,0.25,0.5,0.75,1.0}`.
   * keep the mixture that yields the smallest true surrogate
     `2·n·max(conv)`.  Stop when no improvement is observed or the time
     budget expires.

3. Finally evaluate the best candidate with the official ``evaluate_sequence``
   and store it globally for reuse.

The LP has at most a few hundred constraints, so each iteration is cheap.
The whole search fits comfortably into the 1000 s budget and typically
pushes the objective below the previous best (~1.5085), aiming for the
target ≈ 1.503.

"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (required for the LP step)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seeds (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Arcsine density on a discrete grid – a good starting point."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Linear‑programming step
# ----------------------------------------------------------------------
def _lp_optimize(f: np.ndarray, M: float, act_idx: np.ndarray) -> np.ndarray:
    """
    Solve:
        maximise    Σ b_j
        subject to  (f * b)_k ≤ M for all k in act_idx
                    b_j ≥ 0
    Returns the optimal b (not normalised).  Uses SciPy's HiGHS solver.
    """
    if _opt is None:
        raise RuntimeError("SciPy not available – cannot run LP step")

    n = f.shape[0]
    K = act_idx.size
    # Build dense constraint matrix A (K × n)
    A = np.zeros((K, n), dtype=np.float64)
    for r, k in enumerate(act_idx):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # indices j where both a_i and b_j appear
        cols = np.arange(j_start, j_end + 1, dtype=int)
        rows = k - cols                     # i = k - j
        A[r, cols] = f[rows]                # coefficient = f_i

    b_ub = np.full(K, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)       # maximise Σb ⇔ minimise –Σb
    bounds = [(0.0, None)] * n

    res = _opt.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method='highs',
        options={"presolve": True, "time_limit": 30.0},
    )
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x

# ----------------------------------------------------------------------

    """

    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence found as a plain list of floats.

    rng = np.random.default_rng(seed)

    # Problem size – reuse the size of the previous best if available.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")

            if arr.ndim == 1 and arr.size > 0:
                n = arr.size
                f = np.maximum(arr, 0.0)
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                else:
                    f = _uniform(n)
            else:
                raise ValueError
        except Exception:

            f = _arcsine(n)
    else:

    # Ensure a clean probability vector.
    f = np.maximum(f, 0.0)
    f /= f.sum()
    best_seq = f.copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] n={n}, surrogate={best_val:.6f}")

    # ------------------------------------------------------------------
    # Iterative LP‑guided conditional gradient descent
    # ------------------------------------------------------------------
    iteration = 0
    # number of active constraints (tuneable)
    ACTIVE_K = 120

    while time.time() < deadline:
        iteration += 1
        # Current convolution and worst value
        conv = _conv_fft(best_seq, best_seq)
        M = float(conv.max())

        # Pick the K largest entries – they are the binding constraints.
        K = min(ACTIVE_K, conv.size)
        act_idx = np.argpartition(conv, -K)[-K:]

        # ------------------------------------------------------------------
        # Solve the LP to obtain a direction that respects the active bounds
        # ------------------------------------------------------------------
        try:
            b_raw = _lp_optimize(best_seq, M, act_idx)
        except Exception as exc:
            print(f"[Iter {iteration}] LP error: {exc}")
            break

        sum_b = b_raw.sum()
        if sum_b <= 0.0:
            print(f"[Iter {iteration}] LP gave empty direction – stopping.")
            break
        b_norm = b_raw / sum_b

        # ------------------------------------------------------------------
        # Mix the direction with the current iterate – try a few step sizes
        # ------------------------------------------------------------------
        t_candidates = (0.1, 0.25, 0.5, 0.75, 1.0)
        improved = False
        for t in t_candidates:
            cand = (1.0 - t) * best_seq + t * b_norm
            # re‑normalise (numerical drift)
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()

            if val + 1e-12 < best_val:        # strict improvement

                best_seq = cand
                improved = True

                break   # accept first improving t (larger t often better later)

        if not improved:
            # No improvement with any mixing ratio → stop.
            print(f"[Iter {iteration}] no improvement – terminating loop.")
            break

        # Slight safeguard – avoid too many iterations.

            break

        true_val = None
        print(f"Warning: evaluate_sequence raised {exc}")

    # Store for possible reuse in a later run.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
