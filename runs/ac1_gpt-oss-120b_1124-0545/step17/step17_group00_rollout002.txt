def simulated_annealing(start_seq, n, deadline, rng, initial_temp=0.001, final_temp=1e-7, steps_per_temp=5000):
   cur = start_seq, cur_val
   best = cur, cur_val
   temp = initial_temp
   while time.time()<deadline and temp>final_temp:
        for s in range(steps_per_temp):
            i,j = rng.integers(n,size=2)
            if i == j:
                continue
            # move fraction from i to j
            eps = cur[i] * rng.uniform(0.001,0.01)
            if eps == 0: continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps
            new_val = _objective_surrogate(new)
            delta = new_val - cur_val
            if delta < 0 or rng.random() < math.exp(-delta / temp):
                cur = new
                cur_val = new_val
                if new_val < best_val:
                    best = new
                    best_val = new_val
        temp *= 0.95  # exponential schedule
   return best, best_val

   ``2·n·max(conv)`` (the true objective when the vector is ℓ₁‑normalised).

2. **Exponentiated‑gradient polishing** – mirror‑descent on the exact
   max‑convolution (sub‑gradient is known analytically).  Operates in
   log‑softmax space, guaranteeing the simplex constraint.  A large
   number of iterations (time‑controlled) drives the surrogate down.

3. **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp surrogate
   of the max‑convolution, gradually increasing the smoothing parameter
   β.  The optimisation runs in the same log‑softmax space with exact
   gradients and benefits from the fast quasi‑Newton updates of L‑BFGS‑B.

4. **Simulated‑annealing hill‑climb** – tiny mass‑transfer moves on the
   simplex are accepted according to a cooling schedule, allowing the
   search to escape shallow local minima.

The best sequence found is stored in the global ``height_sequence_1``
so that subsequent calls can reuse it as a seed.
"""

import time
import math
import heapq
import numpy as np

# optional SciPy import (L‑BFGS optimiser)
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (valid for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)==1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families of ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)               # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random generators (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with symmetric concentration = conc."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    Returns the best sequence and its surrogate value seen during the run.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]

        # decreasing step size η_t = η₀ / sqrt(t)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 10_000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space; the returned vector lies on the simplex.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser not available.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         1e4, 2e4, 5e4, 1e5, 2e5, 5e5,
                         1e6, 2e6, 5e6]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)               # always on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                       # for numerical stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # soft‑max weights over the convolution entries
            w = exp_beta / sum_exp

            # grad wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # abort if we exceed the deadline
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         init_temp: float = 1e-3,
                         final_temp: float = 1e-7,
                         steps_per_temp: int = 3000) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer moves with a cooling schedule.
    Returns the best sequence seen and its surrogate value.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    temp = init_temp
    while time.time() < deadline and temp > final_temp:
        for _ in range(steps_per_temp):
            i, j = rng.integers(n, size=2)
            if i == j:
                continue
            # move a random fraction of mass from i to j
            eps = cur[i] * rng.uniform(0.001, 0.01)
            if eps <= 0.0 or cur[i] - eps < 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps
            new_val = _objective_surrogate(new)
            delta = new_val - cur_val
            if delta < 0 or rng.random() < math.exp(-delta / temp):
                cur = new
                cur_val = new_val
                if new_val < best_val:
                    best = new.copy()
                    best_val = new_val
        temp *= 0.92      # geometric cooling
    return best, best_val


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence.  The best sequence found (as a plain Python list)
    is returned and stored globally as ``height_sequence_1``.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and timing
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # 2‑second safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Problem size – a larger n gives a finer discretisation
    # ------------------------------------------------------------------
    n = 8000   # feel free to change; 8 k is still cheap for FFTs

    # ------------------------------------------------------------------
    # 2. Helper to keep the best few seeds (min‑heap on surrogate)
    # ------------------------------------------------------------------
    top_k = 12
    best_heap = []               # stores (-value, vector)

    def _push_candidate(vec: np.ndarray):
        """Insert vec into heap, keeping only the top‑k (lowest) values."""
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if val < -best_heap[0][0]:
                heapq.heappushpop(best_heap, (-val, vec))

    # ------------------------------------------------------------------
    # 3. Deterministic analytic seeds
    # ------------------------------------------------------------------
    _push_candidate(_uniform(n))
    _push_candidate(_triangular(n))
    _push_candidate(_hamming_window(n))
    _push_candidate(_blackman_window(n))
    _push_candidate(_nuttall_window(n))
    for gamma in np.linspace(0.2, 3.0, 12):
        _push_candidate(_sine_window(n, gamma))
    for beta in np.linspace(1.0, 15.0, 10):
        _push_candidate(_kaiser_window(n, beta))
    for sigma in np.linspace(0.08, 0.30, 15):
        _push_candidate(_gaussian(n, sigma))
    for p in np.linspace(0.0, 2.5, 12):
        _push_candidate(_power(n, p))

    # finer beta‑shape grid (U‑shaped and bell‑shaped)
    alphas = np.linspace(0.1, 6.0, 12)
    betas = np.linspace(0.1, 6.0, 12)
    for a in alphas:
        for b in betas:
            _push_candidate(_beta_shape(n, a, b))

    for alpha in np.linspace(0.05, 0.95, 9):
        _push_candidate(_tukey_window(n, alpha))

    for sigma in (0.10, 0.13, 0.16):
        for shift in (0.20, 0.27, 0.34):
            _push_candidate(_double_gaussian(n, sigma, shift))

    # ------------------------------------------------------------------
    # 4. Massive random seeding
    # ------------------------------------------------------------------
    conc_choices = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    random_seed_budget = 6000
    for _ in range(random_seed_budget):
        if time.time() > deadline:
            break
        conc = rng.choice(conc_choices)
        vec = _random_dirichlet(n, conc, rng)
        # random cyclic shift to decorrelate from reversed copy
        shift = rng.integers(n)
        if shift:
            vec = np.roll(vec, shift)
        _push_candidate(vec)

    # exponential draws – heavier tails
    for _ in range(500):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        shift = rng.integers(n)
        if shift:
            vec = np.roll(vec, shift)
        _push_candidate(vec)

    # low‑frequency random curves (smooth shapes)
    for nf in (2, 3, 5, 7):
        for _ in range(250):
            if time.time() > deadline:
                break
            vec = _lowfreq_curve(n, nf, rng)
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
            _push_candidate(vec)

    # ------------------------------------------------------------------
    # 5. Re‑use previous solution if dimensions match
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0:
                prev_norm = prev_arr / prev_arr.sum()
                _push_candidate(prev_norm)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 6. Extract the best seeds (ordered from best to worst)
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seed phase] kept {len(seeds)} candidates, best surrogate = {_objective_surrogate(seeds[0]):.6f}")

    # ------------------------------------------------------------------
    # 7. Global best initialisation
    # ------------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Exponentiated‑gradient polishing (majority of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 25.0:
        eg_time = (deadline - time.time()) * 0.55
        eg_deadline = time.time() + eg_time
        seq_eg, val_eg = _exp_gradient_opt(global_best_seq,
                                           n,
                                           eg_deadline,
                                           max_iters=3_000_000,
                                           eta0=8.0)
        if val_eg < global_best_val:
            global_best_seq = seq_eg
            global_best_val = val_eg
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Continuation L‑BFGS fine‑tuning (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n--- L‑BFGS fine‑tuning ---")
        lbfgs_deadline = deadline - 5.0
        seq_lbfgs = _lbfgs_smooth_optimize(global_best_seq,
                                           n,
                                           lbfgs_deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0, 200.0,
                                                          500.0, 1000.0, 2000.0, 5000.0,
                                                          1e4, 2e4, 5e4, 1e5, 2e5,
                                                          5e5, 1e6, 2e6, 5e6],
                                           max_iter_per_beta=12_000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < global_best_val:
            global_best_seq = seq_lbfgs
            global_best_val = val_lbfgs
            print(f"[L‑BFGS] refined surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 10. Simulated‑annealing hill‑climb (use any leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        remaining = deadline - time.time()
        print("\n--- Simulated Annealing hill‑climb ---")
        seq_sa, val_sa = _simulated_annealing(global_best_seq,
                                              n,
                                              deadline,
                                              rng,
                                              init_temp=8e-4,
                                              final_temp=1e-8,
                                              steps_per_temp=5000)
        if val_sa < global_best_val:
            global_best_seq = seq_sa
            global_best_val = val_sa
            print(f"[SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 11. Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
