def _subgradient_opt(start_seq, n, deadline, max_iters):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]  # surrogate
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # compute subgradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        lr = 0.5 / math.sqrt(it)  # maybe adjust step size
        a -= lr * g
        a = _proj_simplex(a)  # ensure positivity and sum=1
    return best_seq, best_val

def _simulated_annealing_opt(start_seq, n, deadline, max_iters):
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.1 * (1.0 - progress) + 1e-6  # temperature schedule
        sigma = 0.2 * (1.0 - progress) + 1e-4
        proposal_seq = cur_seq + sigma * rng.normal(size=n)  # not logit - we can just perturb in probability space and then reproject
        # ensure positivity and renorm
        proposal_seq = np.maximum(proposal_seq, 0.0)
        s = proposal_seq.sum()
        if s > 0:
            proposal_seq = proposal_seq / s
        else:
            continue
        proposal_val = _objective(proposal_seq)
        delta = proposal_val - cur_val
        if delta < 0 or math.exp(-delta / T) > rng.random():
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 100_000 == 0:
            print(f"[SA] it={it} best={best_val:.6f}")
    return best_seq, best_val

def _subgrad_opt(start_seq, n, deadline, max_iters=500000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # compute subgradient for the active index
        g = np.zeros_like(a)

            break

        T = 0.1 * (1 - progress) + 1e-6
        sigma = 0.2 * (1 - progress) + 1e-4
        proposal = cur_seq + sigma * rng.normal(size=n)

                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 100_000 == 0:

"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm mixes several ideas that proved useful in earlier
experiments:

* a large pool of seeds – analytic windows, Dirichlet draws and, crucially,
  random spectra with *constant* magnitude (the inverse FFT of a flat
  frequency magnitude) which tend to produce a near‑flat auto‑convolution;
* a cheap surrogate objective (2·n·max(conv)) that can be evaluated via FFT;
* a long projected sub‑gradient descent on the exact max‑convolution
  (with optional momentum) to polish the best seed;
* a short simulated‑annealing walk in the simplex for a final tweak;
* careful time‑management so the whole search fits into the 1000 s budget.

The returned list is a probability vector (sum≈1); the official
`evaluate_sequence` is scale‑invariant, so any positive scaling would give
the same score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact surrogate for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Random constant‑magnitude frequency seeds
# ----------------------------------------------------------------------
def _rand_const_phase_seq(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random real non‑negative vector whose Fourier magnitude is constant.
    The raw inverse FFT may contain negatives; we clip and re‑normalise onto the simplex.
    """
    phases = rng.random(n // 2 + 1) * 2.0 * np.pi
    # keep the DC component real (phase = 0) to avoid a systematic shift
    phases[0] = 0.0
    if n % 2 == 0:
        phases[-1] = 0.0  # Nyquist term must be real
    A = np.exp(1j * phases)          # constant magnitude = 1
    a = np.fft.irfft(A, n=n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a / s
    else:
        a = _uniform(n)
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def _subgrad_opt(start_seq: np.ndarray,
                 n: int,
                 deadline: float,
                 max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Long projected sub‑gradient descent on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # simple momentum (optional – improves convergence on noisy sub‑gradients)
    momentum = np.zeros_like(a)
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active max‑convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size decays like 1/√t; momentum smoothes the updates
        lr = 0.25 / math.sqrt(it)
        momentum = beta * momentum + (1.0 - beta) * g
        a -= lr * momentum
        a = _proj_simplex(a)

        if it % 100_000 == 0:
            print(f"[SubGrad] it={it:,d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (simple random walk in the simplex)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Random walk in the probability simplex with a simple cooling schedule."""
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal = cur_seq + sigma * rng.normal(size=n)
        proposal = np.maximum(proposal, 0.0)
        s = proposal.sum()
        if s <= 0.0:
            continue
        proposal = proposal / s

        prop_val = _objective(proposal)
        delta = prop_val - cur_val
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            cur_seq = proposal
            cur_val = prop_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:,d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a short safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – a modest dimension that still offers enough freedom
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # ------------------------------------------------------------------
    # 1. Seed pool construction
    # ------------------------------------------------------------------
    seeds = []

    # Analytic families (already ℓ₁‑normalised)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sin2(n))
    seeds.append(_hann_window(n))
    seeds.append(_arcsine(n))
    for sigma in (0.08, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))
    for p in (0.0, 0.3, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))

    # Random Dirichlet draws (wide range of concentration)
    conc_params = np.logspace(-2, 1, 8)            # 0.01 … 10
    for conc in conc_params:
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)))

    # Random constant‑magnitude frequency seeds
    for _ in range(500):
        seeds.append(_rand_const_phase_seq(n, rng))

    # Include previous best if present (interpolated if size differs)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.size == n:
            seeds.append(arr / arr.sum())
        else:
            # linear interpolation to new length
            idx_old = np.arange(arr.size, dtype=np.float64)
            idx_new = np.linspace(0, arr.size - 1, n)
            new_arr = np.interp(idx_new, idx_old, arr)
            new_arr = np.maximum(new_arr, 0.0)
            s = new_arr.sum()
            if s > 0.0:
                seeds.append(new_arr / s)

    # Normalise everything onto the probability simplex
    clean_seeds = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0.0:
            clean_seeds.append(v / s)
        else:
            clean_seeds.append(_uniform(n))
    seeds = clean_seeds

    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # 2. Quick surrogate ranking – keep only the most promising seeds
    # ------------------------------------------------------------------
    surrogate_vals = [_objective(v) for v in seeds]
    evaluated = list(zip(seeds, surrogate_vals))
    evaluated.sort(key=lambda x: x[1])
    top_k = min(15, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nBest surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Global best tracking
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective(best_seq)
    print(f"\n[Init] current best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy sub‑gradient descent on the best seed (long run)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # estimate a safe number of iterations (≈0.001 s per iteration on this hardware)
        max_iters = int(min(1_200_000, remaining * 300))
        max_iters = max(300_000, max_iters)
        print(f"\n[SubGrad] long descent on current best – {max_iters:,d} iters")
        seq_opt, val_opt = _subgrad_opt(best_seq, n, deadline, max_iters=max_iters)
        if val_opt < best_val:
            best_seq, best_val = seq_opt, val_opt
            print(f"[Update] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Shorter refinements on a few other top seeds (optional)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds[1:4], start=2):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        max_iters = int(min(500_000, remaining * 150))
        max_iters = max(150_000, max_iters)
        print(f"\n[SubGrad] secondary seed #{idx} – {max_iters:,d} iters")
        seq_opt, val_opt = _subgrad_opt(seed_vec, n, deadline, max_iters=max_iters)
        if val_opt < best_val:
            best_seq, best_val = seq_opt, val_opt
            print(f"[Update] new global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(800_000, remaining * 200))
        max_sa = max(200_000, max_sa)
        print(f"\n[SA] final simulated‑annealing – {max_sa:,d} iters")
        seq_sa, val_sa = _simulated_annealing_opt(best_seq, n, deadline, max_iters=max_sa)
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print(f"[SA] final surrogate improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                       # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
