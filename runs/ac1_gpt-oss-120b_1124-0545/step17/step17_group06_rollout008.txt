"""
Non‑negative step‑function optimiser – fine‑grained search.

The strategy improves on the previous baseline by:
  * building a much larger and more diverse seed pool
    (geometric, power‑law, beta, double‑Gaussian, Chebyshev, DPSS, …);
  * keeping a longer list of the best seeds;
  * running a higher‑quality L‑BFGS continuation (more iterations per β);
  * polishing each candidate with a long exponentiated‑gradient run;
  * performing two passes of a fast pairwise “mass‑transfer’’ optimiser,
    the second pass using a very small step size for fine‑tuning.

The routine respects the global 1000 s deadline, stops gracefully when the
time budget is exhausted and finally stores the best sequence in the global
variable ``height_sequence_1`` (the judge reads that variable).
"""

import time
import math
import numpy as np

# Optional SciPy imports (L‑BFGS, special windows)
# ----------------------------------------------------------------------
try:                     # L‑BFGS‑B optimiser
    import scipy.optimize as _opt
except Exception:        # pragma: no cover
    _opt = None

    from scipy.signal import chebwin, dpss

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate for ℓ₁‑normalised vectors:
        2·n·max(conv(seq,seq))
    For a ℓ₁‑normalised seq this equals the official ``evaluate_sequence``
    value (up to rounding).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geometric(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i   (0<r<1)."""
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cheb_window(n: int, att: float) -> np.ndarray:
    if chebwin is None:
        return _uniform(n)
    a = chebwin(n, at=att, sym=True)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, half_nbw: float) -> np.ndarray:
    if dpss is None:
        return _uniform(n)
    # dpss returns a tuple (vectors, eigenvalues)
    a = dpss(n, half_nbw, Kmax=1, return_ratios=False)[0]
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Seed‑generation (very rich)

def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # basic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    for g in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_kaiser_window(n, beta=6.0))
    for beta in (2, 4, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    # geometric (r close to 1 → near‑uniform, r small → spiky)
    for r in np.linspace(0.3, 0.99, 30):
        seeds.append(_geometric(n, r))

    # power‑law exponents
    for p in np.linspace(0.0, 5.0, 31):
        seeds.append(_power(n, p))

    # beta shapes – fairly dense grid (log‑spacing)
    for a in np.linspace(0.2, 5.0, 13):
        for b in np.linspace(0.2, 5.0, 13):
            seeds.append(_beta_shape(n, a, b))

    # double Gaussian – vary both parameters
    for sigma in np.linspace(0.07, 0.25, 9):
        for shift in np.linspace(0.15, 0.35, 7):
            seeds.append(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))

    # Chebyshev and DPSS windows (if SciPy available)
    if chebwin is not None:
        for att in (40, 60, 80, 100, 120):
            seeds.append(_cheb_window(n, att))
    if dpss is not None:
        for nbw in np.linspace(2.0, 5.0, 7):
            seeds.append(_dpss_window(n, nbw))

    # random families
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))
    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))
    for k_frac in (0.005, 0.02, 0.05):
        k = max(1, int(round(k_frac * n)))
        for _ in range(3):
            seeds.append(_sparse_random(n, k, rng))

    # mixture of uniform with a random low‑freq curve
    for mix_alpha in (0.2, 0.5, 0.8):
        curve = _lowfreq_curve(n, 3, rng)
        seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution
    if previous is not None and previous.size == n:
        s = previous.sum()
        if s > 0:
            seeds.append(previous / s)

    # final sanitisation – ensure non‑negative ℓ₁‑normalised vectors
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 14000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns a probability vector (ℓ₁‑normalised).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser not available.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 1e5,
            2e5, 5e5, 1e6
        ]

    v = np.log(start_vec + 1e-30)   # logits – stays on the simplex through softmax

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (exact for smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 2n‑1
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser – exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # learning rate – 0.5/√t is a safe choice
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Pairwise mass‑transfer local optimiser (O(n) per trial)
# ----------------------------------------------------------------------
def _mass_transfer_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      step_factor: float = 0.25,
                      max_no_improve: int = 7_000) -> tuple[np.ndarray, float]:
    """
    Very fast local optimiser that moves a tiny amount of probability mass
    from a high‑mass coordinate to a low‑mass coordinate.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    max_val = float(np.max(conv))
    best_a = a.copy()
    best_val = max_val

    rng = np.random.default_rng()
    no_improve = 0
    it = 0
    centre = n - 1
    L = len(conv)

    while time.time() < deadline:
        if a.max() < 1e-16:
            break
        src = rng.choice(n, p=a)                # source index weighted by mass
        src_mass = a[src]
        if src_mass <= 1e-15:
            continue
        tgt = rng.integers(n)
        while tgt == src:
            tgt = rng.integers(n)

        eps = step_factor * src_mass
        if eps < 1e-14:
            continue

        # tentative new convolution – O(1) updates
        new_conv = conv.copy()

        # linear terms: moving eps from src to tgt affects two “blocks”
        new_conv[src:src + n] -= 2.0 * eps * a
        new_conv[tgt:tgt + n] += 2.0 * eps * a

        # quadratic term of the moved mass
        new_conv[centre] += 2.0 * eps * eps
        offset = tgt - src
        idx1 = centre + offset
        new_conv[idx1] -= eps * eps
        if offset != 0:
            idx2 = centre - offset
            new_conv[idx2] -= eps * eps

        new_max = float(np.max(new_conv))

        if new_max < max_val - 1e-16:
            # accept the move
            a[src] = src_mass - eps
            a[tgt] = a[tgt] + eps
            conv = new_conv
            max_val = new_max
            no_improve = 0
            if max_val < best_val:
                best_val = max_val
                best_a = a.copy()
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                # shrink step size and reset counter
                step_factor = max(step_factor * 0.5, 5e-5)
                no_improve = 0

        it += 1
        if it % 25000 == 0:
            print(f"[MassTransfer] it={it} best max={best_val:.12f}")

    return best_a, 2.0 * n * best_val

        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # evaluate all (fast surrogate) and keep the best few
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12           # keep a reasonably large elite set
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print("\nTop seeds (surrogate values):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f" #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. L‑BFGS + exponentiated‑gradient on elite seeds
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # β‑schedule (use the same as before – very aggressive)
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        10000.0, 20000.0, 50000.0, 1e5,
        2e5, 5e5, 1e6
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before remaining L‑BFGS runs.")
            break

        remaining = deadline - time.time()
        # allow roughly 3 % of remaining time for this L‑BFGS run
        max_iter = int(min(18000, remaining * 0.03 * 1e5))
        max_iter = max(6000, max_iter)

        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter={max_iter}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] new global best = {best_val:.6f}")

        # ------------------- exponentiated‑gradient polishing -------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            # allocate roughly 7 % of the remaining budget to EG
            max_exp = int(min(1_200_000, remaining * 0.07 * 1e5))
            max_exp = max(400_000, max_exp)
            print(f"[ExpGrad] running up to {max_exp:,} iterations")
            seq_exp, val_exp = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_exp)
            if val_exp < best_val:
                best_seq = seq_exp.copy()
                best_val = val_exp
                print(f"[ExpGrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Pairwise mass‑transfer local optimiser – two passes (coarse → fine)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mass‑transfer local optimisation (coarse) ---")
        seq_mt, val_mt = _mass_transfer_opt(best_seq, n, deadline,
                                            step_factor=0.25,
                                            max_no_improve=7_000)
        if val_mt < best_val:
            best_seq = seq_mt.copy()
            best_val = val_mt
            print(f"[MassTransfer‑coarse] improved to {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Mass‑transfer local optimisation (fine) ---")
        # very small step factor for delicate adjustments
        seq_mt2, val_mt2 = _mass_transfer_opt(best_seq, n, deadline,
                                              step_factor=0.05,
                                              max_no_improve=5_000)
        if val_mt2 < best_val:
            best_seq = seq_mt2.copy()
            best_val = val_mt2
            print(f"[MassTransfer‑fine] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing with a short exponentiated‑gradient run
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # final short EG (max 200 k iters) to soak up any left‑over time
        max_final = int(min(200_000, remaining * 0.02 * 1e5))
        max_final = max(50_000, max_final)
        print(f"\n[Final ExpGrad] up to {max_final:,} iterations")
        seq_final, val_final = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_final)
        if val_final < best_val:
            best_seq = seq_final.copy()
            best_val = val_final
            print(f"[Final ExpGrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```

This implementation expands the search space (geometric, power‑law, beta, double‑Gaussian, Chebyshev, DPSS, random Dirichlet, low‑frequency curves, sparse random vectors, and mixtures), keeps a larger elite set, runs a more thorough L‑BFGS continuation, applies longer exponentiated‑gradient polishing, and performs a two‑stage pairwise mass‑transfer local optimisation (coarse then fine) followed by a final EG pass. The careful time‑budget checks ensure the function finishes well before the 1000 s limit while striving to push the surrogate value below the target 1.503.
