"""
Propose a non‑negative step function (list of heights) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq,seq)) / (sum(seq)**2)

The search proceeds in three stages:

1. **Broad seed pool** – a large collection of analytically‑generated
   windows (uniform, triangular, sine, Gaussian, Beta‑shape, etc.) and
   many random Dirichlet / exponential draws.  All seeds are ℓ₁‑normalised
   (sum = 1) and the best *k* seeds (according to the cheap surrogate
   `2·n·max(conv)`) are kept.

2. **Adam optimisation on a smoothed max** – each top seed is refined
   by mirror‑descent on the log‑sum‑exp smooth approximation of the
   maximum convolution value.  The optimisation runs in the log‑softmax
   space (guaranteeing non‑negativity and the simplex constraint) and
   uses an Adam‑style adaptive step.  The “temperature’’ β is increased
   gradually (continuation) so the optimiser first sees a smooth loss
   and then a sharper approximation of the true max.

3. **Exact max polishing** – a final exponentiated‑gradient (mirror‑descent)
   pass on the exact max‑convolution objective brings the solution as
   close as possible to a true local optimum.

The whole routine respects a 1000 s wall‑clock budget, stores the best
sequence it has seen in the global variable ``height_sequence_1`` and
returns it as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Bessel I0, available in NumPy
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _alternating_pattern(n: int, high: float = 1.0, low: float = 0.1) -> np.ndarray:
    """Deterministic alternating high / low pattern, ℓ₁‑normalised."""
    a = np.empty(n, dtype=np.float64)
    a[0::2] = high
    a[1::2] = low
    s = a.sum()
    return a / s


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, keep non‑negative, renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exact max‑convolution mirror descent (exponentiated gradient)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule=None,
                   max_iters_per_beta: int = 1500,
                   lr: float = 0.07):
    """
    Adam optimisation on the smoothed max‑convolution loss.
    Operates in log‑softmax space (variables v = log a) which enforces
    non‑negativity and the simplex constraint.
    Returns the best sequence found (ℓ₁‑normalised) and its surrogate.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0, 20000.0,
                         50000.0, 100000.0, 200000.0]

    # initialise log‑variables
    v = np.log(start_vec + 1e-30)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = None
    best_val = np.inf

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        m = np.zeros_like(v)
        v2 = np.zeros_like(v)
        t = 0
        no_improve = 0

        while time.time() < deadline and t < max_iters_per_beta:
            # -------- forward pass ----------
            a = _softmax(v)                     # current probabilities, sum=1
            conv = _conv_fft(a, a)              # full convolution, length 2n‑1
            max_c = np.max(conv)                # true max (used for shifting)
            shift = max_c

            # smooth max via log‑sum‑exp with temperature β
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp              # soft weights, sum=1

            # gradient w.r.t. a (derived analytically)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule to grad w.r.t. v (log‑softmax)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # -------- Adam update ----------
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v2 / (1.0 - beta2 ** t)

            step = lr / (np.sqrt(v_hat) + eps)
            v = v - step * m_hat

            # -------- objective value ----------
            obj = float(2.0 * n * (shift + np.log(sum_exp) / beta))

            if obj < best_val - 1e-12:
                best_val = obj
                best_seq = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # early break if we stall for many iterations
            if no_improve >= 300:
                break

    # fallback – if nothing succeeded (should not happen) return current vector
    if best_seq is None:
        best_seq = _softmax(v)
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function respects the wall‑clock budget,
    stores the best found sequence in the global variable ``height_sequence_1``,
    and returns it as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # tiny safety margin

    n = 4000                                        # fixed problem size

    # ------------------------------------------------------------------
    # Load previous solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            pa = np.asarray(previous, dtype=np.float64)
            if pa.size == n:
                prev_arr = pa / pa.sum() if pa.sum() > 0 else None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation – a large, diverse pool
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # analytic families (coarse parameter sweeps)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in np.linspace(1.0, 15.0, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.09, 0.26, 14):
        seeds.append(_gaussian(n, sigma))
    for sigma in (0.10, 0.12):
        for shift in (0.20, 0.25, 0.30):
            seeds.append(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))
    for p in np.linspace(0.0, 2.0, 13):
        seeds.append(_power(n, p))
    alphas = np.linspace(0.2, 5.0, 7)
    betas = np.linspace(0.2, 5.0, 7)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))
    for alpha in np.linspace(0.1, 0.9, 7):
        seeds.append(_tukey_window(n, alpha))
    seeds.append(_alternating_pattern(n))

    # random Dirichlet draws – many concentrations, many samples
    conc_levels = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_levels:
        for _ in range(20):                     # 20 draws per concentration
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(15):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(8):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures of uniform and low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # include previous best (if any) and a few perturbed copies
    if prev_arr is not None:
        seeds.append(prev_arr)                        # exact previous
        for _ in range(8):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"\nTop {top_k} surrogate values among seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best from the very best seed
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement (Adam + exact EG polishing)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a time slice for this seed – roughly equal share of the
        # remaining budget, but leave room for later polishing
        remaining = deadline - time.time()
        share = remaining * 0.55 / (top_k - idx)   # 55 % of remaining time
        sub_deadline = time.time() + share

        # ----- Adam optimisation on smoothed max -----
        seq_adam, val_adam = _adam_optimize(seed_vec.copy(),
                                            n,
                                            sub_deadline,
                                            max_iters_per_beta=1800,
                                            lr=0.07)

        # ----- final exact‑max mirror descent (if time permits) -----
        if time.time() < deadline:
            seq_final, val_final = _exp_gradient_opt(seq_adam,
                                                     n,
                                                     deadline,
                                                     max_iters=500_000,
                                                     eta0=5.0)
        else:
            seq_final, val_final = seq_adam, val_adam

        # ----- pick best among raw seed, Adam, final EG -----
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_adam, val_adam),
            (seq_final, val_final)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {idx+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"  >>> New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Optional extra perturb‑and‑refine passes from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        extra_passes = 4
        for p in range(extra_passes):
            if time.time() >= deadline:
                break
            pert = _perturb(global_best_seq, rng, sigma=0.004 + 0.001 * p)
            sub_deadline = time.time() + (deadline - time.time()) * 0.3
            seq_adam, _ = _adam_optimize(pert,
                                         n,
                                         sub_deadline,
                                         max_iters_per_beta=1500,
                                         lr=0.07)
            seq_final, val_final = _exp_gradient_opt(seq_adam,
                                                     n,
                                                     deadline,
                                                     max_iters=300_000,
                                                     eta0=5.0)
            if val_final < global_best_val:
                global_best_seq = seq_final
                global_best_val = val_final
                print(f"[Extra {p+1}] improved global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
