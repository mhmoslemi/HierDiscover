"""
Improved search for a step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Seed generation** – a cheap parametric sweep over a few
   families (exponential, beta, uniform, triangular,
   low‑frequency cosine mixes, random Dirichlet).  The best
   ℓ₁‑normalised seed (sum = 1) is kept.

2. **Adam optimiser on the exact max‑convolution objective** – the
   current best seed is re‑parameterised by logits (softmax) so that
   the ℓ₁‑constraint is always satisfied.  At every iteration a
   sub‑gradient of the (non‑smooth) max‑convolution is computed,
   transformed to the log‑domain, and an Adam update is applied.
   The best point ever seen is stored.

3. **Polish + local hill‑climb** – a short high‑β L‑BFGS run on the
   log‑sum‑exp smooth surrogate followed by a stochastic mass‑transfer
   local search.

All stages respect the 1000 s deadline.  The final sequence is stored
in the global ``height_sequence_1`` for warm starts.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (each returns an ℓ₁‑normalised vector)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)  (ℓ₁‑normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient helpers
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² (sum may be ≠1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the max‑convolution w.r.t. a."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    # each pair (i, rev) contributes twice (i≠rev) → factor 2
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _adam_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.15,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam on the real (non‑smooth) objective.
    The vector is kept on the ℓ₁‑simplex via the softmax re‑parameterisation.
    Returns the best sequence seen and its objective value.
    """
    logits = np.log(start_a + 1e-30)        # unconstrained parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    best_a = start_a.copy()
    best_val = _objective_exact(best_a)

    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1
        a = _softmax(logits)               # ℓ₁‑normalised vector
        # sub‑gradient of max‑convolution
        g = _subgradient_maxconv(a)        # shape (n,)
        # chain rule through softmax
        dot = np.dot(g, a)
        grad_logits = a * (g - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress check
        if t % 5000 == 0:
            cur_val = _objective_exact(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                # (optional) shrink learning‑rate after a good improvement
                lr *= 0.95

    # final evaluation
    final_a = best_a if best_val < _objective_exact(start_a) else start_a
    final_val = _objective_exact(final_a)
    return final_a, final_val


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth approximation of 2·n·max(conv) / (∑a)².
    Returns (obj, grad) – grad w.r.t. logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)               # full linear convolution
    shift = np.max(conv)                 # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient wrt a
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 1500,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Very small stochastic local search (move a fraction of mass)
# ----------------------------------------------------------------------
def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 25_000,
    step_frac: float = 0.012,
) -> tuple[np.ndarray, float]:
    """Hill‑climbing: move a small fraction of mass from i to j."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Seed‑pool construction (diverse and cheap)
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_beta_shape(n, 2.0, 2.0))

    # exponential sweep – λ tuned especially for our problem
    for lam in np.linspace(3e-4, 1.5e-3, 12):
        seeds.append(_exponential_decay(n, lam))

    # random Dirichlet families
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency curves
    for nf in (2, 3, 4):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # pairwise mixtures of random members
    base = np.arange(len(seeds))
    for _ in range(150):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of a few promising seeds
    for _ in range(12):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best (warm‑start)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up – non‑negativity + ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget, stores the best result in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # --------------------------------------------------------------
    # collect seeds
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # --------------------------------------------------------------
    # quick exact evaluation of seeds, keep the elite pool
    # --------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam optimisation on the exact objective
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        iters_adam = int(min(300_000, remaining * 0.45 / 1e-5))  # heuristic
        print("\n--- Adam optimisation (exact max‑conv) ---")
        seq_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            deadline,
            max_iters=iters_adam,
            lr=0.15,
        )
        print(f"[Adam] best = {val_adam:.6f}")
        if val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"[Update] New global best after Adam = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Continuation L‑BFGS polishing (high β)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        per_beta = max(800, int(min(2000, remaining * 0.12 * 1e5)))
        print("\n--- Final polishing L‑BFGS (log‑sum‑exp) ---")
        seq_polish = _lbfgs_smooth_optimize(
            best_seq,
            n,
            deadline,
            beta_schedule=[1e5, 5e5, 1e6, 5e6, 1e7],
            max_iter_per_beta=per_beta,
        )
        val_polish = _objective_exact(seq_polish)
        print(f"[Polish‑L‑BFGS] objective = {val_polish:.6f}")
        if val_polish < best_val:
            best_seq = seq_polish.copy()
            best_val = val_polish
            print(f"[Update] New global best after polishing = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – aggressive stochastic local search
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        iters_loc = int(min(70_000, remaining * 60))
        print("\n--- Stochastic local search (mass transfers) ---")
        seq_loc, val_loc = _local_search(
            best_seq,
            n,
            deadline,
            iters=iters_loc,
            step_frac=0.010,
        )
        print(f"[LocalSearch] best = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq = seq_loc.copy()
            best_val = val_loc
            print(f"[Update] New global best after local search = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging)
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for warm‑starts in future invocations
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
