"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.  

The algorithm consists of the following stages (all respecting the
global 1000 s budget):

1. **Seed generation** – a large and diverse pool of ℓ₁‑normalised
   vectors (uniform, triangular, Gaussian, exponential, beta,
   Dirichlet, sparse, low‑frequency cosine sums, random piecewise
   linear shapes, mixtures and rotations).  The previous best sequence
   (if any) is also injected as a seed.

2. **Elite selection** – evaluate every seed with the exact objective
   ``2·n·max(conv(a,a))`` (the denominator is 1 because the vectors are
   ℓ₁‑normalised) and keep the best few dozen.

3. **Continuation L‑BFGS** – a smooth log‑sum‑exp surrogate of the
   max‑convolution is minimised for each elite seed.  The surrogate is
   solved with L‑BFGS for a schedule of increasing ``β`` values; the
   final ``β`` is very large (≈10⁷) so the surrogate closely matches the
   true max.  The best result of this stage becomes the current global
   best.

4. **Exponentiated‑gradient (mirror descent)** – several random
   Dirichlet restarts are run, each performing many mirror‑descent steps
   on the exact objective using the sub‑gradient of the active lag.
   The best outcome updates the global best.

5. **Stochastic annealed local search** – a temperature‑controlled
   hill‑climber repeatedly transfers a small fraction of mass from one
   index to another (occasionally accepting uphill moves).  This helps
   to escape shallow local minima found by the previous stages.

6. **Final polishing** – another short L‑BFGS run with very high ``β``
   to polish the best candidate.

All stages share a common helper set (FFT‑based convolution,
ℓ₁‑projection, soft‑max, etc.) and constantly check the remaining time.
The final sequence is stored in the global ``height_sequence_1`` variable
so that future invocations can warm‑start from it.

The code below follows the described pipeline and adheres to the
requirements (no closures, top‑level functions only, no I/O apart from
printing progress).
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for large n)."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax producing a probability vector (ℓ₁‑norm = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the probability simplex (non‑negative, sum = 1)."""
    # Standard algorithm: sort, find threshold τ.
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (sum = 1 → denominator drops out)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray, n: int) -> float:
    """
    Compute 2·n·max(conv(a,a)) for a ℓ₁‑normalised non‑negative vector a.
    If a sums to zero (should not happen) returns +inf.
    """
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    # a is already ℓ₁‑normalised in our pipeline, but keep generality:
    a = a / sum_a
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (used by exponentiated‑gradient)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Return a sub‑gradient of max(conv(a,a)) (active lag)."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))          # lag where max is attained
    # Determine which pairs (i, j) contribute to this lag:
    # i + j = idx  ⇔  j = idx - i
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    # Each pair contributes a[i] * a[rev] to the convolution.
    # Derivative w.r.t a[i] = a[rev] + a[rev]? Actually d/d a[i] of sum_j a[j] a[idx-j] = a[rev] + a[rev] = 2*a[rev]
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Returns (smooth_obj, grad_v) where the smooth object approximates
    2·n·max(conv(a,a)) and a = softmax(v) is ℓ₁‑normalised.
    Gradient is taken w.r.t. the un‑constrained logits v.
    """
    a = _softmax(v)                # ℓ₁‑normalised positive vector
    conv = _conv_fft(a, a)          # full length = 2n‑1
    shift = np.max(conv)            # numerical shift for stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    # smooth max ≈ shift + log(sum_exp)/beta
    smooth_max = shift + np.log(sum_exp) / beta
    obj = 2.0 * n * smooth_max

    # gradient w.r.t a:
    w = exp_term / sum_exp          # softmax over lags
    rev_a = a[::-1]                 # reversed a
    full = _conv_fft(rev_a, w)      # length = 3n‑2, but we only need n entries
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray, n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            1e4, 2e4, 5e4, 1e5, 2e5,
            5e5, 1e6, 2e6, 5e6, 1e7,
        ]

    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun, v, method="L-BFGS-B", jac=True,
                                callback=callback,
                                options={"maxiter": max_iter_per_beta,
                                         "disp": False,
                                         "ftol": 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray, n: int,
                          deadline: float,
                          max_iters: int = 500_000,
                          eta0: float = 1.5) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact objective (sub‑gradient style)."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a, n)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / math.sqrt(it)
        # exponential update on the simplex (mirror descent with KL)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 100_000 == 0:
            cur_val = _objective_exact(a, n)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
    # final check
    cur_val = _objective_exact(a, n)
    if cur_val < best_val:
        best_val = cur_val
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Stochastic annealed local search (mass‑transfer moves)
# ----------------------------------------------------------------------
def _annealed_local_search(start_a: np.ndarray, n: int,
                          deadline: float,
                          max_iters: int = 150_000,
                          init_temp: float = 0.05,
                          final_temp: float = 1e-5) -> tuple[np.ndarray, float]:
    """Hill‑climbing with Metropolis acceptance."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a, n)

    rng = np.random.default_rng(314159)

    total_iters = 0
    start_time = time.time()
    while total_iters < max_iters and time.time() < deadline:
        # linear temperature schedule
        elapsed = time.time() - start_time
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        fraction = elapsed / (deadline - start_time)
        T = init_temp * (1.0 - fraction) + final_temp * fraction

        i = rng.integers(n)
        if a[i] <= 1e-15:
            continue
        j = rng.integers(n)
        if i == j:
            continue
        # move a random fraction of mass from i to j
        frac = rng.uniform(0.001, 0.02)
        delta = frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # keep normalization exact (mass conserved)
        cand_val = _objective_exact(cand, n)
        delta_val = cand_val - best_val
        accept = False
        if cand_val < best_val:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < math.exp(-delta_val / max(T, 1e-12)):
                accept = True

        if accept:
            a = cand
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand.copy()
        total_iters += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Analytic shape families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(1e-9, sigma_frac * n)
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random piecewise linear (convex) shape."""
    num_pts = rng.integers(3, 8)  # number of breakpoints
    xs = np.sort(rng.integers(0, n, size=num_pts, replace=False))
    ys = rng.random(num_pts)
    # enforce non‑negativity and create linear interpolation
    a = np.interp(np.arange(n), xs, ys)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = _uniform(n)
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mix = (1.0 - alpha) * a + alpha * b
    s = mix.sum()
    return mix / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.2))
    seeds.append(_gaussian(n, 0.35))
    seeds.append(_exponential_decay(n, 0.001))
    seeds.append(_exponential_decay(n, 0.02))
    seeds.append(_beta_shape(n, 2.0, 2.0))
    seeds.append(_beta_shape(n, 0.8, 0.8))

    # random Dirichlet (various concentrations)
    for _ in range(40):
        conc = rng.uniform(0.01, 2.5)
        seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse
    for frac in (0.02, 0.05, 0.1, 0.2):
        for _ in range(8):
            seeds.append(_random_sparse(n, frac, rng))

    # low‑frequency cosine sums
    for nf in (2, 3, 4, 5, 6):
        for _ in range(12):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # random piecewise linear shapes
    for _ in range(15):
        seeds.append(_piecewise_linear(n, rng))

    # pairwise mixtures of existing seeds
    base = list(range(len(seeds)))
    for _ in range(300):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # rotations of promising seeds
    for _ in range(30):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # previous best (warm‑start)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget, stores the best result in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + float(budget_s) - 0.5   # safety margin

    # problem dimension – 2000 gives a good trade‑off between speed
    # and expressive power (objective is essentially scale‑invariant)
    n = 2000
    print(f"\n=== Optimisation start – n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Seed generation (including possible warm‑start)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # Quick exact evaluation of all seeds, keep elite pool
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for v in seeds:
        val = _objective_exact(v, n)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # Sort seeds by objective and keep top‑k for heavy optimisation
    seeds.sort(key=lambda v: _objective_exact(v, n))
    elite = seeds[:min(20, len(seeds))]
    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on elite seeds
    # ------------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        1e4, 2e4, 5e4, 1e5, 2e5,
        5e5, 1e6, 2e6, 5e6, 1e7,
    ]

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            print("[Time] deadline before finishing L‑BFGS runs.")
            break
        # allocate iterations based on remaining time
        remaining = deadline - time.time()
        per_beta = max(800, int(min(2500, remaining * 0.06 * 1e5)))
        print(f"\n--- L‑BFGS run #{idx+1}/{len(elite)} (max_iter/β={per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=per_beta)
        val_opt = _objective_exact(seq_opt, n)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient restarts
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        restarts = 12
        iters_per_restart = max(50_000, int(min(300_000, remaining * 0.12 / restarts)))
        print("\n--- Exponentiated‑Gradient restarts ---")
        for r in range(restarts):
            if time.time() > deadline:
                break
            conc = rng.uniform(0.02, 1.5)
            start_vec = _random_dirichlet(n, conc, rng)
            seq_eg, val_eg = _exp_gradient_optimize(
                start_vec, n, deadline,
                max_iters=iters_per_restart,
                eta0=2.0,
            )
            print(f"[EG {r+1}/{restarts}] best = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_seq = seq_eg.copy()
                global_best_val = val_eg
                print(f"[Update] New best after EG restart = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – annealed stochastic local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = max(80_000, int(min(250_000, remaining * 0.25)))
        print("\n--- Annealed stochastic local search ---")
        seq_loc, val_loc = _annealed_local_search(
            global_best_seq, n, deadline,
            max_iters=max_iters,
            init_temp=0.07,
            final_temp=1e-5,
        )
        print(f"[Annealed LS] best = {val_loc:.6f}")
        if val_loc < global_best_val:
            global_best_seq = seq_loc.copy()
            global_best_val = val_loc
            print(f"[Update] New best after annealed LS = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final high‑β polishing L‑BFGS
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(800, int(min(2500, remaining * 0.05 * 1e5)))
        print("\n--- Final polishing L‑BFGS (very high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq, n, deadline,
            beta_schedule=[5e5, 1e6, 2e6, 5e6, 1e7],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final, n)
        print(f"[Final L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for future warm‑starts
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
