```
def local_random_search(seq, start_time, deadline, steps, step_size):
    cur_seq = seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    n = len(seq)
    for it in range(steps):
        if time.time()>deadline: break
        # choose two indices to transfer mass
        i = rng.integers(0,n)
        j = rng.integers(0,n)
        if i==j: continue
        # propose moving delta from i to j (or vice versa) 
        delta = step_size * rng.random()
        # ensure not moving more than cur_seq[i]
        delta = min(delta, cur_seq[i])
        # propose new seq
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # don't need projection as sum unchanged and non-neg ensured
        # compute objective
        new_val = _objective_surrogate(new_seq)
        # accept if lower or hill climb with probability
        if new_val < cur_val or rng.random() < 0.001:
            cur_seq = new_seq
            cur_val = new_val
            if cur_val < best_val:
                best_seq = cur_seq.copy()
                best_val = cur_val
        # optionally adapt step size
```

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a, a)) / (sum(a)²)

using a larger problem size, a richer family of analytic seeds
(including Beta‐distributions), an extended L‑BFGS continuation on the
log‑sum‑exp surrogate, long exact sub‑gradient descent and a final
simulated‑annealing polish.  All work respects the 1000 s budget and
stores the best sequence in the global ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import for L‑BFGS‑B
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:          # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N).real[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _clip_and_normalise(v: np.ndarray) -> np.ndarray:
    """Clip values to the evaluator's limits and renormalise to sum = 1."""
    v = np.clip(v, 0.0, 1000.0)
    s = v.sum()
    if s <= 0.0:
        return _uniform(len(v))
    return v / s


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return _clip_and_normalise(a)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return _clip_and_normalise(a)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _clip_and_normalise(a)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    return _clip_and_normalise(a)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    return _clip_and_normalise(a)


def _arcsine(n: int) -> np.ndarray:
    """Approximation of the continuous arcsine density on [0,1]."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    return _clip_and_normalise(a)


def _beta_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric Beta(p, p) density on a grid of size n.
    p > 0; p = 1 gives uniform.
    """
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    eps = 1e-12
    # density ∝ t^{p-1} (1-t)^{p-1}
    a = (t + eps) ** (p - 1) * (1 - t + eps) ** (p - 1)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    return _clip_and_normalise(a)


def _mixture(seq1: np.ndarray, seq2: np.ndarray, w: float) -> np.ndarray:
    """Convex combination w*seq1 + (1‑w)*seq2, then renormalise."""
    a = w * seq1 + (1.0 - w) * seq2
    return _clip_and_normalise(a)


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule=None,
                    max_iter_per_beta: int = 5000) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smooth max‑conv."""
    if _opt is None:                              # pragma: no cover
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0,
        ]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                 # stays on the simplex
            conv = _conv_fft(a, a)                 # full linear conv
            max_c = np.max(conv)
            # log‑sum‑exp stabilised
            shifted = conv - max_c
            exp_beta = np.exp(beta_val * shifted)
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (max_c + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                # soft‑weights, sum(w)=1
            rev_a = a[::-1]
            # gradient of (a*a) w.r.t a is 2·conv(rev_a, w)
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)           # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:                     # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Long sub‑gradient descent on the exact max‑convolution surrogate."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # learning‑rate schedule
        lr = 0.25 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing style refinement (logit‑space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Random walk in logit space with a simple cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted time.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a finer discretisation.
    # ------------------------------------------------------------------
    n = 8000                      # chosen to stay within the budget

    print(f"\n--- Seed generation (n={n}) ---")
    prev = globals().get("height_sequence_1")
    seeds = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sin2(n))
    seeds.append(_hann_window(n))
    seeds.append(_arcsine(n))

    # Gaussian with several widths
    for sigma in (0.08, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))

    # Power‑law family
    for p in (0.0, 0.3, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))

    # Symmetric Beta(p, p) for a dense grid of p ∈ (0.05, 1.0)
    for p in np.linspace(0.05, 0.95, 19):
        seeds.append(_beta_distribution(n, p))

    # Mixtures of uniform and arcsine – allow a gentle tilt.
    for w in np.linspace(0.0, 1.0, 11):
        seeds.append(_mixture(_uniform(n), _arcsine(n), w))

    # Random Dirichlet draws (wide range of concentration)
    conc_params = np.logspace(-2, 1, 12)
    for conc in conc_params:
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)))

    # Re‑use the previous best if it has the same length
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                seeds.append(_clip_and_normalise(arr))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Normalise everything onto the probability simplex
    # ------------------------------------------------------------------
    clean_seeds = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        clean_seeds.append(_clip_and_normalise(v))
    seeds = clean_seeds

    # ------------------------------------------------------------------
    # Quick surrogate ranking – keep only the most promising seeds
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(20, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nBest surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] current best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS refinement (smooth surrogate)
    # ------------------------------------------------------------------
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        20_000.0, 50_000.0, 100_000.0,
        200_000.0, 500_000.0, 1_000_000.0,
        2_000_000.0, 5_000_000.0,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS passes.")
            break

        # allocate a modest slice of the remaining budget for each seed
        remaining = deadline - time.time()
        max_iter_beta = int(min(8_000, remaining * 0.045 * 1e5))
        max_iter_beta = max(4_000, max_iter_beta)

        print(f"\n--- L‑BFGS on seed #{idx+1} (max_iter/β={max_iter_beta}) ---")
        seq_opt = _lbfgs_optimize(seed_vec,
                                  n,
                                  deadline,
                                  beta_schedule=beta_schedule,
                                  max_iter_per_beta=max_iter_beta)

        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] new global best surrogate = {best_val:.6f}")

        # --------------------------------------------------------------
        # Short exponentiated‑gradient polish of the current best
        # --------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_exp = int(min(400_000, remaining * 0.03 * 1e5))
            max_exp = max(200_000, max_exp)
            seq_exp, val_exp = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_exp)
            if val_exp < best_val:
                best_seq = seq_exp.copy()
                best_val = val_exp
                print(f"[ExpGrad] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Heavy sub‑gradient descent (exact max‑conv surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        restarts = [best_seq] + top_seeds[:4]
        for r_idx, start_vec in enumerate(restarts, 1):
            if time.time() > deadline:
                break
            remaining = deadline - time.time()
            max_sub = int(min(1_500_000, remaining * 0.04 * 1e5))
            max_sub = max(500_000, max_sub)
            print(f"\n--- Sub‑gradient descent restart {r_idx} (max_iters={max_sub}) ---")
            seq_sub, val_sub = _subgradient_opt(start_vec,
                                                n,
                                                deadline,
                                                max_iters=max_sub)
            if val_sub < best_val:
                best_seq = seq_sub.copy()
                best_val = val_sub
                print(f"[SubGrad] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(1_200_000, remaining * 0.05 * 1e5))
        max_sa = max(300_000, max_sa)
        print(f"\n--- Simulated‑annealing refinement (iters={max_sa}) ---")
        seq_sa, val_sa = _simulated_annealing_opt(best_seq,
                                                  n,
                                                  deadline,
                                                  max_iters=max_sa)
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                       # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()


# ----------------------------------------------------------------------
# Helper: exponentiated‑gradient (mirror‑descent) optimiser (smooth surrogate)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Short exponentiated‑gradient polish on the exact max‑conv surrogate."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val
