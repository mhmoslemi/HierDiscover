"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
1. Choose a problem size `n`.  Larger `n` gives more degrees of freedom;
   we use `n = 5000` (still fast with FFT).
2. Build a rich pool of ℓ₁‑normalised seed vectors:
   • classic windows (uniform, triangular, sinusoid, Kaiser, Tukey …)
   • super‑Gaussian / generalized normal families
   • exponential‑decay shape
   • random Dirichlet draws with many concentrations
   • mixtures of the above and a few random log‑it vectors.
   The previous best solution is also injected.
3. Evaluate the cheap surrogate `2·n·max(conv)` for every seed and keep the
   `k_best` most promising ones.
4. For each selected seed run a **smooth‑max Adam optimiser**:
   – work in log‑probability space (softmax guarantees a probability simplex)
   – use the log‑sum‑exp smooth approximation to the max‑convolution,
   – increase the smoothing parameter β on a schedule and perform a
     fixed number of Adam steps per β, always respecting the global deadline.
5. Polish the Adam result with **exponentiated‑gradient descent** on the
   exact max‑convolution (mirrored‑descent on the simplex).  This is the
   same sub‑gradient method used previously but we allow many more
   iterations (up to the remaining time).
6. Finally perform a **swap‑based hill‑climber** (randomly swap two
   positions, keep the swap if it lowers the exact surrogate).  This
   inexpensive local search often yields a final improvement.
7. Return the best vector found (as a plain ``list[float]``).  The code
   constantly checks the remaining time and stops cleanly before the
   1000 s limit.

The whole pipeline uses only ``numpy`` and a few small helper utilities,
so it runs comfortably on a single CPU core within the allotted budget.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps ℝⁿ → probability simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Analytic seed families (all return ℓ₁‑normalised vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # modified Bessel I0 (np.i0 exists)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _supergaussian(n: int, sigma_frac: float, beta: float) -> np.ndarray:
    """Generalised Gaussian: exp(-|x|^beta)."""
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-np.abs(i - mu) ** beta / (2.0 * sigma ** beta))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential shape."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_logit(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gaussian logits → softmax."""
    logits = rng.normal(loc=0.0, scale=1.0, size=n)
    a = _softmax(logits)
    return a


def _mix_two(v1: np.ndarray, v2: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * v1 + alpha * v2
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(v1))


def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse pool of ℓ₁‑normalised seed vectors.
    """
    seeds: list[np.ndarray] = []

    # Analytic families ----------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Sine windows with several gammas
    for g in (0.3, 0.5, 1.0, 2.0, 4.0):
        seeds.append(_sine_window(n, gamma=g))

    # Kaiser with a grid of betas
    for beta in (2, 4, 8, 12, 16, 20):
        seeds.append(_kaiser_window(n, beta))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Gaussian and super‑Gaussian families
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    for (sigma, beta) in [(0.12, 4), (0.18, 8), (0.25, 12)]:
        seeds.append(_supergaussian(n, sigma, beta))

    # Exponential decay shapes
    for lam in (0.0005, 0.001, 0.002, 0.004):
        seeds.append(_exp_decay(n, lam))

    # Random Dirichlet draws ------------------------------------------------
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random logit vectors
    for _ in range(4):
        seeds.append(_random_logit(n, rng))

    # Mixtures of random pairs
    for _ in range(6):
        a = _random_dirichlet(n, rng.uniform(0.05, 2.0), rng)
        b = _random_dirichlet(n, rng.uniform(0.05, 2.0), rng)
        alpha = rng.random()
        seeds.append(_mix_two(a, b, alpha))

    # Previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise sanity‑check (already normalised, but guard against NaNs)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0 and np.isfinite(s):
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

def _adam_smooth_optimize(start_vec: np.ndarray,
                          n: int,
                          deadline: float,
                          beta_schedule=None,
                          total_steps: int = 300_000,
                          learning_rate: float = 0.12) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using Adam.
    Works in log‑probability space (softmax guarantees ℓ₁‑normalisation).
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5,
            2e5, 5e5, 1e6, 2e6
        ]

    log_a = np.log(start_vec + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    steps_per_beta = max(1, total_steps // len(beta_schedule))
    t = 0

    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return _softmax(log_a)

            t += 1
            a = _softmax(log_a)                     # ℓ₁‑normalised probabilities
            conv = _conv_fft(a, a)                  # full convolution
            max_c = np.max(conv)

            # smooth‑max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp                  # soft weights that sum to 1

            # gradient w.r.t. a (same derivation as in the L‑BFGS version)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)             # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_log
            v = beta2 * v + (1 - beta2) * (grad_log ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            log_a -= learning_rate * m_hat / (np.sqrt(v_hat) + eps)

    return _softmax(log_a)

# Exponentiated‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000) -> np.ndarray:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # learning rate – classic 1/√t schedule
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress report (quiet in final runs)
        if it % 200_000 == 0:
            print(f"[ExpGrad] iter={it:,} best={best_val:.6f}")

    return best_seq

# Swap‑based permutation hill‑climber (exact surrogate)
# ----------------------------------------------------------------------
def _swap_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_swaps: int = 300_000) -> np.ndarray:
    """
    Randomly swap two entries; keep the swap iff it reduces the exact surrogate.
    Returns the best sequence seen.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _objective_surrogate(cur_seq)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if it % 100_000 == 0:
            print(f"[SwapHill] swaps={it:,} best={best_val:.6f}")

    return best_seq

    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] Deadline reached before finishing all seeds.")
            break

        # Allocate a chunk of time for this seed.
        remaining = deadline - time.time()
        # Roughly spend 20 % of remaining time on Adam for each seed.
        seed_deadline = time.time() + remaining * 0.20

        print(f"\n--- Adam optimisation on seed #{idx+1} ---")
        seq_adam = _adam_smooth_optimize(seed_vec,
                                          n,
                                          seed_deadline,
                                          beta_schedule=beta_sched,
                                          total_steps=200_000,
                                          learning_rate=0.12)
        val_adam = _objective_surrogate(seq_adam)
        print(f"[Adam] surrogate = {val_adam:.6f}")

        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Update] new global best after Adam = {global_best_val:.6f}")

        # --------------------------------------------------------------
        # 4a. Exact max‑convolution polishing via exponentiated gradient
        # --------------------------------------------------------------
        if time.time() < deadline:
            # Give roughly half of the leftover budget to the EG stage.
            eg_deadline = time.time() + (deadline - time.time()) * 0.5
            max_iters = int(min(800_000, (eg_deadline - time.time()) * 400))
            max_iters = max(200_000, max_iters)
            print(f"[ExpGrad] polishing with up to {max_iters:,} iterations")
            seq_exp = _exp_gradient_opt(global_best_seq,
                                        n,
                                        eg_deadline,
                                        max_iters=max_iters)
            val_exp = _objective_surrogate(seq_exp)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final swap‑based hill‑climber
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = int(min(500_000, remaining * 0.25))
        max_swaps = max(150_000, max_swaps)
        print(f"\n--- Final swap hill‑climber (max_swaps={max_swaps}) ---")
        seq_swap = _swap_hill_climb(global_best_seq,
                                    n,
                                    deadline,
                                    max_swaps=max_swaps)
        val_swap = _objective_surrogate(seq_swap)
        if val_swap < global_best_val:
            global_best_seq = seq_swap.copy()
            global_best_val = val_swap
            print(f"[SwapHill] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # make the solution available for a possible next round
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
