    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm:

  2. Evaluate the surrogate objective 2·n·max(conv) (exact when sum = 1)

     optimisation stays on the probability simplex via the softmax.
  4. Polish the L‑BFGS result with an exponentiated‑gradient (mirror‑descent)

  5. If time permits, run a simulated‑annealing walk in log‑softmax space

  6. Return the best vector found (as a plain list).  The vector is stored

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) for a sequence that sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_distribution_general(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta(α,β) density on {0,…,n‑1} using half‑offset.
    Works for any α,β>0 (asymmetric allowed).
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p  (symmetric, zero at the ends)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_uniform_block(n: int, block_size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of `block_size` indices."""
    idx = rng.choice(n, size=block_size, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / block_size
    return a


def _mixture_beta(n: int, components, rng: np.random.Generator) -> np.ndarray:
    """
    Build a mixture of Beta pdfs.
    components = list of (alpha, beta, weight) tuples; weights must sum to 1.
    """
    a = np.zeros(n, dtype=np.float64)
    for alpha, beta, w in components:
        a += w * _beta_distribution_general(n, alpha, beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

    # ------------------------------------------------------------------
    # Symmetric Beta families (α = β)

    for a in (0.3, 0.4, 0.5, 0.6, 0.8, 1.0, 1.2, 1.5, 2.0):

    # ------------------------------------------------------------------

    for _ in range(40):

        b = 10 ** log_b
        seeds.append(_beta_distribution_general(n, a, b))

    # ------------------------------------------------------------------
    # Random mixtures of a few Beta components
    # ------------------------------------------------------------------
    for _ in range(30):

    # ------------------------------------------------------------------
    # Random Dirichlet draws (different concentrations)
    # ------------------------------------------------------------------
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                            # raw
            seeds.append(np.sort(raw)[::-1])             # sorted desc
            # alternating placement
            sorted_desc = np.sort(raw)[::-1]
            left, right = 0, n - 1
            out = np.empty_like(raw)
            for v in sorted_desc:
                if (left + right) % 2 == 0:
                    out[left] = v
                    left += 1
                else:
                    out[right] = v
                    right -= 1
            seeds.append(out)

    # ------------------------------------------------------------------
    # Sparse uniform blocks
    # ------------------------------------------------------------------
    for blk in (80, 120, 160, 200):

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0.0:

    # ------------------------------------------------------------------
    # Normalise all seeds (just in case)
    # ------------------------------------------------------------------
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))

# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 1500) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smoothed max‑conv."""
    if _opt is None:
        # SciPy not available – fall back to a simple mirror‑descent step.
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 3.0, 7.0, 15.0, 30.0, 60.0,
            120.0, 250.0, 500.0, 1000.0, 2500.0,
            5000.0, 10000.0, 25000.0, 50000.0
        ]

    # optimisation variables are logits; the softmax guarantees positivity & unit sum
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            """Smooth surrogate and its gradient."""
            a = _softmax(v_local)                     # stays on simplex
            conv = _conv_fft(a, a)
            m = np.max(conv)
            shift = m                                   # for numerical stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # Smooth approximation to max
            smooth_max = shift + np.log(sum_exp) / beta_val
            obj = 2.0 * n * smooth_max

            # Gradient of smooth max w.r.t. a
            w = exp_beta / sum_exp                     # soft‑weights (sum=1)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # convolution of rev(a) with w
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # length‑n gradient

            # Chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser (exact surrogate)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent polish on the exact max‑conv surrogate."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (pairwise terms)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Random walk in log‑softmax space with a linear cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)

    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.08 * (1.0 - progress) + 1e-6          # temperature
        sigma = 0.14 * (1.0 - progress) + 1e-4      # proposal width

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Small random perturbation (used after a good solution)
# ----------------------------------------------------------------------
def _random_perturb(seq: np.ndarray,
                    n: int,
                    rng: np.random.Generator,
                    magnitude: float = 0.04) -> np.ndarray:
    """Add a tiny random logit perturbation and re‑softmax."""
    logit = np.log(seq + 1e-30)
    perturb = magnitude * rng.normal(size=n)
    new_log = logit + perturb
    return _softmax(new_log)


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and time budget
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a modest safety margin for the final evaluation call
    deadline = start_time + budget_s - 4.0

    # ------------------------------------------------------------------
    # 1. Problem size – a trade‑off between quality and speed.
    #    Larger n gives a finer approximation of the continuous problem.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 2. Seed pool construction
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # ------------------------------------------------------------------
    # 3. Quick surrogate evaluation and selection of promising seeds
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 25                      # keep a relatively tight elite set
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nBest surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 4. Continuation L‑BFGS + Exponentiated‑Gradient polishing
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    beta_schedule = [
        1.0, 3.0, 7.0, 15.0, 30.0, 60.0,
        120.0, 250.0, 500.0, 1000.0, 2500.0,
        5000.0, 10000.0, 25000.0, 50000.0
    ]

    # Run the heavy L‑BFGS only on the top few seeds (time‑budget aware)
    seeds_to_opt = top_seeds[:5]
    for idx, seed_vec in enumerate(seeds_to_opt):
        if time.time() > deadline:
            print("[Time] Deadline reached before finishing L‑BFGS on all elite seeds.")
            break

        print(f"\n--- L‑BFGS on elite seed #{idx + 1} ---")
        # Allocate a slice of remaining time (roughly 6 % per seed)
        remaining = deadline - time.time()
        max_iter_beta = int(min(2500, remaining * 0.06 * 1e5))
        max_iter_beta = max(1200, max_iter_beta)

        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter_beta)

        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq, global_best_val = seq_opt.copy(), val_opt
            print(f"[Update] new global best (L‑BFGS) = {global_best_val:.6f}")

        # ---- Exponentiated‑gradient polish (exact surrogate) ----
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_exp = int(min(300_000, remaining * 0.07 * 1e5))
            max_exp = max(120_000, max_exp)

            seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_exp)
            if val_exp < global_best_val:
                global_best_seq, global_best_val = seq_exp.copy(), val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Random perturbation + short L‑BFGS restart (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random perturbation + restart L‑BFGS ---")
        for rep in range(3):
            if time.time() > deadline:
                break
            perturbed = _random_perturb(global_best_seq, n, rng, magnitude=0.07)

            remaining = deadline - time.time()
            max_iter_beta = int(min(1800, remaining * 0.04 * 1e5))
            max_iter_beta = max(1000, max_iter_beta)

            seq_restart = _lbfgs_smooth_optimize(perturbed,
                                                 n,
                                                 deadline,
                                                 beta_schedule=beta_schedule[:8],
                                                 max_iter_per_beta=max_iter_beta)
            val_restart = _objective_surrogate(seq_restart)
            if val_restart < global_best_val:
                global_best_seq, global_best_val = seq_restart.copy(), val_restart
                print(f"[Restart] new best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing refinement (final phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(800_000, remaining * 0.07 * 1e5))
        max_sa = max(200_000, max_sa)

        print("\n--- Simulated‑annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing_opt(global_best_seq,
                                                  n,
                                                  deadline,
                                                  max_iters=max_sa)
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa.copy(), val_sa
            print(f"[SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next iteration (if any)
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
