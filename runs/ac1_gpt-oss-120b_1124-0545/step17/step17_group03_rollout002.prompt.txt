def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; result lies on the simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _uniform(n: int) -> np.ndarray:

def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()

def _linear_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)  # n, n‑1, …, 1

def _exponential_decay(n: int, lam: float) -> np.ndarray:

def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)

def _gaussian(n: int, sigma_frac: float) -> np.ndarray:

    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:

    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()

    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)

    s = a.sum()
    return a / s

def _rotate(v: np.ndarray, shift: int) -> np.ndarray:

def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Log‑sum‑exp smooth approximation of max(a∗a).
    Returns (obj, grad) where grad is the derivative w.r.t. logits v.
    """
    a = _softmax(v)                     # simplex point
    conv = _conv_fft(a, a)               # full convolution
    shift = np.max(conv)                 # numeric stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # weights of the smooth‑max
    w = exp_term / sum_exp               # length = 2n‑1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,

    v = np.log(start_vec + 1e-30)   # logits

    for beta in beta_schedule:

            return _smooth_objective(v_local, n, beta_val)

def _adam_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta: float,
    max_steps: int,
    lr_initial: float = 0.5,
) -> np.ndarray:

    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta_val = float(beta)

    for _ in range(max_steps):
        if time.time() > deadline:
            break
        loss, grad = _smooth_objective(v, n, beta_val)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)
        lr = lr_initial / np.sqrt(t)
        v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

    return _softmax(v)

def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (sum(a) = 1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient for a single maximizing lag."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = idx - i_range
    g[i_range] = 2.0 * a[rev_idx]
    return g


def _subgradient_maxconv_multi(a: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient that averages over all almost‑maximising lags.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    idxs = np.where(conv >= max_c - eps)[0]
    g = np.zeros_like(a)
    for idx in idxs:
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx - i_range
        g[i_range] += 2.0 * a[rev_idx]
    if len(idxs) > 0:
        g /= len(idxs)
    return g

def _mass_transfer_local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    total_iters: int = 100_000,
    step_frac_start: float = 0.15,
    step_frac_end: float = 0.01,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing: repeatedly move a fraction of mass from one index to another.
    The transferred fraction decays linearly from step_frac_start to step_frac_end.
    """
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(987654321)

    for it in range(total_iters):
        if time.time() > deadline:
            break

        # linearly decreasing step size
        frac = step_frac_start + (step_frac_end - step_frac_start) * (it / total_iters)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue

        delta = frac * a[i]
        if delta <= 0.0:
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        val = _objective_exact(a_candidate)
        if val < best_val:
            best_val = val
            best_a = a_candidate.copy()
            a = a_candidate

    return best_a, best_val
