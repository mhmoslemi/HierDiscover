"""
propose_candidate – a 1000 s meta‑heuristic for the max‑convolution bound.

Main ideas
----------
1. Warm‑start from the best sequence found so far (global variable
   ``height_sequence_1``) – this already beats the uniform bound.
2. Apply a small number of *LP refinement* steps (the classic
   “max‑convolution LP” from the problem statement).  For a fixed
   f, the LP maximises ∑g_j under the linear constraints
   (f∗g)(k) ≤ M (M = max(f∗f)).  The optimal g is blended with f.
   Only a few LP solves are needed – each solve costs O(n²) but with
   n=4000 the sparse Toeplitz matrix fits comfortably in memory.
3. Perform a longer exponentiated‑gradient (mirror descent) run on the
   exact objective.  The gradient is the sub‑gradient of the active
   convolution lag, and the update uses the KL‑projection (softmax)
   onto the ℓ₁‑simplex.
4. Finish with an aggressive stochastic “mass‑transfer” local search
   and a final high‑β L‑BFGS polishing on the smooth log‑sum‑exp
   surrogate.

All stages respect the global deadline (≈1000 s).  The best sequence
encountered is stored in the global variable ``height_sequence_1`` and
returned as a plain list.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex (ℓ₁‑sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# exact objective & auxiliary helpers
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) / (∑a)² (sum may be ≠1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) for the active lag."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# smooth surrogate (log‑sum‑exp) used by continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Smooth approx of 2·n·max(conv) / (∑a)².
    Returns (obj, grad) – grad w.r.t. logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)               # full convolution
    shift = np.max(conv)                 # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # softmax over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 2000,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits (unconstrained)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def fun(v_local):
            return _smooth_objective(v_local, n, beta)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 150_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# stochastic mass‑transfer local search
# ----------------------------------------------------------------------
def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 30_000,
    step_frac: float = 0.015,
) -> tuple[np.ndarray, float]:
    """Hill‑climbing: move a small fraction of mass from i to j."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456789)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
    return best_a, best_val


# ----------------------------------------------------------------------
# LP refinement (the classic “max‑convolution” linear programme)
# ----------------------------------------------------------------------
def _lp_refine(
    f: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 8,
    blend: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Perform a handful of LP refinements:
        maximise   Σ g_j
        subject to  (f * g)(k) ≤ M   for all k,
                    g_j ≥ 0.
    The optimal g is ℓ₁‑normalised and blended into f:
        f ← (1‑blend)·f + blend·g .
    Returns the refined vector and its exact objective value.
    """
    try:
        import cvxpy as cp
    except Exception as exc:  # pragma: no cover
        print("CVXPY not available – skipping LP refinement.")
        return f, _objective_exact(f)

    best_f = f.copy()
    best_val = _objective_exact(best_f)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # current bound M
        M = np.max(_conv_fft(best_f, best_f))

        # build the Toeplitz convolution matrix A (shape: (2n‑1, n))
        # each diagonal offset d (0 ≤ d < n) contains the constant f[d].
        diag_vals = [np.full(n, best_f[d]) for d in range(n)]
        offsets = np.arange(n)
        A = _sp.diags(diag_vals, offsets, shape=(2 * n - 1, n), format="csr")

        # LP: maximise sum(g)  s.t.  A @ g ≤ M
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M * np.ones(2 * n - 1)]
        objective = cp.Maximize(cp.sum(g))
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception as exc:
            print(f"[LP] Solver failed on iteration {it+1}: {exc}")
            break

        if g.value is None:
            print("[LP] No solution returned – aborting refinement.")
            break

        g_vec = np.maximum(g.value, 0.0)
        s = g_vec.sum()
        if s <= 0.0:
            print("[LP] Solution collapsed to zero – aborting.")
            break

        # ℓ₁‑normalise g and blend
        g_norm = g_vec / s
        new_f = (1.0 - blend) * best_f + blend * g_norm
        new_f = np.maximum(new_f, 0.0)
        new_f /= new_f.sum()  # re‑normalise to ℓ₁‑norm = 1

        new_val = _objective_exact(new_f)
        if new_val < best_val:
            best_val = new_val
            best_f = new_f
            print(f"[LP refine {it+1}/{max_iters}] improved objective → {best_val:.6f}")
        else:
            # no improvement – shrink the blend factor
            blend = max(blend * 0.5, 0.001)

        # free the large sparse matrix before the next iteration
        del A

    return best_f, best_val
