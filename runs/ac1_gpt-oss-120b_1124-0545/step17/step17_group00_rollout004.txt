* Running a **short exponentiated‑gradient polish** on the *top‑k*
  seeds (instead of only the very best) to diversify the search.
* Upscaling the best coarse solution to a finer grid (n_large = 4000)
  and applying a **long continuation‑L‑BFGS** optimisation on a smooth
  log‑sum‑exp surrogate with a richer β‑schedule.
* Finishing with a **simulated‑annealing hill‑climb** that moves mass
  between random indices and accepts worsening moves with a temperature
  schedule, allowing the search to escape local minima.
* Adding a few extra analytic families (DPSS, wide‑range beta‑shape,
  wide‑range Gaussian / power exponents) that were missing before.

All steps respect the 1000 s time budget and the simplex
constraint (non‑negative entries summing to 1). The final sequence is
stored in the global variable ``height_sequence_1`` for reuse.
"""

import time
import math
import heapq
import numpy as np

# optional SciPy imports
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

try:
    from scipy.signal.windows import dpss
except Exception:  # pragma: no cover
    dpss = None


# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective (exact – works for any scale)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact evaluation: 2·n·max(conv(a,a)) / (sum(a)^2)."""
    n = len(seq)
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv) / (s * s))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, half_bw: float) -> np.ndarray:
    """DPSS (Slepian) window – falls back to uniform if SciPy missing."""
    if dpss is None:
        return _uniform(n)
    try:
        w = dpss(n, half_bw, return_ratios=False)
        w = np.maximum(w, 0.0)
        s = w.sum()
        return w / s if s > 0 else _uniform(n)
    except Exception:  # pragma: no cover
        return _uniform(n)


# ----------------------------------------------------------------------
# Random generators (return ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a symmetric Dirichlet distribution."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines (clipped to ≥0)."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_binary_mask(n: int, prob: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform weight on a random subset of indices (Bernoulli mask)."""
    mask = rng.random(n) < prob
    if not mask.any():
        mask[0] = True
    a = np.where(mask, 1.0, 0.0)
    s = a.sum()
    return a / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated gradient) on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (pairwise contributions)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser with log‑sum‑exp smoothing
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 15000) -> np.ndarray:
    """L‑BFGS‑B optimisation of the smoothed max‑convolution."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0, 10_000.0,
            20_000.0, 50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0, 2_000_000.0
        ]

    # initialise logits (log‑softmax space)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # stays on simplex
            conv = _conv_fft(a, a)            # full convolution
            max_c = np.max(conv)
            shift = max_c                     # stabilise exponentials
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # soft‑max weights over convolution entries
            w = exp_beta / sum_exp

            # gradient wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated annealing / hill‑climb on the simplex
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Small‑scale SA that moves a random fraction of mass between two indices.
    Accepts worsening moves with probability exp(-Δ/T). Temperature decays
    linearly to 1 % of its initial value.
    """
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    start_time = time.time()
    total_time = deadline - start_time
    if total_time <= 0:
        return best, best_val

    T0 = cur_val * 0.1 + 1e-6               # modest start temperature
    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        frac = max(0.0, 1.0 - elapsed / total_time)
        T = T0 * (0.01 + 0.99 * frac)       # never below 1 % of T0

        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # transfer a random fraction of the mass at i to j
        eps = cur[i] * rng.uniform(0.001, 0.05)   # up to 5 % of a[i]
        if eps <= 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective(new)
        delta = new_val - cur_val
        accept = False
        if delta <= 0.0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / T):
                accept = True
        if accept:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val


# ----------------------------------------------------------------------
# Upscaling helper (linear interpolation)
# ----------------------------------------------------------------------
def _upscale(seq_small: np.ndarray, n_large: int) -> np.ndarray:
    """Interpolate a small sequence to a larger length, keeping total mass = 1."""
    x_small = np.linspace(0.0, 1.0, len(seq_small))
    x_large = np.linspace(0.0, 1.0, n_large)
    interp = np.interp(x_large, x_small, seq_small, left=0.0, right=0.0)
    s = interp.sum()
    if s <= 0.0:
        return _uniform(n_large)
    return interp / s


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the time budget and stores the best found sequence in
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Coarse problem size (fast exploration)
    # ------------------------------------------------------------------
    n_small = 1800          # enough resolution, cheap conv
    top_k = 30              # keep a larger pool for diversification

    # ------------------------------------------------------------------
    # 2. Helper to keep best seeds (min‑heap on objective)
    # ------------------------------------------------------------------
    best_heap = []   # stores (-value, vector)

    def _push_candidate(vec: np.ndarray):
        val = _objective(vec)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if val < -best_heap[0][0]:
                heapq.heappushpop(best_heap, (-val, vec))

    # ------------------------------------------------------------------
    # 3. Deterministic analytic seeds (expanded ranges)
    # ------------------------------------------------------------------
    _push_candidate(_uniform(n_small))
    _push_candidate(_triangular(n_small))

    for gamma in np.linspace(0.2, 5.0, 12):
        _push_candidate(_sine_window(n_small, gamma))

    _push_candidate(_hamming_window(n_small))
    _push_candidate(_blackman_window(n_small))
    _push_candidate(_nuttall_window(n_small))
    _push_candidate(_tukey_window(n_small, 0.5))

    for beta in np.linspace(1.0, 20.0, 10):
        _push_candidate(_kaiser_window(n_small, beta))

    for sigma in np.linspace(0.05, 0.45, 15):
        _push_candidate(_gaussian(n_small, sigma))

    for p in np.concatenate([np.linspace(0.0, 5.0, 11),
                             np.linspace(5.0, 12.0, 8)]):
        _push_candidate(_power(n_small, p))

    # broad beta‑shape search
    alphas = np.concatenate([np.linspace(0.05, 5.0, 11),
                             np.linspace(5.0, 20.0, 6)])
    betas = np.concatenate([np.linspace(0.05, 5.0, 11),
                            np.linspace(5.0, 20.0, 6)])
    for a in alphas:
        for b in betas:
            _push_candidate(_beta_shape(n_small, a, b))

    for bw in np.linspace(2.0, 8.0, 7):
        _push_candidate(_dpss_window(n_small, half_bw=bw))

    # Double‑Gaussian with a wider sweep
    for sigma in (0.08, 0.12, 0.18):
        for shift in (0.15, 0.25, 0.35):
            _push_candidate(_double_gaussian(n_small, sigma, shift))

    # ------------------------------------------------------------------
    # 4. Random seeding (Dirichlet, exponential, low‑freq, binary masks)
    # ------------------------------------------------------------------
    conc_choices = [0.001, 0.005, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    rand_dirichlet_budget = 4000
    for _ in range(rand_dirichlet_budget):
        if time.time() > deadline:
            break
        conc = rng.choice(conc_choices)
        vec = _random_dirichlet(n_small, conc, rng)
        # random cyclic shift for extra diversity
        if rng.random() < 0.3:
            shift = rng.integers(n_small)
            vec = np.roll(vec, shift)
        _push_candidate(vec)

    # exponential draws (heavier tails)
    for _ in range(500):
        if time.time() > deadline:
            break
        vec = _random_exponential(n_small, rng)
        if rng.random() < 0.3:
            vec = np.roll(vec, rng.integers(n_small))
        _push_candidate(vec)

    # low‑frequency cosine curves
    for nf in (2, 3, 5, 8):
        for _ in range(300):
            if time.time() > deadline:
                break
            vec = _lowfreq_curve(n_small, nf, rng)
            if rng.random() < 0.3:
                vec = np.roll(vec, rng.integers(n_small))
            _push_candidate(vec)

    # random binary masks with various densities
    for prob in np.linspace(0.05, 0.5, 10):
        for _ in range(200):
            if time.time() > deadline:
                break
            vec = _random_binary_mask(n_small, prob, rng)
            _push_candidate(vec)

    # ------------------------------------------------------------------
    # 5. Re‑use previous best if compatible
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n_small and prev_arr.sum() > 0:
                _push_candidate(prev_arr / prev_arr.sum())
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 6. Keep the best few seeds
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seed phase] retained {len(seeds)} candidates, best value = {_objective(seeds[0]):.6f}")

    # ------------------------------------------------------------------
    # 7. Exponentiated‑gradient polishing on several top seeds
    # ------------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"[Init] global best value = {global_best_val:.6f}")

    # allocate roughly half of the remaining budget to gradient polishing
    grad_deadline = time.time() + (deadline - time.time()) * 0.48
    for idx, seed_vec in enumerate(seeds[:min(7, len(seeds))]):
        if time.time() > grad_deadline:
            break
        seq, val = _exp_gradient_opt(seed_vec,
                                     n_small,
                                     grad_deadline,
                                     max_iters=800_000,
                                     eta0=5.0)
        if val < global_best_val:
            global_best_seq = seq
            global_best_val = val
            print(f"[ExpGrad #{idx}] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Upscale to the final size
    # ------------------------------------------------------------------
    n_large = 4000
    seq_large = _upscale(global_best_seq, n_large)
    best_val_large = _objective(seq_large)
    print(f"\n[Upscale] objective on n={n_large} = {best_val_large:.6f}")

    # ------------------------------------------------------------------
    # 9. Continuation L‑BFGS fine‑tuning on the large problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("\n--- L‑BFGS fine‑tuning (large) ---")
        lbfgs_deadline = deadline - 30.0
        seq_lbfgs = _lbfgs_smooth_optimize(seq_large,
                                           n_large,
                                           lbfgs_deadline,
                                           beta_schedule=[
                                               1.0, 5.0, 20.0, 80.0, 200.0,
                                               500.0, 1_000.0, 2_000.0,
                                               5_000.0, 10_000.0, 20_000.0,
                                               50_000.0, 100_000.0,
                                               200_000.0, 500_000.0,
                                               1_000_000.0, 2_000_000.0,
                                               5_000_000.0
                                           ],
                                           max_iter_per_beta=18000)
        val_lbfgs = _objective(seq_lbfgs)
        if val_lbfgs < best_val_large:
            seq_large = seq_lbfgs
            best_val_large = val_lbfgs
            print(f"[L‑BFGS] refined objective = {best_val_large:.6f}")

    # ------------------------------------------------------------------
    # 10. Simulated‑annealing hill‑climb (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Simulated‑annealing hill‑climb ---")
        seq_sa, val_sa = _simulated_annealing(seq_large,
                                              n_large,
                                              deadline,
                                              rng)
        if val_sa < best_val_large:
            seq_large = seq_sa
            best_val_large = val_sa
            print(f"[SA] improved to {best_val_large:.6f}")

    # ------------------------------------------------------------------
    # 11. Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(seq_large.tolist())
        print(f"\n=== Finished ===\nCoarse surrogate ≈ {global_best_val:.6f}, "
              f"large surrogate = {best_val_large:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = seq_large.tolist()
    return seq_large.tolist()
