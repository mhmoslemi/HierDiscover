def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, scipy.optimize as opt, scipy.sparse as sp

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0

    n = 3000  # choose moderate n
    # initial seed: use geometric/exponential shape
    sigma = 0.2
    i = np.arange(n)
    a = np.exp(-i / (sigma * n))
    a = a / a.sum()
    f = a.copy()

    # initial compute objective
    conv_f = np.convolve(f, f)
    M = conv_f.max()
    best_f = f.copy()
    best_obj = 2 * n * M / (f.sum()**2) # but sum=1 => 2*n*M
    print(f"Init: obj={best_obj}")

    iter = 0
    while time.time() < deadline:
        iter += 1
        # Build constraint matrix A (size L x n) in sparse csr format
        L = 2 * n - 1
        # Use lil_matrix for efficient row construction
        A = sp.lil_matrix((L, n), dtype=np.float64)
        for i in range(n):
            # positions k where b_j appears with coefficient f_i
            # column j corresponds to variable b_j
            # For each column j, row k = i + j
            # So for fixed i, we assign f_i to rows i: i+n-1 in column j range
            # Equivalent: for j in 0..n-1: row = i + j
            # We can fill by row wise loops maybe slower; maybe we loop over j, but symmetrical.
            # Simpler: for j from 0..n-1: A[i+j, j] = f_i
            #: but constructing manually O(n^2)
        ...
        A = A.tocsr()

        # Solve LP
        c = -np.ones(n)
        b_ub = np.full(L, M)
        bounds = [(0, None)] * n
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"LP failure at iteration {iter}")
            break
        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0:
            print("LP gave zero sum, stop")
            break
        g = b_opt / sum_b  # normalize to sum=1

        # convex combination
        t = 0.3  # mixing fraction
        f_new = (1 - t) * f + t * g
        # compute new objective
        conv_new = np.convolve(f_new, f_new)
        M_new = conv_new.max()
        obj_new = 2 * n * M_new

        print(f"Iter {iter}: obj={obj_new:.6f}, M={M_new:.8f}, sum_b={sum_b:.6f}")

        # if improved, accept; else decrease t or break
        if obj_new < best_obj - 1e-6:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            M = M_new
        else:
            # try smaller t?
            t = t * 0.5
            if t < 1e-3:
                print("No improvement, stopping")
                break
            # maybe update f = (1 - t) * f + t * g anyway ?

    # final evaluate true cost
    seq = best_f.tolist()
    true_val = evaluate_sequence(seq)
    print(f"Final obj = {true_val}")

    globals()["height_sequence_1"] = seq
    return seq

def _lbfgs_optimize(start_vec, n, deadline, beta_schedule):
    if opt is None:
        return start_vec.copy()
    v = np.log(start_vec + 1e-30)
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)
        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shifted = conv - max_c
            exp_beta = np.exp(beta_val * shifted)
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (max_c + np.log(sum_exp) / beta_val)  # surrogate objective
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v
        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()
        try:
            res = opt.minimize(fun=fun, x0=v, method="L-BFGS-B", jac=True, callback=callback,
                               options={'maxiter': 20000, 'ftol': 1e-12, 'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:
            print(f"[L-BFGS] error {e}")
            break
    return _softmax(v)
```

"""
Propose a low‑discrepancy step‑function that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The routine works in four stages:

1.  A rich pool of analytic seeds (uniform, triangular, Gaussian, arcsine,
    Beta‑family, generalized Gaussian, random Dirichlet) is generated and
    normalised on the probability simplex.  The best few seeds are kept.

2.  A smooth log‑sum‑exp surrogate for the max‑convolution is optimised with
    L‑BFGS‑B (projected via a soft‑max parameterisation).  This gives a fast
    “global” polishing of the best seed.

3.  Exact sub‑gradient descent on the true max‑convolution further reduces the
    bound.  The sub‑gradient of the active entry of the convolution is used,
    followed by Euclidean projection onto the simplex.

4.  A final simulated‑annealing walk in log‑probability (logit) space explores
    the remaining neighbourhood and often yields a tiny extra improvement.

All operations respect the non‑negativity and sum‑normalisation constraints,
so the final list can be handed directly to ``evaluate_sequence``.
The global variable ``height_sequence_1`` stores the best result for reuse.
"""

import time
import math
import numpy as np

# optional SciPy – used for L‑BFGS‑B
try:
    import scipy.optimize as _opt
except Exception:          # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – always returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density ∝ 1/√(x(1‑x))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64)
    # centre of each bin
    x = (i + 0.5) / n
    eps = 1e-12
    a = (x + eps) ** (alpha - 1) * (1 - x + eps) ** (alpha - 1)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gen_gaussian(n: int, p: float) -> np.ndarray:
    """Generalised Gaussian ∝ exp(‑|x‑c|ᵖ)."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.exp(-np.abs(i - c) ** p / (c ** p))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Objective helper (exact surrogate – sum is exactly 1 for all candidates)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) (seq is ℓ₁‑normalised)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule=None,
                    max_iter_per_beta: int = 4000) -> np.ndarray:
    """Polish ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smooth surrogate."""
    if _opt is None:                     # pragma: no cover
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0,
        ]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on the simplex
            conv = _conv_fft(a, a)                     # full linear conv
            max_c = np.max(conv)

            # log‑sum‑exp smoothing of the max
            shifted = conv - max_c
            exp_beta = np.exp(beta_val * shifted)
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (max_c + np.log(sum_exp) / beta_val)

            # gradient of the smoothed max
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:                       # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max‑convolution surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Long sub‑gradient descent on the true max‑convolution surrogate."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]          # sum(a)=1 ⇒ denominator =1
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry of the max‑convolution
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.4 / math.sqrt(it)                # diminishing step size
        a -= lr * g
        a = _proj_simplex(a)                    # back to the simplex

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing style refinement (logit space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Random walk in logit space with a simple cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-7       # temperature
        sigma = 0.15 * (1.0 - progress) + 1e-5   # proposal scale

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

    return best_seq, best_val

        print("\n--- L‑BFGS refinement of the current best seed ---")
        seq_opt = _lbfgs_optimize(best_seq,
                                  n,
                                  deadline,
                                  beta_schedule=beta_schedule,
                                  max_iter_per_beta=3000)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] new global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy sub‑gradient descent (exact max‑conv surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(1_500_000, remaining * 0.06 * 1e5))
        max_sub = max(300_000, max_sub)
        print(f"\n--- Sub‑gradient descent (up to {max_sub} iters) ---")
        seq_sub, val_sub = _subgradient_opt(best_seq,
                                            n,
                                            deadline,
                                            max_iters=max_sub)
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"[SubGrad] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(1_200_000, remaining * 0.07 * 1e5))
        max_sa = max(200_000, max_sa)
        print(f"\n--- Simulated‑annealing refinement (iters={max_sa}) ---")
        seq_sa, val_sa = _simulated_annealing_opt(best_seq,
                                                  n,
                                                  deadline,
                                                  max_iters=max_sa)
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    seq_out = best_seq.tolist()
    try:
        true_val = evaluate_sequence(seq_out)
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                       # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible reuse in a later run
    globals()["height_sequence_1"] = seq_out
    return seq_out
