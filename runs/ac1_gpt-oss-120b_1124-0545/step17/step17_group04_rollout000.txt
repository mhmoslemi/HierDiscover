"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

The strategy is:

1. Build a rich pool of analytic and random Dirichlet seeds.
2. Keep the best few seeds (by the exact surrogate 2·n·max(conv)).
3. Run a long *exponentiated‑gradient* (mirror‑descent) optimisation
   on the active constraint (sub‑gradient of the max‑convolution).
   The iterate stays on the probability simplex via a soft‑max
   parametrisation; the learning‑rate decays as 1/√t.
4. When time permits, polish the result with a short L‑BFGS continuation
   on a smooth log‑sum‑exp surrogate.
5. Finish with a simulated–annealing walk in logit space.
6. Return the best vector found; store it globally for the next run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import for L‑BFGS‑B (used only for polishing)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:          # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exact surrogate (the objective when sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(seq, seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule=None,
                    max_iter_per_beta: int = 2000) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smooth max‑conv."""
    if _opt is None:                              # pragma: no cover
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            shifted = conv - max_c
            exp_beta = np.exp(beta_val * shifted)
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (max_c + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp          # soft weights, sum = 1
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)      # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)     # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:                         # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (logit‑space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Random walk in logit space with a simple cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

        for _ in range(4):
            seeds.append(rng.dirichlet(np.full(n, conc)))

    # include the previous best if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                seeds.append(arr / arr.sum())
        except Exception:
            pass

    # normalise everything onto the probability simplex
    def _norm(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            return np.full_like(v, 1.0 / len(v))
        return v / s

    seeds = [_norm(np.asarray(v, dtype=np.float64)) for v in seeds]

    # ------------------------------------------------------------------
    # 2. Quick surrogate ranking and initial best choice
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    for seq in seeds:
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"\n[Init] best surrogate among seeds: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Heavy exponentiated‑gradient descent (mirror descent)
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    log_a = np.log(cur_seq + 1e-30)

    it = 0
    last_print = time.time()
    while time.time() < deadline:
        it += 1

        # convolution and active index
        conv = _conv_fft(cur_seq, cur_seq)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active max‑convolution entry
        g = np.zeros_like(cur_seq)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * cur_seq[rev_idx]

        # learning‑rate schedule (decays like 1/√t)
        eta = 0.5 / math.sqrt(it)

        # mirror‑descent update in logit space
        log_a -= eta * g
        cur_seq = _softmax(log_a)

        # periodic reporting & best‑track
        if it % 20000 == 0:
            cur_val = _objective_surrogate(cur_seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[ExpGrad] it={it:,} new best surrogate {best_val:.6f}")

            now = time.time()
            if now - last_print > 30.0:
                elapsed = now - start_time
                print(f"[ExpGrad] it={it:,} best={best_val:.6f} elapsed={elapsed:.1f}s")
                last_print = now

    print(f"[ExpGrad] finished after {it:,} iters, best surrogate {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional L‑BFGS polish (smooth surrogate) if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # give the L‑BFGS a modest slice of the remaining time
        max_iter_per_beta = int(min(3000, remaining * 0.02 * 1e5))
        max_iter_per_beta = max(1500, max_iter_per_beta)
        print("\n--- L‑BFGS polish ---")
        seq_polished = _lbfgs_optimize(best_seq, n, deadline,
                                       beta_schedule=[1, 5, 20, 80, 200,
                                                      500, 1_000, 2_000,
                                                      5_000, 20_000],
                                       max_iter_per_beta=max_iter_per_beta)
        val_polished = _objective_surrogate(seq_polished)
        print(f"[L‑BFGS] polished surrogate = {val_polished:.6f}")
        if val_polished < best_val:
            best_val = val_polished
            best_seq = seq_polished.copy()
            print(f"[Update] new global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(600_000, remaining * 2000))   # heuristic
        max_sa = max(200_000, max_sa)
        print("\n--- Simulated‑Annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing_opt(best_seq, n, deadline,
                                                  max_iters=max_sa)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()
            print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                       # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
