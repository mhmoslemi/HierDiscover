def _annealing_climb(start_seq,n,deadline,rng):
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val
    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = max(0.001, (1-frac) * 0.01)  # linear cooling from 0.01 to 0
        i,j = rng.integers(n, size=2)
        if i == j:
            continue
        eps = cur[i] * rng.uniform(0.001, 0.02)  # up to 2%
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta/T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val

"""
Propose a step‑function (non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is a three‑stage search:

1️⃣ **Rich seeding** – systematic sampling of many analytic families
   (Beta, power, sine‑window, Gaussian, double‑Gaussian) on a fine grid,
   together with a large number of random Dirichlet draws.  The best
   candidates (according to the exact surrogate 2·n·max(conv) when the
   vector is ℓ₁‑normalised) are kept in a small heap.

2️⃣ **Fine‑tuning** – the best candidate is refined by a continuation
   L‑BFGS‑B optimiser that minimises a smooth log‑sum‑exp surrogate of the
   max‑convolution.  The schedule of the smoothing parameter β is long
   (up to 1 000 000) and each β‑phase runs many L‑BFGS iterations,
   guaranteeing a very accurate approximation of the true objective.

3️⃣ **Local annealing** – a cheap simulated‑annealing style hill‑climb
   moves a tiny amount of mass between two coordinates, occasionally
   accepting non‑improving moves with a temperature that cools to zero.
   This squeezes out the last few digits.

The whole routine respects a user‑provided time budget (default 1000 s)
and stores the best sequence globally in ``height_sequence_1`` for warm‑starting
future runs.
"""

import math
import heapq
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a vector normalised to sum = 1:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = a0 - a1 * np.cos(2 * np.pi * i / (n - 1)) + a2 * np.cos(4 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0 - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random generators (return ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet with all concentration parameters = conc."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smoothed (log‑sum‑exp) objective
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 20000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    The variable lives in log‑softmax space (v), guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0, 1_000_000.0]

    # initialise logits
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # stays on simplex
            conv = _conv_fft(a, a)            # full convolution
            max_c = np.max(conv)
            shift = max_c                     # stabilise exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # soft‑max weights over convolution entries
            w = exp_beta / sum_exp

            # gradient wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb on the exact max‑convolution
# ----------------------------------------------------------------------
def _annealing_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Small mass‑transfer moves with occasional acceptance of worsening moves.
    Returns the best sequence found and its surrogate value.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val
    start = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = max(1e-4, (1 - frac) * 0.02)   # linear cooling, start ~0.02

        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # move a modest fraction of mass from i to j
        eps = cur[i] * rng.uniform(0.001, 0.02)
        if eps <= 0.0 or cur[i] - eps < 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence is stored globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.5

    # problem size (the constant factor of the objective does not depend on n)
    n = 4000

    # ------------------------------------------------------------------
    # heap for the best seeds (kept in min‑heap with negative values)
    # ------------------------------------------------------------------
    top_k = 20
    best_heap: list[tuple[float, np.ndarray]] = []   # stores (-value, vector)

    def _push_candidate(vec: np.ndarray):
        """Insert a candidate into the heap keeping the ``top_k`` best."""
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if val < -best_heap[0][0]:
                heapq.heappushpop(best_heap, (-val, vec))

    # ------------------------------------------------------------------
    # Deterministic analytic seeds (some with a tiny sweep)
    # ------------------------------------------------------------------
    _push_candidate(_uniform(n))
    _push_candidate(_triangular(n))
    _push_candidate(_hamming_window(n))
    _push_candidate(_blackman_window(n))
    _push_candidate(_nuttall_window(n))

    # sine window sweep
    for gamma in np.linspace(0.2, 3.0, 30):
        _push_candidate(_sine_window(n, gamma))

    # kaiser window sweep
    for beta in np.linspace(1.0, 15.0, 12):
        _push_candidate(_kaiser_window(n, beta))

    # gaussian sweep
    for sigma in np.linspace(0.07, 0.30, 25):
        _push_candidate(_gaussian(n, sigma))

    # double Gaussian combos
    for sigma in (0.08, 0.10, 0.12, 0.14):
        for shift in (0.20, 0.30, 0.40):
            _push_candidate(_double_gaussian(n, sigma, shift))

    # power law sweep
    for p in np.linspace(0.0, 2.5, 51):
        _push_candidate(_power(n, p))

    # beta‑shape sweep – a fairly dense log‑grid
    alphas = np.logspace(-0.7, 0.8, 30)      # approx 0.2 … 6.3
    betas   = np.logspace(-0.7, 0.8, 30)
    for a in alphas:
        for b in betas:
            _push_candidate(_beta_shape(n, a, b))

    # ------------------------------------------------------------------
    # Massive random seeding
    # ------------------------------------------------------------------
    conc_choices = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    random_seed_budget = 2500
    for _ in range(random_seed_budget):
        if time.time() > deadline:
            break
        conc = rng.choice(conc_choices)
        vec = _random_dirichlet(n, conc, rng)
        # random cyclic shift to break symmetry
        shift = rng.integers(n)
        if shift:
            vec = np.roll(vec, shift)
        _push_candidate(vec)

    for _ in range(300):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        shift = rng.integers(n)
        if shift:
            vec = np.roll(vec, shift)
        _push_candidate(vec)

    for nf in (2, 3, 5):
        for _ in range(200):
            if time.time() > deadline:
                break
            vec = _lowfreq_curve(n, nf, rng)
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
            _push_candidate(vec)

    # ------------------------------------------------------------------
    # Re‑use previous solution if available and of matching size
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0:
                prev_norm = prev_arr / prev_arr.sum()
                _push_candidate(prev_norm)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Extract the best candidates from the heap (ordered best → worst)
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seed phase] kept {len(seeds)} best candidates, best surrogate = "
          f"{_objective_surrogate(seeds[0]):.7f}")

    # ------------------------------------------------------------------
    # Initialise the global best
    # ------------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.7f}")

    # ------------------------------------------------------------------
    # L‑BFGS fine‑tuning – primary optimisation stage
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n--- L‑BFGS fine‑tuning (primary) ---")
        lbfgs_deadline = deadline - 5.0
        seq_lbfgs = _lbfgs_smooth_optimize(global_best_seq,
                                           n,
                                           lbfgs_deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0, 200.0,
                                                          500.0, 1000.0, 2000.0, 5000.0,
                                                          10000.0, 20000.0, 50000.0,
                                                          100000.0, 200000.0, 500000.0,
                                                          1_000_000.0],
                                           max_iter_per_beta=25000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < global_best_val:
            global_best_seq = seq_lbfgs
            global_best_val = val_lbfgs
            print(f"[L‑BFGS] refined surrogate = {global_best_val:.7f}")

    # ------------------------------------------------------------------
    # Additional local refinement – simulated annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Simulated‑annealing hill‑climb ---")
        seq_hc, val_hc = _annealing_climb(global_best_seq, n, deadline, rng)
        if val_hc < global_best_val:
            global_best_seq = seq_hc
            global_best_val = val_hc
            print(f"[Anneal] improved to {global_best_val:.7f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.7f}, "
              f"true evaluate_sequence = {true_val:.7f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
