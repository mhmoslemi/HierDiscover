def exp_gradient_opt(seq, deadline, max_iters=None):
    a = seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0

    n = len(a)
    while True:
        if max_iters is not None and it >= max_iters:
            break
        if time.time() > deadline:
            break

        # compute conv and max index
        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # gradient
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1)
        rev = kmax - idx
        g[idx] = 2.0 * a[rev]

        # step size
        it += 1
        eta = 0.5 / math.sqrt(it)  # can modify factor
        log_a -= eta * g
        a = _softmax(log_a)  # maintain sum=1

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # optional print
            if it % 200_000 == 0:
                print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

def perm_swap_opt(seq, deadline, max_swaps=None):
    cur_seq = seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    n = len(seq)
    rng = np.random.default_rng()
    it = 0
    while True:
        if max_swaps is not None and it >= max_swaps:
            break
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _objective_surrogate(cur_seq)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        it += 1
        if it % 200_000 == 0:
            print(f"[PermSwap] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val

def reallocate_opt(seq, deadline, moves=None):
    cur_seq = seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    n = len(seq)
    rng = np.random.default_rng()
    it = 0
    # parameter epsilon scaling: choose random epsilon proportion of min of a_i, a_j
    while True:
        if moves is not None and it >= moves:
            break
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount to move: random fraction of a_i
        max_move = cur_seq[i]
        if max_move <= 0:
            continue
        eps = max_move * rng.uniform(0.0, 0.5)  # move up to 50% of source mass
        cur_seq[i] -= eps
        cur_seq[j] += eps
        # renormalize? sum stays same, cause moves preserve sum exactly.
        new_val = _objective_surrogate(cur_seq)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert
            cur_seq[i] += eps
            cur_seq[j] -= eps
        it += 1
        if it % 200_000 == 0:
            print(f"[Realloc] moves={it:,} best={best_val:.6f}")

    return best_seq, best_val

        start_seq = best_vec

    else:

    # Save initial best

    global_best_val = best_val

    # Stage 1: Long exponentiated gradient descent
    # allocate half of remaining time for expgrad
    remaining = deadline - time.time()
    if remaining > 0:
        # allocate approx 60% time for gradient

        print(f"Running exponentiated gradient until {exp_deadline - start:.1f}s remaining.")

        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] Improved to {global_best_val:.6f}")

    # Stage 2: Permutation swap hill climbing
    remaining = deadline - time.time()
    if remaining > 0:
        swap_deadline = time.time() + remaining * 0.7  # use most remaining for swaps

    # Stage 3: Random reallocation moves
    remaining = deadline - time.time()
    if remaining > 1.0:

        print(f"Running random reallocation moves.")
        seq_realloc, val_realloc = reallocate_opt(global_best_seq, realloc_deadline)

            global_best_seq = seq_realloc.copy()

    # Final evaluation using official evaluate_sequence
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())

    except Exception as e:
        print("Error evaluating official:", e)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()

    ``height_sequence_1``).  If absent a diverse pool of ℓ₁‑normalised
    seeds is built and the best one is taken.

2.  A very long exponentiated‑gradient (mirror‑descent with KL‑divergence)
    is run on the exact max‑convolution objective.  The schedule
    ηₜ = 0.5/√t is used and every improvement is remembered.

3.  Two cheap local‑search refinements are applied while time permits:
    (a)  random permutation swaps that keep any improvement,
    (b)  tiny mass‑reallocation moves (move a random ε from one entry to
         another).  Both respect the probability simplex.

All operations respect the global 1000 s budget; the routine stops as
soon as the deadline is reached and returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – maps ℝⁿ → the probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:                     # should never happen, but be safe
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for ℓ₁‑normalised vectors: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generation (used only when no previous solution exists)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a large, diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 1.0, 2.0, 4.0):
        seeds.append(_sine_window(n, gamma=g))
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))
    for a in (0.3, 1.0, 2.0, 4.0):
        for b in (0.3, 1.0, 2.0, 4.0):
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise sanity‑check
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution (exponentiated‑gradient).
    Returns the best vector seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)          # stay in log‑space
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    n = len(a)
    it = 0
    last_improve = 0

    while True:
        # stop conditions
        if max_iters is not None and it >= max_iters:
            break
        if time.time() > deadline:
            break
        # compute full convolution and the active lag
        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active quadratic term
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        g = np.zeros_like(a)
        if i_end >= i_start:
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            rev = kmax - idx
            g[i_start:i_end + 1] = 2.0 * a[rev]

        # step‑size (classic ½·t⁻¹ᐟ² schedule)
        it += 1
        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)            # projection onto simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = it
            if it % 200_000 == 0:
                print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

        # simple early‑stop if no improvement for a long stretch
        if it - last_improve > 500_000:
            break

    return best_seq, best_val


def perm_swap_opt(start_seq: np.ndarray,
                  deadline: float,
                  max_swaps: int | None = None) -> tuple[np.ndarray, float]:
    """
    Randomly swap two entries; keep the swap only if the exact max‑convolution
    improves.  Returns the best sequence seen.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    n = len(cur_seq)

    it = 0
    last_improve = 0
    while True:
        if max_swaps is not None and it >= max_swaps:
            break
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # perform swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _objective_surrogate(cur_seq)

        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
                last_improve = it
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        it += 1
        if it % 100_000 == 0:
            print(f"[PermSwap] swaps={it:,} best={best_val:.6f}")

        if it - last_improve > 500_000:
            break

    return best_seq, best_val


def reallocate_opt(start_seq: np.ndarray,
                   deadline: float,
                   max_moves: int | None = None) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer moves: pick i→j and shift a random ε·a_i.
    Accept only improving moves.  Returns the best vector seen.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    n = len(cur_seq)

    it = 0
    last_improve = 0
    while True:
        if max_moves is not None and it >= max_moves:
            break
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if cur_seq[i] <= 0.0:
            continue

        # move up to 30 % of the source weight (random fraction)
        eps = cur_seq[i] * rng.uniform(0.0, 0.3)
        cur_seq[i] -= eps
        cur_seq[j] += eps

        new_val = _objective_surrogate(cur_seq)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
                last_improve = it
        else:
            # revert
            cur_seq[i] += eps
            cur_seq[j] -= eps

        it += 1
        if it % 150_000 == 0:
            print(f"[Realloc] moves={it:,} best={best_val:.6f}")

        if it - last_improve > 500_000:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects a total time budget (default 1000 s) and returns the
    best sequence found as a plain Python list.
    """
    # --------------------------------------------------------------
    # 0. Initialise RNG and timing
    # --------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 2. Load previous solution if available
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                start_vec = arr / arr.sum()
                print("Starting from previously stored solution.")
            else:
                arr = None
        except Exception:
            arr = None
    else:
        arr = None

    # --------------------------------------------------------------
    # 3. If no previous solution, generate a diverse seed pool
    # --------------------------------------------------------------
    if arr is None:
        print(f"Generating {n}‑dimensional seed pool.")
        seeds = _collect_seeds_extended(n, rng, None)
        evaluated = [(v, _objective_surrogate(v)) for v in seeds]
        evaluated.sort(key=lambda x: x[1])
        start_vec, _ = evaluated[0]
        print(f"Best seed surrogate = {_objective_surrogate(start_vec):.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop
    # --------------------------------------------------------------
    best_seq = start_vec.copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] surrogate = {best_val:.6f}")

    # ----------- Stage 1: long exponentiated‑gradient ----------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        # reserve roughly 60 % of the remaining budget for gradient descent
        grad_deadline = time.time() + remaining * 0.60
        print("Running exponentiated‑gradient descent …")
        seq_exp, val_exp = exp_gradient_opt(best_seq, grad_deadline)
        if val_exp < best_val:
            best_seq, best_val = seq_exp, val_exp
            print(f"[ExpGrad] improved to {best_val:.6f}")

    # ----------- Stage 2: permutation‑swap hill‑climber -------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        swap_deadline = time.time() + remaining * 0.70
        print("Running permutation‑swap local search …")
        seq_swap, val_swap = perm_swap_opt(best_seq, swap_deadline)
        if val_swap < best_val:
            best_seq, best_val = seq_swap, val_swap
            print(f"[PermSwap] improved to {best_val:.6f}")

    # ----------- Stage 3: tiny mass‑reallocation moves ---------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        realloc_deadline = time.time() + remaining
        print("Running random mass‑reallocation moves …")
        seq_realloc, val_realloc = reallocate_opt(best_seq, realloc_deadline)
        if val_realloc < best_val:
            best_seq, best_val = seq_realloc, val_realloc
            print(f"[Realloc] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final official evaluation
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible next round
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
