def build_A(f):
    n = len(f)
    rows = 2*n - 1
    diags = []
    offsets = []
    for d in range(n): # offset = d
        diag_len = min(n, rows - d)  # because rows = 2n-1
        diag = np.full(diag_len, f[d], dtype=np.float64)
        diags.append(diag)
    offsets = np.arange(n)  # d
    A = scipy.sparse.diags(diags, offsets, shape=(rows, n), format='csc')
    return A

"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------

1. **Seeding** – generate a modest collection of analytic windows
   (uniform, sine, Kaiser, Tukey, Gaussian, etc.) and many Dirichlet
   draws.  Keep the best few according to the exact surrogate
   2·n·max(conv) (equivalent to the true objective after ℓ₁‑normalisation).

2. **Linear‑programming “mass‑pumping”** – starting from the best seed,
   repeatedly solve the LP

        maximise   Σ b_j
        subject to  conv(f, b) ≤ M   (component‑wise)
                    b ≥ 0 ,

   where ``f`` is the current sequence, ``M = max(conv(f,f))``.
   The LP pushes as much ℓ₁‑mass as possible while respecting the
   convolution bound.  After scaling the solution back to sum = 1 we
   mix it with the current iterate (convex combination) and keep the
   improvement.  This step dramatically flattens the autocorrelation.

3. **Mirror‑descent polishing** – run exponentiated‑gradient mirror descent
   on the exact max‑convolution (sub‑gradient of the active entry).

4. **Smooth L‑BFGS refinement** – optimise a log‑sum‑exp smooth surrogate
   for the max using L‑BFGS‑B in the log‑softmax parametrisation.

5. **Local hill‑climbing** – a greedy simulated‑annealing style move that
   transfers a tiny amount of mass between two coordinates.

All phases respect the global time budget (default 1000 s).  The best
sequence found is stored in ``height_sequence_1`` for reuse.

The function returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max yielding a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with Σ a = 1:
       2·n·max(conv(a, a))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet with all concentration parameters = conc."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variable lives in log‑softmax space (v) to enforce non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimizer is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0]

    v = np.log(start_vec + 1e-30)  # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:  # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simple hill‑climbing / simulated‑annealing local refinement
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy local search that moves a tiny fraction of mass from i to j.
    Accepts only improving moves.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        eps = cur[i] * rng.uniform(0.001, 0.01)
        if eps <= 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective_surrogate(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val


# ----------------------------------------------------------------------
# Linear‑programming mass‑pumping step (requires SciPy)
# ----------------------------------------------------------------------
def _lp_mass_pump(f: np.ndarray,
                  M: float,
                  mix_coeff: float,
                  deadline: float) -> tuple[np.ndarray, float, bool]:
    """
    One LP improvement step.
    Returns (new_f, new_M, improved_flag).
    """
    if _opt is None or _sp is None:
        return f, M, False

    n = len(f)
    rows = 2 * n - 1

    # Build sparse Toeplitz matrix A where A @ b = conv(f, b)
    diag_vals = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)
    A = _sp.diags(diag_vals, offsets, shape=(rows, n), format='csc')

    # RHS of the inequality constraints
    b_ub = np.full(rows, M, dtype=np.float64)

    # Linear programme: maximise sum(b)  ⇔ minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = _opt.linprog(c,
                           A_ub=A,
                           b_ub=b_ub,
                           bounds=bounds,
                           method='highs',
                           options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
    except Exception:  # pragma: no cover
        return f, M, False

    if not res.success:
        return f, M, False

    b_opt = res.x
    sum_b = b_opt.sum()
    if sum_b <= 1.0 + 1e-12:
        # no extra mass could be injected
        return f, M, False

    # Normalise to ℓ₁‑norm = 1 and try a convex combination with f
    g = b_opt / sum_b          # Σ g = 1

    # Try mixing with decreasing coefficients until we see an improvement
    t = mix_coeff
    improved = False
    while t > 1e-4:
        f_new = (1.0 - t) * f + t * g
        conv_new = _conv_fft(f_new, f_new)
        M_new = float(np.max(conv_new))
        if M_new < M:
            improved = True
            return f_new, M_new, True
        t *= 0.5
    return f, M, False


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0        # safety margin

    # ------------------------------------------------------------------
    # Problem size – a compromise between expressive power and LP cost
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Keep a small heap of the best random seeds (surrogate value)
    # ------------------------------------------------------------------
    top_k = 7
    best_heap = []   # stores (-value, vector)

    def _push_candidate(vec: np.ndarray):
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if val < -best_heap[0][0]:
                heapq.heappushpop(best_heap, (-val, vec))

    # ------------------------------------------------------------------
    # Deterministic analytic seeds
    # ------------------------------------------------------------------
    _push_candidate(_uniform(n))
    _push_candidate(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 7):
        _push_candidate(_sine_window(n, gamma))
    _push_candidate(_hamming_window(n))
    for beta in np.linspace(1.0, 15.0, 5):
        _push_candidate(_kaiser_window(n, beta))
    for sigma in np.linspace(0.09, 0.26, 8):
        _push_candidate(_gaussian(n, sigma))
    for p in np.linspace(0.0, 2.0, 5):
        _push_candidate(_power(n, p))
    alphas = np.linspace(0.2, 5.0, 3)
    betas = np.linspace(0.2, 5.0, 3)
    for a in alphas:
        for b in betas:
            _push_candidate(_beta_shape(n, a, b))
    for alpha in np.linspace(0.1, 0.9, 5):
        _push_candidate(_tukey_window(n, alpha))

    # ------------------------------------------------------------------
    # Random Dirichlet draws (diverse concentration values)
    # ------------------------------------------------------------------
    conc_choices = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for _ in range(1500):
        if time.time() > deadline:
            break
        conc = rng.choice(conc_choices)
        vec = _random_dirichlet(n, conc, rng)
        shift = rng.integers(n)
        if shift:
            vec = np.roll(vec, shift)
        _push_candidate(vec)

    # a few exponential draws
    for _ in range(120):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        shift = rng.integers(n)
        if shift:
            vec = np.roll(vec, shift)
        _push_candidate(vec)

    # low‑frequency cosine curves
    for nf in (2, 3, 5):
        for _ in range(80):
            if time.time() > deadline:
                break
            vec = _lowfreq_curve(n, nf, rng)
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
            _push_candidate(vec)

    # ------------------------------------------------------------------
    # Preserve previous solution if compatible
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                _push_candidate(arr / arr.sum())
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Choose the best seed as starting point for the LP‑based refinement
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    f = seeds[0].copy()
    f = f / f.sum()                         # enforce ℓ₁‑norm = 1
    conv_f = _conv_fft(f, f)
    M = float(np.max(conv_f))
    best_f = f.copy()
    best_val = _objective_surrogate(f)
    print(f"\n[Seed phase] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Linear‑programming mass‑pumping iterations
    # ------------------------------------------------------------------
    lp_iters = 25
    mix_coeff = 0.5          # initial mixing factor (will be halved if needed)
    for it in range(lp_iters):
        if time.time() > deadline - 30.0:
            break
        f, M, improved = _lp_mass_pump(f, M, mix_coeff, deadline)
        if not improved:
            print(f"[LP] iteration {it}: no improvement, stopping LP phase.")
            break
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_f = f.copy()
        print(f"[LP] iter {it:2d} → surrogate = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Exponentiated‑gradient polishing (majority of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        eg_deadline = deadline - 20.0
        seq_eg, val_eg = _exp_gradient_opt(best_f,
                                           n,
                                           eg_deadline,
                                           max_iters=800_000,
                                           eta0=5.0)
        if val_eg < best_val:
            best_val = val_eg
            best_f = seq_eg
            print(f"[ExpGrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS fine‑tuning (if SciPy is available)
    # ------------------------------------------------------------------
    if _opt is not None and time.time() < deadline - 8.0:
        print("\n--- L‑BFGS fine‑tuning ---")
        seq_lbfgs = _lbfgs_smooth_optimize(best_f,
                                           n,
                                           deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                          200.0, 500.0, 1000.0,
                                                          2000.0, 5000.0,
                                                          10000.0, 20000.0,
                                                          50000.0, 100000.0,
                                                          200000.0, 500000.0],
                                           max_iter_per_beta=12000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < best_val:
            best_val = val_lbfgs
            best_f = seq_lbfgs
            print(f"[L‑BFGS] refined surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Hill‑climb ---")
        seq_hc, val_hc = _hill_climb(best_f, n, deadline, rng)
        if val_hc < best_val:
            best_val = val_hc
            best_f = seq_hc
            print(f"[Hill‑climb] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation via the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_f.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse by a subsequent call
    globals()["height_sequence_1"] = best_f.tolist()
    return best_f.tolist()
