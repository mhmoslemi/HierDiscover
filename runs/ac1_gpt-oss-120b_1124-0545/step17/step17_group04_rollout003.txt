"""
Propose a step‑function (non‑negative probability vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The strategy is deliberately different from the previous “L‑BFGS → SA”
pipeline:

1.  Build a diverse pool of analytic and random Dirichlet seeds (all
    normalised to the simplex).
2.  Keep the top few seeds (according to the exact objective) and polish
    each of them with a **large‑scale sub‑gradient descent** on the true
    max‑convolution objective.  The sub‑gradient of `max(conv)` is cheap:
    it only involves the entries that realise the maximum.
3.  The best result obtained so far is further refined by a **simulated‑annealing**
    walk in the log‑probability (softmax) space.
4.  The whole run is driven by a deadline (`budget_s` seconds).  The loops
    continuously check the remaining time and stop early if needed, so the
    function always returns a valid sequence before the timeout.

The implementation works with a modest problem size (`n = 2000`) – this
gives a good trade‑off between the quality of the bound and the number of
iterations we can afford (≈ 2 M cheap sub‑gradient steps fit easily into the
1000 s budget).  All operations stay on the probability simplex, and a
symmetrisation step (`a ← (a + a[::-1]) / 2`) is applied after each update
to keep the solution roughly symmetric, which is known to be beneficial
for autocorrelation problems.

The final sequence is stored in the global `height_sequence_1` variable for
future runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        return a / s
    return _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Objective helpers (exact, assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray, n: int) -> float:
    """Exact objective: 2·n·max(conv(a,a)).  a is assumed to sum to 1."""
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 2_000_000,
                         report_every: int = 200_000) -> tuple[np.ndarray, float]:
    """Large‑scale sub‑gradient descent (projected onto simplex, symmetrised)."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a, n)

    # momentum parameters (optional)
    m = np.zeros_like(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # convolution + max‑index
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        # keep the best seen so far
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional progress report
        if report_every and (it % report_every == 0):
            print(f"[SubGrad] iter={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]

        # simple momentum (beta=0.9)
        m = 0.9 * m + 0.1 * g

        # step‑size schedule (≈ 0.5 / sqrt(t))
        lr = 0.5 / math.sqrt(it)

        # gradient step & projection onto simplex
        a = a - lr * m
        a = _proj_simplex(a)

        # enforce (approximate) symmetry – helps flatten the autocorrelation
        a = (a + a[::-1]) * 0.5
        a = _proj_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (logit space)
# ----------------------------------------------------------------------
def _simulated_annealing(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 600_000,
                         report_every: int = 100_000) -> tuple[np.ndarray, float]:
    """Random walk in logit space, accepting uphill moves with a cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_a + 1e-30)
    cur_a = start_a.copy()
    cur_val = _objective_exact(cur_a, n)
    best_a = cur_a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear cooling
        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        prop_logits = logits + sigma * rng.normal(size=n)
        prop_a = _softmax(prop_logits)
        prop_val = _objective_exact(prop_a, n)

        if prop_val < cur_val or math.exp(-(prop_val - cur_val) / T) > rng.random():
            logits = prop_logits
            cur_a = prop_a
            cur_val = prop_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()

        if report_every and (it % report_every == 0):
            print(f"[SA] iter={it:7d} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine builds many seeds, runs a large
    sub‑gradient descent, finishes with a short simulated‑annealing
    refinement and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – 2000 gives a fast convolution yet still enough
    # degrees of freedom for a good bound.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # 1. Seed pool (rich and diverse)
    # ------------------------------------------------------------------
    seeds = []

    # analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sin2(n))
    seeds.append(_hann_window(n))
    seeds.append(_arcsine(n))
    for sigma in (0.07, 0.12, 0.20, 0.30, 0.45):
        seeds.append(_gaussian(n, sigma))
    for p in (0.0, 0.2, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))

    # random Dirichlet draws – wide concentration range
    conc_params = np.logspace(-2, 1, 14)  # 0.01 … 10
    for conc in conc_params:
        for _ in range(3):
            seeds.append(rng.dirichlet(np.full(n, conc)))

    # attempt to reuse previous global solution (if it has the same length)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0:
                # normalise to ℓ₁ = 1
                seeds.append(prev_arr / prev_arr.sum())
        except Exception:
            pass

    # normalise everything onto the simplex and clean NaNs / negatives
    clean_seeds = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0.0:
            clean_seeds.append(v / s)
        else:
            clean_seeds.append(_uniform(n))
    seeds = clean_seeds

    # ------------------------------------------------------------------
    # 2. Quick ranking – keep only the most promising seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    scored = [(vec, _objective_exact(vec, n)) for vec in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = min(12, len(scored))
    top_seeds = [vec for vec, _ in scored[:top_k]]

    print("\nBest surrogate values among raw seeds:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_exact(best_seq, n)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Sub‑gradient descent on the best seeds (majority of the budget)
    # ------------------------------------------------------------------
    # Allocate most of the remaining time to this phase.
    subgrad_deadline = deadline - 30.0  # reserve ~30 s for SA & final eval
    for idx, seed_vec in enumerate(top_seeds, 1):
        if time.time() > subgrad_deadline:
            print("[Time] deadline reached before finishing sub‑gradient passes.")
            break
        # give each seed a modest budget (e.g. up to 200 k iters)
        remaining = subgrad_deadline - time.time()
        # estimate safe iteration count (≈ 20 k iters per second)
        max_iters = int(min(200_000, remaining * 20000))
        max_iters = max(50_000, max_iters)          # at least some work
        print(f"\n--- Sub‑gradient on seed #{idx} (max_iters={max_iters}) ---")
        seq_opt, val_opt = _subgradient_descent(seed_vec, n,
                                                deadline=subgrad_deadline,
                                                max_iters=max_iters,
                                                report_every=100_000)
        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] new global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Heavy‑weight sub‑gradient descent from the current best
    # ------------------------------------------------------------------
    if time.time() < subgrad_deadline:
        remaining = subgrad_deadline - time.time()
        max_iters = int(min(1_200_000, remaining * 20000))
        max_iters = max(300_000, max_iters)
        print(f"\n--- Final heavy sub‑gradient (max_iters={max_iters}) ---")
        seq_opt, val_opt = _subgradient_descent(best_seq, n,
                                                deadline=subgrad_deadline,
                                                max_iters=max_iters,
                                                report_every=200_000)
        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Final SubGrad] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing fine‑tuning (short, final phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(800_000, remaining * 20000))
        max_sa = max(200_000, max_sa)
        print(f"\n--- Simulated‑annealing refinement (iters={max_sa}) ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, n,
                                              deadline=deadline,
                                              max_iters=max_sa,
                                              report_every=100_000)
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for possible warm‑starts in later runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
