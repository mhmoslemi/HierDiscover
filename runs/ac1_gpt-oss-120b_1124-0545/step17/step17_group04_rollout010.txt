```
v = a.copy()  # parameters

    # compute gradient at y (subgradient at max index)
    conv = _conv_fft(y, y)
    max_idx = np.argmax(conv)
    g = np.zeros_like(y)
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)
    idx = np.arange(i_start, i_end + 1)
    rev_idx = max_idx - idx
    g[idx] = 2.0 * y[rev_idx]

    # gradient step
    v_new = y - lr * g
    v_new = _proj_simplex(v_new)

    t_new = (1 + np.sqrt(1 + 4 * t_prev * t_prev)) / 2
    y = v_new + ((t_prev - 1) / t_new) * (v_new - v)
    v = v_new
    t_prev = t_new
```

```
def _run_nesterov_subgradient(start_vec, deadline, max_iters=1_000_000, base_lr=0.5):
    a = start_vec.copy()
    v = a.copy()   # for momentum (old)
    t = 1.0
    best_a = a.copy()
    best_val = _objective_exact(a)
    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break
        # compute gradient at the "y" point, which is the accelerated point
        # accelerated point y = a + ((t_prev - 1)/t) * (a - v_prev) ??? 
        # Actually Nesterov: 
        # y_k = a_k + ((t_{k-1} - 1)/t_k)*(a_k - a_{k-1})
        # We'll maintain prev_a for momentum.
        # We'll compute g at y.
        conv = _conv_fft(y, y)
        max_idx = int(np.argmax(conv))
        cur_val = 2*n*conv[max_idx]   # since sum^2=1
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # compute gradient g for y
        g = np.zeros_like(y)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * y[rev_idx]
        # step size
        lr = base_lr / np.sqrt(it)
        y_new = y - lr * g
        # project onto simplex => a_new
        a_new = _proj_simplex(y_new)
        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4*t*t))/2
        y = a_new + ((t - 1)/t_new) * (a_new - a)
        a = a_new
        t = t_new
    return best_a, best_val
```

```
def _beta_shape(n, alpha, beta):

    s = a.sum()
    if s > 0:
```

```
def _conv_fft(a, b):
    L = len(a) + len(b) - 1
    N = 1 << ((L - 1).bit_length())
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv
```

```
def _run_nesterov_subgradient(start_vec, n, deadline, max_iters=500_000, base_lr=0.5):
    a = start_vec.copy()
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_val = _objective_exact(a, n)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _auto_conv(y)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # compute gradient at y
        g = np.zeros_like(y)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end+1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * y[rev_idx]

        # step size
        lr = base_lr / np.sqrt(it)
        y_new = y - lr * g
        # project onto simplex for a_new
        a_new = _proj_simplex(y_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        a = a_new
        t = t_new
    return best_a, best_val
```

```
def _mixture_search(seeds, n, deadline, max_iters=200_000):
    best_seq = seeds[0].copy()
    best_val = _objective_exact(best_seq, n)
    top_k = len(seeds)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        m = rng.integers(2, 4)  # 2 or 3
        idx = rng.choice(top_k, size=m, replace=False)
        weights = rng.dirichlet(np.ones(m))
        cand = np.zeros(n, dtype=np.float64)
        for w, i in zip(weights, idx):
            cand += w * seeds[i]
        # already sum to 1 due to seeds sum to 1
        # ensure nonnegative (already)
        # Normalize (due to small floating errors)
        s = cand.sum()
        if s == 0:
            continue
        cand /= s
        val = _objective_exact(cand, n)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            # optionally insert into seeds list if better
            # replace worst seed if val better
            # Find worst in seeds
            worst_idx = np.argmax([_objective_exact(s, n) for s in seeds])
            seed_vals = [_objective_exact(s, n) for s in seeds]
            # Replace if better than worst
            if val < seed_vals[worst_idx]:
                seeds[worst_idx] = cand.copy()
    return best_seq, best_val
```

"""Search for a non‑negative step function minimizing evaluate_sequence.

The algorithm builds a diverse pool of analytic seed sequences, mixes them
randomly, then refines the best candidate with a Nesterov‑accelerated
sub‑gradient descent on the exact max‑convolution objective, followed by a
short exponentiated‑gradient polish and a final simulated‑annealing walk.
All operations stay on the probability simplex, so scaling does not affect
the final score.
"""
import time, math
import numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _auto_conv(a):
    n = a.shape[0]
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return conv

def _proj_simplex(v):
    """Euclidean projection of v onto the simplex {x>=0, sum=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _triangular(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _gaussian(n, sigma_frac):
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _power(n, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _arcsine(n):
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _beta_shape(n, alpha, beta):
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _objective_exact(a, n):
    """Return 2·n·max(conv(a,a))."""
    conv = _auto_conv(a)
    return float(2.0 * n * np.max(conv))

def _mix_search(seeds, n, deadline, max_iters=200_000):
    """Random convex‑combination search among the given seeds."""
    rng = np.random.default_rng()
    best_seq = seeds[0].copy()
    best_val = _objective_exact(best_seq, n)
    seed_vals = [_objective_exact(s, n) for s in seeds]

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        m = rng.integers(2, 4)          # combine 2 or 3 parents
        idx = rng.choice(len(seeds), size=m, replace=False)
        w = rng.dirichlet(np.ones(m))
        cand = np.zeros(n, dtype=np.float64)
        for weight, i in zip(w, idx):
            cand += weight * seeds[i]
        # occasional tiny random jitter
        if rng.random() < 0.05:
            cand += rng.normal(scale=0.005, size=n)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s
        val = _objective_exact(cand, n)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            # replace the worst seed if the new candidate is better
            worst = int(np.argmax(seed_vals))
            if val < seed_vals[worst]:
                seeds[worst] = cand.copy()
                seed_vals[worst] = val
    return best_seq, best_val

def _run_nesterov_subgradient(start_vec, n, deadline,
                              max_iters=500_000, base_lr=0.5):
    """Nesterov‑accelerated sub‑gradient descent on max‑convolution."""
    a = start_vec.copy()
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_val = _objective_exact(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _auto_conv(y)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # gradient of the active entry
        g = np.zeros_like(y)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * y[rev_idx]

        lr = base_lr / np.sqrt(it)
        y_new = y - lr * g
        a_new = _proj_simplex(y_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        a = a_new
        t = t_new

    return best_a, best_val

def _exp_gradient_opt(start_seq, n, deadline, max_iters=300_000):
    """Exponentiated‑gradient (mirror‑descent) polish on the exact max‑conv objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_exact(a, n)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _auto_conv(a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]

        eta = 0.5 / np.sqrt(it)
        log_a -= eta * g
        a = np.exp(log_a)
        a = a / a.sum()

        cur_val = _objective_exact(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

def _simulated_annealing_opt(start_seq, n, deadline, max_iters=200_000):
    """Simple logit‑space simulated‑annealing walk."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_exact(cur_seq, n)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.08 * (1.0 - progress) + 1e-6
        sigma = 0.12 * (1.0 - progress) + 1e-4
        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = np.exp(proposal_logits)
        proposal_seq = proposal_seq / proposal_seq.sum()
        proposal_val = _objective_exact(proposal_seq, n)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 100_000 == 0:
            print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

    # Problem size – 4096 gives a good trade‑off (FFT friendly)
    # ------------------------------------------------------------

        _triangular(n),
        _arcsine(n),

        _gaussian(n, 0.25),
        _power(n, 0.5),

"""Search for a non‑negative step function minimising ``evaluate_sequence``.

The algorithm builds a diverse pool of analytic seed sequences, mixes them
randomly, then refines the best candidate with a Nesterov‑accelerated
sub‑gradient descent on the exact max‑convolution objective, followed by a
short exponentiated‑gradient polish and a final simulated‑annealing walk.
All operations stay on the probability simplex, so scaling does not affect
the final score.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _auto_conv(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a via a real FFT (O(n log n))."""
    n = a.shape[0]
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = int(np.where(cond)[0].max())
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric (or asymmetric) Beta‑type shape on a discrete grid."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n          # values in (0,1)
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exact objective (surrogate is exact because scaling cancels)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray, n: int) -> float:
    """Return 2 · n · max(conv(a,a))."""
    conv = _auto_conv(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Random convex‑combination (mixture) search
# ----------------------------------------------------------------------
def _mix_search(seeds: list[np.ndarray],
                n: int,
                deadline: float,
                max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Explore convex combinations of the given seeds."""
    rng = np.random.default_rng()
    best_seq = seeds[0].copy()
    best_val = _objective_exact(best_seq, n)
    seed_vals = [_objective_exact(s, n) for s in seeds]

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # combine 2 or 3 parents
        m = rng.integers(2, 4)
        idx = rng.choice(len(seeds), size=m, replace=False)
        w = rng.dirichlet(np.ones(m))
        cand = np.zeros(n, dtype=np.float64)
        for wt, i in zip(w, idx):
            cand += wt * seeds[i]
        # occasional tiny jitter
        if rng.random() < 0.05:
            cand += rng.normal(scale=0.005, size=n)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s
        val = _objective_exact(cand, n)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            # replace worst seed if the new candidate is better
            worst = int(np.argmax(seed_vals))
            if val < seed_vals[worst]:
                seeds[worst] = cand.copy()
                seed_vals[worst] = val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Nesterov‑accelerated sub‑gradient descent on the exact max‑conv objective
# ----------------------------------------------------------------------
def _run_nesterov_subgradient(start_vec: np.ndarray,
                              n: int,
                              deadline: float,
                              max_iters: int = 500_000,
                              base_lr: float = 0.5) -> tuple[np.ndarray, float]:
    """Accelerated sub‑gradient method (projected onto the simplex)."""
    a = start_vec.copy()
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_val = _objective_exact(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _auto_conv(y)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # sub‑gradient of the active entry
        g = np.zeros_like(y)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * y[rev_idx]

        lr = base_lr / np.sqrt(it)
        y_new = y - lr * g
        a_new = _proj_simplex(y_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        a = a_new
        t = t_new

    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) polish
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent in log‑space on the exact max‑conv objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_exact(a, n)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _auto_conv(a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]

        eta = 0.5 / np.sqrt(it)
        log_a -= eta * g
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective_exact(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (logit space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Simple log‑it random walk with exponential cooling."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_exact(cur_seq, n)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.08 * (1.0 - progress) + 1e-6
        sigma = 0.12 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = np.exp(proposal_logits)
        proposal_seq = proposal_seq / proposal_seq.sum()
        proposal_val = _objective_exact(proposal_seq, n)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 100_000 == 0:
            print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_seq, best_val
