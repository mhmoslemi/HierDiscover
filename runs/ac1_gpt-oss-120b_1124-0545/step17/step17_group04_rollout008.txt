```python
def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _objective(seq):
    n = len(seq)
    conv = _conv_fft(seq, seq)
    maxb = conv.max()
    return float(2.0 * n * maxb / (seq.sum() ** 2))

def _softmax(v):
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0/v.size)
    return e / s

def _proj_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w
```

The program searches for a non‑negative step‑function `a = (a₀,…,a_{n‑1})` that minimises  

\[
\text{obj}(a)=\frac{2n\;\max (a*a)}{(\sum_i a_i)^2},
\]

where `*` denotes the (full) linear convolution.  
The objective is scale‑invariant, therefore the vector is kept on the probability
simplex (`∑a_i = 1`).  

The optimiser works in three stages:

1. **Diverse seed generation** – analytic families, Dirichlet draws,
   the previous best vector, and *permuted* versions of the analytic vectors
   (including the “alternating‑ends’’ ordering).  
   The best seed (by the exact objective) is kept.

2. **Large‑scale simulated‑annealing / hill‑climbing** – the current best vector
   is repeatedly mutated by either  
   * swapping two entries, or  
   * moving a random amount of mass from one entry to another.  
   A temperature schedule `T(t)` accepts worsening moves with probability
   `exp(-Δ/T)`.  All moves preserve the `ℓ₁`‑norm, so the objective reduces to
   `2·n·max(conv)`.

3. **Final L‑BFGS refinement** – the (smooth) log‑sum‑exp surrogate of the
   max‑convolution is minimised with `scipy.optimize.minimize` (L‑BFGS‑B) for a
   few hundred iterations, starting from the best vector found by annealing.

The whole search respects the 1000 s budget, leaves a safety margin for the
final evaluation, and stores the best vector in the global variable
`height_sequence_1` for reuse.

```python
import time
import numpy as np

# ------------------------------------------------------------
# Low‑level numeric utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ------------------------------------------------------------
# Ordering helpers
# ------------------------------------------------------------
def _alternating_ends(v: np.ndarray) -> np.ndarray:
    """Place the largest entries at the two ends alternately."""
    order = np.argsort(v)[::-1]   # descending
    out = np.empty_like(v)
    left, right = 0, len(v) - 1
    turn = True
    for idx in order:
        if turn:
            out[left] = v[idx]
            left += 1
        else:
            out[right] = v[idx]
            right -= 1
        turn = not turn
    return out


# ------------------------------------------------------------
# Exact objective (uses FFT for speed)
# ------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    maxb = np.max(conv)
    s = seq.sum()
    if s == 0.0:
        return np.inf
    return float(2.0 * n * maxb / (s * s))


# ------------------------------------------------------------
# Seed construction
# ------------------------------------------------------------
def _initial_seed(seed: int, n: int):
    rng = np.random.default_rng(seed)
    candidates = []

    # analytic families
    families = [
        _uniform,
        _triangular,
        lambda n: _gaussian(n, 0.12),
        lambda n: _gaussian(n, 0.25),
        lambda n: _power(n, 0.5),
        lambda n: _power(n, 1.0),
    ]

    for f in families:
        v = f(n)
        candidates.append(v)                     # original ordering
        candidates.append(v[::-1])               # reversed
        candidates.append(_alternating_ends(v))  # alternating ends
        # a random permutation
        perm = rng.permutation(n)
        candidates.append(v[perm])

    # random Dirichlet draws
    concs = np.logspace(-2, 1, 8)               # 0.01 … 10
    for conc in concs:
        for _ in range(3):
            candidates.append(rng.dirichlet(np.full(n, conc)))

    # previous best, if it matches the current size
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.size == n and arr.sum() > 0.0:
            candidates.append(arr / arr.sum())

    # normalise everything onto the simplex and clean NaNs
    clean = []
    for v in candidates:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0.0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ------------------------------------------------------------
# Simulated‑annealing / hill‑climbing optimiser
# ------------------------------------------------------------
def _anneal_opt(start_seq: np.ndarray, start_time: float,
                budget_s: int) -> tuple[np.ndarray, float]:
    """Random swaps + tiny mass moves with a simple cooling schedule."""
    n = len(start_seq)
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    deadline = start_time + budget_s - 2.0

    # Temperature parameters (linear schedule)
    T0 = 1e-3 * cur_val          # start rather low – the problem is already tight
    Tf = 1e-7 * cur_val          # final temperature close to zero

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / budget_s
        T = T0 * (1 - progress) + Tf * progress

        # 70 % swaps, 30 % tiny mass transfers
        if rng.random() < 0.7:
            # ----- swap two entries ------------------------------------------------
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            cur[i], cur[j] = cur[j], cur[i]
            new_val = _objective(cur)
            delta = new_val - cur_val
            if delta < 0 or np.exp(-delta / T) > rng.random():
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = cur.copy()
            else:
                # revert
                cur[i], cur[j] = cur[j], cur[i]
        else:
            # ----- move a small amount of mass from i to j -------------------------
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            if cur[i] == 0.0:
                continue
            # take at most 30 % of the source entry
            max_delta = cur[i] * 0.3
            delta = max_delta * rng.random()
            cur[i] -= delta
            cur[j] += delta
            new_val = _objective(cur)
            delta_obj = new_val - cur_val
            if delta_obj < 0 or np.exp(-delta_obj / T) > rng.random():
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = cur.copy()
            else:
                # revert
                cur[i] += delta
                cur[j] -= delta

    return best, best_val


# ------------------------------------------------------------
# L‑BFGS refinement on the smooth log‑sum‑exp surrogate
# ------------------------------------------------------------
def _lbfgs_refine(start_seq: np.ndarray, deadline: float) -> np.ndarray:
    """A few L‑BFGS‑B steps on the log‑sum‑exp surrogate of max(conv)."""
    try:
        import scipy.optimize as _opt
    except Exception:                         # pragma: no cover
        return start_vec

    n = len(start_seq)
    v = np.log(start_seq + 1e-30)   # log‑softmax parameters

    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        20_000.0, 50_000.0, 100_000.0,
    ]

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                # stays on the simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shifted = conv - max_c               # stabilise log‑sum‑exp
            exp_beta = np.exp(beta_val * shifted)
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (max_c + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                # soft weights, sum=1
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)           # chain rule through softmax
            return obj, grad_v

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': 2000,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except Exception:
            break

    return _softmax(v)


# ------------------------------------------------------------

# ------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Large‑scale simulated‑annealing / hill‑climbing optimiser for the
    ``evaluate_sequence`` objective.

    * Build a rich pool of analytic, random and permuted seeds (size n≈4000).
    * Keep the best seed (exact objective).
    * Run a long annealing loop that swaps entries or moves tiny amounts of mass.
    * Finish with a short L‑BFGS refinement on a smooth surrogate.
    * Return the best sequence found.
    """
    start_time = time.time()
    # keep a short safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------
    # problem size – 4000 gives a nice granularity / speed trade‑off
    # --------------------------------------------------------
    n = 4000

    # --------------------------------------------------------
    # 1. Seed generation and quick exact ranking
    # --------------------------------------------------------
    seeds = _initial_seed(seed, n)
    print("\n--- Seed generation & ranking (n={}) ---".format(n))
    best_seq = None
    best_val = np.inf
    for idx, vec in enumerate(seeds, 1):
        val = _objective(vec)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
        if idx % 20 == 0 or idx == len(seeds):
            print(f"  examined {idx}/{len(seeds)} seeds – current best {best_val:.6f}")

    print(f"\n[Init] best seed objective = {best_val:.6f}")

    # --------------------------------------------------------
    # 2. Simulated‑annealing / hill‑climbing on the best seed
    # --------------------------------------------------------
    best_seq, best_val = _anneal_opt(best_seq, start_time, budget_s)
    print(f"[Anneal] best objective after SA = {best_val:.6f}")

    # --------------------------------------------------------
    # 3. Final L‑BFGS refinement (if time permits)
    # --------------------------------------------------------
    if time.time() < deadline:
        seq_refined = _lbfgs_refine(best_seq, deadline)
        val_refined = _objective(seq_refined)
        print(f"[L‑BFGS] refined objective = {val_refined:.6f}")
        if val_refined < best_val:
            best_seq = seq_refined
            best_val = val_refined

    # 4. Final evaluation with the official evaluator
    # --------------------------------------------------------

        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible reuse by a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```
