def _random_phase_seed(base_vec, rng):
    N = len(base_vec)
    F = np.fft.rfft(base_vec)
    mag = np.abs(F)
    phi = rng.uniform(0, 2*np.pi, size=F.shape)
    phi[0] = 0.0
    if N%2==0:
        phi[-1] = 0.0  # Nyquist real
    # apply random phases to other components
    newF = mag * np.exp(1j * phi)
    a_new = np.fft.irfft(newF, n=N)
    # clip negativity, renorm
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s<=0: return base_vec.copy()
    return a_new / s

"""
search for a non‑negative step‑function (list of heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Diverse seed generation** – deterministic analytic windows (including
   DPSS and Chebyshev windows if SciPy is available), random Dirichlet draws
   with random concentration, random exponential draws, random low‑frequency
   cosine curves and random phase perturbations of known good vectors.
   All seeds are ℓ₁‑normalised.

2. **Quick refinement** – the best few seeds are improved by a short
   exponentiated‑gradient (mirror‑descent) run.  The best of these is then
   polished with a continuation L‑BFGS optimiser that uses a smooth
   log‑sum‑exp approximation of the max‑convolution.

3. **Local hill‑climbing** – a simple random‑mass‑transfer local search that
   moves a small amount of probability from a large entry to a random one.
   Moves are kept only if they improve the exact surrogate objective.

The routine respects a supplied time budget (default 1000 s) and stores the
best sequence found in the global variable ``height_sequence_1`` for reuse.
"""

import time
import math

from operator import itemgetter

# Numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective (exact surrogate for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _random_phase_seed(base_vec: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Keep the magnitude of the FFT of *base_vec* but randomise its phase."""
    N = len(base_vec)
    F = np.fft.rfft(base_vec)
    mag = np.abs(F)

    # generate random phases for interior frequencies
    phi = np.zeros_like(F, dtype=np.float64)
    phi[0] = 0.0
    if N % 2 == 0:
        phi[-1] = 0.0  # Nyquist component must stay real
        interior = slice(1, -1)
    else:
        interior = slice(1, None)

    rand_phase = rng.uniform(0.0, 2.0 * np.pi, size=phi[interior].shape)
    phi[interior] = rand_phase

    newF = mag * np.exp(1j * phi)
    a_new = np.fft.irfft(newF, n=N)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s <= 0.0:
        return base_vec.copy()
    return a_new / s


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space (v) guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        # SciPy not available – fall back to the untouched vector.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         20000.0, 50000.0, 100000.0, 200000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Updates ``log_a ← log_a – η·g`` with η = η₀ / sqrt(t).
    Returns the best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Local hill‑climbing (random mass transfer)
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_moves: int = 10_000) -> np.ndarray:
    """
    Simple local search that repeatedly moves a tiny amount of probability
    from a large entry to a random other entry, keeping moves only if they
    strictly improve the exact surrogate.
    """
    a = seq.copy()
    best_val = _objective_surrogate(a)

    moves = 0
    while time.time() < deadline and moves < max_moves:
        # pick source index proportionally to its mass (more impact)
        i = rng.choice(n, p=a)
        # pick destination uniformly among the remaining positions
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # amount we may move from i (capped to keep positivity)
        max_move = min(a[i], 0.02)   # never move more than 0.02 in one step
        if max_move <= 0.0:
            continue
        eps = rng.random() * max_move  # 0 < eps ≤ max_move

        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        val_new = _objective_surrogate(a_new)
        if val_new < best_val:
            a = a_new
            best_val = val_new
            moves += 1
            # print minimal progress occasionally
            if moves % 100 == 0:
                print(f"  hill‑climb step {moves}, improved to {best_val:.6f}")

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and stores
    the best found sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n:
                prev_arr = arr.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. Seed generation (large, diverse pool)
    # ------------------------------------------------------------------
    seeds = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in np.linspace(0.2, 3.5, 10):
        seeds.append(_sine_window(n, gamma=gamma))
    for sigma in np.linspace(0.07, 0.30, 8):
        seeds.append(_gaussian(n, sigma))
    for beta in np.linspace(2.0, 25.0, 12):
        seeds.append(_kaiser_window(n, beta))
    for alpha in np.linspace(0.05, 0.95, 9):
        seeds.append(_tukey_window(n, alpha))
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))
    for a in np.linspace(0.2, 5.0, 7):
        for b in np.linspace(0.2, 5.0, 7):
            seeds.append(_beta_shape(n, a, b))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))

    # Optional windows from SciPy
    if windows is not None:
        # DPSS (Slepian) windows – Kmax=1 gives the optimal concentration sequence
        for NW in np.linspace(2, 6, 9):
            try:
                dpss = windows.dpss(n, NW, Kmax=1, sym=True)[0]
                dpss = np.maximum(dpss, 0.0)
                dpss = dpss / dpss.sum()
                seeds.append(dpss.astype(np.float64))
            except Exception:
                pass
        # Chebyshev windows (controlled sidelobes)
        for att in [20, 40, 60, 80, 100]:
            try:
                cheb = windows.chebwin(n, at=att, sym=True).astype(np.float64)
                cheb = cheb / cheb.sum()
                seeds.append(cheb)
            except Exception:
                pass

    # Random draws
    for _ in range(30):
        conc = 10.0 ** rng.uniform(-3.0, 1.0)   # 0.001 … 10  (log‑uniform)
        seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(15):
        seeds.append(_random_exponential(n, rng))

    for _ in range(12):
        nf = rng.integers(2, 6)
        seeds.append(_lowfreq_curve(n, nf, rng))

    # Random phase perturbations of known good vectors
    base_uniform = _uniform(n)
    for _ in range(20):
        seeds.append(_random_phase_seed(base_uniform, rng))
    if prev_arr is not None:
        for _ in range(10):
            seeds.append(_random_phase_seed(prev_arr, rng))

    # Perturbations of the previous best (if any)
    if prev_arr is not None:
        seeds.append(prev_arr.copy())
        for _ in range(5):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        # all generators already return ℓ₁‑normalised vectors;
        # tiny numerical errors are harmless.
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=itemgetter(1))
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Short exponentiated‑gradient refinements on a few top seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds[:5]):   # only a handful to stay fast
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.2)
        seq_eg, val_eg = _exp_gradient_opt(seed_vec.copy(),
                                           n,
                                           sub_deadline,
                                           max_iters=80_000,
                                           eta0=6.0)
        if val_eg < global_best_val:
            global_best_seq = seq_eg.copy()
            global_best_val = val_eg
            print(f"[Seed {idx+1}] EG improved global best to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Fine tuning with continuation L‑BFGS (on the current best)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        print("\n--- L‑BFGS fine tuning on current best ---")
        seq_lbfgs = _lbfgs_smooth_optimize(global_best_seq.copy(),
                                           n,
                                           deadline,
                                           beta_schedule=[1, 5, 20, 80, 200,
                                                          500, 1000, 2000,
                                                          5000, 20000, 50000,
                                                          100000, 200000,
                                                          500000, 1000000],
                                           max_iter_per_beta=5000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < global_best_val:
            global_best_seq = seq_lbfgs.copy()
            global_best_val = val_lbfgs
            print(f"L‑BFGS improved global best to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final long EG polishing (uses remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        print("\n--- Final long EG polishing ---")
        seq_final, val_final = _exp_gradient_opt(global_best_seq.copy(),
                                                 n,
                                                 deadline,
                                                 max_iters=500_000,
                                                 eta0=5.0)
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"Long EG improved global best to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Local hill‑climbing (random mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        print("\n--- Local hill‑climbing ---")
        global_best_seq = _hill_climb(global_best_seq,
                                      n,
                                      deadline,
                                      rng,
                                      max_moves=15_000)
        global_best_val = _objective_surrogate(global_best_seq)
        print(f"After hill‑climb surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
