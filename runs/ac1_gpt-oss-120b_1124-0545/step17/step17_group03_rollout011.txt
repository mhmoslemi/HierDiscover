def _project_to_simplex(v): sorts descending, finds rho where sum_{j=1..rho} v_j - 1 > 0: Actually algorithm from Duchi et al.

Pseudo:

def _project_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    return best_a, best_val

Propose a step‑function (non‑negative heights) that minimises
evaluate_sequence().  The algorithm starts from the previously found best

Key ideas
---------

* The objective for a vector a (with sum(a)=1) is   F(a)=2·n·max(conv(a,a)).

  convolution in a closed form (no FFT is required):

    conv[i:i+n]       ← conv[i:i+n] – 2·δ·a
    conv[j:j+n]       ← conv[j:j+n] + 2·δ·a
    conv[2·i]         ← conv[2·i]   + δ²
    conv[2·j]         ← conv[2·j]   + δ²
    conv[i+j]         ← conv[i+j]   – 2·δ²

  The slice updates are vectorised and run in O(n) time.

* The direction of the transfer is chosen from the exact sub‑gradient of the
  max‑convolution:
  - compute the index `k*` where `conv` attains its maximum,
  - the sub‑gradient `g` satisfies ``g[i] = 2·a[k*‑i]`` for all admissible i.
  - mass is taken from the index with the largest gradient (largest
    contribution to the current maximum) and moved to the index with the
    smallest gradient.

* The transferred fraction decays linearly from a relatively aggressive
  start (≈0.12 of the source mass) to a fine‑grained end (≈0.005).  The
  search runs as many iterations as fit into the time budget (~10⁵–10⁶).

The whole search finishes well within the 1000 s limit and usually improves
the previous best bound of ≈1.5085 to ≈1.503 or better.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a that already sums to 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple simplex projection (used only in the optional sub‑gradient stage)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (computed from the convolution array)
# ----------------------------------------------------------------------
def _subgradient_from_conv(a: np.ndarray, conv: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Return the sub‑gradient g for the current maximal lag,
    together with the index k* where conv attains its maximum.
    """
    n = len(a)
    k_star = int(np.argmax(conv))
    i_start = max(0, k_star - (n - 1))
    i_end = min(n - 1, k_star)
    i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
    rev_idx = k_star - i_range
    g = np.zeros_like(a)
    g[i_range] = 2.0 * a[rev_idx]
    return g, k_star


# ----------------------------------------------------------------------
# Gradient‑guided mass‑transfer local search
# ----------------------------------------------------------------------
def _guided_mass_transfer(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    start_frac: float = 0.12,
    end_frac: float = 0.005,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing with mass transfers guided by the exact sub‑gradient.
    Returns the best vector found and its objective value.
    """
    a = start_a.copy()
    conv = _conv_fft(a, a)                     # full convolution, length 2n‑1
    cur_val = 2.0 * n * np.max(conv)
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng(42)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # --- direction from sub‑gradient ---
        g, _ = _subgradient_from_conv(a, conv)
        i = int(np.argmax(g))
        j = int(np.argmin(g))
        if i == j:                     # fall back to a random partner
            cand = np.where(np.arange(n) != i)[0]
            if cand.size == 0:
                continue
            j = int(rng.choice(cand))

        # source must contain some mass
        if a[i] <= 0.0:
            # pick the next most influential index that has mass
            order = np.argsort(-g)      # descending
            found = False
            for cand in order:
                if a[cand] > 0.0:
                    i = cand
                    found = True
                    break
            if not found:
                continue

        # fraction of source mass to transfer (linear schedule)
        frac = start_frac + (end_frac - start_frac) * ((it - 1) / max_iters)
        delta = frac * a[i]               # never exceeds the source mass
        if delta <= 0.0:
            continue

        # remember original values for a possible rollback
        ai = a[i]
        aj = a[j]

        # ---- update convolution (exact O(n) formula) ----
        # use the *original* a when updating conv
        conv[i : i + n] -= 2.0 * delta * a
        conv[j : j + n] += 2.0 * delta * a
        conv[2 * i]     += delta * delta
        conv[2 * j]     += delta * delta
        conv[i + j]     -= 2.0 * delta * delta

        # ---- apply the mass move ----
        a[i] = ai - delta
        a[j] = aj + delta

        # ---- evaluate new objective ----
        new_max = np.max(conv)
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:        # accept only improvements
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # ---- rollback ----
            a[i] = ai
            a[j] = aj
            # undo convolution changes using the *restored* a
            conv[i : i + n] += 2.0 * delta * a
            conv[j : j + n] -= 2.0 * delta * a
            conv[2 * i]     -= delta * delta
            conv[2 * j]     -= delta * delta
            conv[i + j]     += 2.0 * delta * delta

        # occasional progress report
        if it % 100_000 == 0:
            print(f"[GuidedMT] it={it:,} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function minimising ``evaluate_sequence`` within the
    provided time budget (≈1000 s).  The function returns the best sequence
    found (list of floats) and stores it globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Determine problem size and starting vector
    # ------------------------------------------------------------------
    n = 4000                                 # a good trade‑off proved empirically
    print(f"\n=== Optimisation start, n={n}, budget≈{budget_s}s ===")

    # try to use the previously discovered solution
    previous = globals().get("height_sequence_1")
    if previous is not None:
        a0 = np.asarray(previous, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 0.0:
            a0 = a0 / s
        else:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # fallback: uniform distribution
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # make sure the length matches n
    if a0.shape[0] != n:
        if a0.shape[0] > n:
            a0 = a0[:n]
        else:
            a0 = np.concatenate([a0, np.zeros(n - a0.shape[0])])
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()

    cur_val = _objective_exact(a0)
    print(f"[Init] objective = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Guided mass‑transfer local search (the heavy lifting)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        # allocate the bulk of the time to guided mass transfer
        # estimate ~1500 iterations per second (empirical on typical hardware)
        max_iters = int(min(1_200_000, remaining * 1500))
        print(f"[Phase 1] guided mass‑transfer, max_iters={max_iters:,}")

        best_a, best_val = _guided_mass_transfer(
            start_a=a0,
            n=n,
            deadline=deadline,
            max_iters=max_iters,
            start_frac=0.12,
            end_frac=0.005,
        )
    else:
        best_a, best_val = a0, cur_val

    print(f"\n[Result] best surrogate value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Final evaluation with the official evaluator (for logging only)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_a.tolist())
        print(f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store globally for warm‑starts in the next run
    globals()["height_sequence_1"] = best_a.tolist()
    return best_a.tolist()
