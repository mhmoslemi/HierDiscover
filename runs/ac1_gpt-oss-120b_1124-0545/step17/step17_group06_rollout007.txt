def _simulated_annealing(start_seq, n, deadline, max_iters=100000):
    a = start_seq.copy()
    # ensure sum=1
    a = a / a.sum()
    cur_max = np.max(_conv_fft(a, a))
    best_seq = a.copy()
    best_val = cur_max
    rng = np.random.default_rng()
    # schedule: linear cooling
    T0 = 0.01
    T_end = 1e-6
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # exponential cooling
        T = T0 * ((T_end / T0) ** (it / max_iters))
        src = rng.choice(n, p=a)
        src_mass = a[src]
        if src_mass <= 1e-15:
            continue
        # choose epsilon
        eps = src_mass * rng.uniform(0.001, 0.5)
        if eps <= 0:
            continue
        tgt = rng.integers(n)
        if tgt == src:
            continue
        a_new = a.copy()
        a_new[src] -= eps
        a_new[tgt] += eps
        # ensure non-negative
        # compute max conv
        max_new = np.max(_conv_fft(a_new, a_new))
        delta = max_new - cur_max
        if delta < -1e-12 or rng.random() < math.exp(-delta / (T+1e-12)):
            a = a_new
            cur_max = max_new
            if cur_max < best_val:
                best_val = cur_max
                best_seq = a.copy()
        if it % 50000 == 0:
            print(f"[SA] it={it}, best_max={best_val:.6f}")
    return best_seq, best_val

"""
Non‑negative step‑function optimiser (≈ 1000 s budget).

The search proceeds in four stages:

1. **Seed generation** – a very diverse pool of ℓ₁‑normalised vectors is
   built from analytic families (uniform, triangular, sine‑window, Gaussian,
   power‑law, Beta, Tukey, Kaiser, etc.), a few extra extreme Beta shapes,
   sparse Dirichlet draws and random low‑frequency curves.

2. **Mirrored gradient descent** – the best seed is refined by exponentiated‑
   gradient (mirror descent) on the exact max‑convolution objective,
   using a decreasing step‑size.  This gives a fast descent to a good local
   minimum.

3. **Simulated annealing** – starting from the refined candidate a simple
   mass‑transfer Markov chain explores the neighbourhood more globally.
   At each step a random source index transfers a random fraction of its
   mass to a random target; the move is accepted if it lowers the objective
   or with a temperature‑controlled probability otherwise.  The temperature
   decays exponentially.

4. **Pairwise mass‑transfer polishing** – a fast O(n) local optimiser that
   moves tiny amounts of probability mass from a “heavy’’ coordinate to a
   “light’’ one while updating the convolution incrementally.

All phases stop before the global deadline (≈ 1000 s).  The final sequence
is returned as a plain list of floats and stored in the global variable
``height_sequence_1`` for possible warm‑starts in later runs.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Low‑level utilities
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – the exact objective for ℓ₁‑normalised seq."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# -------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# -------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block of length block_frac·n."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture of two ℓ₁‑normalised vectors."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# -------------------------------------------------------------
# Seed assembly (including a few extra extreme families)
# -------------------------------------------------------------
def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_kaiser_window(n, beta=4))
    seeds.append(_kaiser_window(n, beta=8))
    seeds.append(_kaiser_window(n, beta=12))
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random families
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))
    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))
    for k_frac in (0.01, 0.03, 0.07):
        k = max(1, int(round(k_frac * n)))
        for _ in range(2):
            seeds.append(_sparse_random(n, k, rng))
    for bf in (0.05, 0.15, 0.30):
        for _ in range(2):
            seeds.append(_block_uniform(n, bf, rng))
    seeds.append(_alternating_dirichlet(n, rng))

    # Extra extreme Beta shapes (U‑shaped / highly skewed)
    for a in (0.05, 0.10, 0.15):
        for b in (3.0, 5.0, 10.0):
            seeds.append(_beta_shape(n, a, b))
    # Very sparse Dirichlet draws
    for conc in (0.02, 0.03):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Previous solution if available
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Sanitise – ensure ℓ₁‑normalised and non‑negative
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s <= 0:
            continue
        cleaned.append(v / s)
    return cleaned


# -------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser
# -------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # decreasing learning rate
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it} best={best_val:.6f}")

    return best_seq, best_val


# -------------------------------------------------------------
# Pairwise mass‑transfer local optimiser (fast polishing)
# -------------------------------------------------------------
def _mass_transfer_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Very fast local optimiser that moves tiny mass between coordinates."""
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    max_val = float(np.max(conv))
    best_a = a.copy()
    best_val = max_val

    rng = np.random.default_rng()
    step_factor = 0.25          # initial fraction of source mass to move
    no_improve = 0
    it = 0
    center = n - 1
    L = len(conv)

    while time.time() < deadline:
        if a.max() < 1e-16:
            break
        src = rng.choice(n, p=a)                # source index (biased by mass)
        src_mass = a[src]
        if src_mass <= 1e-15:
            continue
        tgt = rng.integers(n)
        if tgt == src:
            continue

        eps = step_factor * src_mass
        if eps < 1e-14:
            continue

        # ---------- tentative new convolution (O(n)) ----------
        new_conv = conv.copy()
        # linear effect of moving eps mass
        new_conv[src:src + n] -= 2.0 * eps * a
        new_conv[tgt:tgt + n] += 2.0 * eps * a
        # quadratic eps*eps correction
        new_conv[center] += 2.0 * eps * eps
        offset = tgt - src
        idx1 = center + offset
        new_conv[idx1] -= eps * eps
        if offset != 0:
            idx2 = center - offset
            new_conv[idx2] -= eps * eps

        new_max = float(np.max(new_conv))

        if new_max < max_val - 1e-16:
            a[src] = src_mass - eps
            a[tgt] += eps
            conv = new_conv
            max_val = new_max
            no_improve = 0
            if max_val < best_val:
                best_val = max_val
                best_a = a.copy()
        else:
            no_improve += 1

        it += 1
        if no_improve > 5000:
            step_factor *= 0.5
            if step_factor < 1e-4:
                step_factor = 0.25
            no_improve = 0

        if it % 20000 == 0:
            print(f"[MassTransfer] it={it} best max={best_val:.12f}")

    return best_a, 2.0 * n * best_val


# -------------------------------------------------------------
# Simulated‑annealing mass‑transfer optimiser
# -------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Global optimiser that repeatedly moves a random fraction of mass from a
    source index (chosen proportionally to its weight) to a random target.
    Moves are accepted according to a simple Metropolis rule with an
    exponentially decaying temperature schedule.
    Returns (best_sequence, best_max_conv).
    """
    a = start_seq.copy()
    a = a / a.sum()
    cur_max = float(np.max(_conv_fft(a, a)))
    best_seq = a.copy()
    best_val = cur_max

    rng = np.random.default_rng()
    T0 = 0.02          # start temperature
    T_end = 1e-6      # final temperature

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = T0 * ((T_end / T0) ** (it / max_iters))

        # pick source (biased by current mass)
        src = rng.choice(n, p=a)
        src_mass = a[src]
        if src_mass <= 1e-15:
            continue

        # choose a random fraction of the source mass to move
        eps = src_mass * rng.uniform(0.001, 0.4)   # 0.1 % – 40 % of src
        if eps <= 0.0:
            continue

        tgt = rng.integers(n)
        if tgt == src:
            continue

        a_new = a.copy()
        a_new[src] -= eps
        a_new[tgt] += eps

        new_max = float(np.max(_conv_fft(a_new, a_new)))
        delta = new_max - cur_max

        accept = False
        if delta < -1e-12:
            accept = True
        else:
            prob = math.exp(-delta / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_max = new_max
            if cur_max < best_val:
                best_val = cur_max
                best_seq = a.copy()

        if it % 50_000 == 0:
            print(f"[SA] it={it} best_max={best_val:.12f}")

    return best_seq, best_val


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and timing
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Warm‑start from previous run (if any)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 3. Seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 4. Exponentiated‑gradient refinement (exact max)
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    remaining = deadline - time.time()
    if remaining > 10:
        max_iters = int(min(600_000, remaining * 2000))  # rough estimate
        print(f"[ExpGrad] running up to {max_iters} iterations")
        seq_exp, val_exp = _exp_gradient_opt(best_seq, n, deadline, max_iters=max_iters)
        if val_exp < best_val:
            best_seq = seq_exp.copy()
            best_val = val_exp
            print(f"[ExpGrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Simulated‑annealing global search
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5:
        max_iters_sa = int(min(2_000_000, remaining * 3000))
        print(f"[SA] running up to {max_iters_sa} iterations")
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline, max_iters=max_iters_sa)
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"[SA] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Pairwise mass‑transfer polishing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5:
        print("\n--- Mass‑transfer local optimisation ---")
        seq_mt, val_mt = _mass_transfer_opt(best_seq, n, deadline)
        if val_mt < best_val:
            best_seq = seq_mt.copy()
            best_val = val_mt
            print(f"[MassTransfer] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
