def _hill_climb(seq, n, deadline, max_moves=20000, step_scale=0.01):
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)
    cur_seq = seq.copy()
    cur_val = best_val
    for move in range(max_moves):
        if time.time() > deadline:
            break
        # random generate move
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # delta = step_scale * cur_seq[i]
        delta = step_scale * cur_seq[i] * rng.random()
        if delta <= 0:
            continue
        # ensure not exceed i's value
        if delta > cur_seq[i]:
            delta = cur_seq[i]
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # no need to re-normalize sum remains same.
        # ensure nonnegative already.
        val = _objective_surrogate(new_seq)
        if val < cur_val:
            cur_seq = new_seq
            cur_val = val
            if val < best_val:
                best_seq = new_seq
                best_val = val
        # else optionally accept with some probability.
    return best_seq, best_val
```

def _logistic_shape(n, mu_frac, scale):
    i = np.arange(n, dtype=np.float64)

    a = 1.0/(1.0 + np.exp((i - mu)/scale))

    a = np.maximum(a, 0)

"""
propose_candidate – high‑precision optimisation of a non‑negative step function

Goal
----
Minimise `evaluate_sequence(seq)` which (for an ℓ₁‑normalised vector) equals  

    2·n·max(conv(seq,seq))

where `conv` is the full linear convolution.
The objective is non‑smooth; we use a blend of:
    * broad parametric seeding,
    * a smooth log‑sum‑exp surrogate with L‑BFGS‑B,
    * stochastic hill‑climbing (small mass transfers),
    * a final exponentiated‑gradient (mirror‑descent) polishing.

The routine respects a total wall‑clock budget (default 1000 s) and stores the
best sequence in the global variable ``height_sequence_1`` for reuse.

All helpers are top‑level (no closures) and the code works with only numpy,
scipy (optional) and the supplied ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:   # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lamb: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lamb * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, mu_frac: float, scale: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mu = mu_frac * (n - 1)
    a = 1.0 / (1.0 + np.exp((i - mu) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _alternating_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    l, r = 0, n - 1
    for idx, v in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[l] = v
            l += 1
        else:
            a[r] = v
            r -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables are logits `v` (soft‑max guarantees non‑negativity and the simplex).
    """
    if _opt is None:
        # SciPy not available – simply return the input.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         20000.0, 50000.0, 100000.0,
                         200000.0, 500000.0, 1e6]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        b = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth approximation of max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(b * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / b)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            # gradient w.r.t. a (full convolution length = 3n‑2)
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={b}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Performs `max_iters` updates or stops when the time budget expires.
    Returns the best sequence found.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (see original code)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (mass transfer)
# ----------------------------------------------------------------------
def _hill_climb_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    max_moves: int = 200_000,
                    init_step: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Repeatedly transfer a small amount of probability mass from one index to
    another, accepting the move only if the surrogate improves.
    """
    seq = start_seq.copy()
    best_seq = seq.copy()
    best_val = _objective_surrogate(seq)
    cur_val = best_val
    step = init_step
    moves = 0

    while time.time() < deadline and moves < max_moves:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # propose moving a fraction of a[i] to a[j]
        delta = step * seq[i] * rng.random()
        if delta <= 0.0:
            continue
        if delta > seq[i]:
            delta = seq[i]

        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # sum is unchanged, non‑negativity guaranteed by construction
        val = _objective_surrogate(new_seq)
        moves += 1

        if val < cur_val:
            seq = new_seq
            cur_val = val
            if val < best_val:
                best_seq = new_seq.copy()
                best_val = val
                # when we improve, gradually shrink the step to fine‑tune
                step = max(step * 0.9, 1e-5)
        else:
            # occasional exploratory moves (simulated‑annealing style)
            # temperature decays linearly with the number of performed moves
            temp = max(0.0, 1.0 - moves / max_moves)
            if rng.random() < temp * 0.01:
                seq = new_seq
                cur_val = val
                step = min(step * 1.05, 0.1)

    return best_seq, best_val

    # random exponential draws
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # alternating Dirichlet (front‑back scattering)
    for conc in (0.05, 0.1, 0.2, 0.5):
        seeds.append(_alternating_dirichlet(n, conc, rng))

    # include the previous best (if any) and a few perturbations
    if prev_seq is not None:
        seeds.append(prev_seq)
        for _ in range(6):
            seeds.append(_perturb(prev_seq, rng, sigma=0.002))

    # ------------------------------------------------------------------
    # 4. Cheap surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    if not evaluated:
        raise RuntimeError("No valid seeds were generated.")

    global_best_seq = evaluated[0][0].copy()
    global_best_val = evaluated[0][1]

    print(f"\nInitial best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Deep optimisation of the current best seed
    # ------------------------------------------------------------------
    # allocate the bulk of the remaining budget to a high‑precision run
    remaining_time = deadline - time.time()
    sub_deadline = time.time() + remaining_time * 0.75

    # 5a – L‑BFGS with an extended β‑schedule
    seq_lbfgs = _lbfgs_smooth_optimize(global_best_seq, n,
                                       sub_deadline,
                                       beta_schedule=[1, 5, 20, 80, 200, 500,
                                                      1000, 2000, 5000, 20000,
                                                      50000, 100000, 200000,
                                                      500000, 1e6],
                                       max_iter_per_beta=6000)
    val_lbfgs = _objective_surrogate(seq_lbfgs)
    if val_lbfgs < global_best_val:
        global_best_seq, global_best_val = seq_lbfgs, val_lbfgs
        print(f"[L‑BFGS] improved surrogate = {global_best_val:.6f}")

    # 5b – Stochastic hill‑climbing local refinement
    if time.time() < deadline:
        print("\n--- Hill‑climbing refinement ---")
        seq_hc, val_hc = _hill_climb_opt(global_best_seq, n,
                                         deadline, rng,
                                         max_moves=200_000,
                                         init_step=0.02)
        if val_hc < global_best_val:
            global_best_seq, global_best_val = seq_hc, val_hc
            print(f"[Hill‑climb] improved surrogate = {global_best_val:.6f}")

    # 5c – Final exponentiated‑gradient polishing (use remaining time)
    if time.time() < deadline:
        print("\n--- Final EG polishing ---")
        seq_final, val_final = _exp_gradient_opt(global_best_seq, n,
                                                 deadline,
                                                 max_iters=500_000,
                                                 eta0=5.0)
        if val_final < global_best_val:
            global_best_seq, global_best_val = seq_final, val_final
            print(f"[Final EG] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation with the provided black‑box
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
