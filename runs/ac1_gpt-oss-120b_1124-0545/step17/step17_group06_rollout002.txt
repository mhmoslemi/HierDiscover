def _guided_mass_transfer_opt(start_seq, n, deadline):
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    max_val = float(np.max(conv))
    best_a = a.copy()
    best_val = max_val

    rng = np.random.default_rng()
    step_factor = 0.5
    no_improve = 0
    it = 0
    while time.time() < deadline:
        # find current max index (recompute if conv changed)
        max_idx = int(np.argmax(conv))
        # subgradient g_i = 2*a[rev_idx] for applicable positions
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        grad = np.zeros_like(a)
        grad[idx_i] = 2.0 * a[rev_idx]

        # source weighting = a * grad (emphasize large contribution)
        src_weight = a * grad
        total_src_weight = src_weight.sum()
        if total_src_weight <= 0:
            src = rng.choice(n, p=a)  # fallback
        else:
            src = rng.choice(n, p=src_weight/total_src_weight)

        # target weighting = 1/(grad+small) but also encourage low a
        inv_grad = 1.0/(grad + 1e-12)
        tgt_weight = inv_grad * (1.0 - a)  # prefer low a also low grad
        # ensure positivity
        tgt_weight = np.maximum(tgt_weight, 0.0)
        # if all zero, choose random
        wsum = tgt_weight.sum()
        if wsum <= 0:
            tgt = rng.integers(n)
        else:
            tgt = rng.choice(n, p=tgt_weight/wsum)

        if src == tgt:
            continue

        src_mass = a[src]
        if src_mass <= 0:
            continue

        eps = step_factor * src_mass
        if eps < 1e-14:
            continue

        # compute new conv as in original
        new_conv = conv.copy()
        new_conv[src:src + n] -= 2.0 * eps * a
        new_conv[tgt:tgt + n] += 2.0 * eps * a
        # quadratic term
        center = n - 1
        new_conv[center] += 2.0 * eps * eps
        offset = tgt - src
        idx1 = center + offset
        new_conv[idx1] -= eps * eps
        if offset != 0:
            idx2 = center - offset
            new_conv[idx2] -= eps * eps

        new_max = float(np.max(new_conv))

        if new_max < max_val - 1e-16:
            # accept
            a[src] = src_mass - eps
            a[tgt] += eps
            conv = new_conv
            max_val = new_max
            no_improve = 0
            if max_val < best_val:
                best_val = max_val
                best_a = a.copy()
        else:
            no_improve += 1

        it += 1
        if no_improve > 5000:
            step_factor = max(step_factor * 0.5, 1e-4)
            no_improve = 0

        if it % 20000 == 0:
            print(f"[GuidedMassTransfer] it={it} best max={best_val:.12f}")

    return best_a, 2.0 * n * best_val

"""
Non‑negative step‑function optimiser (improved).

The optimiser builds a diverse pool of analytic/random seeds,
optimises each seed with a continuation L‑BFGS on a smooth‑max
surrogate, refines the best seed with an exponentiated‑gradient
mirror‑descent and finally performs a *guided* pairwise mass‑transfer
local search that uses the exact sub‑gradient of the max‑convolution
to pick source/target coordinates.  The whole pipeline respects the
global 1000 s deadline and returns the best sequence it found.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# -------------------------------------------------------------
try:

except Exception:  # pragma: no cover

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate of the official objective for ℓ₁‑normalised vectors:
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block of length block_frac·n."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a large, diverse pool of ℓ₁‑normalised vectors.
    """
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    seeds.append(_double_gaussian(n, sigma_frac=0.10, shift_frac=0.30))
    seeds.append(_double_gaussian(n, sigma_frac=0.15, shift_frac=0.25))

    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(10):
        seeds.append(_random_exponential(n, rng))

    for nf in (2, 3, 4, 5):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    for k in (int(0.01 * n), int(0.03 * n), int(0.07 * n)):
        for _ in range(4):
            seeds.append(_sparse_random(n, k, rng))

    for bf in (0.05, 0.15, 0.30):
        for _ in range(4):
            seeds.append(_block_uniform(n, bf, rng))

    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    seeds.append(_alternating_dirichlet(n, rng))

    # previous solution, if compatible
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # sanitise
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# -------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# -------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns a probability vector (ℓ₁‑normalised).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 1e5,
            2e5, 5e5, 1e6
        ]

    v = np.log(start_vec + 1e-30)   # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# -------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent)
# -------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it} best={best_val:.6f}")

    return best_seq, best_val


# -------------------------------------------------------------
# Guided pairwise mass‑transfer local optimiser
# -------------------------------------------------------------
def _guided_mass_transfer_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Fast local search that moves a tiny amount of mass from a
    high‑gradient coordinate to a low‑gradient one, using the exact
    sub‑gradient of the max‑convolution to guide the choice.
    Returns (best_sequence, best_true_value).
    """
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    max_val = float(np.max(conv))
    best_a = a.copy()
    best_val = max_val

    rng = np.random.default_rng()
    step_factor = 0.5          # initial fraction of source mass to move
    no_improve = 0
    it = 0
    center = n - 1

    while time.time() < deadline:
        # recompute the active index (may have changed)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry (as used in ExpGrad)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        grad = np.zeros_like(a)
        grad[idx_i] = 2.0 * a[rev_idx]

        # source weighting: large mass * large gradient
        src_weight = a * grad
        ws = src_weight.sum()
        if ws <= 0:
            src = rng.choice(n, p=a)
        else:
            src = rng.choice(n, p=src_weight / ws)

        # target weighting: prefer low gradient and low current mass
        inv_grad = 1.0 / (grad + 1e-12)
        tgt_weight = inv_grad * (1.0 - a)
        ws_t = tgt_weight.sum()
        if ws_t <= 0:
            tgt = rng.integers(n)
        else:
            tgt = rng.choice(n, p=tgt_weight / ws_t)

        if src == tgt:
            continue

        src_mass = a[src]
        if src_mass <= 0:
            continue

        eps = step_factor * src_mass
        if eps < 1e-14:
            continue

        # ---- compute tentative new convolution (O(n)) ----
        new_conv = conv.copy()
        # linear part
        new_conv[src:src + n] -= 2.0 * eps * a
        new_conv[tgt:tgt + n] += 2.0 * eps * a
        # quadratic correction
        new_conv[center] += 2.0 * eps * eps
        offset = tgt - src
        idx1 = center + offset
        new_conv[idx1] -= eps * eps
        if offset != 0:
            idx2 = center - offset
            new_conv[idx2] -= eps * eps

        new_max = float(np.max(new_conv))

        # ---- accept if improvement ----
        if new_max < max_val - 1e-16:
            a[src] = src_mass - eps
            a[tgt] += eps
            conv = new_conv
            max_val = new_max
            no_improve = 0
            if max_val < best_val:
                best_val = max_val
                best_a = a.copy()
        else:
            no_improve += 1

        it += 1
        if no_improve > 5000:
            step_factor = max(step_factor * 0.5, 1e-4)
            no_improve = 0

        if it % 20000 == 0:
            print(f"[GuidedMass] it={it} best max={best_val:.12f}")

    return best_a, 2.0 * n * best_val

# Main entry point
# -------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size (can be tuned)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Seed generation (including previous solution, if any)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 6                     # keep a small elite set
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS on elite seeds
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        10000.0, 20000.0, 50000.0, 1e5,
        2e5, 5e5, 1e6
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline before finishing L‑BFGS runs.")
            break

        # give each L‑BFGS run a fraction of the remaining budget
        remaining = deadline - time.time()
        per_run = remaining * 0.12          # roughly 12 % per seed
        max_iter = int(min(15000, per_run * 1e5))
        max_iter = max(5000, max_iter)

        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter={max_iter}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] new global best surrogate = {best_val:.6f}")

        # --------------------------------------------------------
        # Short exponentiated‑gradient polishing
        # --------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_exp = int(min(400_000, remaining * 0.07 * 1e5))
            max_exp = max(200_000, max_exp)
            seq_exp, val_exp = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_exp)
            if val_exp < best_val:
                best_seq = seq_exp.copy()
                best_val = val_exp
                print(f"[ExpGrad] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Guided mass‑transfer local optimisation (purely exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Guided mass‑transfer local optimisation ---")
        seq_mt, val_mt = _guided_mass_transfer_opt(best_seq, n, deadline)
        if val_mt < best_val:
            best_seq = seq_mt.copy()
            best_val = val_mt
            print(f"[GuidedMass] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
