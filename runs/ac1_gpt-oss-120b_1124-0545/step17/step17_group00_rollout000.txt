def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a vector normalised to sum = 1:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _alternating_pattern(n: int, even_frac: float) -> np.ndarray:
    """
    Two‑level alternating pattern.
    `even_frac` is the total mass placed on the even positions.
    """
    even_cnt = (n + 1) // 2
    odd_cnt = n // 2
    a = np.empty(n, dtype=np.float64)
    a[0::2] = even_frac / even_cnt
    if odd_cnt > 0:
        a[1::2] = (1.0 - even_frac) / odd_cnt
    else:  # n == 1
        a[0] = 1.0
    return a

def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()

# Exponentiated‑gradient optimiser (mirror‑descent on the exact max‑conv)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 10_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Stops automatically when the time budget expires.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑softmax space to keep the simplex constraint
    log_a = np.log(a + 1e-30)

    # optional momentum (helps a little in practice)
    momentum = np.zeros_like(log_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]          # derivative of a[i] * a[rev_idx]

        # learning‑rate schedule ηₜ = η₀ / √t
        eta = eta0 / math.sqrt(it)

        # simple heavy‑ball momentum
        momentum = 0.9 * momentum - eta * g

        log_a = log_a + momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer moves, accepting worsening moves with a decaying
    temperature.  Returns (best_seq, best_surrogate_value).
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    # start temperature – a tiny fraction of the current value
    T0 = cur_val * 1e-5
    decay = 0.9999
    step = 0

    while time.time() < deadline:
        step += 1
        i, j = rng.integers(n, size=2)
        if i == j:
            continue

        # move a random fraction (0.1 % – 0.5 %) of the mass from i to j
        eps = cur[i] * rng.uniform(0.001, 0.005)
        if eps <= 0.0:
            continue

        new = cur.copy()
        new[i] -= eps
        new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val

        accept = False
        if delta < 0.0:
            accept = True
        else:
            # temperature schedule
            T = T0 * (decay ** step)
            if T > 1e-12 and rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the time budget and stores the best found sequence in
    the global variable ``height_sequence_1`` for reuse.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – a trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Keep the top‑K seeds according to the exact surrogate (2·n·max_conv)
    # ------------------------------------------------------------------
    top_k = 20
    best_heap = []                     # stores (-value, vector)

    def _push_candidate(vec: np.ndarray):
        """Insert a candidate into the heap, keeping the best ``top_k``."""
        if vec.sum() == 0.0:
            return
        norm = vec / vec.sum()
        val = _objective_surrogate(norm)
        if len(best_heap) < top_k:
            heapq.heappush(best_heap, (-val, norm))
        else:
            if val < -best_heap[0][0]:
                heapq.heappushpop(best_heap, (-val, norm))

    # ------------------------------------------------------------------
    # Deterministic analytic seeds (dense grid)
    # ------------------------------------------------------------------
    _push_candidate(_uniform(n))
    _push_candidate(_triangular(n))

    for gamma in np.linspace(0.2, 4.0, 12):
        _push_candidate(_sine_window(n, gamma))

    for sigma in np.linspace(0.07, 0.30, 12):
        _push_candidate(_gaussian(n, sigma))

    for p in np.linspace(0.0, 2.5, 14):
        _push_candidate(_power(n, p))

    for a in np.linspace(0.2, 4.0, 7):
        for b in np.linspace(0.2, 4.0, 7):
            _push_candidate(_beta_shape(n, a, b))

    for alpha in np.linspace(0.05, 0.9, 9):
        _push_candidate(_tukey_window(n, alpha))

    for even_frac in np.linspace(0.30, 0.70, 9):
        _push_candidate(_alternating_pattern(n, even_frac))

    # ------------------------------------------------------------------
    # Massive random seeding
    # ------------------------------------------------------------------
    conc_options = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    random_seed_budget = 6000   # increase a bit compared with the baseline
    for _ in range(random_seed_budget):
        if time.time() >= deadline:
            break
        conc = rng.choice(conc_options)
        vec = _random_dirichlet(n, conc, rng)
        # random cyclic shift – changes correlation with the reversed version
        shift = rng.integers(n)
        if shift:
            vec = np.roll(vec, shift)
        _push_candidate(vec)

    # a few exponential draws (heavier tails)
    for _ in range(500):
        if time.time() >= deadline:
            break
        vec = _random_exponential(n, rng)
        shift = rng.integers(n)
        if shift:
            vec = np.roll(vec, shift)
        _push_candidate(vec)

    # low‑frequency random curves (smooth shapes)
    for nf in (2, 3, 5):
        for _ in range(300):
            if time.time() >= deadline:
                break
            vec = _lowfreq_curve(n, nf, rng)
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
            _push_candidate(vec)

    # ------------------------------------------------------------------
    # Re‑use the previous solution if it matches the size
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0.0:
                _push_candidate(prev_arr / prev_arr.sum())
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Extract the best seeds (ordered from best to worst)
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seed phase] kept {len(seeds)} best candidates, best surrogate = {_objective_surrogate(seeds[0]):.6f}")

    # ------------------------------------------------------------------
    # Global best tracker
    # ------------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Short refinement of the top few seeds (tiny EG runs) – quick
    # ------------------------------------------------------------------
    # Reserve roughly 15 % of the remaining time for this cheap polishing
    time_now = time.time()
    time_left = deadline - time_now
    cheap_budget = time_left * 0.15
    per_seed_deadline = time_now + cheap_budget / min(len(seeds), 5)

    cheap_iters = 200_000
    for idx, seed_vec in enumerate(seeds[:5]):
        sub_deadline = min(deadline - 0.5, per_seed_deadline)
        seq, val = _exp_gradient_opt(seed_vec, n, sub_deadline,
                                     max_iters=cheap_iters,
                                     eta0=5.0)
        if val < global_best_val:
            global_best_seq = seq
            global_best_val = val
            print(f"[Cheap EG {idx+1}] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Main exponentiated‑gradient polishing – consumes the majority of time
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        eg_deadline = deadline - 5.0  # leave a little for final anneal
        print("\n--- Main exponentiated‑gradient polishing ---")
        seq_eg, val_eg = _exp_gradient_opt(global_best_seq,
                                           n,
                                           eg_deadline,
                                           max_iters=10_000_000,
                                           eta0=5.0)
        if val_eg < global_best_val:
            global_best_seq = seq_eg
            global_best_val = val_eg
            print(f"[Main EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional L‑BFGS refinement (smooth max via log‑sum‑exp)
    # ------------------------------------------------------------------
    if _opt is not None and time.time() < deadline - 5.0:
        print("\n--- Optional L‑BFGS refinement (smooth max) ---")
        # we reuse the helper from the baseline (kept simple here)
        def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                                   n: int,
                                   deadline_lb: float) -> np.ndarray:
            v = np.log(start_vec + 1e-30)

            beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                             500.0, 1000.0, 2000.0, 5000.0,
                             10_000.0, 20_000.0, 50_000.0,
                             100_000.0, 200_000.0, 500_000.0]

            for beta in beta_schedule:
                if time.time() >= deadline_lb:
                    break
                beta_val = float(beta)

                def fun(v_local):
                    a = _softmax(v_local)
                    conv = _conv_fft(a, a)
                    max_c = np.max(conv)
                    shift = max_c
                    exp_beta = np.exp(beta_val * (conv - shift))
                    sum_exp = exp_beta.sum()
                    obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

                    w = exp_beta / sum_exp
                    rev_a = a[::-1]
                    full = _conv_fft(rev_a, w)          # length 3n‑2
                    grad_a = 2.0 * full[n - 1:n - 1 + n]

                    dot = np.dot(grad_a, a)
                    grad_v = a * (grad_a - dot)
                    return obj, grad_v

                def callback(_):
                    if time.time() >= deadline_lb:
                        raise KeyboardInterrupt()

                try:
                    res = _opt.minimize(fun=fun,
                                        x0=v,
                                        method='L-BFGS-B',
                                        jac=True,
                                        callback=callback,
                                        options={'maxiter': 15000,
                                                 'disp': False,
                                                 'ftol': 1e-12})
                    v = res.x
                except KeyboardInterrupt:
                    break
                except Exception as exc:
                    print(f"[L‑BFGS] exception {exc} at β={beta_val}")
                    break
            return _softmax(v)

        lbfgs_deadline = deadline - 3.0
        seq_lbfgs = _lbfgs_smooth_optimize(global_best_seq, n, lbfgs_deadline)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < global_best_val:
            global_best_seq = seq_lbfgs
            global_best_val = val_lbfgs
            print(f"[L‑BFGS] refined surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing hill‑climb (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Final simulated‑annealing hill‑climb ---")
        seq_hc, val_hc = _simulated_annealing(global_best_seq,
                                              n,
                                              deadline,
                                              rng)
        if val_hc < global_best_val:
            global_best_seq = seq_hc
            global_best_val = val_hc
            print(f"[Anneal‑climb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
