def _random_hill_climb(seq, best_val, n, deadline, rng):
   a = seq.copy()
   best_seq = a.copy()
   best = best_val
   # we can adapt epsilon schedule: start with step = 0.01 and gradually reduce.
   # We'll try for each iteration:
   # sample epsilon as rng.exponential(scale=step) truncated to <= a[i].
   # optionally adapt step based on time.
   iteration = 0
   while time.time() < deadline:
       iteration+=1
       # Choose donor index weighted by a
       i = rng.choice(n, p=a)  # probability proportional to a_i, donors more likely
       # Choose receiver index uniformly not equal i
       j = rng.integers(0, n-2)
       if j >= i:

def _random_hill_climb(seq, best_val, n, deadline, rng):
    a = seq.copy()
    best_seq = a.copy()
    best = best_val
    iteration=0
    while time.time() < deadline:
        iteration+=1
        # we can gradually decrease step size:
        # define base_step = 0.1 (proportion of donor mass)
        # maybe epsilon = (rng.random() ** 2) * 0.1 * a[i]  # smaller steps
        i = rng.choice(n, p=a)  # donor index weighted by a
        # If a[i] is zero, skip
        if a[i] <= 1e-12:
            continue
        # pick receiver index uniformly not equal i
        j = rng.integers(0, n-2)
        if j >= i:
            j += 1
        max_move = a[i] * 0.4  # at most 40%
        if max_move <= 0:
            continue
        epsilon = rng.random() * max_move
        # create new
        new = a.copy()
        new[i] -= epsilon
        new[j] += epsilon
        # ensure nonnegative
        if new[i] < 0:
            continue
        # Evaluate surrogate
        val = _objective_surrogate(new)
        if val < best:
            best = val
            best_seq = new
            a = new  # move to new location
    return best_seq, best

    def __call__(self, v):

        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        shift = max_c
        exp_beta = np.exp(self.beta * (conv - shift))
        sum_exp = np.sum(exp_beta)
        obj = 2.0 * self.n * (shift + np.log(sum_exp) / self.beta)
        w = exp_beta / sum_exp
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[self.n - 1: self.n - 1 + self.n]
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return obj, grad_v

"""
Search for a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm

1. builds a large diverse pool of analytically‑parameterised seeds
   (windows, power laws, beta‑shapes, random Dirichlet / exponential draws,
   low‑frequency cosine curves and mixtures);
2. evaluates the cheap surrogate 2·n·max(conv) for all seeds and keeps the
   best few;
3. refines each top seed with an exponentiated‑gradient (mirror‑descent)
   pass on the exact max‑convolution, followed by a smooth‑max L‑BFGS‑B
   optimisation (log‑sum‑exp surrogate);
4. runs a simple hill‑climbing random‑move local search on the current
   best sequence;
5. finishes with a long exponentiated‑gradient polishing run.

All steps respect the global time budget (default 1000 s) and the best
sequence is stored in the global variable ``height_sequence_1`` for the
next invocation.
"""

import time
import math
import numpy as np

# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------

    import scipy.optimize as _opt
except Exception:          # pragma: no cover

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (surrogate) for ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector normalised to sum = 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Returns the best sequence found and its surrogate value."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    it = 0
    while it < max_iters:
        it += 1
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser
# ----------------------------------------------------------------------
class _SmoothMaxObj:
    """Callable used by SciPy L‑BFGS‑B; returns smoothed max‑conv objective
    and its gradient w.r.t. the log‑softmax parameters."""
    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, v: np.ndarray):
        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        shift = max_c
        exp_beta = np.exp(self.beta * (conv - shift))
        sum_exp = np.sum(exp_beta)
        obj = 2.0 * self.n * (shift + np.log(sum_exp) / self.beta)

        w = exp_beta / sum_exp
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """Smooth‑max optimisation via L‑BFGS‑B in the log‑softmax space."""
    if _opt is None:
        return _softmax(start_vec)

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         20000.0, 50000.0, 100000.0, 200000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        obj_fun = _SmoothMaxObj(n, beta)
        try:
            res = _opt.minimize(fun=obj_fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Helper: random perturbation of a vector
# ----------------------------------------------------------------------
def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Helper: simple hill‑climbing random‑move local search
# ----------------------------------------------------------------------
def _random_hill_climb(seq: np.ndarray,
                       best_val: float,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass from a donor to a receiver, keeping the sum.
    Accept only improvements; stop when time runs out."""
    a = seq.copy()
    best_seq = a.copy()
    best = best_val

    while time.time() < deadline:
        # choose a donor proportional to its weight (more mass → more chance)
        donor = rng.choice(n, p=a)
        if a[donor] <= 1e-12:
            continue

        # choose a receiver uniformly (different from donor)
        receiver = rng.integers(0, n - 2)
        if receiver >= donor:
            receiver += 1

        # move at most 40 % of the donor's mass
        max_move = a[donor] * 0.4
        epsilon = rng.random() * max_move
        if epsilon <= 0.0:
            continue

        new = a.copy()
        new[donor] -= epsilon
        new[receiver] += epsilon

        # quick feasibility check (non‑negativity is ensured by construction)
        val = _objective_surrogate(new)
        if val < best:
            best = val
            best_seq = new
            a = new

    return best_seq, best


# ----------------------------------------------------------------------
# Small utilities for sorting / min without lambdas
# ----------------------------------------------------------------------
def _pair_val(pair):
    """Return the second element of a (seq, val) tuple."""
    return pair[1]


def _candidate_val(pair):
    """Return the second element of a (seq, val) candidate."""
    return pair[1]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function respects the time budget and stores
    the best found sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    n = 4000  # fixed problem size (chosen to balance quality and speed)

    # ------------------------------------------------------------------
    # Load previous solution if it matches the current size
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None and isinstance(previous, list) and len(previous) == n:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Broad seed generation (diverse families + randomness)
    # ------------------------------------------------------------------
    seeds = []

    # Analytic families – coarse grids
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in np.linspace(0.5, 5.0, 19):
        seeds.append(_sine_window(n, gamma=gamma))

    for sigma in np.linspace(0.06, 0.30, 13):
        seeds.append(_gaussian(n, sigma))

    for beta in np.linspace(0.5, 20.0, 20):
        seeds.append(_kaiser_window(n, beta))

    for p in np.linspace(0.0, 2.0, 21):
        seeds.append(_power(n, p))

    # Beta‑shape grid
    alphas = np.linspace(0.2, 5.0, 6)
    betas = np.linspace(0.2, 5.0, 6)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(15):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(5):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Alternating Dirichlet seed
    seeds.append(_alternating_dirichlet(n, rng))

    # Include perturbed copies of the previous best (if any)
    if prev_arr is not None and prev_arr.sum() > 0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(10):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=_pair_val)
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"\nTop {top_k} surrogate values among {len(seeds)} seeds:")
    for idx in range(top_k):
        print(f"  #{idx + 1}: {evaluated[idx][1]:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement (short EG + optional L‑BFGS)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # Allocate a slice of the remaining budget for this seed
        remaining = deadline - time.time()
        # give each remaining seed roughly equal share of (55 % of) the rest
        share = remaining * 0.55 / (top_k - idx)
        sub_deadline = time.time() + share

        # ----- short exponentiated‑gradient pass -----
        seq_eg, val_eg = _exp_gradient_opt(seed_vec.copy(),
                                           n,
                                           sub_deadline,
                                           max_iters=150_000,
                                           eta0=6.0)

        # ----- L‑BFGS fine‑tuning (if time permits) -----
        if time.time() < deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               n,
                                               deadline,
                                               beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                              200.0, 500.0, 1000.0,
                                                              2000.0, 5000.0,
                                                              20000.0, 50000.0,
                                                              100000.0, 200000.0],
                                               max_iter_per_beta=3000)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs = seq_eg
            val_lbfgs = val_eg

        # Choose the best among seed, EG, L‑BFGS
        candidates = [(seed_vec, _objective_surrogate(seed_vec)),
                      (seq_eg, val_eg),
                      (seq_lbfgs, val_lbfgs)]
        local_best_seq, local_best_val = min(candidates, key=_candidate_val)

        print(f"[Seed {idx + 1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"  >>> New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final random‑move hill climbing (monotone improvements)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final hill climbing ---")
        seq_hc, val_hc = _random_hill_climb(global_best_seq,
                                            global_best_val,
                                            n,
                                            deadline,
                                            rng)
        if val_hc < global_best_val:
            global_best_seq = seq_hc
            global_best_val = val_hc
            print(f"[Hill climb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final long exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        seq_final, val_final = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=500_000,
                                                 eta0=5.0)
        if val_final < global_best_val:
            global_best_seq = seq_final
            global_best_val = val_final
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation with the provided black‑box
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
