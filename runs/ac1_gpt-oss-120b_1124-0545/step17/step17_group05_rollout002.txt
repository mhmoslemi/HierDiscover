"""
Propose a low‑autocorrelation step function.

The algorithm consists of:
1.  A rich set of analytic families (uniform, triangular, sine‑window,
    Gaussian, power‑law, beta‑shape, Tukey, Kaiser, double‑Gaussian) and
    random Dirichlet draws are generated for a fixed length n=4000.
    For families with a continuous parameter a coarse grid is used first.
2.  The best few seeds (by the exact 2·n·max(conv) surrogate) are taken.
3.  Each seed is polished by a long mirror‑descent (exponentiated‑gradient)
    on the *exact* max‑convolution objective.  The schedule η_t = c/√t is
    used and the loop runs as long as time permits (up to several million
    iterations).  The best sequence seen during the run is kept.
4.  After the best candidate is identified, an additional swap‑based
    hill‑climber (randomly swapping two positions) is performed for a
    final fine‑tuning.
5.  The final sequence is returned and stored in the global variable
    ``height_sequence_1`` for possible reuse.

The whole search respects a 1000‑second wall‑clock budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Softmax → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for ℓ₁‑normalised vectors:
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – each returns an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Mirror‑descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000,
                     report_every: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)         # mirror variables (log‑softmax)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        # Time check only every so often to reduce overhead
        if it % 1000 == 0 and time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # Sub‑gradient for the active lag
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx
        g[idx] = 2.0 * a[rev_idx]

        # Step‑size schedule
        eta = 0.5 / math.sqrt(it)

        # Mirror‑descent update in log‑space
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if report_every and (it % report_every == 0):
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Random swap hill‑climber (final polishing)
# ----------------------------------------------------------------------
def _perm_swap_opt(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_swaps: int = 200_000,
                   report_every: int = 50_000) -> tuple[np.ndarray, float]:
    """
    Randomly swap two positions; keep the swap only if it reduces the exact
    max‑convolution.  Returns the best sequence and its surrogate value.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_swaps + 1):
        if it % 1000 == 0 and time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]      # swap
        new_val = _objective_surrogate(cur_seq)

        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if report_every and (it % report_every == 0):
            print(f"[PermSwap] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val

    # Normalise sanity‑check
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Refine one‑parameter families by scalar optimisation (if SciPy is present)
# ----------------------------------------------------------------------
def _refine_one_param(family_func, n: int, param_range, init_vals,
                     deadline: float, max_evals: int = 30):
    """
    Fast scalar optimisation of a single‑parameter family using Brent.
    Returns the best (parameter, vector, objective) found within the budget.
    """
    try:
        import scipy.optimize as _opt
    except Exception:                     # pragma: no cover
        return None                        # no refinement without SciPy

    best = None
    best_obj = float("inf")
    # Use a simple loop over initial guesses and let Brent improve each.
    for start in init_vals:
        if time.time() > deadline:
            break

        def fun(param):
            # Clamp param to the allowed interval to avoid domain errors
            p = float(param)
            if p < param_range[0]:
                p = param_range[0]
            if p > param_range[1]:
                p = param_range[1]
            vec = family_func(n, p)
            return _objective_surrogate(vec)

        # Brent bounded optimisation
        res = _opt.minimize_scalar(fun, bounds=param_range, method='bounded',
                                   options={'maxiter': max_evals, 'xatol': 1e-4})
        if res.success:
            vec = family_func(n, res.x)
            obj = _objective_surrogate(vec)
            if obj < best_obj:
                best_obj = obj
                best = (res.x, vec, obj)
    return best

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under a 1000 s budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # leave a small safety margin

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # --------------------------------------------------------------
    # 2. Load previous solution if any
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 3. Coarse seed generation
    # --------------------------------------------------------------
    print("[Stage 1] Generating coarse seed pool …")
    seeds = _collect_seeds_coarse(n, rng, prev_arr)
    print(f"    collected {len(seeds)} seeds (coarse).")

    # --------------------------------------------------------------
    # 4. One‑parameter refinement (if time permits)
    # --------------------------------------------------------------
    # We'll attempt to refine the families that often perform well:
    # sine_window (γ), gaussian (σ), power (p), tukey (α), kaiser (β)
    # The refinement is cheap because each evaluation is a single FFT.
    best_refined = []
    remaining = deadline - time.time()
    if remaining > 30.0:  # need at least a few seconds
        print("[Stage 2] Refining one‑parameter families …")
        # 4a. sine window γ ∈ [0.1, 6.0]
        gamma_best = _refine_one_param(_sine_window, n,
                                       (0.1, 6.0),
                                       init_vals=np.linspace(0.2, 5.0, 5),
                                       deadline=deadline,
                                       max_evals=25)
        if gamma_best:
            best_refined.append(gamma_best[1])

        # 4b. Gaussian σ ∈ [0.05, 0.5]
        sigma_best = _refine_one_param(_gaussian, n,
                                       (0.05, 0.5),
                                       init_vals=np.linspace(0.08, 0.45, 5),
                                       deadline=deadline,
                                       max_evals=25)
        if sigma_best:
            best_refined.append(sigma_best[1])

        # 4c. Power law p ∈ [0.0, 6.0]
        p_best = _refine_one_param(_power, n,
                                   (0.0, 6.0),
                                   init_vals=np.linspace(0.0, 5.0, 6),
                                   deadline=deadline,
                                   max_evals=25)
        if p_best:
            best_refined.append(p_best[1])

        # 4d. Tukey α ∈ [0.01, 0.99]
        alpha_best = _refine_one_param(_tukey_window, n,
                                       (0.01, 0.99),
                                       init_vals=np.linspace(0.05, 0.95, 5),
                                       deadline=deadline,
                                       max_evals=25)
        if alpha_best:
            best_refined.append(alpha_best[1])

        # 4e. Kaiser β ∈ [0.0, 30.0]
        beta_best = _refine_one_param(_kaiser_window, n,
                                      (0.0, 30.0),
                                      init_vals=np.linspace(2.0, 20.0, 5),
                                      deadline=deadline,
                                      max_evals=25)
        if beta_best:
            best_refined.append(beta_best[1])

        # Append refined vectors to the pool
        seeds.extend(best_refined)
        print(f"    added {len(best_refined)} refined seeds.")

    # --------------------------------------------------------------
    # 5. Quick surrogate evaluation of all seeds, keep the best few
    # --------------------------------------------------------------
    print("[Stage 3] Evaluating surrogate for all seeds …")
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _ in evals[:top_k]]
    print("    top surrogate values:")
    for i, (_, v) in enumerate(evals[:top_k]):
        print(f"      #{i+1:2d}: {v:.6f}")

    # --------------------------------------------------------------
    # 6. Exponentiated‑gradient polishing (main optimisation loop)
    # --------------------------------------------------------------
    # The best surrogate currently becomes the initial candidate for
    # the long mirror‑descent.  We polish each of the top seeds for a
    # short budget first, then concentrate on the best one.
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached during gradient polishing.")
            break

        # Permit a modest number of mirror‑descent iterations for the
        # first few seeds (to diversify) and then a large budget for the
        # current global best.
        remaining = deadline - time.time()
        if idx == 0:
            # aggressive polishing for the very first seed
            max_iters = int(min(1_200_000, remaining * 0.25))
        else:
            max_iters = int(min(300_000, remaining * 0.10))

        max_iters = max(100_000, max_iters)

        print(f"\n[ExpGrad] polishing seed #{idx+1} (max_iters={max_iters:,}) …")
        seq_opt, val_opt = _exp_gradient_opt(seed_vec, n, deadline,
                                             max_iters=max_iters,
                                             report_every=150_000)
        print(f"    after polishing: surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"    ** new global best **  surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Final long mirror‑descent on the current best candidate
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(2_000_000, remaining * 0.45))
        max_iters = max(300_000, max_iters)
        print(f"\n[ExpGrad] final long run on best candidate (max_iters={max_iters:,}) …")
        global_best_seq, global_best_val = _exp_gradient_opt(
            global_best_seq, n, deadline,
            max_iters=max_iters,
            report_every=250_000)
        print(f"    final mirror‑descent best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 8. Permutation‑swap hill‑climber (fine‑tuning)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = int(min(800_000, remaining * 0.20))
        max_swaps = max(200_000, max_swaps)
        print("\n[PermSwap] final local search …")
        seq_perm, val_perm = _perm_swap_opt(global_best_seq, n, deadline,
                                            max_swaps=max_swaps,
                                            report_every=100_000)
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"    ** improved by swap **  surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 9. Official evaluation and return
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:      # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
