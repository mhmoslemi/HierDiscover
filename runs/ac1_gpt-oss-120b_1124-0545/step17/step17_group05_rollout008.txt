"""
Light‑weight low‑autocorrelation search.

We create a sparse “Sidon‑like’’ set of indices S ⊂ {0,…,n‑1}
such that each ordered sum i+j (0≤i,j<n) occurs only a few times.
If |S| = m and the maximal ordered‑pair count is C,
the convolution of the vector a (a_i = 1/m for i∈S, 0 otherwise) satisfies

    max(conv(a,a)) = C / m² ,

and the official objective becomes  2·n·C / m² .
Both a larger support m and a smaller C improve the score.

The algorithm works in three stages:

1. **Random greedy construction** – repeatedly try to insert indices while
   keeping the ordered‑pair count ≤ Cmax (Cmax = 3,4,5 …).  The best
   (smallest surrogate 2·n·C/m²) is kept.

2. **Exponentiated‑gradient polishing** – starting from the best vector,
   we minimise the exact max‑convolution using a mirror‑descent (EG)
   scheme with a sub‑gradient taken from the current maximal lag.

3. **Optional swap hill‑climber** – a cheap permutation search that
   keeps any improvement.

All steps respect the global 1000 s budget.  The final vector is stored
in the global variable ``height_sequence_1`` and returned as a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Greedy construction of a sparse set S with bounded ordered‑pair counts
# ----------------------------------------------------------------------
def _greedy_build_set(n: int, max_allowed: int, rng: np.random.Generator):
    """
    Random greedy insertion of indices.
    Returns (positions array, ordered‑pair count array).
    """
    positions = []                      # selected indices
    sum_counts = np.zeros(2 * n - 1, dtype=np.int32)  # ordered pair histogram

    order = rng.permutation(n)          # random order of attempts
    for i in order:
        # check self‑pair (i,i)
        if sum_counts[i + i] + 1 > max_allowed:
            continue

        # check all cross‑pairs with already selected positions
        feasible = True
        for s in positions:
            if sum_counts[i + s] + 2 > max_allowed:
                feasible = False
                break
        if not feasible:
            continue

        # accept i
        positions.append(i)
        sum_counts[i + i] += 1
        for s in positions[:-1]:        # all previous entries
            sum_counts[i + s] += 2

    return np.array(positions, dtype=np.int32), sum_counts


def _objective_from_set(n: int, m: int, max_counts: int) -> float:
    """Surrogate value for a set S of size m with maximal ordered‑pair count max_counts."""
    if m == 0:
        return float('inf')
    return 2.0 * n * max_counts / (m * m)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution (sub‑gradient from the
    currently active lag).  Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        kmax = int(np.argmax(conv))

        # sub‑gradient for the active lag kmax
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # learning rate – classic 1/√t schedule
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = np.exp(log_a)
        a = a / a.sum()                     # projection onto simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 100_000 == 0:
            print(f"[ExpGrad] iteration={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple swap‑based permutation hill‑climber
# ----------------------------------------------------------------------
def _perm_swap_opt(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_swaps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Randomly swap two entries; keep the swap only if it reduces the exact
    max‑convolution.  Returns (best_sequence, best_surrogate_value).
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_swaps + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_val = _objective_surrogate(cur_seq)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if it % 50_000 == 0:
            print(f"[PermSwap] swaps={it:,} best={best_val:.6f}")

    return best_seq, best_val

    Returns the best sequence (list of floats) found within the time budget.
    """

    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 2. Greedy random construction of sparse supports
    # --------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    trial = 0

    print("\n--- Greedy sparse set construction (n={}) ---".format(n))
    while time.time() < deadline - 5.0:
        trial += 1
        # randomly pick a bound on the ordered‑pair count (3,4 or 5)
        max_allowed = int(rng.integers(3, 6))   # 3 ≤ max_allowed ≤ 5
        positions, sum_counts = _greedy_build_set(n, max_allowed, rng)
        m = len(positions)
        if m == 0:
            continue
        max_counts = int(sum_counts.max())
        val = _objective_from_set(n, m, max_counts)

        if val < best_val:
            best_val = val
            # build the probability vector (equal weight on the support)
            seq = np.zeros(n, dtype=np.float64)
            seq[positions] = 1.0 / m
            best_seq = seq
            print(f"[Trial {trial:4d}] improved surrogate={best_val:.6f} "
                  f"(|S|={m}, Cmax={max_counts})")

    # fall back to a uniform vector if nothing was found (should not happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective_surrogate(best_seq)
        print("[Fallback] using uniform sequence.")

    # --------------------------------------------------------------
    # 3. Exponentiated‑gradient refinement (exact max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # rough estimate: ~0.001 s per iteration for n≈4000 → 1 M it ≈ 1000 s
        # we are much more limited; allocate ≈0.6 s per 10 k it → 60 k it per 6 s
        # use a generous upper bound but stay within the remaining budget.
        max_iters = int(min(600_000, remaining * 2000))
        max_iters = max(200_000, max_iters)   # ensure a useful amount
        print(f"\n--- Exponentiated‑gradient refinement (max_iters={max_iters}) ---")
        best_seq, refined_val = _exp_gradient_opt(best_seq, n, deadline, max_iters=max_iters)
        best_val = refined_val
        print(f"[ExpGrad] final surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. (Optional) Swap‑based permutation hill‑climber
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_swaps = int(min(300_000, remaining * 1500))
        max_swaps = max(100_000, max_swaps)
        print(f"\n--- Permutation‑swap hill‑climber (max_swaps={max_swaps}) ---")
        seq_perm, val_perm = _perm_swap_opt(best_seq, n, deadline, max_swaps=max_swaps)
        if val_perm < best_val:
            best_seq = seq_perm
            best_val = val_perm
            print(f"[PermSwap] improved surrogate = {best_val:.6f}")

        print("Warning: evaluate_sequence raised an exception:", exc)
