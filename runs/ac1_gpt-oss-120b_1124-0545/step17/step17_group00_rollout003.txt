def adam_opt(start_seq, n, deadline, max_iters=2000000, lr=0.1):
    a = start_seq.copy()
    v = np.log(a + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_val = objective_max(a)
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        a = softmax(v)
        conv = conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # subgradient g wrt a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]
        # gradient wrt v
        dot = np.dot(g, a)
        grad_v = a * (g - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_corr = m / (1 - beta1**t)
        v_corr = v_hat / (1 - beta2**t)
        v = v - lr * m_corr / (np.sqrt(v_corr) + eps)
        # compute objective
        cur_val = objective_max(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

"""
Improved step‑function search.

The algorithm proceeds in three phases:

1. **Rich seeding** – a dense sweep over analytic families (Beta,
   Gaussian, power‑law, windows, etc.) together with many random Dirichlet,
   exponential and low‑frequency draws.  The best *k* candidates (by the
   exact surrogate 2·n·max(conv(a,a)) with ∑a=1) are retained.

2. **Heavy optimisation** – starting from the best seed we run a very long
   exponentiated‑gradient (mirror‑descent) on the *exact* max‑convolution,
   followed by a continuation L‑BFGS optimisation of a smooth
   log‑sum‑exp surrogate with an extended β‑schedule.  Both stages respect
   the simplex constraint via a soft‑max parametrisation.

3. **Local refinement** – a greedy hill‑climber (tiny mass transfers)
   squeezes out the last few digits.  The final sequence is evaluated with
   the official ``evaluate_sequence`` and stored in ``height_sequence_1``.

The search is time‑bounded (default 1000 s) and adapts the number of
iterations to the remaining budget.
"""

import time
import heapq
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """
    Exact surrogate for an ℓ₁‑normalised vector (∑a = 1):
        2·n·max(conv(a, a))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random generators (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Updates ``log_a ← log_a – η·g`` with η = η₀ / sqrt(t).
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (sub‑gradient version)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 3_000_000,
              lr: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the non‑smooth max‑convolution objective.
    Works in log‑softmax space (v) guaranteeing non‑negativity and the simplex.
    Returns (best_sequence, best_surrogate_value).
    """
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = _softmax(v)
    best_val = _objective(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # current probability vector
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (same as in EG)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = max_idx - idx
        g[idx] = 2.0 * a[rev]

        # convert to gradient w.r.t. logits v
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_corr = v_hat / (1.0 - beta2 ** t)
        v = v - lr * m_hat / (np.sqrt(v_corr) + eps)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 20_000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    The variable lives in log‑softmax space (v), guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        # extended schedule reaching up to β=2 000 000
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0, 5_000.0,
                         10_000.0, 20_000.0, 50_000.0,
                         100_000.0, 200_000.0, 500_000.0,
                         1_000_000.0, 2_000_000.0]

    # initialise logits
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # stays on simplex
            conv = _conv_fft(a, a)            # full convolution
            max_c = np.max(conv)
            shift = max_c                     # stabilise exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # soft‑max weights over convolution entries
            w = exp_beta / sum_exp

            # gradient wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Simple hill‑climbing (greedy mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer moves on the simplex.  Each step picks two indices i,j
    and moves a random fraction of a[i] to a[j].  Accept only improving
    moves.  Returns the best sequence found and its surrogate value.
    """
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # move a small fraction (0.1 % – 2 %) of the mass at i to j
        eps = cur[i] * rng.uniform(0.001, 0.02)
        if eps <= 0.0 or cur[i] - eps < 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_val = _objective(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new
                best_val = new_val
    return best, best_val


# ----------------------------------------------------------------------
# Candidate handling (heap of the top‑k seeds)
# ----------------------------------------------------------------------
def _push_candidate(vec: np.ndarray,
                    heap: list,
                    top_k: int,
                    n: int):
    """Insert ``vec`` (assumed non‑negative) into ``heap`` keeping the best top_k."""
    if not np.isfinite(vec).all():
        return
    s = vec.sum()
    if s <= 0.0:
        return
    vec = np.maximum(vec, 0.0) / s   # normalise to unit ℓ₁
    val = _objective(vec)
    if len(heap) < top_k:
        heapq.heappush(heap, (-val, vec))
    else:
        if val < -heap[0][0]:
            heapq.heappushpop(heap, (-val, vec))

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the time budget and stores the best found sequence in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Problem size – choose a power‑of‑two for fast FFTs
    # ------------------------------------------------------------------
    n = 4096                     # a modest size with fast FFTs (≈4000 previously)

    # ------------------------------------------------------------------
    # 2. Seed generation (keep the best ``top_k`` candidates)
    # ------------------------------------------------------------------
    top_k = 12
    best_heap = []               # stores (-objective, vector)

    # deterministic analytic families (dense parameter sweeps)
    _push_candidate(_uniform(n), best_heap, top_k, n)
    _push_candidate(_triangular(n), best_heap, top_k, n)

    # sine windows (γ from 0.2 to 3.0)
    for gamma in np.linspace(0.2, 3.0, 15):
        _push_candidate(_sine_window(n, gamma), best_heap, top_k, n)

    # Gaussian families
    for sigma in np.linspace(0.07, 0.30, 24):
        _push_candidate(_gaussian(n, sigma), best_heap, top_k, n)

    # Power‑law families
    for p in np.linspace(0.0, 3.0, 31):
        _push_candidate(_power(n, p), best_heap, top_k, n)

    # Beta‑shape families (coarse grid, then later we will refine)
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            _push_candidate(_beta_shape(n, a, b), best_heap, top_k, n)

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        _push_candidate(_tukey_window(n, alpha), best_heap, top_k, n)

    # Kaiser windows
    for beta in np.linspace(1.0, 15.0, 8):
        _push_candidate(_kaiser_window(n, beta), best_heap, top_k, n)

    # Double Gaussian mixtures
    for sigma in (0.10, 0.12, 0.15):
        for shift in (0.20, 0.25, 0.30, 0.35):
            _push_candidate(_double_gaussian(n, sigma, shift),
                            best_heap, top_k, n)

    # ------------------------------------------------------------------
    # 3. Massive random seeding
    # ------------------------------------------------------------------
    conc_choices = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    random_seed_budget = 4000
    for _ in range(random_seed_budget):
        if time.time() > deadline:
            break
        conc = rng.choice(conc_choices)
        vec = _random_dirichlet(n, conc, rng)
        # random cyclic shift can dramatically change the autocorrelation
        shift = rng.integers(n)
        if shift:
            vec = np.roll(vec, shift)
        _push_candidate(vec, best_heap, top_k, n)

    # exponential draws (heavy‑tailed)
    for _ in range(400):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        shift = rng.integers(n)
        if shift:
            vec = np.roll(vec, shift)
        _push_candidate(vec, best_heap, top_k, n)

    # low‑frequency random curves
    for nf in (2, 3, 5):
        for _ in range(150):
            if time.time() > deadline:
                break
            vec = _lowfreq_curve(n, nf, rng)
            shift = rng.integers(n)
            if shift:
                vec = np.roll(vec, shift)
            _push_candidate(vec, best_heap, top_k, n)

    # ------------------------------------------------------------------
    # 4. Re‑use previously discovered construction (if any)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n and prev_arr.sum() > 0:
                _push_candidate(prev_arr, best_heap, top_k, n)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 5. Extract the best seeds (ordered from best to worst)
    # ------------------------------------------------------------------
    seeds = [vec for _, vec in sorted(best_heap, key=lambda x: -x[0])]
    print(f"\n[Seed phase] kept {len(seeds)} best candidates, best surrogate = {_objective(seeds[0]):.6f}")

    # ------------------------------------------------------------------
    # 6. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Exponentiated‑gradient polishing (majority of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        eg_time = (deadline - time.time()) * 0.55
        eg_deadline = time.time() + eg_time
        seq_eg, val_eg = _exp_gradient_opt(global_best_seq,
                                           n,
                                           eg_deadline,
                                           max_iters=5_000_000,
                                           eta0=5.0)
        if val_eg < global_best_val:
            global_best_seq = seq_eg
            global_best_val = val_eg
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Adam optimisation on the exact max (optional, further refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        adam_time = (deadline - time.time()) * 0.30
        adam_deadline = time.time() + adam_time
        seq_adam, val_adam = _adam_opt(global_best_seq,
                                       n,
                                       adam_deadline,
                                       max_iters=3_000_000,
                                       lr=0.5)
        if val_adam < global_best_val:
            global_best_seq = seq_adam
            global_best_val = val_adam
            print(f"[Adam] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Continuation L‑BFGS fine‑tuning (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0 and _opt is not None:
        print("\n--- L‑BFGS fine‑tuning ---")
        lbfgs_deadline = deadline - 5.0
        seq_lbfgs = _lbfgs_smooth_optimize(global_best_seq,
                                           n,
                                           lbfgs_deadline,
                                           beta_schedule=None,
                                           max_iter_per_beta=30_000)
        val_lbfgs = _objective(seq_lbfgs)
        if val_lbfgs < global_best_val:
            global_best_seq = seq_lbfgs
            global_best_val = val_lbfgs
            print(f"[L‑BFGS] refined surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 10. Final greedy hill‑climb (use any leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n--- Final hill‑climb ---")
        seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline, rng)
        if val_hc < global_best_val:
            global_best_seq = seq_hc
            global_best_val = val_hc
            print(f"[Hill‑climb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 11. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
