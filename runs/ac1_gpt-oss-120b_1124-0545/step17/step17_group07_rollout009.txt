"""
Best‑of‑both‑world optimiser for the step‑function problem.

The routine builds a rich pool of ℓ₁‑normalised “seed” shapes,
runs a short continuation L‑BFGS on a log‑sum‑exp smoothed max‑conv,
then finishes with a long Adam‑style mirror‑descent on the *exact*
max‑convolution (the true objective up to the ℓ₁‑normalisation).
A cheap pair‑wise hill‑climb is applied at the end for final polishing.
The whole search respects the 1000 s budget and stores the best result in
the global variable ``height_sequence_1`` for the next run.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# SciPy is optional – L‑BFGS is only used when available.
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective helpers (surrogate works on ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) for a sequence that sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _double_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta(α,β) on {0,…,n‑1} using a half‑offset to avoid zeros."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 0.5) ** (alpha - 1.0)) * ((n - i - 0.5) ** (beta - 1.0))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _sin_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p  (zero at the ends, symmetric)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n) ** p
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_uniform_block(n: int, block_size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of `block_size` indices."""
    idx = rng.choice(n, size=block_size, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / block_size
    return a


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse collection of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # ----- analytic families ------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Exponential / double‑exponential family
    for rate in (0.001, 0.003, 0.006, 0.01, 0.015):
        seeds.append(_exponential(n, rate))
        seeds.append(_double_exponential(n, rate))

    # Power‑law family
    for p in (0.0, 0.25, 0.5, 0.75, 1.0, 2.0):
        seeds.append(_power(n, p))

    # Beta‑shapes (symmetric)
    for a in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        seeds.append(_beta_distribution(n, a, a))

    # Sin‑power family (peaked at centre, zero at edges)
    for p in (1.0, 2.0, 4.0, 6.0):
        seeds.append(_sin_power(n, p))

    # Dirichlet draws (different concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(3):
            d = _random_dirichlet(n, conc, rng)
            seeds.append(d)                     # raw
            seeds.append(np.sort(d)[::-1])      # sorted descending

    # Sparse block‐uniform seeds
    for blk in (80, 120, 200):
        for _ in range(2):
            seeds.append(_sparse_uniform_block(n, blk, rng))

    # Previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # ----- final clean‑up ---------------------------------------------------
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (log‑sum‑exp smooth max)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 1500,
) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smoothed max‑conv."""
    if _opt is None:
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0,
            3.0,
            7.0,
            15.0,
            30.0,
            60.0,
            120.0,
            250.0,
            500.0,
            1000.0,
        ]

    # optimisation variables are logits; softmax guarantees simplex stay
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_f = float(beta)

        def fun(v_local):
            a = _softmax(v_local)  # stays on simplex
            conv = _conv_fft(a, a)
            m = np.max(conv)
            # smooth max via log‑sum‑exp
            exp_beta = np.exp(beta_f * (conv - m))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (m + np.log(sum_exp) / beta_f)

            # gradient of smooth max w.r.t. a
            w = exp_beta / sum_exp  # soft weights (sum to 1)
            rev_a = a[::-1]
            # d/da of Σ w_k * Σ_i a_i a_{k-i}
            # can be expressed as convolution of rev_a with w
            grad_a = 2.0 * _conv_fft(rev_a, w)[n - 1 : n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_f}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam‑style mirror‑descent on the exact max‑conv surrogate
# ----------------------------------------------------------------------
def _adam_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    alpha0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on logits (mirror‑descent) using the *exact* max‑convolution.
    Returns (best_seq, best_surrogate_value).
    """
    a = start_seq.copy()
    logit = np.log(a + 1e-30)

    # Adam moments
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(logit)

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry (pairwise terms)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # chain rule through softmax (gradient w.r.t logits)
        grad_logit = a * (g - np.dot(g, a))

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_logit
        v = beta2 * v + (1.0 - beta2) * (grad_logit ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha0 / (np.sqrt(v_hat) + eps)
        logit -= step * m_hat

    return best_seq, best_val


# ----------------------------------------------------------------------
# Cheap pair‑wise hill‑climbing (logit swaps)
# ----------------------------------------------------------------------
def _hill_climbing_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Very cheap pair‑wise logit swap hill‑climbing."""
    rng = np.random.default_rng()
    logit = np.log(start_seq + 1e-30)

    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)

        # a tiny logit perturbation that conserves the simplex (add to i, subtract from j)
        sigma = 0.012 / math.sqrt(it)
        delta = rng.normal(scale=sigma)
        new_log = logit.copy()
        new_log[i] += delta
        new_log[j] -= delta

        new_seq = _softmax(new_log)
        new_val = _objective_surrogate(new_seq)

        if new_val < cur_val:
            logit, cur_seq, cur_val = new_log, new_seq, new_val
            if new_val < best_val:
                best_seq, best_val = new_seq.copy(), new_val

    return best_seq, best_val

        max_iter_adam = int(min(2_500_000, remaining * 1200))  # ~0.0008 s per iter empirically
        max_iter_adam = max(200_000, max_iter_adam)
        print(f"\n--- Adam‑mirror‑descent (max_iters={max_iter_adam}) ---")
        seq_adam, val_adam = _adam_opt(
            global_best_seq,
            n,
            deadline,
            max_iters=max_iter_adam,
            alpha0=0.6,
        )
        if val_adam < global_best_val:
            global_best_seq, global_best_val = seq_adam.copy(), val_adam
            print(f"[Adam] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Local hill‑climbing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iter_hc = int(min(500_000, remaining * 800))
        max_iter_hc = max(50_000, max_iter_hc)
        print(f"\n--- Hill‑climbing (max_iters={max_iter_hc}) ---")
        seq_hc, val_hc = _hill_climbing_opt(
            global_best_seq,
            n,
            deadline,
            max_iters=max_iter_hc,
        )
        if val_hc < global_best_val:
            global_best_seq, global_best_val = seq_hc.copy(), val_hc
            print(f"[HillClimb] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next iteration (if any)
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
