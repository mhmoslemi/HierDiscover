"""
Step‑function optimizer.

Goal: minimise  2·n·max(conv(a,a)) / (sum(a)²)  for a non‑negative sequence a.
The algorithm builds a diverse seed pool (analytic families, asymmetric Beta,
Dirichlet draws, previous solution) and selects the most promising candidates.
It then refines them with:

* a continuation L‑BFGS optimiser on a smooth log‑sum‑exp surrogate,
* simple mixing with the reversed sequence and the uniform distribution,
* a long projected sub‑gradient descent on the exact max‑convolution,
* a cheap hill‑climbing pass.

The whole pipeline respects the 1000 s budget and stores the best sequence
in the global variable ``height_sequence_1`` for the next run.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# SciPy is optional – L‑BFGS is used when available.
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (surrogate works on ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) for a sequence that sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by evaluate_sequence (no normalisation)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta(α,β) on {0,…,n‑1} – half‑offset avoids zeros."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 0.5) ** (alpha - 1.0)) * ((n - i - 0.5) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smoothed max‑conv."""
    if _opt is None:
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 3.0, 10.0, 30.0, 100.0,
            300.0, 1000.0, 3000.0, 10000.0,
            30000.0, 100000.0
        ]

    v = np.log(start_vec + 1e-30)          # work in log‑it‑space

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)          # stays on simplex
            conv = _conv_fft(a, a)
            m = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = m
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp          # soft weights
            rev_a = a[::-1]
            # gradient w.r.t. a (convolution of rev_a with w)
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution surrogate
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Long projected sub‑gradient descent on 2·n·max(conv) (exact surrogate)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.3 / math.sqrt(it)          # a slightly larger stepsize than before
        a -= lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very cheap pair‑wise logit swap hill‑climbing
# ----------------------------------------------------------------------
def _hill_climbing_opt(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Cheapest local hill‑climbing in logit space."""
    rng = np.random.default_rng()
    logit = np.log(start_seq + 1e-30)

    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq, best_val = cur_seq.copy(), cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        sigma = 0.02 / math.sqrt(it)
        delta = rng.normal(scale=sigma)
        new_log = logit.copy()
        new_log[i] += delta
        new_log[j] -= delta
        new_seq = _softmax(new_log)
        new_val = _objective_surrogate(new_seq)
        if new_val < cur_val:
            logit, cur_seq, cur_val = new_log, new_seq, new_val
            if new_val < best_val:
                best_seq, best_val = new_seq.copy(), new_val

    return best_seq, best_val

def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse collection of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Gaussian (different bandwidths)
    for sigma in (0.07, 0.14, 0.21, 0.28):
        seeds.append(_gaussian(n, sigma))

    # Asymmetric Beta families – a wide grid of (α,β)
    for a in (0.3, 0.5, 0.8, 1.2, 1.6, 2.0, 3.0):
        for b in (0.5, 1.0, 1.5, 2.5, 4.0):
            seeds.append(_beta_distribution(n, a, b))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.1, 0.3, 0.7, 1.5, 3.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Previous solution (if compatible)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Normalise everything and discard pathological cases
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Mixing helpers (reverse / uniform)
# ----------------------------------------------------------------------
def _mix_with_reverse(seq: np.ndarray, t: float) -> np.ndarray:
    """Convex mix of a sequence and its reversal."""
    mixed = (1.0 - t) * seq + t * seq[::-1]
    total = mixed.sum()
    if total == 0.0:
        return seq.copy()
    return mixed / total


def _mix_with_uniform(seq: np.ndarray, t: float, uniform: np.ndarray) -> np.ndarray:
    """Convex mix of a sequence with the uniform distribution."""
    mixed = (1.0 - t) * seq + t * uniform
    total = mixed.sum()
    if total == 0.0:
        return seq.copy()
    return mixed / total

    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible subsequent runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
