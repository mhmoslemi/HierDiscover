"""
Optimise a non‑negative step‑function (list of heights) to minimise

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search proceeds in three phases:

1. **Diverse seed generation** – many analytic families (sine‐window,
   beta‐distribution, double‑Gaussian, power law, etc.), a dense grid of
   parameters, and many random Dirichlet / exponential draws.  All seeds are
   ℓ₁‑normalised.

2. **Mid‑level refinement** – the best K seeds (by the cheap surrogate
   ``2·n·max(conv)``) are independently refined by a fast exponentiated‑gradient
   (mirror‑descent) pass followed by a modern Adam optimiser on a smooth
   log‑sum‑exp surrogate of the max‑convolution.  The Adam schedule uses a warm
   “temperature” β that is increased according to a preset schedule.

3. **Final polishing** – the overall best candidate is polished with a longer
   Adam run (more steps per β) until the global time budget (default 1000 s)
   is exhausted.

The best sequence found is stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# FFT‑based convolution utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (real output)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# --------------------------------------------------------------
# Simple vector utilities (softmax, projection)
# --------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# --------------------------------------------------------------
# Surrogate objective for ℓ₁‑normalised vectors
# --------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# --------------------------------------------------------------
# Analytic base families (all return ℓ₁‑normalised vectors)
# --------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, forced non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _linear_decreasing(n: int) -> np.ndarray:
    """Deterministic linear‑decreasing shape (a_0 = 2/n, a_{n‑1}=0)."""
    a = np.arange(n, dtype=np.float64)[::-1]  # n‑1, n‑2, ..., 0
    a = a.astype(np.float64)
    a = a * (2.0 / (n * (n - 1)))          # normalise to sum=1
    return a


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Adam optimiser on a smooth log‑sum‑exp surrogate of max(conv)
# --------------------------------------------------------------
def _adam_smooth_optimize(start_vec: np.ndarray,
                          n: int,
                          deadline: float,
                          beta_schedule=None,
                          steps_per_beta: int = 600):
    """
    Adam optimisation in log‑softmax space for the smoothed max‑convolution.
    ``beta_schedule`` is a list of temperatures (higher → sharper).
    Returns the best (softmax‑normalised) vector seen.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1000.0, 2000.0, 5000.0, 20000.0,
                         50000.0, 100000.0, 200000.0]

    # log‑softmax parameters
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    s = np.zeros_like(v)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        beta_val = float(beta)
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return best_seq
            t += 1
            a = _softmax(v)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                         # numerical stabilisation
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp                # soft weights over all k
            # gradient w.r.t. a (see derivations in the original script)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # slice of length n
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)           # chain‑rule through softmax

            # Adam update on v
            m = beta1 * m + (1.0 - beta1) * grad_v
            s = beta2 * s + (1.0 - beta2) * (grad_v ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            s_hat = s / (1.0 - beta2 ** t)
            # step‑size schedule – a modest decay helps stability
            lr = 0.2 / (1.0 + 1e-5 * t)
            v = v - lr * m_hat / (np.sqrt(s_hat) + eps)

            # keep track of the true (non‑smoothed) objective every few steps
            if t % 30 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
    return best_seq

# Simple exponentiated‑gradient (mirror‑descent) optimiser on the exact max
# --------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200_000,
                     eta0: float = 8.0):
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient (only the active convolution entry contributes)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

            seeds.append(_beta_shape(n, a, b))

    # Gaussian families
    for sigma in np.linspace(0.08, 0.30, 12):
        seeds.append(_gaussian(n, sigma_frac=sigma))

    # Double‑Gaussian (two bumps near the ends)
    for sigma in (0.10, 0.12, 0.15):
        for shift in (0.20, 0.25, 0.30):
            seeds.append(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))

    # Linear decreasing shape (purely deterministic)
    seeds.append(_linear_decreasing(n))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random cosine curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of a uniform base with a low‑frequency curve
    for mix_alpha in (0.1, 0.3, 0.5):
        for nf in (2, 3, 5):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Include a few perturbed copies of the previous best (if any)
    if prev_arr is not None and prev_arr.sum() > 0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(6):
            noisy = prev_arr + 0.003 * rng.normal(size=prev_arr.shape)
            noisy = np.maximum(noisy, 0.0)
            s = noisy.sum()
            if s > 0:
                seeds.append(noisy / s)

    # ----------------------------------------------------------
    # 2️⃣ Quick surrogate evaluation of all seeds
    # ----------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    # sort by surrogate (lower is better)
    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ----------------------------------------------------------
    # 3️⃣ Initialise global best
    # ----------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ----------------------------------------------------------
    # 4️⃣ Mid‑level refinement of the top seeds
    # ----------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        # allocate a fair share of the remaining budget
        remain = deadline - time.time()
        # each remaining seed gets roughly an equal share of the remaining time
        sub_deadline = time.time() + remain * 0.55 / max(1, (top_k - idx))

        print(f"\n[Seed {idx+1}/{top_k}] start refinement (deadline in {int(sub_deadline - time.time())} s)")

        # ----- short exponentiated‑gradient pass (fast improvement) -----
        seq_eg, val_eg = _exp_gradient_opt(seed_vec.copy(),
                                           n,
                                           sub_deadline,
                                           max_iters=150_000,
                                           eta0=8.0)

        # ----- Adam smoothing optimisation (more expressive) -----
        if time.time() < sub_deadline:
            seq_adam = _adam_smooth_optimize(seq_eg,
                                             n,
                                             sub_deadline,
                                             beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                            200.0, 500.0, 1000.0,
                                                            2000.0, 5000.0,
                                                            20000.0, 50000.0,
                                                            100000.0, 200000.0],
                                             steps_per_beta=800)
            val_adam = _objective_surrogate(seq_adam)
        else:
            seq_adam = seq_eg
            val_adam = val_eg

        # Gather candidates (original seed, EG, Adam)
        candidates = [(seed_vec, _objective_surrogate(seed_vec)),
                      (seq_eg, val_eg),
                      (seq_adam, val_adam)]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])
        print(f"[Seed {idx+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"  >>> New global best = {global_best_val:.6f}")

    # ----------------------------------------------------------
    # 5️⃣ Final polishing of the global best (long Adam run)
    # ----------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing (long Adam run) ---")
        # give the remaining budget entirely to this final pass
        final_seq = _adam_smooth_optimize(global_best_seq,
                                          n,
                                          deadline,
                                          beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                         200.0, 500.0, 1000.0,
                                                         2000.0, 5000.0,
                                                         20000.0, 50000.0,
                                                         100000.0, 200000.0,
                                                         500000.0],
                                          steps_per_beta=1200)
        final_val = _objective_surrogate(final_seq)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved to surrogate = {global_best_val:.6f}")
        else:
            print(f"[Final] no improvement (best surrogate still = {global_best_val:.6f})")

    # ----------------------------------------------------------
    # 6️⃣ True evaluation and housekeeping
    # ----------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
