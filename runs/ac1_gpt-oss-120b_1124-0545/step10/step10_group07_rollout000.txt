"""
Hybrid LP‑driven refinement + lightweight exponentiated‑gradient fine‑tuning.

1. Initialise a probability vector a (sum = 1).  If a previous construction
   ``height_sequence_1`` of the required length exists it is used, otherwise a
   uniform distribution is taken.

2. Repeatedly solve a linear programme

       maximise   Σ b_i
       subject to  (a * b)_k ≤ M   for all k = 0 … 2n‑2
                   b_i ≥ 0

   where M = max(a * a).  The convolution matrix of a is built once (the
   sparsity pattern does not change) and only the non‑zero values are
   updated each iteration.  If the LP yields a vector b with Σb>1 we can
   increase the total mass while keeping the same bound M – this is the
   driving force for improvement.

3. Mix the current vector a with the normalised LP solution g = b/Σb.
   The mixing ratio τ starts at 1.0 (full replacement) and is halved up to
   five times until the new maximum convolution M′ is strictly smaller
   than the previous M.  After a successful step a is symmetrised
   (a←(a+a[::-1])/2) to exploit the natural symmetry of the problem.

4. When the time budget runs low the LP stage stops and a short
   exponentiated‑gradient descent (mirrored SGD) is run to polish the
   remaining few digits.

The whole routine respects the 1000 s budget, never exceeds the memory
limits (the sparse matrix has at most n² non‑zeros) and returns a list of
non‑negative floats that minimises ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (required for the LP stage)
# ----------------------------------------------------------------------
try:
    from scipy import sparse               # type: ignore
    from scipy.optimize import linprog    # type: ignore
    SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Basic convolution / objective helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """smallest power of two ≥ x"""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """full linear convolution using a real FFT (fast for large n)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Analytic seeds (probability vectors, i.e. sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `alpha`."""
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _make_extended_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of analytic / random probability vectors."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_gaussian(n, 0.12))
    seeds.append(_gaussian(n, 0.20))
    seeds.append(_tent_window(n))

    # a handful of Dirichlet draws with very low concentration → spiky shapes
    for a in [0.02, 0.05, 0.1, 0.2]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, a, rng))

    # reuse previous if compatible
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # normalise & filter
    clean = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0.0:
            clean.append(s / tot)
    return clean


# ----------------------------------------------------------------------
# Exponentiated‑gradient refinement (mirrored SGD, soft‑max approx.)
# ----------------------------------------------------------------------
def _smooth_expgd(start_seq: np.ndarray,
                  deadline: float,
                  max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent in log‑space with a slowly heating soft‑max."""
    n = start_seq.size
    eps = 1e-12
    a = np.clip(start_seq, eps, None)
    a /= a.sum()
    log_a = np.log(a)
    best_seq = a.copy()
    best_val = _objective(a)

    # momentum on the gradient
    mom = np.zeros_like(a)
    beta1 = 0.9
    beta_max = 200.0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)

        # smooth max via soft‑max
        beta = min(beta_max, beta_max * it / max_iters)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w /= w.sum()

        # gradient of the smooth objective
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        g = 2.0 * full[n - 1:n - 1 + n]

        # momentum and step size
        mom = beta1 * mom + (1.0 - beta1) * g
        lr = 0.6 / math.sqrt(it)

        # EG step
        log_a = log_a - lr * mom
        log_a -= log_a.max()            # prevent overflow
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

        # occasional exact eval
        if it % 300 == 0:
            cur = _objective(a)
            if cur < best_val:
                best_val = cur
                best_seq = a.copy()
                print(f"[SmoothEG] it={it:7d} best={cur:.6f}")

    final = _objective(a)
    if final < best_val:
        best_val = final
        best_seq = a.copy()
        print(f"[SmoothEG] final best={final:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming step (core of the new search)
# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray,
             M: float,
             row_idx: np.ndarray,
             col_idx: np.ndarray,
             deadline: float) -> tuple[np.ndarray, bool]:
    """
    Solve the LP   maximise Σ b_i   s.t.   (a * b)_k ≤ M,  b ≥ 0.
    Returns the normalised solution g = b / Σb and a flag indicating whether
    Σb > 1 (i.e. a genuine improvement is possible).  If the LP fails or
    Σb ≤ 1 the flag is False.
    """
    n = a.size
    # data for the sparse constraint matrix (each column repeats the full a)
    data = np.repeat(a, n)                      # length n*n
    A = sparse.csr_matrix((data, (row_idx, col_idx)),
                          shape=(2 * n - 1, n))

    c = -np.ones(n)                             # maximise Σb  → minimise -Σb
    ub = np.full(2 * n - 1, M)                  # (a * b)_k ≤ M
    bounds = [(0.0, None)] * n

    # linprog can be a bit slow on huge problems; we give it a modest
    # iteration budget and let the outer loop handle early termination.
    res = linprog(c,
                  A_ub=A,
                  b_ub=ub,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': min(30, deadline - time.time())})

    if not res.success:
        return a, False

    b = res.x
    sum_b = b.sum()
    if sum_b <= 1.0 + 1e-12:
        return a, False

    g = b / sum_b
    return g, True

# Main entry point required by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm combines a linear‑programming driven refinement with a short
    exponentiated‑gradient polish.  The best vector found is stored globally as
    ``height_sequence_1`` for warm‑starting future runs.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a small safety margin
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length – a compromise between expressiveness and LP size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous construction if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 0) Choose a start distribution (probability vector)
    # ------------------------------------------------------------------
    if previous is not None and previous.shape == (n,):
        a = previous / previous.sum()
        print("[init] using previous construction")
    else:
        # build a small pool of analytic seeds, keep the best one
        seeds = _make_extended_seeds(n, previous)
        vals = [(s, _objective(s)) for s in seeds]
        vals.sort(key=lambda x: x[1])
        a, best_val = vals[0]
        print(f"[init] best analytic seed value={best_val:.6f}")

    best_seq = a.copy()
    best_val = _objective(a)

    # ------------------------------------------------------------------
    # Pre‑compute the sparsity pattern of the convolution matrix (once)
    # ------------------------------------------------------------------
    if SCIPY_AVAILABLE:
        row_idx = np.empty(n * n, dtype=np.int64)
        col_idx = np.empty(n * n, dtype=np.int64)
        pos = 0
        for col in range(n):
            rows = np.arange(col, col + n, dtype=np.int64)
            row_idx[pos:pos + n] = rows
            col_idx[pos:pos + n] = col
            pos += n
    else:
        row_idx = col_idx = None  # will not be used

    # ------------------------------------------------------------------
    # 1) Linear‑programming driven refinement (as much as time permits)
    # ------------------------------------------------------------------
    lp_deadline = deadline - 200.0   # reserve ~200 s for final polish
    it = 0
    while SCIPY_AVAILABLE and time.time() < lp_deadline:
        it += 1
        conv = _conv_fft(a, a)
        M = conv.max()
        g, improved = _lp_step(a, M, row_idx, col_idx, lp_deadline)
        if not improved:
            print("[LP] no further improvement possible – breaking")
            break

        # try to replace a with a mixture of a and g; start with full replacement
        tau = 1.0
        success = False
        for _ in range(5):
            cand = (1.0 - tau) * a + tau * g
            cand = (cand + cand[::-1]) / 2.0                # enforce symmetry
            cand /= cand.sum()
            new_M = _conv_fft(cand, cand).max()
            if new_M < M * (1.0 - 1e-9):                     # strict improvement
                a = cand
                best_seq = a.copy()
                best_val = _objective(a)
                print(f"[LP] iter={it:3d} τ={tau:.3f}  obj={best_val:.6f}")
                success = True
                break
            tau *= 0.5
        if not success:
            # cannot improve with this LP solution – abort LP phase
            print("[LP] mixing failed to improve – aborting LP phase")
            break

    # ------------------------------------------------------------------
    # 2) Light exponentiated‑gradient polish (uses remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remain = deadline - time.time()
        print("[smooth] polishing with EG")
        a_smooth, val_smooth = _smooth_expgd(best_seq,
                                             deadline=deadline,
                                             max_iters=2_000_000)
        if val_smooth < best_val:
            best_seq = a_smooth
            best_val = val_smooth
            print(f"[smooth] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Final sanity – ensure non‑negativity & normalisation
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    if best_seq.sum() == 0.0:
        best_seq = _uniform(n)
    else:
        best_seq /= best_seq.sum()

    final_obj = _objective(best_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
