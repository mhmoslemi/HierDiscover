"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Initial seeds** – try a handful of analytic families (uniform,
   arcsine, Cauchy‑like) for several lengths. Keep the best.

2. **Systematic Beta‑grid** – for each length run a dense grid over the
   Beta(α,β) family (α,β ∈ [0.05,5]) – this family includes the
   arcsine distribution and many U‑shaped densities that flatten the
   self‑convolution. The best configuration is kept.

3. **Heavy random sampling** – draw many Dirichlet vectors with a
   variety of concentration parameters and many random mixtures of two
   Beta pdfs.  Accept improvements.

4. **Long mirror‑descent polishing** – starting from the current best
   candidate run a KL‑mirror‑descent on the exact max‑convolution
   objective.  The tolerance for “active’’ convolution entries is
   decreased exponentially from 10 % down to 10⁻⁹, while the
   exponentiated‑gradient step size decays as 1/√t.  The routine runs
   until the time budget is exhausted.

5. **Optional L‑BFGS refinement** – if SciPy is available, a final
   L‑BFGS‑B optimisation on the smooth log‑sum‑exp approximation of the
   max is performed for a few seconds.

All phases respect the global time budget; the best sequence found is
returned as a plain ``list`` of non‑negative numbers (already normalised
to sum = 1).  The result is stored in the global name
``height_sequence_1`` for possible reuse by subsequent runs.
"""

import time
import numpy as np
from math import sqrt, pi, log, exp
from typing import List

# ---------------------------------------------------------------
# Low‑level convolution utilities (FFT, objective)
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact ``evaluate_sequence`` objective for a *normalised* sequence
    (sum == 1).  Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ---------------------------------------------------------------
# Analytic families (all returned normalised to sum = 1)
# ---------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete approx. of the arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total


def _cauchy_shape(n: int, scale_factor: float = 0.15) -> np.ndarray:
    """Cauchy‑like bump centred in the interval."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = scale_factor * n
    w = 1.0 / (1.0 + ((i - centre) / sigma) ** 2)
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta pdf on a discrete grid (shifted by 0.5 to avoid the exact ends)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total


# ---------------------------------------------------------------
# Random generation utilities
# ---------------------------------------------------------------

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet‑like vector (Gamma/Exponential) with concentration α."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


def _random_beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.uniform(0.05, 5.0)
    b = rng.uniform(0.05, 5.0)
    return _beta_shape(n, a, b)


def _random_mixture_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two independent Beta pdfs with a random weight."""
    w = rng.random()
    a1, b1 = rng.uniform(0.05, 5.0), rng.uniform(0.05, 5.0)
    a2, b2 = rng.uniform(0.05, 5.0), rng.uniform(0.05, 5.0)
    s1 = _beta_shape(n, a1, b1)
    s2 = _beta_shape(n, a2, b2)
    mix = w * s1 + (1.0 - w) * s2
    mix /= mix.sum()
    return mix


# ---------------------------------------------------------------
# Aggressive mirror‑descent polishing (exact max)
# ---------------------------------------------------------------

def _mirror_descent(
    start_seq: np.ndarray,
    deadline: float,
    eps_rel_start: float = 0.1,
    eps_rel_end: float = 1e-9,
    lr_init: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent that minimises the exact max‑convolution objective.
    Active set = all indices whose convolution value lies within a
    relative tolerance of the current maximum.  The tolerance decays
    exponentially from ``eps_rel_start`` to ``eps_rel_end``.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    start_time = time.time()
    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1

        # full self‑convolution
        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        # decay of relative tolerance
        elapsed = time.time() - start_time
        total = deadline - start_time
        fraction = elapsed / total if total > 0 else 0.0
        eps_rel = eps_rel_start * (eps_rel_end / eps_rel_start) ** fraction
        delta = eps_rel * max_val

        # binary mask of active indices
        active = (conv >= max_val - delta).astype(np.float64)

        # aggregated sub‑gradient = 2 * (active ⋆ a_rev)
        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        # exponentiated‑gradient step (KL‑mirror)
        lr = lr_init / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        # periodic exact evaluation
        if (it & 1023) == 0:           # every 1024 iterations
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Mirror] it={it:7d} n={n:5d} best={val:.6f}")

            # early stop if we are very close to the deadline
            if time.time() > deadline - 0.5:
                break

    return best_seq, best_val


# ---------------------------------------------------------------
# Optional L‑BFGS refinement (smooth max)
# ---------------------------------------------------------------

try:
    from scipy.optimize import minimize
except Exception:  # pragma: no cover
    minimize = None


def _lbfgs_refine(start_seq: np.ndarray, deadline: float, beta: float = 2000.0) -> tuple[np.ndarray, float]:
    """
    Run a single L‑BFGS‑B optimisation on the smooth log‑sum‑exp
    approximation of the max‑convolution.  Returns the refined sequence
    and its exact objective.
    """
    if minimize is None:
        return start_seq, _objective(start_seq)

    n = len(start_seq)
    eps = 1e-30

    def fun_and_grad(logits: np.ndarray):
        a = np.exp(logits)
        a /= a.sum()
        conv = _conv_fft(a, a)
        c = np.max(conv)
        exps = np.exp(beta * (conv - c))
        lse = c + (1.0 / beta) * np.log(exps.sum())
        value = 2.0 * n * lse

        # gradient w.r.t. a
        p = exps / exps.sum()
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)
        return value, grad_logits.astype(np.float64)

    x0 = np.log(start_seq + eps)

    class TimeOut(RuntimeError):
        pass

    def callback(xk):
        if time.time() >= deadline:
            raise TimeOut()

    try:
        res = minimize(
            fun=lambda x: fun_and_grad(x)[0],
            x0=x0,
            jac=lambda x: fun_and_grad(x)[1],
            method='L-BFGS-B',
            options={'maxiter': 250, 'disp': False},
            callback=callback,
        )
    except TimeOut:
        best_logits = x0
    else:
        best_logits = res.x

    best_a = np.exp(best_logits)
    best_a /= best_a.sum()
    return best_a, _objective(best_a)

            print(f"[Rnd-Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        # Random Beta shape
        seq = _random_beta_shape(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd-Beta] n={n:5d} val={val:.6f}")

        # Random mixture of Betas
        seq = _random_mixture_beta(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd-Mix] n={n:5d} val={val:.6f}")

        if best_val < 1.45:
            break

    # ------------------------------------------------------------------
    # Phase 4 – aggressive mirror‑descent polishing (majority of time)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # make a copy to avoid accidental in‑place changes
        seq = best_seq.copy()
        final_seq, final_val = _mirror_descent(
            start_seq=seq,
            deadline=deadline,
            eps_rel_start=0.1,
            eps_rel_end=1e-9,
            lr_init=0.5,
        )
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Mirror‑Done] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – optional L‑BFGS refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0 and minimize is not None:
            lbfgs_deadline = time.time() + min(remaining * 0.20, 30.0)
            seq_lbfgs, val_lbfgs = _lbfgs_refine(best_seq, lbfgs_deadline, beta=2000.0)
            if val_lbfgs < best_val:
                best_val = val_lbfgs
                best_seq = seq_lbfgs
                print(f"[L‑BFGS] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will evaluate again anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
