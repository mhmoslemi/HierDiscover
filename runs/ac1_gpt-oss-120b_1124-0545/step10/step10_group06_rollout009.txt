def _lp_step(f, target_sum, time_deadline):
    """
    Given current sequence f (sum = target_sum), compute a new g via linear programming:
    maximize sum(g) subject to nonnegativity and conv(f,g) <= max_conv_f.
    Return normalized g (sum = target_sum).
    """
    n = f.size
    M = np.max(_conv_fft(f, f))
    # Build sparse matrix A (2n-1 x n)
    from scipy.sparse import diags, csr_matrix
    data = []
    offsets = []
    for d in range(n):
        # Place f[d] on offset d diagonal
        data.append(np.full(n, f[d]))
        offsets.append(d)
    A = diags(data, offsets, shape=(2*n-1, n), format='csr')
    # Bounds: each variable >=0
    bounds = [(0, None)] * n
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    # call linear program
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': max(0.0, time_deadline - time.time())})
    if not res.success:
        # fallback: return f unchanged
        return f
    g = res.x
    sum_g = g.sum()
    if sum_g < 1e-12:
        return f
    # normalize to target sum
    g = g * (target_sum / sum_g)
    return g

"""
Propose a step‑function sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The search combines three complementary optimisation stages:

* **Stage 0 – Linear‑programming (LP) sweep**.
  Starting from a feasible simplex point we solve a small LP
  ``max sum(b) s.t. b≥0,  conv(f,b) ≤ M`` (M = max(conv(f,f))).
  The LP‑solution is mixed with the current point (convex combination).
  The LP matrix is built as a sparse Toeplitz convolution matrix
  (only (2n‑1) × n non‑zeros).

* **Stage 1 – Smooth Adam on the simplex**.
  We optimise the smoothed objective
  ``softmax_λ(conv(f,f))`` using an exponentiated‑parameterisation.
  The smoothing parameter λ starts small (≈0.5) for a smooth landscape
  and is multiplied by a factor >1 each iteration so that the optimisation
  gradually focuses on the true non‑smooth max‑convolution.

* **Stage 2 – Simulated‑annealing polishing**.
  A light Metropolis–Hastings walk swaps a tiny amount of mass between
  random coordinates and occasionally performs a global Gaussian jump.

* **Stage 3 – Hard‑max sub‑gradient descent with simplex projection**.
  The exact sub‑gradient of the true max‑convolution is used and the
  iterate is projected back onto the probability simplex after each step.

The routine respects a 1000 s time budget, keeps the best sequence in the
global variable ``height_sequence_1`` and returns it as a plain list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact evaluation identical to ``evaluate_sequence`` (ignores clipping)."""
    s = seq.sum()
    if s < 1e-12:
        return float('inf')
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple seed generators (give a probability vector that sums to 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = logits.max()
    exps = np.exp(logits - m)
    return exps / exps.sum()


def _exponential_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        dec = inc[::-1]
    else:
        dec = np.arange(half, 0, -1, dtype=np.float64)
    a = np.concatenate([inc, dec])
    return a / a.sum()


# ----------------------------------------------------------------------
# Soft‑max weight over the convolution entries (used by Adam)
# ----------------------------------------------------------------------
def _soft_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Return a probability vector of the same length as ``conv``.
    For lam>0 this is a soft‑max, for lam≤0 it is an (uniform) hard‑max."""
    max_c = conv.max()
    if lam <= 0.0:
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.where(active, 1.0 / active.sum(), 0.0)
        return w
    # relative (≤0) to improve numerical stability
    rel = (conv - max_c) / max_c
    exps = np.exp(lam * rel)
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Gradient of the (soft) max‑convolution w.r.t. the probability vector ``a``
# ----------------------------------------------------------------------
def _gradient_wrt_a(a: np.ndarray, lam: float) -> np.ndarray:
    """Gradient of the smoothed max‑convolution (soft‑max weight λ)."""
    n = a.size
    conv = _conv_fft(a, a)                     # length 2n‑1
    w = _soft_max_weights(conv, lam)           # same length
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n]         # slice matching a
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (exponentiated‑gradient parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 500_000,
                   lam_start: float = 0.5,
                   lam_factor: float = 1.002) -> tuple[np.ndarray, float]:
    """Adam on log‑probabilities with a gradually sharpening soft‑max."""
    eps = 1e-30
    logp = np.log(start + eps)                # initialise log‑probabilities
    n = start.size
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = start.copy()
    best_val = _objective(start)

    t = 0
    lam = lam_start

    while t < max_iters and time.time() < deadline:
        t += 1

        # ---- current simplex point ------------------------------------
        logp_shift = logp - np.max(logp)
        a = np.exp(logp_shift)
        a /= a.sum()

        # ---- smoothed gradient ----------------------------------------
        g_a = _gradient_wrt_a(a, lam)

        # chain rule to log‑space
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)

        # ---- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log * g_log)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.15 / (math.sqrt(t) + 1.0)

        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- evaluate true (hard) objective ---------------------------
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:7d}  best={cur_val:.6f}  λ={lam:.3f}")

        lam *= lam_factor

        # occasional tiny random kick to escape plateaus
        if t % 100_000 == 0:
            noise = np.random.default_rng().normal(scale=0.01, size=n)
            logp += noise

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light simulated‑annealing neighbour search (final polishing)
# ----------------------------------------------------------------------
def _anneal_refine(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """Metropolis–Hastings walk that swaps small mass fractions."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0
    n = cur.size

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d}  best={new_val:.6f}")

        # periodic global perturbation
        if it % 3000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0.0:
                new_seq /= total
            else:
                new_seq = _uniform(n)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of ``v`` onto the simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Hard‑max sub‑gradient descent (projected onto simplex)
# ----------------------------------------------------------------------
def _subgradient_descent(start: np.ndarray,
                         deadline: float,
                         max_iters: int = 150_000,
                         step0: float = 0.5) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the exact max‑convolution objective."""
    a = start.copy()
    best = a.copy()
    best_val = _objective(a)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # gradient of the true (hard) max‑convolution
        grad = _gradient_wrt_a(a, lam=0.0)

        eta = step0 / math.sqrt(t)
        a = a - eta * grad
        a = _project_onto_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[Subgrad] it={t:7d}  best={cur_val:.6f}")

        # occasional tiny random shake to escape tiny plateaus
        if t % 30_000 == 0:
            a += np.random.default_rng().normal(scale=0.0005, size=a.shape)
            a = _project_onto_simplex(a)

    return best, best_val


# ----------------------------------------------------------------------
# Linear‑programming sweep (maximise sum(g) s.t. conv(f,g) ≤ M)
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, target_sum: float) -> np.ndarray:
    """One LP step: maximise total mass of ``g`` with conv(f,g) ≤ M.
    Returns ``g`` normalised to ``target_sum`` (or the original ``f`` on failure)."""
    from scipy.sparse import diags, csr_matrix
    from scipy.optimize import linprog

    n = f.size
    M = np.max(_conv_fft(f, f))

    # Build sparse Toeplitz matrix A (shape (2n‑1, n))
    # A_{k,j} = f_{k‑j} if 0 ≤ k‑j < n else 0   (i.e. offset = k‑j)
    data = []
    offsets = []
    for d in range(n):
        data.append(np.full(n, f[d], dtype=np.float64))
        offsets.append(d)
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')

    c = -np.ones(n, dtype=np.float64)          # maximise sum(g)  ⇒ minimise -sum(g)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success or res.x is None:
        # fall back to the input vector
        return f.copy()

    g = res.x
    s = g.sum()
    if s < 1e-12:
        return f.copy()
    # Normalise to the required simplex sum
    g = g * (target_sum / s)
    return g


def _lp_iterative_refine(f: np.ndarray,
                         target_sum: float,
                         steps: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Run a few LP‑updates, each time mixing the result with the current point."""
    best = f.copy()
    best_val = _objective(f)
    t = 0
    while t < steps and time.time() < deadline:
        t += 1
        g = _lp_step(f, target_sum)
        # convex combination (mixing coefficient)
        mix = 0.3
        f = (1.0 - mix) * f + mix * g   # both already have sum = target_sum
        cur_val = _objective(f)
        if cur_val < best_val:
            best_val = cur_val
            best = f.copy()
            print(f"[LP] step={t:2d}  best={cur_val:.6f}")
    return best, best_val
