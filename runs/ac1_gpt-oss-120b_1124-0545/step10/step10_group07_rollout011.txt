"""
Propose a step‑function (non‑negative sequence) with a low
evaluate_sequence() score.

Algorithm
---------
1. Load the previous best construction (global ``height_sequence_1``) or
   build a small diverse set of analytic seeds (uniform, Dirichlet,
   geometric, power‑law, sine/Hann windows).  Keep the seed with the
   lowest exact objective.

2. **Targeted max‑convolution descent** – repeatedly
   * compute the full autocorrelation (a*a) via an FFT,
   * locate the lag where the maximum occurs,
   * compute the exact gradient of that single lag
     (a cheap O(n) operation),
   * take the largest feasible step that keeps all entries non‑negative,
     renormalise to sum = 1 and repeat.
   This monotone descent quickly reduces the peak of the
   autocorrelation, which directly minimises the objective
   ``2·n·max(conv) / (sum(a)^2)``.
   The loop runs until the time budget expires, printing progress
   whenever a new best value is found.  Occasionally a tiny random
   jitter is injected to escape plateaus.

3. A short simulated‑annealing fine‑tune (using the same moves as the
   previous hybrid optimiser) consumes any remaining time.

The code respects the 1000 s budget, never returns ``None``, and stores
the best sequence in ``height_sequence_1`` for reuse by later runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (full linear convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective (exact, no clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    max_conv = conv.max()
    return 2.0 * seq.size * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic seed families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `alpha`."""
    x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Targeted max‑convolution descent
# ----------------------------------------------------------------------


def _max_conv_descent(start_seq: np.ndarray,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Iteratively reduces the peak of the autocorrelation (a*a).
    Each step moves a in the negative gradient of the currently
    maximal lag, clipping negatives and re‑normalising.
    """
    a = start_seq.copy()
    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)

    rng = np.random.default_rng()
    it = 0
    last_improve = 0

    while time.time() < deadline:
        # full autocorrelation and its maximal lag
        conv = _conv_fft(a, a)
        max_val = conv.max()
        k = int(conv.argmax())               # lag of the current peak
        cur_val = 2.0 * n * max_val / (a.sum() ** 2)  # exact objective

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[max_descent] it={it:,} best={best_val:.6f}")
            last_improve = it

        # gradient of the *single* maximal lag (cheap O(n) operation)
        idx = k - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[idx[valid]]

        pos = g > 0
        if not np.any(pos):
            break

        # largest step that keeps a >= 0 (a_i - α g_i >= 0)
        steps = a[pos] / g[pos]
        alpha_max = steps.min()
        if alpha_max <= 0:
            break

        # conservative step (half the maximum admissible)
        alpha = 0.5 * alpha_max

        # take the step, clip negatives and renormalise
        a = a - alpha * g
        np.maximum(a, 0.0, out=a)
        tot = a.sum()
        if tot > 0:
            a /= tot
        else:
            a = _uniform(n)

        # occasional exploration if progress stalls
        if it - last_improve > 5000:
            jitter = rng.normal(scale=1e-3, size=n)
            a = np.clip(a + jitter, 0.0, None)
            tot = a.sum()
            if tot > 0:
                a /= tot
            else:
                a = _uniform(n)
            last_improve = it  # reset stall counter

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight simulated annealing (kept from the previous hybrid)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Metropolis–Hastings walk with a few simple moves."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-7, cur_val * 0.07)
    decay = 0.9994
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        rnd = rng.random()
        if rnd < 0.68:                                   # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = rng.random() * 0.18 * cur[i]
                new_seq = cur.copy()
                new_seq[i] -= delta
                new_seq[j] += delta
            else:
                new_seq = cur.copy()
        elif rnd < 0.86:                                 # swap
            i, j = rng.integers(0, n, size=2)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:                                            # global jitter
            noise = rng.normal(scale=0.0013, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot > 0:
                new_seq /= tot
            else:
                new_seq = _uniform(n)

        new_seq = _project_simplex(new_seq)
        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:,} best={best_val:.6f}")

        if it % 2500 == 0:
            # occasional stronger jump from the current best
            cand = best + rng.normal(scale=0.0035, size=n)
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand)
            cand_val = _objective(cand)
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
                print(f"[Anneal] (jump) it={it:,} best={cand_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found as a plain
    list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length – keep the same as the historic best (4000) unless a
    # different‑sized previous construction is present.
    # ------------------------------------------------------------------
    n = 4000
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.ndim == 1 and previous.size > 0:
                if previous.size != n:
                    n = int(previous.size)
            else:
                previous = None
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 1) Build a modest pool of seeds and keep the best one.
    # ------------------------------------------------------------------
    rng = np.random.default_rng(12345)
    seeds = []

    if previous is not None:
        seeds.append(previous.copy())
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_hann_window(n))
    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0]:
        seeds.append(_random_dirichlet(n, alpha, rng))
    for r in [0.98, 0.985, 0.99, 0.992]:
        seeds.append(_geom(n, r))
    for p in [0.5, 1.0, 1.5, 2.0]:
        seeds.append(_power(n, p))

    # normalise and clean
    clean = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            clean.append(s / tot)
    seeds = clean

    best_seq = None
    best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seq = s.copy()
    print(f"[init] best seed objective {best_val:.6f} (length {len(best_seq)})")

    # ------------------------------------------------------------------
    # 2) Targeted max‑convolution descent – the core optimiser.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n).copy()
        best_val = _objective(best_seq)

    seq_descent, val_descent = _max_conv_descent(best_seq, deadline)
    if val_descent < best_val:
        best_val = val_descent
        best_seq = seq_descent.copy()
        print(f"[max_descent] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Tiny simulated‑annealing polish if time permits.
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline - 1.0:
        seq_ann, val_ann = _anneal_optimize(best_seq, n, deadline, max_iters=None)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = seq_ann.copy()
            print(f"[anneal] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best construction found.
    # ------------------------------------------------------------------
    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
