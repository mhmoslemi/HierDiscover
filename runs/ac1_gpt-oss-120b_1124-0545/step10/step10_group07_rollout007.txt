```python
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def solve_lp_for_b(a, M):
    n = len(a)
    # Build constraint matrix A_ub (2n-1) x n
    # For k from 0 to 2n-2:
    # For j from max(0, k-(n-1)) to min(n-1, k):
    # coefficient of b_j = a_{k - j}
    # Row of size n with zeros elsewhere.

    data = []
    rows = []
    cols = []
    for k in range(2*n-1):
        # b_j contributes a_{k-j} if 0 <= k-j < n
        j_start = max(0, k-(n-1))
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            i = k - j  # index in a
            coeff = a[i]
            rows.append(k)
            cols.append(j)
            data.append(coeff)
    A_ub = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    # bounds for b: (0, None)
    # objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)

    # Use high-level solver
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        b = res.x
        return b
    else:
        # fallback
        return None
```

```python
    n = len(a)
    # Build Ao: For each k, we need conv constraint: sum_i a_i * b_{k-i} <= M
    # We'll build using sparse matrix CSR.
    row_inds = []
    col_inds = []
    data = []
    for k in range(2*n-1):
        # the overlapping region:
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # for each i, b_{k-i} index j = k - i
        # coefficient = a[i]
        # So j ranges from k - i_max to k - i_min.
        # Equivalent: for j from max(0, k-(n-1)) ... min(n-1, k)
        # coefficient = a[k-j]
        for j in range(i_min, i_max+1):
            i = k - j
            row_inds.append(k)
            col_inds.append(j)
            data.append(a[i])
    A_ub = csr_matrix((data, (row_inds, col_inds)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)

    c = -np.ones(n)  # maximize sum(b)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        b = res.x
        # Scale to target sum
        s = b.sum()
        if s > 0:
            b = b * (target_sum / s)
        return b
    else:
        return None
```

def _lp_step(a, M, target_sum):
    n = a.shape[0]
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # j from i_start ... i_end is index for b
        # a_i coefficient where i = k - j
        for j in range(i_start, i_end + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A_ub = csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    b_ub = np.full(2*n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    s = b.sum()
    if s > 0:
        b = b * (target_sum / s)
    return b

"""
Hybrid optimizer with a dedicated linear‑programming (LP) refinement.

The algorithm proceeds as follows:

1. **Seed generation** – create a rich pool of analytic “probability”
   vectors (uniform, geometric, power‑law, classic windows, random
   Dirichlet/soft‑max draws).  All seeds are normalised to sum = 1 and
   the best few are evaluated with the exact objective
   `2·n·max(conv(a,a))/sum(a)²`.

2. **LP‑mixing loop** – starting from the best seed (scaled so that
   `sum(a)=√(2n)`, which makes the objective equal to `max(conv(a,a))`),
   solve a linear program

        maximise   Σ b_j
        subject to  b_j ≥ 0,
                    (a * b)[k] ≤ M   for every k = 0…2n‑2

   where `M = max(conv(a,a))`.  The LP yields a vector `b` that places
   as much mass as possible in locations where `a` is small while
   keeping the convolution bound.  After scaling `b` to the same total
   mass as `a`, we mix the two vectors (`a←(1‑t)·a + t·b`) with a
   decreasing step size `t`.  Repeating this step quickly drives the
   maximal convolution down.

3. **Fine‑grained simulated annealing** – the best vector from the LP
   phase is refined by a Metropolis–Hastings walk with mass‑transfer,
   swap and jitter moves.  The annealer runs until the time budget is
   exhausted.

All steps respect the 1000 s budget, use only non‑negative numbers and
return a plain ``list[float]`` suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (required for the LP step)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    SCIPY_OPT = True
except Exception:  # pragma: no cover
    SCIPY_OPT = False

# Basic utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2).
    The function is scale‑invariant; we do *not* normalise here.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector: sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet with concentration `alpha`."""
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _make_extended_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse set of start vectors (all normalised to sum = 1).
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    # geometric – heavy sweep
    for c in np.linspace(0.5, 12.0, 48):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power law – finer sweep
    for p in np.linspace(0.0, 4.0, 41):
        seeds.append(_power(n, p))

    # combined geometric‑power
    for c in np.linspace(0.5, 12.0, 24):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 9):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 12.0, 24):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classic windows
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 8):
        seeds.append(_tukey_window(n, alpha))
    # Gaussian family – finer sweep
    for sigma_factor in np.linspace(0.06, 0.30, 13):
        seeds.append(_gaussian(n, sigma_factor))
    # Beta‑pdf family
    for alpha in np.linspace(0.3, 2.5, 23):
        seeds.append(_beta_window(n, alpha))

    # Dirichlet draws
    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(4):
            seeds.append(_random_dirichlet(n, alpha, rng))

    # random soft‑max vectors
    for _ in range(10):
        seeds.append(_random_softmax(n, rng))

    # reuse previous construction, if compatible
    if previous is not None and previous.shape == (n,):
        # normalise to sum = 1 before adding
        prev = previous.clip(min=0.0)
        tot = prev.sum()
        if tot > 0:
            seeds.append(prev / tot)

    # final clean‑up
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    return cleaned


# ----------------------------------------------------------------------
# Linear‑programming refinement (the new component)
# ----------------------------------------------------------------------


def _lp_step(a: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ b_j
        subject to  b_j ≥ 0,
                    (a * b)[k] ≤ M      for all k.
    The returned vector is rescales to sum = target_sum.
    """
    if not SCIPY_OPT:
        return None

    n = a.shape[0]
    # Number of constraints = 2n‑1  (full linear convolution length)
    rows = []
    cols = []
    data = []

    # Build the Toeplitz‐like constraint matrix on the fly.
    for k in range(2 * n - 1):
        # valid indices j where 0 ≤ j < n and 0 ≤ k‑j < n
        j_low = max(0, k - (n - 1))
        j_high = min(n - 1, k)
        # (k‑j) ranges over the corresponding a‑indices
        for j in range(j_low, j_high + 1):
            i = k - j          # index in a
            rows.append(k)
            cols.append(j)
            data.append(a[i])

    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)

    # maximise Σ b_j  →  minimise -Σ b_j
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None) for _ in range(n)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:  # pragma: no cover
        # any numerical issue – give up on the LP step
        return None

    if not res.success:
        return None

    b = res.x
    s = b.sum()
    if s > 0:
        b = b * (target_sum / s)
    else:
        # fallback to uniform if something went awry
        b = np.full(n, target_sum / n, dtype=np.float64)

    # Enforce the absolute clip that evaluate_sequence applies (≤1000)
    b = np.clip(b, 0.0, 1000.0)
    # Re‑normalise to exact target_sum after clipping
    b = b * (target_sum / b.sum())
    return b

# Simulated‑annealing fine‑tuning (as in the previous code)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Metropolis–Hastings walk with several move types.
    Returns the best vector found and its objective.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature – a modest fraction of current objective
    T0 = max(1e-7, cur_val * 0.07)
    decay = 0.9994

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        rnd = rng.random()
        if rnd < 0.68:  # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.random() * 0.18 * cur[i]   # 0‑18 % of source mass
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        elif rnd < 0.86:  # swap
            i, j = rng.integers(0, n, size=2)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:  # global jitter
            noise = rng.normal(scale=0.0013, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0:
                new_seq = new_seq / total
            else:
                new_seq = np.full(n, 1.0 / n, dtype=np.float64)

        # always stay on the simplex
        new_seq = _project_simplex(new_seq) * (cur.sum() / new_seq.sum())
        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:8d} best={new_val:.6f}")

        # occasional strong jump to escape plateaus
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.0035, size=n)
            cand = best + jitter
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand) * (best.sum() / cand.sum())
            cand_val = _objective(cand)
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
                print(f"[Anneal] (jump) it={it:8d} best={cand_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a safety margin
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – a compromise between expressiveness and LP speed
    # ------------------------------------------------------------------
    n = 2000                      # length of the step function
    target_sum = math.sqrt(2.0 * n)   # sum = √(2n) ⇒ objective = max(conv)

    # ------------------------------------------------------------------
    # Load a previously stored construction if it matches the current n
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                previous = prev_arr
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # 0) Evaluate previous construction (if any) – keep it if it beats seeds
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    if previous is not None:
        # Scale to the required total mass
        prev_scaled = previous * (target_sum / previous.sum())
        val = _objective(prev_scaled)
        best_seq = prev_scaled.copy()
        best_val = val
        print(f"[prev] value={val:.6f}")

    # ------------------------------------------------------------------
    # 1) Generate analytic seeds and keep the best few
    # ------------------------------------------------------------------
    seeds = _make_extended_seeds(n, previous)
    seed_vals = []
    for s in seeds:
        # scale to target sum (scale invariance of the objective)
        s_scaled = s * target_sum
        val = _objective(s_scaled)
        seed_vals.append((s_scaled, val))
    seed_vals.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    if seed_vals and seed_vals[0][1] < best_val:
        best_seq = seed_vals[0][0].copy()
        best_val = seed_vals[0][1]
        print(f"[seed] new best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) LP‑mixing refinement (the new core)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to uniform if everything failed
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_val = _objective(best_seq)

    # Time budget for the LP phase
    lp_time = (deadline - time.time()) * 0.50  # half of remaining time
    lp_deadline = time.time() + lp_time
    iteration = 0
    t_initial = 0.30   # initial mixing coefficient

    while time.time() < lp_deadline:
        iteration += 1
        a = best_seq
        conv = _conv_fft(a, a)
        M = conv.max()
        # Solve the LP to obtain a complementary vector b
        b = _lp_step(a, M, target_sum)
        if b is None:
            print("[LP] LP step failed – exiting LP phase")
            break

        # Mixing schedule – decreasing step size
        t = t_initial / math.sqrt(iteration)
        new_a = (1.0 - t) * a + t * b
        # Re‑normalise to exact target_sum (tiny rounding errors may accumulate)
        new_a = new_a * (target_sum / new_a.sum())

        new_val = _objective(new_a)
        if new_val < best_val:
            best_val = new_val
            best_seq = new_a.copy()
            print(f"[LP] iter={iteration:3d} improved best={new_val:.6f}")

        # Early stopping if progress stalls
        if iteration >= 15:
            # limit the number of LP iterations to keep runtime reasonable
            break

    # ------------------------------------------------------------------
    # 3) Simulated‑annealing fine‑tuning (uses the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[anneal] starting simulated‑annealing fine‑tuning")
        seq_ann, val_ann = _anneal_optimize(best_seq, n, deadline, max_iters=None)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = seq_ann.copy()
            print(f"[anneal] improved best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # 4) Final safety‑check and export
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_val = _objective(best_seq)

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    globals()["height_sequence_1"] = best_seq.copy()
    # Convert to plain Python list (evaluate_sequence will cast to floats)
    return best_seq.tolist()
