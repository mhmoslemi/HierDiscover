"""
Propose a step‐function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The method is a hybrid optimiser:

* **Sub‑gradient descent on the exact max‑convolution** – we directly minimise
  the worst autocorrelation entry.  The sub‑gradient is obtained analytically
  (the derivative of a single active convolution index is a cheap
  O(n) operation).  After each gradient step the vector is projected onto
  the probability simplex (non‑negative, sum=1).

* **Adam optimiser with a smoothed max** – after the sub‑gradient phase we
  run the smooth‑Adam optimiser from the previous answer (the implementation
  from the reference solution) for a fine‑grained refinement.

* **Light simulated‑annealing polishing** – a short Metropolis walk to escape
  any remaining shallow local minima.

Multiple random restarts are performed; the best sequence found before the
time budget expires is returned.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# FFT utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact evaluation (same formula as evaluate_sequence)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (smooth max → hard max)
# ----------------------------------------------------------------------
def _soft_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights over `conv` with inverse temperature `lam`."""
    max_c = conv.max()
    if lam <= 0.0:
        active = np.isclose(conv, max_c, atol=1e-12)
        cnt = np.count_nonzero(active)
        return np.where(active, 1.0 / cnt, 0.0)
    rel = (conv - max_c) / max_c          # ≤ 0
    exps = np.exp(lam * rel)
    return exps / exps.sum()


def _gradient_wrt_a(a: np.ndarray, lam: float) -> np.ndarray:
    """Gradient of the smoothed max‑convolution w.r.t. probability vector `a`."""
    conv = _conv_fft(a, a)
    w = _soft_max_weights(conv, lam)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length = n + (2n‑1) – 1 = 3n‑2
    n = a.size
    return 2.0 * full[n - 1:n - 1 + n]        # slice that matches `a`


def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 300_000,
                   lam_start: float = 5.0,
                   lam_factor: float = 1.001) -> tuple[np.ndarray, float]:
    """Adam with soft‑max smoothing (same as the reference implementation)."""
    eps = 1e-30
    logp = np.log(start + eps)
    n = start.size
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = start.copy()
    best_val = _objective(start)

    t = 0
    lam = lam_start
    while t < max_iters and time.time() < deadline:
        t += 1
        # current probability vector (soft‑max)
        logp_shift = logp - np.max(logp)
        a = np.exp(logp_shift)
        a /= a.sum()

        g_a = _gradient_wrt_a(a, lam)                # gradient w.r.t. a
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)                      # chain rule

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log * g_log)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.15 / (math.sqrt(t) + 1.0)
        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:7d}  best={cur_val:.6f}  λ={lam:.2f}")

        lam *= lam_factor

        # occasional tiny perturbation to break plateaus
        if t % 25000 == 0:
            noise = np.random.default_rng().normal(scale=0.02, size=n)
            logp += noise

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very light simulated‑annealing neighbour search (final polishing)
# ----------------------------------------------------------------------
def _anneal_refine(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """A cheap Metropolis‑Hastings walk that swaps a small fraction of mass."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    n = cur.size
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val, best = new_val, new_seq.copy()
                print(f"[Anneal] it={it:7d}  best={new_val:.6f}")

        if it % 2000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot > 0.0:
                new_seq /= tot
            else:
                new_seq = np.full(n, 1.0 / n)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val, best = new_val, new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – preserves non‑negativity and sum=1
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex."""
    # shift to make the algorithm robust for negative entries
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Returns a sub‑gradient of max(conv) w.r.t. a.
    The active set are all indices where conv attains its maximum.
    """
    n = a.size
    max_c = conv.max()
    active = np.where(np.isclose(conv, max_c, atol=1e-15, rtol=0.0))[0]

    grad = np.zeros_like(a)
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # vectorised over i
        i = np.arange(i_min, i_max + 1)
        j = k - i
        diag = (i == j)
        # contribution = a_i for diagonal, 2*a_j otherwise
        grad[i] += np.where(diag, a[i], 2.0 * a[j])
    # average over the active set (any convex combination works)
    grad /= max(1, len(active))
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with the smallest evaluation
    value.  The routine respects a time budget (default 1000 s) and returns
    the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final return
    deadline = start_time + budget_s - 2.0

    rng = np.random.default_rng(seed + 12345)

    # ------------------------------------------------------------------
    # Problem size – a balance between expressiveness and speed
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising step function (length {n}) ===")

    # ------------------------------------------------------------------
    # Build a pool of initial seeds (uniform, random, previous if any)
    # ------------------------------------------------------------------
    seeds = []

    # previous solution, if it matches the current length
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                if arr.sum() > 0.0:
                    seeds.append(arr / arr.sum())
        except Exception:
            pass

    # deterministic starters
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))                    # uniform
    seeds.append(_random_softmax_start(n, rng))                          # soft‑max random
    seeds.append(_exponential_decay_start(n, rate=0.001))                # exponential decay
    seeds.append(_triangular_start(n))                                   # triangular

    # ------------------------------------------------------------------
    # Pick the best seed w.r.t. the true objective
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val, best_seq = val, s.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – Sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    phase0_deadline = deadline - 30.0   # reserve ~30 s for later polishing
    if time.time() < phase0_deadline:
        print("[Phase 0] Sub‑gradient descent start")
        a = best_seq.copy()
        step0 = 0.3                      # initial step size
        it = 0
        no_improve = 0
        while time.time() < phase0_deadline:
            it += 1
            conv = _conv_fft(a, a)
            cur_val = 2.0 * n * conv.max()      # sum(a)=1 ⇒ objective = 2·n·maxconv
            if cur_val < best_val - 1e-9:
                best_val, best_seq = cur_val, a.copy()
                print(f"[Phase 0] it={it:7d}  new best={cur_val:.6f}")
                no_improve = 0
            else:
                no_improve += 1

            # compute sub‑gradient of max(conv)
            grad = _subgradient_maxconv(a, conv)

            # simple diminishing step size
            eta = step0 / math.sqrt(it)
            a_try = _project_simplex(a - eta * grad)

            # evaluate trial
            trial_val = 2.0 * n * _conv_fft(a_try, a_try).max()
            if trial_val <= cur_val:
                a = a_try
                if trial_val > cur_val * 0.9999:
                    step0 *= 0.99          # slowly shrink when progress stalls
            else:
                step0 *= 0.5               # reject – reduce step size aggressively
                if step0 < 1e-6:
                    break

            # occasional random kick to escape flat regions
            if no_improve >= 5000:
                noise = rng.normal(scale=0.01, size=n)
                a = _project_simplex(a + noise)
                step0 = max(step0, 0.02)
                no_improve = 0

            if it >= 200_000:
                break

        print(f"[Phase 0] finished – best so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimiser (smooth max) for fine tuning
    # ------------------------------------------------------------------
    if time.time() < deadline - 15.0:
        print("[Phase 1] Adam optimisation (fine tuning)")
        seq_adam, val_adam = _adam_optimize(best_seq, deadline, max_iters=500_000,
                                            lam_start=1.0, lam_factor=1.0005)
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print(f"[Phase 1] improved to {val_adam:.6f}")
        else:
            print("[Phase 1] no improvement")
    else:
        print("[Phase 1] skipped (time)")

    # ------------------------------------------------------------------
    # Phase 2 – Very light simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print("[Phase 2] Simulated‑annealing polishing")
        seq_final, val_final = _anneal_refine(best_seq, deadline)
        if val_final < best_val:
            best_val, best_seq = val_final, seq_final.copy()
            print(f"[Phase 2] improved to {val_final:.6f}")
        else:
            print("[Phase 2] no improvement")
    else:
        print("[Phase 2] skipped (time)")

    # ------------------------------------------------------------------
    # Final normalisation and export
    # ------------------------------------------------------------------
    final_seq = np.maximum(best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        final_seq = final_seq / total

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()


# ----------------------------------------------------------------------
# Helper factories for deterministic seeds (used in initialise)
# ----------------------------------------------------------------------
def _random_softmax_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random soft‑max initialisation."""
    logits = rng.standard_normal(n)
    m = np.max(logits)
    exps = np.exp(logits - m)
    return exps / exps.sum()


def _exponential_decay_start(n: int, rate: float) -> np.ndarray:
    """Deterministic exponential decay (a_i ∝ exp(-rate·i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _triangular_start(n: int) -> np.ndarray:
    """Symmetric triangular shape."""
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    dec = inc[::-1] if n % 2 == 0 else np.arange(half, 0, -1, dtype=np.float64)
    a = np.concatenate([inc, dec])
    return a / a.sum()
