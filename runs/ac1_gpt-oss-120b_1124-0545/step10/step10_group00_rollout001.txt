def _perm_anneal_optimize(start_seq, n, deadline, max_iters=200000):
    # ensure sum=1
    a = start_seq.copy()
    a = a / a.sum()
    cur = a.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    T = cur_val * 0.2  # initial temperature relative to objective
    rng = np.random.default_rng()
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        # cooling schedule
        T = max(T * 0.9999, 1e-12)
        # propose move
        move_type = rng.integers(0,2)  # 0 swap, 1 block reverse
        if move_type == 0:
            i,j = rng.choice(n,size=2,replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        else:
            # block reversal of random segment
            i = rng.integers(0,n-1)
            j = rng.integers(i+1, n)
            new = cur.copy()
            new[i:j] = new[i:j][::-1]
        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                # occasionally print
                if it % 20000 == 0:
                    print(f"[PermAnneal] it={it} best={best_val}")
    return best_seq, best_val

Search for a non‑negative step‑function (a list of heights) that minimises

    2·n·max(conv(seq,seq)) / (sum(seq)²)

within a 1000 s time budget.

The algorithm consists of several stages:

1. **Seed generation** – deterministic analytic shapes, many Dirichlet draws,
   sorted / alternating variants and a few custom exponential families.
2. **Smooth surrogate optimisation** – Adam on log‑softmax parameters,
   using a temperature‑controlled log‑sum‑exp approximation of the true
   max‑convolution.  The temperature is increased gradually so that the
   surrogate converges to the genuine objective.
3. **Permutation annealing** – the values obtained from Adam are kept,
   but their ordering is optimised by a pure‑permutation simulated‑annealing
   that swaps or reverses random blocks.  This exploits the fact that the
   objective depends heavily on the arrangement of large entries.
4. **Exact sub‑gradient polishing** – exponentiated‑gradient (mirror)
   descent on the true max‑convolution value to fine‑tune the final
   sequence.
5. **Final sanity check** – evaluate the produced sequence with the
   official ``evaluate_sequence`` function and store it globally.

All vectors are kept on the probability simplex (ℓ₁‑normalised) – scaling
does not affect the objective.  The implementation stays within the
allowed 2‑CPU limit and respects the 1000 s overall time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for a simplex‑normalised sequence:
        2·n·max(conv(seq,seq))
    (the denominator sum(seq)² equals 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, decay: float) -> np.ndarray:
    """Monotonically decreasing exponential shape."""
    a = np.exp(-decay * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random families & ordering helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)  # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_gaussian(n, sigma_frac=0.40))
    seeds.append(_exponential(n, decay=0.0005))
    seeds.append(_exponential(n, decay=0.001))
    seeds.append(_exponential(n, decay=0.002))

    # power‑law family
    for p in np.linspace(0.0, 4.0, 13):
        seeds.append(_power(n, p))

    # beta family
    for a in np.linspace(0.3, 2.0, 5):
        for b in np.linspace(0.3, 4.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (multiple conc. values)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw order
            seeds.append(_sorted_desc(raw))       # decreasing
            seeds.append(_sorted_asc(raw))        # increasing
            seeds.append(_alternate_order(raw))   # alternating

    # Previous solution if provided
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑softmax parameters using a
    temperature‑controlled log‑sum‑exp surrogate for max(conv).
    The temperature (beta) starts small and is multiplied by a factor
    each iteration up to a large cap, thereby annealing towards the true max.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)               # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8

    beta = 0.5          # initial smoothing temperature
    beta_mul = 1.001    # exponential increase factor
    beta_max = 300.0

    best_seq = start_vec.copy()
    best_val = _objective(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                       # ℓ₁‑normalised probabilities
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)      # exact objective (since sum(a)=1)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # --- smoothed max (log‑sum‑exp) with temperature beta ---
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a  (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

        # convert to log‑space gradient (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature gradually
        if beta < beta_max:
            beta = min(beta * beta_mul, beta_max)

    return best_seq, best_val


def _perm_anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that only permutes the entries (swap / block‑reverse).
    The values themselves are never altered, only their ordering.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)

    best_seq = cur.copy()
    best_val = cur_val

    # Temperature proportional to current objective, slow cooling
    T = cur_val * 0.1 + 1e-12
    cooling = 0.9998

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * cooling, 1e-12)

        # Choose move type
        move_type = rng.integers(0, 2)   # 0=swap, 1=reverse block
        if move_type == 0:
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        else:
            i = rng.integers(0, n - 1)
            j = rng.integers(i + 1, n)
            new = cur.copy()
            new[i:j] = new[i:j][::-1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 20_000 == 0:
                    print(f"[PermAnneal] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _mirror_subgradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the exact
    max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.12  # initial step size factor

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 40_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise onto simplex (positivity already holds)
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats and stores it
    in the global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a few seconds for the final validation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length choice – 4000 strikes a good balance between expressiveness
    # and speed for the heavy convolution calls.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load any previous solution.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – seed generation and quick pruning
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n = {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # Evaluate surrogate (exact) objective for all seeds
    seed_vals = [(vec, _objective(vec)) for vec in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 8
    best_seeds = [vec for vec, _ in seed_vals[:top_k]]
    print(f"Top {top_k} seed objective values:")
    for i, (_, val) in enumerate(seed_vals[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation (smooth surrogate) on the top seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate a modest slice of the remaining budget per Adam run
        max_iters = int(min(200_000, remaining * 60))  # approx 60 it/s
        print(f"\n--- Adam run {idx}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Permutation annealing (pure re‑ordering)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation annealing (ordering optimisation) ---")
        seq_perm, val_perm = _perm_anneal_optimize(global_best_seq, n, deadline,
                                                   max_iters=250_000)
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[overall‑Perm] new global best = {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Exact mirror‑descent polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent polishing (exact max) ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=800_000
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: official evaluate_sequence raised an exception:", exc)
        final_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
