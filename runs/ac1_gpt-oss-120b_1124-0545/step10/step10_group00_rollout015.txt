"""
Propose a step‑function that (hopefully) beats the current best
(≈1.508736) for the official ``evaluate_sequence`` objective.

Approach
--------
1. Build a diverse pool of ℓ₁‑normalised seeds (analytic windows,
   Dirichlet draws, low‑frequency random curves, and the previous best).
2. Quickly screen the pool using a fast FFT‑based surrogate:
       F(seq) = 2·n·max(conv(seq,seq))          (seq is already ℓ₁‑normalised)
   The surrogate is identical to the official objective because the sum is 1.
3. Refine the most promising seeds with a *continuation* L‑BFGS‑B optimiser
   that minimises a smoothed max‑convolution (log‑sum‑exp with increasing
   temperature).  Gradients are obtained analytically, yielding fast
   convergence.
4. Polish the best candidate with a classic simulated‑annealing kernel
   that performs tiny mass‑transfer moves on the simplex.
5. Return the final sequence and store it in the global name
   ``height_sequence_1`` for reuse.

All stages respect the global time budget (default 1000 s).  The code
contains no closures or lambda expressions and works with only NumPy,
SciPy and the supplied ``evaluate_sequence`` function.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (fast O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability‑simplex helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps ℝⁿ → Δⁿ)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _project_to_simplex(a: np.ndarray) -> np.ndarray:
    """Renormalise a non‑negative vector onto the simplex (sum = 1)."""
    s = a.sum()
    if s <= 0.0:
        return np.full_like(a, 1.0 / a.shape[0])
    return a / s


# ----------------------------------------------------------------------
# Surrogate objective (smoothed max) and its gradient
# ----------------------------------------------------------------------
def _smooth_max(conv: np.ndarray, beta: float) -> Tuple[float, np.ndarray]:
    """
    Returns (smooth_max, softmax_weights) where
        smooth_max = (1/β)·log Σ exp(β·conv_i)
    and softmax_weights_i = exp(β·conv_i) / Σ exp(β·conv_j).
    """
    shift = np.max(conv)                     # numerical stability
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    weights = exp_beta / sum_exp
    smooth = shift + np.log(sum_exp) / beta
    return smooth, weights


def _objective_and_grad(v: np.ndarray,
                        n: int,
                        beta: float) -> Tuple[float, np.ndarray]:
    """
    Smooth surrogate of the official objective and its gradient w.r.t.
    the log‑softmax parameters ``v``.
    """
    # a is a probability vector (ℓ₁‑normalised)
    a = _softmax(v)

    # full convolution a * a
    conv = _conv_fft(a, a)

    # smoothed max and its soft‑max weights
    smooth, w = _smooth_max(conv, beta)      # smooth ≈ max(conv)

    # gradient of Σ w_k·conv_k w.r.t. a
    rev_a = a[::-1]
    tmp = _conv_fft(rev_a, w)               # length 3n‑2
    grad_a = 2.0 * tmp[n - 1:n - 1 + n]      # central n entries

    # chain‑rule: ∂F/∂v = a ⊙ (grad_a – (grad_a·a))
    inner = np.dot(grad_a, a)
    grad_v = a * (grad_a - inner)

    # final surrogate value (identical to official when sum(a)=1)
    f_val = 2.0 * n * smooth
    return f_val, grad_v


# ----------------------------------------------------------------------
# Seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve from a few random cosine terms."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _sorted_desc(v: np.ndarray) -> np.ndarray:
    a = np.sort(v)[::-1]
    return a / a.sum()


def _sorted_asc(v: np.ndarray) -> np.ndarray:
    a = np.sort(v)
    return a / a.sum()


def _alternate_order(v: np.ndarray) -> np.ndarray:
    a = np.sort(v)                # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    return out / out.sum()


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> List[np.ndarray]:
    seeds: List[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_gaussian(n, sigma_frac=0.40))

    # power‑law
    for p in np.linspace(0.0, 4.0, 13):
        seeds.append(_power(n, p))

    # beta family
    for a in np.linspace(0.3, 2.0, 5):
        for b in np.linspace(0.3, 4.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (different concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            raw = rng.dirichlet(np.full(n, conc))
            seeds.append(raw)                     # raw order
            seeds.append(_sorted_desc(raw))
            seeds.append(_sorted_asc(raw))
            seeds.append(_alternate_order(raw))

    # Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0.0:
            seeds.append(avg / avg.sum())
    return seeds


# ----------------------------------------------------------------------
# L‑BFGS‑B optimisation with continuation on β (temperature)
# ----------------------------------------------------------------------
def _lbfgs_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule: List[float] = (5.0, 20.0, 80.0),
                    max_iter_per_stage: int = 120) -> Tuple[np.ndarray, float]:
    """
    Refine ``start_vec`` (ℓ₁‑normalised) by minimising the smoothed
    objective with L‑BFGS‑B.  A sequence of increasing temperatures
    (β) is used to approach the true max.
    """
    import scipy.optimize

    # initialise logits from the start vector
    eps = 1e-30
    v = np.log(start_vec + eps)

    best_seq = start_vec.copy()
    best_val = float("inf")

    for beta in beta_schedule:
        # closure for scipy (no lambda as required)
        def func_and_grad(x):
            f, g = _objective_and_grad(x, n, beta)
            return f, g

        opts = dict(maxcor=10, maxfun=max_iter_per_stage,
                    maxiter=max_iter_per_stage, disp=False)

        res = scipy.optimize.minimize(fun=lambda x: func_and_grad(x)[0],
                                      x0=v,
                                      method='L-BFGS-B',
                                      jac=lambda x: func_and_grad(x)[1],
                                      bounds=[(None, None)] * n,
                                      **opts)

        v = res.x                     # continue from the last point
        a = _softmax(v)               # current probability vector
        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if time.time() > deadline:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (fast surrogate)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         init_temp: float = 1e-1,
                         min_temp: float = 1e-12,
                         max_iters: int = 600_000) -> Tuple[np.ndarray, float]:
    """
    Fast SA that performs tiny mass‑transfer moves while evaluating the
    cheap FFT surrogate.  The best candidate is finally evaluated with
    the official ``evaluate_sequence``.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    T = init_temp

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # geometric cooling
        T = max(T * 0.9999, min_temp)

        # ----- propose a move -----
        move_type = rng.integers(0, 3)
        if move_type == 0:                     # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:                   # redistribute a small subset
            k = rng.integers(2, 8)            # subset size
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                   # pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        # ----- evaluate surrogate -----
        new_val = 2.0 * n * np.max(_conv_fft(new, new))
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
        # Periodic progress report
        if it % 50_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted time.
    Returns the best sequence as a plain list of floats and stores it
    in the global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Length of the step function – 4000 is a good compromise.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve previous solution, if any.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list):
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a large seed pool and pick the most promising ones.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Collected {len(seeds)} seeds.")

    # Quick surrogate evaluation (FFT based, no calls to evaluate_sequence)
    evaluated = [(vec, 2.0 * n * np.max(_conv_fft(vec, vec))) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 6
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = 2.0 * n * np.max(_conv_fft(global_best_seq, global_best_seq))
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – L‑BFGS‑B refinement on the top seeds.
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        max_iters_stage = min(300_000, int(remaining * 30))  # generous budget
        print(f"\n--- L‑BFGS‑B run {idx}/{len(best_seeds)} (max_iters≈{max_iters_stage}) ---")
        seq_opt, val_opt = _lbfgs_optimize(seed_vec,
                                           n,
                                           deadline,
                                           beta_schedule=[5.0, 20.0, 80.0],
                                           max_iter_per_stage=150)
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"  [Update] new global best surrogate = {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Heavier simulated‑annealing polish.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(global_best_seq,
                                              n,
                                              deadline,
                                              init_temp=0.2,
                                              max_iters=800_000)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] new global best surrogate = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (the true objective).
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("[Warning] evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(official ≈ {official_val if official_val is not None else 'N/A'}), "
        f"length {len(global_best_seq)}"
    )

    # expose for reuse in subsequent calls
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
