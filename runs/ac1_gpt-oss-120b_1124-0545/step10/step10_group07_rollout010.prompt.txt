def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)

def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))

def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window, β ≥ 0."""
    if not SCIPY_AVAILABLE or beta <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    nminus1 = n - 1
    x = (2.0 * i / nminus1) - 1.0
    a = i0(beta * np.sqrt(1.0 - x * x)) / i0(beta)
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density discretised on mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `alpha`."""
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _smooth_expgd(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using a soft‑max approximation of the max‑convolution.
    The temperature (beta) is increased linearly up to a fixed cap.
    Momentum is added to stabilise the updates.
    """
    # Ensure strictly positive entries to avoid log(0)
    eps = 1e-12
    a = np.clip(start_seq, eps, None)
    a = a / a.sum()

    log_a = np.log(a)
    best_seq = a.copy()
    best_val = _objective(a)

    # momentum parameters
    beta1 = 0.9
    g_mom = np.zeros_like(a)

    # schedule for the soft‑max temperature
    beta_max = 200.0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution of a with itself
        conv = _conv_fft(a, a)

        # soft‑max weight vector (stable)
        beta = min(beta_max, beta_max * it / max_iters)
        conv_max = conv.max()
        shifted = conv - conv_max          # for numerical stability
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient of the smooth objective (weighted sum of conv entries)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        g = 2.0 * full[n - 1:n - 1 + n]

        # momentum update (exponential moving average)
        g_mom = beta1 * g_mom + (1.0 - beta1) * g

        # learning‑rate schedule
        lr = 0.6 / math.sqrt(it)

        # exponentiated‑gradient step in log‑space
        log_a = log_a - lr * g_mom
        log_a -= np.max(log_a)          # prevent overflow
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = _uniform(n)
        else:
            a = a / a_sum

        # occasional exact evaluation (every ~200 iters)
        if it % 200 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SmoothEG] it={it:7d} best={cur_val:.6f}")

    # Final exact evaluation
    final_val = _objective(a)
    if final_val < best_val:
        best_val = final_val
        best_seq = a.copy()
        print(f"[SmoothEG] final best={final_val:.6f}")

    return best_seq, best_val

def _additive_descent(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300000) -> tuple[np.ndarray, float]:
    """
    Simple projected gradient descent using an exact sub‑gradient of the
    max‑convolution (average over the active constraints).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        tol = max(rel_tol, 0.001 * max_val)
        active = np.where(conv >= max_val - tol)[0]

        g = np.zeros_like(a)
        if active.size > 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.12 / math.sqrt(it)
        a_next = a - lr * g
        a_next = _project_simplex(a_next)
        a = a_next

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[AddGD] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val

def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Metropolis–Hastings walk with several move types.
    Returns the best vector found and its objective.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature – a modest fraction of current objective
    T0 = max(1e-7, cur_val * 0.07)
    decay = 0.9994

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        rnd = rng.random()
        if rnd < 0.68:  # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.random() * 0.18 * cur[i]   # 0‑18 % of source mass
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        elif rnd < 0.86:  # swap two entries
            i, j = rng.integers(0, n, size=2)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:  # global Gaussian jitter
            noise = rng.normal(scale=0.0013, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0:
                new_seq = new_seq / total
            else:
                new_seq = _uniform(n)

        new_seq = _project_simplex(new_seq)
        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:8d} best={new_val:.6f}")

        # occasional stronger jump
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.0035, size=n)
            cand = best + jitter
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand)
            cand_val = _objective(cand)
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
                print(f"[Anneal] (jump) it={it:8d} best={cand_val:.6f}")

    return best, best_val
