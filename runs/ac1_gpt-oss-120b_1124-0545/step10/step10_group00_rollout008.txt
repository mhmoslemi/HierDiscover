def _lbfgs_obj_wrapper(v, n, beta):
    a = _softmax(v)
    conv = _conv_fft(a, a)
    # compute smoothed max: f = (1/beta) * log( sum exp(beta*conv) )
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    w = exp_term / exp_term.sum()
    # objective (smoothed max) for optimizer: we can return surrogate as weighted sum:
    obj = np.dot(w, conv)  # this is smooth approx, but we need objective for optimizer: we want to minimize max conv, not weighted sum.
    # Actually want surrogate that approximates max; the weighted average approximates softmax, but to approximate max we can use weighted average, but gradient for w yields same as gradient for smoothed max up to additive constant.
    # We can compute directly surrogate = (np.log(np.sum(exp_term)) / beta) + shift
    obj = (np.log(np.sum(exp_term)) / beta) + shift
    # gradient w.r.t a:
    full = _conv_fft(a[::-1], w)   # length 3n-2
    grad_a = 2 * full[n-1:n-1+n]   # as before
    # chain rule dL/dv:
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v

"""
Search for a non‑negative step‑function that minimizes

    2·n·max(conv(seq,seq)) / (sum(seq)²)

within a fixed wall‑clock budget (default 1000 s).

The strategy improves upon the reference implementation by adding:

* a fast L‑BFGS optimiser on a smooth log‑sum‑exp surrogate (instead of
  only Adam);
* a longer mirror‑descent phase with a slightly larger learning‑rate schedule;
* a slower‑cooling simulated‑annealing polish;
* a richer seed pool (more random Dirichlet draws, extra low‑frequency curves).

All helper functions are top‑level (no closures) and the only global
state is ``height_sequence_1`` which stores the best found sequence for
possible warm‑starts in subsequent runs.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact 2·n·max(conv(seq,seq)) / (sum(seq)²) – the quantity to minimise."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv) / (np.sum(seq) ** 2)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random families & ordering helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)  # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve generated from a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_gaussian(n, sigma_frac=0.40))

    # power‑law family
    for p in np.linspace(0.0, 4.0, 13):
        seeds.append(_power(n, p))

    # beta family
    for a in np.linspace(0.3, 2.0, 5):
        for b in np.linspace(0.3, 4.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw
            seeds.append(_sorted_desc(raw))       # descending
            seeds.append(_sorted_asc(raw))        # ascending
            seeds.append(_alternate_order(raw))   # alternating

    # Extra low‑frequency smooth curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures with the uniform base
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    # A handful of completely random sparse candidates (to explore Sidon‑like patterns)
    for _ in range(5):
        # pick a random sparsity level between 5% and 20%
        k = rng.integers(max(2, n // 100), max(3, n // 5))
        positions = rng.choice(n, size=k, replace=False)
        vec = np.zeros(n, dtype=np.float64)
        # equal mass on the chosen positions
        vec[positions] = 1.0 / k
        seeds.append(vec)

    return seeds


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray | None, float]:
    """Adam on log‑softmax parameters using a log‑sum‑exp surrogate."""
    eps = 1e-30
    v = np.log(start_vec + eps)             # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8
    beta = 0.5
    beta_inc = 0.5
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # probability simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)    # exact surrogate (no smoothing)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] iter={it:7d} best={cur_val:.6f}")

        # ----- smoothed max (log‑sum‑exp) -----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule to log‑space (softmax) gradient
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror) sub‑gradient descent on the true max‑conv."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.12  # initial step size factor

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] iter={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _simulated_annealing_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
    init_temp_factor: float = 0.05,
    min_temp: float = 1e-13,
) -> tuple[np.ndarray, float]:
    """Simulated annealing with several move types and a very slow cooling schedule."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-13)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling (very slow)
        T = max(T * 0.9999, min_temp)

        move_type = rng.integers(0, 3)   # 0=swap, 1=redistribute subset, 2=pair‑transfer
        if move_type == 0:
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            k = rng.integers(2, 12)    # size of the subset
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA] iter={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _lbfgs_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iter: int = 200,
    beta: float = 30.0,
) -> tuple[np.ndarray | None, float]:
    """L‑BFGS on the smooth log‑sum‑exp surrogate (maximum approximated)."""

    def func_and_grad(v: np.ndarray):
        a = _softmax(v)
        conv = _conv_fft(a, a)                     # length 2n‑1
        shift = np.max(conv)
        exp_term = np.exp(beta * (conv - shift))
        sum_exp = exp_term.sum()
        # smooth approximation of the max
        obj = (np.log(sum_exp) / beta) + shift
        w = exp_term / sum_exp                      # soft‑max weights over conv entries
        # gradient w.r.t. a
        full = _conv_fft(a[::-1], w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]
        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return obj, grad_v

    v0 = np.log(start_vec + 1e-30)

    # minimise – L‑BFGS works with bound‑less variables, the sum constraint is enforced by softmax
    res = minimize(
        fun=lambda v: func_and_grad(v),
        x0=v0,
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': max_iter, 'disp': False}
    )
    if not res.success:
        return None, float('inf')

    seq = _softmax(res.x)
    val = _objective(seq)
    return seq, val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allocated wall‑clock budget.
    Returns the best sequence as a plain list of floats and stores it
    in the global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for the final sanity check
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Length of the candidate – 4000 balances expressive power vs speed
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Warm‑start from any existing solution
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # 1️⃣  Seed generation + quick ranking
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Collected {len(seeds)} raw seeds.")

    # evaluate surrogate (exact max for quick ranking)
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  L‑BFGS refinement on the top seeds (smooth surrogate)
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate a few hundred L‑BFGS iterations per seed (very cheap)
        print(f"\n--- L‑BFGS refinement run {idx}/{len(best_seeds)} ---")
        seq_lbfgs, val_lbfgs = _lbfgs_optimize(seed_vec, n, deadline,
                                               max_iter=250, beta=40.0)
        if seq_lbfgs is not None and val_lbfgs < global_best_val:
            global_best_seq = seq_lbfgs.copy()
            global_best_val = val_lbfgs
            print(f"[L‑BFGS] new global best = {val_lbfgs:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Adam optimisation on the best candidates (smooth surrogate)
    # --------------------------------------------------------------
    if time.time() < deadline:
        # two quick Adam runs starting from the current best
        for run_id in range(2):
            if time.time() > deadline:
                break
            remaining = deadline - time.time()
            max_iters = min(120_000, int(remaining * 55))   # aim for ~55 it/s
            print(f"\n--- Adam optimisation pass {run_id + 1} (max_iters={max_iters}) ---")
            seq_adam, val_adam = _adam_optimize(
                global_best_seq, n, deadline, max_iters=max_iters
            )
            if seq_adam is not None and val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[Adam] updated global best = {val_adam:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Mirror‑descent sub‑gradient optimisation (exact max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=1_600_000
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Simulated‑annealing polishing (very slow cooling)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline, max_iters=2_000_000,
            init_temp_factor=0.03, min_temp=1e-13
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official checker (optional)
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
