import time, numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _objective(seq):
    n = len(seq)
    return float(2.0 * n * np.max(_conv_fft(seq, seq)))

def _softmax(v):
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

def _proj_simplex(v):
    # Euclidean projection onto simplex
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w
```

def _adam_optimize(start_vec, n, deadline, max_iters=500_000, beta_max=1000.0):
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta_temp = 5.0
    beta_mul = 1.001
    best_seq = None
    best_val = float('inf')
    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        beta_temp = min(beta_temp * beta_mul, beta_max)
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum==0.0:
            continue
        w = w / w_sum
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1:n-1+n]
        # transform to log-softmax gradient
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # clip
        max_norm=6.0
        norm = np.linalg.norm(grad_v)
        if norm>max_norm:
            grad_v = grad_v * (max_norm / norm)
        # Adam update
        beta1=0.9; beta2=0.999; eps=1e-12; lr0=0.8
        lr = lr0 / np.sqrt(it)
        m = beta1*m + (1-beta1)*grad_v
        v2 = beta2*v2 + (1-beta2)*(grad_v**2)
        m_hat = m/(1-beta1**it)
        v2_hat = v2/(1-beta2**it)
        v = v - lr * m_hat / (np.sqrt(v2_hat)+eps)
    return best_seq, best_val

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective (scale‑invariant, assumes non‑negative entries)."""
    n = len(seq)
    return float(2.0 * n * np.max(_conv_fft(seq, seq)))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex Σ v = 1, v ≥ 0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    k = rng.integers(3, 31)               # 3 … 30 non‑zero entries
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


def _mixture_uniform_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.random()
    g = _gaussian(n, sigma_frac=rng.uniform(0.05, 0.4))
    u = _uniform(n)
    a = lam * u + (1.0 - lam) * g
    return a / a.sum()


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 800_000,
                   beta_max: float = 800.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max‑convolution.
    Returns (best_sequence, best_value) where best_value is the *exact*
    objective value (computed with _objective).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)           # logits – stay on simplex after softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0                        # initial smooth‑max temperature
    beta_mul = 1.001                       # how fast it rises
    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                    # ℓ1‑normalised candidate
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:,d} best={best_val:.6f}")

        # raise temperature very slowly to approach true max
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # full convolution of a with itself
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                          # softmax over convolution entries

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # map to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses a Polyak step size (based on the best value seen so far).
    Returns the best sequence encountered and its exact value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[SubGrad] it={it:,d} best={best_val:.6f}")

        # sub‑gradient for the active index (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Very local simulated‑annealing moves: select a small random subset of
    coordinates and re‑allocate their total mass according to a Dirichlet draw.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)               # move 2 … 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:,d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation utilities
# ----------------------------------------------------------------------


def _collect_diverse_seeds(n: int,
                           rng: np.random.Generator,
                           previous: np.ndarray | None,
                           top_k: int = 20,
                           deadline: float = 0.0) -> list[np.ndarray]:
    """
    Generate a large, diverse pool of ℓ₁‑normalised candidates.
    Returns the `top_k` seeds with the smallest exact objective.
    """
    top_vals = []   # list of (value, seq)

    def try_insert(val, seq):
        """Maintain a sorted list of the best `top_k` entries."""
        if len(top_vals) < top_k:
            top_vals.append((val, seq))
            top_vals.sort(key=lambda x: x[0])
        elif val < top_vals[-1][0]:
            top_vals[-1] = (val, seq)
            top_vals.sort(key=lambda x: x[0])

    # deterministic families (always worth keeping)
    families = [
        _uniform,
        _triangular,
        lambda n: _trapezoidal(n, width_fraction=0.15),
        lambda n: _sine_window(n, gamma=1.0),
        lambda n: _sine_window(n, gamma=2.0),
        lambda n: _sine_window(n, gamma=3.0),
        lambda n: _gaussian(n, sigma_frac=0.12),
        lambda n: _gaussian(n, sigma_frac=0.24),
        lambda n: _tukey_window(n, alpha=0.2),
        lambda n: _tukey_window(n, alpha=0.4),
    ]
    for fn in families:
        seq = fn(n)
        val = _objective(seq)
        try_insert(val, seq)

    # geometric families
    for r in np.linspace(0.2, 0.98, 12):
        for fn in (_geom, _sym_geom):
            seq = fn(n, r)
            val = _objective(seq)
            try_insert(val, seq)

    # power‑law family
    for p in np.linspace(0.0, 5.0, 18):
        seq = _power(n, p)
        val = _objective(seq)
        try_insert(val, seq)

    # random Dirichlet draws (wide range of concentrations)
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    for conc in concs:
        draws = 120 if conc < 0.01 else 70
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = _dirichlet_seed(n, conc, rng)
            val = _objective(seq)
            try_insert(val, seq)

    # sparse random supports
    for _ in range(500):
        if time.time() > deadline:
            break
        seq = _sparse_seed(n, rng)
        val = _objective(seq)
        try_insert(val, seq)

    # random beta shapes
    for _ in range(600):
        if time.time() > deadline:
            break
        alpha = rng.uniform(0.05, 5.0)
        beta = rng.uniform(0.05, 5.0)
        seq = _beta_shape(n, alpha, beta)
        val = _objective(seq)
        try_insert(val, seq)

    # random sine windows
    for _ in range(300):
        if time.time() > deadline:
            break
        gamma = rng.uniform(0.2, 5.0)
        seq = _sine_window(n, gamma=gamma)
        val = _objective(seq)
        try_insert(val, seq)

    # random Gaussian windows
    for _ in range(300):
        if time.time() > deadline:
            break
        sigma_frac = rng.uniform(0.05, 0.4)
        seq = _gaussian(n, sigma_frac=sigma_frac)
        val = _objective(seq)
        try_insert(val, seq)

    # random Tukey windows
    for _ in range(200):
        if time.time() > deadline:
            break
        alpha = rng.uniform(0.05, 0.9)
        seq = _tukey_window(n, alpha=alpha)
        val = _objective(seq)
        try_insert(val, seq)

    # mixture of uniform & Gaussian
    for _ in range(200):
        if time.time() > deadline:
            break
        seq = _mixture_uniform_gaussian(n, rng)
        val = _objective(seq)
        try_insert(val, seq)

    # previous solution (if any) – give it a chance again
    if previous is not None and previous.shape == (n,):
        seq = np.asarray(previous, dtype=np.float64)
        if seq.sum() > 0:
            seq = seq / seq.sum()
            val = _objective(seq)
            try_insert(val, seq)

    return [seq for _, seq in top_vals]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    (default 1000 s) and stores the best sequence in the global variable
    ``height_sequence_1`` for possible reuse.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Length – a power of two (good for FFT) and still reasonably fine‑grained.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Stage 0 – abundant seed generation.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # allocate roughly 15 % of the budget to seeding
    seed_deadline = min(deadline, start_time + budget_s * 0.15)
    seeds = _collect_diverse_seeds(n, rng, previous,
                                   top_k=20,
                                   deadline=seed_deadline)
    print(f"Collected {len(seeds)} promising seeds.")

    # ------------------------------------------------------------------
    # Stage 1 – short Adam refinements (quick exploration).
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Short Adam refinements (stage 1) ---")
    for idx, seed_vec in enumerate(seeds):
        if time.time() > deadline:
            break
        it_budget = int(80_000)   # cheap runs, enough to move away from poor locals
        print(f"[Seed {idx + 1}/{len(seeds)}] – short Adam")
        seq_adam, val_adam = _adam_optimize(
            seed_vec, n, deadline, max_iters=it_budget, beta_max=400.0
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  New global best after short Adam: {best_val:.6f}")

    # fallback if nothing succeeded (unlikely)
    if best_seq is None:
        best_seq = seeds[0].copy()
        best_val = _objective(best_seq)
        print(f"Fallback to first seed: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – longer Adam optimisation on the current best.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        it_budget = int(min(1_500_000, remaining * 1_000))   # heuristic
        print("\n--- Long Adam optimisation (stage 2) ---")
        seq_adam, val_adam = _adam_optimize(
            best_seq, n, deadline, max_iters=it_budget, beta_max=800.0
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  Updated global best after long Adam: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – intensive projected sub‑gradient descent.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        it_budget = int(min(1_500_000, remaining * 1_000))
        print("\n--- Projected sub‑gradient descent (stage 3) ---")
        seq_sub, val_sub = _subgrad_optimize(
            best_seq, n, deadline, max_iters=it_budget
        )
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"  Updated global best after sub‑grad: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final simulated‑annealing polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        it_budget = int(min(800_000, remaining * 1_000))
        print("\n--- Simulated annealing polishing (stage 4) ---")
        seq_sa, val_sa = _simulated_annealing(
            best_seq, n, deadline, max_iters=it_budget,
            init_temp_factor=0.05
        )
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"  Updated global best after SA: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"\nFinal score reported by evaluate_sequence: {final_score:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective (internal) ≈ {best_val:.6f} "
        f"(vector length {len(best_seq)})"
    )

    # Store for possible reuse and return a plain Python list.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
