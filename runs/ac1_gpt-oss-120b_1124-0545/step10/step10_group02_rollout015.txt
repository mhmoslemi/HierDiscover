def solve_lp_for_g(f, M, n):
    import scipy.optimize as opt
    # Build A_ub as lists for sparse representation
    # We'll use CSR matrix.
    row_ptr = [0]
    data = []
    col = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # indices i_start..i_end inclusive
        length = i_end - i_start + 1
        if length > 0:
            # Append f[i_start:i_end+1] to data
            vals = f[i_start:i_end+1]
            data.extend(vals.tolist())
            col.extend(range(i_start, i_end+1))
        row_ptr.append(row_ptr[-1] + length)
    A_ub = sp.csr_matrix((data, col, row_ptr), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum -> minimize -sum
    bounds = [(0, None)] * n
    # Solve
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
    if res.success:
        return res.x
    else:
        return None

"""
Propose a step function that beats the known bound (~1.509) for
``evaluate_sequence``.

The search consists of three cheap but effective stages that run well
within the 1000 s budget.

1. **Dense seed sweep** – a large collection of analytic families
   (geometric, symmetric‑geometric, trapezoidal, sine‑window, power‑law,
   Gaussian, Tukey, Beta) is sampled densely on the simplex
   (ℓ₁‑normalisation).  The best 30 seeds are retained.

2. **Convex mixing** – every pair of the retained seeds is mixed with
   mixing weights t∈[0,1] (21 equally spaced points).  Because
   ``conv((1‑t)f+t g) = (1‑t)conv(f)+(t)conv(g)`` the maximum of the
   convex combination can be strictly smaller than the individual
   maxima when the peaks occur at different shifts.  The best mixed
   candidate is kept.

3. **Light hill‑climbing** – starting from the best mixed sequence a
   simple stochastic coordinate update is performed: a small amount of
   mass is transferred from a random index i to a random index j.
   The move is accepted if it improves the objective, otherwise it is
   rejected (with a tiny probability of uphill acceptance to escape
   plateaus).  The loop runs while time permits (a few‑ten‑thousand
   iterations) and updates the global best whenever a better value is
   found.

All vectors are kept on the probability simplex (sum = 1) so the
objective reduces to ``2·n·max(conv)`` and the scale‑invariance of the
original problem is respected.  The final sequence is stored in the
global ``height_sequence_1`` variable for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT version, O(n log n))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large n)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(vec: np.ndarray) -> float:
    """Exact objective for an ℓ₁‑normalised vector (sum == 1)."""
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return a vector whose entries sum to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (0 < r < 1)."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

    cur_val = global_best_val
    it = 0
    # maximum number of iterations – tuned to stay well within the time budget
    max_iters = 40000
    while it < max_iters and time.time() < deadline:
        it += 1
        # pick a donor index with non‑trivial mass
        i = rng.integers(n)
        if cur[i] < 1e-12:
            continue
        # pick a recipient (different)
        j = rng.integers(n)
        if i == j:
            continue
        # transfer a small fraction of the donor's mass
        delta = cur[i] * rng.uniform(0.0, 0.05)
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # numerical safety
        if new[i] < 0.0:
            continue
        val = _objective(new)
        # accept only if strictly better (tiny probability for uphill moves)
        if val < cur_val or rng.random() < 0.001:
            cur = new
            cur_val = val
            if val < global_best_val:
                global_best_val = val
                global_best_seq = cur.copy()
                print(f"[Hill] iter {it:6d} new best = {val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – evaluate with the official function.
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    # Store for possible later reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
