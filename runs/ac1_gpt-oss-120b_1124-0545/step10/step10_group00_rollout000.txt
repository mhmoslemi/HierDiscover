"""
Propose a step‑function that minimizes

    2·n·max(conv(seq,seq)) / (sum(seq)²)

using a continuation‑L‑BFGS optimizer on a smoothed max‑convolution
surrogate.  The algorithm

1. builds a diverse pool of ℓ₁‑normalised seeds (analytic windows,
   sparse random spikes, low‑frequency curves, Dirichlet draws);
2. keeps the best few seeds according to the exact surrogate
   2·n·max(conv);
3. for each seed runs a continuation in the temperature β:
   minimize  (2·n/β)·log‑sum‑exp(β·conv(a,a))  with a = softmax(v)
   by L‑BFGS‑B (the gradient is obtained analytically via a convolution);
4. optionally refines the result with a few mirror‑subgradient steps;
5. returns the best sequence (list of floats) and stores it in the
   global variable ``height_sequence_1`` for reuse.

The method is fast (FFT‑based convolutions) and works well within the
1000 s budget, typically improving the bound below the target 1.503.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Convolution utilities (FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple surrogate objective (exact max‑convolution)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)) for a vector with sum = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Softmax and smoothed‑max surrogate (log‑sum‑exp)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _smooth_max_obj_grad(v: np.ndarray, n: int, beta: float):
    """
    Return (objective, gradient) of the smooth max surrogate

        f(v) = (2·n/β)·log‑sum‑exp(β·conv(a,a))

    where a = softmax(v).  Gradient is with respect to v.
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)                     # length 2n‑1
    shift = np.max(conv)                       # for numerical stability
    exp_terms = np.exp(beta * (conv - shift))
    exp_sum = np.sum(exp_terms)
    lse = shift + np.log(exp_sum)              # log‑sum‑exp
    obj = 2.0 * n * lse / beta

    # weights for the convex combination in the gradient
    w = exp_terms / exp_sum                     # sums to 1

    # gradient w.r.t. a:  grad_a = 2 * (rev_a * w) (central n entries)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    return obj, grad_v


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the smooth surrogate (continuation in β)
# ----------------------------------------------------------------------
def _lbfgs_continuation(start_seq: np.ndarray,
                        n: int,
                        betas,
                        max_iter_per_beta: int,
                        deadline: float):
    """
    Run a continuation L‑BFGS optimiser.
    start_seq – ℓ₁‑normalised seed (sum = 1).
    betas – iterable of temperature values (increasing).
    Returns the final sequence (still ℓ₁‑normalised) and its exact surrogate.
    """
    from scipy.optimize import minimize

    # initialise logits (log‑softmax parameters)
    v = np.log(start_seq + 1e-30)

    # a tiny helper to abort when we run out of time
    class TimeOut(RuntimeError):
        pass

    def fun(v_vec):
        if time.time() > deadline:
            raise TimeOut()
        obj, grad = _smooth_max_obj_grad(v_vec, n, beta_cur)
        return obj, grad

    for beta_cur in betas:
        try:
            res = minimize(fun,
                           v,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': max_iter_per_beta,
                                    'disp': False})
            v = res.x
        except TimeOut:
            break  # out of the outer loop as well

    final_seq = _softmax(v)          # ℓ₁‑normalised automatically
    return final_seq, _objective(final_seq)


# ----------------------------------------------------------------------
# Mirror‑subgradient refinement (optional, cheap)
# ----------------------------------------------------------------------
def _mirror_subgradient_refine(start_seq: np.ndarray,
                               n: int,
                               deadline: float,
                               max_iters: int = 200_000):
    """
    A few steps of exponentiated‑gradient (mirror) sub‑gradient descent
    on the exact max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 20_000 == 0:
                print(f"[Mirror‑refine] it={it:,d} best={cur_val:.6f}")

        # sub‑gradient for the active lag
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generators (including sparse random spikes)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve built from a few cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_spike(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """
    k spikes of equal mass placed on k distinct random indices.
    If k > n we fall back to uniform.
    """
    if k >= n:
        return _uniform(n)
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    a[idx] = 1.0 / k
    return a


def _collect_seeds(n: int, rng: np.random.Generator, previous):
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_gaussian(n, sigma_frac=0.35))
    seeds.append(_beta_shape(n, 0.5, 0.5))
    seeds.append(_beta_shape(n, 0.8, 1.2))
    seeds.append(_beta_shape(n, 1.2, 0.8))

    # low‑frequency smooth curves
    for nf in (2, 3, 4):
        seeds.append(_lowfreq_curve(n, nf, rng))

    # random Dirichlet draws (different concentration parameters)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            raw = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            seeds.append(raw)                       # raw order
            seeds.append(np.sort(raw)[::-1])        # descending
            seeds.append(np.sort(raw))              # ascending

    # sparse spikes – these are often very effective
    for k in (8, 16, 32, 64):
        seeds.append(_sparse_spike(n, k, rng))

    # mixture of a uniform background with a low‑frequency curve
    for mix_alpha in (0.2, 0.4, 0.6):
        curve = _lowfreq_curve(n, 3, rng)
        mixed = (1.0 - mix_alpha) * _uniform(n) + mix_alpha * curve
        mixed /= mixed.sum()
        seeds.append(mixed)

    # previous solution (if any)
    if previous is not None and isinstance(previous, (list, np.ndarray)):
        prev = np.asarray(previous, dtype=np.float64)
        if prev.shape == (n,) and prev.sum() > 0:
            seeds.append(prev / prev.sum())

    return seeds

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence (list of floats) and stores it in the
    global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a tiny safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length of the step‑function – 4000 offers a good trade‑off.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Gather seeds and keep the most promising few.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n = {n}) ---")
    previous = globals().get("height_sequence_1")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # evaluate the exact surrogate (= 2·n·max(conv)) for each seed
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS optimisation on the smooth max surrogate.
    # ------------------------------------------------------------------
    beta_schedule = [5.0, 15.0, 50.0, 200.0, 800.0]
    max_iter_per_beta = 250   # per β, per seed (adjusted to stay within budget)

    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        print(f"\n--- L‑BFGS continuation run {idx}/{len(best_seeds)} ---")
        seq_opt, val_opt = _lbfgs_continuation(seed_vec,
                                               n,
                                               beta_schedule,
                                               max_iter_per_beta,
                                               deadline)
        print(f"  [L‑BFGS] surrogate = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"  [overall] new global best = {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Light mirror‑subgradient polishing (few thousand iterations only).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑subgradient polishing ---")
        seq_ref, val_ref = _mirror_subgradient_refine(global_best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=80_000)
        if val_ref < global_best_val:
            global_best_seq = seq_ref.copy()
            global_best_val = val_ref
            print(f"[overall‑Mirror] new global best = {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse by later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
