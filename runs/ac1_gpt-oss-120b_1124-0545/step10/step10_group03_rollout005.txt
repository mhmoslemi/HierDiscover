"""
Propose a low‑value step‑function using a two‑phase scheme:

1. **Active‑set linear programming** – given a current shape `f`,
   solve the LP 
        maximise   Σ g_i
        subject to conv(f,g) ≤ M   (M = max(conv(f,f))),
                    g ≥ 0
   The LP gives a direction `g` that can carry more mass without
   exceeding the current peak `M`.  Mixing `f` with the normalised
   solution lowers the peak.  The process is repeated with a
   decreasing mixing factor `t`.  Only a few LP solves are needed;
   building the convolution matrix is O(n²) but n≈3000 is still fast.

2. **Local refinement** – after the LP phase a short Adam optimiser
   on a smooth log‑sum‑exp surrogate fine‑tunes the shape.

Both stages keep the vector on the probability simplex, so the
official `evaluate_sequence` can be called directly.  The routine
respects a 1000 s budget and stores the best sequence in the global
`height_sequence_1` for reuse.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags, csr_matrix

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence when the sequence is ℓ₁‑normalised.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple analytic families (only uniform is required as a fallback)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Adam optimiser (used for a final polishing stage)
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max‑convolution.
    `start_vec` is assumed to be ℓ₁‑normalised.
    Returns (best_sequence, best_objective).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.001
    beta_max = 250.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming utilities
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1) × n matrix A such that A @ g = conv(f, g)
    (linear convolution, zero‑padding outside the interval).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # Each diagonal offset d (0 … n‑1) is filled with the constant f[d]
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=int)
    A = diags(data, offsets, shape=(rows, n), format='csr')
    return A


def _lp_step(f: np.ndarray, M: float, deadline: float) -> np.ndarray | None:
    """
    Solve the LP  max Σ g_i  s.t.  conv(f,g) ≤ M , g ≥ 0 .
    Returns the optimal g (not normalised) or None on failure.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)                     # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)             # minimise -Σg ⇒ maximise Σg
    bounds = [(0, None)] * n

    # HiGHS handles sparse A_ub efficiently
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
    except Exception as exc:
        print("linprog exception:", exc)
        return None

    if not res.success:
        print("LP failed:", res.message)
        return None
    return res.x


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine first refines an existing sequence
    (or a uniform start) via a few LP‑based mixing steps, then runs a
    short Adam polish.  The best sequence is stored in the global variable
    `height_sequence_1` and returned as a plain list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Length – 3000 gives a good trade‑off between resolution and LP size.
    # ------------------------------------------------------------------
    n = 3000

    # ------------------------------------------------------------------
    # Starting point – reuse the previously discovered sequence if possible.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) >= 2:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size > n:
                arr = arr[:n]                # truncate
            elif arr.size < n:
                # pad with zeros (they will stay zero after normalisation)
                arr = np.concatenate([arr, np.zeros(n - arr.size, dtype=np.float64)])
            f = arr
            if f.sum() > 0:
                f = f / f.sum()
            else:
                f = _uniform(n)
        except Exception:
            f = _uniform(n)
    else:
        f = _uniform(n)

    best_f = f.copy()
    best_val = _objective(best_f)
    print(f"Initial objective (n={n}): {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – active‑set LP mixing.
    # ------------------------------------------------------------------
    t = 0.5                               # initial mixing weight
    t_min = 0.02
    decay = 0.94
    itr = 0
    while time.time() < deadline - 30.0:   # keep ~30 s for the final Adam stage
        itr += 1
        conv_ff = _conv_fft(best_f, best_f)
        M = conv_ff.max()
        g0 = _lp_step(best_f, M, deadline)
        if g0 is None:
            print("LP step aborted.")
            break
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            print("LP returned zero‑mass vector.")
            break
        g = g0 / sum_g0                     # normalise to sum = 1

        # convex combination
        new_f = (1.0 - t) * best_f + t * g
        new_f = _proj_simplex(new_f)        # re‑project to the simplex

        new_val = _objective(new_f)
        print(f"[LP‑iter {itr:3d}] t={t:.3f} → val={new_val:.6f}")

        if new_val < best_val:
            best_val = new_val
            best_f = new_f
            print(f"  *** improvement: {best_val:.6f}")
        else:
            # no improvement – reduce the mixing factor more aggressively
            t = max(t * 0.5, t_min)

        # geometric decay of the mixing factor
        t = max(t * decay, t_min)

    # ------------------------------------------------------------------
    # Phase 2 – short Adam polishing (uses the surrogate smooth‑max).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_adam_iters = int(3e5)          # enough for a decent polish
        print("\n--- Adam polishing ---")
        seq_adam, val_adam = _adam_optimize(best_f, n, deadline, max_iters=max_adam_iters)
        if seq_adam is not None and val_adam < best_val:
            best_f = seq_adam
            best_val = val_adam
            print(f"Adam improvement: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_f.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f}  (length {len(best_f)})")
    globals()["height_sequence_1"] = best_f.tolist()
    return best_f.tolist()
