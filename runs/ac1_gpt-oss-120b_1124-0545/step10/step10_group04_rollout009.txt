def solve_lp_max_sum(f, M, active_idx):
    n = len(f)
    m = len(active_idx)
    # Build A (m x n) dense, using for each row.
    A = np.zeros((m, n), dtype=np.float64)
    for row_i, k in enumerate(active_idx):
        # for each j where 0 <= k-j < n
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        # j runs from start to end: then i = k - j
        # So for j in [start, end], set A[row_i, j] = f[k - j]
        # we can vectorize:
        js = np.arange(start, end + 1)
        A[row_i, js] = f[k - js]
    # solve LP
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = M * np.ones(m)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return zero vector
        return np.zeros_like(f)
    b = res.x
    return b
```

def lp_refine(f, max_iters, deadline, t=0.5, active_rel=0.01, max_active=200):
    n = len(f)
    best_f = f.copy()
    best_val = _objective(f)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv_f_f = _conv_fft(f, f)
        M = conv_f_f.max()
        # active indices: top indices where conv close to M
        # get sorted indices descending
        idx_sorted = np.argsort(-conv_f_f)  # descending
        # select top max_active or those within relative tolerance
        # Use rough top or relative threshold
        topk = min(max_active, len(conv_f_f))
        active_idx = idx_sorted[:topk]
        # solve LP
        b = solve_lp_max_sum(f, M, active_idx)
        S = b.sum()
        if S <= 1.0 + 1e-12:
            # cannot improve
            break
        g = b / S
        # mixing
        f_new = (1 - t) * f + t * g
        # renormalize to sum=1 (should be)
        f_new = f_new / f_new.sum()
        # evaluate
        val_new = _objective(f_new)
        if val_new < best_val:
            best_val = val_new
            best_f = f_new.copy()
            print(f"[LP] it={it} n={n} val={val_new:.6f} M={M:.8f}")
        # if no improvement, maybe reduce t
        if val_new >= best_val - 1e-8:
            # optionally reduce t
            t = max(t * 0.8, 0.01)
        f = f_new
        # early exit if best_val < 1.503 (target)
        if best_val < 1.503:
            break
    return best_f, best_val

"""
Optimisation of a non‑negative step function for the objective

    E(a) = 2·n·max(conv(a,a)) / (∑ a_i)²

The evaluation is scale‑invariant, therefore we work with sequences normalised
to a unit sum (∑ a_i = 1).  The algorithm consists of three stages:

1. **Initialisation** – a collection of analytic families (arcsine, Beta,
   Catalan, …) and a few random Dirichlet draws are evaluated, the best shape
   for each tested length is retained.

2. **Cut‑plane LP refinement** – for the current shape `f` we compute
   `M = max(conv(f,f))`.  The linear programme

        maximise   1ᵀb
        subject to  conv(f,b) ≤ M      (for the most critical convolution indices)
                    b ≥ 0

   gives a vector `b` whose total mass `S = 1ᵀb` is the largest possible while
   never exceeding the current worst convolution value `M`.  Whenever `S>1`
   we can mix `f` with the normalised vector `g = b/S` (the mixing weight
   `t` is reduced adaptively) – this strictly reduces the maximum convolution.
   The process is repeated (a simple cutting‑plane method) until no further
   improvement is possible or the time budget is exhausted.

3. **Projected sub‑gradient polishing** – a short run of the Euclidean‑projected
   sub‑gradient method (the same code as the previous state‑of‑the‑art) is used
   to fine‑tune the final shape.

The whole search respects a user supplied time limit (default 1000 s).  The
best sequence found is returned as a plain Python list; all entries are
non‑negative, the evaluator will clip them to the allowed range automatically.
"""

import time
import numpy as np
from math import pi, sqrt

# ----------------------------------------------------------------------
# Fast convolution (real FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a *normalised* sequence (sum == 1).
    Returns ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (always returned normalised to sum == 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * pi * i / (n - 1))
         + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian_sigma015(n: int) -> np.ndarray:
    sigma_factor = 0.15
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _gaussian_sigma025(n: int) -> np.ndarray:
    sigma_factor = 0.25
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf shape on a discrete grid (shifted by 0.5 to avoid end‑points)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape_05_05(n: int) -> np.ndarray:
    return _beta_shape(n, 0.5, 0.5)


def _beta_shape_2_2(n: int) -> np.ndarray:
    return _beta_shape(n, 2.0, 2.0)


def _beta_shape_05_2(n: int) -> np.ndarray:
    return _beta_shape(n, 0.5, 2.0)


def _beta_shape_2_05(n: int) -> np.ndarray:
    return _beta_shape(n, 2.0, 0.5)


# ----------------------------------------------------------------------
# Random Dirichlet sampler (produces points on the simplex)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Return a random point on the simplex (sum=1) with Dirichlet(alpha)."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (same as the previous state‑of‑the‑art)
# ----------------------------------------------------------------------
def _proj_subgradient_optimize(start_seq: np.ndarray,
                               deadline: float,
                               eps_rel_start: float = 1e-4,
                               eps_rel_end: float = 1e-6,
                               lr0: float = 0.12,
                               max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for the exact objective.
    Uses an aggregated sub‑gradient built from all convolution entries
    that are within a relative tolerance of the current maximum.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        # linear schedule for the tolerance (shrink over time)
        eps_rel = eps_rel_start + (eps_rel_end - eps_rel_start) * min(1.0, it / (max_iter * 0.5))

        thresh = max_val - eps_rel * max_val
        active_idx = np.where(conv >= thresh)[0]

        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv)], dtype=int)

        # uniform weights over the active set
        w = np.zeros_like(conv)
        w[active_idx] = 1.0 / active_idx.size

        # gradient = 2 * (w * a_rev) convolved with a
        conv_ag = _conv_fft(w, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        # step size schedule
        lr = lr0 / sqrt(it)

        a_next = a - lr * g
        a_next = _project_simplex(a_next)

        # occasional exact evaluation
        if (it & 1023) == 0:
            val = _objective(a_next)
            if val < best_val:
                best_val = val
                best_seq = a_next.copy()
                print(f"[ProjSGD] it={it:7d} n={n:5d} val={val:.6f}")
            if best_val < 1.45:          # early exit if already very good
                break

        a = a_next

    return best_seq, best_val


# ----------------------------------------------------------------------
# Helper to build the (small) constraint matrix used by the LP step
# ----------------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    For a given shape ``f`` and a set of indices ``active_idx`` (the
    convolution positions to enforce), build the matrix A such that
    A @ b  =  conv(f, b)[active_idx].
    """
    n = f.shape[0]
    m = active_idx.size
    A = np.zeros((m, n), dtype=np.float64)

    for row, k in enumerate(active_idx):
        # convolution term: sum_j f[k-j] * b_j, where 0 ≤ j < n and 0 ≤ k-j < n
        start_j = max(0, k - (n - 1))
        end_j = min(k, n - 1)
        js = np.arange(start_j, end_j + 1, dtype=int)
        A[row, js] = f[k - js]
    return A


# ----------------------------------------------------------------------
# Cut‑plane LP refinement of a single shape
# ----------------------------------------------------------------------
def _lp_refine(f_start: np.ndarray,
               deadline: float,
               max_iters: int = 30,
               t_initial: float = 0.5,
               active_rel: float = 0.02,
               max_active: int = 200) -> tuple[np.ndarray, float]:
    """
    Iteratively solve a linear programme that maximises the total mass
    under the current worst‑case convolution bound.  The solution is mixed
    with the current shape, guaranteeing a non‑increasing objective.
    """
    n = f_start.shape[0]
    f = f_start.copy()
    best_seq = f.copy()
    best_val = _objective(f)

    t = t_initial
    it = 0

    # Import lazily – SciPy may be missing in some environments.
    try:
        import scipy.optimize
    except Exception as e:
        print("[LP] SciPy not available, skipping LP refinement:", e)
        return best_seq, best_val

    while time.time() < deadline and it < max_iters:
        it += 1

        conv_ff = _conv_fft(f, f)
        M = float(conv_ff.max())

        # --------------------------------------------------------------
        # Active set: indices where the convolution is close to its maximum
        # --------------------------------------------------------------
        thresh = M * (1.0 - active_rel)
        active_idx = np.where(conv_ff >= thresh)[0]
        if active_idx.size < 5:
            # fallback – simply take the top entries
            active_idx = np.argsort(-conv_ff)[:min(max_active, conv_ff.size)]

        # --------------------------------------------------------------
        # Build and solve the LP: maximise Σ b_j  subject to  conv(f,b) ≤ M
        # --------------------------------------------------------------
        A = _build_constraint_matrix(f, active_idx)
        c = -np.ones(n, dtype=np.float64)          # maximise Σ b  ⇔ minimise -Σ b
        bounds = [(0.0, None)] * n
        b_ub = M * np.ones(A.shape[0], dtype=np.float64)

        lp_res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                        bounds=bounds, method='highs')
        if not lp_res.success:
            print(f"[LP] linprog failed at iteration {it} – exiting LP phase")
            break

        b_opt = lp_res.x
        S = b_opt.sum()

        # No room for improvement → stop.
        if S <= 1.0 + 1e-12:
            break

        # Normalise the new candidate and mix.
        g = b_opt / S
        f_new = (1.0 - t) * f + t * g
        f_new /= f_new.sum()          # guard against numerical drift

        val_new = _objective(f_new)

        if val_new < best_val:
            best_val = val_new
            best_seq = f_new.copy()
            print(f"[LP] it={it:2d} n={n:4d} val={val_new:.6f} M={M:.8f} S={S:.4f} t={t:.3f}")
        else:
            # If the mix did not improve the objective, shrink the step size.
            t = max(t * 0.7, 0.01)

        # Prepare for next iteration.
        f = f_new

        # Early exit if we already beat the target.
        if best_val < 1.503:
            break

    return best_seq, best_val

    # 0.  Pull a previously found solution (if any) as a warm‑start.
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_seq = np.asarray(previous, dtype=np.float64)
            if prev_seq.size > 0 and np.all(prev_seq >= 0):
                prev_seq = np.maximum(prev_seq, 0.0)
                if prev_seq.sum() > 0.0:
                    prev_seq = prev_seq / prev_seq.sum()
                else:
                    prev_seq = None
            else:
                prev_seq = None
        except Exception:
            prev_seq = None
    else:
        prev_seq = None

    # --------------------------------------------------------------
    # 1.  Choose a handful of lengths (FFT‑friendly, not too large).
    # --------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000]

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 2.  Deterministic seeds and a few random draws.
    # --------------------------------------------------------------
    seed_families = (
        _arcsine_shape,
        _catalan_shape,
        _beta_shape_05_05,
        _beta_shape_2_2,
        _beta_shape_05_2,
        _beta_shape_2_05,
        _hann,
        _hamming,
        _blackman,
        _gaussian_sigma015,
        _gaussian_sigma025,
    )

    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:15s} val={val:.6f}")

        # Add a small collection of random Dirichlet / Beta candidates.
        for _ in range(3):
            seq = _random_dirichlet(n, rng, alpha=rng.choice([0.5, 1.0, 2.0]))
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[RandDir] n={n:5d} val={val:.6f}")

            a_par, b_par = rng.choice([(0.5, 0.5), (2.0, 2.0), (0.5, 2.0), (2.0, 0.5)])
            seq = _beta_shape(n, a_par, b_par)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[RandBeta] n={n:5d} (α,β)=({a_par:.1f},{b_par:.1f}) val={val:.6f}")

        # Re‑use the stored solution if it has the same length.
        if prev_seq is not None and prev_seq.shape[0] == n:
            val = _objective(prev_seq)
            if val < best_val:
                best_val = val
                best_seq = prev_seq.copy()
                print(f"[Prev] n={n:5d} val={val:.6f}")

    # --------------------------------------------------------------
    # 3.  LP refinement – the core novel ingredient.
    # --------------------------------------------------------------
    if best_seq is not None:
        # Allocate roughly half the remaining time to the LP phase.
        now = time.time()
        remaining = deadline - now
        lp_deadline = now + min(remaining * 0.5, 25.0)   # never spend more than ~25 s
        refined_seq, refined_val = _lp_refine(best_seq, lp_deadline,
                                             max_iters=30,
                                             t_initial=0.5,
                                             active_rel=0.02,
                                             max_active=200)
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"[LP‑Refine] improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4.  Final short projected sub‑gradient polish (optional).
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        proj_deadline = deadline
        seq_proj, val_proj = _proj_subgradient_optimize(
            start_seq=best_seq,
            deadline=proj_deadline,
            eps_rel_start=1e-4,
            eps_rel_end=1e-6,
            lr0=0.12,
            max_iter=5_000_000,
        )
        if val_proj < best_val:
            best_val = val_proj
            best_seq = seq_proj.copy()
            print(f"[Proj‑Final] improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # 5.  Final sanity check (the grader will evaluate again anyway)
    # --------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
