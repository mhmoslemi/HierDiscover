Define function proj_simplex(v): typical algorithm:

def proj_simplex(v):
    # ensure non-negative sum = 1
    v = np.maximum(v, 0) # we need positivity anyway but algorithm deals with any real.
    if v.sum() == 0:
        return np.full_like(v, 1.0/v.size)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _interpolate_sequence(seq, new_n):
    old_n = len(seq)
    old_x = np.linspace(0, 1, old_n, endpoint=False) + 0.5/old_n
    new_x = np.linspace(0, 1, new_n, endpoint=False) + 0.5/new_n
    new_vals = np.interp(new_x, old_x, seq)
    new_vals /= new_vals.sum()

Hybrid step‑function optimizer
------------------------------

We attack the problem of minimizing

    2·n·max(conv(a,a)) / (sum(a)²)

by a multi‑phase search that fits comfortably into a 1000 s budget.

Phases
------

   semi‑circular and power‑sine shape).

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to evaluate_sequence for a normalised sequence (sum == 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # from "Efficient Projections onto the L1‑ball"
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(v.size) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    return coeffs / total if total > 0 else _uniform(n)


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()

    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = a0 - a1 * np.cos(2 * pi * i / (n - 1)) + a2 * np.cos(4 * pi * i / (n - 1))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = w.sum()
    return w / total if total > 0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    total = w.sum()
    return w / total if total > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    total = w.sum()
    return w / total if total > 0 else _uniform(n)


def _sin_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(pi * i / n)
    total = w.sum()
    return w / total if total > 0 else _uniform(n)


def _semicircle_shape(n: int) -> np.ndarray:
    """Discrete semi‑circular (Wigner) density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    radius = n / 2.0
    x = (i - radius) / radius
    w = np.sqrt(1 - x * x)
    total = w.sum()
    return w / total if total > 0 else _uniform(n)


def _power_sin_shape(n: int, p: float = 2.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(pi * i / n) ** p
    total = w.sum()
    return w / total if total > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random generation utilities
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet‑like draw (Gamma/Exp) with concentration α."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else _uniform(n)


def _random_beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.uniform(0.1, 5.0)
    b = rng.uniform(0.1, 5.0)
    return _beta_shape(n, a, b)


def _random_mixture_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    w = rng.random()
    a1, b1 = rng.uniform(0.1, 5.0), rng.uniform(0.1, 5.0)
    a2, b2 = rng.uniform(0.1, 5.0), rng.uniform(0.1, 5.0)
    s1 = _beta_shape(n, a1, b1)
    s2 = _beta_shape(n, a2, b2)
    mix = w * s1 + (1.0 - w) * s2
    return mix / mix.sum()


def _interpolate_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence to a new length."""
    old_n = len(seq)
    old_x = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    new_x = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq / new_seq.sum()


def _apply_power(seq: np.ndarray, p: float) -> np.ndarray:
    """Element‑wise power transform a ↦ aᵖ (re‑normalised)."""
    if p == 1.0:
        return seq.copy()
    transformed = np.maximum(seq, 0) ** p
    s = transformed.sum()
    return transformed / s if s > 0 else seq.copy()


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 5000.0,
    max_iter: int = 5_000_000,
    lr0: float = 0.03,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth (log‑sum‑exp) approximation of max(conv(a,a)).
    Uses Adam on the log‑softmax (logits) parametrisation.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # soft‑max
        a = np.exp(logits)
        a = a / a.sum()

        conv = _conv_fft(a, a)

        # temperature schedule (linear ramp)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()  # soft max weights

        # gradient wrt a
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional exact evaluation
        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

            if best_val < 1.44:  # early exit if already very good
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS‑B refinement (smooth max with large β)
# ----------------------------------------------------------------------
def _lbfgs_refine(start_seq: np.ndarray, deadline: float, beta: float = 5000.0) -> tuple[np.ndarray, float]:
    if minimize is None:
        return start_seq, _objective(start_seq)

    n = len(start_seq)
    eps = 1e-30

    def fun_and_grad(logits: np.ndarray):
        a = np.exp(logits)
        a = a / a.sum()
        conv = _conv_fft(a, a)
        c = np.max(conv)
        exps = np.exp(beta * (conv - c))
        lse = c + (1.0 / beta) * np.log(exps.sum())
        value = 2.0 * n * lse

        p = exps / exps.sum()
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1 : n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)
        return value, grad_logits.astype(np.float64)

    x0 = np.log(start_seq + eps)

    class TimeOut(RuntimeError):
        pass

    def callback(xk):
        if time.time() >= deadline:
            raise TimeOut()

    try:
        res = minimize(
            fun=lambda x: fun_and_grad(x)[0],
            x0=x0,
            jac=lambda x: fun_and_grad(x)[1],
            method="L-BFGS-B",
            options={"maxiter": 200, "disp": False},
            callback=callback,
        )
    except TimeOut:
        best_logits = x0
    else:
        best_logits = res.x

    a = np.exp(best_logits)
    a /= a.sum()
    return a, _objective(a)


# ----------------------------------------------------------------------
# Aggressive aggregated mirror‑descent polishing
# ----------------------------------------------------------------------
def _aggregated_mirror_descent(
    start_seq: np.ndarray,
    deadline: float,
    eps_rel_start: float = 0.02,
    eps_rel_end: float = 1e-7,
    lr_factor: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent where the sub‑gradient aggregates *all* entries whose
    convolution value is within a decaying relative tolerance of the running
    maximum.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    start = time.time()
    it = 0
    eps = 1e-30
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        # exponentially decaying tolerance
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total if total > 0 else 0.0
        eps_rel = eps_rel_start * (eps_rel_end / eps_rel_start) ** frac
        delta = eps_rel * max_val

        active_idx = np.where(conv >= max_val - delta)[0]
        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1 : n - 1 + n]

        lr = lr_factor / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)  # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[AggMD] it={it:7d} n={n:5d} best={val:.6f}")

            if time.time() > deadline - 0.5:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    `evaluate_sequence` objective, respecting the supplied time budget.
    Returns the best sequence as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Re‑use a previously computed solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length candidates (FFT‑friendly)
    # ------------------------------------------------------------------
    lengths = [2000, 2500, 3000, 3500, 4000, 5000, 6000, 8000, 10000, 12000, 15000, 20000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic families (seed generation)
    # ------------------------------------------------------------------
    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.20),
        lambda n: _gaussian(n, sigma_factor=0.25),
        _arcsine_shape,
        _sin_shape,
        _semicircle_shape,
        lambda n: _beta_shape(n, 0.5, 0.5),
        lambda n: _beta_shape(n, 2.0, 2.0),
        lambda n: _beta_shape(n, 0.5, 2.0),
        lambda n: _beta_shape(n, 2.0, 0.5),
        lambda n: _power_sin_shape(n, 1.5),
        lambda n: _power_sin_shape(n, 2.5),
    )

    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:18s} val={val:.6f}")

        # reuse previous solution if same length
        if previous is not None and previous.shape[0] == n:
            val = _objective(previous)
            if val < best_val:
                best_val, best_seq = val, previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – heavy random sampling (Dirichlet + Beta families)
    # ------------------------------------------------------------------
    now = time.time()
    sample_time = (deadline - now) * 0.30  # ~30% of remaining budget
    sample_deadline = now + sample_time

    alphas = [0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]
    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Rnd‑Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        seq = _random_beta_shape(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Rnd‑Beta] n={n:5d} val={val:.6f}")

        seq = _random_mixture_beta(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Rnd‑Mix] n={n:5d} val={val:.6f}")

        if best_val < 1.44:  # early cut‑off
            break

    # ------------------------------------------------------------------
    # Phase 3 – scaling to larger lengths via interpolation
    # ------------------------------------------------------------------
    if best_seq is not None:
        base_len = len(best_seq)
        larger = [L for L in lengths if L > base_len]
        for L in larger:
            if time.time() >= deadline:
                break
            seq = _interpolate_sequence(best_seq, L)
            val = _objective(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[Scale] n={L:5d} val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – exponent‑power transforms of the current best
    # ------------------------------------------------------------------
    if best_seq is not None:
        powers = [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.4, 1.6, 2.0]
        for p in powers:
            if time.time() >= deadline:
                break
            seq = _apply_power(best_seq, p)
            val = _objective(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[Power] p={p:.2f} val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Smoothed Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 30.0:
            adam_deadline = time.time() + remaining * 0.40
            # collect a few diversified starts
            candidates = [best_seq]
            # two random Dirichlet starts with same length
            for _ in range(2):
                cand = _random_dirichlet(len(best_seq), rng, alpha=rng.uniform(0.2, 3.0))
                candidates.append(cand)
            # one extra larger‑length start (if possible)
            larger_len = [L for L in lengths if L > len(best_seq)]
            if larger_len:
                L_big = int(rng.choice(larger_len))
                cand_big = _random_dirichlet(L_big, rng, alpha=1.0)
                candidates.append(cand_big)

            for idx, cand in enumerate(candidates):
                if time.time() >= adam_deadline:
                    break
                seq_opt, val_opt = _smooth_adam_optimize(
                    start_seq=cand,
                    deadline=adam_deadline,
                    beta_start=5.0,
                    beta_end=5000.0,
                    max_iter=5_000_000,
                    lr0=0.04,
                )
                if val_opt < best_val:
                    best_val, best_seq = val_opt, seq_opt.copy()
                    print(f"[Adam Refine] idx={idx} val={val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – optional L‑BFGS‑B refinement
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0 and minimize is not None:
            lbfgs_deadline = time.time() + remaining * 0.20
            seq_lb, val_lb = _lbfgs_refine(best_seq, lbfgs_deadline, beta=5000.0)
            if val_lb < best_val:
                best_val, best_seq = val_lb, seq_lb.copy()
                print(f"[L‑BFGS] improved best to {val_lb:.6f}")

    # ------------------------------------------------------------------
    # Phase 7 – aggressive aggregated mirror‑descent polishing
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _aggregated_mirror_descent(
            start_seq=best_seq,
            deadline=deadline,
            eps_rel_start=0.02,
            eps_rel_end=1e-7,
            lr_factor=1.2,
        )
        if final_val < best_val:
            best_val, best_seq = final_val, final_seq.copy()
            print(f"[Agg‑MD Final] improved best to {final_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will evaluate again anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised:", exc)

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
