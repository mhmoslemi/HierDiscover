"""
Propose a non‑negative step‑function that improves the bound for
``evaluate_sequence``.  

The search consists of several complementary stages that respect a strict
time budget:

1. **Diversified seed pool** – a large collection of analytic families
   (beta, sine‑window, Tukey, geometric, power‑law, Gaussian, trapezoidal,
   uniform, triangular) together with random convex combinations of them.
2. **Coarse screening** – evaluate every seed with the exact max‑convolution
   objective (fast FFT) and keep the few best.
3. **Adam on a smooth surrogate** – for each top seed run Adam optimisation
   on the log‑sum‑exp smooth approximation of the max, slowly sharpening the
   temperature (β up to 2000).  This quickly moves the iterate into a better
   basin.
4. **Projected sub‑gradient descent** – a long Polyak‑style sub‑gradient
   descent on the true max‑convolution objective, staying on the simplex.
5. **Random local mass‑transfer search** – very cheap stochastic hill‑climbing
   that repeatedly moves a small amount of mass between two coordinates.
6. **Final sanity check** – compute the official value with ``evaluate_sequence``.

All stages continuously check the deadline (default 1000 s) and stop
gracefully when the budget is exhausted.  The best sequence found is stored
in the global variable ``height_sequence_1`` for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation for ℓ₁‑normalised sequences (sum == 1):
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):          # numerical safety
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric decay from both ends."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ i^(−p), i starts at 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped distribution on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey (cosine‑tapered) window."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc) and normalise to sum = 1."""
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 500_000, beta_max: float = 2000.0):
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    ``start_vec`` must already be ℓ₁‑normalised (sum = 1).
    Returns the best sequence seen and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)               # logits → simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.003

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] iter={it:,} best={best_val:.6f}")

        # increase temperature (sharpen the smooth max)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # weighted soft‑max over the convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                         # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]               # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip gradient norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 2_000_000, init_lr: float = 0.20):
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses a Polyak‑style diminishing step size.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] iter={it:,} best={cur_val:.6f}")

        # sub‑gradient for the active convolution index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size (no knowledge of optimum → use 1/√t scaling)
        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _random_local_search(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 300_000, step_factor: float = 0.15):
    """
    Very cheap stochastic hill‑climbing that moves a small amount of mass
    between two random coordinates while staying on the simplex.
    Accepts uphill moves with a slowly cooling temperature.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rng = np.random.default_rng()
    T = best_val * 0.02          # initial temperature (relative)
    decay = 0.99995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * decay, 1e-12)

        src = rng.integers(0, n)
        if a[src] <= 1e-16:
            continue
        dst = rng.integers(0, n - 1)
        if dst >= src:
            dst += 1

        delta = rng.random() * step_factor * a[src]   # move at most step_factor·src
        new = a.copy()
        new[src] -= delta
        new[dst] += delta

        cur_val = _objective(new)

        # accept if improvement or according to Metropolis rule
        if cur_val < best_val or rng.random() < np.exp(-(cur_val - best_val) / T):
            a = new
            if cur_val < best_val:
                best_val = cur_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[Local] iter={it:,} best={best_val:.6f}")

    return best_seq, best_val

        # random parameters for each chosen family
        combo = np.zeros(n, dtype=np.float64)
        for w, func in zip(weights, chosen):
            # pick a random admissible parameter set for the function
            if func is _trapezoidal:
                wf = rng.uniform(0.02, 0.45)
                seq = func(n, width_fraction=wf)
            elif func is _sine_window:
                gamma = rng.uniform(0.0, 3.0)
                seq = func(n, gamma=gamma)
            elif func is _tukey_window:
                alpha = rng.uniform(0.0, 0.95)
                seq = func(n, alpha=alpha)
            elif func is _geom or func is _sym_geom:
                r = rng.uniform(0.2, 0.99995)
                seq = func(n, r)
            elif func is _power:
                p = rng.uniform(0.0, 5.0)
                seq = func(n, p)
            elif func is _beta_shape:
                a = rng.uniform(0.05, 4.0)
                b = rng.uniform(0.05, 4.0)
                seq = func(n, a, b)
            elif func is _gaussian:
                sigma = rng.uniform(0.08, 0.45)
                seq = func(n, sigma)
            else:   # uniform or triangular – no parameters
                seq = func(n)
            combo += w * seq
        combo = combo / combo.sum() if combo.sum() > 0 else _uniform(n)
        seeds.append(combo)

    # Include previous solution if available
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # 1 s safety margin

    # ------------------------------------------------------------------
    # Length of the candidate – 6000 is a sweet spot for quality vs speed.
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Stage 1 – generate a large, diverse pool of seeds.
    # ------------------------------------------------------------------
    print(f"\n--- Collecting seeds (length={n}) ---")
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    # Quick evaluation of all seeds
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])

    top_k = min(6, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation on the smooth surrogate.
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds[:4]):   # up to four runs
        if time.time() > deadline:
            break
        print(f"\n--- Adam optimisation run {idx+1}/{len(best_seeds[:4])} ---")
        seq_adam, val_adam = _adam_optimize(
            seed_vec, n, deadline,
            max_iters=500_000,
            beta_max=2000.0
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Projected sub‑gradient refinement (the heavy part).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # Rough budget: about 1 ms per iteration → allow up to a few million steps
        max_iters = int(min(3_000_000, max(500_000, remaining * 1500)))
        print(f"\n--- Sub‑gradient refinement (max_iters={max_iters}) ---")
        seq_proj, val_proj = _subgradient_proj(
            global_best_seq, n, deadline,
            max_iters=max_iters,
            init_lr=0.25
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[overall‑SubProj] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Random local mass‑transfer search (cheap hill‑climbing).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random local search (mass transfer) ---")
        seq_local, val_local = _random_local_search(
            global_best_seq, n, deadline,
            max_iters=400_000,
            step_factor=0.12
        )
        if val_local < global_best_val:
            global_best_seq = seq_local.copy()
            global_best_val = val_local
            print(f"[overall‑Local] new global best = {val_local:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length={len(global_best_seq)})"
    )

    # Store for potential reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
