"""
Propose a non‑negative step function minimizing ``evaluate_sequence``.

The search consists of four phases:

1. Deterministic analytic seeds (uniform, Hann, Beta, etc.) on several
   FFT‑friendly lengths.
2. Light random sampling (Dirichlet & Beta shapes) to obtain a good
   starting shape.
3. A multi‑stage L‑BFGS optimisation of a smooth log‑sum‑exp
   approximation of the max‑convolution.  The variable is the
   soft‑max logits, guaranteeing the simplex constraint.
4. A final exact projected sub‑gradient polishing (the same routine as
   the reference implementation) and a few random perturbations to
   escape possible local minima.

All phases respect the overall time budget (default 1000 s) and the
best sequence found is returned as a plain ``list[float]``.
"""

# ---------------------------------------------------------------
# Imports & low‑level utilities
# ---------------------------------------------------------------
import time
import numpy as np
from math import pi, sqrt

# ---------------------------------------------------------------
# Convolution helpers (FFT based, O(n log n))
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a *normalised* sequence
    (sum == 1).  Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ---------------------------------------------------------------
# Analytic seed families (all returned normalised to sum = 1)
# ---------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * pi * i / (n - 1))
         + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian_sigma015(n: int) -> np.ndarray:
    sigma_factor = 0.15
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _gaussian_sigma025(n: int) -> np.ndarray:
    sigma_factor = 0.25
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf shape on a discrete grid (shifted by 0.5 to avoid end‑points)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape_05_05(n: int) -> np.ndarray:
    return _beta_shape(n, 0.5, 0.5)


def _beta_shape_2_2(n: int) -> np.ndarray:
    return _beta_shape(n, 2.0, 2.0)


def _beta_shape_05_2(n: int) -> np.ndarray:
    return _beta_shape(n, 0.5, 2.0)


def _beta_shape_2_05(n: int) -> np.ndarray:
    return _beta_shape(n, 2.0, 0.5)

# Random Dirichlet sampling
# ---------------------------------------------------------------

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Return a random point on the simplex (sum=1) with Dirichlet(alpha)."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ---------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ---------------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ---------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
# ---------------------------------------------------------------

def _proj_subgradient_optimize(start_seq: np.ndarray,
                               deadline: float,
                               eps_rel_start: float = 1e-4,
                               eps_rel_end: float = 1e-6,
                               lr0: float = 0.12,
                               max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for the exact objective.
    Uses an aggregated sub‑gradient built from all convolution entries
    that are within a relative tolerance of the current maximum.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        eps_rel = eps_rel_start + (eps_rel_end - eps_rel_start) * min(1.0, it / (max_iter * 0.5))
        thresh = max_val - eps_rel * max_val
        active_idx = np.where(conv >= thresh)[0]

        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv)], dtype=int)

        w = np.zeros_like(conv)
        w[active_idx] = 1.0 / active_idx.size

        conv_ag = _conv_fft(w, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        lr = lr0 / sqrt(it)
        a_next = a - lr * g
        a_next = _project_simplex(a_next)

        if (it & 1023) == 0:
            val = _objective(a_next)
            if val < best_val:
                best_val = val
                best_seq = a_next.copy()
                print(f"[ProjSGD] it={it:7d} n={n:5d} val={val:.6f}")
            if best_val < 1.45:
                break

        a = a_next

    return best_seq, best_val

# Smooth log‑sum‑exp objective + L‑BFGS optimisation
# ---------------------------------------------------------------

# Import is optional – if SciPy is unavailable we simply skip this phase.
try:
    from scipy.optimize import fmin_l_bfgs_b
except Exception:  # pragma: no cover
    fmin_l_bfgs_b = None


def _lbfgs_optimize(start_seq: np.ndarray,
                    beta: float,
                    max_iter: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth log‑sum‑exp approximation of max(conv(a,a)) using
    L‑BFGS‑B.  Variables are logits (soft‑max), ensuring the simplex constraint.
    """
    if fmin_l_bfgs_b is None:
        # SciPy not available – fall back to the input sequence.
        return start_seq, _objective(start_seq)

    eps = 1e-30
    logits = np.log(start_seq + eps)
    n = len(start_seq)

    def f_and_grad(z):
        a = np.exp(z)
        a_sum = a.sum()
        a = a / a_sum

        conv = _conv_fft(a, a)
        maxc = conv.max()
        # stable log‑sum‑exp
        exp_term = np.exp(beta * (conv - maxc))
        sumexp = exp_term.sum()
        f_val = maxc + np.log(sumexp) / beta

        p = exp_term / sumexp
        g_a = 2.0 * _conv_fft(p, a[::-1])[n - 1:n - 1 + n]
        dot = np.dot(g_a, a)
        grad_z = a * (g_a - dot)
        return f_val, grad_z

    def callback(z):
        if time.time() > deadline:
            raise RuntimeError("Timeout")

    try:
        x_opt, _, _ = fmin_l_bfgs_b(f_and_grad,
                                    logits,
                                    maxiter=max_iter,
                                    iprint=-1,
                                    callback=callback)
    except RuntimeError:
        # timed out – return the current best (the start sequence)
        return start_seq, _objective(start_seq)

    a_opt = np.exp(x_opt)
    a_opt = a_opt / a_opt.sum()
    return a_opt, _objective(a_opt)


def _lbfgs_multi_stage(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Apply several L‑BFGS stages with increasing β (smoother → sharper).
    Each stage gets a fraction of the remaining time budget.
    """
    betas = [10.0, 50.0, 200.0, 800.0]
    seq = start_seq
    best_seq = seq
    best_val = _objective(seq)

    for i, beta in enumerate(betas):
        now = time.time()
        remaining = deadline - now
        if remaining <= 5.0:
            break
        # allocate roughly 15 % of the remaining budget per stage
        stage_deadline = now + remaining * 0.15
        seq, val = _lbfgs_optimize(seq, beta=beta, max_iter=3000,
                                   deadline=stage_deadline)
        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"[L‑BFGS stage {i}] β={beta:.1f} val={val:.6f}")
    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:

    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine follows several phases:
      1. deterministic analytic seeds,
      2. modest random Dirichlet / Beta sampling,
      3. multi‑stage L‑BFGS on a smooth log‑sum‑exp surrogate,
      4. exact projected sub‑gradient polishing with occasional random
         perturbations.
    The best sequence found within the time budget is returned as a list.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Lengths to explore (FFT‑friendly, but not too large for the LP‑free phases)
    # ------------------------------------------------------------------
    lengths = [2000, 2500, 3000, 3500, 4000, 5000, 6000, 8000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic seeds
    # ------------------------------------------------------------------
    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        _gaussian_sigma015,
        _gaussian_sigma025,
        _arcsine_shape,
        _beta_shape_05_05,
        _beta_shape_2_2,
        _beta_shape_05_2,
        _beta_shape_2_05,
    )

    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

        # Reuse a previously stored sequence if length matches
        previous = globals().get("height_sequence_1")
        if isinstance(previous, (list, tuple, np.ndarray)):
            prev = np.asarray(previous, dtype=np.float64)
            if prev.shape == (n,):
                prev = np.maximum(prev, 0.0)
                if prev.sum() > 0.0:
                    prev = prev / prev.sum()
                    val = _objective(prev)
                    if val < best_val:
                        best_val = val
                        best_seq = prev.copy()
                        print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – modest random sampling (≈10 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.10
    sample_deadline = now + time_for_sampling

    alphas = [0.5, 1.0, 2.0]
    beta_params = [(0.3, 0.3), (0.5, 0.5), (1.0, 1.0),
                   (2.0, 2.0), (0.5, 2.0), (2.0, 0.5)]

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random‑Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        a_par, b_par = rng.choice(beta_params)
        seq = _beta_shape(n, a_par, b_par)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random‑Beta] n={n:5d} (α,β)=({a_par:.2f},{b_par:.2f}) val={val:.6f}")

        if best_val < 1.45:    # early stop if already very good
            break

    # ------------------------------------------------------------------
    # Phase 3 – multi‑stage L‑BFGS on a smooth surrogate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq_opt, val_opt = _lbfgs_multi_stage(best_seq, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt
            print(f"[L‑BFGS] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – exact projected sub‑gradient polishing + random nudges
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 5.0:
            proj_deadline = time.time() + remaining * 0.80
            seq_proj, val_proj = _proj_subgradient_optimize(
                start_seq=best_seq,
                deadline=proj_deadline,
                eps_rel_start=1e-4,
                eps_rel_end=1e-7,
                lr0=0.20,
                max_iter=8_000_000,
            )
            if val_proj < best_val:
                best_val = val_proj
                best_seq = seq_proj
                print(f"[Proj‑Refine] improved best to {best_val:.6f}")

            # occasional random perturbations to escape possible plateaus
            while time.time() < deadline - 1.0:
                # small Gaussian jitter, then re‑project onto simplex
                noise = rng.normal(scale=0.02, size=best_seq.shape)
                pert = best_seq + noise
                pert = _project_simplex(pert)
                # short polishing run

                pert_opt, pert_val = _proj_subgradient_optimize(
                    start_seq=pert,
                    deadline=short_deadline,
                    eps_rel_start=1e-4,
                    eps_rel_end=1e-6,
                    lr0=0.15,
                    max_iter=500_000,
                )

                    best_val = pert_val
                    best_seq = pert_opt
                    print(f"[Perturb] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check (the grader will evaluate again anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
