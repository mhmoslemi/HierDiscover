def _lbfgsb_optimize(v0, n, deadline, beta=50.0, max_iter=200):
    import scipy.optimize
    # define functions
    def func_and_grad(v):
        a = _softmax(v)
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        # compute smoothed max using log-sum-exp
        # compute w = exp(beta*(conv - shift))

        grad_a = 2.0 * _conv_fft(a[::-1], w)[n-1:n-1+n]

        return obj, grad_v

        obj, _ = func_and_grad(v)
        return obj
    def grad(v):
        _, g = func_and_grad(v)
        return g

    def callback(xk):

"""
Hybrid optimisation for the step‑function attack.

The search proceeds in three phases within the given time budget:

1. **Seed generation** – a large, diverse pool of ℓ₁‑normalised candidates is
   created from analytic windows, Dirichlet draws and low‑frequency curves.
   The best few seeds (according to the exact surrogate objective
   `2·n·max(conv(seq,seq))`) are kept.

2. **Local refinement** – each top seed is refined with a smooth
   `log‑sum‑exp` surrogate of the max‑convolution using the
   L‑BFGS‑B optimiser (variables are log‑softmax parameters).  The
   surrogate uses a temperature `beta` to keep the problem differentiable.
   The resulting sequence is evaluated with the true max‑convolution and
   the best is retained.

3. **Polishing** – the current best sequence is further improved by a
   few‑million‑step mirror‑descent (exponentiated‑gradient) sub‑gradient
   optimisation on the *exact* max‑convolution, followed by a short
   simulated‑annealing pass.

All steps respect the hard time limit (default 1000 s).  The final
sequence is stored in the global variable ``height_sequence_1`` and also
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax producing a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """
    Exact surrogate used throughout the search:
        2·n·max(conv(seq,seq))   (seq is ℓ₁‑normalised, i.e. Σseq = 1)
    The factor 2·n is retained because it does not affect ordering.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Random families & ordering helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    """Return a copy sorted in non‑increasing order and re‑normalised."""
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    """Return a copy sorted in non‑decreasing order and re‑normalised."""
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Largest, smallest, 2nd‑largest, 2nd‑smallest, …"""
    a = np.sort(vec)          # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve obtained by a few random cosines."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_gaussian(n, sigma_frac=0.40))

    # power‑law family
    for p in np.linspace(0.0, 4.0, 13):
        seeds.append(_power(n, p))

    # beta family
    for a in np.linspace(0.3, 2.0, 5):
        for b in np.linspace(0.3, 4.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (different concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw random order
            seeds.append(_sorted_desc(raw))       # monotone decreasing
            seeds.append(_sorted_asc(raw))        # monotone increasing
            seeds.append(_alternate_order(raw))   # alternating pattern

    # Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixture of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

def _mirror_subgradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the exact
    max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.12  # initial step‑size factor

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _simulated_annealing_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
    init_temp_factor: float = 0.03,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Lightweight simulated‑annealing polish.  Moves are swaps, Dirichlet
    reshuffles of a small random subset and pair‑wise mass transfers.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 3)   # 0=swap, 1=redistribute subset, 2=pair‑transfer
        if move_type == 0:
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            k = rng.integers(2, 10)    # size of the subset
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _lbfgsb_optimize(
    v0: np.ndarray,
    n: int,
    deadline: float,
    beta: float = 50.0,
    max_iter: int = 250,
) -> tuple[np.ndarray | None, float]:
    """
    L‑BFGS‑B optimisation of the *smooth* surrogate
        2·n·log‑sum‑exp_β(conv(a,a))
    where a = softmax(v).  Returns the ℓ₁‑normalised vector and its exact
    surrogate value, or (None, inf) if the optimiser is interrupted.
    """
    try:
        import scipy.optimize
    except Exception as exc:
        print("SciPy not available – skipping L‑BFGS phase:", exc)
        return None, float("inf")

    def func_and_grad(v):
        a = _softmax(v)                     # probability simplex
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        # smoothed max via log‑sum‑exp
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        approx_max = shift + np.log(w_sum) / beta
        obj = 2.0 * n * approx_max

        # gradient w.r.t. a
        w = w / w_sum
        grad_a = 2.0 * _conv_fft(a[::-1], w)[n - 1:n - 1 + n]
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)         # chain rule through softmax
        return obj, grad_v

    def f(v):
        val, _ = func_and_grad(v)
        return val

    def g(v):
        _, grad = func_and_grad(v)
        return grad

    # callback to enforce the hard time limit
    def callback(xk):
        if time.time() > deadline:
            raise RuntimeError("time budget exceeded (L‑BFGS)")

    try:
        res = scipy.optimize.minimize(
            f,
            v0,
            jac=g,
            method="L-BFGS-B",
            callback=callback,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
    except RuntimeError as e:
        print("[L‑BFGS] terminated early:", e)
        return None, float("inf")
    except Exception as e:
        print("[L‑BFGS] optimisation error:", e)
        return None, float("inf")

    if not res.success:
        print("[L‑BFGS] warning – optimizer did not converge:", res.message)

    a_opt = _softmax(res.x)
    val = _objective(a_opt)    # evaluate with the *exact* max‑convolution
    return a_opt, val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence as a list of floats and stores it globally as
    ``height_sequence_1`` for reuse.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # Keep a small safety margin for final sanity checks
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length – 4000 is a good trade‑off (used by the reference solution)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Previous best (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list):
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # 1️⃣  Seed generation and quick ranking
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    best_seq, best_val = evaluated[0]
    print(f"\n[Init] Best seed surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Local refinement with L‑BFGS (smooth surrogate)
    # ------------------------------------------------------------------
    print("\n--- L‑BFGS‑B refinement on top seeds ---")
    for idx in range(top_k):
        if time.time() > deadline:
            break
        seq = evaluated[idx][0]
        v0 = np.log(seq + 1e-30)   # initialise logits
        a_opt, val_opt = _lbfgsb_optimize(v0, n, deadline, beta=50.0, max_iter=250)
        if a_opt is not None and val_opt < best_val:
            best_seq = a_opt
            best_val = val_opt
            print(f"[L‑BFGS] new best surrogate {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Mirror‑descent sub‑gradient optimisation (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_mirror = int(min(300_000, remaining * 120))  # rough budget allocation
        print("\n--- Mirror‑descent polishing ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            best_seq, n, deadline, max_iters=max_iters_mirror
        )
        if val_mirror < best_val:
            best_seq = seq_mirror
            best_val = val_mirror
            print(f"[Mirror] new best surrogate {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Simulated‑annealing final polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_sa = int(min(400_000, remaining * 120))
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            best_seq, n, deadline, max_iters=max_iters_sa
        )
        if val_sa < best_val:
            best_seq = seq_sa
            best_val = val_sa
            print(f"[SA] new best surrogate {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate (≈) {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
