"""
Propose a step‑function sequence that minimizes the provided
evaluate_sequence() objective.

Algorithm
---------
1. Generate a modest diverse pool of ℓ₁‑normalised seeds
   (uniform, triangular, Gaussians with large σ, a few Dirichlet draws,
   and the previous best if available).

2. For each seed run a *restricted linear‑program* refinement:
   •  At the current iterate f we keep only the K largest values of
      the self‑convolution f∗f (K≈400).  For those indices k we add the
      linear constraints  Σ_i f_i·g_{k‑i} ≤ M .
   •  Variables are the new sequence g ≥ 0 (ℓ₁‑norm fixed to 1) and the
      bound M.  The LP minimizes M.
   •  The solution g is taken as the next iterate.  The process is
      repeated a few (≈10) times – this is the block‑coordinate descent
      described in the statement.

   The LP is tiny (≈K constraints, N+1 variables) and solved with
   SciPy’s HiGHS interior‑point method, which is very fast.

3. Keep the best sequence seen over all seeds.

4. Perform a final smooth‑max Adam optimisation (log‑sum‑exp surrogate
   of the max‑convolution) to polish the result.  The Adam optimiser
   works on the simplex via a soft‑max parametrisation.

5. Return the best sequence (as a plain Python list) and store it in the
   global variable ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact evaluation: 2·n·max(conv(seq,seq)).  Assumes sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    Returns (best_sequence, best_value) where best_value is the exact
    objective (not the surrogate).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.001
    beta_max = 250.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # on‑simplex representation
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        # increase temperature slowly
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)                  # full conv
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                              # softmax over conv entries

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (active‑set)
# ----------------------------------------------------------------------


def _lp_refine(f_seq: np.ndarray,
               max_iters: int,
               deadline: float,
               top_k_active: int = 400) -> tuple[np.ndarray, float]:
    """
    Perform a few block‑coordinate steps:
    given the current sequence f, solve the LP
        minimise M
        s.t.   Σ_i f_i·g_{k-i} ≤ M    for the K most critical indices k
               Σ_j g_j = 1
               g_j ≥ 0
    The solution g becomes the next iterate.
    Returns the best sequence encountered and its exact objective value.
    """
    seq = f_seq.copy()
    n = len(seq)
    best_seq = seq.copy()
    best_val = _objective(seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # 1) identify the K indices where the self‑convolution is largest
        conv_self = _conv_fft(seq, seq)
        max_k = min(top_k_active, conv_self.size)
        active_idx = np.argpartition(-conv_self, max_k - 1)[:max_k]

        # 2) build the LP (variables: g_0 … g_{n‑1}, M)
        m = len(active_idx)
        A_ub = np.zeros((m, n + 1), dtype=np.float64)
        for row, k in enumerate(active_idx):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            j_range = np.arange(j_start, j_end + 1, dtype=int)
            A_ub[row, j_range] = f_seq[k - j_range]          # f_i coefficients
            A_ub[row, n] = -1.0                               # -M
        b_ub = np.zeros(m, dtype=np.float64)

        # equality Σ g_j = 1
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0], dtype=np.float64)

        # bounds: g_j ≥ 0, M ≥ 0
        bounds = [(0.0, None)] * n + [(0.0, None)]

        # objective: minimise M
        c = np.zeros(n + 1, dtype=np.float64)
        c[-1] = 1.0

        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})

        if not res.success:
            # Should rarely happen – abort the refinement loop.
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        g = res.x[:n]
        # Numerical clean‑up
        g = np.clip(g, 0.0, None)
        s = g.sum()
        if s == 0.0:
            g = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            g = g / s

        seq = g
        cur_val = _objective(seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = seq.copy()
            print(f"[LP] iter {it} best={best_val:.6f}")

        # Prepare for the next iteration
        f_seq = seq

    return best_seq, best_val

def _gaussian_seed(n: int, sigma_frac: float) -> np.ndarray:
    """Return an ℓ₁‑normalised discrete Gaussian (σ = sigma_frac·n)."""
    if sigma_frac <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _triangular_seed(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises ``evaluate_sequence``.  The routine respects the supplied
    time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a few seconds safety margin

    # ------------------------------------------------------------------
    # Length of the construction.  4000 works well for the FFT utilities.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Assemble a small but diverse set of seeds.
    # ------------------------------------------------------------------
    seeds = []

    # Uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Triangular shape
    seeds.append(_triangular_seed(n))

    # Gaussian windows with fairly large σ (these are the most promising)
    for sigma_frac in (0.25, 0.30, 0.35, 0.40, 0.45, 0.48, 0.50):
        seeds.append(_gaussian_seed(n, sigma_frac))

    # Random Dirichlet draws with several concentration parameters
    for conc in (0.01, 0.05, 0.10, 0.20, 0.50, 1.0):
        for _ in range(2):
            d = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            seeds.append(d / d.sum())

    # Re‑use the previously found sequence if it exists (gives a warm start)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = prev_arr / prev_arr.sum()
                seeds.append(prev_arr)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – active‑set LP refinement on each seed.
    # ------------------------------------------------------------------
    print("\n--- LP refinement of seeds ---")
    for idx, seed_seq in enumerate(seeds):
        if time.time() > deadline:
            break
        print(f"[Seed {idx + 1}/{len(seeds)}]")
        seq = seed_seq.copy()
        if seq.sum() == 0.0:
            continue
        seq = seq / seq.sum()
        # Run a modest number of LP steps (10 – 15) – cheap yet powerful.
        seq, val = _lp_refine(seq,
                              max_iters=12,
                              deadline=deadline,
                              top_k_active=400)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"  New global best after LP: {best_val:.6f}")

    # ------------------------------------------------------------------
    # If for any reason we have no candidate, fall back to uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)

    # ------------------------------------------------------------------
    # Stage 2 – final Adam polishing (uses the smooth‑max surrogate).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final Adam polishing ---")
        seq_adam, val_adam = _adam_optimize(best_seq,
                                            n,
                                            deadline,
                                            max_iters=350_000)
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  Updated global best after Adam: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # Store for possible reuse in subsequent runs.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
