def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to evaluate_sequence for a *normalised* sequence (sum == 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    """Flat window."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * math.pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * math.pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0 - a1 * np.cos(2 * math.pi * i / (n - 1))
         + a2 * np.cos(4 * math.pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑pdf on the interior of the interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _sin_shape(n: int) -> np.ndarray:
    """Half‑sine bump (zero at the ends)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(math.pi * i / n)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _spectral_factored(n: int) -> np.ndarray:
    """
    Spectral‑factorisation seed.
    We zero‑pad a length‑n vector to M = 2*n-1, take the DFT of the
    uniform target, compute a complex square‑root and inverse‑transform.
    The result is truncated to the first n entries, negative parts are
    set to zero and the vector is re‑normalised.
    """
    M = 2 * n - 1
    # target uniform convolution (length M, sum = 1)
    u = np.full(M, 1.0 / M, dtype=np.float64)

    # DFT of target
    U_hat = np.fft.rfft(u, n=M)

    # complex square‑root (preserves Hermitian symmetry automatically)
    A_hat = np.sqrt(U_hat)

    # inverse DFT → candidate impulse response of length M
    a_full = np.fft.irfft(A_hat, n=M)

    # keep only the first n entries, discard the rest (they are forced to zero)
    a = a_full[:n]

    # enforce non‑negativity and renormalise
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet‑like draw (Gamma/Exp) with concentration α."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


def _random_beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:

    b = rng.uniform(0.1, 5.0)
    return _beta_shape(n, a, b)


def _random_mixture_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two independent beta pdfs (random mixing weight)."""
    w = rng.random()
    a1, b1 = rng.uniform(0.1, 5.0), rng.uniform(0.1, 5.0)
    a2, b2 = rng.uniform(0.1, 5.0), rng.uniform(0.1, 5.0)
    s1 = _beta_shape(n, a1, b1)
    s2 = _beta_shape(n, a2, b2)
    mix = w * s1 + (1.0 - w) * s2
    mix /= mix.sum()
    return mix

def _smooth_adam_optimize(start_seq: np.ndarray,
                          deadline: float,
                          beta_start: float = 5.0,
                          beta_end: float = 1000.0,
                          max_iter: int = 5_000_000,
                          lr0: float = 0.03) -> (np.ndarray, float):
    """
    Minimise a smooth (log‑sum‑exp) approximation of max(conv(a,a)).
    Uses Adam on the log‑softmax (logits) parametrisation.
    """
    n = len(start_seq)
    eps = 1e-30

    logits = np.log(start_seq + eps)          # initialise logits
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # soft‑max → probability vector
        a = np.exp(logits)
        a /= a.sum()

        conv = _conv_fft(a, a)                # linear auto‑convolution

        # linear temperature schedule
        beta = beta_start + (beta_end - beta_start) * min(1.0,
                                                          it / (max_iter * 0.6))

        # smooth‑max (log‑sum‑exp)
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()          # soft‑weights, sum = 1

        # gradient w.r.t. a  (see derivation in the original code)
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate decays like 1/√it
        lr = lr0 / math.sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # periodic exact evaluation
        if (it & 1023) == 0:                # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

            if best_val < 1.45:            # early exit on a very good bound
                break

    return best_seq, best_val

    eps = 1e-30

    def fun_and_grad(logits: np.ndarray):
        a = np.exp(logits)
        a /= a.sum()
        conv = _conv_fft(a, a)
        c = np.max(conv)
        exps = np.exp(beta * (conv - c))
        lse = c + (1.0 / beta) * np.log(exps.sum())
        value = 2.0 * n * lse

        # gradient w.r.t. a
        p = exps / exps.sum()
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)
        return value, grad_logits.astype(np.float64)

    x0 = np.log(start_seq + eps)

    class TimeOut(RuntimeError):
        pass

    def callback(xk):
        if time.time() >= deadline:
            raise TimeOut()

    try:
        res = minimize(fun=lambda x: fun_and_grad(x)[0],
                       x0=x0,
                       jac=lambda x: fun_and_grad(x)[1],
                       method='L-BFGS-B',
                       options={'maxiter': 200, 'disp': False},
                       callback=callback)
    except TimeOut:
        best_logits = x0
    else:
        best_logits = res.x

    best_a = np.exp(best_logits)
    best_a /= best_a.sum()
    best_val = _objective(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive KL‑mirror‑descent polishing
# ----------------------------------------------------------------------


def _aggregated_mirror_descent(start_seq: np.ndarray,
                               deadline: float,
                               eps_rel_start: float = 0.01,
                               eps_rel_end: float = 1e-6,
                               lr_factor: float = 1.0) -> (np.ndarray, float):
    """
    KL‑mirror descent where the sub‑gradient aggregates *all* entries whose
    convolution value is within a relative tolerance of the running maximum.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    t0 = time.time()
    it = 0
    eps = 1e-30
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        # exponential decay of the relative tolerance
        elapsed = time.time() - t0
        total = deadline - t0
        frac = elapsed / total if total > 0 else 0.0
        eps_rel = eps_rel_start * (eps_rel_end / eps_rel_start) ** frac
        delta = eps_rel * max_val

        active_idx = np.where(conv >= max_val - delta)[0]
        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        # aggregated sub‑gradient = 2 * (active ⋆ a_rev)
        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        # exponentiated‑gradient step (KL‑mirror)
        lr = lr_factor / math.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)   # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[AggMD] it={it:7d} n={n:5d} best={val:.6f}")

            if time.time() > deadline - 0.5:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny stochastic hill‑climbing (mass‑re‑distribution)
# ----------------------------------------------------------------------


def _hill_climb(seq: np.ndarray,
                deadline: float,
                max_trials: int = 5_000_000,
                delta: float = 0.002) -> (np.ndarray, float):
    """
    Simple stochastic polishing:
    repeatedly pick two indices i≠j, move a small amount ε from
    i to j (ε ≤ delta and ≤ a_i) and keep the move if it improves the
    exact objective.
    """
    n = len(seq)
    best_seq = seq.copy()
    best_val = _objective(best_seq)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_trials:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose a transfer from i to j
        max_transfer = min(delta, best_seq[i])
        if max_transfer <= 0:
            continue
        eps = rng.random() * max_transfer
        cand = best_seq.copy()
        cand[i] -= eps
        cand[j] += eps
        # renormalise (numerically it stays normalised)
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand
            print(f"[Hill] it={it:7d} n={n:5d} val={val:.6f}")

            # early break if we beat the target already
            if best_val < 1.503:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective, respecting the supplied time budget.
    Returns the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # Re‑use a previously computed solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Problem sizes – a selection of FFT‑friendly lengths (up to 12 000)
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000, 8000, 10000, 12000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic families (seed generation)
    # ------------------------------------------------------------------
    seed_families = (
        _spectral_factored,
        _catalan_shape if ' _catalan_shape' in globals() else _uniform,  # placeholder
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.12),
        lambda n: _gaussian(n, sigma_factor=0.20),
        lambda n: _gaussian(n, sigma_factor=0.30),
        _sin_shape,
        lambda n: _beta_shape(n, 0.5, 0.5),
        lambda n: _beta_shape(n, 2.0, 2.0),
        lambda n: _beta_shape(n, 0.5, 2.0),
        lambda n: _beta_shape(n, 2.0, 0.5),
    )

    for n in lengths:
        for fn in seed_families:
            try:
                seq = fn(n)
            except Exception:
                continue
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:18s} val={val:.6f}")

        # reuse a previous solution of the same length, if any
        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – random sampling (Dirichlet + random Beta shapes)
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.30        # ~30 % of remaining budget
    sample_deadline = now + time_for_sampling

    alphas = [0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]
    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))

        # Dirichlet draw with random concentration
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd-Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        # Random Beta pdf
        seq = _random_beta_shape(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd-Beta] n={n:5d} val={val:.6f}")

        # Random mixture of two Betas
        seq = _random_mixture_beta(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd-Mix] n={n:5d} val={val:.6f}")

        if best_val < 1.45:            # early stop if already very good
            break

    # ------------------------------------------------------------------
    # Phase 3 – Smoothed Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # keep a few diverse restart candidates
        candidates = [best_seq]
        top_len = len(best_seq)

        # two fresh Dirichlet starts of the same length
        for _ in range(2):
            cand = _random_dirichlet(top_len, rng, alpha=rng.choice(alphas))
            candidates.append(cand)

        # one larger‑length start (if possible, to explore bigger n)
        larger_lengths = [L for L in lengths if L > top_len]
        if larger_lengths:
            L_big = int(rng.choice(larger_lengths))
            cand_big = _random_dirichlet(L_big, rng, alpha=1.0)
            candidates.append(cand_big)

        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.25   # ≈25 % of remaining time

        for idx, cand in enumerate(candidates):
            seq_adam, val_adam = _smooth_adam_optimize(
                start_seq=cand,
                deadline=adam_deadline,
                beta_start=5.0,
                beta_end=1000.0,
                max_iter=5_000_000,
                lr0=0.03,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = seq_adam
                print(f"[Adam‑Refine {idx}] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – optional L‑BFGS refinement (if SciPy is available)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0 and minimize is not None:
            lbfgs_deadline = time.time() + remaining * 0.20
            seq_lbfgs, val_lbfgs = _lbfgs_refine(best_seq, lbfgs_deadline, beta=1000.0)
            if val_lbfgs < best_val:
                best_val = val_lbfgs
                best_seq = seq_lbfgs
                print(f"[L‑BFGS] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – aggressive aggregated mirror‑descent polishing
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _aggregated_mirror_descent(
            start_seq=best_seq,
            deadline=deadline,
            eps_rel_start=0.01,
            eps_rel_end=1e-6,
            lr_factor=1.0,
        )
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Agg‑MD] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – tiny stochastic hill‑climbing (final fine‑tune)
    # ------------------------------------------------------------------
    if best_seq is not None:
        hc_deadline = deadline - 0.2   # keep a tiny safety margin
        if hc_deadline > time.time():
            seq_hc, val_hc = _hill_climb(best_seq, hc_deadline,
                                         max_trials=2_000_000,
                                         delta=0.0015)
            if val_hc < best_val:
                best_val = val_hc
                best_seq = seq_hc
                print(f"[Hill‑Climb] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will evaluate again anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:      # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )

    # expose the result for possible reuse
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
