"""
Improved 1000‑second search for the low‑value step‑function.

The optimiser works in four stages:

  1. **Rich seed pool** – a large collection of analytic families
     (uniform, triangular, sine‑/beta‑/Gaussian‑windows, power‑law,
     random Dirichlet draws, sparse supports and mixtures).  The 20
     best seeds (by the exact objective) are kept.

  2. **Fast sub‑gradient refinement** – each kept seed undergoes a
     short projected sub‑gradient descent (≈2·10⁵ iterations) that
     quickly pushes the maximal convolution down.

  3. **Long refinement** – the current global best is refined for the
     remaining time by a long sub‑gradient run (up to a few million
     iterations).  The step size follows a Polyak rule using the best
     value seen so far.

  4. **Annealing polish** – a lightweight simulated‑annealing walk
     (few × 10⁵ moves) is applied as a final polish.

All convolutions are performed with a real FFT (O(n log n)), the
iterates are kept on the probability simplex by Euclidean projection,
and the routine respects the supplied time budget.  The best sequence
found is stored in the global variable ``height_sequence_1`` for reuse
and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` (scale‑invariant).
    The vector is assumed to be non‑negative; it does *not* enforce a
    specific ℓ₁‑norm – that is irrelevant for the ratio.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a symmetric Dirichlet distribution."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random support with 3‑30 non‑zero entries."""
    k = rng.integers(3, 31)
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


def _mixture_uniform_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Blend uniform and a Gaussian window."""
    lam = rng.random()
    g = _gaussian(n, sigma_frac=rng.uniform(0.05, 0.4))
    u = _uniform(n)
    a = lam * u + (1.0 - lam) * g
    return a / a.sum()


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _subgrad_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient w.r.t. the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size (using the best value seen so far)
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.2 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.03,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Tiny‑step simulated‑annealing on the exact objective."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.9997, min_temp)

        # choose a small subset and re‑allocate its total mass
        k = rng.integers(2, 13)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total == 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def _collect_diverse_seeds(n: int, rng: np.random.Generator,
                          previous: np.ndarray | None,

    top_vals = []          # list of (value, seq)

    def try_insert(val, seq):
        if len(top_vals) < top_k:
            top_vals.append((val, seq))
            top_vals.sort(key=lambda x: x[0])
        elif val < top_vals[-1][0]:
            top_vals[-1] = (val, seq)
            top_vals.sort(key=lambda x: x[0])

    # Deterministic families (always worth evaluating)
    families = [
        _uniform,
        _triangular,
        lambda n: _sine_window(n, gamma=0.5),
        lambda n: _sine_window(n, gamma=1.0),
        lambda n: _sine_window(n, gamma=2.0),
        lambda n: _gaussian(n, sigma_frac=0.10),
        lambda n: _gaussian(n, sigma_frac=0.20),
        lambda n: _gaussian(n, sigma_frac=0.30),
        lambda n: _beta_shape(n, alpha=0.5, beta=0.5),
        lambda n: _beta_shape(n, alpha=2.0, beta=2.0),
        lambda n: _power_law(n, p=0.5),
        lambda n: _power_law(n, p=1.0),
        lambda n: _power_law(n, p=2.0),
    ]
    for fn in families:
        seq = fn(n)
        try_insert(_objective(seq), seq)

    # Random Dirichlet draws with many concentration levels
    for conc in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        draws = 80 if conc < 0.02 else 40
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = _dirichlet_seed(n, conc, rng)
            try_insert(_objective(seq), seq)

    # Sparse random supports
    for _ in range(300):
        if time.time() > deadline:
            break
        seq = _sparse_seed(n, rng)
        try_insert(_objective(seq), seq)

    # Mixtures and random blends
    for _ in range(250):
        if time.time() > deadline:
            break
        seq = _mixture_uniform_gaussian(n, rng)
        try_insert(_objective(seq), seq)

    # Random beta windows
    for _ in range(200):
        if time.time() > deadline:
            break
        a = rng.uniform(0.2, 4.0)
        b = rng.uniform(0.2, 4.0)
        seq = _beta_shape(n, alpha=a, beta=b)
        try_insert(_objective(seq), seq)

    # Random sine‑window exponents
    for _ in range(200):
        if time.time() > deadline:
            break
        g = rng.uniform(0.5, 3.0)
        seq = _sine_window(n, gamma=g)
        try_insert(_objective(seq), seq)

    # Random Gaussian bandwidths
    for _ in range(200):
        if time.time() > deadline:
            break
        sf = rng.uniform(0.05, 0.35)
        seq = _gaussian(n, sigma_frac=sf)
        try_insert(_objective(seq), seq)

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        seq = np.asarray(previous, dtype=np.float64)
        if seq.sum() > 0:
            seq = seq / seq.sum()
            try_insert(_objective(seq), seq)

    # Return only the sequences sorted by increasing objective value
    return [seq for _, seq in top_vals]

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises ``evaluate_sequence``.  The routine respects the supplied
    time budget (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Length – a power of two gives fast FFTs while keeping a large
    # search space.  4096 works well for the hardware used in the
    # evaluation harness.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Stage 0 – burst of diverse seeds.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    seed_deadline = min(deadline, start_time + budget_s * 0.22)
    seeds = _collect_diverse_seeds(
        n, rng, previous, top_k=20, deadline=seed_deadline
    )
    print(f"Collected {len(seeds)} promising seeds (best seed value ≈ {_objective(seeds[0]):.6f})")

    # ------------------------------------------------------------------
    # Stage 1 – quick sub‑gradient refinement of each seed.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Quick sub‑gradient refinements (stage 1) ---")
    for idx, seed_vec in enumerate(seeds):
        if time.time() > deadline:
            break
        max_iters = 120_000
        print(f"[Seed {idx + 1}/{len(seeds)}] start quick refinement")
        seq_sub, val_sub = _subgrad_optimize(
            seed_vec, n, deadline, max_iters=max_iters
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = seq_sub.copy()
            print(f"  New global best after quick sub‑grad: {best_val:.6f}")

    # If something went wrong, fall back to the best raw seed.
    if best_seq is None:
        best_seq = seeds[0].copy()
        best_val = _objective(best_seq)
        print(f"Fallback to first raw seed: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long sub‑gradient run on the current best.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        long_iters = int(min(3_000_000, remaining / 0.00025))  # rough bound
        print("\n--- Long sub‑gradient refinement (stage 2) ---")
        seq_sub, val_sub = _subgrad_optimize(
            best_seq, n, deadline, max_iters=long_iters
        )
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"  Updated global best after long sub‑grad: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final simulated‑annealing polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing (stage 3) ---")
        seq_sa, val_sa = _simulated_annealing(
            best_seq, n, deadline, max_iters=400_000, init_temp_factor=0.04
        )
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"  Updated global best after SA: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(best_seq.tolist())
        print(f"\nOfficial evaluator returned: {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"with length {len(best_seq)}"
    )

    # Store for possible reuse and return a plain Python list.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
