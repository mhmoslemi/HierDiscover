    def _upsample(seq, n):
        m = len(seq)
        if m == n:
            return seq.copy()
        x_old = np.linspace(0, 1, m)
        x_new = np.linspace(0, 1, n)
        new_seq = np.interp(x_new, x_old, seq)
        return new_seq / new_seq.sum()  # ensure sum = 1

"""
Optimiser for the step‑function attack.

The objective (scale‑invariant) is 2·n·max(conv(a,a)) with the constraint
∑a = 1, a_i ≥ 0.  We treat the vector as a probability distribution on
{0,…,n‑1}.  The algorithm consists of three stages:

1.  Build several deterministic / random seeds (including an up‑sampled
    previous best) and keep the one with the smallest true objective.
2.  Run a long Adam optimiser on the simplex.  The max‑convolution is
    smoothed by a soft‑max with an inverse‑temperature λ that is
    increased slowly (λ←λ·λ_factor) so the optimiser first sees a smooth
    landscape and later focuses on the true (hard) max.  Adam works on the
    log‑softmax parameters, guaranteeing that the iterate always stays on
    the simplex.
3.  Finish with a light Metropolis‑Hastings (simulated‑annealing) walk
    that swaps a small amount of mass between random coordinates and
    occasionally performs a global Gaussian perturbation.

The search space is taken with n = 8000 (twice the size of the previous
run) – this gives more flexibility while keeping the per‑iteration cost
reasonable (FFT of length 16384).  The whole routine respects the given
time budget (default 1000 s) and stores the best vector found in the
global name ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (full mode)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective (used for reporting and final comparison)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)^2).  Assumes seq ≥ 0."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Soft‑max weighting of convolution entries (λ = inverse temperature)
# ----------------------------------------------------------------------
def _soft_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Return a probability vector w that is a soft‑max over `conv`.
    For lam → ∞ the hard max (uniform over ties) is recovered.
    """
    max_c = conv.max()
    if lam <= 0.0:
        # hard max – uniform over all entries that achieve the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        cnt = max(1, np.count_nonzero(active))
        w = np.where(active, 1.0 / cnt, 0.0)
        return w

    # work with relative values for stability
    rel = (conv - max_c) / max_c          # ≤ 0
    exps = np.exp(lam * rel)
    w = exps / exps.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution w.r.t. the probability vector
# ----------------------------------------------------------------------
def _gradient_wrt_a(a: np.ndarray, lam: float) -> np.ndarray:
    """∇_a (soft‑max_λ max(conv(a,a))) ."""
    n = a.size
    conv = _conv_fft(a, a)               # length = 2n‑1
    w = _soft_max_weights(conv, lam)     # same length
    rev_a = a[::-1]
    # rev(a) ⊛ w  (full convolution)
    full = _conv_fft(rev_a, w)           # length = n + (2n‑1) – 1 = 3n‑2
    # extract the slice that aligns with a
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 500_000,
                   lam_start: float = 0.5,
                   lam_factor: float = 1.0005) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex.
    The iterate `a` is obtained from log‑probabilities via a soft‑max.
    The smoothing parameter λ begins at `lam_start` and is multiplied by
    `lam_factor` each iteration.
    Returns the best sequence (probability vector) seen and its true
    objective value.
    """
    eps = 1e-30
    logp = np.log(start + eps)          # initialise log‑probabilities
    n = start.size
    m = np.zeros_like(logp)             # first moment
    v = np.zeros_like(logp)             # second moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = start.copy()
    best_val = _objective(start)

    t = 0
    lam = lam_start

    while t < max_iters and time.time() < deadline:
        t += 1

        # ----- current point on simplex -----
        logp_shift = logp - np.max(logp)          # for numerical stability
        a = np.exp(logp_shift)
        a /= a.sum()

        # ----- gradient of the smoothed objective -----
        g_a = _gradient_wrt_a(a, lam)

        # chain rule: ∂/∂logp  (soft‑max) = a * (g_a – <a,g_a>)
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log * g_log)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.15 / (math.sqrt(t) + 1.0)          # diminishing step size
        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- evaluate the *true* (hard‑max) objective -----
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:7d}  best={cur_val:.6f}  λ={lam:.2e}")

        # ----- temperature schedule -----
        lam *= lam_factor

        # ----- occasional random kick if stuck -----
        if t % 25000 == 0:
            # small perturbation of log‑probabilities
            noise = np.random.default_rng().normal(scale=0.02, size=n)
            logp += noise

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light simulated‑annealing polishing
# ----------------------------------------------------------------------
def _anneal_refine(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Metropolis–Hastings walk that moves a tiny fraction of mass between
    random coordinates and occasionally performs a global Gaussian jump.
    The temperature decays geometrically.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    n = cur.size
    T0 = max(1e-6, cur_val * 0.04)   # initial temperature proportional to current value
    decay = 0.9995

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # ----- local move: transfer a small fraction from i to j -----
        i, j = rng.integers(0, n, size=2)
        if i != j:
            delta = rng.random() * 0.02 * cur[i]   # at most 2 % of the source mass
            delta = min(delta, cur[i])
            if delta > 0:
                new_seq = cur.copy()
                new_seq[i] -= delta
                new_seq[j] += delta
                new_val = _objective(new_seq)

                accept = (new_val < cur_val) or \
                         (rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)))
                if accept:
                    cur, cur_val = new_seq, new_val
                    if new_val < best_val:
                        best, best_val = new_seq.copy(), new_val
                        print(f"[Anneal] it={it:7d}  best={new_val:.6f}")

        # ----- global perturbation every few thousand steps -----
        if it % 2000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0:
                new_seq /= total
            else:
                new_seq = _random_softmax(n, rng)

            new_val = _objective(new_seq)
            accept = (new_val < cur_val) or \
                     (rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)))
            if accept:
                cur, cur_val = new_seq, new_val
                if new_val < best_val:
                    best, best_val = new_seq.copy(), new_val
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Helper seed generators
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Flat probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gaussian logits → soft‑max."""
    logits = rng.standard_normal(n)
    m = np.max(logits)
    exps = np.exp(logits - m)
    return exps / exps.sum()


def _exponential_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    """Increasing then decreasing linear shape."""
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        dec = inc[::-1]
    else:
        dec = np.arange(half, 0, -1, dtype=np.float64)
    a = np.concatenate([inc, dec])
    return a / a.sum()


def _upsample(seq: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    if seq.size == n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, seq.size)
    x_new = np.linspace(0.0, 1.0, n)
    new_seq = np.interp(x_new, x_old, seq)
    # Numerical noise could introduce tiny negatives – clip just in case
    new_seq = np.clip(new_seq, 0.0, None)
    total = new_seq.sum()
    if total > 0:
        new_seq /= total
    else:
        new_seq = _uniform(n)
    return new_seq
