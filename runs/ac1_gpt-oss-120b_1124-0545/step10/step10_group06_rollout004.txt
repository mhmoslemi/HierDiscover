def adam_optimize(start, deadline):
    eps = 1e-30
    logp = np.log(start + eps)
    n = start.size
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)
    beta1=0.9; beta2=0.999; eps_adam=1e-8
    best_seq = start.copy()
    best_val = _objective(start)
    t=0
    lam=0.5
    lam_factor = 1.0003

    while time.time() < deadline:
        t+=1
        # compute a
        max_log = np.max(logp)
        a = np.exp(logp - max_log)
        a /= a.sum()
        # compute gradient
        conv = _conv_fft(a, a)
        # compute softmax weights
        max_c = conv.max()
        # compute weights w in a stable way
        # approx softmax using lam, but lam may be moderate
        rel = (conv - max_c) / max_c  # could be nan if max_c=0? but conv nonzero.
        exps = np.exp(lam * rel)
        w = exps / exps.sum()
        # compute gradient wrt a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1:n-1+n]
        # chain rule to logp
        dot = np.dot(a, grad_a)
        grad_log = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log * grad_log)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = 0.15 / (np.sqrt(t) + 1.0)
        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # evaluate objective
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:6d} best={cur_val:.6f} lam={lam:.2f}")
        lam *= lam_factor
        # optional random perturbation if stagnation - maybe skip.
    return best_seq, best_val

"""
Propose a non‑negative step function (probability vector) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The algorithm runs three optimisation phases within the given time budget:

1. **Adam on the simplex** – optimisation of a smoothed version of the
   max‑convolution using a log‑softmax (exponentiated) parametrisation.
   The smoothing parameter λ starts small (≈0.5) and grows very slowly
   (factor 1.0003), allowing the optimiser to explore the landscape before
   focusing on the true non‑smooth objective.

2. **Projected sub‑gradient descent on the true objective** – after Adam the
   exact sub‑gradient of the L∞‑norm of the convolution is used (hard max).
   Each step moves the probability vector opposite to the sub‑gradient and
   projects back onto the probability simplex.

3. **Light simulated‑annealing polish** – a Metropolis–Hastings walk that
   swaps small mass fractions between random coordinates and occasionally
   performs a global Gaussian jump.  This phase finishes the search.

All phases respect the overall time budget (default 1000 s) and keep the
best sequence encountered.  The final sequence is stored in the global
variable ``height_sequence_1`` and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective (exact – same as evaluate_sequence but without clipping)
# ----------------------------------------------------------------------
def _objective(p: np.ndarray) -> float:
    """Exact objective for a probability vector p (sum(p)=1, p≥0)."""
    n = p.size
    conv = _conv_fft(p, p)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. 2008
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, non‑negative)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed families (return a probability vector that sums to 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    exps = np.exp(logits - m)
    return exps / exps.sum()


def _exponential_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    dec = inc[::-1] if n % 2 == 0 else np.arange(half, 0, -1, dtype=np.float64)
    a = np.concatenate([inc, dec])
    return a / a.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smoothed max‑convolution.
    Returns the best probability vector found and its objective value.
    """
    eps = 1e-30
    logp = np.log(start + eps)
    n = start.size
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = start.copy()
    best_val = _objective(start)

    t = 0
    lam = 0.5               # start with a gentle smoothing
    lam_factor = 1.0003

    while time.time() < deadline:
        t += 1

        # ---- compute current probability vector (stable softmax) ----
        shift = logp - np.max(logp)
        a = np.exp(shift)
        a /= a.sum()

        # ---- gradient of the smoothed max‑convolution ----
        conv = _conv_fft(a, a)
        max_c = conv.max()
        # soft‑max weights (relative to the current max)
        rel = (conv - max_c) / max_c        # ≤ 0
        exps = np.exp(lam * rel)
        w = exps / exps.sum()

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice that matches a

        # ---- back‑propagation to log‑probabilities (mirror descent) ----
        dot = np.dot(a, grad_a)
        grad_log = a * (grad_a - dot)

        # ---- Adam update (bias‑corrected) ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.15 / (math.sqrt(t) + 1.0)

        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- evaluate true objective (hard max) ----
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:7d}  best={cur_val:.6f}  λ={lam:.2f}")

        # ---- increase smoothing parameter slowly ----
        lam *= lam_factor

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------
def _subgradient_optimize(start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on 2·n·max(conv(p,p)).
    Uses the exact (hard) max, projects onto the simplex after each step.
    """
    p = start.copy()
    n = p.size
    best_seq = p.copy()
    best_val = _objective(p)

    t = 0
    step0 = 0.10           # initial step size (will decay as 1/√t)

    while time.time() < deadline:
        t += 1

        conv = _conv_fft(p, p)
        max_idx = int(conv.argmax())

        # gradient of the max‑convolution (hard max)
        w = np.zeros_like(conv)
        w[max_idx] = 1.0
        rev_p = p[::-1]
        full = _conv_fft(rev_p, w)
        grad = 2.0 * full[n - 1:n - 1 + n]

        step = step0 / math.sqrt(t)
        p_new = p - step * grad
        p_new = _project_simplex(p_new)

        cur_val = _objective(p_new)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = p_new.copy()
            print(f"[Subgrad] it={t:7d}  best={cur_val:.6f}")

        p = p_new

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light simulated‑annealing polishing
# ----------------------------------------------------------------------
def _anneal_refine(start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis–Hastings walk that swaps a small fraction of mass
    between random coordinates and occasionally performs a global Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    n = cur.size
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # ---- small local move -------------------------------------------------
        i, j = rng.integers(0, n, size=2)
        if i != j:
            delta = rng.random() * 0.02 * cur[i]       # up to 2 % of i‑mass
            delta = min(delta, cur[i])
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            new_val = _objective(new_seq)

            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d}  best={new_val:.6f}")

        # ---- occasional global Gaussian perturbation ---------------------------
        if it % 2000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0.0:
                new_seq /= total
            else:
                new_seq = _uniform(n)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises evaluate_sequence().
    The function respects a time budget (default 1000 s) and returns the best
    vector found as a list of floats.  The best sequence is stored in the global
    variable ``height_sequence_1``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # safety margin

    rng = np.random.default_rng(seed + 12345)

    # ------------------------------------------------------------------
    # Problem size – pick a size that balances expressiveness and speed.
    # ------------------------------------------------------------------
    # Try to keep the previous length if we already have a result.
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            n = prev_arr.shape[0]
        except Exception:
            n = 4000
    else:
        n = 4000

    print(f"\n=== Optimising step function (length {n}) ===")

    # ------------------------------------------------------------------
    # Build initial seeds (probability vectors)
    # ------------------------------------------------------------------
    seeds = []

    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                if prev_arr.sum() > 0:
                    seeds.append(prev_arr / prev_arr.sum())
        except Exception:
            pass

    seeds.extend([
        _uniform(n),
        _random_softmax(n, rng),
        _exponential_decay(n, rate=0.001),
        _triangular(n),
    ])

    # ------------------------------------------------------------------
    # Choose the best seed according to the exact objective
    # ------------------------------------------------------------------
    best_seed = None
    best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seed = s
    print(f"[Init] best initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Allocate time to the three phases
    # ------------------------------------------------------------------
    total_remaining = deadline - time.time()
    phase1_deadline = time.time() + 0.55 * total_remaining   # Adam
    phase2_deadline = None
    phase3_deadline = None

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation (smooth max)
    # ------------------------------------------------------------------
    if time.time() < phase1_deadline:
        print("[Phase 1] Adam optimisation start")
        seq_adam, val_adam = _adam_optimize(best_seed, phase1_deadline)
        print(f"[Phase 1] best after Adam = {val_adam:.6f}")
        best_seq = seq_adam
        best_val = val_adam
    else:
        seq_adam, val_adam = best_seed, best_val
        best_seq, best_val = seq_adam, val_adam
        print("[Phase 1] skipped (no time)")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent on the true L∞ objective
    # ------------------------------------------------------------------
    phase2_deadline = time.time() + 0.30 * (deadline - time.time())
    if time.time() < phase2_deadline:
        print("[Phase 2] Projected sub‑gradient descent")
        seq_sub, val_sub = _subgradient_optimize(best_seq, phase2_deadline)
        if val_sub < best_val:
            best_seq, best_val = seq_sub, val_sub
            print(f"[Phase 2] improved to {val_sub:.6f}")
        else:
            print("[Phase 2] no improvement")
    else:
        print("[Phase 2] skipped (no time)")

    # ------------------------------------------------------------------
    # Phase 3 – Light Simulated Annealing polishing
    # ------------------------------------------------------------------
    phase3_deadline = deadline
    if time.time() < phase3_deadline - 0.5:
        print("[Phase 3] Simulated‑annealing polishing")
        seq_final, val_final = _anneal_refine(best_seq, phase3_deadline)
        if val_final < best_val:
            best_seq, best_val = seq_final, val_final
            print(f"[Phase 3] improved to {val_final:.6f}")
        else:
            print("[Phase 3] no improvement")
    else:
        print("[Phase 3] skipped (no time)")

    # ------------------------------------------------------------------
    # Final normalisation and export
    # ------------------------------------------------------------------
    final_seq = np.maximum(best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
        total = final_seq.sum()
    final_seq = final_seq / total

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
