def propose_candidate(seed=123, budget_s=1000, **kwargs):
    import time, numpy as np, scipy.optimize as opt
    start = time.time()
    deadline = start + budget_s - 1

    # set n
    n = 4000

    # create rng
    rng = np.random.default_rng(seed)

    # define functions: same as prior _uniform, _triangular, etc.

    # Build basis
    basis = []
    basis.append(_uniform(n))
    basis.append(_triangular(n))
    for wf in np.linspace(0.05,0.4,8):
        basis.append(_trapezoidal(n, width_fraction=float(wf)))
    for gamma in np.linspace(0.0, 2.5, 6):
        basis.append(_sine_window(n, gamma=float(gamma)))
    for alpha in np.linspace(0.0, 0.9, 5):
        basis.append(_tukey_window(n, alpha=float(alpha)))
    for r in (0.5, 0.7, 0.85, 0.95, 0.98):
        basis.append(_geom(n, r))
        basis.append(_sym_geom(n, r))
    for p in (0.5,1.0,1.5,2.0,3.0):
        basis.append(_power(n, p))
    for a,b in [(0.5,0.5),(1.0,2.0),(2.0,1.0),(2.0,2.0)]:
        basis.append(_beta_shape(n,a,b))
    for sigma_frac in (0.12,0.18,0.24):
        basis.append(_gaussian(n, sigma_frac))

    # add random Dirichlet with various conc
    for conc in (0.1, 0.5, 1.0, 2.0):
        for _ in range(2):
            basis.append(_random_dirichlet(n, conc, rng))
    # add previous solution if exists and matches length
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape[0] == n:
                prev_norm = prev_arr / prev_arr.sum()
                basis.append(prev_norm)
        except Exception:
            pass

    basis = [b for b in basis if b.size == n]  # ensure length
    # Stack into matrix shape (B,n)
    basis_mat = np.vstack(basis)
    B = basis_mat.shape[0]

    # mixture objective
    def mixture_obj(logits):
        w = _softmax(np.array(logits))
        a = np.dot(w, basis_mat)  # (n,)
        # normalization
        s = a.sum()
        if s <= 0:
            return np.inf
        a = a / s
        return _objective(a)  # surrogate objective

    # DE bounds
    bound = (-5.0, 5.0)
    bounds = [bound] * B

    # time check before DE
    de_maxiter = 150
    try:
        result = opt.differential_evolution(
            mixture_obj,
            bounds,
            maxiter=de_maxiter,
            popsize=15,
            polish=False,
            disp=False,
            seed=seed,
            workers=1,  # single thread due to GIL
            updating='deferred',
            init='latinhypercube',
            # callback to check time
            callback=lambda x, convergence: time.time() > deadline
        )
    except Exception as e:
        print("DE exception:", e)
        result = None

    # get best sequence
    if result is not None and result.success:
        best_logits = result.x
        best_w = _softmax(best_logits)
        best_seq = np.dot(best_w, basis_mat)
        best_seq = best_seq / best_seq.sum()
        best_val = _objective(best_seq)
        print("[DE] best surrogate value:", best_val)
    else:
        # fallback to previous best random seed

"""
Improved search for a non‑negative step‑function that minimises
``evaluate_sequence``.

The algorithm builds a modest dictionary of analytic “basis” vectors
(e.g. uniform, trapezoidal, geometric, beta‑shaped …) and then searches
the convex hull of these bases with a global differential‑evolution
optimizer (scipy).  The resulting mixture is fine‑tuned with the Adam
surrogate optimiser and a short projected sub‑gradient descent.  All
operations respect a strict time budget (default 1000 s) and the best
sequence is stored as ``height_sequence_1`` for later reuse.
"""

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Global containers used by the DE callback (no closures are allowed)
# ----------------------------------------------------------------------
GLOBAL_BASIS_MAT = None          # type: np.ndarray | None
GLOBAL_DEADLINE = None           # type: float | None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised sequences (∑a = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors of length n)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Optimisers

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 500_000,
                   beta_max: float = 800.0) -> tuple[np.ndarray | None, float]:
    """Adam optimiser on the log‑sum‑exp surrogate.
    Parameters are logits of the soft‑max representation.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # initialise logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0          # start smooth
    beta_mul = 1.003

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        # anneal temperature (increase β)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip gradient norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 1_000_000,
                      init_lr: float = 0.20) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val

def _mixture_obj(logits: np.ndarray) -> float:
    """Surrogate objective for a convex combination of the basis vectors."""
    global GLOBAL_BASIS_MAT
    w = _softmax(np.array(logits))
    a = np.dot(w, GLOBAL_BASIS_MAT)        # shape (n,)
    s = a.sum()
    if s <= 0.0:
        return np.inf
    a = a / s
    return _objective(a)


def _de_time_callback(xk, convergence) -> bool:
    """Terminate DE when the global deadline is exceeded."""
    global GLOBAL_DEADLINE
    return time.time() > GLOBAL_DEADLINE

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin
    # expose deadline to the DE callback
    global GLOBAL_DEADLINE
    GLOBAL_DEADLINE = deadline

    # ------------------------------------------------------------------
    # Length – 4000 is a good compromise (used in the reference code).
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build a heterogeneous dictionary of basis vectors (all ℓ₁‑normalised).
    # ------------------------------------------------------------------
    bases = []

    # deterministic families (few parameter values)
    bases.append(_uniform(n))
    bases.append(_triangular(n))

    for wf in np.linspace(0.05, 0.40, 8):
        bases.append(_trapezoidal(n, width_fraction=float(wf)))

    for gamma in np.linspace(0.0, 2.5, 6):
        bases.append(_sine_window(n, gamma=float(gamma)))

    for alpha in np.linspace(0.0, 0.9, 5):
        bases.append(_tukey_window(n, alpha=float(alpha)))

    for r in (0.5, 0.7, 0.85, 0.95, 0.98):
        bases.append(_geom(n, r))
        bases.append(_sym_geom(n, r))

    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        bases.append(_power(n, p))

    for a, b in [(0.5, 0.5), (1.0, 2.0), (2.0, 1.0), (2.0, 2.0)]:
        bases.append(_beta_shape(n, a, b))

    for sigma_frac in (0.12, 0.18, 0.24):
        bases.append(_gaussian(n, sigma_frac))

    # random Dirichlet / exponential draws – add a few of each
    for conc in (0.1, 0.5, 1.0, 2.0):
        for _ in range(2):
            bases.append(_random_dirichlet(n, conc, rng))
    for _ in range(4):
        bases.append(_random_exponential(n, rng))

    # previous solution (if present and compatible)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_norm = prev_arr / prev_arr.sum()
                bases.append(prev_norm)
        except Exception:
            pass

    # Stack into a matrix (B × n)
    basis_mat = np.vstack(bases)
    B = basis_mat.shape[0]

    # expose to DE helper
    global GLOBAL_BASIS_MAT
    GLOBAL_BASIS_MAT = basis_mat

    print(f"\n--- Basis matrix: {B} vectors of length {n} ---")

    # ------------------------------------------------------------------
    # Stage 1 – global optimisation in the (convex) hull via DE.
    # ------------------------------------------------------------------
    print("\n--- Differential Evolution (global search) ---")
    bounds = [(-5.0, 5.0)] * B
    # The callback will stop DE as soon as the deadline is reached.
    try:
        result = differential_evolution(
            _mixture_obj,
            bounds,
            maxiter=150,
            popsize=15,
            polish=False,
            disp=False,
            seed=seed,
            callback=_de_time_callback,
        )
    except Exception as exc:
        print("DE failed:", exc)
        result = None

    if result is not None and result.success:
        best_logits = result.x
        best_weights = _softmax(best_logits)
        best_seq = np.dot(best_weights, basis_mat)
        best_seq = best_seq / best_seq.sum()
        best_val = _objective(best_seq)
        print(f"[DE] best surrogate value = {best_val:.6f}")
    else:
        # fallback – take the best single basis (the one with smallest surrogate)
        print("[DE] fallback to best individual basis.")
        evaluated = [(b, _objective(b)) for b in bases]
        evaluated.sort(key=lambda x: x[1])
        best_seq = evaluated[0][0].copy()
        best_val = evaluated[0][1]
        print(f"[fallback] value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – local refinement with Adam on the smooth surrogate.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(remaining * 2000)               # generous budget estimate
        max_iters = max(200_000, min(max_iters, 800_000))
        print(f"\n--- Adam refinement (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(best_seq, n, deadline,
                                            max_iters=max_iters,
                                            beta_max=800.0)
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam
            best_val = val_adam
            print(f"[Adam] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final projected sub‑gradient descent (true max objective).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(remaining * 1500)
        max_iters = max(300_000, min(max_iters, 2_000_000))
        print(f"\n--- Sub‑gradient refinement (max_iters={max_iters}) ---")
        seq_proj, val_proj = _subgradient_proj(best_seq, n, deadline,
                                               max_iters=max_iters,
                                               init_lr=0.20)
        if val_proj < best_val:
            best_seq = seq_proj
            best_val = val_proj
            print(f"[SubProj] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Sanity check – official evaluation.
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # Store for possible later reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
