def run_annealing(start_seq, deadline, n, rng):
    # a: copy.
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_val = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_val = cur_val

    # Temperature schedule
    T0 = cur_val * 0.08 # maybe
    T = T0
    decay = 0.9997
    it = 0
    max_fraction = 0.25 # max delta fraction initial.

    # Periodic renorm?
    # We'll update T and max_fraction each iteration: max_frac = max_fraction * (T/T0)

    # Use jitter sigma schedule: start sigma = 0.02, decreasing with T.

    while time.time() < deadline:
        it += 1
        # temperature decay
        T = T0 * (decay ** it)
        # compute max fraction for delta
        move_frac = max_fraction * (T / T0)  # linear decay.

        # choose move
        if rng.random() < 0.9:
            # mass transfer
            # source index weighted by a
            if a.sum() == 0:
                break
            src = rng.choice(n, p=a / a.sum())
            dest = rng.integers(0, n)
            if dest == src:
                continue
            delta = move_frac * a[src] * rng.random()
            if delta <= 0:
                continue
            # apply move
            apply_transfer(a, conv, src, dest, delta)
            new_val = 2.0 * n * conv.max()
            # acceptance
            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_seq = a.copy()
                    print(f"[Anneal] it={it} best={best_val:.6f}")
            else:
                # revert
                apply_transfer(a, conv, dest, src, delta)
        else:
            # jitter
            sigma = move_frac * 0.05  # small
            new_a = a + rng.normal(scale=sigma, size=n)
            new_a = np.clip(new_a, 0.0, None)
            new_a = _project_simplex(new_a)
            conv_new = _conv_fft(new_a, new_a)
            new_val = 2.0 * n * conv_new.max()
            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                a = new_a
                conv = conv_new
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_seq = a.copy()
                    print(f"[Anneal] it={it} best={best_val:.6f}")

        # occasional stronger jitter: every 5000 steps
        if it % 5000 == 0:
            # stronger jitter
            sigma = move_frac * 0.2
            new_a = a + rng.normal(scale=sigma, size=n)
            new_a = np.clip(new_a, 0.0, None)

            conv_new = _conv_fft(new_a, new_a)
            new_val = 2.0 * n * conv_new.max()
            if new_val < best_val:
                best_val = new_val
                best_seq = new_a.copy()
                print(f"[Anneal] (strong jump) it={it} best={best_val:.6f}")

    return best_seq, best_val

"""
Hybrid search for a low‑value step‑function.

The optimizer consists of three phases:

1. **Rich seeding** – a large library of analytic windows,
   geometric / power‑law families and many random Dirichlet draws
   (with tiny concentration for sparse vectors) is built.
   All candidates are normalised to sum = 1 and the best few are kept.

2. **Fast simulated‑annealing** – starting from the best seed a
   specialised annealer works directly on the convolution vector.
   The dominant move type is a *mass‑transfer*: a small amount of
   probability mass is moved from a source index to a destination
   index and the convolution is updated in O(n) using slice
   arithmetic (no FFT per move).  A secondary *jitter* move
   performs a global Gaussian perturbation and recomputes the
   convolution via FFT.  Temperature decays very slowly, enabling a
   thorough exploration of the simplex.

3. **Optional polishing** – a few projected‑gradient steps are run
   (exact sub‑gradient of the max‑convolution) for a short time
   to sharpen the final answer.

The whole routine respects the 1000 s budget, stores the best
sequence in the global variable ``height_sequence_1`` and returns it
as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families (return probability vectors, i.e. sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `alpha`."""
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _make_extended_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Diverse set of normalised start vectors (sum = 1)."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    # geometric families
    for c in np.linspace(0.5, 12.0, 36):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 4.0, 33):
        seeds.append(_power(n, p))

    # combined geometric‑power
    for c in np.linspace(0.5, 12.0, 20):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 7):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 12.0, 20):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classic windows
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 7):
        seeds.append(_tukey_window(n, alpha))

    # Gaussian family
    for sigma_factor in np.linspace(0.06, 0.30, 11):
        seeds.append(_gaussian(n, sigma_factor))

    # Sparse Dirichlet draws – very low concentration → spikes
    for alpha in [0.02, 0.05, 0.1, 0.2]:
        for _ in range(8):
            seeds.append(_random_dirichlet(n, alpha, rng))

    # Moderately sparse Dirichlet
    for alpha in [0.5, 0.8, 1.2]:
        for _ in range(4):
            seeds.append(_random_dirichlet(n, alpha, rng))

    # Re‑use previous construction if supplied
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final clean‑up
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned


# ----------------------------------------------------------------------
# Fast mass‑transfer update (works on the full convolution vector)
# ----------------------------------------------------------------------


def _apply_mass_transfer(a: np.ndarray, conv: np.ndarray,
                         src: int, dst: int, delta: float) -> None:
    """Move `delta` mass from src → dst, updating `a` and its full convolution."""
    n = a.shape[0]
    # slice updates – use original a values (still untouched)
    conv[src:src + n] -= 2.0 * delta * a
    conv[dst:dst + n] += 2.0 * delta * a

    # d*d term
    i2 = src + src
    j2 = dst + dst
    ij = src + dst
    conv[i2] += delta * delta
    conv[j2] += delta * delta
    conv[ij] -= 2.0 * delta * delta

    # finally move the probability mass
    a[src] -= delta
    a[dst] += delta


# ----------------------------------------------------------------------
# Specialized simulated annealing (mass‑transfer + jitter)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with cheap O(n) convolution updates.
    Returns (best_sequence, best_value).
    """
    n = start_seq.size
    a = start_seq.copy()
    # full convolution (full length = 2n‑1)
    conv = _conv_fft(a, a)
    cur_val = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_val = cur_val

    # annealing schedule
    T0 = max(1e-9, cur_val * 0.08)      # initial temperature
    decay = 0.9996                      # very slow cooling
    it = 0
    max_frac_initial = 0.30            # max fraction of source mass moved initially

    while time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)
        # linearly shrink the allowed transfer size with temperature
        max_frac = max_frac_initial * (T / T0)

        # ------------------------------------------------------------------
        # Primary move: mass transfer
        # ------------------------------------------------------------------
        if rng.random() < 0.87:
            # choose source proportional to its mass (avoid zero‑mass sources)
            if a.max() == 0.0:
                break
            src = rng.choice(n, p=a)          # a already sums to 1
            dst = rng.integers(0, n)           # uniform destination
            if src == dst:
                continue
            # draw delta ≤ max_frac * a[src]
            delta = max_frac * a[src] * rng.random()
            if delta <= 1e-15:
                continue

            # apply move
            _apply_mass_transfer(a, conv, src, dst, delta)
            new_val = 2.0 * n * conv.max()

            # Metropolis acceptance
            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_seq = a.copy()
                    print(f"[Anneal] it={it:8d} best={best_val:.6f}")
            else:
                # revert
                _apply_mass_transfer(a, conv, dst, src, delta)

        # ------------------------------------------------------------------
        # Secondary move: global jitter (full recomputation)
        # ------------------------------------------------------------------
        else:
            # jitter magnitude tied to temperature
            sigma = max_frac * 0.07
            cand = a + rng.normal(scale=sigma, size=n)
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand)
            conv_cand = _conv_fft(cand, cand)
            new_val = 2.0 * n * conv_cand.max()
            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                a = cand
                conv = conv_cand
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_seq = a.copy()
                    print(f"[Anneal] it={it:8d} best={best_val:.6f}")

        # occasional stronger exploration
        if it % 3000 == 0:
            sigma = max_frac * 0.17
            cand = best_seq + rng.normal(scale=sigma, size=n)
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand)
            conv_cand = _conv_fft(cand, cand)
            cand_val = 2.0 * n * conv_cand.max()
            if cand_val < best_val:
                best_val = cand_val
                best_seq = cand.copy()
                a = cand
                conv = conv_cand
                cur_val = cand_val
                print(f"[Anneal] (strong jump) it={it:8d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple projected sub‑gradient descent (optional polishing)
# ----------------------------------------------------------------------


def _projected_subgradient(start_seq: np.ndarray,
                           deadline: float,
                           max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient steps on the true max‑convolution objective."""
    a = start_seq.copy()
    n = a.size
    best_seq = a.copy()
    best_val = 2.0 * n * _conv_fft(a, a).max()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_c = conv.max()
        # tolerance to collect the active set of constraints
        tol = max(1e-12, 0.001 * max_c)
        active = np.where(conv >= max_c - tol)[0]

        # sub‑gradient for each active lag
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.12 / math.sqrt(it)
        a_next = a - lr * g
        a_next = _project_simplex(a_next)

        cur_val = 2.0 * n * _conv_fft(a_next, a_next).max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_next.copy()
            print(f"[ProjGD] it={it:7d} best={best_val:.6f}")

        a = a_next

    return best_seq, best_val

def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The best sequence found within the time
    budget is returned as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Length of the sequence – 4000 is a good trade‑off.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Retrieve any previous solution (helps to keep progress across runs)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # --------------------------------------------------------------
    # 0) Inspect previous construction (if any)
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    if previous is not None and previous.shape == (n,):
        val = _objective(previous)
        best_seq = previous.copy()
        best_val = val
        print(f"[prev] objective={val:.6f}")

    # --------------------------------------------------------------
    # 1) Build a broad pool of analytic / random seeds.
    # --------------------------------------------------------------
    seeds = _make_extended_seeds(n, previous)
    # Evaluate all seeds (exact objective)
    seed_evals = [(s, _objective(s)) for s in seeds]
    seed_evals.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [s for s, _ in seed_evals[:top_k]]

    if seed_evals[0][1] < best_val:
        best_val = seed_evals[0][1]
        best_seq = seed_evals[0][0].copy()
        print(f"[seed] new best={best_val:.6f}")

    # --------------------------------------------------------------
    # 2) Fast annealing starting from the best seed.
    #    Allocate the majority of the remaining budget to this phase.
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)

    # time budgeting
    now = time.time()
    time_left = deadline - now
    anneal_time = time_left * 0.80          # 80 % for annealing
    anneal_deadline = time.time() + anneal_time

    rng = np.random.default_rng(seed + 123)   # independent rng for the annealer

    print("[anneal] starting fast annealing")
    seq_ann, val_ann = _anneal_optimize(best_seq, anneal_deadline, rng)
    if val_ann < best_val:
        best_val = val_ann
        best_seq = seq_ann.copy()
        print(f"[anneal] improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Optional short projected sub‑gradient polishing.
    # --------------------------------------------------------------
    now = time.time()
    if now < deadline - 3.0:
        proj_deadline = deadline - 2.0
        print("[projGD] starting brief projected sub‑gradient polishing")
        seq_proj, val_proj = _projected_subgradient(best_seq, proj_deadline,
                                                   max_iters=150_000)
        if val_proj < best_val:
            best_val = val_proj
            best_seq = seq_proj.copy()
            print(f"[projGD] improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4) Fallback – should never happen, but guarantees a return.
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    # store for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
