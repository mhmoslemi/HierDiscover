def _lp_maximize_sum(f, M, target_sum=1.0):
    import scipy.sparse as sp
    from scipy.optimize import linprog

    n = len(f)
    # Build A_ub
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        start = max(0, k - (n-1))
        end = min(n-1, k)
        # compute indices range
        # vect = f[start_rev...]? Actually need coefficient f[k - j] for each j
        for j in range(start, end+1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    b = np.full(2*n - 1, M, dtype=np.float64)

    c = -np.ones(n)  # minimize -sum(g) => maximize sum(g)
    bounds = [(0, None)] * n

    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})  # time limit per solve optional?

    if not result.success:
        return None

    g = result.x

    sum_g = g.sum()
    if sum_g == 0:
        return None
    # normalise to target_sum
    g = g * (target_sum / sum_g)
    return g

"""
Propose a non‑negative step function minimizing evaluate_sequence.

The method consists of three phases:
1.  Initialise a normalized distribution (from a previous run or a simple
    exponential decay).
2.  Repeatedly solve a linear program that maximises the ℓ₁‑norm of a
    partner vector g under the constraint  f * g ≤ M (the current maximal
    auto‑convolution of f).  Mixing f and g shrinks the maximal
    auto‑convolution of the mixture.
3.  Polish the result with the Adam / sub‑gradient / simulated‑annealing
    pipeline already used by the reference solution.

All steps respect the global 1000 s wall‑clock budget and finally return
the best sequence found as a plain list of floats.
"""
import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution with a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact value of evaluate_sequence (ignoring clipping)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_conv = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_conv / (sum_a ** 2))


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Sparse matrix A such that A @ g = f * g (full convolution)."""
    n = len(f)
    rows = 2 * n - 1
    data = []
    offsets = []
    # each diagonal d ≥ 0 contains the constant f[d]
    for d in range(n):
        length = min(rows - d, n)            # number of elements on this diagonal
        data.append(np.full(length, f[d], dtype=np.float64))
        offsets.append(d)
    A = sp.diags(data, offsets, shape=(rows, n), format='csr')
    return A


def _lp_maximize_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ g_j   s.t.   f * g ≤ M  (element‑wise),   g ≥ 0.
    Returns a normalised vector (ℓ₁‑norm = 1) or None on failure.
    """
    n = len(f)
    A = _conv_matrix(f)            # shape (2n‑1, n)
    b = np.full(A.shape[0], M, dtype=np.float64)

    # objective: minimise –Σ g_j  → maximise Σ g_j
    c = -np.ones(n, dtype=np.float64)

    # bounds g_j ≥ 0
    bounds = [(0.0, None)] * n

    # use the high‑performance HiGHS solver
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if not res.success:
        return None
    g = res.x
    s = g.sum()
    if s == 0:
        return None
    return g * (1.0 / s)          # normalise to unit sum


# ----------------------------------------------------------------------
# Optimisation primitives (same as the reference implementation)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection on the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(seq,seq)) (assumes Σseq = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _logspace_adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray | None, float]:
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits
    m = np.zeros_like(v)                        # 1st moment
    v_sq = np.zeros_like(v)                     # 2nd moment
    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8
    beta = 0.5
    beta_inc = 0.1
    beta_max = 200.0
    lr0 = 1.0
    best_seq, best_val = None, float("inf")
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # smoothed max
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]        # central n entries
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        lr = lr0 / np.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)
        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)
    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # sub‑gradient of the active term
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        lr = 0.20 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)
    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.06,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq, best_val = cur.copy(), cur_val
    T = max(init_temp_factor * cur_val, 1e-12)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.9999, min_temp)
        if rng.random() < 0.7:
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps
        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur, cur_val = new, new_val
            if new_val < best_val:
                best_seq, best_val = new.copy(), new_val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Stage 0 – initialise a start vector of moderate length.
    # ------------------------------------------------------------------
    n_small = 2000          # size for the LP stage (fast enough)
    prev = globals().get("height_sequence_1")
    if prev is not None and len(prev) >= n_small:
        # down‑sample the previous best by taking every second entry
        f = np.asarray(prev[: n_small * 2 : 2], dtype=np.float64)
        if f.sum() == 0:
            f = np.full(n_small, 1.0 / n_small, dtype=np.float64)
        else:
            f = f / f.sum()
        print("[Init] Loaded previous solution (down‑sampled).")
    else:
        # simple exponential decay – parameters tuned later
        c = 0.001
        i = np.arange(n_small, dtype=np.float64)
        f = np.exp(-c * i)
        f = f / f.sum()
        print("[Init] Built exponential seed.")

    best_f = f.copy()
    best_val = _objective(best_f)
    print(f"[Init] Objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – linear‑programming shrink‑max‑convolution loop.
    # ------------------------------------------------------------------
    lp_iters = 15
    for it in range(1, lp_iters + 1):
        if time.time() > deadline:
            break
        # current max convolution
        conv_f = _conv_fft(best_f, best_f)
        M = conv_f.max()
        # solve LP
        g = _lp_maximize_sum(best_f, M)
        if g is None:
            print("[LP] Solver failed – exiting LP phase.")
            break
        # mixing weight (decaying)
        t = 0.2 * (0.85 ** (it - 1))
        new_f = (1.0 - t) * best_f + t * g
        new_f = new_f / new_f.sum()

"""
Propose a non‑negative step function that minimises the
``evaluate_sequence`` objective.

The search proceeds in three phases:

1. **Initialisation** – a short (≈2000‑element) distribution is built
   either from the previously best sequence (down‑sampled) or from a
   simple exponential decay.  The vector is ℓ₁‑normalised.

2. **Linear‑programming contraction** – repeated LP solves maximise the
   ℓ₁‑norm of a partner vector *g* under the linear constraint
   ``f * g ≤ M`` where ``M = max(f * f)``.  Mixing the current vector
   ``f`` with the optimiser ``g`` (convex combination) provably reduces
   the maximal auto‑convolution of the mixture.  Fifteen such
   iterations are performed; the LP uses a sparse Toeplitz matrix
   (`scipy.sparse`) and the high‑performance HiGHS solver.

3. **Gradient polishing** – the refined result is up‑sampled to the
   reference length (4000) and fed into the log‑space Adam,
   exact sub‑gradient descent and simulated‑annealing pipeline that
   was used in the previous state‑of‑the‑art code.  The three stages
   respect the global 1000 s wall‑clock budget.

The function returns the best sequence found as a plain list of floats
(and stores it in the global name ``height_sequence_1`` for possible
reuse).  All intermediate steps print progress information.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` (ignores clipping)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_conv = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_conv / (sum_a ** 2))


# ----------------------------------------------------------------------
# Linear‑programming step (maximise Σg subject to f * g ≤ M)
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Sparse matrix A with A @ g = f * g (full convolution)."""
    n = len(f)
    rows = 2 * n - 1
    data = []
    offsets = []
    for d in range(n):                         # diagonal offset d ≥ 0
        length = min(rows - d, n)               # entries on this diagonal
        data.append(np.full(length, f[d], dtype=np.float64))
        offsets.append(d)
    A = sp.diags(data, offsets, shape=(rows, n), format="csr")
    return A


def _lp_maximize_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ g_j   s.t.   f * g ≤ M   (element‑wise),   g ≥ 0.

    Returns a unit‑ℓ₁‑norm vector (or ``None`` on failure).
    """
    n = len(f)
    A = _conv_matrix(f)                # (2n‑1) × n
    b = np.full(A.shape[0], M, dtype=np.float64)

    # maximise Σ g → minimise −Σ g
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if not res.success:
        return None
    g = res.x
    s = g.sum()
    if s == 0.0:
        return None
    return g * (1.0 / s)               # normalise to Σg = 1


# ----------------------------------------------------------------------
# Gradient‑based refinement (copied from the reference implementation)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(seq,seq)) – assumes Σseq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _logspace_adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray | None, float]:
    eps = 1e-30
    v = np.log(start_vec + eps)               # logits
    m = np.zeros_like(v)
    v_sq = np.zeros_like(v)
    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8
    beta = 0.5
    beta_inc = 0.1
    beta_max = 200.0
    lr0 = 1.0
    best_seq, best_val = None, float("inf")
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        lr = lr0 / np.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)
        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)
    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        lr = 0.20 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)
    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.06,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq, best_val = cur.copy(), cur_val
    T = max(init_temp_factor * cur_val, 1e-12)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.9999, min_temp)
        if rng.random() < 0.7:
            k = rng.integers(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps
        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur, cur_val = new, new_val
            if new_val < best_val:
                best_seq, best_val = new.copy(), new_val
    return best_seq, best_val
