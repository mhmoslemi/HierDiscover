"""
Optimiser for the step‑function evaluation metric.

The algorithm works in three stages:

1. **Diverse seeds** – in addition to the usual uniform / random /
   triangular / exponential seeds we also build a large *Sidon* set.
   A Sidon set has the property that all pairwise sums i+j are distinct,
   therefore each convolution entry receives at most one product.
   For a set of size `m` with equal weights the objective equals
   ``2·n / m²``; a larger Sidon set immediately yields a very good bound.

2. **Adam on the simplex** – starting from the best seed we run a
   soft‑max (exponentiated‑gradient) Adam optimiser.  The soft‑max
   temperature λ is increased slowly; when it exceeds a configurable cap
   we switch to the exact (hard) max, i.e. we use a genuine sub‑gradient.
   A few short random restarts are performed if progress stalls.

3. **Fine‑grained annealing** – a cheap Metropolis–Hastings walk that
   swaps tiny amounts of mass between random coordinates and occasionally
   performs a global Gaussian jump.  It finishes the optimisation with a
   final polishing pass.

All phases respect the global time budget (default 1000 s).  The function
returns the best sequence found as a plain list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact evaluation used for reporting (same formula as evaluate_sequence)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple seed families (return a probability vector that sums to 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    exps = np.exp(logits - m)
    return exps / exps.sum()


def _exponential_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)  (then normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    """Increasing then decreasing linear shape."""
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    dec = inc[::-1] if n % 2 == 0 else np.arange(half, 0, -1, dtype=np.float64)
    a = np.concatenate([inc, dec])
    return a / a.sum()


# ----------------------------------------------------------------------
# Greedy construction of a large Sidon‑like set (all i+j distinct)
# ----------------------------------------------------------------------
def _greedy_sidon_set(n: int, attempts: int, rng: np.random.Generator):
    """
    Returns a binary mask (0/1) of length n whose support is a Sidon set.
    The function tries `attempts` random orderings and keeps the largest set.
    """
    best_mask = np.zeros(n, dtype=np.float64)
    best_len = 0
    max_sum = 2 * n - 1                       # possible sums 0 … 2n‑2
    for _ in range(attempts):
        order = rng.permutation(n)
        used = np.zeros(max_sum, dtype=np.bool_)
        mask = np.zeros(n, dtype=np.float64)
        cur_len = 0
        for idx in order:
            conflict = False
            # check sums with already chosen indices
            existing = np.flatnonzero(mask)
            for j in existing:
                if used[idx + j]:
                    conflict = True
                    break
            if conflict:
                continue
            # self‑sum
            if used[idx + idx]:
                continue
            # accept idx
            mask[idx] = 1.0
            cur_len += 1
            # mark all new sums
            for j in existing:
                used[idx + j] = True
                used[j + idx] = True          # redundant but keeps logic clear
            used[idx + idx] = True
        if cur_len > best_len:
            best_len = cur_len
            best_mask = mask.copy()
    # normalise to a probability vector
    if best_len > 0:
        best_mask /= best_len
    return best_mask, best_len


# ----------------------------------------------------------------------
# Soft‑max weights over the convolution entries
# ----------------------------------------------------------------------
def _soft_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Return a probability vector w (sums to 1) that is a soft‑max over the
    entries of `conv`.  The argument `lam` is an inverse temperature.
    For lam ≤ 0 a hard max (uniform weight on all ties) is returned.
    """
    max_c = conv.max()
    if lam <= 0.0:
        active = np.isclose(conv, max_c, atol=1e-12)
        cnt = np.count_nonzero(active)
        if cnt == 0:
            return np.ones_like(conv) / conv.size
        return np.where(active, 1.0 / cnt, 0.0)
    rel = (conv - max_c) / max_c          # ≤ 0
    exps = np.exp(lam * rel)
    w = exps / exps.sum()
    return w


def _gradient_wrt_a(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the (soft) max‑convolution with respect to the probability
    vector `a`.  Implements   g = 2 * rev(a) ⊛ w,
    where w are the soft‑max weights over the convolution entries.
    """
    conv = _conv_fft(a, a)
    w = _soft_max_weights(conv, lam)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)               # length = 3n‑2
    n = a.size
    grad = 2.0 * full[n - 1: n - 1 + n]       # slice that matches a
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 400_000,
                   lam_start: float = 0.5,
                   lam_factor: float = 1.001,
                   lam_cap: float = 200.0,
                   restart_interval: int = 60_000) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑probabilities (soft‑max) of `start`.  The smoothing
    parameter λ grows multiplicatively (lam_factor) until `lam_cap` is
    reached; after that we use the exact (hard) max (lam ≤ 0).  Every
    `restart_interval` iterations a tiny Gaussian perturbation is added
    to the logits if no improvement has been seen for a while.
    """
    eps = 1e-30
    logp = np.log(start + eps)
    n = start.size

    m = np.zeros_like(logp)      # first moment
    v = np.zeros_like(logp)      # second moment

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = start.copy()
    best_val = _objective(start)

    t = 0
    lam = lam_start
    last_best = best_val
    stagnant = 0

    while t < max_iters and time.time() < deadline:
        t += 1

        # current probability vector (stable soft‑max)
        logp_shift = logp - np.max(logp)
        a = np.exp(logp_shift)
        a /= a.sum()

        # gradient of (soft) max‑convolution
        g_a = _gradient_wrt_a(a, lam)

        # chain rule to obtain gradient w.r.t. log‑probabilities
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log * g_log)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.13 / (math.sqrt(t) + 1.0)
        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # true objective (hard max)
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            stagnant = 0
            print(f"[Adam] it={t:7d}  best={cur_val:.6f}  λ={lam:.2f}")
        else:
            stagnant += 1

        # increase λ (but do not exceed cap)
        if lam < lam_cap:
            lam *= lam_factor
        else:
            lam = -1.0                     # force hard‑max from now on

        # occasional random kick‑restart if we have been stagnant
        if t % restart_interval == 0 and stagnant > restart_interval // 2:
            noise = np.random.default_rng().normal(scale=0.02, size=n)
            logp += noise
            stagnant = 0

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light simulated‑annealing neighbour search (final polishing)
# ----------------------------------------------------------------------
def _anneal_refine(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Tiny Metropolis–Hastings walk that swaps a small fraction of mass."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9996
    it = 0
    n = cur.size

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d}  best={new_val:.6f}")

        # every ~2500 steps try a global Gaussian perturbation
        if it % 2500 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot > 0.0:
                new_seq /= tot
            else:
                new_seq = _uniform(n)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The function respects the time budget (default 1000 s) and returns the best
    vector found as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    n = 4000
    print(f"\n=== Optimising step function (length {n}) ===")

    # ------------------------------------------------------------------
    # 1.  Build a collection of seeds (including a large Sidon‑like set)
    # ------------------------------------------------------------------
    seeds = []

    # use previously stored result if it matches the current length
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        prev_arr = np.asarray(previous, dtype=np.float64)
        if prev_arr.shape == (n,):
            prev_arr = np.maximum(prev_arr, 0.0)
            if prev_arr.sum() > 0:
                seeds.append(prev_arr / prev_arr.sum())

    # deterministic cheap seeds
    seeds.extend([
        _uniform(n),
        _random_softmax(n, rng),
        _exponential_decay(n, rate=0.001),
        _triangular(n),
    ])

    # a greedy Sidon‑like construction – try several attempts
    sidon_mask, sidon_len = _greedy_sidon_set(n, attempts=250, rng=rng)
    if sidon_len >= 2:                     # need at least two points to be meaningful
        seeds.append(sidon_mask)

    # pick the best initial seed according to the true objective
    best_seed = None
    best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seed = s
    print(f"[Init] best initial objective = {best_val:.6f}  (support size {np.count_nonzero(best_seed>0):d})")

    # ------------------------------------------------------------------
    # 2.  Phase‑1 – Adam optimisation (several short restarts)
    # ------------------------------------------------------------------
    phase1_deadline = deadline - 30.0            # keep ~30 s for polishing
    if time.time() < phase1_deadline:
        remaining = phase1_deadline - time.time()
        # number of independent Adam runs we can afford (roughly 5‑6)
        runs = max(1, int(remaining // 120))      # ~2 min per run
        print(f"[Phase 1] planning {runs} Adam runs")
        current_best_seq = best_seed
        current_best_val = best_val

        for r in range(runs):
            run_deadline = time.time() + remaining / runs
            # slight random perturbation of the start for each restart
            start_vec = current_best_seq.copy()
            if r > 0:                           # after the first run, jitter the start
                eps = rng.normal(scale=0.02, size=n)
                start_vec = np.clip(start_vec + eps, 0.0, None)
                start_vec /= start_vec.sum()
            seq, val = _adam_optimize(start_vec, run_deadline,
                                      max_iters=400_000,
                                      lam_start=0.5,
                                      lam_factor=1.001,
                                      lam_cap=180.0,
                                      restart_interval=70_000)
            if val < current_best_val:
                current_best_val = val
                current_best_seq = seq.copy()
                print(f"[Phase 1] run {r+1}/{runs} – new global best = {val:.6f}")
            remaining = deadline - 30.0 - time.time()
            if remaining <= 5.0:
                break
        seq_adam, val_adam = current_best_seq, current_best_val
        print(f"[Phase 1] best after Adam = {val_adam:.6f}")
    else:
        seq_adam, val_adam = best_seed, best_val
        print("[Phase 1] skipped (insufficient time)")

    # ------------------------------------------------------------------
    # 3.  Phase‑2 – Light simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("[Phase 2] Simulated‑annealing polishing")
        seq_final, val_final = _anneal_refine(seq_adam, deadline)
        if val_final < val_adam:
            seq_adam, val_adam = seq_final, val_final
            print(f"[Phase 2] improved to {val_final:.6f}")
        else:
            print("[Phase 2] no improvement")
    else:
        print("[Phase 2] no time left")

    # ------------------------------------------------------------------
    # 4.  Normalise and expose the result
    # ------------------------------------------------------------------
    final_seq = np.maximum(seq_adam, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
        total = final_seq.sum()
    final_seq /= total

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
