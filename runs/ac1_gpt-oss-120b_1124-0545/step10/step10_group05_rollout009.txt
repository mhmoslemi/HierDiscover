"""
Propose a non‑negative step‑function that minimizes ``evaluate_sequence``.

Strategy
--------
1. *Broad sampling* – evaluate many analytic families (uniform, Beta,
   Gaussian, Hann, etc.) for several lengths (2000‑10000).  Keep the best
   candidate.

2. *Heavy random sampling* – draw Dirichlet vectors with a large range
   of concentration parameters and mixtures of Beta pdfs.  Again keep the
   global best.

3. *Smoothed Adam* – start from the best few candidates and minimise a
   smooth‐max approximation of ``max(conv(a,a))``.  The temperature
   (β) is exponentially increased up to 5000, the number of iterations
   is limited only by the remaining time budget.

4. *Aggressive KL‑mirror descent* – after Adam we run a pure
   sub‑gradient step for the true max (no log‑sum‑exp smoothing).  The
   step size follows a 1/√t schedule and the active set tolerance
   decays exponentially from 1 % to 10⁻⁹.

5. Return the best sequence seen (already normalised to sum = 1).  The
   result is stored in the global name ``height_sequence_1`` for possible
   reuse in later runs.
"""

import time
import numpy as np
from math import sqrt, pi, log, exp
from typing import List

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence for a *normalised* sequence (sum == 1).
    Returns   2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    return coeffs / total if total != 0 else _uniform(n)


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = a0 - a1 * np.cos(2 * pi * i / (n - 1)) + a2 * np.cos(4 * pi * i / (n - 1))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = w.sum()
    return w / total if total != 0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    total = w.sum()
    return w / total if total != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    total = w.sum()
    return w / total if total != 0 else _uniform(n)


def _sin_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(pi * i / n)
    total = w.sum()
    return w / total if total != 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random generation utilities
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet‑like vector with concentration α."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s != 0 else _uniform(n)


def _random_beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.uniform(0.1, 5.0)
    b = rng.uniform(0.1, 5.0)
    return _beta_shape(n, a, b)


def _random_mixture_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two independent Beta pdfs."""
    w = rng.random()
    a1, b1 = rng.uniform(0.1, 5.0), rng.uniform(0.1, 5.0)
    a2, b2 = rng.uniform(0.1, 5.0), rng.uniform(0.1, 5.0)
    s1 = _beta_shape(n, a1, b1)
    s2 = _beta_shape(n, a2, b2)
    mix = w * s1 + (1.0 - w) * s2
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(start_seq: np.ndarray,
                          deadline: float,
                          beta_start: float = 5.0,
                          beta_end: float = 5000.0,
                          max_iter: int = 10_000_000,
                          lr0: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth (log‑sum‑exp) approximation of ``max(conv(a,a))``.
    Adam is applied to the log‑softmax (logits) parametrisation.
    """
    n = len(start_seq)
    eps = 1e-30

    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        a = np.exp(logits)
        a /= a.sum()

        conv = _conv_fft(a, a)

        # linearly increase the temperature
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        # log‑sum‑exp smooth‑max
        c = np.max(conv)
        ex = np.exp(beta * (conv - c))
        p = ex / ex.sum()

        # gradient of the smooth max w.r.t. a
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # periodic exact evaluation
        if (it & 1023) == 0:          # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

            # early stop if we already passed a very low threshold
            if best_val < 1.45:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Aggressive KL‑mirror descent (true max, no smoothing)
# ----------------------------------------------------------------------
def _expgrad_max_descent(start_seq: np.ndarray,
                         deadline: float,
                         eps_rel_start: float = 0.01,
                         eps_rel_end: float = 1e-9,
                         lr_factor: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the *exact* max of the convolution.
    The active set tolerance shrinks exponentially, allowing the algorithm
    to start with a coarse approximation and finish with a precise fix.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    start_time = time.time()
    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        # exponential decay of relative tolerance
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        eps_rel = eps_rel_start * (eps_rel_end / eps_rel_start) ** frac
        delta = eps_rel * max_val

        active = (conv >= max_val - delta).astype(np.float64)

        # sub‑gradient of the (weighted) max = 2 * (active ⋆ a_rev)
        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        # KL‑mirror step (exponentiated gradient)
        lr = lr_factor / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise numerics
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[ExpGrad] it={it:7d} n={n:5d} best={val:.6f}")

            # stop a little before we run out of time
            if time.time() > deadline - 0.5:
                break

    return best_seq, best_val

except Exception:               # pragma: no cover
    minimize = None


def _lbfgs_refine(start_seq: np.ndarray, deadline: float, beta: float = 5000.0) -> tuple[np.ndarray, float]:
    """L‑BFGS‑B on the log‑sum‑exp smooth max (if SciPy is available)."""
    if minimize is None:
        return start_seq, _objective(start_seq)

    n = len(start_seq)
    eps = 1e-30

    def fun_and_grad(logits: np.ndarray):
        a = np.exp(logits)
        a /= a.sum()
        conv = _conv_fft(a, a)
        c = np.max(conv)
        exps = np.exp(beta * (conv - c))
        lse = c + (1.0 / beta) * np.log(exps.sum())
        val = 2.0 * n * lse

        p = exps / exps.sum()
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)
        return val, grad_logits.astype(np.float64)

    x0 = np.log(start_seq + eps)

    class TimeOut(RuntimeError):
        pass

    def callback(xk):
        if time.time() >= deadline:
            raise TimeOut()

    try:
        res = minimize(
            fun=lambda x: fun_and_grad(x)[0],
            x0=x0,
            jac=lambda x: fun_and_grad(x)[1],
            method='L-BFGS-B',
            options={'maxiter': 200, 'disp': False},
            callback=callback,
        )
    except TimeOut:
        best_logits = x0
    else:
        best_logits = res.x

    best_a = np.exp(best_logits)
    best_a /= best_a.sum()
    return best_a, _objective(best_a)
