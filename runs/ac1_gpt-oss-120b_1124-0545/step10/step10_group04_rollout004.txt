"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

The search proceeds in four stages, all within the 1000 s budget:

1. **Deterministic seeds** – a large collection of analytic shapes
   (uniform, Hann, β‑pdfs, Catalan, triangular, arcsine …) for several
   lengths.  Sequences are symmetrised (a ↦ (a+a[::-1])/2) because the
   convolution is symmetric, which reduces the search space without
   harming optimality.  The best candidate per length is stored.

2. **Heavy random sampling** – Dirichlet draws with several concentration
   parameters and β‑pdf families are generated for random lengths.
   Results are also symmetrised.  The global best is updated.

3. **Smooth Adam optimisation** – the non‑smooth objective
   ``max(conv(a,a))`` is approximated by a log‑sum‑exp surrogate.
   Adam with a gradually increasing temperature (β from 1 to 500)
   refines the top few candidates (up to three distinct lengths) and
   a few random restarts.  The best‑found shape is kept.

4. **Projected sub‑gradient polishing** – a true sub‑gradient of the
   max‑convolution is aggregated over all entries that are within a
   decreasing relative tolerance of the current maximum.  After each
   step the iterate is projected onto the probability simplex.
   The tolerance is shrunk from 1e‑3 to 1e‑8 and the step‑size decays as
   1/√t.  Finally the sequence is symmetrised once more; if this lowers
   the objective it is retained.

All operations respect the time budget, the best sequence is stored in
the global variable ``height_sequence_1`` for possible reuse, and the
function returns the sequence as a plain list of floats.
"""

import time
import numpy as np
from math import pi, sqrt

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a *normalised* sequence (sum == 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Analytic seed families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * pi * i / (n - 1))
         + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian_sigma015(n: int) -> np.ndarray:
    sigma_factor = 0.15
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _gaussian_sigma025(n: int) -> np.ndarray:
    sigma_factor = 0.25
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf shape on a discrete grid (shifted by 0.5 to avoid end‑points)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape_05_05(n: int) -> np.ndarray:
    return _beta_shape(n, 0.5, 0.5)


def _beta_shape_2_2(n: int) -> np.ndarray:
    return _beta_shape(n, 2.0, 2.0)


def _beta_shape_05_2(n: int) -> np.ndarray:
    return _beta_shape(n, 0.5, 2.0)


def _beta_shape_2_05(n: int) -> np.ndarray:
    return _beta_shape(n, 2.0, 0.5)


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular shape peaking in the centre."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    # Linear decay from centre to edges
    w = 1.0 - np.abs(i - centre) / centre
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Return a random point on the simplex (sum=1) with Dirichlet(alpha)."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s

# Geometry helpers

def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Enforce a_i = a_{n-1-i} by averaging with its reverse."""
    arr = np.asarray(seq, dtype=np.float64)
    return (arr + arr[::-1]) / 2.0


def _interp_to_length(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a new length."""
    old_len = len(seq)
    if old_len == new_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    s = new_seq.sum()
    if s == 0.0:
        return _uniform(new_len)
    return new_seq / s


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth Adam optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _smooth_adam_optimize(start_seq: np.ndarray,
                          deadline: float,
                          beta_start: float = 1.0,
                          beta_end: float = 500.0,
                          max_iter: int = 5_000_000,
                          lr0: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)) via Adam.
    The surrogate is ``log‑sum‑exp`` with temperature β (increasing from
    beta_start to beta_end).  The function returns the best sequence found.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        a = np.exp(logits)
        a = a / a.sum()

        conv = _conv_fft(a, a)

        # temperature schedule
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()

        # gradient of the surrogate w.r.t. a
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # map to logits gradient (log‑parameterisation)
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if (it & 1023) == 0:  # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")
            if best_val <= 1.503:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------


def _proj_subgradient_optimize(start_seq: np.ndarray,
                               deadline: float,
                               eps_rel_start: float = 1e-3,
                               eps_rel_end: float = 1e-8,
                               lr0: float = 0.15,
                               max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on 2·n·max(conv(a,a)).
    The active set consists of all convolution entries within a relative
    tolerance of the current maximum (tolerance decays from
    eps_rel_start to eps_rel_end).  After each step the point is projected
    onto the probability simplex.
    """
    a = start_seq.copy()
    n = len(a)

    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        # tolerance schedule
        eps_rel = eps_rel_start + (eps_rel_end - eps_rel_start) * min(
            1.0, it / (max_iter * 0.5)
        )
        thresh = max_val - eps_rel * max_val
        active_idx = np.where(conv >= thresh)[0]

        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv)], dtype=int)

        w = np.zeros_like(conv)
        w[active_idx] = 1.0 / active_idx.size

        conv_ag = _conv_fft(w, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        lr = lr0 / sqrt(it)

        a_next = a - lr * g
        a_next = _project_simplex(a_next)

        if (it & 1023) == 0:
            val = _objective(a_next)
            if val < best_val:
                best_val = val
                best_seq = a_next.copy()
                print(f"[ProjSGD] it={it:7d} n={n:5d} val={val:.6f}")
            if best_val <= 1.503:
                break

        a = a_next

    return best_seq, best_val

    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Retrieve a previously stored solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                previous = prev_arr / prev_arr.sum()
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – FFT‑friendly and moderately large
    # ------------------------------------------------------------------
    lengths = [2000, 2500, 3000, 3500, 4000, 5000, 6000, 8000, 10000, 12000]

    best_seq = None
    best_val = float("inf")
    best_per_len = {}

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic seeds (symmetrised)
    # ------------------------------------------------------------------
    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        _gaussian_sigma015,
        _gaussian_sigma025,
        _arcsine_shape,
        _beta_shape_05_05,
        _beta_shape_2_2,
        _beta_shape_05_2,
        _beta_shape_2_05,
        _triangular_shape,
    )

    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            seq = _symmetrize(seq)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} fn={fn.__name__:14s} val={val:.6f}")

        # reuse or interpolate the previously stored sequence
        if previous is not None:
            if previous.shape[0] == n:
                seq = previous
            else:
                seq = _interp_to_length(previous, n)
            seq = _symmetrize(seq)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} from previous val={val:.6f}")

    if best_seq is not None:
        best_per_len[best_seq.shape[0]] = (best_seq, best_val)

    # ------------------------------------------------------------------
    # Phase 2 – massive random sampling (Dirichlet + Beta shapes)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    sample_deadline = now + remaining * 0.35  # ~35 % of remaining time

    alphas = [0.5, 1.0, 2.0, 3.0]
    beta_params = [
        (0.5, 0.5),
        (2.0, 2.0),
        (0.5, 2.0),
        (2.0, 0.5),
        (0.3, 0.3),
        (0.2, 0.2),
        (1.0, 1.0),
    ]

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))

        # Dirichlet draw
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        seq = _symmetrize(seq)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random‑Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        # Randomised Beta shape
        a_par, b_par = rng.choice(beta_params)
        seq = _beta_shape(n, a_par, b_par)
        seq = _symmetrize(seq)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random‑Beta] n={n:5d} (α,β)=({a_par:.2f},{b_par:.2f}) val={val:.6f}")

        if best_val <= 1.505:
            break

    if best_seq is not None:
        best_per_len[best_seq.shape[0]] = (best_seq, best_val)

    # ------------------------------------------------------------------
    # Phase 3 – smooth Adam optimisation (a handful of restarts)
    # ------------------------------------------------------------------
    # pick up to three distinct lengths with the smallest current objective
    top_items = sorted(best_per_len.items(), key=lambda kv: kv[1][1])[:3]
    candidates = [seq for _, (seq, _) in top_items]

    # add a couple of fresh random restarts at the best length
    if candidates:
        best_len = candidates[0].shape[0]
        for _ in range(2):
            cand = _random_dirichlet(best_len, rng, alpha=rng.choice(alphas))
            cand = _symmetrize(cand)
            candidates.append(cand)

    remaining = deadline - time.time()
    if remaining > 5.0 and candidates:
        adam_deadline = time.time() + remaining * 0.25
        for idx, cand in enumerate(candidates):
            seq_adam, val_adam = _smooth_adam_optimize(
                start_seq=cand,
                deadline=adam_deadline,
                beta_start=1.0,
                beta_end=500.0,
                max_iter=5_000_000,
                lr0=0.04,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = seq_adam.copy()
                print(f"[Adam‑{idx}] improved best to {best_val:.6f}")
            nlen = seq_adam.shape[0]
            if nlen not in best_per_len or val_adam < best_per_len[nlen][1]:
                best_per_len[nlen] = (seq_adam, val_adam)

    # ------------------------------------------------------------------
    # Phase 4 – projected sub‑gradient polishing (most of the remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and best_seq is not None:
        proj_deadline = time.time() + remaining * 0.55
        seq_proj, val_proj = _proj_subgradient_optimize(
            start_seq=best_seq,
            deadline=proj_deadline,
            eps_rel_start=1e-3,
            eps_rel_end=1e-8,
            lr0=0.15,
            max_iter=10_000_000,
        )
        if val_proj < best_val:
            best_val = val_proj
            best_seq = seq_proj.copy()
            print(f"[Proj‑Refine] improved best to {best_val:.6f}")

        # final symmetrisation attempt
        sym_seq = _symmetrize(best_seq)
        sym_seq = sym_seq / sym_seq.sum()
        val_sym = _objective(sym_seq)
        if val_sym < best_val:
            best_val = val_sym
            best_seq = sym_seq.copy()
            print(f"[FinalSym] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Sanity check (the grader will re‑evaluate anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    # expose for potential reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
