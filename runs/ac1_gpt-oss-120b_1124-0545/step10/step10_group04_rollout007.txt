def _subgradient_optimize(seq, deadline, best_val_external=None, max_iters=10_000_000):
    n = len(seq)
    cur_seq = seq.copy()
    best_seq = cur_seq.copy()
    best_val = _objective(cur_seq)
    if best_val_external is not None and best_val_external < best_val:
        best_val = best_val_external
        best_seq = seq.copy()

    it = 0
    no_improve = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute conv and gradient
        conv = _conv_fft(cur_seq, cur_seq)
        max_val = conv.max()
        # active set tolerance
        eps_rel = 1e-9
        active = np.where(conv >= max_val - eps_rel*max_val)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)], dtype=int)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size
        grad_conv = _conv_fft(w, cur_seq[::-1])
        grad = 2.0 * grad_conv[n-1:n-1+n]

        cur_obj = float(2 * n * max_val)   # our objective formula; note that evaluate_sequence multiplies 2*n*maxconv/(sum^2) and sum=1.
        # compute step size
        if best_val < cur_obj:
            # use Polyak step to aim for best_val
            step = (cur_obj - best_val) / (np.linalg.norm(grad)**2 + 1e-12)
        else:
            step = 0.0

        # fallback if step too small or zero
        if step <= 0.0 or np.isnan(step):
            step = 1e-4 / (np.sqrt(it) + 1.0)

        # cap step
        max_step = 0.5
        if step > max_step:
            step = max_step

        # try step with backtracking (max 5 reductions)
        for _ in range(5):
            new_seq = cur_seq - step * grad
            # project onto simplex
            new_seq = _project_simplex(new_seq)
            # compute new objective
            new_conv = _conv_fft(new_seq, new_seq)
            new_max = new_conv.max()
            new_obj = float(2 * n * new_max)
            if new_obj <= cur_obj:
                cur_seq = new_seq
                cur_obj = new_obj
                break
            else:
                step *= 0.5

        # update best
        if cur_obj < best_val - 1e-12:
            best_val = cur_obj
            best_seq = cur_seq.copy()
            no_improve = 0
            print(f"[PolyakSGD] it={it} n={n} val={best_val:.6f}")
        else:
            no_improve += 1
        # if stuck a while, random perturbation
        if no_improve >= 30000:
            # small random jitter
            perturb = _random_dirichlet(n, np.random.default_rng(), alpha=0.1)
            # mix with current seq
            cur_seq = 0.99 * cur_seq + 0.01 * perturb
            cur_seq = _project_simplex(cur_seq)
            no_improve = 0
        # optionally check if time near deadline
    return best_seq, best_val

"""
Propose a non‑negative step‑function (a probability vector) that minimises
``evaluate_sequence``.  

The method consists of three stages:

1. **Quick deterministic seeds** – a few classic window shapes (uniform,
   Hann, Blackman, Beta‑pdfs, etc.) are normalised and the best among them is
   kept as a start point.

2. **Optional random Dirichlet perturbations** – a short burst of random
   Dirichlet draws (different concentration parameters) is performed to see
   whether a better start can be found.

3. **Polyak‑style projected sub‑gradient optimisation** – the exact non‑smooth
   objective `2·n·max(conv(a,a))` (with `sum(a)=1`) is minimised on the simplex.
   At each iteration the gradient of the active maximal convolution terms is
   computed, a Polyak step length
   `step = (f(x)‑f_best)/‖g‖²` is tried, and the new point is projected back
   onto the probability simplex.  A simple back‑tracking line‑search guarantees
   monotone descent.  When no improvement is observed for many iterations a
   tiny random jitter is added to escape stagnation.

All heavy work is done with FFT‑based convolutions; the algorithm respects the
user supplied time budget (default 1000 s) and returns the best sequence
encountered.
"""

import time
import numpy as np

# ---------------------------------------------------------------------------
# Low‑level utilities
# ---------------------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a normalised sequence (sum == 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Simple analytic families (all returned normalised to sum = 1)
# ---------------------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    return w / w.sum()


def _gaussian_sigma015(n: int) -> np.ndarray:
    sigma_factor = 0.15
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = w.sum()
    return w / s if s != 0.0 else _uniform(n)


def _gaussian_sigma025(n: int) -> np.ndarray:

    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = w.sum()
    return w / s if s != 0.0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    s = w.sum()
    return w / s if s != 0.0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf sampled on a discrete grid (shifted by 0.5 to avoid endpoints)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    s = w.sum()
    return w / s if s != 0.0 else _uniform(n)

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a point on the simplex (sum = 1) with Dirichlet(alpha)."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s != 0.0 else _uniform(n)


# ---------------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ---------------------------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Polyak projected sub‑gradient optimiser (exact max‑convolution)
# ---------------------------------------------------------------------------

def _polyak_subgradient_optimize(start_seq: np.ndarray,
                                 deadline: float,
                                 max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with Polyak step size on the exact objective.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    no_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iters:
        it += 1

        # convolution and active set
        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        eps_rel = 1e-12
        active = np.where(conv >= max_val - eps_rel * max_val)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)], dtype=int)

        # weight vector for the active indices (uniform)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient of the exact max‑convolution term
        grad = 2.0 * _conv_fft(w, a[::-1])[n - 1:n - 1 + n]

        cur_obj = 2.0 * n * max_val

        # Polyak step length (if we have a strictly better reference)
        if best_val < cur_obj:
            step = (cur_obj - best_val) / (np.linalg.norm(grad) ** 2 + 1e-12)
        else:
            step = 0.0

        # fall back to a diminishing step if needed
        if step <= 0.0 or np.isnan(step):
            step = 0.1 / (np.sqrt(it) + 1.0)

        # clamp step size to a safe interval
        if step > 0.5:
            step = 0.5

        # simple back‑tracking line search (max 5 attempts)
        improved = False
        for _ in range(5):
            new_a = a - step * grad
            new_a = _project_simplex(new_a)
            new_obj = _objective(new_a)
            if new_obj <= cur_obj:
                a = new_a
                cur_obj = new_obj
                improved = True
                break
            step *= 0.5

        # keep the best ever seen
        if cur_obj < best_val - 1e-12:
            best_val = cur_obj
            best_seq = a.copy()
            no_improve = 0
