"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

The search consists of five stages, each respecting the global time
budget (≈ 1000 s):

1. **Analytic seed families** – a collection of deterministic shapes
   (uniform, Catalan, Hann, Hamming, Blackman, Gaussian,
   arcsine, half‑sine and two families of parametrised shapes:
   Beta‑pdf and a simple power‑law).  All families are normalised to
   sum = 1 and evaluated for several FFT‑friendly lengths.

2. **Heavy random sampling** – draw many Dirichlet vectors with a
   broad range of concentration parameters and many random Beta‑pdf
   and power‑law shapes.  The best sample is kept.

3. **Smoothed Adam optimisation** – starting from the best candidates
   we run Adam on a smooth log‑sum‑exp approximation of the max‑convolution
   (temperature β linearly annealed from 5 to 2000).  The optimiser works
   on logits (log‑softmax parametrisation) and uses a decreasing learning
   rate  η≈0.02/√t.  The number of iterations is limited only by the
   remaining time.

4. **Aggressive mirror‑descent polishing** – KL‑mirror descent on the
   exact max–convolution (no smoothing).  The active set contains *all*
   convolution entries within a relative tolerance that decays
   exponentially from 1 % to 10⁻⁶.  The step size is 1/√t.

5. **Local stochastic refinement** – “hill‑climbing’’ on the logits:
   repeatedly add Gaussian noise with a temperature that decays from
   ``sigma_start`` to ``sigma_end`` and keep the perturbation if it
   improves the exact objective.

All stages keep track of the best sequence found so far.  The final
sequence is returned as a plain Python list; its sum is ≈ 1 (the exact
scaling does not matter for the objective because the evaluator
cancels it out).  The function prints progress information and is fully
self‑contained – it does not rely on any external files or network
access.
"""

import time
import numpy as np
from math import sqrt, pi, log, exp
from typing import List

# optional SciPy
try:
    from scipy.optimize import minimize
except Exception:  # pragma: no cover
    minimize = None


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (efficient for n up to ~20000).
    Returns the full convolution of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for a *normalised* sequence (sum == 1):
        2 * n * max(conv(seq, seq))
    where conv is the linear self‑convolution.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * float(np.max(conv))


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  Implements the O(n log n) algorithm.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal, return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = a0 - a1 * np.cos(2 * pi * i / (n - 1)) + a2 * np.cos(4 * pi * i / (n - 1))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _sin_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(pi * i / n)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _power_shape(n: int, p: float) -> np.ndarray:
    """
    Weight ∝ ((i+0.5)*(n-i-0.5))**p   (symmetric, interior‑peaked for p>0,
    edge‑peaked for p<0).  Returned vector is normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    base = i * (n - i)
    # protect against zero for extreme p (though base>0 for all i)
    w = np.power(base, p)
    if w.sum() == 0.0 or np.any(np.isnan(w)) or np.any(np.isinf(w)):
        return _uniform(n)
    return w / w.sum()

# Random generation utilities

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


def _random_beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.uniform(0.05, 10.0)  # extended range
    b = rng.uniform(0.05, 10.0)
    return _beta_shape(n, a, b)


def _random_power_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(-4.0, 4.0)
    return _power_shape(n, p)


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax parameterisation)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    max_iter: int = 5_000_000,
    lr0: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth (log‑sum‑exp) approximation of max(conv(a, a)).
    Operates on log‑softmax (logits) to stay on the simplex.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)  # initialise logits

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # soft‑max to obtain a probability vector
        a = np.exp(logits)
        a = a / a.sum()

        # convolution
        conv = _conv_fft(a, a)

        # temperature schedule (linear annealing)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        # smooth‑max (log‑sum‑exp)
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()

        # gradient w.r.t. a
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule ∝ 1/√it
        lr = lr0 / sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional exact evaluation
        if (it & 1023) == 0:  # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

            if best_val < 1.45:   # early stop if already excellent
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Aggressive KL‑mirror‑descent polishing (exact max)
# ----------------------------------------------------------------------
def _aggregated_mirror_descent(
    start_seq: np.ndarray,
    deadline: float,
    eps_rel_start: float = 0.01,
    eps_rel_end: float = 1e-6,
    lr_factor: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent where the sub‑gradient aggregates *all* convolution
    entries that are within a relative tolerance of the current maximum.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    t0 = time.time()
    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        # exponential decay of the tolerance
        elapsed = time.time() - t0
        total = deadline - t0
        frac = elapsed / total if total > 0 else 0.0
        eps_rel = eps_rel_start * (eps_rel_end / eps_rel_start) ** frac
        delta = eps_rel * max_val

        active_idx = np.where(conv >= max_val - delta)[0]
        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        # aggregated sub‑gradient = 2 * (active ⋆ a_rev)
        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        # exponentiated‑gradient step (KL‑mirror)
        lr = lr_factor / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # stabilise numerics
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[AggMD] it={it:7d} n={n:5d} best={val:.6f}")

            # break if time is almost exhausted
            if time.time() > deadline - 0.5:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Local stochastic refinement (logit perturbations)
# ----------------------------------------------------------------------
def _local_random_refine(
    start_seq: np.ndarray,
    deadline: float,
    sigma_start: float = 0.5,
    sigma_end: float = 1e-4,
    attempts: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing on logits: add Gaussian noise with a temperature
    that decays from sigma_start to sigma_end, keep perturbation if
    it improves the exact objective.
    """
    n = len(start_seq)
    eps = 1e-30
    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    logit = np.log(best_seq + eps)

    it = 0
    while time.time() < deadline and it < attempts:
        it += 1
        # annealed sigma
        frac = it / attempts
        sigma = sigma_start * (sigma_end / sigma_start) ** frac

        pert = logit + rng.normal(0.0, sigma, size=n)
        cand = np.exp(pert)
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            logit = pert  # accept new logits
            print(f"[RandRefine] it={it:7d} n={n:5d} val={val:.6f}")

        # early break if we have already found a very good value
        if best_val < 1.45:
            break

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence as a plain list.
    """
    global rng
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Re‑use a previously computed solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Candidate lengths (FFT‑friendly) – we focus on the range that
    # previously gave good results, but also explore a couple of larger
    # sizes for completeness.
    # ------------------------------------------------------------------
    lengths = [2000, 2500, 3000, 3500, 4000, 5000, 6000, 8000, 10000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic families
    # ------------------------------------------------------------------
    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.25),
        _arcsine_shape,
        _sin_shape,
        lambda n: _beta_shape(n, 0.5, 0.5),
        lambda n: _beta_shape(n, 2.0, 2.0),
        lambda n: _beta_shape(n, 0.5, 2.0),
        lambda n: _beta_shape(n, 2.0, 0.5),
        lambda n: _power_shape(n, -1.5),   # edge‑heavy power law
        lambda n: _power_shape(n,  1.0),   # interior‑heavy power law
    )
    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

        # try reusing a previously stored solution of the same length
        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – intensive random sampling
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.30   # 30 % of remaining budget
    sample_deadline = now + time_for_sampling

    alphas = [0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]
    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))

        # Dirichlet draw with random concentration
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd‑Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        # Random Beta pdf
        seq = _random_beta_shape(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd‑Beta] n={n:5d} val={val:.6f}")

        # Random power‑law shape
        seq = _random_power_shape(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd‑Pow] n={n:5d} val={val:.6f}")

        # early exit if we already beat the target comfortably
        if best_val < 1.45:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best_seq is not None:
        # collect a few diverse restart points
        candidates = [best_seq]
        top_len = len(best_seq)

        # two fresh Dirichlet starts of the same length
        for _ in range(2):

            candidates.append(cand)

            candidates.append(cand_big)

        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.25   # ~25 % of remaining

        for idx, cand in enumerate(candidates):
            seq_adam, val_adam = _smooth_adam_optimize(
                start_seq=cand,
                deadline=adam_deadline,
                beta_start=5.0,
                beta_end=2000.0,
                max_iter=5_000_000,
                lr0=0.02,

    # ------------------------------------------------------------------
    # Phase 4 – Aggressive mirror‑descent polishing (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        md_deadline = time.time() + remaining * 0.30   # 30 % for MD
        seq_md, val_md = _aggregated_mirror_descent(
            start_seq=best_seq,
            deadline=md_deadline,
            eps_rel_start=0.01,
            eps_rel_end=1e-6,
            lr_factor=1.0,
        )
        if val_md < best_val:
            best_val = val_md
            best_seq = seq_md
            print(f"[Agg‑MD] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Local random logit perturbation refinement
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            seq_ref, val_ref = _local_random_refine(
                start_seq=best_seq,
                deadline=time.time() + remaining * 0.30,
                sigma_start=0.5,
                sigma_end=1e-5,
                attempts=200_000,
            )
            if val_ref < best_val:
                best_val = val_ref
                best_seq = seq_ref
                print(f"[Rand‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will evaluate again anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
