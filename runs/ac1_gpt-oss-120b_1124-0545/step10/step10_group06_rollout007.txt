def _subgradient_descent(start, deadline, max_iters=200000, init_step=1.0):
    a = start.copy()
    n = a.size
    best = a.copy()
    best_val = _objective(a)
    step0 = init_step
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute convolution
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        # compute gradient of max entry
        # gradient = 2 * rev(a) shifted by k
        grad = np.zeros_like(a)
        # indices i where k - i in range
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad[mask] = 2.0 * a[idx[mask]]
        # project gradient onto simplex tangent: subtract mean
        grad_proj = grad - grad.mean()
        # step size decreasing
        step = step0 / math.sqrt(t)
        # take a step in descending direction
        a_new = a - step * grad_proj
        # project back to simplex
        a_new = _project_to_simplex(a_new)
        # evaluate
        val = _objective(a_new)
        if val < best_val:
            best_val = val
            best = a_new.copy()
            # print progress occasionally
            if t % 5000 == 0:
                print(f"[Subgrad] it={t} best={val:.6f}")
        a = a_new
    return best, best_val
```

def _project_to_simplex(v):
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # Find rho
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1  # zero-index
    theta = cssv[ rho ] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _project_to_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a three‑phase pipeline:

1. **Adam on the simplex (soft‑max parametrisation)** – a smooth
   approximation of the max‑convolution is optimised while the
   temperature λ is increased slowly.  This quickly finds a good basin.

2. **Hard‑max sub‑gradient descent** – after the smoothing is removed,
   an exact sub‑gradient of the true objective is taken on the simplex
   (projected gradient descent).  A decreasing step size and occasional
   random kicks allow it to escape tiny plateaus and shave off the last
   bits of the bound.

3. **Light simulated‑annealing polish** – a Metropolis walk that swaps
   small amounts of mass between coordinates (plus occasional Gaussian
   jumps) finishes the optimisation.

Multiple deterministic seeds (uniform, exponential decay, triangular,
random soft‑max…) are tried; the best sequence seen over the whole
budget (default 1000 s) is stored in the global ``height_sequence_1`` and
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors using FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the objective (same as evaluate_sequence for a
    normalised probability vector).  No clipping – the optimiser works
    with the true mathematical formula.
    """
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex helpers
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed families (return a probability vector that sums to 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    # stable soft‑max
    m = np.max(logits)
    exps = np.exp(logits - m)
    return exps / exps.sum()


def _exponential_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        dec = inc[::-1]
    else:
        dec = np.arange(half, 0, -1, dtype=np.float64)
    a = np.concatenate([inc, dec])
    return a / a.sum()


def _sinusoidal(n: int) -> np.ndarray:
    """f(i) ∝ sin²(π·(i+0.5)/n), normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    return a / a.sum()


# ----------------------------------------------------------------------
# Soft‑max (smooth) max utilities for Adam phase
# ----------------------------------------------------------------------
def _soft_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Soft‑max weights over entries of `conv`.  Large λ -> hard max.
    """
    max_c = conv.max()
    if lam <= 0.0:
        active = np.isclose(conv, max_c, atol=1e-12)
        cnt = np.count_nonzero(active)
        return np.where(active, 1.0 / cnt, 0.0)
    rel = (conv - max_c) / max_c          # ≤ 0
    exps = np.exp(lam * rel)
    return exps / exps.sum()


def _gradient_wrt_a(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution with respect to the vector
    a (which is already on the simplex).  Implements
    g = 2·rev(a) ⊛ w where w are the soft‑max weights.
    """
    n = a.size
    conv = _conv_fft(a, a)
    w = _soft_max_weights(conv, lam)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)               # length = 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n]       # slice that aligns with a
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 300_000,
                   lam_start: float = 2.0,
                   lam_factor: float = 1.0005) -> tuple[np.ndarray, float]:
    """
    Adam on log‑probabilities with a slowly increasing temperature λ.
    Returns the best probability vector found within the provided time.
    """
    eps = 1e-30
    logp = np.log(start + eps)
    n = start.size
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = start.copy()
    best_val = _objective(start)

    t = 0
    lam = lam_start
    last_improve = 0

    while t < max_iters and time.time() < deadline:
        t += 1

        # current simplex point via stable soft‑max
        logp_shift = logp - np.max(logp)
        a = np.exp(logp_shift)
        a /= a.sum()

        # gradient of the smoothed objective
        g_a = _gradient_wrt_a(a, lam)

        # chain rule to gradient wrt log‑probabilities
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log * g_log)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.15 / (math.sqrt(t) + 1.0)   # simple schedule
        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # exact (non‑smoothed) objective for bookkeeping
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = 0
            print(f"[Adam] it={t:7d}  best={cur_val:.6f}  λ={lam:.2f}")
        else:
            last_improve += 1

        # increase smoothing temperature
        lam *= lam_factor

        # occasional tiny random kick to get out of shallow plateaus
        if last_improve > 30_000:
            noise = np.random.default_rng().normal(scale=0.02, size=n)
            logp += noise
            last_improve = 0

    return best_seq, best_val


# ----------------------------------------------------------------------
# Hard‑max sub‑gradient descent on the true objective
# ----------------------------------------------------------------------
def _subgradient_descent(start: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         init_step: float = 0.8,
                         no_imp_limit: int = 10_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true L∞‑objective.
    Uses the exact max‑convolution gradient (hard max) and projects
    back onto the simplex after each step.
    """
    a = start.copy()
    n = a.size
    best = a.copy()
    best_val = _objective(a)

    no_improve = 0
    rng = np.random.default_rng()

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # full convolution (needed to locate the active index)
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active max entry: g_i = 2 * a_{k-i}
        grad = np.zeros_like(a)
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad[mask] = 2.0 * a[idx[mask]]

        # project gradient onto the tangent space of the simplex
        grad_proj = grad - grad.mean()

        # decreasing step size (standard sub‑gradient schedule)
        step = init_step / math.sqrt(t)

        # gradient step + projection
        a_new = a - step * grad_proj
        a_new = _project_to_simplex(a_new)

        val = _objective(a_new)
        if val < best_val:
            best_val = val
            best = a_new.copy()
            no_improve = 0
            if t % 5_000 == 0:
                print(f"[Subgrad] it={t:7d}  new best={val:.6f}")
        else:
            no_improve += 1

        # occasional random kick when stuck
        if no_improve >= no_imp_limit:
            noise = rng.normal(scale=0.01, size=n)
            a_new = a + noise
            a_new = np.clip(a_new, 0.0, None)
            a_new = _project_to_simplex(a_new)
            no_improve = 0

        a = a_new

    return best, best_val


# ----------------------------------------------------------------------
# Light simulated‑annealing polishing (unchanged from the previous
# implementation, only tiny parameter tweaks)
# ----------------------------------------------------------------------
def _anneal_refine(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0
    n = cur.size

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a small fraction from i to j
        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d}  best={new_val:.6f}")

        # every ~2000 steps try a global Gaussian perturbation
        if it % 2000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            s = new_seq.sum()
            if s > 0.0:
                new_seq /= s
            else:
                new_seq = _uniform(n)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val

# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative step function under the given budget.
    Returns the sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0        # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.  4000 is a good trade‑off; larger sizes give
    # more degrees of freedom but each iteration becomes slower.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising step function (length {n}) ===")

    # ------------------------------------------------------------------
    # Build a pool of seeds (including the previous global result if it
    # matches the current length)
    # ------------------------------------------------------------------
    seeds = []

    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                if arr.sum() > 0:
                    seeds.append(arr / arr.sum())
        except Exception:
            pass

    # deterministic and random seeds
    seeds.extend([
        _uniform(n),
        _random_softmax(n, rng),
        _exponential_decay(n, rate=0.001),
        _exponential_decay(n, rate=0.003),
        _triangular(n),
        _sinusoidal(n),
    ])

    # ------------------------------------------------------------------
    # Keep the best initial seed according to the true objective
    # ------------------------------------------------------------------
    best_seed = None
    best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seed = s.copy()
    print(f"[Init] best initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimiser (smooth max)
    # ------------------------------------------------------------------
    phase1_deadline = deadline - 35.0   # reserve ~35 s for later phases
    if time.time() < phase1_deadline:
        print("[Phase 1] Adam optimisation start")
        seq_adam, val_adam = _adam_optimize(best_seed, phase1_deadline,
                                            max_iters=300_000,
                                            lam_start=2.0,
                                            lam_factor=1.0005)
        print(f"[Phase 1] best after Adam = {val_adam:.6f}")
    else:
        seq_adam, val_adam = best_seed, best_val
        print("[Phase 1] skipped (insufficient time)")

    # ------------------------------------------------------------------
    # Phase 2 – Hard‑max sub‑gradient descent
    # ------------------------------------------------------------------
    phase2_deadline = deadline - 15.0
    if time.time() < phase2_deadline:
        print("[Phase 2] Sub‑gradient descent start")
        seq_sgd, val_sgd = _subgradient_descent(seq_adam, phase2_deadline,
                                                max_iters=250_000,
                                                init_step=0.8,
                                                no_imp_limit=15_000)
        print(f"[Phase 2] best after sub‑grad = {val_sgd:.6f}")
    else:
        seq_sgd, val_sgd = seq_adam, val_adam
        print("[Phase 2] skipped (insufficient time)")

    # ------------------------------------------------------------------
    # Phase 3 – Light simulated‑annealing polish
    # ------------------------------------------------------------------
    seq_final, val_final = seq_sgd, val_sgd
    if time.time() < deadline - 0.5:
        print("[Phase 3] Simulated‑annealing polishing")
        seq_final, val_final = _anneal_refine(seq_sgd, deadline)
        if val_final < val_sgd:
            print(f"[Phase 3] improved to {val_final:.6f}")
        else:
            print("[Phase 3] no improvement")
    else:
        print("[Phase 3] no time left for polishing")

    # ------------------------------------------------------------------
    # Final normalisation and export
    # ------------------------------------------------------------------
    final_seq = np.maximum(seq_final, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(final_seq.size)
        total = final_seq.sum()
    final_seq = final_seq / total

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")

    # store for possible warm‑starts in the next run
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
