def _penalty_optimize(a0, n, deadline):
    a = a0.copy()
    best_a = a0.copy()
    best_val = _objective_surrogate(a)
    # start with current max = best_val/(2*n)
    M = _conv_fft(a,a).max()
    # schedule: reduce M by factor each outer iteration
    while time.time() < deadline:
        # try new M_target = M * (1 - delta) where delta small (like 0.001)
        M_target = M * 0.998  # 0.2% reduction
        # run inner gradient descent attempts
        for inner in range(2000): # limited
            if time.time() > deadline: break
            conv = _conv_fft(a, a)
            excess = conv - M_target
            excess = np.where(excess > 0, excess, 0.0)
            # compute gradient:
            # weight w = 2 * excess (since derivative of excess^2 w.r.t a)
            w = 2 * excess
            # gradient = conv_rev of w
            grad = 2 * _conv_fft(a[::-1], w)[n-1:n-1+n]
            # step size adapt small
            step = 0.01 / np.sqrt(inner+1)
            a = a - step * grad
            a = _proj_simplex(a)
            # check if max(conv) <= M_target (+ tolerance)
            conv = _conv_fft(a, a)
            cur_max = conv.max()
            if cur_max <= M_target*(1+1e-6):
                # success in achieving lower M
                M = cur_max
                best_a = a.copy()
                best_val = 2*n*cur_max
                break
        else:

"""
Propose a non‑negative step function that minimises

    2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm follows an enhanced multi‑stage scheme:

1. **Seed pool** – a richer collection of analytically‑derived and random
   distributions (uniform, triangular, windows, beta‑shapes, Dirichlet,
   low‑frequency curves, two‑block patterns, …) all ℓ₁‑normalised.

2. **Log‑space Adam** – optimisation on the log‑softmax parametrisation.
   The surrogate is the true max‑convolution (no log‑sum‑exp smoothing):
   the smoothed max is gradually sharpened by raising a temperature β.

3. **Exact sub‑gradient descent** – a projected sub‑gradient descent
   directly on the true ℓ∞‑objective with a Polyak‑type step‑size schedule.

4. **Penalty‑based feasibility descent** – starting from the current best,
   we repeatedly lower a target peak `M` and minimise the quadratic penalty
   `∑ max(0, conv‑M)²` by gradient descent on the simplex.  This aggressively
   drives the maximal convolution value down.

5. **Greedy local refinement** – random pair‑wise mass transfers that are
   kept only if they improve the true (max‑convolution) objective.

6. **Optional upscale** – if time permits, the best solution is interpolated
   to a larger length (e.g. 8000) and the refinement stages are rerun.

All stages respect the global deadline (default 1000 s). The final best
sequence is stored globally as ``height_sequence_1`` and returned as a plain

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) (sum(seq) is assumed = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) -
                                            2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    if _scipy_kaiser is not None:
        a = _scipy_kaiser(n, beta)
        s = a.sum()
        return a / s if s > 0 else _uniform(n)

    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    if _scipy_i0 is not None:
        a = _scipy_i0(beta * np.sqrt(1 - y * y))
    else:
        t = beta * np.sqrt(1 - y * y)
        a = 1.0 + (t ** 2) / 4.0 + (t ** 4) / 64.0 + (t ** 6) / 2304.0 + (t ** 8) / 147456.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components; result is ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _two_block(n: int, block_len: int, gap: int) -> np.ndarray:
    """Two disjoint uniform blocks of equal mass."""
    a = np.zeros(n, dtype=np.float64)
    if block_len * 2 + gap > n:
        # fallback to uniform
        return _uniform(n)
    start1 = (n - (2 * block_len + gap)) // 2
    start2 = start1 + block_len + gap
    a[start1:start1 + block_len] = 1.0 / block_len
    a[start2:start2 + block_len] = 1.0 / block_len
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection – richer than the previous version
# ----------------------------------------------------------------------
def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.15, 0.45, 7):
        seeds.append(_gaussian(n, sigma))

    # Power‑law family
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta family – broader grid (including extreme values)
    alphas = np.concatenate([np.linspace(0.05, 0.2, 4), np.linspace(0.2, 5.0, 9)])
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential (heavy‑tailed)
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Two‑block patterns (different gaps / lengths)
    for blk in (500, 800, 1200):
        for gap in (200, 400, 800):
            seeds.append(_two_block(n, blk, gap))

    # Mixtures of uniform with a low‑frequency curve
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # Normalise & clean
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

# Optimisation primitives
# ----------------------------------------------------------------------
def _logspace_adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on the log‑softmax parametrisation.
    Uses the true max‑convolution (via a slowly increasing temperature β).
    Returns the best vector (ℓ₁‑normalised) found and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits
    m = np.zeros_like(v)                        # first moment
    v_sq = np.zeros_like(v)                     # second moment

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5          # initial smoothing temperature
    beta_inc = 0.3
    beta_max = 250.0   # final temperature – close to true max

    lr0 = 1.0           # initial learning rate (log‑space)
    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # Softmax to enforce simplex
        a = _softmax(v)

        # Compute surrogate (actual max)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[LogAdam] it={it:6d} best={cur_val:.6f}")

        # Smoothed max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # Gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]        # central n entries

        # Chain rule to logits
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update (learning‑rate decays like 1/√t)
        lr = lr0 / np.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # Temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_500_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Uses a decreasing step‑size and Euclidean projection onto the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # Sub‑gradient (same pattern as used in the original code)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Step size (Polyak‑style constant / sqrt(t))
        lr = 0.20 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _penalty_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_outer: int = 30,
    inner_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Decrease a target peak M via quadratic penalty minimisation:
        P(a) = Σ max(0, conv(a,a) - M)²
    Gradient descent with simplex projection is used in the inner loop.
    """
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    M = conv.max()
    best_seq = a.copy()
    best_val = 2.0 * n * M

    for outer in range(max_outer):
        if time.time() > deadline:
            break
        # Reduce target by a small factor (more aggressive when far)
        reduction = 0.005 if M > best_val / (2.0 * n) else 0.001
        target = M * (1.0 - reduction)

        for it in range(1, inner_iters + 1):
            if time.time() > deadline:
                break
            conv = _conv_fft(a, a)
            excess = conv - target
            excess = np.where(excess > 0, excess, 0.0)

            # If no excess, we have satisfied the current target
            if not excess.any():
                M = target
                best_seq = a.copy()
                best_val = 2.0 * n * M
                print(f"[Penalty] achieved target M={M:.6e} at outer={outer}")
                break

            w = 2.0 * excess            # gradient of penalty term
            grad = 2.0 * _conv_fft(a[::-1], w)[n - 1:n - 1 + n]

            # Simple diminishing step size
            lr = 0.12 / np.sqrt(it)
            a = a - lr * grad
            a = _proj_simplex(a)

        # Re‑evaluate after inner loop
        conv = _conv_fft(a, a)
        cur_max = conv.max()
        if cur_max < M - 1e-12:
            M = cur_max
            best_seq = a.copy()
            best_val = 2.0 * n * M
            print(f"[Penalty] improved M={M:.6e} after outer={outer}")
        else:
            # No improvement – stop early
            break

    return best_seq, best_val


def _local_greedy_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climbing: move a small fraction of mass from a
    heavy contributor to a light one, keeping the move only if it reduces
    the true max‑convolution objective.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] <= 1e-12:
            continue
        eps = rng.random() * cur[i] * 0.04   # move up to 4 % of source mass
        if eps <= 1e-12:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps

        new_val = _objective_surrogate(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur.copy()
                if it % 5000 == 0:
                    print(f"[Greedy] it={it:5d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Length selection – start with 4000 (good trade‑off)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – seed generation & cheap surrogate ranking
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} candidate seeds.")

    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 8
    best_seeds = [vec for vec, _ in evals[:top_k]]

    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – log‑space Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate a fraction of the remaining time to this Adam run
        max_iters = int(min(600_000, remaining * 0.12 * 1e5))
        print(f"\n--- Log‑space Adam run {idx+1}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_sgd, val_sgd = _logspace_adam_optimize(
            seed_vec, n, deadline, max_iters=max_iters
        )
        if seq_sgd is not None and val_sgd < global_best_val:
            global_best_seq = seq_sgd.copy()
            global_best_val = val_sgd
            print(f"[overall‑Adam] new global best = {val_sgd:.6f}")

        # ------------------------------------------------------------------
        # Sub‑gradient projection (exact max) – modest budget per seed
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        max_iters_sub = int(min(1_800_000, remaining * 0.30 * 1e5))
        if max_iters_sub > 0:
            seq_proj, val_proj = _subgradient_proj(
                global_best_seq, n, deadline, max_iters=max_iters_sub
            )
            if val_proj < global_best_val:
                global_best_seq = seq_proj.copy()
                global_best_val = val_proj
                print(f"[overall‑SubProj] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – penalty‑based descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Penalty‑based peak reduction ---")
        remaining = deadline - time.time()
        max_outer = int(min(30, remaining * 0.10))
        seq_pen, val_pen = _penalty_optimize(
            global_best_seq, n, deadline,
            max_outer=max_outer,
            inner_iters=1500
        )
        if val_pen < global_best_val:
            global_best_seq = seq_pen.copy()
            global_best_val = val_pen
            print(f"[Penalty] new global best = {val_pen:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – greedy local refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Greedy local refinement ---")
        remaining = deadline - time.time()
        max_iters = int(min(80_000, remaining * 1e4))
        seq_greedy, val_greedy = _local_greedy_refine(
            global_best_seq, n, deadline, max_iters=max_iters
        )
        if val_greedy < global_best_val:
            global_best_seq = seq_greedy.copy()
            global_best_val = val_greedy
            print(f"[Greedy] new global best = {val_greedy:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – optional upscale if enough time remains
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:   # need at least ~30 s for upscale work
        n2 = n * 2   # double the length
        print(f"\n--- Upscaling to n={n2} and polishing ---")
        # Linear interpolation of the current best to the larger grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n2, endpoint=False)
        upscaled = np.interp(x_new, x_old, global_best_seq)
        upscaled = upscaled / upscaled.sum()
        # Run a short penalty optimisation on the larger problem
        deadline2 = deadline
        seq_up, val_up = _penalty_optimize(
            upscaled, n2, deadline2,
            max_outer=12,
            inner_iters=1200
        )
        # If the upscaled version is better, adopt it (and rescale back)
        if val_up < global_best_val:
            # Rescale back to original length for final output
            x_up = np.linspace(0, 1, n2, endpoint=False)
            downscaled = np.interp(x_old, x_up, seq_up)
            downscaled = downscaled / downscaled.sum()
            # final local refinement on original size
            seq_final, val_final = _local_greedy_refine(
                downscaled, n, deadline2, max_iters=40_000
            )
            if val_final < global_best_val:
                global_best_seq = seq_final.copy()
                global_best_val = val_final
                print(f"[Upscale] adopted improved solution = {val_final:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
