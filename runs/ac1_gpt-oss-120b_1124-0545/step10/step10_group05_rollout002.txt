def _hill_climbing(seq, deadline, rng):
    n = len(seq)
    best_seq = seq.copy()
    best_val = _objective(seq)
    step_scale = 0.2
    while time.time() < deadline:

        new_seq[j] += delta

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm (≈1000 s total)
--------------------------------
1. **Deterministic analytic seeds** – generate many classic windows
   (uniform, Hann, Hamming, Blackman, Gaussian, arcsine, sinusoid,
   Catalan) plus a few extra shapes (triangular, parabolic,
   exponential decay, inverse‑distance).  Keep the best one.

2. **Heavy random sampling** – draw Dirichlet vectors with many
   concentration parameters, random Beta pdfs and mixtures of 2‑3
   Beta components.  Keep the global best.

3. **Smoothed Adam optimisation** – run several Adam restarts on a
   log‑sum‑exp smooth approximation of ``max(conv(a,a))`` with a
   temperature schedule up to β = 2000.  The Adam updates are performed
   on logits (log‑softmax) to guarantee non‑negativity and unit sum.

4. **Optional L‑BFGS‑B refinement** – one L‑BFGS‑B run on the same
   smooth objective (if SciPy is available).

5. **Aggressive aggregated mirror‑descent** – KL‑mirror descent that
   aggregates *all* near‑maximal convolution entries with an
   exponentially decaying tolerance.

6. **Stochastic hill‑climbing fine‑tuning** – repeatedly move a tiny
   amount of mass from a random index to another (preserving the sum)
   and keep improvements; the step size decays with time.

All phases respect a global 1000 s budget.  The best sequence found
is stored in the global ``height_sequence_1`` variable for possible
reuse and returned as a plain Python list.
"""

import time

from math import sqrt, log, pi, exp

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a *normalised* sequence
    (sum == 1).  Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * float(np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0 - a1 * np.cos(2 * pi * i / (n - 1)) + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


def _sin_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(pi * i / n)
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = 1.0 - np.abs(i - centre) / centre
    w[w < 0] = 0.0
    return w / w.sum()


def _parabolic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = 1.0 - ((i - centre) / centre) ** 2
    w[w < 0] = 0.0
    return w / w.sum()


def _exp_decay(n: int, rate: float = 0.001) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - centre))
    return w / w.sum()


def _inverse_square(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    dist = np.abs(i - centre) + 1.0
    w = 1.0 / (dist ** 2)
    return w / w.sum()

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet‑like vector with concentration α."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


def _random_beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.uniform(0.1, 5.0)
    b = rng.uniform(0.1, 5.0)
    return _beta_shape(n, a, b)


def _random_mixture_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two independent beta pdfs."""
    w = rng.random()
    a1, b1 = rng.uniform(0.1, 5.0), rng.uniform(0.1, 5.0)
    a2, b2 = rng.uniform(0.1, 5.0), rng.uniform(0.1, 5.0)
    s1 = _beta_shape(n, a1, b1)
    s2 = _beta_shape(n, a2, b2)
    mix = w * s1 + (1.0 - w) * s2
    mix /= mix.sum()
    return mix


def _random_mixture_beta_k(n: int, rng: np.random.Generator, k: int = 3) -> np.ndarray:
    """Mixture of *k* beta pdfs with Dirichlet weights."""
    weights = rng.dirichlet(np.ones(k))
    mix = np.zeros(n, dtype=np.float64)
    for idx in range(k):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        mix += weights[idx] * _beta_shape(n, a, b)
    mix /= mix.sum()
    return mix


# ---------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax parametrisation)
# ---------------------------------------------------------------

def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    max_iter: int = 5_000_000,
    lr0: float = 0.03,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth (log‑sum‑exp) approximation of max(conv(a,a)).
    Uses Adam on the log‑softmax (logits) parametrisation.
    """
    n = len(start_seq)
    eps = 1e-30

    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # soft‑max to obtain a probability vector (ensures non‑negativity & sum = 1)
        a = np.exp(logits)
        a /= a.sum()

        # self‑convolution
        conv = _conv_fft(a, a)

        # temperature schedule (linear rise)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        # smooth‑max via log‑sum‑exp (stable)
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()               # soft weights, sum=1

        # gradient w.r.t. a
        conv_pa = _conv_fft(p, a[::-1])               # length = 3n‑2
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional exact evaluation
        if (it & 1023) == 0:          # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

            if best_val < 1.45:        # early exit if near theoretical bound
                break

    return best_seq, best_val


# ---------------------------------------------------------------
# Optional L‑BFGS‑B refinement (smooth max with large β)
# ---------------------------------------------------------------

def _lbfgs_refine(start_seq: np.ndarray, deadline: float, beta: float = 2000.0) -> tuple[np.ndarray, float]:
    """
    Run a single L‑BFGS‑B optimisation on the smooth log‑sum‑exp max.
    Returns the refined sequence and its exact objective.
    """
    if minimize is None:
        return start_seq, _objective(start_seq)

    n = len(start_seq)
    eps = 1e-30

    def fun_and_grad(logits: np.ndarray):
        a = np.exp(logits)
        a /= a.sum()
        conv = _conv_fft(a, a)
        c = np.max(conv)
        exps = np.exp(beta * (conv - c))
        lse = c + (1.0 / beta) * np.log(exps.sum())
        value = 2.0 * n * lse

        # gradient w.r.t. a
        p = exps / exps.sum()
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)
        return value, grad_logits.astype(np.float64)

    x0 = np.log(start_seq + eps)

    class TimeOut(RuntimeError):
        pass

    def callback(xk):
        if time.time() >= deadline:
            raise TimeOut()

    try:
        res = minimize(
            fun=lambda x: fun_and_grad(x)[0],
            x0=x0,
            jac=lambda x: fun_and_grad(x)[1],
            method='L-BFGS-B',
            options={'maxiter': 200, 'disp': False},
            callback=callback,
        )
    except TimeOut:
        best_logits = x0
    else:
        best_logits = res.x

    best_a = np.exp(best_logits)
    best_a /= best_a.sum()
    best_val = _objective(best_a)
    return best_a, best_val

# Aggressive aggregated mirror‑descent polishing
# ---------------------------------------------------------------

def _aggregated_mirror_descent(
    start_seq: np.ndarray,
    deadline: float,
    eps_rel_start: float = 0.01,
    eps_rel_end: float = 1e-6,
    lr_factor: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent where the sub‑gradient aggregates *all* entries
    whose convolution value is within a relative tolerance of the running
    maximum.  The tolerance decays exponentially during the run.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    t0 = time.time()
    it = 0
    eps = 1e-30
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        elapsed = time.time() - t0
        total = deadline - t0
        frac = elapsed / total if total > 0 else 0.0
        eps_rel = eps_rel_start * (eps_rel_end / eps_rel_start) ** frac
        delta = eps_rel * max_val

        active_idx = np.where(conv >= max_val - delta)[0]
        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        lr = lr_factor / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[AggMD] it={it:7d} n={n:5d} best={val:.6f}")

            if time.time() > deadline - 0.5:
                break

    return best_seq, best_val

# Stochastic hill‑climbing fine‑tuning
# ---------------------------------------------------------------

def _hill_climb(start_seq: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple hill climbing: repeatedly move a small amount of mass
    from a random index i to another j (preserving the total sum).
    The step size decays linearly with elapsed time.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    n = len(a)

    t0 = time.time()
    total = max(deadline - t0, 1e-9)

    while time.time() < deadline:
        elapsed = time.time() - t0
        frac = elapsed / total
        # step size goes from 0.2 to 0.0005 of the donor mass
        step_max = 0.20
        step_min = 0.0005
        step = step_max * (1 - frac) + step_min * frac

        i, j = rng.choice(n, size=2, replace=False)
        donor = a[i]
        if donor <= 0.0:
            continue
        max_move = step * donor
        if max_move <= 0.0:
            continue
        delta = rng.uniform(0.0, max_move)

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        val = _objective(new_a)
        if val < best_val:
            best_val = val
            best_seq = new_a.copy()
            a = new_a
            # occasional progress report
            if (int(elapsed * 100) % 2000) == 0:
                print(f"[HillClimb] t={elapsed:.1f}s val={val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective under the given time budget.
    Returns the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Re‑use a previously computed solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Problem sizes – a selection of FFT‑friendly lengths
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000, 8000, 10000, 12000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic families (seed generation)
    # ------------------------------------------------------------------
    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.25),
        _arcsine_shape,
        _sin_shape,
        lambda n: _beta_shape(n, 0.5, 0.5),
        lambda n: _beta_shape(n, 2.0, 2.0),
        lambda n: _beta_shape(n, 0.5, 2.0),
        lambda n: _beta_shape(n, 2.0, 0.5),
        _triangular,
        _parabolic,
        lambda n: _exp_decay(n, rate=0.001),
        lambda n: _exp_decay(n, rate=0.005),
        _inverse_square,
    )

    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")
        # try to reuse a previously stored solution of the same length
        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – heavy random sampling (Dirichlet + random Beta shapes)
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.30   # about 30 % of remaining budget
    sample_deadline = now + time_for_sampling

    alphas = [0.1, 0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0]
    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))

        # Dirichlet draw with random concentration
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd-Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        # Random Beta pdf
        seq = _random_beta_shape(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd-Beta] n={n:5d} val={val:.6f}")

        # Random mixture of two Betas
        seq = _random_mixture_beta(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd-Mix2] n={n:5d} val={val:.6f}")

        # Random mixture of three Betas
        seq = _random_mixture_beta_k(n, rng, k=3)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd-Mix3] n={n:5d} val={val:.6f}")

        # early stop if already very good
        if best_val < 1.45:
            break

    # ------------------------------------------------------------------
    # Phase 3 – Smoothed Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        candidates = [best_seq]
        top_len = len(best_seq)

        # two fresh Dirichlet starts of the same length
        for _ in range(2):
            cand = _random_dirichlet(top_len, rng, alpha=rng.choice(alphas))
            candidates.append(cand)

        # one larger‑length start (if possible)
        larger_lengths = [L for L in lengths if L > top_len]
        if larger_lengths:
            L_big = int(rng.choice(larger_lengths))
            cand_big = _random_dirichlet(L_big, rng, alpha=1.0)
            candidates.append(cand_big)

        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.25   # ~25 % of remaining time

        for idx, cand in enumerate(candidates):
            seq_adam, val_adam = _smooth_adam_optimize(
                start_seq=cand,
                deadline=adam_deadline,
                beta_start=5.0,
                beta_end=2000.0,
                max_iter=6_000_000,
                lr0=0.03,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = seq_adam
                print(f"[Adam‑Refine {idx}] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – optional L‑BFGS‑B refinement
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0 and minimize is not None:
            lbfgs_deadline = time.time() + remaining * 0.20
            seq_lbfgs, val_lbfgs = _lbfgs_refine(best_seq, lbfgs_deadline, beta=2000.0)
            if val_lbfgs < best_val:
                best_val = val_lbfgs
                best_seq = seq_lbfgs
                print(f"[L‑BFGS] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – aggressive aggregated mirror‑descent polishing
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _aggregated_mirror_descent(
            start_seq=best_seq,
            deadline=deadline,
            eps_rel_start=0.01,
            eps_rel_end=5e-7,
            lr_factor=1.2,
        )
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Agg‑MD] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – stochastic hill‑climbing fine‑tuning
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 1.0:
            hill_deadline = time.time() + remaining
            seq_hc, val_hc = _hill_climb(best_seq, hill_deadline, rng)
            if val_hc < best_val:
                best_val = val_hc
                best_seq = seq_hc
                print(f"[HillClimb] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will evaluate again anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
