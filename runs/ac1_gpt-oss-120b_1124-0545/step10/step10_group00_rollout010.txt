    best_seq = start_vec.copy()
    best_val = _objective(start_vec)
    v0 = np.log(start_vec + 1e-30)
    beta = beta_start
    for stage in range(stages):
        if time.time() > deadline: break
        # ensure beta not exceed beta_end
        beta = min(beta_end, beta_start * (beta_end/beta_start) ** (stage/(stages-1)) )
        # define objective function closure using beta variable
        def func_and_grad(v):
            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_conv = np.max(conv)
            # compute smooth max
            shift = max_conv
            exp_term = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_term)
            smooth = shift + np.log(sum_exp) / beta
            obj = 2.0 * n * smooth
            # gradient

            grad_a = _grad_wrt_a(a, conv, weights)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            if time.time() > deadline:
                raise TimeOutException()

def _targeted_move(cur, conv_max_idx, rng):
    n = len(cur)
    # choose i that participates in convolution at max_idx: i ranges max(0, max_idx-(n-1)) to min(n-1, max_idx)
    i_start = max(0, conv_max_idx - (n - 1))
    i_end = min(n - 1, conv_max_idx)
    # list of participating indices
    idx_i = np.arange(i_start, i_end + 1)
    # compute contributions a_i * a_{max_idx - i}
    partner = conv_max_idx - idx_i
    contrib = cur[idx_i] * cur[partner]
    # probability proportional to contribution; choose i
    if contrib.sum() == 0.0:
        return None
    prob = contrib / contrib.sum()
    i = rng.choice(idx_i, p=prob)
    j = conv_max_idx - i
    # transfer small epsilon from i and j to random other positions equally
    eps = 0.01 * min(cur[i], cur[j])  # small proportion
    if eps <= 0: 
        return None
    new = cur.copy()
    new[i] -= eps
    new[j] -= eps
    # distribute eps to two random other indices (to keep sum unchanged)
    k1, k2 = rng.choice(n, size=2, replace=False)
    new[k1] += eps/2
    new[k2] += eps/2
    # ensure nonnegative
    new = np.maximum(new, 0.0)
    # renorm
    s = new.sum()
    if s > 0:
        new /= s
    else:
        new = _uniform(n)
    return new
```

Actually moving eps from i and j reduces product a_i*a_j by O(eps*(a_i + a_j)) controlling peak.

def _simulated_annealing_optimize(start_seq, n, deadline, max_iters=1_200_000, init_temp_factor=0.03, min_temp=1e-12):
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    T = max(init_temp_factor * cur_val, 1e-12)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.9995, min_temp)
        # choose move type: more target
        p = rng.random()
        if p < 0.4:
            # targeted move
            conv = _conv_fft(cur, cur)
            max_idx = int(np.argmax(conv))
            new = _targeted_move(cur, max_idx, rng)
            if new is None:
                continue
        elif p < 0.7:
            # swap
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        else:
            # redistrib subset
            k = rng.integers(2, 10)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 20000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")
    return best_seq, best_val

"""
Search for a non‑negative step‑function that minimises

    2·n·max(conv(seq,seq)) / (sum(seq)²)

within a 1000 s time budget.  The algorithm consists of:

1.  A large, diverse seed pool (analytic windows, random Dirichlet draws,
    low‑frequency cosine curves, mixtures etc.).
2.  Refinement of the best seeds by a smooth‑max L‑BFGS optimisation
    (log‑softmax parametrisation, progressively increasing the
    temperature β of the log‑sum‑exp surrogate).
3.  A long mirror‑descent (exponentiated‑gradient) sub‑gradient optimisation
    on the exact max‑convolution objective.
4.  Simulated‑annealing polishing that also applies a *targeted* move:
    mass is taken from the two coefficients that most contribute to the
    current maximal convolution lag and redistributed elsewhere.  

The final sequence is stored in the global ``height_sequence_1`` variable.
"""

import numpy as np

# Optional SciPy import (used for L‑BFGS)

# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    v = v - np.max(v)
    e = np.exp(v)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic shape families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random families & ordering helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)          # asc
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve by summing a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_gaussian(n, sigma_frac=0.40))

    # Power‑law family
    for p in np.linspace(0.0, 4.0, 13):
        seeds.append(_power(n, p))

    # Beta family
    for a in np.linspace(0.3, 2.0, 5):
        for b in np.linspace(0.3, 4.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws with various concentration values
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw order
            seeds.append(_sorted_desc(raw))
            seeds.append(_sorted_asc(raw))
            seeds.append(_alternate_order(raw))

    # Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if compatible)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Objective (surrogate – exact max after ℓ₁ normalisation)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)) for a normalised sequence (sum=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Mirror‑descent sub‑gradient optimiser (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the true max‑conv objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.12  # initial step‑size factor

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise (positivity already holds)
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Targeted SA move (reduces the contribution to the current maximal lag)
# ----------------------------------------------------------------------
def _targeted_move(cur: np.ndarray, max_idx: int, rng: np.random.Generator) -> np.ndarray | None:
    """Take a small amount of mass from the two coordinates that contribute most
    to the convolution at ``max_idx`` and spread it elsewhere."""
    n = len(cur)
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    partner = max_idx - idx_i
    contrib = cur[idx_i] * cur[partner]
    if contrib.sum() == 0.0:
        return None
    prob = contrib / contrib.sum()
    i = rng.choice(idx_i, p=prob)
    j = max_idx - i
    # amount to shift – 5 % of the smaller of the two values (capped)
    eps = 0.05 * min(cur[i], cur[j])
    if eps <= 0.0:
        return None
    new = cur.copy()
    new[i] -= eps
    new[j] -= eps
    # redistribute the taken mass uniformly to two random locations
    k1, k2 = rng.choice(n, size=2, replace=False)
    new[k1] += eps / 2.0
    new[k2] += eps / 2.0
    # project back onto simplex (tiny numerical drift possible)
    new = np.maximum(new, 0.0)
    s = new.sum()
    if s > 0.0:
        new /= s
    else:
        new = _uniform(n)
    return new


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser (with targeted move)
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_200_000,
    init_temp_factor: float = 0.03,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """SA that mixes simple swaps, Dirichlet re‑assignments and a targeted
    peak‑reduction move."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        move_type = rng.random()

        if move_type < 0.35:
            # targeted move
            conv = _conv_fft(cur, cur)
            max_idx = int(np.argmax(conv))
            new = _targeted_move(cur, max_idx, rng)
            if new is None:
                continue
        elif move_type < 0.65:
            # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        else:
            # redistribute a random subset using a Dirichlet draw
            k = rng.integers(2, 10)
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 30_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (uses SciPy if available)
# ----------------------------------------------------------------------
class _TimeOutException(RuntimeError):
    pass


def _grad_wrt_a(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """Gradient of Σ w_k·conv_k w.r.t. a (central n entries)."""
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    n = len(a)
    return 2.0 * full[n - 1:n - 1 + n]


def _lbfgs_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    total_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Smooth‑max L‑BFGS on the log‑softmax parametrisation.
    A geometric schedule of β (temperature) values is used.
    """
    if not _SCIPY_AVAILABLE:
        # fall back to the start vector unchanged
        return start_vec.copy(), _objective(start_vec)

    best_seq = start_vec.copy()
    best_val = _objective(best_seq)

    # start from logits (add tiny eps for numerical stability)
    v = np.log(start_vec + 1e-30)

    # geometric schedule for β – from a modest temperature up to a large one
    beta_vals = np.geomspace(5.0, 2000.0, num=6)

    # allocate iterations per stage (roughly equal)
    iters_per_stage = max(1, total_iters // len(beta_vals))

    for stage, beta in enumerate(beta_vals, 1):
        if time.time() > deadline:
            break

        # --------------------------------------------------------------
        # objective & gradient for the current β
        # --------------------------------------------------------------
        def fun(v_local, beta_local=beta):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_term = np.exp(beta_local * (conv - shift))
            sum_exp = np.sum(exp_term)
            smooth = shift + np.log(sum_exp) / beta_local
            return 2.0 * n * smooth

        def grad(v_local, beta_local=beta):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_term = np.exp(beta_local * (conv - shift))
            sum_exp = np.sum(exp_term)
            if sum_exp == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = exp_term / sum_exp
            grad_a = _grad_wrt_a(a, w)
            dot = np.dot(grad_a, a)
            return a * (grad_a - dot)

        # --------------------------------------------------------------
        # callback for time‑out handling
        # --------------------------------------------------------------
        def callback(xk):
            if time.time() > deadline:
                raise _TimeOutException()

        # --------------------------------------------------------------
        # run L‑BFGS for this stage
        # --------------------------------------------------------------
        try:
            res = minimize(
                fun=fun,
                x0=v,
                jac=grad,
                method="L-BFGS-B",
                callback=callback,
                options={"maxiter": iters_per_stage, "disp": False},
            )
            v = res.x  # warm‑start next stage
        except _TimeOutException:
            break
        except Exception:  # pragma: no cover
            # any other numerical issue – abort further L‑BFGS
            break

        # evaluate exact objective for the current solution
        a_opt = _softmax(v)
        val = _objective(a_opt)
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            print(f"[LBFGS] stage {stage}/{len(beta_vals)} β={beta:.1f} new best {val:.6f}")

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats and stores it
    in the global name ``height_sequence_1`` for possible reuse.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # reserve a couple of seconds for the final sanity check
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Length choice – more points give a lower theoretical bound.
    # 5000 is a good compromise between expressive power and speed.
    # --------------------------------------------------------------
    n = 5000

    # --------------------------------------------------------------
    # Previous best (if any) – resample to current length if needed.
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape[0] != n:
                # linear interpolation to the new length
                x_old = np.linspace(0, 1, prev_arr.shape[0])
                x_new = np.linspace(0, 1, n)
                prev_arr = np.interp(x_new, x_old, prev_arr)
            prev_arr = np.maximum(prev_arr, 0.0)
            if prev_arr.sum() > 0:
                prev_arr = prev_arr / prev_arr.sum()
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # --------------------------------------------------------------
    # Stage 1 – seed generation & quick surrogate evaluation
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 6
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – L‑BFGS smooth‑max refinement (if SciPy is available)
    # --------------------------------------------------------------
    if time.time() < deadline and _SCIPY_AVAILABLE:
        print("\n--- L‑BFGS refinement ---")
        for idx, seed_vec in enumerate(best_seeds, 1):
            if time.time() > deadline:
                break
            seq_l, val_l = _lbfgs_optimize(seed_vec, n, deadline, total_iters=2000)
            if val_l < global_best_val:
                global_best_seq = seq_l.copy()
                global_best_val = val_l
                print(f"[LBFGS] new global best after seed {idx} = {val_l:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – Mirror‑descent sub‑gradient optimisation
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # estimate a modest number of iterations (≈250 it/s on typical hardware)
        mirror_iters = int(min(2_000_000, remaining * 250))
        print("\n--- Mirror‑descent optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=mirror_iters
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – Simulated‑annealing polishing (with targeted move)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sa_iters = int(min(2_000_000, remaining * 250))
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline, max_iters=sa_iters
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # --------------------------------------------------------------
    # Final sanity check with the official evaluator (optional)
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
