"""
Optimise a non‑negative step function for minimise
``evaluate_sequence`` (i.e. minimise 2·n·max(conv(a,a)) with sum(a)=1).

The search combines several complementary ideas:

1. **Analytic families** – a wide set of classic windows (uniform,
   Hann, Hamming, Blackman, Gaussian, arcsine, sinusoid, β‑pdfs).  All
   lengths are normalised and evaluated.

2. **Systematic β‑mixture grid** – a dense grid of two‑component β‑mixture
   shapes (α,β ∈ {0.2,0.5,1.0,2.0,5.0}, weight w ∈ {0,0.25,0.5,0.75,1}).
   This explores a very rich family of curves with only a few parameters
   and quickly finds a good shape, often better than random sampling.

3. **Smooth Adam optimisation** – from the best candidates a
   log‑soft‑max parametrisation is optimised using Adam on a
   log‑sum‑exp approximation of the max.  A high temperature schedule
   (β from 10 to 2000) yields a very sharp approximation of the true
   objective.

4. **Optional L‑BFGS‑B refinement** (if SciPy is available) on the same
   smooth objective.

5. **Aggressive KL‑mirror‑descent polishing** – a sub‑gradient mirror
   descent that aggregates *all* near‑maximal convolution entries and
   gradually tightens the tolerance.

All phases respect a global 1000 s budget and periodically report the
current best bound.  The final sequence is returned as a plain list of
floats and stored in the global ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np
from math import sqrt, pi, log, exp
from typing import List

# optional SciPy for L‑BFGS‑B
try:
    from scipy.optimize import minimize
except Exception:  # pragma: no cover
    minimize = None


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to evaluate_sequence for a *normalised* sequence (sum == 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = a0 - a1 * np.cos(2 * pi * i / (n - 1)) + a2 * np.cos(4 * pi * i / (n - 1))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _sin_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(pi * i / n)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta‑pdf (shifted by 0.5 to avoid the ends)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


# ----------------------------------------------------------------------
# Random generation utilities
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet‑like vector with concentration α."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


def _random_beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.uniform(0.1, 5.0)
    b = rng.uniform(0.1, 5.0)
    return _beta_shape(n, a, b)


def _random_mixture_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two independent β‑pdfs with a random weight."""
    w = rng.random()
    a1, b1 = rng.uniform(0.1, 5.0), rng.uniform(0.1, 5.0)
    a2, b2 = rng.uniform(0.1, 5.0), rng.uniform(0.1, 5.0)
    s1 = _beta_shape(n, a1, b1)
    s2 = _beta_shape(n, a2, b2)
    mix = w * s1 + (1.0 - w) * s2
    mix /= mix.sum()
    return mix


def _beta_mixture_shape(
    n: int,
    a1: float,
    b1: float,
    a2: float,
    b2: float,
    w: float,
) -> np.ndarray:
    """Mixture w·Beta(a1,b1) + (1‑w)·Beta(a2,b2).  Returns a normalised PMF."""
    s1 = _beta_shape(n, a1, b1)
    s2 = _beta_shape(n, a2, b2)
    mix = w * s1 + (1.0 - w) * s2
    sm = mix.sum()
    if sm == 0.0:
        return _uniform(n)
    return mix / sm


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 10.0,
    beta_end: float = 2000.0,
    max_iter: int = 5_000_000,
    lr0: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth (log‑sum‑exp) approximation of max(conv(a,a)).
    Uses Adam on the log‑softmax (logits) parametrisation.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        a = np.exp(logits)
        a = a / a.sum()

        conv = _conv_fft(a, a)

        # linear temperature schedule
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        # smooth‑max via log‑sum‑exp
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()

        # gradient w.r.t. a
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1 : n - 1 + n]

        # back‑prop through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional exact evaluation
        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

            if best_val < 1.45:   # early exit if already very good
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS refinement (smooth max with large β)
# ----------------------------------------------------------------------
def _lbfgs_refine(start_seq: np.ndarray, deadline: float, beta: float = 2000.0) -> tuple[np.ndarray, float]:
    if minimize is None:
        return start_seq, _objective(start_seq)

    n = len(start_seq)
    eps = 1e-30

    def fun_and_grad(logits: np.ndarray):
        a = np.exp(logits)
        a = a / a.sum()
        conv = _conv_fft(a, a)
        c = np.max(conv)
        exps = np.exp(beta * (conv - c))
        lse = c + (1.0 / beta) * np.log(exps.sum())
        value = 2.0 * n * lse

        p = exps / exps.sum()
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1 : n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)
        return value, grad_logits.astype(np.float64)

    x0 = np.log(start_seq + eps)

    class TimeOut(RuntimeError):
        pass

    def callback(xk):
        if time.time() >= deadline:
            raise TimeOut()

    try:
        res = minimize(
            fun=lambda x: fun_and_grad(x)[0],
            x0=x0,
            jac=lambda x: fun_and_grad(x)[1],
            method="L-BFGS-B",
            options={"maxiter": 200, "disp": False},
            callback=callback,
        )
    except TimeOut:
        best_logits = x0
    else:
        best_logits = res.x

    best_a = np.exp(best_logits)
    best_a /= best_a.sum()
    best_val = _objective(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive aggregated mirror‑descent polishing
# ----------------------------------------------------------------------
def _aggregated_mirror_descent(
    start_seq: np.ndarray,
    deadline: float,
    eps_rel_start: float = 0.01,
    eps_rel_end: float = 1e-6,
    lr_factor: float = 1.0,
) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    t0 = time.time()
    it = 0
    eps = 1e-30
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        elapsed = time.time() - t0
        total = deadline - t0
        frac = elapsed / total if total > 0 else 0.0
        eps_rel = eps_rel_start * (eps_rel_end / eps_rel_start) ** frac
        delta = eps_rel * max_val

        active_idx = np.where(conv >= max_val - delta)[0]
        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1 : n - 1 + n]

        lr = lr_factor / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)  # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[AggMD] it={it:7d} n={n:5d} best={val:.6f}")

            if time.time() > deadline - 0.5:
                break

    return best_seq, best_val

# Systematic β‑mixture grid search
# ----------------------------------------------------------------------
def _grid_beta_mixture_search(
    lengths: List[int],
    alphas: List[float],
    betas: List[float],
    ws: List[float],
    best_seq: np.ndarray,
    best_val: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Exhaustively evaluate a modest grid of two‑component beta mixtures.
    The grid is tiny (5⁴·5 ≈ 3 k points per length) and therefore cheap.
    """
    for n in lengths:
        for a1 in alphas:
            for b1 in betas:
                for a2 in alphas:
                    for b2 in betas:
                        for w in ws:
                            if time.time() >= deadline:
                                return best_seq, best_val
                            seq = _beta_mixture_shape(n, a1, b1, a2, b2, w)
                            val = _objective(seq)
                            if val < best_val:
                                best_val = val
                                best_seq = seq.copy()
                                print(
                                    f"[GridBeta] n={n:5d} a1={a1:.2f} b1={b1:.2f} "
                                    f"a2={a2:.2f} b2={b2:.2f} w={w:.2f} val={val:.6f}"
                                )
    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective, respecting a 1000 s budget.
    Returns the best sequence found as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Re‑use a previously computed solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Problem sizes – a selection of FFT‑friendly lengths
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000, 8000, 10000, 12000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic families
    # ------------------------------------------------------------------
    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.25),
        _arcsine_shape,
        _sin_shape,
        lambda n: _beta_shape(n, 0.5, 0.5),
        lambda n: _beta_shape(n, 2.0, 2.0),
        lambda n: _beta_shape(n, 0.5, 2.0),
        lambda n: _beta_shape(n, 2.0, 0.5),
    )

    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

        # reuse stored solution of matching length if it exists
        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – systematic β‑mixture grid search
    # ------------------------------------------------------------------
    grid_deadline = time.time() + (deadline - time.time()) * 0.20
    alphas_grid = [0.2, 0.5, 1.0, 2.0, 5.0]
    betas_grid = alphas_grid[:]          # same set
    ws_grid = [0.0, 0.25, 0.5, 0.75, 1.0]
    best_seq, best_val = _grid_beta_mixture_search(
        lengths=lengths,
        alphas=alphas_grid,
        betas=betas_grid,
        ws=ws_grid,
        best_seq=best_seq,
        best_val=best_val,
        deadline=grid_deadline,
    )

    # ------------------------------------------------------------------
    # Phase 3 – random Dirichlet exploration (lightweight)
    # ------------------------------------------------------------------
    alphas_dir = [0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]
    rand_deadline = time.time() + (deadline - time.time()) * 0.15
    while time.time() < rand_deadline:
        n = int(rng.choice(lengths))
        alpha = float(rng.choice(alphas_dir))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd‑Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        seq = _random_beta_shape(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd‑Beta] n={n:5d} val={val:.6f}")

        seq = _random_mixture_beta(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd‑Mix] n={n:5d} val={val:.6f}")

        if best_val < 1.45:
            break

    # ------------------------------------------------------------------
    # Phase 4 – Smoothed Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        candidates = [best_seq]

        # two fresh Dirichlet restarts of the same length
        top_len = len(best_seq)
        for _ in range(2):
            cand = _random_dirichlet(top_len, rng, alpha=rng.choice(alphas_dir))
            candidates.append(cand)

        # one larger‑length restart (if possible)
        larger_lengths = [L for L in lengths if L > top_len]
        if larger_lengths:
            L_big = int(rng.choice(larger_lengths))
            cand_big = _random_dirichlet(L_big, rng, alpha=1.0)
            candidates.append(cand_big)

        adam_deadline = time.time() + (deadline - time.time()) * 0.25
        for idx, cand in enumerate(candidates):
            seq_adam, val_adam = _smooth_adam_optimize(
                start_seq=cand,
                deadline=adam_deadline,
                beta_start=10.0,
                beta_end=2000.0,
                max_iter=5_000_000,
                lr0=0.02,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = seq_adam
                print(f"[Adam‑Refine {idx}] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – optional L‑BFGS‑B refinement
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0 and minimize is not None:
            lbfgs_deadline = time.time() + remaining * 0.15
            seq_lbfgs, val_lbfgs = _lbfgs_refine(best_seq, lbfgs_deadline, beta=2000.0)
            if val_lbfgs < best_val:
                best_val = val_lbfgs
                best_seq = seq_lbfgs
                print(f"[L‑BFGS] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – aggressive aggregated mirror‑descent polishing
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _aggregated_mirror_descent(
            start_seq=best_seq,
            deadline=deadline,
            eps_rel_start=0.01,
            eps_rel_end=1e-6,
            lr_factor=1.0,
        )
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Agg‑MD] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 7 – a last short Adam polish (sharper temperature)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 1.0:
            final_adam_seq, final_adam_val = _smooth_adam_optimize(
                start_seq=best_seq,
                deadline=deadline,
                beta_start=50.0,
                beta_end=3000.0,
                max_iter=3_000_000,
                lr0=0.015,
            )
            if final_adam_val < best_val:
                best_val = final_adam_val
                best_seq = final_adam_seq
                print(f"[Final‑Adam] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will evaluate again anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )

    # expose the result for possible reuse in a later run
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
