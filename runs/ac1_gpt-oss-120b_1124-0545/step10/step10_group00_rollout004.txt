def _mirror_descent(start_seq, n, deadline, max_iters=1_500_000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    lr0 = 0.12
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[Mirror] it={it} best={cur_val:.6f}")
        # subgradient: compute g
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s
    return best_seq, best_val

"""
Propose a step‑function that minimizes

    2·n·max(conv(seq,seq)) / (sum(seq)²)

using a three‑stage pipeline:

1.  A diverse seed pool (analytic families + random Dirichlet draws) is
    generated and the best few seeds are kept using a fast FFT‑based
    surrogate (2·n·max(conv) with ∑a=1).

2.  Each seed is refined with a continuation of L‑BFGS‑B on a smoothed
    log‑sum‑exp surrogate of the max‑convolution.  The parameter β
    (temperature) is increased (5 → 20 → 100 → 300) so that the smooth
    objective converges to the true max.  The optimisation works in the
    log‑softmax (log‑probability) space, guaranteeing non‑negativity and
    ℓ₁‑normalisation without explicit constraints.

3.  The best candidate is further polished with an exponentiated‑gradient
    (mirror‑descent) sub‑gradient method on the exact max‑convolution,
    followed by a cheap permutation‑hill‑climb to exploit the fact that
    the ordering of the coefficients influences the convolution peak.

All stages respect a total time budget (default 1000 s).  The final
sequence is stored in the global name ``height_sequence_1`` and returned
as a plain list of Python floats.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """Surrogate objective 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised seq (∑seq=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)          # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve from a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,

    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_gaussian(n, sigma_frac=0.40))

    # power‑law family
    for p in np.linspace(0.0, 4.0, 13):
        seeds.append(_power(n, p))

    # beta family
    for a in np.linspace(0.3, 2.0, 5):
        for b in np.linspace(0.3, 4.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (different concentration parameters)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw random order
            seeds.append(_sorted_desc(raw))       # monotone decreasing
            seeds.append(_sorted_asc(raw))        # monotone increasing
            seeds.append(_alternate_order(raw))   # alternating pattern

    # Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# L‑BFGS + smooth‑max optimisation
# ----------------------------------------------------------------------
def _lbfgs_objective(v: np.ndarray, beta: float, n: int):
    """
    Returns (smooth_max, grad_v) where the smooth max approximates
    max(conv(a,a)) with a = softmax(v).  The gradient is taken w.r.t. v.
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    # smooth approximation of max
    smooth_max = shift + np.log(w_sum) / beta

    # probability distribution over convolution entries (log‑sum‑exp gradient)
    w = exp_term / w_sum

    # gradient w.r.t. a (linear convolution of reversed a with w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)              # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]    # central n entries

    # chain rule: ∂/∂v  (softmax)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return smooth_max, grad_v


def _lbfgs_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule: list[float],
                    max_iter_per_stage: int = 200) -> tuple[np.ndarray, float]:
    """
    Optimise a start vector using a continuation of L‑BFGS‑B on the
    smooth‑max surrogate.  Returns the best probability vector found
    and its surrogate value (2·n·max(conv)).
    """
    v = np.log(start_vec + 1e-30)  # logits
    best_a = _softmax(v)
    best_val = _objective(best_a)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # Allow a few hundred L‑BFGS iterations; scale with remaining time
        iters = min(max_iter_per_stage, max(10, int(remaining * 30)))
        res = scipy.optimize.minimize(
            _lbfgs_objective,
            v,
            args=(beta, n),
            jac=True,
            method='L-BFGS-B',
            options={'maxiter': iters, 'disp': False}
        )
        v = res.x
        a = _softmax(v)
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L‑BFGS] beta={beta:.1f} new surrogate = {val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent sub‑gradient optimisation (exact max‑conv)
# ----------------------------------------------------------------------
def _mirror_descent(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) descent directly on the true max‑conv.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    lr0 = 0.12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise (positivity already holds)
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation hill‑climb (permutes the entries, preserving the multiset)
# ----------------------------------------------------------------------
def _permutation_hillclimb(seq: np.ndarray,
                           deadline: float,
                           max_attempts: int = 8000) -> np.ndarray:
    """
    Tries random permutations of the current multiset of coefficients and
    keeps any that improve the surrogate objective.
    """
    n = len(seq)
    best_seq = seq.copy()
    best_val = _objective(best_seq)
    rng = np.random.default_rng()
    attempts = 0

    while attempts < max_attempts and time.time() < deadline:
        perm = rng.permutation(n)
        cand = best_seq[perm]
        cand_val = _objective(cand)
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand
            # occasional progress report
            if attempts % 2000 == 0:
                print(f"[Permute] attempt {attempts} new surrogate {cand_val:.6f}")
        attempts += 1

    return best_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats and stores it
    in the global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length of the candidate.  5000 provides a good trade‑off.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Retrieve previous solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a diverse seed pool and pick the most promising
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – L‑BFGS continuation on smoothed max‑conv
    # ------------------------------------------------------------------
    beta_schedule = [5.0, 20.0, 100.0, 300.0]
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        print(f"\n--- L‑BFGS optimisation run {idx}/{len(best_seeds)} ---")
        seq_opt, val_opt = _lbfgs_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule,
            max_iter_per_stage=250  # a moderate number of L‑BFGS iterations per β
        )
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[overall‑L‑BFGS] new global best surrogate = {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Mirror‑descent sub‑gradient refinement (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent refinement ---")
        seq_mirror, val_mirror = _mirror_descent(
            global_best_seq, n, deadline, max_iters=1_200_000
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best surrogate = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Permutation hill‑climb (exploits ordering freedom)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation hill‑climb polishing ---")
        seq_perm = _permutation_hillclimb(
            global_best_seq, deadline, max_attempts=8000
        )
        val_perm = _objective(seq_perm)
        if val_perm < global_best_val:
            global_best_seq = seq_perm.copy()
            global_best_val = val_perm
            print(f"[overall‑Permute] new global best surrogate = {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

This implementation expands on the previous pipeline by adding an explicit L‑BFGS optimisation on a smoothed max‑convolution surrogate (with a continuation in the temperature β), followed by a longer mirror‑descent phase and a permutation‑based polishing step.  
All helper functions are defined at the top level, there are no filesystem or network calls, and the total runtime respects the 1000‑second budget.
