"""
Propose a non‑negative step‑function (probability vector) that minimizes
evaluate_sequence().  The algorithm uses a richer set of analytic families,
a finer sampling of the Beta(α,β) family, and a longer, more aggressive
Adam+sub‑gradient+simulated‑annealing refinement.  It works with a length
of 6000 (a good trade‑off between quality and speed) and respects the
1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for ℓ₁‑normalised sequences (sum == 1):
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max (always on the simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangle (tent) distribution."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    """Flat top with linear ramps."""
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """One‑sided geometric decay, r∈(0,1)."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    """sin⁽ᵞ⁾ window."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on a discrete grid."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Truncated Gaussian."""
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey (cosine‑tapered) window."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw (parameter conc > 0)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential draw with mean 1."""
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 200_000,
                   beta_max: float = 1500.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max‑convolution.
    start_vec must already be ℓ₁‑normalised.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)           # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.003

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")

        # increase temperature (smoothness -> sharpness)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip gradient norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 2_000_000,
                      init_lr: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active index (the whole sum of overlapping terms)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step size
        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing that redistributes mass among a random subset.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # number of coordinates to reshuffle
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a heterogeneous pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # trapezoidal sweep (finer)
    for wf in np.linspace(0.05, 0.45, 9):
        seeds.append(_trapezoidal(n, width_fraction=wf))

    # sine windows – extended exponent range
    for gamma in np.linspace(0.0, 3.5, 15):
        seeds.append(_sine_window(n, gamma=gamma))

    # Tukey windows – many α values
    for alpha in np.linspace(0.0, 0.98, 20):
        seeds.append(_tukey_window(n, alpha=alpha))

    # geometric families – include ratios extremely close to 1
    r_vals = np.concatenate([
        np.linspace(0.2, 0.97, 12),
        np.linspace(0.97, 0.99995, 12)
    ])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # Power‑law family (dense sampling)
    for p in np.linspace(0.0, 5.0, 31):
        seeds.append(_power(n, p))

    # Beta shape – dense grid around (0.5,0.5) and broader
    for a in np.linspace(0.2, 2.5, 13):
        for b in np.linspace(0.2, 2.5, 13):
            seeds.append(_beta_shape(n, a, b))

    # Random draws of Beta parameters (continuous exploration)
    for _ in range(30):
        a = rng.uniform(0.05, 2.8)
        b = rng.uniform(0.05, 2.8)
        seeds.append(_beta_shape(n, a, b))

    # Gaussian windows (different sigmas)
    for sigma_frac in (0.10, 0.15, 0.20, 0.25, 0.30, 0.35):
        seeds.append(_gaussian(n, sigma_frac))

    # Random Dirichlet draws – wide concentration range
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # Mixtures of two power laws (extra flexibility)
    for _ in range(15):
        p1 = rng.uniform(0.0, 5.0)
        p2 = rng.uniform(0.0, 5.0)
        w = rng.random()
        mix = w * _power(n, p1) + (1.0 - w) * _power(n, p2)
        mix = mix / mix.sum()
        seeds.append(mix)

    # Previous solution (interpolated to current length if necessary)
    if previous is not None:
        try:
            prev = np.asarray(previous, dtype=np.float64)
            if prev.shape[0] != n:
                # Linear interpolation to new size
                x_old = np.linspace(0.0, 1.0, prev.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                prev = np.interp(x_new, x_old, prev)
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s > 0:
                seeds.append(prev / s)
        except Exception:
            pass

    return seeds

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # a small safety margin

    # try to reuse a previous solution (if any)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length choice – 6000 gives a good balance between quality and speed.
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Stage 1 – seed generation & quick screening
    # ------------------------------------------------------------------
    print(f"\n--- Collecting seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])

    top_k = min(12, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation on the most promising seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds[:5]):   # up to five Adam runs
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate a generous but bounded number of iterations
        max_iters = int(remaining * 2000)
        max_iters = max(80_000, min(max_iters, 500_000))
        print(f"\n--- Adam optimisation run {idx+1}/{len(best_seeds[:5])} "
              f"max_iters={max_iters} ---")
        seq_adam, val_adam = _adam_optimize(
            seed_vec, n, deadline,
            max_iters=max_iters,
            beta_max=1800.0
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Projected sub‑gradient refinement (bulk of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(remaining * 1200)
        max_iters = max(200_000, min(max_iters, 3_000_000))
        print(f"\n--- Sub‑gradient refinement (max_iters={max_iters}) ---")
        seq_proj, val_proj = _subgradient_proj(
            global_best_seq, n, deadline,
            max_iters=max_iters,
            init_lr=0.18
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[overall‑SubProj] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing polishing (final touch)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(remaining * 1500)
        max_iters = max(150_000, min(max_iters, 1_200_000))
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=max_iters,
            init_temp_factor=0.05
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the official evaluator
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # store for possible later reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
