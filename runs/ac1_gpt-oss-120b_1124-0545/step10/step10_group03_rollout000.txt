"""
genetic_convolution_optimize.py

Search for a non‑negative step‑function that minimizes
    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm proceeds as follows:

1. Choose a length `n` (default 5000) – large enough to give many
   degrees of freedom but still cheap to FFT‑convolve.
2. Create a diverse initial population of ℓ₁‑normalised candidates.
   The pool contains analytic families (uniform, triangular, cosine‑U,
   beta‑ends), random Dirichlet draws with many concentration parameters,
   and sparse supports.
3. Run a simple steady‑state genetic algorithm:
   * tournament selection,
   * convex crossover,
   * Gaussian mutation,
   * projection onto the probability simplex.
   The best individual seen throughout the run is kept.
4. When the time budget is almost exhausted, polish the best individual
   with a few hundred thousand steps of Adam optimisation on a smooth‑max
   surrogate of the max‑convolution (the same surrogate used in the
   reference implementation).
5. Return the final sequence as a plain Python list and store it in the
   global variable `height_sequence_1` for possible reuse.

The whole routine respects the supplied `budget_s` (default 1000 s) and
stops just before the deadline, guaranteeing a valid answer.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, N: int) -> np.ndarray:
    """Full linear convolution via real FFT (pre‑allocated size N)."""
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)
    # Trim to linear length
    L = len(x) + len(y) - 1
    return conv[:L]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_u(n: int, power: float = 2.0) -> np.ndarray:
    """U‑shaped window: |cos|^power, high at the ends, low in the centre."""
    i = np.arange(n, dtype=np.float64)
    theta = np.pi * i / (n - 1)
    a = np.abs(np.cos(theta)) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_ends(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution with shape parameters < 1 gives mass at the ends."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet draw with concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random sparse vector with 3‑30 non‑zero entries."""
    k = rng.integers(3, 31)  # inclusive lower, exclusive upper
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray, fft_len: int) -> float:
    """
    Exact objective value (scale‑invariant):
        2·n·max(conv(seq,seq)) / (sum(seq)^2)
    ``seq`` is assumed to be non‑negative.
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq, fft_len)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray, n: int, fft_len: int,
                   deadline: float, max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    Returns the best sequence (ℓ₁‑normalised) and its exact objective.
    """
    eps = 1e-30
    # we optimise in the soft‑max (log‑its) space so we stay on the simplex
    v = np.log(start_vec + eps)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0          # start temperature for smooth‑max
    beta_mul = 1.001
    beta_max = 250.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)               # current candidate (ℓ₁‑normalised)
        cur_val = _objective(a, fft_len)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        # increase temperature slowly – we approach the true max
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # --- smooth max gradient ---------------------------------
        conv = _conv_fft(a, a, fft_len)          # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                               # softmax over conv entries

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w, fft_len)     # length 3n‑2 (same FFT len)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # transform gradient to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip gradient norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    if best_seq is None:
        best_seq = _softmax(v)
        best_val = _objective(best_seq, fft_len)

    return best_seq, best_val

# Genetic algorithm core
# ----------------------------------------------------------------------


def _initial_population(n: int, rng: np.random.Generator,
                        fft_len: int, deadline: float,
                        pop_size: int = 30) -> list[np.ndarray]:
    """
    Generate a diverse initial population of size `pop_size`.
    Each member is ℓ₁‑normalised.
    """
    candidates = []

    def consider(seq: np.ndarray):
        """Insert seq into candidate list respecting pop_size."""
        if seq.ndim != 1 or seq.shape[0] != n:
            return
        val = _objective(seq, fft_len)
        if len(candidates) < pop_size:
            candidates.append((val, seq))
            candidates.sort(key=lambda x: x[0])
        elif val < candidates[-1][0]:
            candidates[-1] = (val, seq)
            candidates.sort(key=lambda x: x[0])

    # deterministic families
    consider(_uniform(n))
    consider(_triangular(n))
    consider(_cosine_u(n, power=2.0))
    consider(_cosine_u(n, power=4.0))
    consider(_cosine_u(n, power=6.0))

    # beta‑ends with a few extreme parameters
    for a in (0.2, 0.3, 0.5):
        for b in (0.2, 0.3, 0.5):
            consider(_beta_ends(n, alpha=a, beta=b))

    # random Dirichlet draws with varied concentration
    concs = [0.001, 0.005, 0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 30 if conc < 0.01 else 20
        for _ in range(draws):
            if time.time() > deadline:
                break
            consider(_dirichlet_seed(n, conc, rng))

    # sparse seeds
    for _ in range(40):
        if time.time() > deadline:
            break
        consider(_sparse_seed(n, rng))

    # keep only the sequences, sorted
    return [seq for _, seq in sorted(candidates, key=lambda x: x[0])]


def _genetic_search(initial_pop: list[np.ndarray],
                    n: int, fft_len: int,
                    deadline: float,
                    max_iters: int = 800_000) -> np.ndarray:
    """
    Simple steady‑state GA that runs until `deadline` or `max_iters`.
    Returns the best sequence ever seen.
    """
    rng = np.random.default_rng()
    population = initial_pop.copy()
    pop_size = len(population)

    # store objectives alongside individuals to avoid recomputation
    pop_vals = [_objective(ind, fft_len) for ind in population]
    best_idx = int(np.argmin(pop_vals))
    best_seq = population[best_idx].copy()
    best_val = pop_vals[best_idx]

    # parameters
    tournament_k = 5
    mutation_scale = 0.01   # Gaussian std. as a fraction of 1 (since sum=1)
    mutation_prob = 0.2

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # --- tournament selection for two parents -----------------
        def select_parent():
            indices = rng.choice(pop_size, size=tournament_k, replace=False)
            chosen = min(indices, key=lambda idx: pop_vals[idx])
            return population[chosen]

        p1 = select_parent()
        p2 = select_parent()

        # --- convex crossover ------------------------------------
        lam = rng.random()
        child = lam * p1 + (1.0 - lam) * p2

        # --- mutation --------------------------------------------
        if rng.random() < mutation_prob:
            noise = rng.normal(scale=mutation_scale, size=n)
            child += noise

        # enforce non‑negativity and ℓ₁ normalisation
        child = np.maximum(child, 0.0)
        if child.sum() == 0:
            child = _uniform(n)
        else:
            child = child / child.sum()

        # evaluate child
        child_val = _objective(child, fft_len)

        # replacement if better than worst in pop
        worst_idx = int(np.argmax(pop_vals))
        if child_val < pop_vals[worst_idx]:
            population[worst_idx] = child
            pop_vals[worst_idx] = child_val

            # update global best
            if child_val < best_val:
                best_val = child_val
                best_seq = child.copy()
                if it % 5000 == 0:
                    print(f"[GA] it={it:7d} best={best_val:.6f}")

    print(f"[GA] finished after {it} iterations – best value {best_val:.6f}")
    return best_seq

    )

    return best_seq.tolist()
