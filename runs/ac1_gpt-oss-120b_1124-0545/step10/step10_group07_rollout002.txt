def _subgradient_adam(start_seq, n, deadline, max_iters=2_000_000):

    # start with a = start_seq (sum=1).

    a = start_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8

    best_seq = a.copy()
    best_val = _objective(a)
    iteration=0

    while time.time() < deadline and iteration < max_iters:
        iteration +=1

        # compute convolution
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        # identify active indices within tolerance
        tol = max(0.001*max_conv, 1e-12)  # dynamic tolerance
        active = np.where(conv >= max_conv - tol)[0]
        # gradient for each active index similarly combine equally
        # w is weight vector of size 2n-1: each active idx weight=1/|active|
        w = np.zeros(2*n-1, dtype=np.float64)
        w[active] = 1.0 / active.size
        # gradient = 2 * rev_a * w (select appropriate segment)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        g = 2.0 * full[n-1:n-1+n]

        # Adam update for log domain:
        # Use multiplicative update: a_i = a_i * exp(-lr * g_i). We'll incorporate Adam's adaptive learning.
        # Compute g_hat for Adam.
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        # learning rate schedule: decreasing sqrt / sqrt(t) maybe.
        lr = 0.5 / np.sqrt(iteration)

"""
Hybrid non‑convex optimizer for the self‑convolution L∞‑norm.

The algorithm consists of five stages:

1. **Seed generation** – a large curated pool of analytic probability
   vectors (uniform, windows, beta, geometric, exponential, etc.) is
   built.  All vectors are normalised to sum = 1 and the best few seeds
   are kept.

2. **Sub‑gradient Adam descent** – we minimise the exact objective  
   `2·n·max(conv(a,a))` (the denominator is constant after normalisation).
   The non‑smooth max is replaced by the sub‑gradient obtained from the
   set of (almost) active convolution entries.  An Adam optimiser is
   applied in the log‑domain (multiplicative updates) together with a
   projection onto the probability simplex.

3. **Fine‑grained simulated‑annealing** – a Metropolis–Hastings walk
   with several move types (mass transfer, swap, global jitter) is run
   on the remaining time budget to escape possible shallow minima.

4. **Result handling** – the best sequence seen throughout the run is
   stored in the global name ``height_sequence_1`` and returned as a
   plain list of Python floats.

The whole search respects a 1000 s budget; intermediate progress and
objective values are printed for monitoring.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – FFT based convolution (real‑valued)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (full) using real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return a *probability* vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density discretised on mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_window(n: int, lam: float) -> np.ndarray:
    """Exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-lam * np.abs(i - centre))
    return a / a.sum()


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `alpha`."""
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()

    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, alpha, rng))

    # random soft‑max draws
    for _ in range(12):
        seeds.append(_random_softmax(n, rng))

    # reuse previous construction if provided
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final clean‑up – clip and renorm
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    return cleaned


# ----------------------------------------------------------------------
# Sub‑gradient Adam optimiser (log‑domain, simplex projection)
# ----------------------------------------------------------------------


def _adam_subgrad(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam on the exact non‑smooth objective using the active‑set sub‑gradient.
    Multiplicative updates are performed in log‑space, followed by a simplex
    projection.  The best sequence encountered is returned.
    """
    a = start_seq.copy()
    eps = 1e-12

    # initialise Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ---- convolution and active set ----
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        # indices whose value is within 0.1 % of the maximum are treated as active
        tol = max(1e-12, 0.001 * max_conv)
        active = np.where(conv >= max_conv - tol)[0]

        # ---- sub‑gradient of the max ----
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[active] = 1.0 / active.size
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        g = 2.0 * full[n - 1:n - 1 + n]   # gradient of the (soft) max

        # ---- Adam update (log‑domain) ----
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # learning‑rate schedule (decreasing ~ 1/√t)
        lr = 0.6 / math.sqrt(it)

        delta = -lr * m_hat / (np.sqrt(v_hat) + eps)
        # keep the multiplicative step well‑behaved
        delta = np.clip(delta, -0.5, 0.5)

        # multiplicative update in log‑space
        a = a * np.exp(delta)
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = _uniform(n)
        else:
            a = a / a_sum

        # ---- occasional exact evaluation ----
        if it % 250 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

    # final check
    final_val = _objective(a)
    if final_val < best_val:
        best_val = final_val
        best_seq = a.copy()
        print(f"[Adam] final best={final_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (same spirit as the previous code)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Metropolis–Hastings walk with several move types."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # start temperature – a small fraction of the current objective
    T0 = max(1e-7, cur_val * 0.07)
    decay = 0.9994
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        rnd = rng.random()
        if rnd < 0.66:                 # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.random() * 0.18 * cur[i]
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        elif rnd < 0.84:               # swap
            i, j = rng.integers(0, n, size=2)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:                          # global jitter
            noise = rng.normal(scale=0.0013, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)

        new_seq = _project_simplex(new_seq)
        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:8d} best={new_val:.6f}")

        # occasional stronger jump from the current best
        if it % 2500 == 0:
            cand = best + rng.normal(scale=0.0035, size=n)
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand)
            cand_val = _objective(cand)
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
                print(f"[Anneal] (jump) it={it:8d} best={cand_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  Returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin (must return before the hard timeout)
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Length – a compromise between expressiveness and speed
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Retrieve a previously found construction, if any
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # --------------------------------------------------------------
    # 0) evaluate the previous construction (if available)
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    if previous is not None and previous.shape == (n,):
        val = _objective(previous)
        best_seq = previous.copy()
        best_val = val
        print(f"[prev] val={val:.6f}")

    # --------------------------------------------------------------
    # 1) Build a large pool of analytic seeds
    # --------------------------------------------------------------
    seeds = _make_extended_seeds(n, previous)
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    if seed_vals[0][1] < best_val:
        best_val = seed_vals[0][1]
        best_seq = seed_vals[0][0].copy()
        print(f"[seed] new best={best_val:.6f}")

    # --------------------------------------------------------------
    # 2) Sub‑gradient Adam descent (major part of the budget)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)

    time_now = time.time()
    time_left = deadline - time_now
    # allocate roughly 60 % of the remaining time to Adam
    adam_deadline = time_now + 0.60 * time_left
    print("[adam] starting sub‑gradient Adam optimisation")
    seq_adam, val_adam = _adam_subgrad(best_seq, n, adam_deadline,
                                      max_iters=500_000)
    if val_adam < best_val:
        best_val = val_adam
        best_seq = seq_adam.copy()
        print(f"[adam] improved best to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 3) Final simulated‑annealing fine‑tuning (remaining time)
    # --------------------------------------------------------------
    time_now = time.time()
    if time_now < deadline:
        print("[anneal] starting simulated‑annealing fine‑tuning")
        seq_ann, val_ann = _anneal_optimize(best_seq, n, deadline,
                                            max_iters=None)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = seq_ann.copy()
            print(f"[anneal] improved best to {val_ann:.6f}")

    # --------------------------------------------------------------
    # 4) Fallback (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
