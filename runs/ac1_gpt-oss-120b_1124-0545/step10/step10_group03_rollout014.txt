"""
Improved search for a low‑value step‑function.

The optimizer works in four stages, each respecting the global time budget:

1. **Diversified seeding** – a rich collection of analytical families (uniform,
   triangular, quadratic, sine‑squared, Hann, Kaiser, Blackman, Gaussian,
   Tukey, Beta, power‑law, sparse Dirichlet draws, mixtures).  The 𝑘 best
   candidates (by the exact objective) are retained.

2. **Long Adam optimisation** – each seed is refined with Adam on a
   log‑sum‑exp surrogate of the max‑convolution.  The temperature (smooth‑max
   parameter) is increased aggressively up to 𝛽≈500 so that the surrogate
   converges to the true ℓ∞‑norm.  The best refined sequence is kept.

3. **Projected sub‑gradient descent** – a Polyak step‑size scheme directly
   minimises the true max‑convolution on the probability simplex.
   Multiple restarts (small random perturbations of the current best) are
   performed while time permits.

4. **Simulated‑annealing polishing** – tiny Dirichlet re‑allocations on a
   randomly chosen subset of coordinates finish the search.

All vectors are kept ℓ₁‑normalised (sum = 1).  Because the official evaluator
is scale‑invariant, the returned list can be used directly.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, …)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used during the search.
    ``seq`` is assumed to be ℓ₁‑normalised (sum = 1).  The return value
    equals 2·n·max(conv(seq, seq)), which is the same quantity that
    ``evaluate_sequence`` returns for a normalised input.
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _quadratic(n: int) -> np.ndarray:
    """Symmetric quadratic “parabolic” window."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sin_squared(n: int) -> np.ndarray:
    """a_i = sin²(π·i/(n‑1)).  Zero at the ends, peak in the centre."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1)) ** 2
    return a / a.sum()


def _hann(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    return a / a.sum()


def _kaiser_window(n: int, beta: float = 5.0) -> np.ndarray:
    """Kaiser window with shape parameter β."""
    a = np.kaiser(n, beta)
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    """Blackman window (β≈0.16)."""
    a = np.blackman(n)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    k = rng.integers(3, 31)  # 3 … 30 non‑zero entries
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


def _mixture_uniform_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.random()
    g = _gaussian(n, sigma_frac=rng.uniform(0.05, 0.4))
    u = _uniform(n)
    a = lam * u + (1.0 - lam) * g
    return a / a.sum()


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 800_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    Parameters are the logits of a soft‑max (probability) representation.
    Returns (best_sequence, best_value) where best_value = exact objective.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 2.0                # start temperature (smooth‑max)
    beta_mul = 1.010               # geometric increase
    beta_max = 500.0               # final temperature – very close to true max

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # probability vector on the simplex
        a = _softmax(v)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")

        # raise temperature slowly
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # ---- gradient of the smooth max ---------------------------------
        conv = _conv_fft(a, a)                     # length 2n‑1
        shift = np.max(conv)                       # numerical stabiliser
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                                 # softmax over convolution entries

        # gradient w.r.t. a of Σ w_k·conv_k  (linear conv)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # transform to logits
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping for stability
        max_norm = 5.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses a Polyak step size based on the best value seen so far.
    Returns the best sequence encountered and its exact value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.2 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Very light simulated‑annealing that repeatedly redistributes mass among a
    small random subset of coordinates while respecting the simplex.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)                 # move 2 … 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 20000 == 0:
                    print(f"[SA]   it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             scale: float = 0.05) -> np.ndarray:
    """
    Small random perturbation that keeps the vector on the simplex.
    ``scale`` controls the typical magnitude of the move.
    """
    n = len(seq)
    # pick a random subset (≈10 % of the coordinates)
    k = max(2, int(0.1 * n))
    idx = rng.choice(n, size=k, replace=False)
    # draw a Dirichlet of size k and blend with the old mass
    total = seq[idx].sum()
    if total == 0.0:
        return seq.copy()
    new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
    new = seq.copy()
    # “mix” old and new allocations to keep changes modest
    new[idx] = (1.0 - scale) * seq[idx] + scale * new_alloc
    # re‑normalise to sum = 1 (projection is cheap thanks to simplex form)
    return _proj_simplex(new)


# ----------------------------------------------------------------------
# Seed generation utilities
# ----------------------------------------------------------------------


def _collect_diverse_seeds(n: int, rng: np.random.Generator,
                          previous: np.ndarray | None,
                          top_k: int = 20,
                          deadline: float = 0.0) -> list[np.ndarray]:
    """
    Generate a diverse pool of ℓ₁‑normalised candidates and keep the ``top_k``
    with the smallest exact objective value.
    """
    top_vals = []          # list of (value, seq)

    def try_insert(val, seq):
        """Maintain a sorted list of the best `top_k` entries."""
        if len(top_vals) < top_k:
            top_vals.append((val, seq))
            top_vals.sort(key=lambda x: x[0])
        elif val < top_vals[-1][0]:
            top_vals[-1] = (val, seq)
            top_vals.sort(key=lambda x: x[0])

    # Deterministic families (always worth keeping)
    families = [
        _uniform,
        _triangular,
        _quadratic,
        _sin_squared,
        _hann,
        lambda n: _kaiser_window(n, beta=4.0),
        lambda n: _kaiser_window(n, beta=8.0),
        _blackman_window,
        lambda n: _tukey_window(n, alpha=0.2),
        lambda n: _tukey_window(n, alpha=0.4),
        lambda n: _gaussian(n, sigma_frac=0.12),
        lambda n: _gaussian(n, sigma_frac=0.24),
    ]

    for fn in families:
        seq = fn(n)
        val = _objective(seq)
        try_insert(val, seq)

    # Geometric families
    for r in np.linspace(0.25, 0.95, 12):
        for fn in (_geom, _sym_geom):
            seq = fn(n, r)
            val = _objective(seq)
            try_insert(val, seq)

    # Power‑law family
    for p in np.linspace(0.0, 5.0, 30):
        seq = _power(n, p)
        val = _objective(seq)
        try_insert(val, seq)

    # Random Dirichlet draws – a wide range of concentration parameters
    concs = [0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 1.0, 3.0, 5.0]
    for conc in concs:
        draws = 120 if conc < 0.01 else 70
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = _dirichlet_seed(n, conc, rng)
            val = _objective(seq)
            try_insert(val, seq)

    # Sparse random supports
    for _ in range(400):
        if time.time() > deadline:
            break
        seq = _sparse_seed(n, rng)
        val = _objective(seq)
        try_insert(val, seq)

    # Random beta shapes
    for _ in range(400):
        if time.time() > deadline:
            break
        alpha = rng.uniform(0.1, 5.0)
        beta = rng.uniform(0.1, 5.0)
        seq = _beta_shape(n, alpha, beta)
        val = _objective(seq)
        try_insert(val, seq)

    # Random sine‑window exponents
    for _ in range(250):
        if time.time() > deadline:
            break
        gamma = rng.uniform(0.3, 5.0)
        seq = _sine_window(n, gamma=gamma)
        val = _objective(seq)
        try_insert(val, seq)

    # Random Gaussian windows
    for _ in range(250):
        if time.time() > deadline:
            break
        sigma_frac = rng.uniform(0.04, 0.4)
        seq = _gaussian(n, sigma_frac=sigma_frac)
        val = _objective(seq)
        try_insert(val, seq)

    # Random Tukey windows
    for _ in range(200):
        if time.time() > deadline:
            break
        alpha = rng.uniform(0.05, 0.9)
        seq = _tukey_window(n, alpha=alpha)
        val = _objective(seq)
        try_insert(val, seq)

    # Mixture of uniform & Gaussian
    for _ in range(200):
        if time.time() > deadline:
            break
        seq = _mixture_uniform_gaussian(n, rng)
        val = _objective(seq)
        try_insert(val, seq)

    # Previous solution (if any) – give it a second chance
    if previous is not None and previous.shape == (n,):
        seq = np.asarray(previous, dtype=np.float64)
        if seq.sum() > 0:
            seq = seq / seq.sum()
            val = _objective(seq)
            try_insert(val, seq)

    # Return the sequences sorted by increasing value
    return [seq for _, seq in top_vals]

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget.
    The best sequence found is stored in the global name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # --------------------------------------------------------------
    # Length – 4000 gives a good trade‑off between resolution and speed.
    # --------------------------------------------------------------
    n = 4000

    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # Give seed collection about 20 % of the total budget
    seed_deadline = min(deadline, start_time + budget_s * 0.20)
    seeds = _collect_diverse_seeds(
        n, rng, previous, top_k=20, deadline=seed_deadline
    )
    print(f"Collected {len(seeds)} promising seeds.")

    # --------------------------------------------------------------
    # Stage 1 – medium‑length Adam runs on each seed.
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Adam refinements (stage 1) ---")
    for idx, seed_vec in enumerate(seeds):
        if time.time() > deadline:
            break
        print(f"[Seed {idx + 1}/{len(seeds)}]")
        seq_adam, val_adam = _adam_optimize(
            seed_vec, n, deadline, max_iters=500_000
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  New global best after Adam: {best_val:.6f}")

    # If nothing improved, fall back to the very best raw seed.
    if best_seq is None:
        best_seq = seeds[0].copy()
        best_val = _objective(best_seq)
        print(f"Fallback to first seed: {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – extended Adam optimisation on the current best.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long Adam optimisation (stage 2) ---")
        seq_adam, val_adam = _adam_optimize(
            best_seq, n, deadline, max_iters=2_000_000
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  Updated global best after long Adam: {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – multi‑restart projected sub‑gradient descent.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient descent (stage 3) ---")
        # first, a direct run from the current best
        seq_sub, val_sub = _subgrad_optimize(
            best_seq, n, deadline, max_iters=1_500_000
        )
        if val_sub < best_val:
            best_seq, best_val = seq_sub, val_sub
            print(f"  Updated best after sub‑grad 1: {best_val:.6f}")

        # a few random perturbation restarts
        restarts = 4
        for r in range(restarts):
            if time.time() > deadline:
                break
            pert = _perturb(best_seq, rng, scale=0.07)
            seq_sub, val_sub = _subgrad_optimize(
                pert, n, deadline, max_iters=800_000
            )
            if val_sub < best_val:
                best_seq, best_val = seq_sub, val_sub
                print(f"  Updated best after sub‑grad restart {r+1}: {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – final simulated‑annealing polishing.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing (stage 4) ---")
        seq_sa, val_sa = _simulated_annealing(
            best_seq, n, deadline, max_iters=800_000,
            init_temp_factor=0.06
        )
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print(f"  Updated best after SA: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # Store for possible reuse and return a plain Python list.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```

The routine is deliberately aggressive with its optimisation budget, uses many complementary families for seeding, performs long Adam runs with a high‑temperature schedule, applies a Polyak‑style sub‑gradient refinement with several random restarts, and finishes with a brief simulated‑annealing polishing stage.  
All intermediate vectors stay on the probability simplex, guaranteeing that the final call to ``evaluate_sequence`` never triggers the guarding checks.
