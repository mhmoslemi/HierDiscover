def _subgradient_descent(a, n, deadline):
    best_a = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # subgradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end+1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]
        # step size: decreasing
        it += 1
        lr = 0.1 / np.sqrt(it)
        a = a - lr * g
        # projection onto simplex
        a = _proj_simplex(a)
    return best_a, best_val

"""
Propose a step‑function (non‑negative list) that minimizes

    2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm proceeds in three phases:

1. **Rich seed generation** – deterministic analytic windows (Gaussian,
   Beta, Kaiser, Tukey, power‑law, sine, etc.) together with many random
   Dirichlet draws.  The parameters are explored on a finer grid than the
   reference solution.

2. **Log‑space Adam on a smooth max‑approximation** – the sequence is
   optimised in the log‑softmax (simplex) space using a temperature‑annealed
   log‑sum‑exp surrogate for the max‑convolution.  Several restarts are
   performed; the best result is kept.

3. **Long exact sub‑gradient descent** – the true max‑convolution objective
   (without smoothing) is minimised by a classic sub‑gradient method with
   Euclidean projection onto the probability simplex.  This step runs for
   as long as the time budget permits and typically yields a noticeable
   improvement over the Adam phase.

4. (Optional) **Stochastic annealing** – a short aggressive annealing pass
   is applied as a final polish.

All stages respect the global deadline (`budget_s` seconds).  The best
vector found is stored in the global name ``height_sequence_1`` and also
returned as a plain list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(seq,seq)) for a simplex‑normalised seq (Σseq=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic shape families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    if p < 0:
        p = 0.0
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) -
                                            2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.signal import kaiser as _kaiser
        a = _kaiser(n, beta)
    except Exception:
        # fallback: pure‑numpy using modified Bessel function approximation
        i = np.arange(n, dtype=np.float64)
        alpha = (n - 1) / 2.0
        y = (i - alpha) / alpha
        try:
            from scipy.special import i0 as _i0
            a = _i0(beta * np.sqrt(1 - y * y))
        except Exception:
            t = beta * np.sqrt(1 - y * y)
            a = 1.0 + (t ** 2) / 4.0 + (t ** 4) / 64.0 + (t ** 6) / 2304.0 + (t ** 8) / 147456.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few low‑frequency cosine components, enforce non‑negativity."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Seed collection – richer sampling

def _collect_seeds_extended(n: int, rng: np.random.Generator, previous: np.ndarray | None):
    seeds = []

    # Deterministic families (fine parameter grids)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (0.5, 1.0, 1.5, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=gamma))
    for sigma in np.linspace(0.1, 0.55, 12):
        seeds.append(_gaussian(n, sigma))
    for p in np.linspace(0.0, 8.0, 33):
        seeds.append(_power(n, p))
    for beta in (1, 2, 4, 6, 8, 10, 12, 15):
        seeds.append(_kaiser_window(n, beta))
    alphas = np.linspace(0.2, 4.0, 15)
    betas = np.linspace(0.2, 4.0, 15)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))
    for tuk_alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, tuk_alpha))

    # Random draws
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with a low‑freq curve
    for mix_alpha in (0.1, 0.3, 0.5, 0.7):
        for nf in (2, 4, 6):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous best (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.shape == (n,) and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # Normalise everything (just in case)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

def _logspace_adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                            max_iters: int = 400_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on the log‑softmax representation.
    Uses a temperature‑annealed log‑sum‑exp surrogate for max‑convolution.
    Returns best vector (ℓ₁‑normalised) and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                # logits
    m = np.zeros_like(v)                       # first moment
    v_sq = np.zeros_like(v)                    # second moment
    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    # Temperature schedule (start low, end high)
    beta = 0.5
    beta_inc = (5000.0 - 0.5) / max_iters      # linear increase
    lr0 = 1.0                                   # learning‑rate in log space

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # current simplex point
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[LogAdam] it={it:6d} best={cur_val:.6f}")

        # smoothed max via log‑sum‑exp with current temperature
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of Σ w_k * conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                      # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # gradient w.r.t. logits (log‑softmax)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / np.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        beta = min(beta + beta_inc, 5000.0)

    return best_seq, best_val


def _subgradient_descent(start_seq: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution.
    Runs until time expires (or a very large iteration limit).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while True:
        if time.time() > deadline:
            break
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # construct sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size – start larger, then decay as 1/√t
        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Aggressive annealing that redistributes mass on random subsets.
    Operates on the exact max‑convolution objective.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    # initial temperature proportional to current objective
    T = max(0.06 * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # very slow geometric cooling
        T = max(T * 0.9999, 1e-12)

        if rng.random() < 0.7:          # reshuffle a random subset
            k = rng.integers(2, 12)          # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                           # tiny pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)  # up to 5 % of source
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within ``budget_s`` seconds.

    The function returns the best sequence as a list of Python floats and
    stores it in the global name ``height_sequence_1``.
    """
    # ------------------------------------------------------------------
    # Initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final evaluator call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a reasonable length (4000 works well with the FFT code)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Phase 1 – seed generation and cheap surrogate ranking
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # Evaluate surrogate (very cheap) and keep the top few
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in evals[:top_k]]

    print(f"Top {top_k} surrogate values (approximate):")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Log‑space Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break

        remaining = deadline - time.time()
        # allocate a small budget to each Adam run (approx 5‑10 % of remaining)
        max_iters = int(min(500_000, remaining * 0.07 * 1e5))
        if max_iters <= 0:
            break

        print(f"\n--- Log‑space Adam run {idx+1}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _logspace_adam_optimize(seed_vec, n, deadline,
                                                     max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Long exact sub‑gradient descent on the best candidate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient descent (exact max) ---")
        seq_sub, val_sub = _subgradient_descent(global_best_seq, n, deadline)
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[overall‑SubGrad] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated annealing polishing (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_iters_sa = int(min(1_500_000, remaining * 1e5))
        seq_sa, val_sa = _simulated_annealing(global_best_seq, n, deadline,
                                              max_iters=max_iters_sa)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
