def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _objective(seq):
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _softmax(v):
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

def _proj_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------- Analytic families -------------------
def _uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _triangular(n):
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _trapezoidal(n, width_fraction=0.2):
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)
    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]
    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _geom(n, r):
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _sym_geom(n, r):
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _power(n, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _sine_window(n, gamma=1.0):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _beta_shape(n, alpha, beta):
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _gaussian(n, sigma_frac):
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _tukey_window(n, alpha):
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _exp_decay(n, rate):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _arcsine(n):
    # Equivalent to beta(0.5,0.5)
    i = np.arange(1, n+1, dtype=np.float64)
    a = 1.0 / np.sqrt(i * (n - i + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _dirichlet_seed(n, conc, rng):
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _sparse_seed(n, rng):
    k = rng.integers(3, 31)  # 3 … 30 non‑zero entries
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a

def _mixture_uniform_gaussian(n, rng):
    lam = rng.random()
    g = _gaussian(n, sigma_frac=rng.uniform(0.05, 0.4))
    u = _uniform(n)
    a = lam * u + (1.0 - lam) * g
    return a / a.sum()

# ------------------- Optimisers -------------------
def _adam_optimize(start_vec, n, deadline, max_iters=500_000):
    eps = 1e-30
    v = np.log(start_vec + eps)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta_temp = 5.0
    beta_mul = 1.001
    beta_max = 2000.0   # higher temperature for tighter max approximation
    best_seq = None
    best_val = float('inf')
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[Adam] it={it} best={best_val:.6f}")
        # increase temperature
        beta_temp = min(beta_temp * beta_mul, beta_max)
        # convolution and smoothed gradient
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)   # length 3n-2
        grad_a = 2.0 * full[n - 1:n - 1 + n]
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        norm = np.linalg.norm(grad_v)
        max_norm = 6.0
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)
    return best_seq, best_val

def _subgrad_optimize(start_seq, n, deadline, max_iters=300_000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] it={it} best={best_val:.6f}")
        # compute sub‑gradient for active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)
        a = a - step * g
        a = _proj_simplex(a)
    return best_seq, best_val

def _collect_diverse_seeds(n, rng, previous, top_k=15, deadline=0.0):
    top_vals = []
    def try_insert(val, seq):
        if len(top_vals) < top_k:
            top_vals.append((val, seq))
            top_vals.sort(key=lambda x: x[0])
        elif val < top_vals[-1][0]:
            top_vals[-1] = (val, seq)
            top_vals.sort(key=lambda x: x[0])
    # deterministic families
    families = [
        _uniform,
        _triangular,
        lambda n: _trapezoidal(n, 0.15),
        lambda n: _sine_window(n, 1.0),
        lambda n: _sine_window(n, 2.0),
        lambda n: _sine_window(n, 3.0),
        lambda n: _beta_shape(n, 0.5, 0.5),
        lambda n: _beta_shape(n, 0.2, 0.2),
        lambda n: _beta_shape(n, 0.1, 0.1),
        lambda n: _gaussian(n, 0.12),
        lambda n: _gaussian(n, 0.24),
        lambda n: _tukey_window(n, 0.3),
        lambda n: _tukey_window(n, 0.6),
        lambda n: _exp_decay(n, 0.001),
    ]
    for fn in families:
        seq = fn(n)
        val = _objective(seq)
        try_insert(val, seq)
    # geometric families
    for r in np.linspace(0.2, 0.98, 12):
        for fn in (_geom, _sym_geom):
            seq = fn(n, r)
            val = _objective(seq)
            try_insert(val, seq)
    # power law
    for p in np.linspace(0.0, 5.0, 25):
        seq = _power(n, p)
        val = _objective(seq)
        try_insert(val, seq)
    # random Dirichlet draws
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 60 if conc < 0.01 else 40
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = _dirichlet_seed(n, conc, rng)
            val = _objective(seq)
            try_insert(val, seq)
    # sparse supports
    for _ in range(200):
        if time.time() > deadline:
            break
        seq = _sparse_seed(n, rng)
        val = _objective(seq)
        try_insert(val, seq)
    # random beta shapes
    for _ in range(250):
        if time.time() > deadline:
            break
        alpha = rng.uniform(0.05, 5.0)
        beta = rng.uniform(0.05, 5.0)
        seq = _beta_shape(n, alpha, beta)
        val = _objective(seq)
        try_insert(val, seq)
    # random sine windows
    for _ in range(150):
        if time.time() > deadline:
            break
        gamma = rng.uniform(0.2, 5.0)
        seq = _sine_window(n, gamma=gamma)
        val = _objective(seq)
        try_insert(val, seq)
    # random Gaussian windows
    for _ in range(150):
        if time.time() > deadline:
            break
        sigma_frac = rng.uniform(0.05, 0.4)
        seq = _gaussian(n, sigma_frac=sigma_frac)
        val = _objective(seq)
        try_insert(val, seq)
    # random Tukey windows
    for _ in range(120):
        if time.time() > deadline:
            break
        alpha = rng.uniform(0.05, 0.9)
        seq = _tukey_window(n, alpha=alpha)
        val = _objective(seq)
        try_insert(val, seq)
    # mixture uniform+gaussian
    for _ in range(120):
        if time.time() > deadline:
            break
        seq = _mixture_uniform_gaussian(n, rng)
        val = _objective(seq)
        try_insert(val, seq)
    # previous solution
    if previous is not None and isinstance(previous, (list, np.ndarray)):
        prev = np.asarray(previous, dtype=np.float64)
        if prev.shape == (n,):
            seq = prev / prev.sum()
            val = _objective(seq)
            try_insert(val, seq)
    return [seq for _, seq in top_vals]

def propose_candidate(seed: int = 123, budget_s: int = 1000):
    """
    Evolutionary‑plus‑gradient search for a non‑negative step‑function that
    minimises 2·n·max(convolution)/sum².
    The routine builds a diverse set of seeds, refines the best with a
    long Adam run, then runs a simple genetic algorithm that repeatedly
    crosses the current population and applies tiny mass‑reallocation
    mutations.  The best individual is finally polished with a short
    sub‑gradient descent and a final Adam pass.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # keep a safety margin

    n = 4000
    print(f"\n=== Starting search, n={n}, budget={budget_s}s ===")
    # ---------- seed generation ----------
    seed_deadline = min(deadline, start_time + budget_s * 0.20)
    previous = globals().get("height_sequence_1")
    seeds = _collect_diverse_seeds(n, rng, previous, top_k=15, deadline=seed_deadline)
    print(f"Collected {len(seeds)} diverse seeds (kept top {len(seeds)})")

    # ---------- short Adam refinement of each seed ----------
    best_seq = None
    best_val = float('inf')
    print("\n--- Short Adam refinement of seeds ---")
    for idx, s in enumerate(seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(s, n, deadline, max_iters=120_000)
        if seq is not None and val < best_val:
            best_seq, best_val = seq, val
            print(f"[Seed {idx+1}] improved to {best_val:.6f}")

    if best_seq is None:
        best_seq = seeds[0].copy()
        best_val = _objective(best_seq)
        print(f"Fallback to first seed, value={best_val:.6f}")

    # ---------- extended Adam on the current best ----------
    if time.time() < deadline:
        print("\n--- Extended Adam on global best ---")
        seq, val = _adam_optimize(best_seq, n, deadline, max_iters=800_000)
        if seq is not None and val < best_val:
            best_seq, best_val = seq, val
            print(f"  Updated best after long Adam: {best_val:.6f}")

    # ---------- initialise population for GA ----------
    pop = [seq for seq in seeds]  # copies already normalized
    # Ensure we have enough individuals
    pop = pop[:20]  # keep at most 20
    pop_vals = [_objective(p) for p in pop]
    # sort by fitness (lower is better)
    pop = sorted(zip(pop_vals, pop), key=lambda x: x[0])
    best_seq = pop[0][1].copy()
    best_val = pop[0][0]
    print(f"\nInitial GA population best value: {best_val:.6f}")

    # ---------- simple genetic algorithm ----------
    print("\n--- Genetic optimisation phase ---")
    while time.time() < deadline:
        # select two parents from the top half of the population
        top = pop[:max(1, len(pop)//2)]
        p1 = rng.choice([ind for _, ind in top])
        p2 = rng.choice([ind for _, ind in top])
        child = (p1 + p2) * 0.5
        child = _proj_simplex(child)
        # mutation: re‑allocate mass on a few random coordinates
        k = rng.integers(2, 7)
        idx = rng.choice(n, size=k, replace=False)
        total = child[idx].sum()
        if total > 0.0:
            child[idx] = rng.dirichlet(np.full(k, 1.0)) * total
        # evaluate
        child_val = _objective(child)
        # replace worst if child is better
        if child_val < pop[-1][0]:
            pop[-1] = (child_val, child)
            pop.sort(key=lambda x: x[0])
            if child_val < best_val:
                best_seq, best_val = child.copy(), child_val
                print(f"[GA] New global best {best_val:.6f}")
        # occasional polishing of the current best with a few Adam steps
        if time.time() > deadline - 15.0:
            # time is about to run out; break after polishing
            break

    # ---------- final polishing ----------
    if time.time() < deadline:
        print("\n--- Final polishing with Adam ---")
        seq, val = _adam_optimize(best_seq, n, deadline, max_iters=200_000)
        if seq is not None and val < best_val:
            best_seq, best_val = seq, val
            print(f"  After final Adam best {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final polishing with sub‑gradient descent ---")
        seq, val = _subgrad_optimize(best_seq, n, deadline, max_iters=200_000)
        if val < best_val:
            best_seq, best_val = seq, val
            print(f"  After final sub‑grad best {best_val:.6f}")

    # ---------- sanity check ----------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"\nFinal evaluate_sequence score = {final_score:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    print(f"\n=== Search finished. Best 2·n·max = {best_val:.6f} ===")
    return best_seq.tolist()

"""
Evolutionary‑plus‑gradient search for a non‑negative step‑function
minimising  2·n·max(convolution)/sum².

The routine builds a diverse pool of analytically‑derived seeds,
refines each seed briefly with Adam, then runs a simple genetic
algorithm (crossover + tiny Dirichlet re‑allocations) that constantly
improves the population.  The current best individual is polished
with a longer Adam run and a short projected sub‑gradient descent.
All vectors are kept on the probability simplex (sum = 1) so the
objective that is optimised is exactly ``2·n·max(conv)``.
"""
import time
import operator
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n·log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for a simplex‑normalised sequence (sum = 1):
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σv=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)
    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]
    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    """Beta(0.5,0.5) – classical arcsine shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i * (n - i + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Random seed generation utilities

def _dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    k = rng.integers(3, 31)               # 3 … 30 non‑zero entries
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


def _mixture_uniform_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.random()
    g = _gaussian(n, sigma_frac=rng.uniform(0.05, 0.4))
    u = _uniform(n)
    a = lam * u + (1.0 - lam) * g
    return a / a.sum()

# Optimisers

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    Returns the best sequence seen (still ℓ₁‑normalised) and its exact
    objective value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)               # logits – softmax stays on simplex
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # temperature ramp (smooth‑max → true max)
    beta_temp = 5.0
    beta_mul = 1.001
    beta_max = 2000.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # current probability vector
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")

        # raise temperature slowly
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # smoothed max gradient
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip gradient norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses a Polyak‑style step size based on the best value seen so far.
    Returns the best sequence encountered and its exact value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size (uses the best value known so far)
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed collection (keeps the top‑k candidates)
# ----------------------------------------------------------------------


def _collect_diverse_seeds(n: int, rng: np.random.Generator,
                          previous: np.ndarray | None,
                          top_k: int = 15,
                          deadline: float = 0.0) -> list[np.ndarray]:
    """
    Generate a rich pool of ℓ₁‑normalised candidates and retain the
    `top_k` with the smallest exact objective.
    """
    top_vals = []          # list of (value, seq)

    def try_insert(val, seq):
        """Maintain a sorted list of the best `top_k` entries."""
        if len(top_vals) < top_k:
            top_vals.append((val, seq))
            top_vals.sort(key=operator.itemgetter(0))
        elif val < top_vals[-1][0]:
            top_vals[-1] = (val, seq)
            top_vals.sort(key=operator.itemgetter(0))

    # deterministic analytic families
    families = [
        _uniform,
        _triangular,
        _trapezoidal_015,
        _sine_window_1,
        _sine_window_2,
        _sine_window_3,
        _beta_05_05,
        _beta_02_02,
        _beta_01_01,
        _gaussian_012,
        _gaussian_024,
        _tukey_03,
        _tukey_06,
        _exp_decay_001,
        _arcsine,
    ]
    for fn in families:
        seq = fn(n)
        val = _objective(seq)
        try_insert(val, seq)

    # geometric families
    for r in np.linspace(0.2, 0.98, 12):
        for fn in (_geom, _sym_geom):
            seq = fn(n, r)
            val = _objective(seq)
            try_insert(val, seq)

    # power‑law family
    for p in np.linspace(0.0, 5.0, 25):
        seq = _power(n, p)
        val = _objective(seq)
        try_insert(val, seq)

    # random Dirichlet draws
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        draws = 60 if conc < 0.01 else 40
        for _ in range(draws):
            if time.time() > deadline:
                break
            seq = _dirichlet_seed(n, conc, rng)
            val = _objective(seq)
            try_insert(val, seq)

    # sparse random supports
    for _ in range(200):
        if time.time() > deadline:
            break
        seq = _sparse_seed(n, rng)
        val = _objective(seq)
        try_insert(val, seq)

    # random Beta shapes
    for _ in range(250):
        if time.time() > deadline:
            break
        alpha = rng.uniform(0.05, 5.0)
        beta = rng.uniform(0.05, 5.0)
        seq = _beta_shape(n, alpha, beta)
        val = _objective(seq)
        try_insert(val, seq)

    # random sine windows
    for _ in range(150):
        if time.time() > deadline:
            break
        gamma = rng.uniform(0.2, 5.0)
        seq = _sine_window(n, gamma=gamma)
        val = _objective(seq)
        try_insert(val, seq)

    # random Gaussian windows
    for _ in range(150):
        if time.time() > deadline:
            break
        sigma_frac = rng.uniform(0.05, 0.4)
        seq = _gaussian(n, sigma_frac=sigma_frac)
        val = _objective(seq)
        try_insert(val, seq)

    # random Tukey windows
    for _ in range(120):
        if time.time() > deadline:
            break
        alpha = rng.uniform(0.05, 0.9)
        seq = _tukey_window(n, alpha=alpha)
        val = _objective(seq)
        try_insert(val, seq)

    # mixture of uniform & Gaussian
    for _ in range(120):
        if time.time() > deadline:
            break
        seq = _mixture_uniform_gaussian(n, rng)
        val = _objective(seq)
        try_insert(val, seq)

    # previous solution (if any)
    if previous is not None and isinstance(previous, (list, np.ndarray)):
        prev = np.asarray(previous, dtype=np.float64)
        if prev.shape == (n,):
            seq = prev / prev.sum()
            val = _objective(seq)
            try_insert(val, seq)

    return [seq for _, seq in top_vals]

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000):
    """
    Evolutionary‑plus‑gradient search for a non‑negative step‑function
    minimising the official objective.  The routine respects the given
    time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Length – 4000 gives a good trade‑off between resolution and speed.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Search started (n={n}, budget={budget_s}s) ===")

    # ------------------------------------------------------------------
    # Stage 0 – generate a diverse set of seeds.
    # ------------------------------------------------------------------
    seed_deadline = min(deadline, start_time + budget_s * 0.20)
    previous = globals().get("height_sequence_1")
    seeds = _collect_diverse_seeds(n, rng, previous,
                                   top_k=15, deadline=seed_deadline)
    print(f"Collected {len(seeds)} promising seeds (kept top {len(seeds)}).")

    # ------------------------------------------------------------------
    # Stage 1 – brief Adam refinements of each seed.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Short Adam refinement of seeds ---")
    for idx, s in enumerate(seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(s, n, deadline, max_iters=120_000)
        if seq is not None and val < best_val:
            best_seq, best_val = seq, val
            print(f"[Seed {idx+1}] improved to {best_val:.6f}")

    if best_seq is None:
        best_seq = seeds[0].copy()
        best_val = _objective(best_seq)
        print(f"Fallback to first seed, value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long Adam run on the current best.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long Adam on current best ---")
        seq, val = _adam_optimize(best_seq, n, deadline, max_iters=800_000)
        if seq is not None and val < best_val:
            best_seq, best_val = seq, val
            print(f"  Updated best after long Adam: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – initialise a small population for a genetic search.
    # ------------------------------------------------------------------
    pop = [p.copy() for p in seeds[:20]]          # keep at most 20 individuals
    pop_vals = [_objective(p) for p in pop]
    pop = sorted(zip(pop_vals, pop), key=operator.itemgetter(0))
    best_seq = pop[0][1].copy()
    best_val = pop[0][0]
    print(f"\nGenetic initial best value: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simple genetic algorithm (crossover + tiny mutations).
    # ------------------------------------------------------------------
    print("\n--- Genetic optimisation phase ---")
    while time.time() < deadline:
        # pick two parents from the top half of the current population
        top = pop[:max(1, len(pop) // 2)]
        parent1 = rng.choice([ind for _, ind in top])
        parent2 = rng.choice([ind for _, ind in top])
        child = (parent1 + parent2) * 0.5
        child = _proj_simplex(child)

        # tiny Dirichlet re‑allocation on a few random coordinates
        k = rng.integers(2, 7)
        idx = rng.choice(n, size=k, replace=False)
        total = child[idx].sum()
        if total > 0.0:
            child[idx] = rng.dirichlet(np.full(k, 1.0)) * total

        child_val = _objective(child)

        # replace the worst individual if the child is better
        if child_val < pop[-1][0]:
            pop[-1] = (child_val, child)
            pop.sort(key=operator.itemgetter(0))
            if child_val < best_val:
                best_seq, best_val = child.copy(), child_val
                print(f"[GA] New global best {best_val:.6f}")

        # when the deadline is near we break to allow polishing
        if time.time() > deadline - 15.0:
            break

    # ------------------------------------------------------------------
    # Stage 5 – final polishing with Adam and a sub‑gradient pass.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final Adam polishing ---")
        seq, val = _adam_optimize(best_seq, n, deadline, max_iters=200_000)
        if seq is not None and val < best_val:
            best_seq, best_val = seq, val
            print(f"  After final Adam best {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final sub‑gradient polishing ---")
        seq, val = _subgrad_optimize(best_seq, n, deadline, max_iters=200_000)
        if val < best_val:
            best_seq, best_val = seq, val
            print(f"  After final sub‑grad best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"\nFinal evaluate_sequence score = {final_score:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    print(f"\n=== Search finished. Best 2·n·max = {best_val:.6f} ===")
    return best_seq.tolist()
