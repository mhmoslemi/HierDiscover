def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(vec: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv) for a probability vector."""
    n = len(vec)
    conv = _conv_fft(vec, vec)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max (maps logits to a point on the simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """One‑sided geometric series (0 < r < 1)."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet with identical concentration parameters."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_spikes(n: int, rng: np.random.Generator,
                   min_k: int = 5, max_k: int = 20) -> np.ndarray:
    """Few random non‑zero entries (Dirichlet on the chosen support)."""
    k = rng.integers(min_k, max_k + 1)
    idx = rng.choice(n, size=k, replace=False)
    values = rng.exponential(scale=1.0, size=k)
    vec = np.zeros(n, dtype=np.float64)
    vec[idx] = values
    vec /= vec.sum()
    return vec

# Optimisers

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 300_000,
                   beta_max: float = 2000.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp (soft‑max) surrogate.
    The variables are logits (unconstrained) – the soft‑max enforces
    positivity and the simplex condition.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)      # initialise logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # temperature schedule (deterministic annealing)
    beta_temp = 5.0
    beta_mul = 1.003

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        # increase temperature gradually
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # ----- compute smoothed gradient -----
        conv = _conv_fft(a, a)
        shift = np.max(conv)                # for numerical stability
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t a (see derivation in the paper)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip gradient norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 2_000_000,
                      init_lr: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution objective.
    Uses a Polyak‑style diminishing step size.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active coordinate (see original code)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)   # enforce feasibility

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 600_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple SA that repeatedly reshuffles mass on a random subset of
    coordinates (Dirichlet re‑allocation).  Accepts worse moves with
    probability exp(-Δ/T) where the temperature T decays slowly.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # modify between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed pool generation (coarse stage)
# ----------------------------------------------------------------------


def _seed_pool(n: int, rng: np.random.Generator,
               previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a heterogeneous collection of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # trapezoidal (random widths)
    for wf in rng.uniform(0.02, 0.5, size=8):
        seeds.append(_trapezoidal(n, width_fraction=float(wf)))

    # sine windows (random exponent)
    for gamma in rng.uniform(0.1, 6.0, size=8):
        seeds.append(_sine_window(n, gamma=float(gamma)))

    # Tukey windows (random α)
    for alpha in rng.uniform(0.0, 0.95, size=8):
        seeds.append(_tukey_window(n, alpha=float(alpha)))

    # geometric families
    for r in rng.uniform(0.2, 0.9999, size=8):
        seeds.append(_geom(n, r=float(r)))
        seeds.append(_sym_geom(n, r=float(r)))

    # power‑law family
    for p in rng.uniform(0.0, 6.0, size=8):
        seeds.append(_power(n, p=float(p)))

    # Beta shapes – explore log‑space near 0 and moderate values
    for _ in range(12):
        a = 10.0 ** rng.uniform(-3, 0)   # 0.001 … 1
        b = 10.0 ** rng.uniform(-3, 0)
        seeds.append(_beta_shape(n, a=float(a), beta=float(b)))

    # Gaussian windows (different σ)
    for sigma_frac in (0.08, 0.12, 0.18, 0.24, 0.30):
        seeds.append(_gaussian(n, sigma_frac=float(sigma_frac)))

    # Random Dirichlet draws (wide concentration range)
    for conc in (0.001, 0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc=float(conc), rng=rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Sparse random spikes
    for _ in range(12):
        seeds.append(_sparse_spikes(n, rng))

    # Previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds


def _upsample(vec_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation from length len(vec_small) to n_large, re‑normalised."""
    x_small = np.linspace(0.0, 1.0, len(vec_small))
    x_large = np.linspace(0.0, 1.0, n_large)
    up = np.interp(x_large, x_small, vec_small)
    if up.sum() == 0.0:
        return _uniform(n_large)
    up /= up.sum()
    return up

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # leave a small safety margin

    # ------------------------------------------------------------------
    # Try to reuse a previously discovered solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list):
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – coarse exploration on a moderate length
    # ------------------------------------------------------------------
    n_small = 2000
    print(f"\n=== Stage 1: coarse search (n={n_small}) ===")
    seeds = _seed_pool(n_small, rng, previous)
    print(f"Generated {len(seeds)} candidate seeds.")

    # evaluate all seeds (fast surrogate)
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])

    top_k = min(12, len(evaluated))
    best_seeds_small = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (small n):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:02d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upsample and smooth optimisation (Adam)
    # ------------------------------------------------------------------
    n_large = 4000
    print(f"\n=== Stage 2: Adam optimisation (n={n_large}) ===")
    global_best_seq = None
    global_best_val = float("inf")

    for idx, seed_vec in enumerate(best_seeds_small[:6]):   # up to six Adam runs
        if time.time() > deadline:
            break
        print(f"\n[Adam] start run {idx+1}/{len(best_seeds_small[:6])}")
        seq_large = _upsample(seed_vec, n_large)
        seq_opt, val_opt = _adam_optimize(
            seq_large,
            n_large,
            deadline,
            max_iters=300_000,
            beta_max=2000.0,
        )
        if seq_opt is not None and val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Adam] new global best = {val_opt:.6f}")

    # If no Adam run improved anything, fall back to the best up‑sampled seed
    if global_best_seq is None:
        print("[Info] Adam did not improve; using best upsampled seed.")
        global_best_seq = _upsample(best_seeds_small[0], n_large)
        global_best_val = _objective(global_best_seq)

    # ------------------------------------------------------------------
    # Stage 3 – projected sub‑gradient refinement (major time consumer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # approximate how many sub‑gradient steps we can afford
        # (empirically ~2 µs per iteration for n=4000)
        max_iters = int(remaining / 2e-6)
        max_iters = max(200_000, min(max_iters, 3_000_000))
        print(f"\n=== Stage 3: sub‑gradient refinement (max_iters={max_iters}) ===")
        seq_proj, val_proj = _subgradient_proj(
            global_best_seq,
            n_large,
            deadline,
            max_iters=max_iters,
            init_lr=0.20,
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[SubProj] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n=== Stage 4: simulated‑annealing polishing ===")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq,
            n_large,
            deadline,
            max_iters=500_000,
            init_temp_factor=0.05,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final check with the official evaluation function
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence returned: {official_val:.6f}")
    except Exception as exc:
        official_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate (≈) = {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # store for possible later reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
