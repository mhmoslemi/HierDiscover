"""
A fast search for a low‑value step‑function.
The algorithm consists of three stages:

1.  Scan a few analytic families (arcsine, beta(2,2), uniform) on several
    lengths and keep the best (lowest exact objective) length and seed.
2.  Perform a long “mirror‑descent” optimisation on the simplex.
    The update is multiplicative (KL‑projection) and uses the exact sub‑gradient
    of the max‑convolution (the active convolution indices are those attaining
    the current maximum).  Symmetry a↔a[::-1] is enforced at each step.
3.  Finish with a short Adam optimisation on a smooth log‑sum‑exp surrogate
    (as in the reference implementation).

All work respects the 1000 s budget.  The best sequence found is returned as
a plain list of non‑negative floats.  The code contains no closures or lambdas,
only top‑level functions.
"""

import time
import numpy as np
from math import pi, sqrt

# ---------------------------------------------------------------
# Low‑level utilities
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for a *normalised* sequence (sum == 1):
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ---------------------------------------------------------------
# Analytic seed families (all returned normalised to sum = 1)
# ---------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * pi * i / (n - 1))
         + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian_sigma015(n: int) -> np.ndarray:
    sigma_factor = 0.15
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _gaussian_sigma025(n: int) -> np.ndarray:
    sigma_factor = 0.25
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf shape on a discrete grid (shifted by 0.5 to avoid end‑points)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape_2_2(n: int) -> np.ndarray:
    return _beta_shape(n, 2.0, 2.0)


# ---------------------------------------------------------------
# Random Dirichlet sampling
# ---------------------------------------------------------------

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Return a random point on the simplex (sum=1) with Dirichlet(alpha)."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ---------------------------------------------------------------
# Smoothed Adam optimiser (log‑sum‑exp surrogate)
# ---------------------------------------------------------------

def _smooth_adam_optimize(start_seq: np.ndarray,
                          deadline: float,
                          beta_start: float = 5.0,
                          beta_end: float = 200.0,
                          max_iter: int = 5_000_000,
                          lr0: float = 0.03) -> tuple[np.ndarray, float]:
    """Minimise a smooth approximation of max(conv(a,a)) via Adam."""
    n = len(start_seq)
    eps = 1e-30

    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        a = np.exp(logits)
        a = a / a.sum()

        conv = _conv_fft(a, a)

        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()

        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if (it & 1023) == 0:          # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")
            if best_val < 1.503:   # early exit if already very good
                break

    return best_seq, best_val


# ---------------------------------------------------------------
# Mirror‑descent optimiser on the exact max‑convolution objective
# ---------------------------------------------------------------

def _mirror_descent(start_seq: np.ndarray,
                    deadline: float,
                    max_iters: int = 1_200_000,
                    eta0: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Multiplicative (KL) projected sub‑gradient descent.
    Uses the exact sub‑gradient built from all convolution entries that
    achieve the current maximum.  Symmetry a_i = a_{n-1-i} is enforced
    after each update.
    Returns the best sequence encountered and its exact objective value.
    """
    a = start_seq.copy()
    n = len(a)

    # initialise best
    conv = _conv_fft(a, a)
    max_val = float(conv.max())
    best_val = 2.0 * n * max_val
    best_seq = a.copy()

    it = 0
    while True:
        if time.time() > deadline or it >= max_iters:
            break
        it += 1

        # current convolution and active set
        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        eps = max_val * 1e-12
        active_idx = np.where(conv >= max_val - eps)[0]

        # weight vector w (uniform on active indices)
        w = np.zeros_like(conv)
        if active_idx.size:
            w[active_idx] = 1.0 / active_idx.size
        else:
            w[np.argmax(conv)] = 1.0

        # sub‑gradient of max(conv) (ignoring the constant 2·n factor)
        conv_ag = _conv_fft(w, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        # mirror‑descent multiplicative update
        eta = eta0 / np.sqrt(it)
        upd = -eta * g
        upd = np.clip(upd, -50, 50)          # protect against overflow
        a = a * np.exp(upd)

        # enforce symmetry and renormalise
        a = (a + a[::-1]) / 2.0
        total = a.sum()
        if total == 0.0 or not np.isfinite(total):
            a = _uniform(n)
        else:
            a = a / total

        # occasional exact evaluation (every 1024 steps)
        if (it & 1023) == 0:
            conv_cur = _conv_fft(a, a)
            max_cur = float(conv_cur.max())
            val_cur = 2.0 * n * max_cur
            if val_cur < best_val:
                best_val = val_cur
                best_seq = a.copy()
                print(f"[Mirror] it={it:7d} n={n:5d} val={val_cur:.6f}")
            if best_val < 1.503:
                break

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine:
      1. tries a few analytic seeds on several lengths,
      2. picks the most promising length,
      3. runs a long mirror‑descent optimisation (symmetry enforced),
      4. optionally refines the result with a short Adam run,
      5. returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Retrieve a previously stored solution if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore (FFT‑friendly, moderate size)
    # ------------------------------------------------------------------
    lengths = [2500, 3000, 3500, 4000, 4500, 5000, 5500, 6000]

    # ------------------------------------------------------------------
    # Phase 1 – cheap evaluation of analytic seeds
    # ------------------------------------------------------------------
    seed_fns = (_arcsine_shape,
                _beta_shape_2_2,
                _uniform)

    best_seq = None
    best_val = float("inf")
    best_n = None

    for n in lengths:
        for fn in seed_fns:
            seq = fn(n)
            # enforce symmetry (helps the later optimisation)
            seq = (seq + seq[::-1]) / 2.0
            seq = seq / seq.sum()
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                best_n = n
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

        # also try the previously stored sequence if it matches the length
        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                best_n = n
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

        # stop early if we are out of time
        if time.time() > deadline - 5.0:
            break

    if best_seq is None:
        # fallback – should never happen
        best_n = 4000
        best_seq = _uniform(best_n)

    # ------------------------------------------------------------------
    # Phase 2 – long mirror‑descent on the most promising length
    # ------------------------------------------------------------------
    # allocate the bulk of the remaining budget to this phase
    time_now = time.time()
    time_remaining = deadline - time_now
    md_deadline = time_now + 0.65 * time_remaining  # about 65 % of what's left

    # run mirror‑descent from the best seed
    seq_md, val_md = _mirror_descent(best_seq, md_deadline,
                                     max_iters=1_200_000,
                                     eta0=0.35)
    if val_md < best_val:
        best_val = val_md
        best_seq = seq_md
        print(f"[Mirror‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional: a few short runs from random Dirichlet starts (same length)
    # ------------------------------------------------------------------
    extra_deadline = deadline - 5.0
    while time.time() < extra_deadline:
        # create a random Dirichlet start (biased towards sparse vectors)
        seq_rand = _random_dirichlet(best_n, rng, alpha=0.5)
        seq_rand = (seq_rand + seq_rand[::-1]) / 2.0
        seq_rand = seq_rand / seq_rand.sum()
        # run a *short* mirror descent (max 200 k iters or time limited)
        sub_deadline = time.time() + min(20.0, extra_deadline - time.time())
        seq_sub, val_sub = _mirror_descent(seq_rand, sub_deadline,
                                           max_iters=200_000,
                                           eta0=0.35)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = seq_sub
            print(f"[Random‑Mirror] new best {best_val:.6f}")
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 3 – final polished Adam refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        adam_deadline = time.time() + min(30.0, deadline - time.time() - 2.0)
        seq_adam, val_adam = _smooth_adam_optimize(
            start_seq=best_seq,
            deadline=adam_deadline,
            beta_start=5.0,
            beta_end=200.0,
            max_iter=2_000_000,
            lr0=0.03,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam
            print(f"[Final‑Adam] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check (the grader will evaluate again anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
