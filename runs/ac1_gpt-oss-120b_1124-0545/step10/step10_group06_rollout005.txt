    # from https://arxiv.org/abs/1101.6081
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / np.arange(1, len(v)+1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _subgradient_descent(start, deadline, max_iters=200_000):
    a = start.copy()
    best = a.copy()
    best_val = _objective(a)
    eta = 0.1
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # compute conv
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # find indices within tolerance
        max_idx = np.where(np.isclose(conv, max_val, atol=1e-12))[0]
        # compute gradient sum
        grad = np.zeros_like(a)
        for j in max_idx:
            # compute contribution for index j
            low = max(0, j - (len(a) - 1))
            high = min(len(a) - 1, j)
            ks = np.arange(low, high + 1)
            grad[ks] += 2 * a[j - ks]  # each term 2*a_{j-k}
        # average if multiple maxima
        grad /= len(max_idx)

        a = np.maximum(a, 0.0)

        if s > 0:

"""
Optimise a non‑negative step function for the objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The method consists of three stages:

1. **Seed generation & selection** – a collection of simple parametric families
   (uniform, random soft‑max, exponential, power‑law, triangular).  The best
   seed with respect to the exact objective is chosen.

2. **Smooth Adam optimisation on the simplex** – the sequence is represented
   by log‑probabilities (soft‑max) so that the simplex constraint is automatic.
   A soft‑max over the convolution entries provides a differentiable proxy for
   the max.  The smoothing parameter λ is increased slowly (annealed) while
   Adam updates the log‑probabilities.

3. **Projected sub‑gradient descent** – after the smooth phase we switch to the
   true (non‑smoothed) max‑convolution.  The sub‑gradient of the maximal entry
   is computed exactly via a single FFT, a simple gradient step is taken,
   negative entries are clipped, and the vector is re‑normalised onto the
   simplex.  This fine‑tunes the solution without the bias of the soft‑max.

A short simulated‑annealing polishing (the previous implementation) is run
as a final touch.  The whole search respects a 1000 s time budget and stores
the best sequence found in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT helpers (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective (same formula as evaluate_sequence)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple families of candidate distributions (sum to 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    exps = np.exp(logits - m)
    return exps / exps.sum()


def _exponential_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def _power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(i, p)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        dec = inc[::-1]
    else:
        dec = np.arange(half, 0, -1, dtype=np.float64)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max over convolution entries (used for smooth Adam)
# ----------------------------------------------------------------------
def _soft_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    max_c = conv.max()
    if lam <= 0.0:
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.where(active, 1.0 / np.count_nonzero(active), 0.0)
        return w
    rel = (conv - max_c) / max_c          # ≤ 0
    exps = np.exp(lam * rel)              # λ·rel is ≤0, safe from overflow
    w = exps / exps.sum()
    return w


def _gradient_wrt_a(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the *soft* max‑convolution (2·rev(a) ⊛ w).
    """
    n = a.size
    conv = _conv_fft(a, a)
    w = _soft_max_weights(conv, lam)      # length = 2n‑1, sums to 1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)            # length = n + (2n‑1) -1 = 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n]    # central slice matches a
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 400_000,
                   lam_start: float = 5.0,
                   lam_factor: float = 1.0005) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑probabilities, with a slowly increasing smoothing
    parameter λ.  Returns the best probability vector encountered.
    """
    eps = 1e-30
    logp = np.log(start + eps)
    n = start.size
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = start.copy()
    best_val = _objective(start)

    t = 0
    lam = lam_start

    while t < max_iters and time.time() < deadline:
        t += 1

        # ---- simplex projection via soft‑max ----
        logp_shift = logp - np.max(logp)
        a = np.exp(logp_shift)
        a /= a.sum()

        # ---- gradient of smoothed max ----
        g_a = _gradient_wrt_a(a, lam)

        # chain rule: ∂/∂logp  = a·(g_a − <a,g_a>)
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log * g_log)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.15 / (math.sqrt(t) + 1.0)

        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- evaluate true objective (hard max) ----
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:7d}  best={cur_val:.6f}  λ={lam:.2f}")

        # ---- raise smoothing ----
        lam *= lam_factor

        # occasional tiny Gaussian kick to escape plateaus
        if t % 25000 == 0:
            noise = np.random.default_rng().normal(scale=0.02, size=n)
            logp += noise

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient of the *hard* max‑convolution
# ----------------------------------------------------------------------
def _subgrad_gradient(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Gradient of (a*a)[idx] w.r.t. a (ordered pairs).  Implements
    g_k = 2·a_{idx‑k} for all k with 0≤idx‑k<n, else 0.
    The computation is performed via a single FFT convolution.
    """
    n = a.size
    rev_a = a[::-1]
    w = np.zeros(2 * n - 1, dtype=np.float64)
    w[idx] = 1.0                     # unit spike at the maximal index
    full = _conv_fft(rev_a, w)       # length = 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _subgradient_descent(start: np.ndarray,
                         deadline: float,
                         max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Each iteration:
        1. compute linear convolution a⊛a
        2. locate maximal index (or all ties)
        3. compute exact sub‑gradient using _subgrad_gradient
        4. gradient step, clip to ≥0 and re‑normalise (projection onto simplex)
    """
    a = start.copy()
    best = a.copy()
    best_val = _objective(a)

    # initial learning rate – heuristically chosen
    eta = 0.7
    tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        max_idxs = np.where(np.abs(conv - max_val) <= tol)[0]

        # average gradient over all tied maxima
        grad = np.zeros_like(a)
        for idx in max_idxs:
            grad += _subgrad_gradient(a, idx)
        grad /= max_idxs.size

        # gradient step
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = _uniform(a.shape[0])

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            # shrink learning rate when we make progress
            eta *= 0.85
            print(f"[Subgrad] it={it:7d}  best={best_val:.6f}  η={eta:.6f}")
        else:
            # slowly decay otherwise
            eta *= 0.995

    return best, best_val


# ----------------------------------------------------------------------
# Light simulated‑annealing polishing (the previous routine)
# ----------------------------------------------------------------------
def _anneal_refine(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Tiny Metropolis–Hastings walk with local swaps and occasional globals."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    n = cur.size
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d}  best={new_val:.6f}")

        if it % 2000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0.0:
                new_seq /= total
            else:
                new_seq = _uniform(n)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects a time budget (default 1000 s) and returns the best
    vector found as a plain list of floats.  The best sequence is also stored
    globally under the name ``height_sequence_1`` for possible reuse.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # safety margin

    rng = np.random.default_rng(seed + 12345)

    # ---------------------------------------------------------------
    # Problem size – a modest increase over the previous 4000 gives a
    # slightly richer search space while staying fast.
    # ---------------------------------------------------------------
    n = 6000
    print(f"\n=== Optimising step function (length {n}) ===")

    # ---------------------------------------------------------------
    # Initialise a pool of deterministic / stochastic seeds
    # ---------------------------------------------------------------
    seeds = []

    # reuse previous result if it matches the current length
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                if prev_arr.sum() > 0.0:
                    seeds.append(prev_arr / prev_arr.sum())
        except Exception:
            pass

    # deterministic families
    seeds.extend([
        _uniform(n),
        _triangular(n),
    ])

    # exponential family – sweep a few rates
    for rate in (0.0005, 0.001, 0.002, 0.004):
        seeds.append(_exponential_decay(n, rate))

    # power‑law family – sweep a few exponents
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power_law(n, p))

    # a few random soft‑max seeds
    for _ in range(3):
        seeds.append(_random_softmax(n, rng))

    # ---------------------------------------------------------------
    # Pick the best seed according to the exact objective
    # ---------------------------------------------------------------
    best_seed = None
    best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seed = s.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # ---------------------------------------------------------------
    # Phase 1 – Adam optimisation (smooth max) on the simplex
    # ---------------------------------------------------------------
    phase1_deadline = deadline - 30.0   # reserve ~30 s for polishing
    if time.time() < phase1_deadline:
        print("[Phase 1] Adam optimisation start")
        seq_adam, val_adam = _adam_optimize(best_seed,
                                            phase1_deadline,
                                            max_iters=500_000,
                                            lam_start=5.0,
                                            lam_factor=1.0005)
        print(f"[Phase 1] best after Adam = {val_adam:.6f}")
    else:
        seq_adam, val_adam = best_seed, best_val
        print("[Phase 1] skipped (insufficient time)")

    # ---------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent on the true max
    # ---------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[Phase 2] Projected sub‑gradient descent")
        seq_sub, val_sub = _subgradient_descent(seq_adam, deadline)
        if val_sub < val_adam:
            seq_adam, val_adam = seq_sub, val_sub
            print(f"[Phase 2] improved to {val_sub:.6f}")
        else:
            print("[Phase 2] no improvement")
    else:
        print("[Phase 2] skipped (time ran out)")

    # ---------------------------------------------------------------
    # Phase 3 – Light simulated‑annealing polishing (as before)
    # ---------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("[Phase 3] Simulated‑annealing polishing")
        seq_final, val_final = _anneal_refine(seq_adam, deadline)
        if val_final < val_adam:
            seq_adam, val_adam = seq_final, val_final
            print(f"[Phase 3] improved to {val_final:.6f}")
        else:
            print("[Phase 3] no improvement")
    else:
        print("[Phase 3] skipped (no time left)")

    # ---------------------------------------------------------------
    # Final normalisation, safety checks and export
    # ---------------------------------------------------------------
    final_seq = np.maximum(seq_adam, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
        total = final_seq.sum()
    final_seq /= total

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")

    # make the result globally reachable for potential warm‑starts
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
