"""
Propose a step‑function (non‑negative list of heights) that minimizes

    2·n·max(conv(seq,seq)) / (sum(seq)²)

The search proceeds in three stages:

1. **Seed generation** – a very diverse pool of ℓ₁‑normalised vectors
   (analytic windows, DPSS windows, Dirichlet draws, low‑frequency
   random curves, mixtures, and the previous best solution).

2. **Refinement** – each of the best seeds is refined by
   *log‑space Adam* (smooth max via log‑sum‑exp) and a long
   *projected sub‑gradient* descent on the exact max‑convolution.
   The number of iterations is adjusted dynamically so that the
   whole search respects the global deadline.

3. **Polishing** – an aggressive simulated‑annealing loop that reshuffles
   mass on random subsets and accepts uphill moves with a slowly‑cooling
   temperature.

All heavy work uses FFT‑based convolution (O(n log n)).  The algorithm
stores the best sequence in the global name ``height_sequence_1`` so
that a later run can start from it.

The function ``propose_candidate`` conforms to the required signature.
"""

import time

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large n)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) assuming Σseq = 1.
    The denominator (sum²) is omitted because the sequence will always be
    ℓ₁‑normalised.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic window families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (
                1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1))
            )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2, a3 = 0.355768, 0.487396, 0.144232, 0.012604
    a = (
        a0
        - a1 * np.cos(2 * np.pi * i / (n - 1))
        + a2 * np.cos(4 * np.pi * i / (n - 1))
        - a3 * np.cos(6 * np.pi * i / (n - 1))
    )
    return a / a.sum()

def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window; falls back to a pure‑numpy implementation if SciPy is absent."""
    if _scipy_kaiser is not None:
        a = _scipy_kaiser(n, beta)
        a = np.maximum(a, 0.0)
        return a / a.sum()
    # fallback – use modified Bessel I0 if available, else a crude series
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    if _scipy_i0 is not None:
        a = _scipy_i0(beta * np.sqrt(1 - y * y))
    else:
        t = beta * np.sqrt(1 - y * y)
        a = 1.0 + (t ** 2) / 4.0 + (t ** 4) / 64.0 + (t ** 6) / 2304.0 + (t ** 8) / 147456.0
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Discrete Prolate Spheroidal (Slepian) window – first DPSS eigenvector."""
    if _scipy_dpss is None:
        # fall back to uniform if DPSS unavailable
        return _uniform(n)
    # we request only the first eigenvector
    w = _scipy_dpss(n, nw, Kmax=1, sym=True)[0]
    w = np.maximum(w, 0.0)  # protect against tiny negatives
    s = w.sum()
    return w / s if s > 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with a scalar concentration parameter."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase, clip to ≥0, renormalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, then renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a large collection of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))

    # Kaiser windows – a broader sweep
    for beta in np.linspace(2, 12, 11):
        seeds.append(_kaiser_window(n, beta))

    # Gaußian windows – finer sigma sweep
    for sigma in np.linspace(0.10, 0.50, 9):
        seeds.append(_gaussian(n, sigma))

    # Power‑law family
    for p in np.linspace(0.0, 5.0, 31):
        seeds.append(_power(n, p))

    # Beta family – systematic grid (coarse → fine)
    alphas = np.linspace(0.3, 4.5, 9)
    betas = np.linspace(0.3, 4.5, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows – finer granularity
    for alpha in np.linspace(0.05, 0.95, 19):
        seeds.append(_tukey_window(n, alpha))

    # DPSS windows – different time‑bandwidth products
    if _scipy_dpss is not None:
        for nw in np.linspace(2.0, 6.0, 9):
            seeds.append(_dpss_window(n, nw))

    # Random Dirichlet draws – many concentration parameters
    for _ in range(20):
        conc = rng.uniform(0.01, 5.0)
        seeds.append(_random_dirichlet(n, conc, rng))

    # Heavy‑tailed exponential draws
    for _ in range(10):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(8):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Convex mixtures of the uniform window with a low‑frequency curve
    for mix_alpha in (0.15, 0.30, 0.45, 0.60, 0.75):
        for nf in (2, 4, 6):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.shape == (n,) and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # Final clean‑up – guarantee ℓ₁‑normalisation and non‑negativity
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        cleaned.append(v / s if s > 0 else _uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _logspace_adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on the log‑softmax parametrisation.
    Uses a log‑sum‑exp surrogate for the max‑convolution.
    Returns the best ℓ₁‑normalised vector seen and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)               # logits
    m = np.zeros_like(v)                      # first moment
    v_sq = np.zeros_like(v)                   # second moment

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5               # start temperature
    beta_inc = 0.05
    beta_max = 500.0

    lr0 = 1.0                # learning rate in log‑space
    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # probability vector on simplex
        a = _softmax(v)

        # surrogate objective (ordering only)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[LogAdam] it={it:7d} best={cur_val:.6f}")

        # --- smoothed max via log‑sum‑exp ---
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

        # convert to gradient on logits (log‑softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update (learning‑rate decays as 1/√t)
        lr = lr0 / np.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule – slowly warm up
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses a 1/√t step‑size and Euclidean projection onto the simplex.
    Returns the best vector found and its surrogate value (2·n·max(conv)).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active max‑lag
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step size
        lr = 0.25 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-14,
) -> tuple[np.ndarray, float]:
    """
    Aggressive annealing that periodically reshuffles the mass of a small
    random subset of coordinates.  Accepts uphill moves with a slowly
    cooling schedule.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    # Initial temperature proportional to the current objective value
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # Very slow geometric cooling
        T = max(T * 0.99995, min_temp)

        # Choose move type
        if rng.random() < 0.75:                     # reshuffle a random subset
            k = rng.integers(2, 15)                # size 2 … 14
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                        # pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.03)  # move at most 3 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimizes the
    ``evaluate_sequence`` objective within the allotted time.

    The algorithm:
        1. builds a very rich seed pool (analytic windows, DPSS, random draws);
        2. refines a handful of the best seeds with log‑space Adam, then a long
           projected sub‑gradient descent;
        3. finishes with an aggressive simulated‑annealing polishing phase.

    The best sequence found is stored globally as ``height_sequence_1`` and
    returned as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin – the evaluator itself performs cheap checks
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length of the step function – 4000 gives a good speed / expressivity trade‑off
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution (if available) to enrich the seed pool
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(previous, list):
        arr = np.asarray(previous, dtype=np.float64)
        if arr.shape == (n,) and arr.sum() > 0.0:
            prev_vec = arr / arr.sum()

    # ------------------------------------------------------------------
    # Stage 1 – generate a huge and diverse set of seeds, evaluate the cheap surrogate,
    # and keep the very best few for expensive refinement.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_seeds_extended(n, rng, prev_vec)
    print(f"Generated {len(seeds)} candidate seeds.")

    # surrogate ranking (objective without the denominator – exact ordering)
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 8
    best_seeds = [vec for vec, _ in evals[:top_k]]

    print(f"\nTop {top_k} surrogate values:")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – refinement of the elite seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break

        # --- Log‑space Adam -------------------------------------------------
        remaining = deadline - time.time()
        # allocate a modest slice of the remaining budget (≈15 % of time)
        time_for_adam = min(remaining * 0.15, 25.0)          # at most 25 s per seed
        max_iters_adam = int(min(800_000, time_for_adam * 12_000))  # rough 12 k iters per sec
        print(f"\n--- Log‑space Adam run {idx+1}/{len(best_seeds)} (max_iters={max_iters_adam}) ---")
        seq_adam, val_adam = _logspace_adam_optimize(
            seed_vec, n, deadline, max_iters=max_iters_adam
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # --- Projected sub‑gradient descent ---------------------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # allocate the bulk of the remaining time to this very cheap iteration
        max_iters_sub = int(min(3_000_000, remaining * 7_000))  # ~7 k iters per sec
        print(f"[SubProj] running up to {max_iters_sub} iterations")
        seq_sub, val_sub = _subgradient_proj(
            global_best_seq, n, deadline, max_iters=max_iters_sub
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[overall‑SubProj] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – aggressive simulated‑annealing polishing (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_iters_sa = int(min(4_000_000, remaining * 8_000))
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=max_iters_sa
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator (optional but cheap)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose the best sequence for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
