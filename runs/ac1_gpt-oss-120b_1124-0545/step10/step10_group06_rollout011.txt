def mix_and_optimize(f, g, objective_fn, max_steps=5):
    # We'll try to find alpha in [0,1] that reduces objective.
    # Start with alpha = 0.5, test lower/higher if needed.
    # Use binary search or simple grid.
    best_seq = f
    best_val = objective_fn(f)
    for alpha in np.linspace(0.0, 1.0, num=11)[1:]:
        new_seq = (1-alpha) * f + alpha * g
        new_seq = np.maximum(new_seq, 0)
        total = new_seq.sum()
        if total == 0:
            continue
        new_seq /= total
        val = objective_fn(new_seq)
        if val < best_val:
            best_val = val
            best_seq = new_seq
    return best_seq, best_val

def _pgd_finetune(seq, deadline, max_iters=200000, step=0.02, decay=0.9995):
    cur = seq.copy()
    n = len(cur)
    cur_val = _objective(cur)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute conv and hardmax weights
        conv = _conv_fft(cur, cur)
        max_c = conv.max()
        # Identify max indices
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.where(active, 1.0/np.count_nonzero(active), 0.0)
        # gradient = 2 * rev(cur) * w
        rev_cur = cur[::-1]
        grad = 2.0 * _conv_fft(rev_cur, w)

        grad = grad[n-1:n-1 + n]
        # Gradient descent step: move opposite direction
        step_size = step * (decay ** it)

        # project onto simplex
        new_seq = _project_onto_simplex(new_seq)
        new_val = _objective(new_seq)
        if new_val < cur_val:

def _adam_optimize(start, deadline, max_iters=200_000, lam_start=0.5, lam_factor=1.0005, patience=20000):
    eps = 1e-30
    logp = np.log(start + eps)
    n = start.size
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)
    beta1 = 0.9; beta2 = 0.999; eps_adam=1e-8

    best_seq = start.copy()
    conv = _conv_fft(start, start)
    best_val = 2.0 * n * conv.max()

    last_improve = 0

    t = 0
    lam = lam_start

    while t < max_iters and time.time() < deadline:
        t += 1

        # compute a
        logp_shift = logp - np.max(logp)
        a = np.exp(logp_shift)
        a_sum = a.sum()
        if a_sum == 0:
            a = _uniform(n)
        else:
            a = a / a_sum

        # gradient
        g_a = _gradient_wrt_a(a, lam)

        # gradient w.r.t. logp (softmax)
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)

        m = beta1 * m + (1 - beta1) * g_log
        v = beta2 * v + (1 - beta2) * (g_log * g_log)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = 0.3 / (math.sqrt(t) + 1.0)

        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # compute true objective
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = t
            print(f"[Adam] it={t:7d} best={cur_val:.6f} lam={lam:.2f}")

        # add restart
        if t - last_improve > patience:
            # add Gaussian noise
            noise = np.random.default_rng().normal(scale=0.05, size=n)
            logp += noise
            last_improve = t
            print(f"[Adam] Restarted at it={t}")

        lam *= lam_factor

    return best_seq, best_val

"""
Propose a step‑function (non‑negative vector) with a low value of

    2·n·max(conv(a,a)) / (sum(a)**2)

The search consists of:

1. Several short Adam runs on the simplex (soft‑max parametrisation) with a
   smoothly increasing soft‑max temperature.  Restarts are triggered if the
   best value does not improve for a while.
2. A light simulated‑annealing (Metropolis) polishing phase.
3. A final projected‑gradient descent phase that uses the exact hard‑max
   gradient of the convolution peak.  The update is projected onto the
   probability simplex each step.

All phases respect the overall time budget (default 1000 s).  The best
sequence found is stored in the global ``height_sequence_1`` and returned
as a plain list of floats.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective (exact – the same as evaluate_sequence for a normalised sum)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple random/simplex helpers
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    exps = np.exp(logits - m)
    return exps / exps.sum()


def _exponential_decay(n: int, rate: float = 0.001) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        dec = inc[::-1]
    else:
        dec = np.arange(half, 0, -1, dtype=np.float64)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – needed for the final PGD phase
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("vector expected")
    n = v.shape[0]

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform
        return _uniform(n)
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max weights over convolution entries (smooth approximation of max)
# ----------------------------------------------------------------------
def _soft_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max (probability) vector over conv, temperature 1/lam."""
    max_c = conv.max()
    if lam <= 0.0:
        active = np.isclose(conv, max_c, atol=1e-12)
        cnt = np.count_nonzero(active)
        if cnt == 0:
            return np.zeros_like(conv)
        return np.where(active, 1.0 / cnt, 0.0)

    rel = (conv - max_c) / max_c          # ≤ 0
    exps = np.exp(lam * rel)
    w = exps / exps.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of the (soft) max‑convolution w.r.t. a (probability vector)
# ----------------------------------------------------------------------
def _gradient_wrt_a(a: np.ndarray, lam: float) -> np.ndarray:
    """Grad of smoothed max(conv(a,a)) w.r.t. a (returns length‑n vector)."""
    n = a.size
    conv = _conv_fft(a, a)
    w = _soft_max_weights(conv, lam)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                     # length = 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n]             # slice matching a
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 200_000,
                   lam_start: float = 0.5,
                   lam_factor: float = 1.0005,
                   patience: int = 20_000) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑probabilities (soft‑max) of `start`.
    ``lam`` controls the smoothness of the max.  A restart (Gaussian
    perturbation of the logits) is performed if the best value has not
    improved for ``patience`` iterations.
    """
    eps = 1e-30
    logp = np.log(start + eps)               # logits
    n = start.size
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = start.copy()
    best_val = _objective(start)
    last_improve = 0
    t = 0
    lam = lam_start
    rng = np.random.default_rng()

    while t < max_iters and time.time() < deadline:
        t += 1

        # current probability vector (soft‑max, stable)
        logp_shift = logp - np.max(logp)
        a = np.exp(logp_shift)
        a_sum = a.sum()
        if a_sum == 0:
            a = _uniform(n)
        else:
            a = a / a_sum

        # gradient of smooth max‑convolution
        g_a = _gradient_wrt_a(a, lam)

        # chain‑rule: grad w.r.t. logits
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log * g_log)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.30 / (math.sqrt(t) + 1.0)     # modest step size

        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # true (non‑smoothed) objective
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = t
            print(f"[Adam] it={t:7d} best={cur_val:.6f} λ={lam:.2f}")

        # restart if stuck
        if t - last_improve > patience:
            noise = rng.normal(scale=0.04, size=n)
            logp += noise
            last_improve = t
            print(f"[Adam] restart at it={t}")

        lam *= lam_factor

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light simulated‑annealing polishing
# ----------------------------------------------------------------------
def _anneal_refine(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis–Hastings walk that moves a small fraction of mass
    between random coordinates, with occasional global Gaussian jumps.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    n = cur.size
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # shift a small amount from i to j
        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # every ~2000 steps: global Gaussian perturbation
        if it % 2000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0:
                new_seq /= total
            else:
                new_seq = _uniform(n)
            new_val = _objective(new_seq)
            accept = (new_val < cur_val) or (rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Projected gradient descent on the *hard* max‑convolution peak
# ----------------------------------------------------------------------
def _pgd_finetune(start: np.ndarray,
                  deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.05,
                  decay: float = 0.9999) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent using the exact gradient of max(conv(a,a)).
    Each step is projected onto the probability simplex.
    """
    cur = start.copy()
    n = cur.size
    cur_val = _objective(cur)
    rng = np.random.default_rng()
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        # hard‑max weights
        conv = _conv_fft(cur, cur)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        cnt = np.count_nonzero(active)
        if cnt == 0:
            w = np.zeros_like(conv)
        else:
            w = np.where(active, 1.0 / cnt, 0.0)

        # gradient of the peak (2·rev(cur)⊛w)
        rev_cur = cur[::-1]
        full = _conv_fft(rev_cur, w)
        grad = 2.0 * full[n - 1:n - 1 + n]

        # descent step
        step = step0 * (decay ** it)
        new_seq = cur - step * grad
        new_seq = _project_onto_simplex(new_seq)

        new_val = _objective(new_seq)
        if new_val < cur_val:
            cur = new_seq
            cur_val = new_val

        # occasional random jitter to escape tiny plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.001, size=n)
            cand = cur + jitter
            cand = np.clip(cand, 0.0, None)
            cand /= cand.sum()
            cand_val = _objective(cand)
            if cand_val < cur_val:
                cur = cand
                cur_val = cand_val

    return cur, cur_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` (or equivalently the formulation in this file).
    The function respects a time budget (default 1000 s) and returns the best
    vector found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    rng = np.random.default_rng(seed + 12345)

    # ---------------------------------------------------------------
    # Problem size – fixed to 4000 (a good trade‑off between expressiveness
    # and speed).  Could be increased for a longer budget.
    # ---------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising step function (length {n}) ===")

    # ---------------------------------------------------------------
    # Initial seeds (previous result if available + a few deterministic ones)
    # ---------------------------------------------------------------
    seeds = []
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                if prev_arr.sum() > 0:
                    seeds.append(prev_arr / prev_arr.sum())
        except Exception:
            pass

    seeds.extend([
        _uniform(n),
        _random_softmax(n, rng),
        _exponential_decay(n, rate=0.001),
        _triangular(n),
    ])

    best_seq = None
    best_val = float("inf")
    for s in seeds:
        v = _objective(s)
        if v < best_val:
            best_val = v
            best_seq = s.copy()
    print(f"[Init] best initial objective = {best_val:.6f}")

    # ---------------------------------------------------------------
    # Phase 1 – several short Adam runs (different random starts)
    # ---------------------------------------------------------------
    num_restarts = 3
    for run in range(num_restarts):
        if time.time() >= deadline - 30.0:
            break
        print(f"[Phase 1] Adam run {run + 1}/{num_restarts}")

        # random start (unless this is the first run and we already have a good seed)
        if run == 0:
            start_vec = best_seq.copy()
        else:
            start_vec = _random_softmax(n, rng)

        seq_adam, val_adam = _adam_optimize(start_vec, deadline,
                                            max_iters=200_000,
                                            lam_start=0.5,
                                            lam_factor=1.0005,
                                            patience=15_000)

        if val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam.copy()
            print(f"[Phase 1] new overall best = {val_adam:.6f}")

    # ---------------------------------------------------------------
    # Phase 2 – simulated‑annealing polishing
    # ---------------------------------------------------------------
    if time.time() < deadline - 10.0:
        print("[Phase 2] Simulated‑annealing polishing")
        seq_anneal, val_anneal = _anneal_refine(best_seq, deadline,
                                                max_iters=200_000)
        if val_anneal < best_val:
            best_val = val_anneal
            best_seq = seq_anneal.copy()
            print(f"[Phase 2] improved to {val_anneal:.6f}")
        else:
            print("[Phase 2] no improvement")
    else:
        print("[Phase 2] skipped (time)")

    # ---------------------------------------------------------------
    # Phase 3 – final exact projected‑gradient descent (hard max)
    # ---------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[Phase 3] Final PGD fine‑tuning")
        seq_pgd, val_pgd = _pgd_finetune(best_seq, deadline,
                                         max_iters=200_000,
                                         step0=0.05,
                                         decay=0.9999)
        if val_pgd < best_val:
            best_val = val_pgd
            best_seq = seq_pgd.copy()
            print(f"[Phase 3] improved to {val_pgd:.6f}")
        else:
            print("[Phase 3] no improvement")
    else:
        print("[Phase 3] skipped (time)")

    # ---------------------------------------------------------------
    # Final normalisation and export
    # ---------------------------------------------------------------
    final_seq = np.maximum(best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
        total = final_seq.sum()
    final_seq = final_seq / total

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")

    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
