"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The code combines several ideas that go beyond the reference implementation:

* **Multiple diverse seeds** – uniform, random softmax, exponential decay,
  triangular, a deterministic Sidon‑like set and a fast random‑sparse seed.
  The best seed (according to the true objective) is taken as a start.

* **Smoothed‑max Adam optimisation on the simplex** – the objective is
  approximated by a soft‑max over the convolution entries.  The smoothing
  temperature `λ` starts very small (highly smooth) and is increased very
  slowly (`λ ← λ·λ_factor`).  This lets the optimiser explore a smooth
  landscape first and then gradually focus on the true non‑smooth max.
  The update is performed on log‑probabilities (exponentiated‑gradient /
  mirror‑descent) which automatically respects non‑negativity and the
  simplex constraint.

* **More iterations / slower temperature schedule** – the reference used
  `lam_start=5.0` and `lam_factor=1.001`.  Here we start at `0.5` and
  increase with `λ_factor=1.0007`, keeping the optimisation “soft’’ for far
  longer.  Up to 800 000 Adam steps are performed (subject to the time
  budget).

* **Extended simulated‑annealing polishing** – after Adam the current best
  vector is refined by a Metropolis–Hastings walk that transfers a small
  fraction of mass between random coordinates, performs occasional global
  Gaussian jumps and uses a slowly decaying temperature schedule.  The
  walk runs for as many iterations as time permits (up to 500 000 steps).

* **Time‑aware orchestration** – the whole routine respects the supplied
  budget (default 1000 s).  Phases are truncated automatically when the
  deadline is reached, guaranteeing a return value before timeout.

The final result is normalised to sum to 1, clipped to the interval
`[0, 1000]` (the limits enforced by ``evaluate_sequence``) and stored in
the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Convolution & objective
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used for reporting (same formula as evaluate_sequence)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families (return a probability vector that sums to 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    # stable soft‑max
    m = np.max(logits)
    exps = np.exp(logits - m)
    return exps / exps.sum()


def _exponential_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)  (then normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    """Increasing then decreasing linear shape."""
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    dec = inc[::-1] if n % 2 == 0 else np.arange(half, 0, -1, dtype=np.float64)
    a = np.concatenate([inc, dec])
    return a / a.sum()


def _sidon_like(n: int, m: int) -> np.ndarray:
    """Deterministic Sidon‑like support (binary)."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a / a.sum()


def _random_sparse(n: int, rng: np.random.Generator, sparsity: float = 0.07) -> np.ndarray:
    """Few random non‑zero entries (weights drawn from Exp(1))."""
    m = max(2, int(sparsity * n))
    idx = rng.choice(n, size=m, replace=False)
    vals = rng.exponential(scale=1.0, size=m)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max over convolution entries (used for smoothing)
# ----------------------------------------------------------------------
def _soft_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Probability vector w (sum = 1) that is a soft‑max of `conv`
    with temperature λ (larger λ → sharper)."""
    max_c = conv.max()
    if lam <= 0.0:
        active = np.isclose(conv, max_c, atol=1e-12)
        cnt = np.count_nonzero(active)
        return np.where(active, 1.0 / cnt, 0.0)
    # scale relative to the maximum for stability
    rel = (conv - max_c) / max_c          # ≤ 0
    exps = np.exp(lam * rel)
    return exps / exps.sum()


def _gradient_wrt_a(a: np.ndarray, lam: float) -> np.ndarray:
    """Gradient of smoothed max‑convolution with respect to probability vector a."""
    n = a.size
    conv = _conv_fft(a, a)
    w = _soft_max_weights(conv, lam)          # shape = 2n‑1
    rev_a = a[::-1]
    # full convolution rev_a ⊛ w
    full = _conv_fft(rev_a, w)                # length = 3n‑2
    # extract the slice that corresponds to the original a
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 800_000,
                   lam_start: float = 0.5,
                   lam_factor: float = 1.0007,
                   lr0: float = 0.18) -> Tuple[np.ndarray, float]:
    """Run Adam on the log‑probabilities of `start` while slowly hardening
    the soft‑max approximation of the max‑convolution.  Returns the best
    probability vector found and its true objective value."""
    eps = 1e-30
    logp = np.log(start + eps)                # shape (n,)
    n = start.size
    m1 = np.zeros_like(logp)                  # first moment
    v1 = np.zeros_like(logp)                  # second moment

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = start.copy()
    best_val = _objective(start)

    t = 0
    lam = lam_start

    while t < max_iters and time.time() < deadline:
        t += 1

        # ------- current probability vector (soft‑max) -------
        logp_shift = logp - np.max(logp)       # for numerical stability
        a = np.exp(logp_shift)
        a /= a.sum()

        # ------- gradient of the smoothed objective -------
        g_a = _gradient_wrt_a(a, lam)          # d/da (soft‑max approximation)

        # chain rule for log‑probabilities (mirror‑descent)
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)                # ∂f/∂logp

        # ------- Adam updates (bias‑corrected) -------
        m1 = beta1 * m1 + (1.0 - beta1) * g_log
        v1 = beta2 * v1 + (1.0 - beta2) * (g_log * g_log)

        m_hat = m1 / (1.0 - beta1 ** t)
        v_hat = v1 / (1.0 - beta2 ** t)

        # learning‑rate schedule (slow decay)
        lr = lr0 / (math.sqrt(t) + 1.0)

        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- true (hard) objective -------
        # reuse the convolution already computed in the gradient (conv = a*a)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:7d}  best={cur_val:.6f}  λ={lam:.2f}")

        # ------- temperature schedule -------
        lam *= lam_factor

        # occasional tiny Gaussian kick‑restart to escape plateaux
        if t % 50000 == 0:
            rng = np.random.default_rng()
            logp += rng.normal(scale=0.015, size=n)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (more aggressive than the reference)
# ----------------------------------------------------------------------
def _anneal_refine(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 500_000) -> Tuple[np.ndarray, float]:
    """Metropolis–Hastings walk that redistributes a tiny fraction of the
    mass between two random coordinates and occasionally adds a global
    Gaussian perturbation.  The temperature decays geometrically."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    n = cur.size
    # initial temperature roughly 5 % of the current objective
    T0 = max(1e-8, cur_val * 0.05)
    decay = 0.9994
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # ----- local move: shift a small fraction between two indices -----
        i, j = rng.integers(0, n, size=2)
        if i != j:
            delta = rng.random() * 0.015 * cur[i]      # up to 1.5 % of mass at i
            delta = min(delta, cur[i])
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            new_val = _objective(cand)
            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur, cur_val = cand, new_val
                if new_val < best_val:
                    best, best_val = cand.copy(), new_val
                    print(f"[Anneal] it={it:7d}  best={new_val:.6f}")

        # ----- global jump every few thousand steps -----
        if it % 2500 == 0:
            noise = rng.normal(scale=0.0015, size=n)
            cand = cur + noise
            cand = np.clip(cand, 0.0, None)
            total = cand.sum()
            if total > 0.0:
                cand /= total
            else:
                cand = _uniform(n)
            new_val = _objective(cand)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur, cur_val = cand, new_val
                if new_val < best_val:
                    best, best_val = cand.copy(), new_val
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val

            print(f"[Phase 2] improved to {val_final:.6f}")
        else:
            print("[Phase 2] no improvement")
    else:
        print("[Phase 2] no time left")

    # ---------------------------------------------------------------
    # Normalise, enforce evaluator limits and export
    # ---------------------------------------------------------------
    final_seq = np.maximum(seq_adam, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
        total = final_seq.sum()
    final_seq = final_seq / total

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")

    # store for potential warm‑starts in future calls
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
