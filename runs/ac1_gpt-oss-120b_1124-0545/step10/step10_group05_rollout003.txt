"""
Optimise a non‑negative step function for the ``evaluate_sequence`` objective.

The search consists of several inexpensive phases that explore different
lengths, analytic families and random Dirichlet / Beta draws.
The best candidate is then polished with:

* A smoothed Adam optimiser (log‑sum‑exp approximation of the max‑convolution).
* A projected sub‑gradient descent that works directly on the exact max
  (the active set of near‑maximal convolution entries is shrunk
  exponentially during the run).
* A small “restart‑and‑polish’’ loop that perturbs the current best
  solution and runs the sub‑gradient method again.

All phases respect a global 1000 s time‑budget; the best sequence found is
returned as a plain Python ``list`` of non‑negative numbers (already
normalised to sum = 1).  The result is stored in the global variable
``height_sequence_1`` for possible reuse in later runs.
"""

import time
import numpy as np
from math import sqrt, pi, log, exp

# ---------------------------------------------------------------
# Low‑level utilities
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a *normalised* sequence
    (sum == 1).  Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * float(conv.max())

# Simplex projection (used by additive sub‑gradient method)
# ---------------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑d array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → projection is the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0 - a1 * np.cos(2 * pi * i / (n - 1)) + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf shape on a discrete grid (shifted by 0.5 to avoid ends)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _sin_shape(n: int) -> np.ndarray:
    """Half‑sine bump, zero at the ends, peak in the centre."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(pi * i / n)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet‑like vector (Gamma/Exp) with concentration α."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


def _random_beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.uniform(0.1, 5.0)
    b = rng.uniform(0.1, 5.0)
    return _beta_shape(n, a, b)


def _random_mixture_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two independent beta pdfs, mixed with a random weight."""
    w = rng.random()
    a1, b1 = rng.uniform(0.1, 5.0), rng.uniform(0.1, 5.0)
    a2, b2 = rng.uniform(0.1, 5.0), rng.uniform(0.1, 5.0)
    s1 = _beta_shape(n, a1, b1)
    s2 = _beta_shape(n, a2, b2)
    mix = w * s1 + (1.0 - w) * s2
    mix /= mix.sum()
    return mix

# Smoothed Adam optimiser (log‑softmax parametrisation)
# ---------------------------------------------------------------

def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 1200.0,
    max_iter: int = 5_000_000,
    lr0: float = 0.03,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth log‑sum‑exp approximation of max(conv(a,a)).
    Operates on the logits of a soft‑max (so the iterate always stays
    on the simplex).  Returns the best sequence found before ``deadline``.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # soft‑max
        a = np.exp(logits)
        a /= a.sum()

        # convolution
        conv = _conv_fft(a, a)

        # temperature schedule (linear)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        # smooth‑max via log‑sum‑exp
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()   # soft weights, sum=1

        # gradient w.r.t. a (2 * (p * a) convolved with reversed a)
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate decay ∝ 1/√it
        lr = lr0 / sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional exact evaluation
        if (it & 1023) == 0:          # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

        # early exit if we have already hit a very good bound
        if best_val < 1.45:
            break

    return best_seq, best_val


# ---------------------------------------------------------------
# Projected sub‑gradient descent (additive updates)
# ---------------------------------------------------------------

def _subgradient_descent(
    start_seq: np.ndarray,
    deadline: float,
    eps_start: float = 0.02,
    eps_end: float = 1e-6,
    lr_base: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Simple additive sub‑gradient descent on the exact max‑convolution.
    The active set consists of those entries that are within a relative
    tolerance ``eps_rel`` of the current maximum; ``eps_rel`` decays
    exponentially from ``eps_start`` to ``eps_end``.
    The update is projected onto the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    t0 = time.time()

    n = len(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        # exponential decay of the relative tolerance
        elapsed = time.time() - t0
        total = deadline - t0
        frac = elapsed / total if total > 0 else 0.0
        eps_rel = eps_start * (eps_end / eps_start) ** frac

        # active set (binary mask on the full convolution)
        active = (conv >= max_val * (1.0 - eps_rel)).astype(np.float64)

        # gradient of the sum of the active entries
        conv_ag = _conv_fft(active, a[::-1])
        grad = 2.0 * conv_ag[n - 1:n - 1 + n]

        # step size
        lr = lr_base / sqrt(it)

        # additive update + projection onto simplex
        a_new = a - lr * grad
        a_new = _project_simplex(a_new)

        # periodic exact check
        if (it & 1023) == 0:
            val = _objective(a_new)
            if val < best_val:
                best_val = val
                best_seq = a_new.copy()
                print(f"[SubGrad] it={it:7d} n={n:5d} val={val:.6f}")

        a = a_new

    return best_seq, best_val

# Perturb‑and‑polish (multiple restarts of sub‑gradient descent)
# ---------------------------------------------------------------

def _restart_polish(
    base_seq: np.ndarray,
    deadline: float,
    n_restarts: int = 5,
    perturb_scale: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Starting from ``base_seq`` perform a few random perturbations and
    polish each of them with ``_subgradient_descent``.  The best result
    is returned.
    """
    best_seq = base_seq.copy()
    best_val = _objective(best_seq)

    n = len(base_seq)
    rng = np.random.default_rng(12345)

    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # generate a Gaussian perturbation, keep non‑negative and renormalise
        perturb = rng.normal(scale=perturb_scale, size=n)
        cand = best_seq + perturb
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            cand = _uniform(n)
        else:
            cand /= cand_sum

        # allocate a small slice of the remaining budget for polishing
        sub_deadline = time.time() + min(12.0, remaining * 0.3)
        cand_seq, cand_val = _subgradient_descent(cand, sub_deadline,
                                                   eps_start=0.02,
                                                   eps_end=1e-6,
                                                   lr_base=0.5)
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand_seq
            print(f"[Restart] improved to {best_val:.6f}")

        if n_restarts <= 0:
            break
        n_restarts -= 1

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Perform a timed search for a step‑function minimising
    ``evaluate_sequence``.  Returns the best sequence found as a plain
    ``list`` of non‑negative numbers (already normalised to sum = 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Re‑use a previously stored solution if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:          # pragma: no cover
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore (FFT‑friendly, modest memory)
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 8000]   # a handful of candidates

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic families
    # ------------------------------------------------------------------
    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.25),
        _arcsine_shape,
        _sin_shape,
        lambda n: _beta_shape(n, 0.5, 0.5),
        lambda n: _beta_shape(n, 2.0, 2.0),
        lambda n: _beta_shape(n, 0.5, 2.0),
        lambda n: _beta_shape(n, 2.0, 0.5),
    )

    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

        # reuse previous solution if it matches the current length
        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – heavy random sampling (Dirichlet + random Beta shapes)
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.30   # about 30 % of remaining budget
    sample_deadline = now + time_for_sampling

    alphas = [0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))

        # Dirichlet draw with random concentration
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd-Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        # Random Beta pdf
        seq = _random_beta_shape(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd-Beta] n={n:5d} val={val:.6f}")

        # Random mixture of two Betas
        seq = _random_mixture_beta(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd-Mix] n={n:5d} val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Smoothed Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # build a small pool of restart candidates
        candidates = [best_seq]
        top_len = len(best_seq)

        for _ in range(2):
            cand = _random_dirichlet(top_len, rng, alpha=rng.choice(alphas))
            candidates.append(cand)

        # optionally try a larger length if any
        larger_lengths = [L for L in lengths if L > top_len]
        if larger_lengths:
            L_big = int(rng.choice(larger_lengths))
            cand_big = _random_dirichlet(L_big, rng, alpha=1.0)
            candidates.append(cand_big)

        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.25   # ~25 % of remaining time

        for idx, cand in enumerate(candidates):
            seq_adam, val_adam = _smooth_adam_optimize(
                start_seq=cand,
                deadline=adam_deadline,
                beta_start=5.0,
                beta_end=1200.0,
                max_iter=5_000_000,
                lr0=0.03,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = seq_adam
                print(f"[Adam‑Refine {idx}] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Projected sub‑gradient descent (long polishing phase)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.45   # bulk of the rest
        seq_sub, val_sub = _subgradient_descent(
            start_seq=best_seq,
            deadline=sub_deadline,
            eps_start=0.02,
            eps_end=1e-6,
            lr_base=0.5,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = seq_sub
            print(f"[SubGrad‑Polish] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Small perturb‑and‑polish restarts (exploit remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            seq_restart, val_restart = _restart_polish(
                base_seq=best_seq,
                deadline=deadline,
                n_restarts=8,
                perturb_scale=0.02,
            )
            if val_restart < best_val:
                best_val = val_restart
                best_seq = seq_restart
                print(f"[Restart‑Polish] new best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will re‑evaluate anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
