```
def evaluate_sequence(sequence):
    # validation (list, nonempty, each element numeric, no NaN/inf)
    # convert each to float
    # protect against negative: sequence = [max(0, x) for x in sequence]
    # protect large: [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01: return inf
    return float(2 * n * max_b / (sum_a**2))
```

```python
import time
import numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _objective(seq):
    # expects numpy ndarray, sum may not be 1
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv) / (np.sum(seq) ** 2))

def _softmax(v):
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

def _proj_simplex(v):
    """Euclidean projection onto the simplex Σ v_i = 1, v_i>=0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n):
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n, gamma=1.0):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n, sigma_frac):
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n, alpha, beta):
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n, conc, rng):
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_seed(n, rng):
    k = rng.integers(3, 31)
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


def _mixture_uniform_gaussian(n, rng):
    lam = rng.random()
    g = _gaussian(n, sigma_frac=rng.uniform(0.05, 0.4))
    u = _uniform(n)
    a = lam * u + (1.0 - lam) * g
    return a / a.sum()

def _adam_optimize(start_vec, n, deadline, max_iters=500_000,
                   beta_initial=5.0, beta_mul=1.001, beta_max=500.0):
    """Adam optimiser on smooth max (log‑sum‑exp) surrogate."""
    eps = 1e-30
    v = np.log(start_vec + eps)  # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = beta_initial

    best_seq = None
    best_val = float("inf")
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # periodic deadline check
        if it % 2000 == 0 and time.time() > deadline:
            break

        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        # temperature increase
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # compute convolution
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient wrt a (linear convolution with reversed a)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)  # length 3n-2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # transform gradient to logits space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip gradient norm
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq, n, deadline, max_iters=500_000):
    """Projected sub‑gradient descent on the exact max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        if it % 2000 == 0 and time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

        # sub‑gradient for active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val

"""
Improved search for a low‑value step‑function.

The routine builds a modest pool of diverse ℓ₁‑normalised seeds,
refines each seed with a longer Adam run (smooth‑max surrogate),
then applies a heavy‑weight Adam on the current best, followed by
projected sub‑gradient descent, simulated‑annealing polishing and a
final random‑local‑search phase.  All stages respect the supplied time
budget (default 1000 s) and the best sequence is stored in the global
variable ``height_sequence_1`` for reuse.

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the official objective (scale‑invariant).
    Assumes ``seq`` is a non‑negative NumPy array.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv) / (np.sum(seq) ** 2))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σv_i=1, v_i≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all ℓ₁‑normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    k = rng.integers(3, 31)               # 3 … 30 non‑zero entries
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


def _mixture_uniform_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.random()
    g = _gaussian(n, sigma_frac=rng.uniform(0.05, 0.4))
    u = _uniform(n)
    a = lam * u + (1.0 - lam) * g
    return a / a.sum()

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 500_000,
                   beta_initial: float = 5.0,
                   beta_mul: float = 1.001,
                   beta_max: float = 500.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate.
    Parameters are the logits of the soft‑max representation.
    Returns (best_sequence, best_value) where best_value is the exact
    objective value (not the surrogate).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = beta_initial

    best_seq = None
    best_val = float("inf")
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        if it % 2000 == 0 and time.time() > deadline:
            break

        a = _softmax(v)                         # probability vector on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        # raise temperature slowly so we approach the true max
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # exact convolution of a with itself
        conv = _conv_fft(a, a)                  # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                               # softmax over convolution entries

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]    # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip gradient norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses a Polyak‑style step size based on the best value seen so far.
    Returns the best sequence encountered and its exact value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        if it % 2000 == 0 and time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

        # sub‑gradient w.r.t. the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size (using currently best known value)
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val

    Search for a non‑negative step‑function (list of floats) that
    minimises ``evaluate_sequence``.  The routine respects the supplied
    time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Length – 4096 (power of two) offers a good trade‑off between speed
    # and resolution for the FFT‑based utilities.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Stage 0 – abundant seed generation.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    seed_deadline = min(deadline, start_time + budget_s * 0.20)
    seeds = _collect_diverse_seeds(n, rng, previous,
                                   top_k=6, deadline=seed_deadline)
    print(f"Collected {len(seeds)} promising seeds.")

    # ------------------------------------------------------------------
    # Stage 1 – medium‑length Adam runs on each seed.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Adam refinement (stage 1) ---")
    for idx, seed_vec in enumerate(seeds):
        if time.time() > deadline:
            break
        print(f"[Seed {idx+1}/{len(seeds)}]")
        seq_adam, val_adam = _adam_optimize(
            seed_vec, n, deadline,
            max_iters=300_000,
            beta_initial=5.0,
            beta_mul=1.001,
            beta_max=300.0,
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  New global best after Adam: {best_val:.6f}")

    # Fallback to the first seed if nothing improved (unlikely)
    if best_seq is None:
        best_seq = seeds[0].copy()
        best_val = _objective(best_seq)
        print(f"Fallback seed value: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long Adam optimisation on the current global best.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long Adam optimisation (stage 2) ---")
        seq_adam, val_adam = _adam_optimize(
            best_seq, n, deadline,
            max_iters=1_200_000,
            beta_initial=5.0,
            beta_mul=1.001,
            beta_max=800.0,
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  Updated best after long Adam: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – projected sub‑gradient descent.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient descent (stage 3) ---")
        seq_sub, val_sub = _subgrad_optimize(
            best_seq, n, deadline,
            max_iters=800_000,
        )
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"  Updated best after sub‑grad: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing (stage 4) ---")
        seq_sa, val_sa = _simulated_annealing(
            best_seq, n, deadline,
            max_iters=600_000,
            init_temp_factor=0.06,
        )
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"  Updated best after SA: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – tiny random local search (uses remaining budget).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random local search (stage 5) ---")
        seq_loc, val_loc = _random_local_search(
            best_seq, n, deadline,
            max_iters=300_000,
            sigma=0.01,
        )
        if val_loc < best_val:
            best_seq = seq_loc.copy()
            best_val = val_loc
            print(f"  Updated best after local search: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )

    # Store for possible reuse and return a plain Python list.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```
