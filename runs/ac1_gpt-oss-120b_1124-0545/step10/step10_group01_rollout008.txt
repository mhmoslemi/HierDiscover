def _l_bfgs_smooth_opt(start_vec, n, beta_schedule, maxiter_per_stage, deadline):
    v = np.log(start_vec + 1e-30)  # initial logits
    best_vec = start_vec
    best_val = _objective_surrogate(start_vec)

    for beta in beta_schedule:
        def func(theta):
            a = _softmax(theta)  # n vector
            conv = _conv_fft(a, a)
            # compute smoothed max expectation
            shift = np.max(conv)
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0.0:
                obj = 2.0 * n * shift
                grad_theta = np.zeros_like(theta)
                return obj, grad_theta
            w /= w_sum
            # objective surrogate: weighted sum
            obj = np.dot(w, conv)  # expectation
            # gradient wrt a:
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            # gradient wrt theta as before
            dot = np.dot(grad_a, a)
            grad_theta = a * (grad_a - dot)
            # convert to objective 2 n * obj maybe? Actually real objective uses 2 n * max. But we minimize weighted sum, which we will later compute actual value. Minimizing weighted sum approximates min max.
            return 2.0 * n * obj, grad_theta
        # call minimize
        res = scipy.optimize.minimize(func, v, method='L-BFGS-B', jac=True,
                                      options={'maxiter': maxiter_per_stage, 'disp': False})
        v = res.x
        a = _softmax(v)
        # evaluate true objective
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_vec = a.copy()
        # continue to next beta
        # optionally break if deadline near
        if time.time() > deadline: break

    return best_vec, best_val
```

def _lbfgs_continuation(v0, n, betas, deadline, maxiter_per_beta=50):
    v = v0.copy()
    best_seq = None
    best_val = float('inf')
    for beta in betas:
        if time.time() > deadline:
            break

        # define function for minimize
        def fun(theta):
            a = _softmax(theta)
            conv = _conv_fft(a, a)
            # compute softmax weights
            shift = np.max(conv)
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0.0:
                obj = 2.0 * n * shift
                grad_theta = np.zeros_like(theta)
                return obj, grad_theta
            w /= w_sum
            obj = 2.0 * n * np.dot(w, conv)   # surrogate

            # gradient wrt a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            dot = np.dot(grad_a, a)
            grad_theta = a * (grad_a - dot)
            return obj, grad_theta

        # Use SciPy minimize
        import scipy.optimize
        # we pass Jacobian via return tuple
        res = scipy.optimize.minimize(fun, v, method='L-BFGS-B',
                                      jac=True,
                                      options={'maxiter': maxiter_per_beta, 'disp': False})
        v = res.x
        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # report
            print(f"[L-BFGS beta={beta}] best surrogate={cur_val:.6f}")
    return best_seq, best_val

"""Search for step-function coefficients minimizing evaluate_sequence

The algorithm:
  1. Generate a diverse set of ℓ1‑normalised candidate shapes (analytic windows,
     Dirichlet draws, low‑frequency random curves, mixtures, previous solution).
  2. Keep the best few seeds w.r.t. the true objective.
  3. For each seed run a continuation of smooth convex optimisation:
     - Parametrise the vector as a soft‑max of logits (guarantees non‑negativity
       and sum‑to‑1).
     - Use a log‑sum‑exp surrogate for the max‑convolution; its gradient
       is computed analytically.
     - Minimise the surrogate with SciPy's L‑BFGS‑B, gradually increasing the
       smoothing temperature β (0.5 → 200).  This yields fast convergence on the
       convex problem ``min_ a  max_k a*a (k)``.
  4. Refine the best result with a few hundred thousand exact sub‑gradient steps
     on the true max‑convolution (projected onto the probability simplex).
  5. Finally run a short simulated‑annealing polishing pass.
  6. Return the best sequence found (list of floats)."""

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (complexity O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) – the true objective when sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic shape families (all return ℓ₁‑normalised vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) -
                                            2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    if _scipy_kaiser is not None:
        a = _scipy_kaiser(n, beta)
        s = a.sum()
        return a / s if s > 0 else _uniform(n)
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    if _scipy_i0 is not None:
        a = _scipy_i0(beta * np.sqrt(1 - y * y))
    else:
        t = beta * np.sqrt(1 - y * y)
        a = 1.0 + (t ** 2) / 4.0 + (t ** 4) / 64.0 + (t ** 6) / 2304.0 + (t ** 8) / 147456.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; result is ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _plateau(n: int, plateau_frac: float = 0.4) -> np.ndarray:
    """Linear ramp up, flat plateau, linear ramp down."""
    if plateau_frac <= 0 or plateau_frac >= 1:
        return _uniform(n)
    ramp_len = int((1 - plateau_frac) * n / 2)
    plateau_len = n - 2 * ramp_len
    a = np.concatenate([
        np.linspace(0.0, 1.0, ramp_len, endpoint=False),
        np.ones(plateau_len),
        np.linspace(1.0, 0.0, ramp_len, endpoint=False)
    ])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _collect_seeds_extended(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Gather a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.15, 0.45, 7):
        seeds.append(_gaussian(n, sigma))

    # Power‑law family
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta family grid
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Plateau shapes
    for frac in (0.2, 0.4, 0.6):
        seeds.append(_plateau(n, plateau_frac=frac))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential (heavy‑tailed)
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with a low‑frequency curve
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # Normalise everything and guard against zero vectors
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

# Optimisation primitives

def _lbfgs_continuation(v0: np.ndarray, n: int, betas: list[float], deadline: float,
                        maxiter_per_beta: int = 40) -> tuple[np.ndarray, float]:
    """
    Smooth convex optimisation by L‑BFGS‑B on the soft‑max parametrisation.
    Starts from logits ``v0`` (log of a probability vector) and successively
    increases the smoothing temperature ``beta``. Returns the best vector found
    (ℓ₁‑normalised) and its true objective value.
    """
    v = v0.copy()
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for beta in betas:
        if time.time() > deadline:
            break

        # define objective+gradient for SciPy
        def fun(theta: np.ndarray):
            a = _softmax(theta)                           # probabilities
            conv = _conv_fft(a, a)                         # length 2n‑1
            shift = np.max(conv)
            # smoothed weights (soft‑max over convolution values)
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0.0:
                obj = 2.0 * n * shift
                grad_theta = np.zeros_like(theta)
                return obj, grad_theta
            w = w / w_sum
            # surrogate objective (weighted average of conv entries)
            obj = 2.0 * n * np.dot(w, conv)

            # gradient wrt a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                     # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]           # central n entries

            # gradient wr.t. logits (softmax Jacobian)
            dot = np.dot(grad_a, a)
            grad_theta = a * (grad_a - dot)
            return obj, grad_theta

        # Run L‑BFGS‑B with a modest iteration budget
        res = scipy.optimize.minimize(fun, v,
                                      method='L-BFGS-B',
                                      jac=True,
                                      options={'maxiter': maxiter_per_beta,
                                               'disp': False})
        v = res.x
        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L‑BFGS β={beta:.1f}] new best surrogate = {cur_val:.6f}")

    if best_seq is None:
        # fallback – return the softmax of the final logits
        best_seq = _softmax(v)
        best_val = _objective_surrogate(best_seq)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Uses a decreasing step‑size 0.2 / sqrt(t) and Euclidean projection onto
    the probability simplex after each step.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 400_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Aggressive annealing that reshuffles mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9999, min_temp)

        if rng.random() < 0.7:
            k = rng.integers(2, 12)          # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

"""
Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm outline
-----------------
1. **Rich seed pool** – deterministic windows (uniform, triangular,
   sine, Kaiser, Gaussian, beta, Tukey, plateau, …) plus random Dirichlet,
   exponential and low‑frequency cosine curves.  All candidates are
   ℓ₁‑normalised (sum = 1).

2. **Continuation L‑BFGS** – parametrize a vector as a soft‑max of logits
   (guarantees non‑negativity and the simplex).  For a temperature β the
   objective is the log‑sum‑exp surrogate of the max‑convolution:
   `obj(θ) = 2·n·∑ w_k·conv_k`, where
   `w = softmax(β·conv)`.  The gradient is derived analytically:
   `∇_θ obj = a·(∇_a obj − (∇_a obj·a))`.  Starting from β≈0.5 we
   increase β geometrically up to 200, optimising each stage with
   SciPy’s L‑BFGS‑B (few tens of iterations – fast because each step needs
   only two FFT‑convolutions).

3. **Exact sub‑gradient refinement** – after L‑BFGS we perform a modest
   projected sub‑gradient descent on the true max‑convolution objective
   (convex, because each quadratic form is PSD).  The step size decays as
   `0.2/√t`.

4. **Simulated annealing polishing** – an aggressive annealing that reshuffles
   mass on random subsets and accepts uphill moves with a slow cooling schedule.

The routine respects the global deadline (default 1000 s) and returns the
best sequence found as a plain ``list[float]``.  The final sequence is also
stored in the global name ``height_sequence_1`` for possible reuse.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) – the true objective when sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) -
                                            2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    if _scipy_kaiser is not None:
        a = _scipy_kaiser(n, beta)
        s = a.sum()
        return a / s if s > 0 else _uniform(n)

    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    if _scipy_i0 is not None:
        a = _scipy_i0(beta * np.sqrt(1 - y * y))
    else:
        t = beta * np.sqrt(1 - y * y)
        a = 1.0 + (t ** 2) / 4.0 + (t ** 4) / 64.0 + (t ** 6) / 2304.0 + (t ** 8) / 147456.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; result is ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _plateau(n: int, plateau_frac: float = 0.4) -> np.ndarray:
    """Linear ramp‑up, flat plateau, linear ramp‑down; ℓ₁‑normalised."""
    if plateau_frac <= 0 or plateau_frac >= 1:
        return _uniform(n)
    ramp_len = int((1 - plateau_frac) * n / 2)
    plateau_len = n - 2 * ramp_len
    a = np.concatenate([
        np.linspace(0.0, 1.0, ramp_len, endpoint=False),
        np.ones(plateau_len),
        np.linspace(1.0, 0.0, ramp_len, endpoint=False)
    ])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _collect_seeds_extended(n: int, rng: np.random.Generator,
                           previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.15, 0.45, 7):
        seeds.append(_gaussian(n, sigma))

    # Power‑law family
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta family grid
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Plateau shapes
    for frac in (0.2, 0.4, 0.6):
        seeds.append(_plateau(n, plateau_frac=frac))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential (heavy‑tailed)
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with a low‑frequency curve
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # Normalise everything and guard against zero vectors
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _lbfgs_continuation(v0: np.ndarray, n: int, betas: list[float],
                        deadline: float, maxiter_per_beta: int = 40
                        ) -> tuple[np.ndarray, float]:
    """
    Smooth optimisation via L‑BFGS‑B on the log‑softmax parametrisation.
    Starts from logits ``v0`` and runs a temperature schedule given by ``betas``.
    Returns the best vector (ℓ₁‑normalised) found and its true objective.
    """
    v = v0.copy()
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for beta in betas:
        if time.time() > deadline:
            break

        def fun(theta: np.ndarray):
            a = _softmax(theta)                     # simplex vector
            conv = _conv_fft(a, a)                  # length 2n‑1
            shift = np.max(conv)
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0.0:
                obj = 2.0 * n * shift
                grad_theta = np.zeros_like(theta)
                return obj, grad_theta
            w = w / w_sum
            obj = 2.0 * n * np.dot(w, conv)         # smoothed surrogate

            # gradient wrt a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)              # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_theta = a * (grad_a - dot)
            return obj, grad_theta

        res = scipy.optimize.minimize(fun, v,
                                      method='L-BFGS-B',
                                      jac=True,
                                      options={'maxiter': maxiter_per_beta,
                                               'disp': False})
        v = res.x
        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L‑BFGS β={beta:.1f}] new best surrogate = {cur_val:.6f}")

    if best_seq is None:
        best_seq = _softmax(v)
        best_val = _objective_surrogate(best_seq)
    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    After each step we project onto the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.20 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 400_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Aggressive annealing that reshuffles mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9999, min_temp)

        if rng.random() < 0.7:                     # reshuffle a random subset
            k = rng.integers(2, 12)                # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                                      # tiny pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats; the result is also
    stored in the global name ``height_sequence_1`` for later reuse.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for final checks

    # ------------------------------------------------------------------
    # Choose vector length – 4000 offers a good balance between expressivity
    # and computational cost.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Generating seeds (n={n}) ---")
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Total seeds generated: {len(seeds)}")

    # Evaluate surrogate (cheap) and keep the top performers
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 8
    best_seeds = [vec for vec, _ in evals[:top_k]]

    print(f"\nTop {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS optimisation
    # ------------------------------------------------------------------
    betas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0]

    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break

        print(f"\n--- L‑BFGS run {idx+1}/{len(best_seeds)} ---")
        v0 = np.log(seed_vec + 1e-30)          # logits for soft‑max
        seq_opt, val_opt = _lbfgs_continuation(v0, n, betas,
                                               deadline,
                                               maxiter_per_beta=50)
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_seq = seq_opt.copy()
            print(f"[Overall] New global best after L‑BFGS = {val_opt:.6f}")

        # ------------------------------------------------------------------
        # Short exact sub‑gradient refinement
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 0:
            max_iters_sub = int(min(800_000, remaining * 8_000))  # heuristic
            if max_iters_sub > 0:
                seq_proj, val_proj = _subgradient_proj(global_best_seq, n,
                                                       deadline,
                                                       max_iters=max_iters_sub)
                if val_proj < global_best_val:
                    global_best_val = val_proj
                    global_best_seq = seq_proj.copy()
                    print(f"[Overall] New global best after sub‑proj = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(global_best_seq, n, deadline)
        if val_sa < global_best_val:
            global_best_val = val_sa
            global_best_seq = seq_sa.copy()
            print(f"[Overall] Final global best after SA = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
          f"true evaluate_sequence = {true_val}")

    # store for possible reuse in subsequent runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
