def solve_lp_given_f(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(g) == 1]
    # convolution constraints
    # Build linear constraints for each k:
    for k in range(2*n - 1):
        # sum_{i} f[i] * g[k-i] <= M
        # i runs from max(0, k-(n-1)) to min(k, n-1)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # Build expression
        expr = cp.sum(cp.multiply(f[i_start:i_end+1], g[k - np.arange(i_start, i_end+1)]))

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
    return g.value, prob.value

"""
Improved search for a non‑negative step‑function that minimises
``evaluate_sequence``.

The algorithm proceeds in three stages:

1. **Diverse seed pool** – analytic families (uniform, triangular,
   trapezoidal, geometric, power‑law, beta, Gaussian, Tukey, sine) are
   complemented by many random constructions:
   Dirichlet draws with a wide range of concentrations, sparse vectors
   with a few random non‑zero entries and mixtures of two Beta shapes.
   All seeds are ℓ¹‑normalised and the best few (by the exact surrogate
   2·n·max(conv)) are kept.

2. **Smooth optimisation** – the retained seeds are refined with a
   Adam optimiser that works on the logits of the soft‑max
   parametrisation.  The loss is a log‑sum‑exp surrogate of the
   convolution, with a temperature that is annealed up to a large
   value (β≈3000) so that the surrogate closely follows the true
   max‑convolution.  A generous iteration budget is used while respecting
   the global time limit.

3. **Fine polishing** – the best candidate from Adam is further
   improved by projected sub‑gradient descent on the genuine max‑convolution
   objective, followed by a short simulated‑annealing phase that
   redistributes mass among random coordinate subsets.

The routine returns the best sequence found as a plain Python list.
It also stores the result in the global variable ``height_sequence_1``
so that later runs can reuse it as a warm start.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate used during the search (ℓ₁‑normalised vectors)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – stays on the simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_mix(n: int, a1: float, b1: float, a2: float, b2: float, p: float) -> np.ndarray:
    """Mixture of two Beta distributions."""
    w1 = np.clip(p, 0.0, 1.0)
    w2 = 1.0 - w1
    v1 = _beta_shape(n, a1, b1)
    v2 = _beta_shape(n, a2, b2)
    a = w1 * v1 + w2 * v2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_sparse(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k random non‑zero positions, positive random weights, ℓ₁‑normalised."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k) + 1e-12  # avoid exact zeros
    seq = np.zeros(n, dtype=np.float64)
    seq[idx] = vals
    s = seq.sum()
    return seq / s if s > 0 else _uniform(n)

# Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 500_000,
                   beta_max: float = 2000.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    Parameters are logits of the soft‑max representation (ensuring ℓ₁‑norm 1).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # initialise logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.001

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:,} best={best_val:.6f}")

        # increase temperature (sharper approximation of max)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of weighted sum Σ w_k·conv_k  w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)            # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip gradient norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 2_000_000,
                      init_lr: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses a Polyak‑style diminishing step size.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:,} best={cur_val:.6f}")

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size
        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 600_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing that redistributes mass among a random
    subset of coordinates (Dirichlet reshuffle).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:,} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------


def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a heterogeneous pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for wf in np.linspace(0.05, 0.40, 8):
        seeds.append(_trapezoidal(n, width_fraction=wf))

    for gamma in np.linspace(0.0, 2.5, 9):
        seeds.append(_sine_window(n, gamma=gamma))

    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    r_vals = np.concatenate([
        np.linspace(0.2, 0.98, 12),
        np.linspace(0.98, 0.9999, 10)
    ])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 18):
        seeds.append(_power(n, p))

    for a in np.linspace(0.1, 3.0, 6):
        for b in np.linspace(0.1, 3.0, 6):
            seeds.append(_beta_shape(n, a, b))

    for sigma_frac in (0.12, 0.18, 0.24, 0.30, 0.36):
        seeds.append(_gaussian(n, sigma_frac))

    # random Dirichlet draws – wide concentration range
    for conc in np.logspace(-3.0, 1.0, 10):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # random sparse vectors
    for k in (30, 50, 80, 120, 180, 250):
        for _ in range(3):
            seeds.append(_random_sparse(n, k, rng))

    # mixtures of two beta families
    for _ in range(15):
        a1, b1, a2, b2 = rng.uniform(0.1, 5.0, 4)
        p = rng.random()
        seeds.append(_beta_mix(n, a1, b1, a2, b2, p))

    # shuffles of a few existing seeds – break symmetry
    for s in seeds[:20]:
        if rng.random() < 0.15:
            perm = rng.permutation(n)
            seeds.append(s[perm])

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # Retrieve a previously found solution (if present)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length choice – a larger n gives a finer discretisation.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Stage 1 – generate a large, diverse seed pool and retain the best.
    # ------------------------------------------------------------------
    print(f"\n--- Collecting seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = []
    for vec in seeds:
        val = _objective(vec)
        evaluated.append((val, vec))

    evaluated.sort(key=lambda x: x[0])
    top_k = min(12, len(evaluated))
    best_seeds = [vec for _, vec in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (val, _) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # Initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – intensive Adam optimisation on the most promising seeds.
    # ------------------------------------------------------------------
    # Allocate time for Adam runs; we use at most two seeds (most expensive).
    for idx, seed_vec in enumerate(best_seeds[:2]):
        if time.time() > deadline:
            break
        print(f"\n--- Adam optimisation run {idx+1}/{2} ---")
        # Estimate a generous iteration budget based on remaining time
        remaining = deadline - time.time()
        est_iters = int(remaining * 1500)           # ~1500 it/s on typical hardware
        max_iters = max(300_000, min(est_iters, 1_200_000))
        seq_adam, val_adam = _adam_optimize(
            seed_vec, n, deadline,
            max_iters=max_iters,
            beta_max=3000.0
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # early exit if we already reached the target
        if global_best_val <= 1.503:
            print("Target reached during Adam stage.")
            break

    # ------------------------------------------------------------------
    # Stage 3 – refined projected sub‑gradient descent.
    # ------------------------------------------------------------------
    if time.time() < deadline and global_best_val > 1.503:
        remaining = deadline - time.time()
        max_iters = int(remaining * 1500)
        max_iters = max(500_000, min(max_iters, 3_000_000))
        print(f"\n--- Sub‑gradient refinement (max_iters={max_iters}) ---")
        seq_proj, val_proj = _subgradient_proj(
            global_best_seq, n, deadline,
            max_iters=max_iters,
            init_lr=0.30
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[overall‑SubProj] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final simulated‑annealing polish.
    # ------------------------------------------------------------------
    if time.time() < deadline and global_best_val > 1.503:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=600_000,
            init_temp_factor=0.06
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Sanity check – evaluate with the official function.
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for later reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
