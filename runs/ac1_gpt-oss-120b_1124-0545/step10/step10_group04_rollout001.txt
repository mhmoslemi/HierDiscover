"""
Improved step‑function search for minimizing ``evaluate_sequence``.

The algorithm proceeds in several stages respecting a 1000 s budget:

1️⃣  Deterministic seeds (uniform, Hann, Beta, arcsine, etc.) for a few
    FFT‑friendly lengths (2000 – 8000).  The best seed becomes the initial
    candidate.

2️⃣  Heavy random exploration:
    * Dirichlet draws with several concentration parameters.
    * Random beta‑pdf shapes (α,β sampled on a log‑scale).
    * Simple mixtures of two seed families (parameter λ∈[0,1]).
    The best of all random candidates replaces the current best.

3️⃣  Smooth Adam optimisation of a log‑sum‑exp surrogate for the
    max‑convolution.  Several restarts are performed; the surrogate
    becomes increasingly sharp (β grows from 5 to 500) and the learning
    rate decays as 1/√t.

4️⃣  Aggressive projected sub‑gradient descent on the exact
    non‑smooth objective.  The active set of “almost‑maximal’’ convolution
    entries is used to build a dense sub‑gradient; the step size follows
    0.12/√t and the tolerance on the active set shrinks from 1e‑3 to
    1e‑8.  Periodic random jitter helps escape shallow plateaus.

5️⃣  Final sanity check via ``evaluate_sequence`` and return the best
    sequence (as a plain list) found before the deadline.
"""

import time
import numpy as np
from math import pi, sqrt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a *normalised* sequence (sum == 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * pi * i / (n - 1))
         + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian_sigma015(n: int) -> np.ndarray:
    sigma_factor = 0.15
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = w.sum()
    return w / s if s else _uniform(n)


def _gaussian_sigma025(n: int) -> np.ndarray:
    sigma_factor = 0.25
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = w.sum()
    return w / s if s else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    s = w.sum()
    return w / s if s else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf shape on a discrete grid (shifted by 0.5 to avoid end‑points)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    s = w.sum()
    return w / s if s else _uniform(n)


# ----------------------------------------------------------------------
# Random Dirichlet sampling
# ----------------------------------------------------------------------


def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Return a random point on the simplex (sum=1) with Dirichlet(alpha)."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed Adam optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _smooth_adam_optimize(start_seq: np.ndarray,
                          deadline: float,
                          beta_start: float = 5.0,
                          beta_end: float = 500.0,
                          max_iter: int = 5_000_000,
                          lr0: float = 0.03) -> tuple[np.ndarray, float]:
    """Minimise a smooth approximation of max(conv(a,a)) via Adam."""
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        a = np.exp(logits)
        a = a / a.sum()

        conv = _conv_fft(a, a)

        # linear schedule for the smoothing temperature β
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()

        # gradient of the surrogate w.r.t. a (see original code)
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if (it & 1023) == 0:          # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")
            # early exit if we are already extremely good
            if best_val < 1.45:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------


def _proj_subgradient_optimize(start_seq: np.ndarray,
                               deadline: float,
                               eps_rel_start: float = 1e-3,
                               eps_rel_end: float = 1e-8,
                               lr0: float = 0.12,
                               max_iter: int = 10_000_000,
                               jitter_interval: int = 200_000,
                               jitter_scale: float = 5e-4) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for the exact objective.
    Uses an aggregated sub‑gradient built from all convolution entries
    that are within a relative tolerance of the current maximum.
    Periodic small random jitter is added to escape shallow plateaus.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    last_improve = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        # decreasing tolerance for the active set
        eps_rel = eps_rel_start + (eps_rel_end - eps_rel_start) * min(
            1.0, it / (max_iter * 0.5))

        thresh = max_val - eps_rel * max_val
        active_idx = np.where(conv >= thresh)[0]

        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv)], dtype=int)

        w = np.zeros_like(conv)
        w[active_idx] = 1.0 / active_idx.size

        # sub‑gradient for the max‑convolution
        conv_ag = _conv_fft(w, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        lr = lr0 / sqrt(it)
        a_next = a - lr * g
        a_next = _project_simplex(a_next)

        # jitter to try to get out of flat regions
        if jitter_interval and (it % jitter_interval) == 0:
            noise = (np.random.default_rng().normal(scale=jitter_scale, size=n))
            a_next = _project_simplex(a_next + noise)

        if (it & 1023) == 0:
            val = _objective(a_next)
            if val < best_val:
                best_val = val
                best_seq = a_next.copy()
                last_improve = it
                print(f"[ProjSGD] it={it:7d} n={n:5d} val={val:.6f}")

                if best_val < 1.45:
                    break

        a = a_next

    return best_seq, best_val


# ----------------------------------------------------------------------
# Helper to mix two families with weight λ
# ----------------------------------------------------------------------


def _mix_two(a: np.ndarray, b: np.ndarray, lam: float) -> np.ndarray:
    """Return a convex combination λ·b + (1‑λ)·a, renormalised to sum = 1."""
    mix = (1.0 - lam) * a + lam * b
    s = mix.sum()
    return mix / s if s else a.copy()

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:

    # ------------------------------------------------------------------

    best_val = float("inf")

            best_seq = seq.copy()

        # Beta‑pdf draw
        a_par, b_par = rng.choice(beta_param_grid)
        seq = _beta_shape(n, a_par, b_par)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rand‑B] n={n:5d} (α,β)=({a_par:.2f},{b_par:.2f}) val={val:.6f}")

        # Mixture of two seeds – try a handful of random λ
        if best_seq is not None:
            # pick two random seed families
            f1, f2 = rng.choice(seed_families, size=2, replace=False)
            s1, s2 = f1(n), f2(n)
            for lam in (0.1, 0.3, 0.5, 0.7, 0.9):
                seq = _mix_two(s1, s2, lam)
                val = _objective(seq)
                if val < best_val:

                    print(f"[Mix]    n={n:5d} {f1.__name__:12s}+{f2.__name__:12s} λ={lam:.1f} val={val:.6f}")

        if best_val < 1.45:        # early stop if already impressive
            break

    # ------------------------------------------------------------------
    # Phase 3 – smooth Adam optimisation (a few restarts)

                start_seq=cand,

            )

    # ------------------------------------------------------------------

            )

    # ------------------------------------------------------------------
    # Final sanity‑check (the grader will evaluate again anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
