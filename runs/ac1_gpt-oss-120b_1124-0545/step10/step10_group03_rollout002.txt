"""
Improved search for a low‑value step‑function.

The algorithm proceeds in four phases within the allotted time budget.

1. **Extensive seed generation** – a large, diverse pool of ℓ₁‑normalised
   candidates is built from analytic families (uniform, sine‑windows,
   Gaussians, Beta‑shapes, geometric, triangular, trapezoidal) as well as
   random Dirichlet and sparse draws.  The 30 best seeds (by the exact
   objective) are retained.

2. **Medium‑length Adam refinements** – each retained seed is fed to an
   Adam optimiser that minimises a smooth log‑sum‑exp surrogate of the
   max‑convolution.  The temperature is slowly increased up to a large
   value (≈1000) so the surrogate approaches the true max.  The best
   result of this stage becomes the global incumbent.

3. **Intensive local optimisation** – the incumbent is improved further
   by (a) a long Adam run, (b) projected sub‑gradient descent on the
   exact max‑convolution objective, and (c) a short simulated‑annealing
   polishing step.

4. **Final hill‑climbing polishing** – a simple random‑reallocation
   neighbourhood is explored for a few seconds; any improvement is
   kept.

All heavy lifting (FFT convolution, softmax, simplex projection) is
implemented with NumPy; the code respects the time budget (default
1000 s) and stores the best sequence in the global variable
``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a sequence that already sums to 1.
    Returns 2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (probability vector, sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all ℓ₁‑normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    k = rng.integers(3, 31)  # 3 … 30 non‑zero entries
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


def _mixture_uniform_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.random()
    g = _gaussian(n, sigma_frac=rng.uniform(0.05, 0.4))
    u = _uniform(n)
    a = lam * u + (1.0 - lam) * g
    return a / a.sum()

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 500_000,
                   beta_max: float = 1000.0,
                   beta_mul: float = 1.0005,
                   lr0: float = 0.8) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on the log‑sum‑exp surrogate.
    ``start_vec`` must already sum to 1.
    Returns (best_sequence, best_value) where best_value is the *exact*
    objective (not the surrogate).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0                       # initial smoothing
    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                   # current probability vector
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        # raise temperature slowly (approaches true max)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # smooth‑max of the convolution
        conv = _conv_fft(a, a)            # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                       # softmax over convolution entries

        # gradient w.r.t. a of Σ w_k * conv_k :
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)       # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # transform gradient to logits space (log‑softmax)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping (stability)
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update (decaying learning rate)
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *exact* max‑convolution objective.
    Uses a Polyak‑style step size based on the best value seen so far.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient corresponding to the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step (uses current best)
        grad_norm_sq = np.dot(g, g) + 1e-12
        step = (cur_val - best_val) / grad_norm_sq
        if step <= 0.0:
            step = 0.1 / np.sqrt(it)

        a = a - step * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 300_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Very light simulated‑annealing: repeatedly pick a small random
    subset of indices and reshuffle the total mass among them.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)                 # size of the moved block
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _random_neighbour(seq: np.ndarray, rng: np.random.Generator,
                     max_block: int = 8) -> np.ndarray:
    """
    Generate a neighbouring sequence by moving a small amount of mass
    between a randomly chosen block of coordinates.
    """
    n = seq.shape[0]
    k = rng.integers(2, max_block + 1)
    idx = rng.choice(n, size=k, replace=False)
    total = seq[idx].sum()
    if total <= 0.0:
        return seq
    new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
    new = seq.copy()
    new[idx] = new_alloc
    return new

def _collect_diverse_seeds(n: int, rng: np.random.Generator,
                          previous: np.ndarray | None,
                          top_k: int,

    """
    Build a large pool of candidate sequences and keep the ``top_k``
    with the smallest exact objective.
    """
    top_vals = []        # list of (value, seq)

    def try_insert(val, seq):
        if len(top_vals) < top_k:
            top_vals.append((val, seq))
            top_vals.sort(key=lambda x: x[0])
        elif val < top_vals[-1][0]:
            top_vals[-1] = (val, seq)
            top_vals.sort(key=lambda x: x[0])

    # Deterministic analytic families (always evaluated)
    families = [
        _uniform,
        _triangular,
        lambda n: _trapezoidal(n, width_fraction=0.15),
        lambda n: _trapezoidal(n, width_fraction=0.30),
        lambda n: _sine_window(n, gamma=0.2),
        lambda n: _sine_window(n, gamma=0.5),
        lambda n: _sine_window(n, gamma=1.0),
        lambda n: _sine_window(n, gamma=2.0),
        lambda n: _gaussian(n, sigma_frac=0.07),
        lambda n: _gaussian(n, sigma_frac=0.12),
        lambda n: _gaussian(n, sigma_frac=0.20),
        lambda n: _tukey_window(n, alpha=0.1),
        lambda n: _tukey_window(n, alpha=0.3),
        lambda n: _tukey_window(n, alpha=0.6),
    ]
    for fn in families:
        seq = fn(n)
        val = _objective(seq)
        try_insert(val, seq)

    # Geometric families
    for r in np.linspace(0.2, 0.95, 12):
        for fn in (_geom, _sym_geom):
            seq = fn(n, r)
            val = _objective(seq)
            try_insert(val, seq)

    # Power‑law family
    for p in np.linspace(0.0, 5.0, 20):
        seq = _power(n, p)
        val = _objective(seq)
        try_insert(val, seq)

    # Random Dirichlet draws with diverse concentrations

            try_insert(val, seq)

    # Sparse supports

        seq = _sparse_seed(n, rng)
        val = _objective(seq)
        try_insert(val, seq)

    # Random beta shapes

        if time.time() > deadline:

        alpha = rng.uniform(0.05, 5.0)
        beta = rng.uniform(0.05, 5.0)
        seq = _beta_shape(n, alpha, beta)
        val = _objective(seq)
        try_insert(val, seq)

    # Random sine exponents
    for _ in range(200):
        if time.time() > deadline:
            break
        gamma = rng.uniform(0.1, 5.0)
        seq = _sine_window(n, gamma=gamma)
        val = _objective(seq)
        try_insert(val, seq)

    # Random Gaussian widths

        if time.time() > deadline:
            break
        sigma_frac = rng.uniform(0.05, 0.30)
        seq = _gaussian(n, sigma_frac=sigma_frac)
        val = _objective(seq)
        try_insert(val, seq)

    # Random Tukey alphas

        alpha = rng.uniform(0.05, 0.9)
        seq = _tukey_window(n, alpha=alpha)

        seq = _mixture_uniform_gaussian(n, rng)

        try_insert(val, seq)

    # Previous best (if any)
    if previous is not None and previous.shape == (n,):
        seq = np.asarray(previous, dtype=np.float64)
        if seq.sum() > 0:
            seq = seq / seq.sum()
            val = _objective(seq)
            try_insert(val, seq)

    # Return only the sequences (sorted by increasing objective)
    return [seq for _, seq in top_vals]

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget and stores the best
    sequence in the global variable ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    # ------------------------------------------------------------------
    # Length – 5000 offers a good trade‑off between resolution and speed.
    # ------------------------------------------------------------------
    n = 5000
    print(f"\n=== Starting search (n={n}, budget={budget_s}s) ===")

    # ------------------------------------------------------------------
    # Existing solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 0 – seed generation (≈30 % of the time)
    # ------------------------------------------------------------------
    seed_deadline = min(deadline, start_time + budget_s * 0.30)
    print("\n--- Phase 0: seed generation ---")
    seeds = _collect_diverse_seeds(n, rng, previous,
                                   top_k=30, deadline=seed_deadline)
    print(f"Collected {len(seeds)} seeds (kept top‑30).")

    # ------------------------------------------------------------------
    # Phase 1 – short Adam runs on each seed
    # ------------------------------------------------------------------
    print("\n--- Phase 1: short Adam refinements ---")
    best_seq = None
    best_val = float("inf")
    for idx, seed_vec in enumerate(seeds):
        if time.time() > deadline:
            break
        print(f"[Seed {idx+1}/{len(seeds)}]")
        seq_adam, val_adam = _adam_optimize(
            start_vec=seed_vec,
            n=n,
            deadline=deadline,
            max_iters=200_000,
            beta_max=500.0,
            beta_mul=1.001,
            lr0=0.8,
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  New global best after short Adam: {best_val:.6f}")

    # Fallback – should never happen
    if best_seq is None:
        best_seq = seeds[0].copy()
        best_val = _objective(best_seq)
        print(f"Fallback to first seed: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long Adam optimisation on the incumbent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 2: long Adam optimisation ---")
        seq_adam, val_adam = _adam_optimize(
            start_vec=best_seq,
            n=n,
            deadline=deadline,
            max_iters=1_200_000,
            beta_max=1500.0,
            beta_mul=1.0007,
            lr0=0.6,
        )
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"  Updated global best after long Adam: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 3: projected sub‑gradient descent ---")
        seq_sub, val_sub = _subgrad_optimize(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=800_000,
        )
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"  Updated global best after sub‑grad: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 4: simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=400_000,
            init_temp_factor=0.06,
        )
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"  Updated global best after SA: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final random‑neighbour hill‑climbing (few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 5: final hill‑climbing polish ---")
        cur = best_seq.copy()
        cur_val = _objective(cur)
        best_local = cur.copy()
        best_local_val = cur_val
        max_hc_iters = 150_000
        for it in range(1, max_hc_iters + 1):
            if time.time() > deadline:
                break
            new = _random_neighbour(cur, rng, max_block=6)
            new_val = _objective(new)
            if new_val < cur_val:
                cur = new
                cur_val = new_val
                if new_val < best_local_val:
                    best_local = new.copy()
                    best_local_val = new_val
            if it % 20_000 == 0:
                print(f"[HC] it={it:6d} best_local={best_local_val:.6f}")

        if best_local_val < best_val:
            best_seq = best_local
            best_val = best_local_val
            print(f"  Updated global best after hill‑climbing: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check using the official evaluator
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )

    # Store the result for possible reuse in a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
