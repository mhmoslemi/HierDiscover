def _proj_simplex(v):
    # ensures v's entries sum to 1 and are nonnegative
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Optimisation of a non‑negative step function for the objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Initial seeding** – several deterministic and random simple
   probability vectors are generated (uniform, random soft‑max, exponential,
   triangular).  The best one with respect to the true objective is kept.

2. **Adam on the simplex** – the probabilities are parametrised by a
   log‑soft‑max (so the sum‑to‑one constraint is automatically satisfied).
   A smooth approximation of the maximum (soft‑max with an “inverse‑temperature”
   λ) is gradually sharpened while Adam updates the log‑weights.  The schedule
   of λ is very slow (λ←λ·1.0005) so that the optimiser first sees a smooth
   landscape and only at the end focuses on the true non‑smooth objective.
   The optimiser runs until the time budget is exhausted (or a hard iteration
   limit is hit).

3. **Hard‑max projected gradient descent** – a short refinement that uses the
   exact sub‑gradient of the true max‑convolution (λ=0).  After each gradient
   step the iterate is projected back onto the probability simplex.

All heavy work (convolutions, gradients) is performed with FFTs for
O(n log n) speed.  The routine is completely self‑contained, respects the
global 1000 s budget and stores the best sequence in the global variable
``height_sequence_1``.  The final list of floats is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective (exact evaluation, same as evaluate_sequence)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective value for a probability vector (sum(seq)=1)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple seeds (return a probability vector that sums to 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    exps = np.exp(logits - m)
    return exps / exps.sum()


def _exponential_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        dec = inc[::-1]
    else:
        dec = np.arange(half, 0, -1, dtype=np.float64)
    a = np.concatenate([inc, dec])
    return a / a.sum()


def _gaussian_blob(n: int, sigma: float = 0.2) -> np.ndarray:
    x = np.linspace(-0.5, 0.5, n, dtype=np.float64)
    a = np.exp(-0.5 * (x / sigma) ** 2)
    return a / a.sum()


# ----------------------------------------------------------------------
# Soft‑max weights over the convolution entries
# ----------------------------------------------------------------------
def _soft_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Return a probability vector w (sums to 1) that is a soft‑max over `conv`.
    For lam<=0 a hard max (uniform over ties) is used.
    """
    max_c = conv.max()
    if lam <= 0.0:
        active = np.isclose(conv, max_c, atol=1e-12)
        cnt = np.count_nonzero(active)
        if cnt == 0:
            return np.full_like(conv, 1.0 / conv.size)
        return np.where(active, 1.0 / cnt, 0.0)

    rel = (conv - max_c) / max_c  # ≤ 0
    exps = np.exp(lam * rel)
    w = exps / exps.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of the (smoothed) max‑convolution w.r.t. a
# ----------------------------------------------------------------------
def _gradient_wrt_a(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the λ‑smoothed max‑convolution with respect to a.
    Implements   g = 2 * rev(a) ⊛ w   where w are the (soft) max‑weights.
    """
    n = a.size
    conv = _conv_fft(a, a)
    w = _soft_max_weights(conv, lam)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length = 3n‑2
    grad = 2.0 * full[n - 1 : n - 1 + n]      # slice that aligns with a
    return grad


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Duchi et al.)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety: return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 500_000,
                   lam_start: float = 0.1,
                   lam_factor: float = 1.0005,
                   lr_const: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑probabilities (soft‑max) of `start`.
    A smooth max is sharpened by gradually increasing `lam`.
    Returns best probability vector found and its exact objective value.
    """
    eps = 1e-30
    logp = np.log(start + eps)          # shape (n,)
    n = start.size
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = start.copy()
    best_val = _objective(start)

    t = 0
    lam = lam_start

    while t < max_iters and time.time() < deadline:
        t += 1

        # current probability vector (soft‑max, numerically stable)
        logp_shift = logp - np.max(logp)
        a = np.exp(logp_shift)
        a /= a.sum()

        # gradient of the smoothed objective
        g_a = _gradient_wrt_a(a, lam)

        # chain rule to log‑probabilities (softmax Jacobian)
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)

        # Adam updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log * g_log)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_const / (math.sqrt(t) + 1.0)

        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # exact objective (hard max) for bookkeeping
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:7d}  best={cur_val:.6f}  λ={lam:.2f}")

        # increase smoothing parameter slowly
        lam *= lam_factor

        # occasional tiny Gaussian kick to escape plateaus
        if t % 25000 == 0:
            noise = np.random.default_rng().normal(scale=0.02, size=n)
            logp += noise

    return best_seq, best_val


# ----------------------------------------------------------------------
# Hard‑max projected gradient refinement (exact sub‑gradient)
# ----------------------------------------------------------------------
def _hard_refine(start: np.ndarray,
                 deadline: float,
                 max_iters: int = 5000,
                 eta0: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent using the true sub‑gradient (λ=0).
    Very short but often reduces the final objective noticeably.
    """
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        grad = _gradient_wrt_a(cur, lam=0.0)   # exact max‑gradient
        eta = eta0 / math.sqrt(it + 1)
        cur = _proj_simplex(cur - eta * grad)
        cur_val = _objective(cur)
        if cur_val < best_val:
            best_val = cur_val
            best = cur.copy()
            print(f"[HardRefine] it={it+1:5d}  best={best_val:.6f}")

    return best, best_val
