"""
Propose a non‑negative step function that improves the bound on
``evaluate_sequence``.

The algorithm consists of three stages:

1. **Diverse seed generation** – a large pool of analytically‑defined
   windows (triangular, Tukey, geometric, power‑law, Beta, Gaussian,
   Hann, Hamming, Blackman, …) plus many random draws (Dirichlet,
   exponentiated uniforms, smoothed noises).  All vectors are ℓ₁‑normalised
   (sum == 1) because the evaluation is homogeneous.

2. **Heavy continuous optimisation** – the best few seeds are refined
   with a high‑temperature Adam optimiser that minimises the
   log‑sum‑exp surrogate of the autocorrelation (the surrogate tends to
   the true max as the temperature grows).  Afterwards a projected
   sub‑gradient descent is run on the true max‑convolution objective.

3. **Stochastic polishing** – a short simulated‑annealing phase
   redistributes mass among random subsets of coordinates.

All stages respect the supplied time budget (default 1000 s).  The final
sequence is stored in the global ``height_sequence_1`` variable for
later reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact evaluation for the homogeneous objective (max convolution)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r ∈ (0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ i^{‑p} (p≥0)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Alpha = 0 → rectangular, alpha = 1 → Hann."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return a / a.sum()

    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1))

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet with equal concentration."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_powered_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Take uniform[0,1) draw, raise to a random power, renormalise."""
    a = rng.random(n).astype(np.float64)
    p = rng.uniform(0.3, 4.0)
    a = a ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 1_000_000,
                   beta_max: float = 1000.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    The vector ``v`` holds logits of a soft‑max representation, guaranteeing
    that the iterate stays on the probability simplex.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # initialise logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0                      # starting temperature
    beta_mul = 1.002                     # growth factor

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                      # points on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        # increase temperature (smooth → sharp)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)               # full linear convolution
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip gradient norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 5_000_000,
                      init_lr: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    The step size follows a Polyak‑type 1/√t schedule.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active lag
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple Dirichlet‑reshuffle simulated annealing.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)               # change between 2 and 12 entries
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------


def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a heterogeneous pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families (some parameters swept finely)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Trapezoidal – wide sweep
    for wf in np.linspace(0.01, 0.45, 12):
        seeds.append(_trapezoidal(n, width_fraction=wf))

    # Tukey – many α
    for alpha in np.linspace(0.0, 0.95, 12):
        seeds.append(_tukey_window(n, alpha=alpha))

    # Power‑law
    for p in np.linspace(0.0, 6.0, 16):
        seeds.append(_power(n, p))

    # Sine‑window exponent
    for gamma in np.linspace(0.0, 5.0, 11):
        seeds.append(_sine_window(n, gamma=gamma))

    # Geometric ratios
    r_vals = np.concatenate([
        np.linspace(0.2, 0.97, 12),
        np.linspace(0.97, 0.9999, 12)
    ])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # Beta shapes
    for a in np.linspace(0.1, 5.0, 8):
        for b in np.linspace(0.1, 5.0, 8):
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows
    for sigma_frac in (0.12, 0.18, 0.24, 0.30, 0.36, 0.42):
        seeds.append(_gaussian(n, sigma_frac))

    # Classical windows
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    # Random Dirichlet draws with varied concentration
    for _ in range(25):
        conc = 10 ** rng.uniform(-3.0, 1.0)      # from 0.001 to 10
        seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # Random power‑transformed uniform draws
    for _ in range(20):
        seeds.append(_random_powered_uniform(n, rng))

    # Previous best (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5               # safety margin

    # Retrieve any previously discovered solution
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # Choose a length that balances expressive power and speed.
    # --------------------------------------------------------------
    n = 6000
    print(f"\n=== Starting search: length = {n}, time budget = {budget_s}s ===")

    # --------------------------------------------------------------
    # Stage 1 – generate a rich pool of seeds and keep the best.
    # --------------------------------------------------------------
    print("\n--- Seed generation ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values after seed stage:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – heavy continuous optimisation (Adam + sub‑gradient).
    # --------------------------------------------------------------

    # Adam optimisation on the top few seeds
    max_adam_runs = min(3, len(best_seeds))
    for idx in range(max_adam_runs):
        if time.time() > deadline:
            break
        print(f"\n--- Adam optimisation run {idx + 1}/{max_adam_runs} ---")
        # Allocate a generous iteration budget but stay within time
        remaining = deadline - time.time()
        # Rough estimate: ~0.0006 s per iteration for n≈6000
        max_iters = int(min(2_000_000, remaining * 1500))
        seq_adam, val_adam = _adam_optimize(
            best_seeds[idx], n, deadline,
            max_iters=max_iters,
            beta_max=2000.0
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # Sub‑gradient refinement – use the bulk of the leftover time
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(5_000_000, remaining * 1500))
        max_iters = max(500_000, max_iters)      # guarantee a minimum effort
        print(f"\n--- Sub‑gradient refinement (max_iters={max_iters}) ---")
        seq_proj, val_proj = _subgradient_proj(
            global_best_seq, n, deadline,
            max_iters=max_iters,
            init_lr=0.20
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[overall‑SubProj] new global best = {val_proj:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – stochastic polishing (simulated annealing).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=600_000,
            init_temp_factor=0.06
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # --------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # --------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:                       # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for possible later reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
