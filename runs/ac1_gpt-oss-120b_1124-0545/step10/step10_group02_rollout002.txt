"""
Propose a non‑negative step‑function that beats the current bound (~1.509).

Algorithm
---------
1. Choose a modest length n0 = 2000 (fast enough for many gradient steps).
2. Build a large heterogeneous pool of ℓ₁‑normalised seeds:
   • deterministic analytic families (uniform, triangular, trapezoidal,
     sine‑window, Tukey, geometric, power‑law, beta, Gaussian);
   • a handful of random Dirichlet draws with random concentration parameters.
3. For every seed run a soft‑max‑Adam optimiser that minimises a
   smooth approximation of max (a∗a).  The optimiser works in the
   log‑softmax (mirror‑descent) space, keeps the best sequence and
   respects the time budget.
4. Upsample the best n0‑vector to n₁ = 4000 (linear interpolation,
   re‑normalise) – a finer discretisation lets the convolution peak
   approach its continuous optimum.
5. Refine the up‑sampled vector with a long projected sub‑gradient
   descent on the true max‑convolution objective (Polyak step‑size).
6. If time permits, upsample once more to n₂ = 8000 and run a short
   sub‑gradient phase.
7. A final simulated‑annealing polish redistributes a small amount of
   mass among a random subset of coordinates.
8. The best candidate is evaluated with the official ``evaluate_sequence``,
   printed and stored in the global variable ``height_sequence_1`` for
   future runs.

All heavy loops check the remaining time and abort gracefully, guaranteeing
a return before the 1000 s deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact evaluation for ℓ₁‑normalised sequences (sum == 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (r∈(0,1))."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet with a single concentration parameter."""
    a = rng.dirichlet(np.full(n, conc))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 300_000,
                   beta_max: float = 1500.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate for max(a∗a).
    Parameters are logits of the soft‑max representation.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)      # initial logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # temperature schedule – start smooth, finish sharp
    beta_temp = 5.0
    beta_mul = 1.003

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                 # stays on simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        # temperature increase
        beta_temp = min(beta_temp * beta_mul, beta_max)

        shift = np.max(conv)            # for numerical stability
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)      # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip gradient norm for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 4_000_000,
                      init_lr: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses a Polyak‑style diminishing step size.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak‑style diminishing step size
        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 600_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing that reshuffles mass among a random subset
    of coordinates (Dirichlet redistribution).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)               # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length, then re‑normalise."""
    old_n = len(seq)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0.0:
        new_seq = np.full(new_n, 1.0 / new_n, dtype=np.float64)
    else:
        new_seq /= s
    return new_seq.astype(np.float64)


def _deterministic_seeds(n: int, rng: np.random.Generator,
                         previous: np.ndarray | None) -> list[np.ndarray]:
    """Generate a diverse fixed set of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # Core families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for wf in np.linspace(0.05, 0.40, 8):
        seeds.append(_trapezoidal(n, width_fraction=wf))

    for gamma in np.linspace(0.0, 2.5, 9):
        seeds.append(_sine_window(n, gamma=gamma))

    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    r_vals = np.concatenate([
        np.linspace(0.2, 0.98, 12),
        np.linspace(0.98, 0.9999, 10)
    ])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 18):
        seeds.append(_power(n, p))

    for a in np.linspace(0.1, 3.0, 6):
        for b in np.linspace(0.1, 3.0, 6):
            seeds.append(_beta_shape(n, a, b))

    for sigma_frac in (0.12, 0.18, 0.24, 0.30, 0.36):
        seeds.append(_gaussian(n, sigma_frac))

    # Random Dirichlet with a few concentrations
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponentials
    for _ in range(4):
        seeds.append(_random_exponential(n, rng))

    # Previous solution, if any
    if previous is not None and previous.shape == (n,):
        s = previous.sum()
        if s > 0:
            seeds.append(previous / s)

    return seeds


def _random_seeds(n: int, rng: np.random.Generator, count: int = 8) -> list[np.ndarray]:
    """Create a handful of Dirichlet draws with random concentration."""
    seeds: list[np.ndarray] = []
    for _ in range(count):
        # concentration uniformly sampled in log‑space between 1e‑3 and 1e2
        logc = rng.uniform(-3.0, 2.0)
        conc = 10 ** logc
        seeds.append(_random_dirichlet(n, conc, rng))
    return seeds

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Length schedule
    # ------------------------------------------------------------------
    n0 = 2000   # fast stage
    n1 = 4000   # refined stage
    n2 = 8000   # optional ultra‑refined stage

    # ------------------------------------------------------------------
    # Stage 0 – collect deterministic seeds (length n0)
    # ------------------------------------------------------------------
    print("\n--- Stage 0: seed collection (n =", n0, ") ---")
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    seeds = _deterministic_seeds(n0, rng, previous)
    # add a few random Dirichlet seeds for extra diversity
    seeds.extend(_random_seeds(n0, rng, count=12))
    print(f"Generated {len(seeds)} candidate seeds (including random).")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation on each seed (quick runs)
    # ------------------------------------------------------------------
    best_global_seq = None
    best_global_val = float("inf")

    # allocate a modest iteration budget per seed depending on remaining time
    max_iters_per_seed = 180_000   # empirically ~15 s for n0=2000

    for idx, seed_vec in enumerate(seeds):
        if time.time() > deadline:
            break
        print(f"\n[Adam] Seed {idx + 1}/{len(seeds)} (n={n0})")
        seq, val = _adam_optimize(seed_vec, n0, deadline,
                                  max_iters=max_iters_per_seed,
                                  beta_max=1500.0)
        if seq is not None and val < best_global_val:
            best_global_val = val
            best_global_seq = seq.copy()
            print(f"[Adam] New best surrogate = {val:.6f}")

    # If by any chance no Adam run succeeded (should not happen), fallback
    if best_global_seq is None:
        best_global_seq = _uniform(n0)
        best_global_val = _objective(best_global_seq)

    print(f"\n[Stage 1] best surrogate after Adam = {best_global_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upscale to n1 and perform a long sub‑gradient phase
    # ------------------------------------------------------------------
    seq_n1 = _upsample(best_global_seq, n1)
    print(f"\n--- Stage 2: sub‑gradient refinement (n={n1}) ---")
    remaining = deadline - time.time()
    # Use a large iteration count but bound by remaining time
    max_iters_sub = int(min(5_000_000, max(500_000, remaining * 1400)))
    print(f"Allocated {max_iters_sub:,} sub‑gradient iterations.")
    seq_proj, val_proj = _subgradient_proj(seq_n1, n1, deadline,
                                          max_iters=max_iters_sub,
                                          init_lr=0.25)
    if val_proj < best_global_val:
        best_global_val = val_proj
        best_global_seq = seq_proj.copy()
        print(f"[SubProj] New global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – optional second upscale (n2) if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_n2 = _upsample(best_global_seq, n2)
        print(f"\n--- Stage 3: extra sub‑gradient on n={n2} ---")
        remaining = deadline - time.time()
        max_iters_sub2 = int(min(2_000_000, max(200_000, remaining * 800)))
        seq_proj2, val_proj2 = _subgradient_proj(seq_n2, n2, deadline,
                                                max_iters=max_iters_sub2,
                                                init_lr=0.20)
        if val_proj2 < best_global_val:
            best_global_val = val_proj2
            best_global_seq = seq_proj2.copy()
            print(f"[SubProj‑2] New global best = {val_proj2:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = len(best_global_seq)
        print(f"\n--- Stage 4: simulated annealing (n={cur_n}) ---")
        seq_sa, val_sa = _simulated_annealing(best_global_seq, cur_n,
                                              deadline,
                                              max_iters=600_000,
                                              init_temp_factor=0.04)
        if val_sa < best_global_val:
            best_global_val = val_sa
            best_global_seq = seq_sa.copy()
            print(f"[SA] New global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(best_global_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {best_global_val:.6f} "
        f"(length {len(best_global_seq)})"
    )

    # Store for possible later reuse
    globals()["height_sequence_1"] = best_global_seq.tolist()
    return best_global_seq.tolist()
