"""
Improved search for a low‑max‑convolution step‐function.

The algorithm consists of four phases:

1. **Analytic seeds** – a wide collection of closed‑form shapes
   (uniform, Catalan, Hann, Beta, …).  Each seed is normalised to
   sum = 1 and its objective 2·n·max(conv) is evaluated.

2. **Random sampling** – Dirichlet draws and Beta‑pdf shapes are
   generated for a few short lengths.  The best of these seeds becomes
   the starting point for the later phases.

3. **Smooth Adam optimisation** – we minimise a log‑sum‑exp
   approximation of `max(conv(a,a))`.  The smoothing parameter β is
   increased from 5 to 2000 and the Adam schedule runs for many
   iterations (≈ 10 M) while respecting the remaining time budget.

4. **Projected sub‑gradient polishing** – an exact sub‑gradient for
   the non‑smooth objective is built from all convolution entries that
   are within a relative tolerance of the current maximum.  The step
   size follows the classic 1/√t schedule and every 1024 steps the
   exact objective is recomputed.

   For modest sizes (n ≤ 3000) an additional *Frank–Wolfe* style
   refinement is performed: we solve a linear programme
   `max Σ g_j  s.t.  (f * g) ≤ M` (with `M = max(conv(f,f))`), obtain a
   new direction `g`, and mix it with the current iterate.
   This often yields a noticeable drop in the objective.

All helpers are top‑level; no closures or lambdas are used.
The routine respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain list of non‑negative floats.
"""

import time
import numpy as np
from math import pi, sqrt
from scipy.optimize import linprog

# ---------------------------------------------------------------
# Low‑level utilities
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective of `evaluate_sequence` for a normalised sequence."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ---------------------------------------------------------------
# Analytic seed families (all returned normalised to sum = 1)
# ---------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * pi * i / (n - 1))
         + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian_sigma(factor: float, n: int) -> np.ndarray:
    sigma = factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _gaussian_sigma015(n: int) -> np.ndarray:
    return _gaussian_sigma(0.15, n)


def _gaussian_sigma025(n: int) -> np.ndarray:
    return _gaussian_sigma(0.25, n)


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf on a discrete grid, shifted by 0.5 to avoid end‑points."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape_05_05(n: int) -> np.ndarray:
    return _beta_shape(n, 0.5, 0.5)


def _beta_shape_2_2(n: int) -> np.ndarray:
    return _beta_shape(n, 2.0, 2.0)


def _beta_shape_05_2(n: int) -> np.ndarray:
    return _beta_shape(n, 0.5, 2.0)


def _beta_shape_2_05(n: int) -> np.ndarray:
    return _beta_shape(n, 2.0, 0.5)


# ---------------------------------------------------------------
# Random Dirichlet / Beta sampling
# ---------------------------------------------------------------

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a point on the simplex using a Dirichlet distribution."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ---------------------------------------------------------------
# Smooth Adam optimisation (log‑sum‑exp surrogate)
# ---------------------------------------------------------------

def _smooth_adam_optimize(start_seq: np.ndarray,
                          deadline: float,
                          beta_start: float = 5.0,
                          beta_end: float = 2000.0,
                          max_iter: int = 10_000_000,
                          lr0: float = 0.03) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp approximation of the max‑convolution."""
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        a = np.exp(logits)
        a = a / a.sum()
        conv = _conv_fft(a, a)

        # schedule for β (smoothness)
        frac = min(1.0, it / (max_iter * 0.6))
        beta = beta_start + (beta_end - beta_start) * frac

        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()               # soft‑max weights

        # gradient of the smoothed max = 2 * (p * a_rev) convolved with a
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # convert to gradient w.r.t. logits (mirror‑descent style)
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if (it & 1023) == 0:          # every 1024 iterations
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")
            if best_val < 1.45:      # premature stop if already excellent
                break

    return best_seq, best_val


# ---------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ---------------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ---------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
# ---------------------------------------------------------------

def _proj_subgradient_optimize(start_seq: np.ndarray,
                               deadline: float,
                               eps_rel_start: float = 1e-4,
                               eps_rel_end: float = 1e-7,
                               lr0: float = 0.12,
                               max_iter: int = 12_000_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent for the exact (non‑smooth) objective."""
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        # tolerance that shrinks linearly from start to end
        eps_rel = eps_rel_start + (eps_rel_end - eps_rel_start) * min(1.0,
                                                                     it / (max_iter * 0.5))
        thresh = max_val - eps_rel * max_val
        active_idx = np.where(conv >= thresh)[0]
        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv)], dtype=int)

        # uniform weighting over the active set (standard sub‑gradient)
        w = np.zeros_like(conv)
        w[active_idx] = 1.0 / active_idx.size

        # gradient = 2 * (w * a_rev) convolved with a
        conv_ag = _conv_fft(w, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        lr = lr0 / sqrt(it)
        a_next = a - lr * g
        a_next = _project_simplex(a_next)

        if (it & 1023) == 0:
            val = _objective(a_next)
            if val < best_val:
                best_val = val
                best_seq = a_next.copy()
                print(f"[ProjSGD] it={it:7d} n={n:5d} val={val:.6f}")
            if best_val < 1.45:
                break

        a = a_next

    return best_seq, best_val


# ---------------------------------------------------------------
# Frank‑Wolfe / linear‑programming refinement (optional, small n)
# ---------------------------------------------------------------

def _lp_refine(f: np.ndarray,
               max_iters: int,
               deadline: float) -> tuple[np.ndarray, float]:
    """Perform a few Frank‑Wolfe steps using a linear programme.

    The LP:   maximise Σ g_j
              subject to A g ≤ M·1 ,  g ≥ 0
    where A is the convolution matrix of `f` and M = max(conv(f,f)).
    """
    n = f.shape[0]
    L = 2 * n - 1
    best_seq = f.copy()
    best_val = _objective(f)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        M = float(np.max(_conv_fft(f, f)))
        # Build dense convolution matrix A (size L×n)
        # column j contains f placed at rows j … j+n‑1
        A = np.zeros((L, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f
        # Solve the LP (HiGHS interior‑point)
        res = linprog(c=-np.ones(n),
                      A_ub=A,
                      b_ub=np.full(L, M),
                      bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            print("[LP] linear programme failed")
            break
        g = np.maximum(res.x, 0.0)
        if g.sum() == 0.0:
            break
        g = g / g.sum()
        # step size – classic 2/(k+2) schedule
        t = 2.0 / (it + 2.0)
        f = (1.0 - t) * f + t * g
        f = f / f.sum()
        val = _objective(f)
        if val < best_val:
            best_val = val
            best_seq = f.copy()
            print(f"[LP] it={it:3d} n={n:5d} val={val:.6f}")
        # early stopping if improvement stalls
        if it > 0 and abs(val - best_val) < 1e-7:
            break
    return best_seq, best_val

def propose_candidate(seed: int = 123,

                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine follows several phases, ending with a final sanity check.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a little safety margin

    # ------------------------------------------------------------------
    # Use a previously stored sequence if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore (FFT‑friendly, up to ~16 k)
    # ------------------------------------------------------------------
    lengths = [2000, 2500, 3000, 3500, 4000, 6000, 8000, 12000, 16000]

    # ------------------------------------------------------------------
    # Containers for the current best solution
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic seeds
    # ------------------------------------------------------------------
    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        _gaussian_sigma015,
        _gaussian_sigma025,
        _arcsine_shape,
        _beta_shape_05_05,
        _beta_shape_2_2,
        _beta_shape_05_2,
        _beta_shape_2_05,
    )

    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                best_n = n
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

        # reuse previous if same length
        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                best_n = n
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – heavy random sampling (Dirichlet + Beta) for short lengths
    # ------------------------------------------------------------------
    alphas = [0.5, 1.0, 2.0, 3.0]
    beta_params = [(0.3, 0.3), (0.5, 0.5), (1.0, 1.0),
                   (2.0, 2.0), (0.5, 2.0), (2.0, 0.5), (3.0, 3.0)]

    while time.time() < deadline * 0.45:      # use ~45 % of remaining time
        n = int(rng.choice([2000, 2500, 3000, 3500, 4000]))
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            best_n = n
            print(f"[Random-Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        a_par, b_par = rng.choice(beta_params)
        seq = _beta_shape(n, a_par, b_par)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            best_n = n
            print(f"[Random-Beta] n={n:5d} (α,β)=({a_par:.2f},{b_par:.2f}) val={val:.6f}")

        if best_val < 1.45:
            break

    # ------------------------------------------------------------------
    # Phase 3 – smooth Adam optimisation (few restarts)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0 and best_seq is not None:
        # we will run Adam on the current best and on a couple of fresh restarts
        candidates = [best_seq]
        for _ in range(2):

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
