"""
Search for a non‑negative step function that minimizes `evaluate_sequence`.

Algorithm
---------
1. Build a large heterogeneous pool of ℓ₁‑normalised candidates:
   - deterministic analytic families (uniform, triangular, trapezoidal, geometric,
     symmetric geometric, power‑law, sine‑window, Tukey, Gaussian, Beta).
   - random Dirichlet draws with many concentration parameters.
   - random sparse “spike’’ vectors (few non‑zero entries).
   - the previously discovered best solution (if any).

2. Evaluate all seeds with the exact surrogate `2·n·max(conv)` (the denominator
   is 1 because every candidate is ℓ₁‑normalised).  Keep the top‑k seeds.

3. For each top seed run a continuation of L‑BFGS on the smooth
   log‑sum‑exp surrogate of the max‑convolution:
        f(v) = (2·n/β)·log⎡∑_i exp(β·conv_i(v))⎤,
   where `v` are logits (softmax(v) = a).  The schedule of temperatures is
   β = [5, 30, 120, 500]; each stage gets a few hundred L‑BFGS iterations.
   The best vector from this phase becomes the new global candidate.

4. Refine the current best with a long projected sub‑gradient descent on the
   true objective 2·n·max(conv).  The step‑size follows a Polyak‑style
   1/√t decay and the iterate is projected onto the probability simplex after
   each step.

5. Polish the refined vector with a simple simulated‑annealing that
   reshuffles mass among a random subset of coordinates.

6. Evaluate the final vector with the official `evaluate_sequence`,
   store it globally as `height_sequence_1`, and return it as a list.

The routine stops as soon as the allotted wall‑clock budget is exhausted
(the default is 1000 s).  All helper functions are defined at module level to
avoid closures.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector (ℓ₁‑norm 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """One‑sided geometric decay (r∈(0,1])."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc) and normalise."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k random indices receive random positive mass; rest are zero."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    mass = rng.random(k)  # uniform (0,1)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = mass
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Objective helpers (exact surrogate)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for ℓ₁‑normalised sequences."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth surrogate for L‑BFGS
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Smooth surrogate f(v) = (2·n/β)·log∑_i exp(β·conv_i)
    where a = softmax(v) and conv = a * a (full convolution).
    Returns (f, grad_v) where grad_v is the gradient w.r.t. logits v.
    """
    a = _softmax(v)                     # ℓ₁‑normalised
    conv = _conv_fft(a, a)               # length 2n‑1

    # log‑sum‑exp in a numerically stable way
    u = beta * conv
    u_max = np.max(u)
    exp_shifted = np.exp(u - u_max)
    sum_exp = exp_shifted.sum()
    logsumexp = u_max + np.log(sum_exp)

    f = (2.0 * n / beta) * logsumexp

    # softmax weights (p_i = exp(β·conv_i)/∑exp)
    p = exp_shifted / sum_exp

    # gradient w.r.t. a of Σ p_i·conv_i  (same as weighted sum)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, p)                # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

    # chain rule: ∂f/∂a = 2·n·grad_a  (see derivation in analysis)
    grad_f_a = 2.0 * n * grad_a

    # gradient w.r.t. logits v (softmax Jacobian)
    dot = np.dot(grad_f_a, a)
    grad_v = a * (grad_f_a - dot)

    return f, grad_v


# ----------------------------------------------------------------------
# L‑BFGS optimiser (continuation over β)
# ----------------------------------------------------------------------


def _lbfgs_optimize(start_vec: np.ndarray, n: int, deadline: float,
                    betas=(5.0, 30.0, 120.0, 500.0),
                    maxiter_per_beta: int = 150):
    """
    Perform a continuation of L‑BFGS on the smooth surrogate.
    Returns the best vector (ℓ₁‑normalised) found and its exact objective.
    """
    try:
        from scipy import optimize as opt
    except Exception as exc:
        print("[WARN] SciPy not available – skipping L‑BFGS stage:", exc)
        return start_vec, _objective(start_vec)

    # Initialise logits safely (avoid log(0))
    eps = 1e-30
    v = np.log(start_vec + eps)

    best_seq = start_vec.copy()
    best_val = _objective(best_seq)

    for beta in betas:
        if time.time() > deadline:
            break
        # minimise f(v) with supplied gradient
        res = opt.minimize(_smooth_obj_and_grad,
                           v,
                           args=(n, beta),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_beta,
                                    'disp': False})
        v = res.x
        a = _softmax(v)                  # stays on the simplex
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        # print progress
        print(f"[L‑BFGS] beta={beta:.1f}  iters={res.nit}  best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 3_000_000,
                      init_lr: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for the exact objective 2·n·max(conv).
    Uses a Polyak‑style diminishing step size.
    Returns the best encountered sequence and its objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubProj] it={it:7d}  best={best_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step size
        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val

# Simulated‑annealing polish
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple SA that redistributes mass among a random small subset of indices.
    Returns the best sequence seen and its objective.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # move between 2 and 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d}  best={best_val:.6f}")

    return best_seq, best_val

def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse collection of ℓ₁‑normalised candidates.
    """
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for wf in np.linspace(0.05, 0.40, 8):
        seeds.append(_trapezoidal(n, width_fraction=wf))

    for gamma in np.linspace(0.0, 2.5, 9):
        seeds.append(_sine_window(n, gamma=gamma))

    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    # geometric families – include ratios extremely close to 1
    r_vals = np.concatenate([
        np.linspace(0.2, 0.98, 12),
        np.linspace(0.98, 0.9999, 10)
    ])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # power‑law family
    for p in np.linspace(0.0, 5.0, 18):
        seeds.append(_power(n, p))

    # beta shapes – modest grid
    for a in np.linspace(0.1, 3.0, 6):
        for b in np.linspace(0.1, 3.0, 6):
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows
    for sigma_frac in (0.12, 0.18, 0.24, 0.30, 0.36):
        seeds.append(_gaussian(n, sigma_frac))

    # Random Dirichlet draws – wide concentration range
    conc_vals = [0.001, 0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # Random sparse spikes
    for k in (2, 4, 8, 16, 32):
        for _ in range(6):
            seeds.append(_random_spikes(n, k, rng))

    # Include previous best if available
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    returns the best sequence as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # Retrieve previous solution if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose a length that balances expressivity & speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – generate a large heterogeneous pool and keep the best.
    # ------------------------------------------------------------------
    print(f"\n--- Seed collection (n={n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])

    top_k = min(12, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"\nTop {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # Initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best exact = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth L‑BFGS continuation on top seeds.
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds[:5]):   # try a few seeds
        if time.time() > deadline:
            break
        print(f"\n--- L‑BFGS optimisation run {idx+1}/{len(best_seeds[:5])} ---")
        seq_lbfgs, val_lbfgs = _lbfgs_optimize(
            seed_vec, n, deadline,
            betas=(5.0, 30.0, 120.0, 500.0),
            maxiter_per_beta=150
        )
        if val_lbfgs < global_best_val:
            global_best_seq = seq_lbfgs.copy()
            global_best_val = val_lbfgs
            print(f"[overall‑L‑BFGS] new global best = {val_lbfgs:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long projected sub‑gradient refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # Heuristic: allocate most of the remaining time to sub‑gradient.
        max_iters = int(remaining * 1200)   # ~0.8 ms per iteration empirically
        max_iters = max(500_000, min(max_iters, 4_000_000))
        print(f"\n--- Sub‑gradient refinement (max_iters={max_iters}) ---")
        seq_sub, val_sub = _subgradient_proj(
            global_best_seq, n, deadline,
            max_iters=max_iters,
            init_lr=0.25
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[overall‑SubProj] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final simulated‑annealing polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=600_000,
            init_temp_factor=0.04
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and bookkeeping.
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = None

    print(
        f"\n=== Finished ===\nBest exact surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # Store for possible later reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

This implementation expands the search space, replaces the Adam stage with a more powerful L‑BFGS continuation on a smooth surrogate, allocates more time to a robust sub‑gradient refinement, and finishes with a stochastic annealing polish—all while respecting the 1000 s budget. The best sequence is saved as `height_sequence_1` for future runs.
