def _local_search(seq, n, deadline):
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_fft(a, a)
        max_idx = np.argmax(conv)
        cur_val = 2.0 * n * conv[max_idx]
        # Determine contributions
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idxs = np.arange(i_start, i_end+1)
        rev = max_idx - idxs
        contrib = a[idxs] * a[rev]
        # Find i_max where contribution high
        # Compute probability distribution proportional to contrib
        if contrib.sum()==0:
            # random move
            i = rng.integers(n)
            j = rng.integers(n)
            # simple swap
            new_a = a.copy(); new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            # choose i from idxs weighted by contrib
            probs = contrib / contrib.sum()
            i_chosen = rng.choice(idxs, p=probs)
            # partner index
            j_partner = max_idx - i_chosen
            # We'll move epsilon from i_chosen to a random other index not equal
            eps = min(0.01 * a[i_chosen], a[i_chosen])  # small fraction
            # choose target index not equal to i_chosen or j_partner
            target = rng.integers(n)
            while target == i_chosen:
                target = rng.integers(n)
            new_a = a.copy()
            new_a[i_chosen] -= eps
            new_a[target] += eps
        # renormalize to sum 1 (should already sum)
        new_val = _objective(new_a)
        if new_val < cur_val or np.random.rand() < 0.001:
            a = new_a
            if new_val < best_val:
                best_val = new_val
                best_seq = a.copy()
    return best_seq, best_val
```

Algorithm details for local search:

1. While time < deadline:

    - conv = _conv_fft(a, a)

    - max_idx = argmax(conv), cur_val = conv[max_idx]

    - Determine overlapped pairs: i_start = max(0, max_idx - (n - 1)); i_end = min(n - 1, max_idx)

    - idxs = np.arange(i_start, i_end+1)

    - rev = max_idx - idxs

    - contributions = a[idxs] * a[rev]

    - Identify i_max = idxs[np.argmax(contributions)]

    - Determine move amount eps = min(epsilon * a[i_max], 0.5 * a[i_max]) where epsilon = maybe 0.05.

    - Pick a target index j (randomly) not equal i_max or rev(i_max) (if distinct) ; optionally could be index with minimal contributions to current max (i.e., where a[j] is minimal). Choose j from low-value indices: e.g., randomly pick among lowest 10% of values.

    - new_a = a.copy(); new_a[i_max] -= eps; new_a[j] += eps

    - Normalize: sum stays same.

    - Compute new_val = _objective(new_a); if new_val < cur_val, accept; else accept with probability exp(-(new_val - cur_val)/T) with T slowly cooling.

   Temperature schedule: start T = 0.1 * cur_val maybe.

   We'll define T = max(init_temp * (0.99 ** it), min_temp)

   Accept if improvement else maybe occasional uphill.

   Keep best.

2. Return best.

def _adam_optimize_schedule(start_vec, n, deadline, max_iters=200_000):
    eps = 1e-30
    v = np.log(start_vec + eps)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1=0.9
    beta2=0.999
    lr0=0.8
    beta_start=5.0
    beta_end=300.0

    best_seq=None
    best_val=float('inf')
    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break
        a = _softmax(v)
        conv = _conv_fft(a,a)
        cur_val = 2.0*n*np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # compute temperature:
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t
        # smoothed max
        shift = np.max(conv)
        w = np.exp(beta*(conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum
        # gradient:
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1:n-1+n]
        # log-space grad
        dot = np.dot(grad_a, a)
        grad_v = a*(grad_a - dot)
        # Adam update
        m = beta1*m + (1-beta1)*grad_v
        v_hat = beta2*v_hat + (1-beta2)*(grad_v**2)
        m_hat = m/(1-beta1**it)
        v_hat_corr = v_hat/(1-beta2**it)
        lr = lr0/np.sqrt(it)
        v = v - lr*m_hat/(np.sqrt(v_hat_corr)+eps)
    return best_seq, best_val
```

Note: We compute objective as cur_val (2*n*max(conv)). That's same as surrogate we want to minimize.

Our evaluation of surrogate at each iteration equals actual because we compute exact max conv via conv. So we don't need surrogate; we just use that.

Thus we can simplify: use the exact max conv each iteration, but we incorporate smoothed gradient for subgradient of max conv using log-sum-exp weight w. This gradient approximates exact gradient increasingly as beta grows.

def _max_reduce_local_search(start_seq, n, deadline):
    rng = np.random.default_rng()
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    cur_val = best_val
    it = 0
    T_init = 0.1 * cur_val
    T = T_init
    min_temp = 1e-12
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        # contributions
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idxs = np.arange(i_start, i_end + 1)
        rev = max_idx - idxs
        contrib = a[idxs] * a[rev]
        if contrib.sum() > 0:
            probs = contrib / contrib.sum()
            i = rng.choice(idxs, p=probs)
            # partner:
            j_partner = rev[np.where(idxs == i)[0][0]]
        else:
            i = rng.integers(n)
            j_partner = None
        # choose target index far from i and j_partner: sample from low-mass indices
        # Compute low-mass threshold:
        low_threshold = np.percentile(a, 20)  # bottom 20%
        low_idxs = np.where(a <= low_threshold)[0]
        if low_idxs.size == 0:
            target = rng.integers(n)
        else:
            target = rng.choice(low_idxs)
        if target == i:
            continue
        eps = min(0.02 * a[i], a[i])  # move up to 2% of mass at i
        if eps <= 0:
            continue
        new_a = a.copy()
        new_a[i] -= eps
        new_a[target] += eps
        # renorm not needed, sum unchanged
        new_val = _objective(new_a)
        delta = new_val - cur_val
        # cooling
        T = max(min_temp, T * 0.9999)  # slow cooling
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a.copy()
                # maybe print progress
                if it % 10000 == 0:
                    print(f"[Local] it={it:,} best={best_val:.6f}")
    return best_seq, best_val

```python
"""
Algorithm:
1. Generate a diverse set of ℓ₁‑normalised candidate step‑functions
   (analytic windows, random Dirichlet draws, spike patterns, alternating orders).
2. Keep the best few seeds according to a fast FFT‑based surrogate (exact max‑conv).
3. Perform a long Adam optimisation on the log‑softmax parameters.
   The temperature for the smoothed‑max surrogate grows exponentially from a
   low value (≈5) to a high value (≈300) so the optimiser first sees a smooth
   landscape and later refines the true max.  The optimiser stops automatically
   when its time budget expires.
4. Refine the result with exponentiated‑gradient (mirror) sub‑gradient descent
   directly on the true max‑convolution objective.
5. Polish the solution with a targeted simulated‑annealing / hill‑climbing
   routine that repeatedly moves a small amount of mass away from the pairs
   that realise the current worst autocorrelation.
6. Return the best sequence found and store it globally for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised sequence."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple numerical helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random families & ordering helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)          # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


def _alternating_desc(vec: np.ndarray) -> np.ndarray:
    """Place the largest half on even indices, the rest on odd indices."""
    a = np.sort(vec)[::-1]
    n = len(a)
    evens = (n + 1) // 2
    odds = n // 2
    out = np.empty_like(a)
    out[::2][:evens] = a[:evens]
    out[1::2][:odds] = a[evens:evens+odds]
    s = out.sum()
    return out / s if s > 0 else _uniform(n)


def _spike_random(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Place m equal spikes on random locations."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=m, replace=False)
    a[idx] = 1.0 / m
    return a


def _spike_random_amplitudes(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Place m spikes with random Dirichlet amplitudes."""
    amp = rng.dirichlet(np.full(m, 1.0))
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=m, replace=False)
    a[idx] = amp
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_gaussian(n, sigma_frac=0.40))

    # power‑law family
    for p in np.linspace(0.0, 4.0, 13):
        seeds.append(_power(n, p))

    # beta family
    for a in np.linspace(0.3, 2.0, 5):
        for b in np.linspace(0.3, 4.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (different concentration parameters)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw random order
            seeds.append(_sorted_desc(raw))       # monotone decreasing
            seeds.append(_sorted_asc(raw))        # monotone increasing
            seeds.append(_alternate_order(raw))   # alternating pattern

    # Spike patterns – a few sharp peaks
    for m in (5, 10, 20):
        seeds.append(_spike_random(n, m, rng))
        seeds.append(_spike_random_amplitudes(n, m, rng))

    # Alternating large‑small ordering of a uniform seed
    seeds.append(_alternating_desc(_uniform(n)))

    # Previous solution (if any and of matching length)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

# Optimisers

def _adam_optimize_schedule(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam on log‑softmax parameters.
    The smoothing temperature for the log‑sum‑exp surrogate grows
    exponentially from beta_start to beta_end across the allotted iterations.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)             # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8
    beta_start = 5.0
    beta_end = 300.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50_000 == 0:
                print(f"[Adam] it={it:,} best={cur_val:.6f}")

        # temperature schedule (exponential growth)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # convert to log‑space gradient (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

    return best_seq, best_val


def _mirror_subgradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the true max‑conv objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.12  # initial step‑size factor

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:,} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex (positivity already holds)
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _max_reduce_local_search(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Targeted hill‑climbing focused on the pair of indices that realise the current
    maximal autocorrelation.  Small amounts of mass are moved away from those
    positions and transferred to low‑mass locations; occasional uphill moves are
    allowed via a slowly cooling temperature.
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    cur_val = best_val

    it = 0
    T = max(0.1 * cur_val, 1e-12)
    min_temp = 1e-12

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        # contributions of each pair (i, max_idx-i)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idxs = np.arange(i_start, i_end + 1)
        rev = max_idx - idxs
        contrib = a[idxs] * a[rev]

        if contrib.sum() > 0.0:
            probs = contrib / contrib.sum()
            i = rng.choice(idxs, p=probs)
        else:
            i = rng.integers(n)

        # pick a target index among the lowest 20 % of masses
        low_thr = np.percentile(a, 20)
        low_idxs = np.where(a <= low_thr)[0]
        if low_idxs.size == 0:
            target = rng.integers(n)
        else:
            target = rng.choice(low_idxs)

        if target == i:
            continue

        # move a modest fraction of the mass at i
        eps = 0.02 * a[i]          # 2 % of the current value
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= eps
        new_a[target] += eps

        new_val = _objective(new_a)
        delta = new_val - cur_val

        # accept if improvement or with Metropolis probability
        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, min_temp)):
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a.copy()
                if it % 10_000 == 0:
                    print(f"[Local] it={it:,} best={best_val:.6f}")

        # geometric cooling
        T = max(T * 0.9999, min_temp)

    return best_seq, best_val

"""
Algorithm:
1. Build a diversified pool of ℓ₁‑normalised candidate step‑functions:
   - analytic windows (uniform, triangular, sine, Gaussian, Tukey …)
   - random Dirichlet draws (raw, sorted, alternating)
   - a few “spike’’ patterns (sharp equal or random‑amplitude peaks)
   - an alternating large‑small ordering of a uniform seed.
2. Keep the best few seeds according to a fast FFT‑based exact max‑convolution
   surrogate (the same quantity that the official evaluator uses).
3. Run a long Adam optimisation on the log‑softmax parameters.
   The smoothing temperature for the log‑sum‑exp surrogate grows
   exponentially from a low value (≈5) to a high value (≈300), so the optimiser
   first sees a smooth landscape and later refines the true max.
4. Refine the result with exponentiated‑gradient (mirror) sub‑gradient descent
   directly on the true max‑convolution objective.
5. Polish the solution with a targeted hill‑climbing routine that repeatedly
   moves a small amount of mass away from the pair of indices that realise the
   current worst autocorrelation, accepting occasional uphill moves via a
   slowly‑cooling temperature.
6. Return the best sequence found and store it globally as ``height_sequence_1``
   for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised sequence."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple numerical helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random families & ordering helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)          # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


def _alternating_desc(vec: np.ndarray) -> np.ndarray:
    """Largest half on even indices, the rest on odd indices."""
    a = np.sort(vec)[::-1]
    n = len(a)
    evens = (n + 1) // 2
    odds = n // 2
    out = np.empty_like(a)
    out[::2] = a[:evens]
    out[1::2] = a[evens:evens + odds]
    s = out.sum()
    return out / s if s > 0 else _uniform(n)


def _spike_random(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Place m equal spikes on random locations."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=m, replace=False)
    a[idx] = 1.0 / m
    return a


def _spike_random_amplitudes(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Place m spikes with random Dirichlet amplitudes."""
    amp = rng.dirichlet(np.full(m, 1.0))
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=m, replace=False)
    a[idx] = amp
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_gaussian(n, sigma_frac=0.40))

    # power‑law family
    for p in np.linspace(0.0, 4.0, 13):
        seeds.append(_power(n, p))

    # beta family
    for a in np.linspace(0.3, 2.0, 5):
        for b in np.linspace(0.3, 4.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (different concentration parameters)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw order
            seeds.append(_sorted_desc(raw))       # decreasing
            seeds.append(_sorted_asc(raw))        # increasing
            seeds.append(_alternate_order(raw))   # alternating

    # Spike patterns – a few sharp peaks
    for m in (5, 10, 20):
        seeds.append(_spike_random(n, m, rng))
        seeds.append(_spike_random_amplitudes(n, m, rng))

    # Alternating large‑small ordering of a uniform seed
    seeds.append(_alternating_desc(_uniform(n)))

    # Previous solution (if any and of matching length)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize_schedule(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam on log‑softmax parameters.
    The smoothing temperature for the log‑sum‑exp surrogate grows
    exponentially from beta_start to beta_end across the allotted iterations.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)             # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8
    beta_start = 5.0
    beta_end = 300.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50_000 == 0:
                print(f"[Adam] it={it:,} best={cur_val:.6f}")

        # temperature schedule (exponential growth)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # smoothed max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # convert to log‑space gradient (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

    return best_seq, best_val


def _mirror_subgradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the true max‑conv objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.12  # initial step‑size factor

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[Mirror] it={it:,} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex (positivity already holds)
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _max_reduce_local_search(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Targeted hill‑climbing that alleviates the current maximal autocorrelation.
    Small amounts of mass are moved away from the pair of indices that realise the
    maximal convolution value, with a slowly cooling Metropolis acceptance rule.
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    cur_val = best_val

    it = 0
    T = max(0.1 * cur_val, 1e-12)
    min_temp = 1e-12

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        # contributions of each pair (i, max_idx-i)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idxs = np.arange(i_start, i_end + 1)
        rev = max_idx - idxs
        contrib = a[idxs] * a[rev]

        if contrib.sum() > 0.0:
            probs = contrib / contrib.sum()
            i = rng.choice(idxs, p=probs)
        else:
            i = rng.integers(n)

        # choose a low‑mass target (bottom 20 % of the distribution)
        low_thr = np.percentile(a, 20)
        low_idxs = np.where(a <= low_thr)[0]
        if low_idxs.size == 0:
            target = rng.integers(n)
        else:
            target = rng.choice(low_idxs)

        if target == i:
            continue

        # shift a modest fraction of the mass at i
        eps = 0.02 * a[i]          # 2 % of the current entry
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= eps
        new_a[target] += eps

        new_val = _objective(new_a)
        delta = new_val - cur_val

        # Metropolis acceptance
        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, min_temp)):
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a.copy()
                if it % 10_000 == 0:
                    print(f"[Local] it={it:,} best={best_val:.6f}")

        # geometric cooling
        T = max(T * 0.9999, min_temp)

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns a plain list of floats and stores the result globally as
    ``height_sequence_1`` for possible reuse.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # Keep a small safety margin for the final sanity check.
    total_budget = max(0.0, budget_s - 2.0)
    deadline = start_time + total_budget

    # ------------------------------------------------------------------
    # Length choice – a power‑of‑two friendly size.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Load previous best (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – generate seeds and evaluate them quickly
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation (temperature grows)
    # ------------------------------------------------------------------
    adam_deadline = start_time + total_budget * 0.20   # ~20 % of total budget
    print("\n--- Adam optimisation stage ---")
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > adam_deadline:
            break
        # Rough estimate of how many iterations we can afford.
        remaining = adam_deadline - time.time()
        max_iters = int(max(30_000, remaining * 350))   # ~350 it/s estimate
        max_iters = min(max_iters, 300_000)
        print(f"[Adam] seed {idx}/{len(best_seeds)} max_iters={max_iters}")
        seq_adam, val_adam = _adam_optimize_schedule(
            seed_vec, n, adam_deadline, max_iters=max_iters
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Mirror‑descent sub‑gradient optimisation
    # ------------------------------------------------------------------
    mirror_deadline = adam_deadline + total_budget * 0.55   # ~55 % of total budget
    if time.time() < mirror_deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, mirror_deadline, max_iters=5_000_000
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Targeted local polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Targeted local search polishing ---")
        seq_local, val_local = _max_reduce_local_search(
            global_best_seq, n, deadline
        )
        if val_local < global_best_val:
            global_best_seq = seq_local.copy()
            global_best_val = val_local
            print(f"[Local] new global best = {val_local:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator (optional)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
