"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Adam on the simplex** – we optimise a soft‑max parametrisation
   of the vector using a smooth approximation of the max‑operator.
   The smoothing parameter λ starts small (very smooth) and is
   increased slowly, so the optimiser first sees a benign landscape
   and then gradually focuses on the true non‑smooth objective.

2. **Projected sub‑gradient descent** – after the Adam run we switch
   to a hard max (sub‑gradient) and perform a simple projected gradient
   descent on the probability simplex.  The sub‑gradient for the
   max‑convolution is known analytically:
   `g_i = 2 * a[t_max - i]` (zero when the index is out of range).

3. **Light simulated‑annealing polishing** – a short Metropolis walk
   that swaps small amounts of mass between random coordinates and
   occasionally performs a global Gaussian jump.

All three phases respect the overall 1000 s budget.  The best vector
found is stored in the global variable ``height_sequence_1`` and
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (full mode)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective (exact – same formula as evaluate_sequence with sum=1)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return the exact evaluation value (sum is assumed to be 1)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple seed generators (return a probability vector that sums to 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    exps = np.exp(logits - m)
    return exps / exps.sum()


def _exponential_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    dec = inc[::-1] if n % 2 == 0 else np.arange(half, 0, -1, dtype=np.float64)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max over convolution entries (used by Adam)
# ----------------------------------------------------------------------
def _soft_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Soft‑max weights w_i  (sum w_i = 1) over the entries of `conv`.
    lam plays the role of an inverse temperature.
    """
    max_c = conv.max()
    if lam <= 0.0:
        # hard max – uniform on all exact ties
        active = np.isclose(conv, max_c, atol=1e-12)
        cnt = np.count_nonzero(active)
        w = np.where(active, 1.0 / cnt, 0.0)
        return w

    # relative distance from the maximum, scaled for stability
    rel = (conv - max_c) / max_c   # ≤ 0
    exps = np.exp(lam * rel)
    w = exps / exps.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of the (soft) max‑convolution w.r.t. a
# ----------------------------------------------------------------------
def _gradient_wrt_a(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution with respect to `a`.
    Implements   g = 2 * rev(a) ⊛ w,
    where w are the soft‑max weights over the convolution entries.
    """
    n = a.size
    conv = _conv_fft(a, a)
    w = _soft_max_weights(conv, lam)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length = n + (2n-1) -1 = 3n-2
    grad = 2.0 * full[n - 1:n - 1 + n]        # extract slice that matches a
    return grad


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Duchi et al.)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum x = 1}."""
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform
        return _uniform(len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 1 – Adam on the simplex (soft max)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   lam_start: float = 0.2,
                   lam_factor: float = 1.0005,
                   lr_const: float = 0.18) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parametrisation.
    Returns the best probability vector found and its exact objective.
    """
    eps = 1e-30
    logp = np.log(start + eps)          # initialise logits
    n = start.size
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = start.copy()
    best_val = _objective(start)

    it = 0
    lam = lam_start

    while time.time() < deadline:
        it += 1

        # soft‑max to get current probabilities
        logp_shift = logp - np.max(logp)
        a = np.exp(logp_shift)
        a /= a.sum()

        # gradient of smoothed max‑convolution
        g_a = _gradient_wrt_a(a, lam)

        # chain rule: ∂L/∂logp = a * (g_a - a·g_a)
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log * g_log)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule
        lr = lr_const / (math.sqrt(it) + 1.0)

        # update logits
        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # evaluate true (hard‑max) objective
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:7d}  best={cur_val:.6f}  λ={lam:.2f}")

        # slowly tighten the soft‑max
        lam *= lam_factor

        # occasional tiny Gaussian perturbation to escape flat plateaus
        if it % 50000 == 0:
            noise = np.random.default_rng().normal(scale=0.015, size=n)
            logp += noise

    return best_seq, best_val


# ----------------------------------------------------------------------
# Phase 2 – Projected sub‑gradient descent on the hard max
# ----------------------------------------------------------------------
def _subgradient_descent(start: np.ndarray,
                         deadline: float,
                         step0: float = 0.12,
                         decay: float = 0.9996) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on the non‑smooth objective.
    After each step we project back onto the simplex.
    """
    a = start.copy()
    best = a.copy()
    best_val = _objective(a)

    it = 0
    step = step0

    while time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_val = conv.max()
        t = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a:  g_i = 2 * a[t - i] (if in range)
        g = np.zeros_like(a)
        lo = max(0, t - (a.size - 1))
        hi = min(a.size - 1, t)
        # i runs from lo..hi, j = t - i is automatically in range
        idx = np.arange(lo, hi + 1, dtype=int)
        g[idx] = 2.0 * a[t - idx]

        # gradient descent step
        a = a - step * g

        # project onto simplex
        a = _proj_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[SubGrad] it={it:7d}  best={cur_val:.6f}")

        # step size decay
        step *= decay
        if step < 1e-8:
            step = 1e-8

        # occasional random kick to avoid becoming stuck
        if it % 30000 == 0:
            rng = np.random.default_rng()
            noise = rng.normal(scale=0.001, size=a.size)
            a = a + noise
            a = np.clip(a, 0.0, None)
            a = _proj_simplex(a)

    return best, best_val


# ----------------------------------------------------------------------
# Phase 3 – Light simulated‑annealing polishing (same idea as before)
# ----------------------------------------------------------------------
def _anneal_refine(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.035)
    decay = 0.9995
    it = 0
    n = cur.size

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        if i != j:
            delta = rng.random() * 0.018 * cur[i]
            delta = min(delta, cur[i])
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta

            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d}  best={new_val:.6f}")

        # every few thousand steps try a global perturbation
        if it % 2500 == 0:
            noise = rng.normal(scale=0.0015, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() > 0:
                new_seq /= new_seq.sum()
            else:
                new_seq = _uniform(n)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects a time budget (default
    1000 s) and returns the best vector found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a small safety margin

    rng = np.random.default_rng(seed + 12345)

    # ---------------------------------------------------------------
    # Problem size – a larger n gives a tighter theoretical lower bound
    # ---------------------------------------------------------------
    n = 8000
    print(f"\n=== Optimising step function (length {n}) ===")

    # ---------------------------------------------------------------
    # Build a collection of initial seeds
    # ---------------------------------------------------------------
    seeds = []

    # use previous run if available (maybe of a different length)
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        prev_arr = np.asarray(previous, dtype=np.float64)
        if prev_arr.ndim == 1:
            # either truncate or pad with zeros to reach length n
            if prev_arr.size >= n:
                prev_arr = prev_arr[:n]
            else:
                pad = np.zeros(n - prev_arr.size, dtype=np.float64)
                prev_arr = np.concatenate([prev_arr, pad])
            prev_arr = np.clip(prev_arr, 0.0, None)
            if prev_arr.sum() > 0:
                prev_arr /= prev_arr.sum()
                seeds.append(prev_arr)

    # deterministic cheap seeds
    seeds.extend([
        _uniform(n),
        _random_softmax(n, rng),
        _exponential_decay(n, rate=0.0015),
        _triangular(n),
    ])

    # pick the best seed according to the exact objective
    best_seed = None
    best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seed = s
    print(f"[Init] best initial objective = {best_val:.6f}")

    # ---------------------------------------------------------------
    # Phase 1 – Adam optimisation (smooth → hard max)
    # ---------------------------------------------------------------
    phase1_deadline = deadline - 200.0   # reserve time for later phases
    if time.time() < phase1_deadline:
        print("[Phase 1] Adam optimisation start")
        seq_adam, val_adam = _adam_optimize(best_seed, phase1_deadline)
        print(f"[Phase 1] best after Adam = {val_adam:.6f}")
    else:
        seq_adam, val_adam = best_seed, best_val
        print("[Phase 1] skipped (insufficient time)")

    # ---------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent on the hard max
    # ---------------------------------------------------------------
    phase2_deadline = deadline - 60.0
    if time.time() < phase2_deadline:
        print("[Phase 2] Sub‑gradient descent")
        seq_sub, val_sub = _subgradient_descent(seq_adam, phase2_deadline)
        if val_sub < val_adam:
            seq_adam, val_adam = seq_sub, val_sub
            print(f"[Phase 2] improved to {val_sub:.6f}")
        else:
            print("[Phase 2] no improvement")
    else:
        print("[Phase 2] skipped (time exhausted)")

    # ---------------------------------------------------------------
    # Phase 3 – Very light simulated‑annealing polishing
    # ---------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("[Phase 3] Simulated‑annealing polishing")
        seq_final, val_final = _anneal_refine(seq_adam, deadline)
        if val_final < val_adam:
            seq_adam, val_adam = seq_final, val_final
            print(f"[Phase 3] improved to {val_final:.6f}")
        else:
            print("[Phase 3] no improvement")
    else:
        print("[Phase 3] skipped (no time left)")

    # ---------------------------------------------------------------
    # Final normalisation and export
    # ---------------------------------------------------------------
    final_seq = np.maximum(seq_adam, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
        total = final_seq.sum()
    final_seq = final_seq / total

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
