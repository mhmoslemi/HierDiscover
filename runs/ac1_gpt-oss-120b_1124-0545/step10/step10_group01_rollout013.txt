"""
Propose a non‑negative step function that minimizes

    2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm works in three stages:

1️⃣  **Seed generation** – a very diverse pool of ℓ₁‑normalised vectors,
    including classic windows (Gaussian, Beta, Tukey, Kaiser …),
    random Dirichlet draws, low‑frequency Fourier curves,
    random piece‑wise linear shapes and mixtures of the above.
    The best few seeds (by the cheap surrogate 2·n·max(conv)) are kept.

2️⃣  **Log‑space Adam optimisation** – the sequence is parametrised by logits,
    a soft‑max turns them into a probability vector,
    and a smoothed‐max (log‑sum‑exp) surrogate of the convolution peak
    is minimised with Adam.  The temperature β is increased slowly up to 1000,
    so the surrogate converges to the true max.

3️⃣  **Exact sub‑gradient descent** – works on the true max‑convolution.
    After each step the vector is projected onto the probability simplex.

4️⃣  **Stochastic annealing** – a final aggressive polishing phase that
    reshuffles mass on random subsets and performs pair‑wise transfers,
    accepting uphill moves with a slowly‑cooling schedule.

All stages are time‑aware: they stop as soon as the global deadline
(≈ 1000 s) is reached.  The best sequence found is stored in the global
name ``height_sequence_1`` for possible reuse.  The final true objective
value (via the given ``evaluate_sequence``) is printed.
"""

import time

# ----------------------------------------------------------------------
# FFT utilities (linear convolution via real FFT)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate 2·n·max(conv(seq,seq)) assuming Σseq = 1.
    Returns a value that respects the ordering of candidates.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:

def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1))
                                            - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _kaiser_window(n: int, beta: float) -> np.ndarray:
    # import lazily – fall back to a pure‑numpy implementation if SciPy is missing
    try:
        from scipy.signal import kaiser as _kaiser
        a = _kaiser(n, beta)

        # rough fallback based on the modified Bessel function I0
        try:
            from scipy.special import i0 as _i0
        except Exception:
            _i0 = None

        y = (i - alpha) / alpha
        if _i0 is not None:

        else:
            t = beta * np.sqrt(1 - y * y)
            a = 1.0 + (t ** 2) / 4.0 + (t ** 4) / 64.0 + (t ** 6) / 2304.0 + (t ** 8) / 147456.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitudes/phases, then clip."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_piecewise_linear(n: int, knots: int, rng: np.random.Generator) -> np.ndarray:
    """Random piece‑wise linear curve (non‑negative)."""
    if knots < 2:
        return _uniform(n)
    xs = np.sort(rng.choice(n, size=knots, replace=False))
    heights = rng.uniform(0.0, 1.0, size=knots)
    a = np.interp(np.arange(n), xs, heights, left=0.0, right=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_fourier_square(n: int, terms: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random Fourier series (real), squared to enforce positivity,
    then ℓ₁‑normalised.
    """
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for _ in range(terms):
        amp = rng.normal()
        freq = rng.uniform(0.0, 0.5)    # relative to Nyquist
        phase = rng.uniform(0.0, 2.0 * np.pi)
        a += amp * np.sin(2.0 * np.pi * freq * t + phase)
    a = a ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_sparse_dirichlet(n: int, sparsity: float,
                             conc: float, rng: np.random.Generator) -> np.ndarray:
    """
    Choose a random subset of size ⌈sparsity·n⌉ and put a Dirichlet
    distribution on that support.
    """
    k = max(1, int(np.ceil(sparsity * n)))
    idx = rng.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = rng.dirichlet(np.full(k, conc))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed pool construction (more diverse than the previous version)
# ----------------------------------------------------------------------


def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a rich pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.15, 0.45, 7):
        seeds.append(_gaussian(n, sigma))

    # power‑law family
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # beta family – systematic grid
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential (heavy‑tailed)
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # random piecewise‑linear shapes
    for knots in (3, 5, 8):
        for _ in range(3):
            seeds.append(_random_piecewise_linear(n, knots, rng))

    # random Fourier‑square shapes
    for terms in (2, 4, 6):
        for _ in range(3):
            seeds.append(_random_fourier_square(n, terms, rng))

    # sparse Dirichlet families
    for sparsity in (0.05, 0.1, 0.2):
        for conc in (0.1, 0.5, 1.0):
            for _ in range(3):
                seeds.append(_random_sparse_dirichlet(n, sparsity, conc, rng))

    # mixtures of two analytic families (beta + gaussian, etc.)
    for _ in range(8):
        a = _beta_shape(n, rng.uniform(0.2, 5.0), rng.uniform(0.2, 5.0))
        b = _gaussian(n, rng.uniform(0.1, 0.4))
        alpha = rng.uniform(0.1, 0.9)
        seeds.append((1 - alpha) * a + alpha * b)

    # previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final clean‑up – normalise all vectors, replace pathological ones with uniform
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

def _logspace_adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on the log‑softmax parametrisation.
    Uses a log‑sum‑exp surrogate for the max‑convolution.
    Returns the best vector (ℓ₁‑normalised) found and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)                 # first moment
    v_sq = np.zeros_like(v)              # second moment

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5            # start temperature for soft‑max of the conv
    beta_inc = 0.15
    beta_max = 1000.0

    lr0 = 1.0             # initial learning rate in log‑space
    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                       # current probability vector
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)      # surrogate (no normalisation needed)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[LogAdam] it={it:6d} best={cur_val:.6f}")

        # smoothed max via log‑sum‑exp (temperature β)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to gradient on logits (log‑softmax)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update (learning‑rate decays like 1/√t)
        lr = lr0 / np.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule – approach β_max gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_500_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Uses a decreasing step‑size and Euclidean projection onto the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak‑style step‑size
        lr = 0.25 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-13,
) -> tuple[np.ndarray, float]:
    """
    Aggressive simulated annealing.  Two move types:
      – reshuffle a random subset (Dirichlet redistribution)
      – transfer a small chunk between two coordinates.
    Accepts uphill moves with a slowly‑cooling schedule.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.99995, min_temp)   # slower cooling

        if rng.random() < 0.68:          # subset reshuffle
            k = rng.integers(3, 14)      # size 3 … 13
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                            # pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.04)   # ≤ 4 % of source
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` inside the given time budget.
    Returns the best sequence as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Length of the vector – 4000 works well (good trade‑off)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – collect a very diverse set of seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # cheap surrogate ranking
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 15
    best_seeds = [vec for vec, _ in evals[:top_k]]

    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – log‑space Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break

        # allocate a fair slice of remaining time for this Adam run
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 0.10 * 1e5))
        print(f"\n--- Log‑Adam run {idx+1}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_adam, val_adam = _logspace_adam_optimize(
            seed_vec, n, deadline, max_iters=max_iters
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best (surrogate) = {val_adam:.6f}")

        # --------------------------------------------------------------
        # Sub‑gradient descent (exact max) – tighter budget
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        max_iters_sub = int(min(5_000_000, remaining * 0.40 * 1e5))
        if max_iters_sub > 0:
            seq_sub, val_sub = _subgradient_proj(
                global_best_seq, n, deadline, max_iters=max_iters_sub
            )
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[overall‑SubProj] new global best (surrogate) = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated annealing polishing (longest phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_iters_sa = int(min(2_500_000, remaining * 1e5))
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=max_iters_sa
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] new global best (surrogate) = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final true evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
