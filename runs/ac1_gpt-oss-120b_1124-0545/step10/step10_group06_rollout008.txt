def _project_simplex(v):
    # shift to ensure positivity and sum=1
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _project_simplex(v):
    # Ensure non-negative after projection onto simplex sum=1
    # Follow Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _subgrad_descent(start, deadline, max_iters=200000):
    a = start.copy()
    best_a = a.copy()
    best_val = _objective(a)
    n = a.size
    t = 0
    # lower bound
    lower = 2.0 * n / (2 * n - 1)   # small improvement but use as theoretical infimum 1?
    lower = max(1.0, lower)  # min bound ~1
    while t < max_iters and time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        M = conv.max()
        # choose max index (could be multiple)
        # to get subgradient direction, we can use average over all max indices
        max_idxs = np.flatnonzero(conv >= M - 1e-14)
        # choose one randomly
        k = max_idxs[np.random.randint(len(max_idxs))]
        # compute grad_M (size n)
        # grad_i = 2 * a[k-i] if 0 <= k-i < n else 0
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad_M = np.zeros_like(a)
        grad_M[mask] = 2.0 * a[j_arr[mask]]
        # gradient of objective F = 2 n M
        grad_F = 2.0 * n * grad_M
        # compute step size using Polyak
        cur_val = 2.0 * n * M
        # ensure lower bound not exceed cur_val
        target = lower
        # step = (cur_val - target) / (np.linalg.norm(grad_F)**2)
        denom = np.dot(grad_F, grad_F)
        if denom == 0:
            step = 0.0
        else:
            step = (cur_val - target) / denom
        # maybe clip step to max step e.g., 0.5 to avoid overshoot
        # but step may be too large, use safe factor
        step = max(0.0, min(step, 1.0))
        # update
        a_new = a - step * grad_F
        # project onto simplex
        a = _project_simplex(a_new)
        # evaluate objective occasionally
        if t % 1000 == 0 or step == 0.0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[Subgrad] it={t:7d} best={val:.6f}")
    return best_a, best_val

"""
Propose a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm has three stages:

1. **Adam on the simplex** – a smooth‑max surrogate is optimised with a
   soft‑max (log‑probability) parametrisation.  The smoothing parameter λ
   is raised slowly so the optimiser first sees a smooth landscape and
   finally the true max‑convolution objective.

2. **Projected sub‑gradient descent** – after the Adam phase the exact
   max‑convolution is used.  The sub‑gradient of the objective
   (`2·n·max(conv)`) is known analytically, so we take a Polyak‑type step
   and project back onto the probability simplex (Duchi‑et‑al. projection).

3. **Light simulated‑annealing polishing** – a short Metropolis walk
   swaps a tiny amount of mass between random coordinates and occasionally
   performs a global Gaussian jump.

All phases respect a total time budget (default 1000 s).  The best sequence
found is stored globally as ``height_sequence_1`` and returned as a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact evaluation (the same formula used by evaluate_sequence)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple seed families (return a probability vector that sums to 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    exps = np.exp(logits - m)
    return exps / exps.sum()


def _exponential_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        dec = inc[::-1]
    else:
        dec = np.arange(half, 0, -1, dtype=np.float64)
    a = np.concatenate([inc, dec])
    return a / a.sum()


# ----------------------------------------------------------------------
# Soft‑max weights (used by Adam)
# ----------------------------------------------------------------------
def _soft_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Return a probability vector w that approximates soft‑max over conv."""
    max_c = conv.max()
    if lam <= 0.0:
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.where(active, 1.0 / np.count_nonzero(active), 0.0)
        return w
    rel = (conv - max_c) / max_c          # ≤ 0
    exps = np.exp(lam * rel)
    w = exps / exps.sum()
    return w


def _gradient_wrt_a(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution w.r.t. the probability vector `a`.
    Implements   g = 2 * rev(a) ⊛ w   where w are the soft‑max weights.
    """
    n = a.size
    conv = _conv_fft(a, a)
    w = _soft_max_weights(conv, lam)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length = n + (2n‑1) – 1 = 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n]        # slice that matches `a`
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 300_000,
                   lam_start: float = 5.0,
                   lam_factor: float = 1.001) -> tuple[np.ndarray, float]:
    """Adam on the log‑probabilities of `start`."""
    eps = 1e-30
    logp = np.log(start + eps)          # shape (n,)
    n = start.size
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = start.copy()
    best_val = _objective(start)

    t = 0
    lam = lam_start

    while t < max_iters and time.time() < deadline:
        t += 1

        # current probability vector (soft‑max, stable)
        logp_shift = logp - np.max(logp)
        a = np.exp(logp_shift)
        a /= a.sum()

        # smoothed gradient
        g_a = _gradient_wrt_a(a, lam)

        # chain rule to get gradient w.r.t. log‑probabilities
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log * g_log)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.15 / (math.sqrt(t) + 1.0)

        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:7d}  best={cur_val:.6f}  λ={lam:.2f}")

        lam *= lam_factor

        # occasional tiny random kick
        if t % 25000 == 0:
            noise = np.random.default_rng().normal(scale=0.02, size=n)
            logp += noise

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def _subgrad_descent(start: np.ndarray,
                     deadline: float,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Polyak‑type sub‑gradient descent on the true objective.
    Uses the exact max‑convolution to obtain a sub‑gradient,
    takes a step (cur‑opt)/(‖g‖²) and projects onto the simplex.
    """
    a = start.copy()
    best_a = a.copy()
    best_val = _objective(a)
    n = a.size

    # a safe theoretical lower bound: 2n/(2n‑1) → 1 as n→∞
    lower_bound = max(1.0, 2.0 * n / (2 * n - 1))

    rng = np.random.default_rng()
    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1

        conv = _conv_fft(a, a)
        M = conv.max()
        # pick a random index among those attaining the max
        max_idxs = np.flatnonzero(conv >= M - 1e-14)
        k = max_idxs[rng.integers(len(max_idxs))]

        # sub‑gradient of M (size n)
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        grad_M = np.zeros_like(a)
        grad_M[mask] = 2.0 * a[j[mask]]

        # gradient of the full objective F = 2·n·M
        grad_F = 2.0 * n * grad_M

        cur_val = 2.0 * n * M
        denom = np.dot(grad_F, grad_F)
        if denom == 0.0:
            step = 0.0
        else:
            step = (cur_val - lower_bound) / denom
            # keep step modest to avoid wild jumps
            step = max(0.0, min(step, 0.5))

        a_new = a - step * grad_F
        a = _project_simplex(a_new)

        # occasional reporting
        if t % 1000 == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[Subgrad] it={t:7d}  best={val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Very light simulated‑annealing neighbour search (final polishing)
# ----------------------------------------------------------------------
def _anneal_refine(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis–Hastings walk that swaps a small fraction of mass
    between random coordinates and occasionally performs a global Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0
    n = cur.size

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # small transfer from i to j
        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d}  best={new_val:.6f}")

        # global perturbation every ~2000 steps
        if it % 2000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0.0:
                new_seq /= total
            else:
                new_seq = _uniform(n)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val
