def solve_lp_for_f(f, M, active_idx):
    n = len(f)
    rows = []
    cols = []
    data = []
    for idx, k in enumerate(active_idx):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # j indices where b_j appears
        for j in range(i_start, i_end+1):
            rows.append(idx)
            cols.append(j)
            i_f = k - j
            data.append(f[i_f])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    # linear program
    c = -np.ones(n)  # maximize sum -> minimize -sum
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(len(active_idx), M), bounds=bounds, method='highs')
    if not res.success:
        # fallback: uniform distribution
        b = np.full(n, 1.0 / n)
    else:
        b = res.x
    return b
```

"""
Propose a non‑negative step‑function (list of heights) that minimises
evaluate_sequence().  The algorithm proceeds in four stages:

1. **Initial seeds** – a small set of analytic shapes (uniform, Hann,
   Catalan, arcsine, Beta…) are evaluated on a few lengths (4000, 5000,
   6000).  The best of them (or a previously stored solution) seeds the
   optimisation.

2. **Smooth Adam optimisation** – the log‑sum‑exp surrogate of the
   max‑convolution is minimised with Adam.  The temperature (β) is
   annealed from a modest value to a large one (≈2000) so that the
   surrogate approaches the true max.  All iterates are projected onto
   the probability simplex and symmetrised (a[i]=a[n‑1‑i]).

3. **Frank‑Wolfe–style LP refinement** – given the current shape f,
   we solve a tiny linear programme:
        maximise   Σ b_j
        subject to b_j ≥ 0,
                   (f * b)_k ≤ M   for every k where (f * f)_k attains the
                                   current maximum M.
   The solution b is normalised to a simplex point g and mixed with f
   (f←(1−t)f+t g, t≈0.5).  The LP uses only the *active* convolution
   indices, so the constraint matrix stays tiny.  The step is repeated
   while time permits.

4. **High‑β soft‑max gradient descent** – a final projected gradient
   descent on the exact (non‑smooth) objective, using a soft‑max
   approximation with a linearly increasing β.  This polishes the last
   few digits.

Throughout the run the best sequence (and its objective value) is
tracked.  All operations respect the 1000 s budget and return a plain
list of floats ready for evaluate_sequence().
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via FFT (real)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to evaluate_sequence for a *normalised* sequence (sum == 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry v[i] = v[-i-1] and re‑project onto the simplex."""
    w = (v + v[::-1]) / 2.0
    return _project_simplex(w)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:

    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0 - a1 * np.cos(2 * np.pi * i / (n - 1)) + a2 * np.cos(4 * np.pi * i / (n - 1)))
    return w / w.sum()


def _gaussian_sigma025(n: int) -> np.ndarray:

    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)

    if w.sum() == 0.0:

    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) pdf (shifted by 0.5 to avoid the exact boundaries)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a point on the simplex (sum=1) using a Dirichlet(alpha) distribution."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s

def _smooth_adam_optimize(start_seq: np.ndarray,
                          deadline: float,
                          beta_start: float = 5.0,
                          beta_end: float = 2000.0,
                          max_iter: int = 5_000_000,
                          lr0: float = 0.03) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv(a,a)).
    The iterate a is kept on the simplex and symmetrised after each step.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        a = np.exp(logits)
        a = a / a.sum()
        a = _symmetrize(a)          # keep shape symmetric

        conv = _conv_fft(a, a)

        # linear schedule for β
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()

        # gradient w.r.t. a (via softmax of convolutions)
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if (it & 1023) == 0:            # every 1024 iterations
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")
            if best_val < 1.503:       # target reached → stop early
                break
    return best_seq, best_val

# Frank‑Wolfe style Linear‑Program refinement

def _solve_lp_max_sum(f: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray:
    """
    Solve   maximise Σ b_j   s.t.   b≥0  and  (f * b)_k ≤ M
    only for the indices in ``active_idx`` (where f*f attains its current
    maximum).  Returns the optimal b (not normalised).
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for row, k in enumerate(active_idx):
        # b_j participates iff 0 ≤ j ≤ n‑1 and 0 ≤ k‑j ≤ n‑1
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(row)
            cols.append(j)
            data.append(f[k - j])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    c = -np.ones(n)                     # maximise Σ b => minimise -Σ b
    b_ub = np.full(len(active_idx), M)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                method='highs')
    if not res.success:
        # fallback: uniform distribution
        return np.full(n, 1.0 / n)
    return res.x


def _lp_refinement(start_seq: np.ndarray,
                   deadline: float,
                   mix_factor: float = 0.5,
                   max_iter: int = 30) -> tuple[np.ndarray, float]:
    """
    Repeatedly improve the sequence by solving the small LP described above
    and mixing the solution with the current iterate.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_fft(a, a)
        M = float(conv.max())

        # active constraints – all positions within a tiny relative tolerance
        eps_rel = 1e-9
        thresh = M - eps_rel * M
        active_idx = np.where(conv >= thresh)[0]
        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv)], dtype=int)

        b = _solve_lp_max_sum(a, M, active_idx)

        # normalise b to the simplex
        sb = b.sum()
        if sb <= 0:
            b = _uniform(len(a))
        else:
            b = b / sb

        # convex combination + projection & symmetrisation
        a_next = (1.0 - mix_factor) * a + mix_factor * b
        a_next = _project_simplex(a_next)
        a_next = _symmetrize(a_next)

        val = _objective(a_next)
        if val < best_val:
            best_val = val
            best_seq = a_next.copy()
            print(f"[LP] iter={it:2d} val={val:.6f}")

        a = a_next

        # stop early if improvement stalls
        if it >= 5 and abs(val - best_val) < 1e-7:
            break
    return best_seq, best_val

def _softmax_gradient_descent(start_seq: np.ndarray,
                              deadline: float,
                              beta_start: float = 20.0,
                              beta_end: float = 3000.0,
                              max_iter: int = 4_000_000,
                              lr0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact max‑convolution objective,
    using a soft‑max weighting that becomes sharper as β grows.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_fft(a, a)
        max_c = conv.max()

        # β schedule (linear)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / max_iter)

        # soft‑max weights (numerically stable)
        exp_term = np.exp(beta * (conv - max_c))
        w = exp_term / exp_term.sum()

        # gradient = 2 * (w * a_rev) convolved with a
        conv_ag = _conv_fft(w, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        # step size
        lr = lr0 / math.sqrt(it)

        a = a - lr * g
        a = _project_simplex(a)
        a = _symmetrize(a)

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[SoftmaxGD] it={it:7d} val={val:.6f}")
            if best_val < 1.503:      # stop as soon as we beat the target
                break

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Determine lengths to explore (FFT‑friendly, medium‑size)
    # ------------------------------------------------------------------
    lengths = [4000, 5000, 6000]   # larger lengths gave diminishing returns
    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 0 – deterministic analytic seeds (and possible previous solution)
    # ------------------------------------------------------------------
    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        _gaussian_sigma025,
        _arcsine_shape,
        _beta_shape_05_05,
        _beta_shape_2_2,
        _beta_shape_05_2,
        _beta_shape_2_05,
    )

    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

    # ------------------------------------------------------------------
    # Include a previously stored solution if available
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                prev_arr = prev_arr / prev_arr.sum()
                val = _objective(prev_arr)
                if val < best_val:
                    best_val = val
                    best_seq = prev_arr.copy()
                    print(f"[Prev] reused sequence val={val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Phase 1 – smooth Adam optimisation (few restarts)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0 and best_seq is not None:
        adam_deadline = time.time() + remaining * 0.35
        n = best_seq.size
        candidates = [best_seq]
        # add a handful of random Dirichlet restarts
        for _ in range(3):
            cand = _random_dirichlet(n, rng, alpha=rng.choice([0.5, 1.0, 2.0]))
            candidates.append(cand)

        for idx, cand in enumerate(candidates):
            seq_adam, val_adam = _smooth_adam_optimize(
                start_seq=cand,
                deadline=adam_deadline,
                beta_start=5.0,
                beta_end=2000.0,
                max_iter=4_000_000,
                lr0=0.03,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = seq_adam.copy()
                print(f"[Adam‑Refine {idx}] new best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Frank‑Wolfe style LP refinement (active constraints only)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and best_seq is not None:
        lp_deadline = time.time() + remaining * 0.40
        seq_lp, val_lp = _lp_refinement(best_seq, lp_deadline,
                                         mix_factor=0.5, max_iter=30)
        if val_lp < best_val:
            best_val = val_lp
            best_seq = seq_lp.copy()
            print(f"[LP‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – high‑β soft‑max projected gradient descent (polishing)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0 and best_seq is not None:
        seq_final, val_final = _softmax_gradient_descent(
            start_seq=best_seq,
            deadline=time.time() + remaining,
            beta_start=20.0,
            beta_end=3000.0,
            max_iter=8_000_000,
            lr0=0.2,
        )
        if val_final < best_val:
            best_val = val_final
            best_seq = seq_final.copy()
            print(f"[Final‑Polish] best now {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (evaluate_sequence will be called by the grader)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
