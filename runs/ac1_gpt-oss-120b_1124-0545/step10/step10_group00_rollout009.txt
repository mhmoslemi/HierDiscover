  draws.  The best few seeds are kept for further optimisation.

* Optimise the smooth “log‑sum‑exp” surrogate of the true objective with
  Adam (the surrogate is a differentiable upper bound on the max‑convolution).

* Refine the best candidate with exact sub‑gradient mirror descent
  (exponentiated‑gradient) that directly minimises the true max‑convolution.

* Polish the result with a specialised simulated‑annealing (SA) that
  combines generic swaps, block redistributions, pair‑mass transfers and a
  **targeted** move that actively reduces the current peak of the convolution.

The whole search stops after the allotted time budget (default 1000 s) and
returns the best sequence found.  The resulting list is also stored in the
global name ``height_sequence_1`` for possible reuse in subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised sequence
    (i.e. Σseq = 1).  The factor 2·n does not affect ordering of candidates.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all ℓ¹‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)  (lam ≥ 0)"""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random families & ordering helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)  # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))


# ----------------------------------------------------------------------
# Low‑frequency smooth curves (useful as seeds)
# ----------------------------------------------------------------------
def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection (much richer than the reference implementation)
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ¹‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # finer sweep of sine windows
    for gamma in np.linspace(0.5, 5.0, 10):
        seeds.append(_sine_window(n, gamma=gamma))

    # finer sweep of Gaussians
    for sigma_frac in np.linspace(0.05, 0.60, 12):
        seeds.append(_gaussian(n, sigma_frac=sigma_frac))

    # power‑law sweep
    for p in np.linspace(0.0, 4.0, 21):
        seeds.append(_power(n, p))

    # beta‑distribution grid (more points)
    alphas = np.linspace(0.3, 2.5, 9)
    betas = np.linspace(0.3, 5.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows (dense grid)
    for alpha in np.linspace(0.05, 0.95, 10):
        seeds.append(_tukey_window(n, alpha))

    # exponential decays (helpful for asymmetric shapes)
    for lam in np.linspace(0.0, 0.005, 7):  # lam tiny because n is large
        seeds.append(_exponential_decay(n, lam))

    # Random Dirichlet draws – different concentration parameters
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw random order
            seeds.append(_sorted_desc(raw))       # decreasing
            seeds.append(_sorted_asc(raw))        # increasing
            seeds.append(_alternate_order(raw))   # alternating pattern

    # Low‑frequency random curves – act as smooth seeds
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray | None, float]:
    """Adam optimisation on the log‑softmax parameters using a smoothed‑max."""
    eps = 1e-30
    v = np.log(start_vec + eps)  # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8
    beta = 0.5          # start temperature
    beta_inc = 0.5
    beta_max = 500.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                 # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam] iter={it:7d} best={cur_val:.6f}")

        # --- smoothed max via log‑sum‑exp ---
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                    # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]          # central n entries

        # convert to log‑space gradient (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # temperature schedule (slowly tighten the surrogate)
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 3_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the exact
    max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50_000 == 0:
                print(f"[Mirror] iter={it:7d} best={cur_val:.6f}")

        # sub‑gradient corresponding to the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update (exponentiated gradient)
        lr = 0.12 / np.sqrt(it)          # step‑size decays
        a = a * np.exp(-lr * g)

        # renormalise onto the simplex (positivity preserved)
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _simulated_annealing_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_500_000,
    init_temp_factor: float = 0.03,
    min_temp: float = 1e-12,
    target_move_prob: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing with four move types:
        0 – swap two entries
        1 – redistribute the mass of a small random subset (Dirichlet)
        2 – pair‑mass transfer (split between two positions)
        3 – *targeted* reduction of the current convolution peak
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9995, min_temp)

        # compute current convolution once per iteration (needed for target move)
        conv_cur = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv_cur))

        move_type = rng.random()
        if move_type < target_move_prob:
            # ---- TARGETED MOVE -------------------------------------------------
            # Identify the pair (i,j) contributing most to the peak.
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            j_range = max_idx - i_range
            contrib = cur[i_range] * cur[j_range]
            # ignore i=j (self‑product) – it is harder to split cleanly
            mask = i_range != j_range
            if not np.any(mask):
                # fall back to a generic move
                move_type = 0.0
            else:
                i_range = i_range[mask]
                j_range = j_range[mask]
                contrib = contrib[mask]
                idx = int(np.argmax(contrib))
                i = int(i_range[idx])
                j = int(j_range[idx])

                # decide how much mass to shift
                delta = min(cur[i], cur[j]) * 0.10   # move 10 % of the smaller partner
                if delta <= 0.0:
                    move_type = 0.0
                else:
                    # choose a new pair (p,q) that does NOT recreate the same sum
                    attempts = 0
                    while attempts < 5:
                        p = rng.integers(0, n)
                        q = rng.integers(0, n)
                        if p + q != max_idx and p != i and q != j:
                            break
                        attempts += 1
                    else:
                        # failed to find a clean pair – swap instead
                        move_type = 0.0

        if move_type >= target_move_prob:
            # ---- NON‑TARGETED MOVES --------------------------------------------
            move_type = rng.integers(0, 3)   # 0,1,2
        # ---- Execute the chosen move -----------------------------------------
        if move_type == 0:
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            k = rng.integers(2, 10)          # size of the subset
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        elif move_type == 2:
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]
        else:   # targeted move (move_type == 3)
            new = cur.copy()
            new[i] -= delta
            new[j] -= delta
            new[p] += delta
            new[q] += delta

        new_val = _objective(new)
        delta_val = new_val - cur_val
        if delta_val < 0.0 or rng.random() < np.exp(-delta_val / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 100_000 == 0:
                    print(f"[SA] iter={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats and stores it
    in the global name ``height_sequence_1`` for possible reuse.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # Reserve a couple of seconds for the final sanity check.
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length choice – 4000 gives a good balance between expressive power
    # and evaluation speed (the reference solution also used 4000).
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Previous best (if any) – enables warm‑starts across separate runs.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a wide range of analytic and random seeds,
    #           keep the top few according to the cheap surrogate.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation on the smooth surrogate.
    # Allocate roughly 15 % of the remaining time per seed (capped).
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # Rough estimate: 30 % of the remaining budget for Adam runs
        max_iters = min(250_000, int(remaining * 40))
        print(f"\n--- Adam optimisation run {idx}/{len(best_seeds)} "
              f"(max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_optimize(
            seed_vec, n, deadline, max_iters=max_iters
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Mirror‑descent sub‑gradient optimisation (exact max)
    #-------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent sub‑gradient optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=3_000_000
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[overall‑Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing polishing with targeted moves
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing (targeted) ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline, max_iters=2_500_000,
            target_move_prob=0.35
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator (optional)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
