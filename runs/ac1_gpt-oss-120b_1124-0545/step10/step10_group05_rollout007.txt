def _hardmax_mirror_descent(start_seq, deadline):
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # if max appears in multiple positions, randomly pick one
        i_max = int(np.flatnonzero(conv == max_val)[np.random.randint(len(np.flatnonzero(conv==max_val)))])
        # compute gradient for that index:
        # create one-hot vector p of length 2n-1
        p = np.zeros(2 * n - 1, dtype=np.float64)
        p[i_max] = 1.0
        grad = 2.0 * _conv_fft(p, a[::-1])[n-1:n-1+n]
        # step size schedule
        lr = 0.1 / np.sqrt(it)   # tune
        # line search:
        dec_factor = 0.5
        for ls in range(5):
            log_a = np.log(a + 1e-30) - lr * grad
            log_a -= np.max(log_a)
            a_new = np.exp(log_a)
            a_new /= a_new.sum()
            val_new = _objective(a_new)
            if val_new <= best_val:
                a = a_new
                if val_new < best_val:
                    best_val = val_new
                    best_seq = a_new.copy()
                    print(f"[HardMD] it={it} val={val_new:.6f}")
                break
            else:
                lr *= dec_factor
                if lr < 1e-12:
                    break
    return best_seq, best_val
```

def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    n = len(f)
    L = 2*n - 1
    # Build A_ub matrix as sparse
    from scipy.sparse import diags
    # diag_vals is list of arrays shape (n,) each filled with f[d]
    diag_vals = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    A_ub = diags(diag_vals, offsets=range(n), shape=(L, n), format='csr')
    b = np.full(L, M, dtype=np.float64)
    # solve LP
    res = opt.linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    g = res.x
    if g.sum() <= 0:
        return None
    return g / g.sum()
```

"""
Improved search for a low‑value step function.
------------------------------------------------
Key ideas
---------

* Use the best sequence already found (global ``height_sequence_1``) as a
  starting point.
* Perform a fast *hard‑max* mirror‑descent: at each iteration the exact
  index of the maximal convolution value is used to compute a sub‑gradient.
  An exponentiated‑gradient (KL‑mirror) update keeps the iterate on the
  probability simplex and the step‑size decays as 1/√t.
* When SciPy is available, run a few iterations of a linear‑program
  “LP‑step’’ (the classic Frank‑Wolfe/LP‑update described in the problem
  statement).  For a fixed f, the LP maximises Σg_j subject to
  (f∗g)_i ≤ M (M = max(f∗f)).  The solution g is mixed with f
  (small convex combination) and the process is repeated – this reliably
  lowers the maximal self‑convolution.
* Finish with the aggressive aggregated mirror‑descent from the previous
  baseline to polish the result.
* All phases respect the global 1000 s budget and continually keep the
  best sequence seen so far.

The algorithm is fully self‑contained, uses only NumPy (and SciPy when
available) and returns a plain Python list of non‑negative numbers.
"""

import time
import numpy as np
from math import sqrt, pi

# ----------------------------------------------------------------------
# Optional SciPy imports (LP step)
# ----------------------------------------------------------------------
try:
    import scipy.sparse as sp
    import scipy.optimize as opt
    HAVE_SCIPY = True
except Exception:  # pragma: no cover
    HAVE_SCIPY = False

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a *normalised* sequence
    (sum == 1).  Returns ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = a0 - a1 * np.cos(2 * pi * i / (n - 1)) + a2 * np.cos(4 * pi * i / (n - 1))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return w / w.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    return w / w.sum()


# ----------------------------------------------------------------------
# Hard‑max mirror‑descent (sub‑gradient on the exact max)
# ----------------------------------------------------------------------


def _hardmax_mirror_descent(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient method that always uses the *exact* maximal convolution entry.
    Mirror‑descent is performed in the log‑domain (KL‑projection) and the step
    size decays as 1/√iteration.  Returns the best sequence seen.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()

        # pick one of the (possibly) multiple maximisers uniformly at random
        max_idxs = np.flatnonzero(conv == max_val)
        i_max = int(np.random.choice(max_idxs))

        # one‑hot vector for the active index
        p = np.zeros(2 * n - 1, dtype=np.float64)
        p[i_max] = 1.0

        # sub‑gradient of the (hard) max w.r.t. a
        grad = 2.0 * _conv_fft(p, a[::-1])[n - 1:n - 1 + n]

        # learning‑rate schedule
        lr = 0.15 / np.sqrt(it)

        # simple back‑tracking line search (max 5 attempts)
        improved = False
        lr_try = lr
        for _ in range(5):
            log_a = np.log(a + eps) - lr_try * grad
            log_a -= np.max(log_a)          # numeric stabilisation
            a_new = np.exp(log_a)
            a_new /= a_new.sum()
            val = _objective(a_new)

            if val < best_val:
                best_val = val
                best_seq = a_new.copy()
                a = a_new
                improved = True
                print(f"[HardMD] it={it:7d} val={val:.6f}")
                break
            else:
                lr_try *= 0.5

        if not improved:
            # perform a tiny step anyway to keep moving
            log_a = np.log(a + eps) - lr_try * grad
            log_a -= np.max(log_a)
            a = np.exp(log_a)
            a /= a.sum()

        # early exit if we have already reached a very good value
        if best_val < 1.45:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming (LP) update – the “Frank‑Wolfe/LP’’ step described
# ----------------------------------------------------------------------


def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ g_j   subject to   g ≥ 0  and  (f∗g)_i ≤ M   (∀i).
    Returns the optimiser g normalised to sum = 1, or ``None`` on failure.
    """
    if not HAVE_SCIPY:  # pragma: no cover
        return None

    n = len(f)
    L = 2 * n - 1

    # Build the Toeplitz‑type constraint matrix A_ub (size L×n).
    # Row i, column j contains f[i‑j] if 0 ≤ i‑j < n, otherwise 0.
    # This is exactly a collection of n diagonals with constant value f[d].
    diag_vals = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    A_ub = sp.diags(diag_vals, offsets=range(n), shape=(L, n), format='csr')
    b = np.full(L, M, dtype=np.float64)

    # Linear program: minimise -Σ g_j  (i.e. maximise Σ g_j)
    res = opt.linprog(
        c=-np.ones(n, dtype=np.float64),
        A_ub=A_ub,
        b_ub=b,
        bounds=[(0, None)] * n,
        method='highs',
        options={'presolve': True}
    )
    if not res.success:
        return None

    g = res.x
    if g.sum() <= 0.0:
        return None
    return g / g.sum()


def _lp_descent(start_seq: np.ndarray, deadline: float,
                t: float = 0.25, max_iter: int = 15) -> tuple[np.ndarray, float]:
    """
    Repeated LP‑steps (as above) with a small convex combination.
    ``t`` is the mixing weight (f←(1‑t)f + t g) and may be gradually reduced.
    """
    if not HAVE_SCIPY:  # pragma: no cover
        return start_seq, _objective(start_seq)

    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        M = _conv_fft(a, a).max()
        g = _lp_step(a, M)
        if g is None:
            break

        # convex combination and re‑normalisation
        a = (1.0 - t) * a + t * g
        a /= a.sum()

        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[LP] it={it:3d} val={val:.6f}")
        # slowly shrink the mixing factor
        t = max(t * 0.90, 0.02)

        if best_val < 1.45:  # early stop
            break

    return best_seq, best_val

# Aggressive aggregated mirror‑descent (from the reference code)

def _aggregated_mirror_descent(
    start_seq: np.ndarray,
    deadline: float,
    eps_rel_start: float = 0.01,
    eps_rel_end: float = 1e-6,
    lr_factor: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent where the sub‑gradient aggregates *all* near‑maximal
    convolution entries.  The tolerance decays exponentially.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    t0 = time.time()
    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        # exponential decay of the relative tolerance
        elapsed = time.time() - t0
        total = deadline - t0
        frac = elapsed / total if total > 0 else 0.0
        eps_rel = eps_rel_start * (eps_rel_end / eps_rel_start) ** frac
        delta = eps_rel * max_val

        active_idx = np.where(conv >= max_val - delta)[0]
        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        lr = lr_factor / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:           # every 1024 iterations
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[AggMD] it={it:7d} best={val:.6f}")

            if time.time() > deadline - 0.5:
                break

    return best_seq, best_val
