"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(seq,seq)) / (sum(seq)²)

using a richer search pipeline:

* diverse analytic & random seeds (including additional exponential &
  cosine‑power families);
* Adam optimisation on a log‑softmax parametrisation with a
  temperature‑controlled log‑sum‑exp surrogate for the max‑convolution;
* mirror‑descent (exponentiated‑gradient) sub‑gradient descent that
  aggregates the sub‑gradients of all near‑maximal convolution entries;
* a longer simulated‑annealing polishing phase with extra move types;
* dynamic time‑budget handling and early exit when the target value
  (1.503) is reached.

All helpers are defined at module level – no closures or lambdas – and
the function returns a plain ``list[float]`` that can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised
    sequence (Σ seq = 1).  The factor 2·n is kept because it does not
    affect the ordering of candidates.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

# Analytic shape families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) with a final reversal to keep mass away from edge."""
    if rate <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (cos(π * i / (2*n)))^p, zero‑clipped."""
    if p <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * i / (2.0 * n)) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Random families & ordering helpers

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector of length n with concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    s = a.sum()
    return a / s if s > 0 else _uniform(len(vec))


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """Largest, smallest, 2nd‑largest, 2nd‑smallest, …"""
    a = np.sort(vec)          # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    s = out.sum()
    return out / s if s > 0 else _uniform(len(vec))

# Low‑frequency smooth seeds
# ----------------------------------------------------------------------
def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Seed collection (diverse family of ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_gaussian(n, sigma_frac=0.40))
    seeds.append(_exponential_decay(n, rate=0.001))
    seeds.append(_exponential_decay(n, rate=0.005))
    seeds.append(_cosine_power(n, p=1.0))
    seeds.append(_cosine_power(n, p=2.0))
    seeds.append(_cosine_power(n, p=4.0))

    # power‑law family
    for p in np.linspace(0.0, 4.0, 13):
        seeds.append(_power(n, p))

    # beta family (broader search)
    for a in np.linspace(0.1, 3.0, 7):
        for b in np.linspace(0.1, 5.0, 9):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in (0.05, 0.1, 0.2, 0.4, 0.6, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw order
            seeds.append(_sorted_desc(raw))       # decreasing
            seeds.append(_sorted_asc(raw))        # increasing
            seeds.append(_alternate_order(raw))   # alternating

    # Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

def _adam_optimize(start_vec: np.ndarray, n: int,
                   deadline: float, max_iters: int = 200_000) -> tuple[np.ndarray | None, float]:
    """
    Adam on log‑softmax parameters with a log‑sum‑exp surrogate for max(conv).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8
    beta = 0.5           # start temperature
    beta_inc = 0.5
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # --- smooth max via log‑sum‑exp weighting ---
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # Gradient of Σ w_k·conv_k wrt a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # Chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # increase temperature gradually
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_subgradient_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 2_000_000,
                                 epsilon_frac: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) descent using a sub‑gradient that
    aggregates over all convolution entries whose value lies within
    `epsilon_frac`·max of the current maximum.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    lr0 = 0.12  # base learning rate

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        cur_val = 2.0 * n * max_val

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

        # indices that are “almost maximal”
        thr = max_val - epsilon_frac * max_val
        ks = np.nonzero(conv >= thr)[0]

        # accumulate sub‑gradient over all these ks
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]

        # multiplicative update
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # renormalise (projection onto simplex is just division)
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _simulated_annealing_optimize(start_seq: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 1_500_000,
                                 init_temp_factor: float = 0.03,
                                 min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated annealing with three move types:
    * swap two entries,
    * re‑allocate mass on a random subset (Dirichlet),
    * transfer mass between a random pair.
    Additionally, a small Gaussian‑blur is occasionally applied to
    encourage smoothness.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cooling schedule
        T = max(T * 0.9995, min_temp)

        move_type = rng.integers(0, 4)   # 0=swap, 1=redistribute, 2=pair‑transfer, 3=blur
        if move_type == 0:
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        elif move_type == 1:
            k = rng.integers(2, 12)    # subset size
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        elif move_type == 2:
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total <= 0.0:
                continue
            split = rng.dirichlet([1.0, 1.0]) * total
            new = cur.copy()
            new[i], new[j] = split[0], split[1]
        else:  # blur move – apply a tiny 3‑point smoothing kernel
            kernel = np.array([0.1, 0.8, 0.1], dtype=np.float64)
            new = np.convolve(cur, kernel, mode='same')
            # after blur, renormalise to keep total mass the same
            new = new * (cur.sum() / new.sum())

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats and stores it
    in the global name ``height_sequence_1`` for possible reuse.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for the final sanity check
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length choice – 5000 provides a good trade‑off between expressive
    # power and speed (still well within the time budget).
    # ------------------------------------------------------------------
    n = 5000
    print(f"\n[Info] Using length n = {n}")

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any) for seeding
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – seed generation & quick ranking
    # ------------------------------------------------------------------
    print("\n--- Stage 1: seed generation ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation (smooth surrogate) on selected seeds
    # ------------------------------------------------------------------
    print("\n--- Stage 2: Adam optimisation ---")
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        # allocate a modest chunk of remaining time per Adam run
        remaining = deadline - time.time()
        max_iters = int(min(200_000, remaining * 45))  # ~45 iters per second
        print(f"\n[Adam] run {idx}/{len(best_seeds)} – max_iters={max_iters}")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Mirror‑descent sub‑gradient optimisation (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Stage 3: Mirror‑descent optimisation ---")
        seq_mirror, val_mirror = _mirror_subgradient_optimize(
            global_best_seq, n, deadline, max_iters=1_800_000, epsilon_frac=0.001
        )
        if val_mirror < global_best_val:
            global_best_seq = seq_mirror.copy()
            global_best_val = val_mirror
            print(f"[Mirror] new global best = {val_mirror:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing polishing (longer run)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Stage 4: Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_optimize(
            global_best_seq, n, deadline, max_iters=1_800_000,
            init_temp_factor=0.04, min_temp=1e-14
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Optional final refinement with a short Adam pass (helps escape
    # tiny plateaus after SA)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Stage 5: Final Adam refinement ---")
        remaining = deadline - time.time()
        max_iters = int(min(100_000, remaining * 45))
        seq_final, val_final = _adam_optimize(global_best_seq, n, deadline, max_iters=max_iters)
        if seq_final is not None and val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Final Adam] new global best = {val_final:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check using the official evaluator
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
