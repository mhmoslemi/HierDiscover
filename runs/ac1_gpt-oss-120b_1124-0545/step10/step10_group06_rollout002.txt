def solve_lp_max_sum(f, M):
    n = f.size
    # Build A_ub shape (2n-1, n)
    # Precompute container of rows
    # Use SciPy's sparse matrix to improve speed maybe.
    from scipy.sparse import csr_matrix

    # Build data, row, col.
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i from i_start to i_end inclusive
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            data.append(f[i])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    # b_ub = M * np.ones(2n-1)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n) # maximize sum(b)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        return None
    return res.x

"""
Improved optimiser for the step‑function bound.

The search is split into three stages under the overall time budget:

1. **Multi‑restart Adam on the simplex** – a few short Adam runs from
   different random seeds (log‑softmax parametrisation).  The smoothing
   parameter λ grows very slowly so the optimiser sees a smooth landscape
   for most of the iterations and only switches to the true max‑convolution
   at the end.

2. **Long simulated‑annealing (SA) polishing** – a Metropolis walk that
   repeatedly transfers a small amount of mass between two coordinates,
   swaps values, adds a global jitter and occasionally redistributes mass
   among three coordinates.  The temperature schedule is adaptive and the
   walk continues until the time budget for this phase is exhausted.

3. **Exponential‑gradient sub‑gradient refinement** – a final pass that
   uses the exact sub‑gradient of the (hard) max‑convolution and updates
   the probability vector with an exponentiated‑gradient step, keeping it
   on the simplex.  This cleans up the remaining tiny irregularities.

All stages keep the best sequence seen so far.  The final sequence is
normalised to sum to one, stored in the global ``height_sequence_1`` and
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).  Assumes sum>0."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed generators (return probability vectors that sum to 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    exps = np.exp(logits - m)
    return exps / exps.sum()


def _exponential_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    dec = inc[::-1] if n % 2 == 0 else np.arange(half, 0, -1, dtype=np.float64)
    a = np.concatenate([inc, dec])
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max weighting for the smoothed max‑convolution
# ----------------------------------------------------------------------
def _soft_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max over convolution entries, lam is an inverse temperature."""
    max_c = conv.max()
    if lam <= 0.0:
        # hard max – uniform distribution over all exact ties
        active = np.isclose(conv, max_c, atol=1e-12)
        cnt = np.count_nonzero(active)
        if cnt == 0:
            return np.zeros_like(conv)
        return np.where(active, 1.0 / cnt, 0.0)
    # stable computation
    rel = (conv - max_c) / max_c      # ≤ 0
    exps = np.exp(lam * rel)
    return exps / exps.sum()


def _gradient_wrt_a(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of (soft) max‑convolution w.r.t. probability vector `a`.
    Implements   g = 2·rev(a) ⊛ w,
    where w are the soft‑max weights over the convolution entries.
    """
    n = a.size
    conv = _conv_fft(a, a)
    w = _soft_max_weights(conv, lam)          # shape = 2n‑1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length = 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n]        # slice that matches `a`
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 250_000,
                   lam_start: float = 5.0,
                   lam_factor: float = 1.0008) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on log‑probabilities (soft‑max) of `start`.
    The smoothing parameter λ is increased slowly so the optimiser starts
    on a smooth approximation of the max‑convolution and finishes on the
    true (hard) objective.
    Returns the best probability vector (sums to 1) and its objective.
    """
    eps = 1e-30
    logp = np.log(start + eps)                # initialise log‑probabilities
    n = start.size
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = start.copy()
    best_val = _objective(start)

    t = 0
    lam = lam_start

    while t < max_iters and time.time() < deadline:
        t += 1

        # current simplex point
        logp_shift = logp - np.max(logp)       # for stability
        a = np.exp(logp_shift)
        a /= a.sum()

        # gradient of smoothed max‑convolution
        g_a = _gradient_wrt_a(a, lam)

        # chain rule to log‑space
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log * g_log)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.15 / (math.sqrt(t) + 1.0)      # slowly decaying step size
        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # evaluate true (non‑smoothed) objective
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:7d}  best={cur_val:.6f}  λ={lam:.2f}")

        # sharpen the soft‑max
        lam *= lam_factor

        # occasional tiny Gaussian kick to escape plateaus
        if t % 25000 == 0:
            noise = np.random.default_rng().normal(scale=0.02, size=n)
            logp += noise

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light simulated‑annealing neighbour search (final polishing)
# ----------------------------------------------------------------------
def _anneal_refine(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Metropolis–Hastings walk that moves a small fraction of probability
    mass between coordinates, occasionally performs global jitter, and
    accepts moves according to a decreasing temperature schedule.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)           # initial temperature
    decay = 0.9999
    it = 0
    n = cur.size

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # ---- propose a neighbour ----
        move_type = rng.integers(0, 4)      # 0: transfer, 1: swap, 2: jitter, 3: three‑way split
        new_seq = cur.copy()

        if move_type == 0:                  # transfer a small amount from i to j
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = rng.random() * 0.05 * cur[i]  # up to 5 % of source mass
                new_seq[i] -= delta
                new_seq[j] += delta

        elif move_type == 1:                # swap two entries
            i, j = rng.integers(0, n, size=2)
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]

        elif move_type == 2:                # global jitter
            noise = rng.normal(scale=0.001, size=n)
            new_seq += noise
            new_seq = np.clip(new_seq, 0.0, None)

        else:                               # three‑way split
            i, j, k = rng.integers(0, n, size=3)
            if i != j:
                delta = rng.random() * 0.05 * cur[i]
                split = rng.random()
                new_seq[i] -= delta
                new_seq[j] += delta * split
                new_seq[k] += delta * (1.0 - split)

        # renormalise onto the simplex
        total = new_seq.sum()
        if total <= 0.0:
            continue
        new_seq /= total

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d}  best={new_val:.6f}")

        # every ~2000 steps try a global perturbation (jump)
        if it % 2000 == 0:
            noise = rng.normal(scale=0.002, size=n)
            jump = cur + noise
            jump = np.clip(jump, 0.0, None)
            jump_sum = jump.sum()
            if jump_sum > 0.0:
                jump /= jump_sum
            else:
                jump = _uniform(n)
            jump_val = _objective(jump)
            if jump_val < cur_val or rng.random() < math.exp(-(jump_val - cur_val) / (T + 1e-12)):
                cur = jump
                cur_val = jump_val
                if jump_val < best_val:
                    best_val = jump_val
                    best = jump.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={jump_val:.6f}")

    return best, best_val

# Sub‑gradient exponentiated‑gradient polish (hard max)
# ----------------------------------------------------------------------
def _subgradient_refine(start: np.ndarray,
                        deadline: float,
                        max_iters: int = 150_000,
                        eta_start: float = 0.08,
                        eta_decay: float = 0.99995) -> tuple[np.ndarray, float]:
    """
    Uses the exact sub‑gradient of the hard max‑convolution.
    The update is an exponentiated‑gradient step which automatically
    stays on the simplex.
    """
    a = start.copy()
    best = a.copy()
    best_val = _objective(a)
    it = 0
    eta = eta_start

    while it < max_iters and time.time() < deadline:
        it += 1
        grad = _gradient_wrt_a(a, lam=0.0)              # hard max (sub‑gradient)
        a = a * np.exp(-eta * grad)                    # exponentiated update
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = _uniform(a.size)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[Subgrad] it={it:7d}  best={cur_val:.6f}")

        eta *= eta_decay

    return best, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The algorithm runs several short Adam restarts, a long simulated‑annealing
    polishing phase and a final exponentiated‑gradient sub‑gradient phase,
    always respecting the supplied time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ---------------------------------------------------------------
    # Problem size – a trade‑off between expressiveness and speed
    # ---------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising step function (length {n}) ===")

    # ---------------------------------------------------------------
    # Build a collection of seed probability vectors
    # ---------------------------------------------------------------
    seeds = []

    # 1) re‑use previous result if available and of correct size
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                if prev_arr.sum() > 0.0:
                    seeds.append(prev_arr / prev_arr.sum())
        except Exception:
            pass

    # 2) deterministic seeds
    seeds.extend([
        _uniform(n),
        _random_softmax(n, rng),
        _exponential_decay(n, rate=0.001),
        _triangular(n),
    ])

    # 3) a few extra random softmax seeds for diversity
    for _ in range(2):
        seeds.append(_random_softmax(n, np.random.default_rng(rng.integers(1_000_000))))

    # keep the best legal seed according to the true objective
    best_seq = None
    best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seq = s.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # ---------------------------------------------------------------
    # Phase 1 – multi‑restart Adam optimisation (≈ 45 % of budget)
    # ---------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.45
    if time.time() < phase1_deadline - 1.0:
        print("[Phase 1] Multi‑restart Adam")
        # limit each Adam run to a modest iteration budget
        max_iters_per_run = 150_000
        for idx, seed_vec in enumerate(seeds):
            if time.time() >= phase1_deadline - 1.0:
                break
            print(f"[Phase 1] Adam run {idx+1}/{len(seeds)}")
            seq_adam, val_adam = _adam_optimize(
                seed_vec,
                phase1_deadline,
                max_iters=max_iters_per_run,
                lam_start=5.0,
                lam_factor=1.0008,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = seq_adam.copy()
                print(f"[Phase 1] new best = {val_adam:.6f}")
    else:
        print("[Phase 1] skipped (insufficient time)")

    # ---------------------------------------------------------------
    # Phase 2 – long simulated‑annealing polishing (≈ 40 % of budget)
    # ---------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.85
    if time.time() < phase2_deadline - 1.0:
        print("[Phase 2] Simulated‑annealing refinement")
        seq_sa, val_sa = _anneal_refine(best_seq, phase2_deadline, max_iters=400_000)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()
            print(f"[Phase 2] improved to {val_sa:.6f}")
    else:
        print("[Phase 2] skipped (insufficient time)")

    # ---------------------------------------------------------------
    # Phase 3 – final sub‑gradient exponentiated‑gradient polishing
    # ---------------------------------------------------------------
    if time.time() < deadline - 0.2:
        print("[Phase 3] Sub‑gradient exponentiated‑gradient polish")
        seq_sub, val_sub = _subgradient_refine(best_seq, deadline, max_iters=150_000)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = seq_sub.copy()
            print(f"[Phase 3] improved to {val_sub:.6f}")
    else:
        print("[Phase 3] skipped (insufficient time)")

    # ---------------------------------------------------------------
    # Final normalisation and export
    # ---------------------------------------------------------------
    final_seq = np.maximum(best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
        total = final_seq.sum()
    final_seq /= total

    final_obj = _objective(final_seq)
    print("\n=== Finished ===")
    print(f"Best objective ≈ {final_obj:.6f} (length {n})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
