def _adam_optimize(start, deadline, max_iters=200000, lam_start=5.0, lam_factor=1.001, switch_iter=50000):
    logp = np.log(start + 1e-30)
    n = start.size
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    best_seq = start.copy()
    best_val = _objective(start)
    t = 0
    lam = lam_start
    while t < max_iters and time.time() < deadline:
        t += 1
        # softmax
        logp_shift = logp - np.max(logp)
        a = np.exp(logp_shift)
        a /= a.sum()
        # gradient
        if t < switch_iter:
            g_a = _gradient_wrt_a_soft(a, lam)
        else:
            g_a = _gradient_wrt_a_hard(a)
        # chain rule
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * g_log
        v = beta2 * v + (1 - beta2) * (g_log * g_log)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = 0.15 / (math.sqrt(t) + 1.0)
        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # evaluate objective
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:7d}  best={cur_val:.6f}  λ={lam:.2f}  mode={'soft' if t<switch_iter else 'hard'}")
        # update lam if using soft
        if t < switch_iter:
            lam *= lam_factor
        # occasional random perturbation per 25000 steps if stagnation? Could skip or keep.
        if t % 25000 == 0 and t > switch_iter: # after switch
            # small perturbation to escape plateaus
            noise = np.random.default_rng().normal(scale=0.02, size=n)
            logp += noise
    return best_seq, best_val

"""
Improved optimisation for the step‑function lower bound.

The algorithm consists of three stages:

1. **Soft‑max Adam** – a few thousand iterations with a smoothed
   max‑convolution (soft‑max) to get into a good basin.
2. **Hard‑max Adam** – the same Adam optimiser but now using the exact
   sub‑gradient of the true max‑convolution (no smoothing).  This
   converges faster to a (near‑)optimal extreme point.
3. **Light simulated‑annealing polishing** – a Metropolis walk that
   swaps small amounts of mass between random coordinates and occasionally
   performs a global Gaussian perturbation.

The optimiser works on the simplex via an exponentiated‑gradient (log‑soft‑max)
parameterisation, which guarantees non‑negativity and unit sum.
A short time budget is reserved for each stage so that the whole routine
finishes well inside the 1000 s limit.  The best sequence found is stored
in the global ``height_sequence_1`` variable and returned as a plain list of
floats (values are already normalised, the evaluator is scale‑invariant).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (full mode)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective (the same as evaluate_sequence for sum(a)=1)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) (sum(seq) is assumed equal to 1)."""
    n = seq.size
    conv = _conv_fft(seq, seq)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Seed generators – return a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    exps = np.exp(logits - m)
    return exps / exps.sum()


def _exponential_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        dec = inc[::-1]
    else:
        dec = np.arange(half, 0, -1, dtype=np.float64)
    a = np.concatenate([inc, dec])
    return a / a.sum()


def _sidon_like(n: int, m: int) -> np.ndarray:
    """Very sparse binary pattern (useful as a rough seed)."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a / a.sum()


# ----------------------------------------------------------------------
# Gradient of (soft) max‑convolution
# ----------------------------------------------------------------------
def _soft_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries (lam > 0)."""
    max_c = conv.max()
    if lam <= 0.0:
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.where(active, 1.0 / np.count_nonzero(active), 0.0)
        return w
    # avoid overflow – the exponent is always ≤ 0
    rel = (conv - max_c) / max_c   # ≤ 0
    exps = np.exp(lam * rel)
    w = exps / exps.sum()
    return w


def _gradient_wrt_a_soft(a: np.ndarray, lam: float) -> np.ndarray:
    """Gradient of the smoothed max‑convolution."""
    conv = _conv_fft(a, a)
    w = _soft_max_weights(conv, lam)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                     # length = n + (2n-1) -1 = 3n-2
    n = a.size
    grad = 2.0 * full[n - 1:n - 1 + n]            # slice that aligns with a
    return grad


def _gradient_wrt_a_hard(a: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_i (a*a)_i.
    It is 2·a reflected about the index that attains the maximum.
    """
    n = a.size
    conv = _conv_fft(a, a)
    k = int(conv.argmax())          # index where the max occurs

    g = np.zeros_like(a)
    if k < n:
        # i = 0 … k  →  g[i] = 2·a[k‑i]
        g[:k + 1] = 2.0 * a[k::-1]
    else:
        # k ≥ n → only the tail overlaps
        start = k - (n - 1)        # first i that sees a non‑zero partner
        length = n - start
        g[start:] = 2.0 * a[::-1][:length]
    return g


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 400_000,
                   lam_start: float = 5.0,
                   lam_factor: float = 1.01,
                   soft_iters: int = 50_000) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑probabilities.
    * First `soft_iters` steps use the smooth max‑convolution.
    * Remaining steps use the exact hard‑max sub‑gradient.
    """
    eps = 1e-30
    logp = np.log(start + eps)          # shape (n,)
    n = start.size

    m = np.zeros_like(logp)             # 1st moment
    v = np.zeros_like(logp)             # 2nd moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = start.copy()
    best_val = _objective(start)

    t = 0
    lam = lam_start

    while t < max_iters and time.time() < deadline:
        t += 1

        # ------- probability vector from log‑probs -------
        logp_shift = logp - np.max(logp)
        a = np.exp(logp_shift)
        a /= a.sum()

        # ------- gradient of the objective -------
        if t <= soft_iters:
            g_a = _gradient_wrt_a_soft(a, lam)
            lam = min(lam * lam_factor, 1e6)   # prevent overflow
        else:
            g_a = _gradient_wrt_a_hard(a)

        # chain rule for the log‑softmax parametrisation
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log * g_log)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.15 / (math.sqrt(t) + 1.0)

        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- keep the best -------
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            mode = "soft" if t <= soft_iters else "hard"
            print(f"[Adam] it={t:7d}  best={cur_val:.6f}  mode={mode}")

        # ------- occasional tiny random kick -------
        if t % 25000 == 0 and t > soft_iters:
            noise = np.random.default_rng().normal(scale=0.02, size=n)
            logp += noise

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very light simulated‑annealing polishing
# ----------------------------------------------------------------------
def _anneal_refine(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Metropolis–Hastings walk that swaps a small fraction of mass between
    random coordinates and occasionally makes a global Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0
    n = cur.size

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a small fraction from i to j
        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d}  best={new_val:.6f}")

        # periodic global perturbation
        if it % 2000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0.0:
                new_seq /= total
            else:
                new_seq = _uniform(n)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects a time budget (default
    1000 s) and returns the best vector found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    rng = np.random.default_rng(seed + 12345)

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a richer search space but costs more
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising step function (length {n}) ===")

    # ------------------------------------------------------------------
    # Initialise a collection of seeds (probability vectors)
    # ------------------------------------------------------------------
    seeds = []

    # if a previous solution of the same length exists, reuse it
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,) and arr.sum() > 0:
                arr = np.maximum(arr, 0.0)
                arr = arr / arr.sum()
                seeds.append(arr)
        except Exception:
            pass

    # deterministic seeds
    seeds.extend([
        _uniform(n),
        _random_softmax(n, rng),
        _exponential_decay(n, rate=0.001),
        _triangular(n),
        _sidon_like(n, m=800),      # a sparse binary pattern
    ])

    # pick the best among those seeds (according to the exact objective)
    best_seed = None
    best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seed = s
    print(f"[Init] best initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimiser (soft → hard max)
    # ------------------------------------------------------------------
    phase1_deadline = deadline - 30.0   # reserve ~30 s for polishing
    if phase1_deadline > time.time():
        print("[Phase 1] Adam optimisation start")
        seq_adam, val_adam = _adam_optimize(
            best_seed,
            phase1_deadline,
            max_iters=400_000,
            lam_start=5.0,
            lam_factor=1.01,
            soft_iters=50_000,
        )
        print(f"[Phase 1] best after Adam = {val_adam:.6f}")
    else:
        seq_adam, val_adam = best_seed, best_val
        print("[Phase 1] skipped – no time left")

    # ------------------------------------------------------------------
    # Phase 2 – Simulated‑annealing polishing (uses remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("[Phase 2] Simulated‑annealing polishing")
        seq_final, val_final = _anneal_refine(seq_adam, deadline, max_iters=600_000)
        if val_final < val_adam:
            seq_adam, val_adam = seq_final, val_final
            print(f"[Phase 2] improved to {val_final:.6f}")
        else:
            print("[Phase 2] no improvement")
    else:
        print("[Phase 2] no time left for annealing")

    # ------------------------------------------------------------------
    # Normalise (the evaluator is scale‑invariant, but we keep sum=1)
    # ------------------------------------------------------------------
    final_seq = np.maximum(seq_adam, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
        total = final_seq.sum()
    final_seq /= total

    # ------------------------------------------------------------------
    # Final verification using the public ``evaluate_sequence`` function
    # ------------------------------------------------------------------
    try:
        # evaluate_sequence is assumed to be available in the environment
        final_obj = evaluate_sequence(final_seq.tolist())
    except Exception:
        final_obj = _objective(final_seq)   # fallback (equivalent for sum=1)

    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")

    # store for possible reuse in a later run
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
