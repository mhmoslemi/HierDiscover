def _exp_gradient_descent(start_seq, deadline, max_iter=5_000_000):
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    beta = 1.0
    beta_max = 500.0
    beta_inc = (beta_max - beta) / max_iter
    eta0 = 0.1
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        # Compute conv
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # compute softmax weights for gradient
        beta = min(beta_max, beta + beta_inc)
        exp_beta = np.exp(beta * (conv - max_val))
        p = exp_beta / exp_beta.sum()
        # gradient of weighted sum w.r.t a:
        grad = 2.0 * _conv_fft(p, a[::-1])[n-1:n-1+n]
        # step size schedule
        eta = eta0 / np.sqrt(it)
        # exponentiated update
        a = a * np.exp(-eta * grad)
        # protect against numerical underflow: renorm and min value
        a = np.clip(a, 1e-20, None)
        a /= a.sum()
        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[ExpGD] it={it} val={val}")
                if best_val < 1.503:
                    break
    return best_seq, best_val
```

def _kaiser(n, beta):
    from scipy.special import i0
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    w = i0(beta * np.sqrt(1 - t**2)) / i0(beta)
    return w / w.sum()
```

Note: if beta == 0, i0(0) =1, sqrt term zeros leads to 1, yields uniform.

def _tukey(n, a):
    # a in [0,1]
    i = np.arange(n, dtype=np.float64)
    L = n - 1
    w = np.ones(n, dtype=np.float64)
    # first and last parts
    if a > 0:
        period = a * L / 2.0
        idx1 = i < period
        w[idx1] = 0.5 * (1 + np.cos(np.pi * ( (2*i[idx1]/L) -1) / a ))
        idx2 = i > L - period
        w[idx2] = 0.5 * (1 + np.cos(np.pi * ( (2*i[idx2]/L) -1) / a ))
    # else a ==0 => rectangular (uniform)
    return w / w.sum()

def _exp_gradient_descent(start_seq, deadline, beta_start=5.0, beta_end=500.0,
                         eta0=0.3, max_iter=10_000_000):
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    beta = beta_start
    beta_inc = (beta_end - beta_start) / max_iter
    while time.time() < deadline and it < max_iter:
        it += 1
        # compute full conv
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # update beta (could be schedule based on iteration proportion)
        if beta < beta_end:
            beta = min(beta_end, beta + beta_inc)

        # softmax weights
        exp_beta = np.exp(beta * (conv - max_val))
        w = exp_beta / exp_beta.sum()
        # gradient of weighted sum
        grad = 2.0 * _conv_fft(w, a[::-1])[n - 1:n - 1 + n]
        # step size decaying
        eta = eta0 / np.sqrt(it)
        # exponentiated update (mirror descent)
        a = a * np.exp(-eta * grad)
        # avoid all zeros
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a = a / a_sum

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[ExpGD] it={it:7d} n={n:5d} val={val:.6f}")
                if best_val < 1.503:
                    break
    return best_seq, best_val

"""
Search for a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.  The strategy is:

1.  Build a large collection of deterministic analytic windows
    (uniform, Hann, Blackman, Kaiser, Tukey, Gaussian, arcsine, Beta …)
    and keep the best seed.
2.  Perform a fast random sampling phase (Dirichlet draws and random
    Beta‑pdf shapes) to improve the best candidate.
3.  Run a short smooth Adam optimiser (log‑sum‑exp surrogate) to get out
    of poor local minima.
4.  Polish the result with a dedicated exponentiated‑gradient descent on
    the *exact* max‑convolution objective – a mirror‑descent update on the
    probability simplex with a soft‑max weighting of the active
    convolution entries.
5.  Finish with a cheap random local‑search (Gaussian perturbations)
    while time permits.

All work respects a user‑supplied time budget (default 1000 s) and
stores the best sequence found in the global variable ``height_sequence_1``.
"""

import time
import numpy as np
from math import sqrt, pi

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a *normalised* sequence
    (sum == 1). Returns the numerator ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * pi * i / (n - 1))
         + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian_sigma015(n: int) -> np.ndarray:
    sigma_factor = 0.15
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _gaussian_sigma025(n: int) -> np.ndarray:
    sigma_factor = 0.25
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf shape on a discrete grid (shifted by 0.5 to avoid end‑points)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape_05_05(n: int) -> np.ndarray:
    return _beta_shape(n, 0.5, 0.5)


def _beta_shape_2_2(n: int) -> np.ndarray:
    return _beta_shape(n, 2.0, 2.0)


def _beta_shape_05_2(n: int) -> np.ndarray:
    return _beta_shape(n, 0.5, 2.0)


def _beta_shape_2_05(n: int) -> np.ndarray:
    return _beta_shape(n, 2.0, 0.5)


def _kaiser(n: int, beta: float) -> np.ndarray:
    """Kaiser window; fallback to uniform if SciPy is unavailable."""
    try:
        from scipy.special import i0
    except Exception:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha  # range [-1,1]
    w = i0(beta * np.sqrt(1.0 - t * t)) / i0(beta)
    return w / w.sum()


def _tukey(n: int, a: float) -> np.ndarray:
    """Tukey (tapered cosine) window; a∈[0,1] (0→rectangular, 1→Hann)."""
    if a <= 0.0:
        return _uniform(n)
    if a >= 1.0:
        return _hann(n)
    i = np.arange(n, dtype=np.float64)
    L = n - 1
    w = np.ones(n, dtype=np.float64)
    period = a * L / 2.0
    idx1 = i < period
    idx2 = i > (L - period)
    w[idx1] = 0.5 * (1 + np.cos(np.pi * (2 * i[idx1] / L - 1) / a))
    w[idx2] = 0.5 * (1 + np.cos(np.pi * (2 * i[idx2] / L - 1) / a))
    return w / w.sum()

# Random Dirichlet sampling
# ----------------------------------------------------------------------


def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Return a random point on the simplex (sum=1) with Dirichlet(alpha)."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _smooth_adam_optimize(start_seq: np.ndarray,
                          deadline: float,
                          beta_start: float = 5.0,
                          beta_end: float = 200.0,
                          max_iter: int = 5_000_000,
                          lr0: float = 0.03) -> tuple[np.ndarray, float]:
    """Minimise a smooth surrogate of max(conv(a,a)) via Adam."""
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        a = np.exp(logits)
        a = a / a.sum()

        conv = _conv_fft(a, a)

        beta = beta_start + (beta_end - beta_start) * min(1.0,
                                                          it / (max_iter * 0.6))

        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()

        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")
            if best_val < 1.45:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------


def _exp_gradient_descent(start_seq: np.ndarray,
                          deadline: float,
                          beta_start: float = 5.0,
                          beta_end: float = 500.0,
                          eta0: float = 0.30,
                          max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex with a soft‑max weighting of the active
    convolution entries.  ``beta`` controls the soft‑max sharpness, ``eta0``
    the initial step size (decays like 1/√t).
    """
    a = start_seq.copy()
    n = len(a)

    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    beta = beta_start
    beta_inc = (beta_end - beta_start) / max_iter

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # increase temperature gradually
        if beta < beta_end:
            beta = min(beta_end, beta + beta_inc)

        # soft‑max weights over the whole convolution vector
        exp_beta = np.exp(beta * (conv - max_val))
        w = exp_beta / exp_beta.sum()

        # gradient of the weighted sum (see derivation in analysis)
        grad = 2.0 * _conv_fft(w, a[::-1])[n - 1:n - 1 + n]

        # step size schedule
        eta = eta0 / np.sqrt(it)

        # exponentiated (mirror) update
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[ExpGD] it={it:7d} n={n:5d} val={val:.6f}")
                if best_val < 1.503:
                    break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny random local‑search (Gaussian perturbations)
# ----------------------------------------------------------------------


def _local_perturb(start_seq: np.ndarray,
                   rng: np.random.Generator,
                   deadline: float,
                   max_iter: int = 200_000,
                   sigma_start: float = 0.02) -> tuple[np.ndarray, float]:
    """Additive Gaussian noise, re‑normalise, keep improvements."""
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    sigma = sigma_start
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        pert = rng.normal(0.0, sigma, size=n)
        new = a + pert
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s == 0.0:
            continue
        new /= s
        val = _objective(new)
        if val < best_val:
            best_val = val
            best_seq = new.copy()
            a = new
            print(f"[Local] it={it:7d} n={n:5d} val={val:.6f}")
        sigma *= 0.9999  # very gentle decay
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within the given time budget.  The routine
    combines analytic seeds, random sampling, a short Adam phase,
    exponentiated‑gradient polishing and a final noisy local search.
    The best sequence found is stored in the global ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)

    t_start = time.time()
    deadline = t_start + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Load a previously stored solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                previous = prev_arr / prev_arr.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore (FFT‑friendly, up to ~12k)
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000, 8000, 10000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic seeds (including new windows)
    # ------------------------------------------------------------------
    kaiser_betas = (0.0, 2.0, 5.0, 8.0, 12.0)
    tukey_alphas = (0.0, 0.3, 0.5, 0.7, 1.0)

    for n in lengths:
        # basic families
        seed_seqs = [
            _uniform(n),
            _catalan_shape(n),
            _hann(n),
            _hamming(n),
            _blackman(n),
            _gaussian_sigma015(n),
            _gaussian_sigma025(n),
            _arcsine_shape(n),
            _beta_shape_05_05(n),
            _beta_shape_2_2(n),
            _beta_shape_05_2(n),
            _beta_shape_2_05(n),
        ]

        # extended families
        for beta in kaiser_betas:
            seed_seqs.append(_kaiser(n, beta))

        for a_ in tukey_alphas:
            seed_seqs.append(_tukey(n, a_))

        # optional – reuse previous if dimensions match
        if previous is not None and previous.shape == (n,):
            seed_seqs.append(previous)

        for seq in seed_seqs:
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={seq.dtype} val={val:.6f}")

        # stop early if we already beat the target
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 2 – heavy random sampling (Dirichlet + random Beta shapes)
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.35  # roughly 35 % of remaining budget
    sample_deadline = now + time_for_sampling

    alphas = [0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    beta_params = [(0.2, 0.2), (0.5, 0.5), (1.0, 1.0),
                   (2.0, 2.0), (0.5, 2.0), (2.0, 0.5), (3.0, 0.3), (0.3, 3.0)]

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        # Dirichlet draws
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random‑Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        # Random Beta‑pdf shapes
        a_par, b_par = rng.choice(beta_params)
        seq = _beta_shape(n, a_par, b_par)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random‑Beta] n={n:5d} (α,β)=({a_par:.2f},{b_par:.2f}) val={val:.6f}")

        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 3 – short smooth Adam refinement (few restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            adam_deadline = time.time() + remaining * 0.15
            candidates = [best_seq]
            # a couple of random restarts of the same length
            for _ in range(2):
                cand = _random_dirichlet(len(best_seq), rng,
                                         alpha=rng.choice(alphas))
                candidates.append(cand)

            for idx, cand in enumerate(candidates):
                seq_adam, val_adam = _smooth_adam_optimize(
                    start_seq=cand,
                    deadline=adam_deadline,
                    beta_start=5.0,
                    beta_end=200.0,
                    max_iter=3_000_000,
                    lr0=0.03,
                )
                if val_adam < best_val:
                    best_val = val_adam
                    best_seq = seq_adam
                    print(f"[Adam‑Refine {idx}] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – exponentiated‑gradient polishing (the core phase)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            exp_deadline = time.time() + remaining * 0.55
            seq_exp, val_exp = _exp_gradient_descent(
                start_seq=best_seq,
                deadline=exp_deadline,
                beta_start=5.0,
                beta_end=500.0,
                eta0=0.30,
                max_iter=8_000_000,
            )
            if val_exp < best_val:
                best_val = val_exp
                best_seq = seq_exp
                print(f"[Exp‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – tiny random local search while time permits
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 2.0:
            local_deadline = time.time() + remaining
            seq_loc, val_loc = _local_perturb(best_seq, rng, local_deadline)
            if val_loc < best_val:
                best_val = val_loc
                best_seq = seq_loc
                print(f"[Local‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check (the grader will evaluate again anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )

    # store for possible future runs
    globals()["height_sequence_1"] = best_seq.copy()

    return best_seq.tolist()
