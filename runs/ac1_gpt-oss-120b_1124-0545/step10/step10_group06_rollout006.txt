"""
Search for a non‑negative step function that minimises
``evaluate_sequence`` (i.e. 2·n·max(conv(a,a))/(sum(a)**2)).

The routine consists of three stages and respects the 1000 s budget:

1. **Initial seeds** – a few deterministic and random probability vectors.
   The best one (according to the exact objective) is kept.

2. **Iterative LP expansion** – given the current distribution f (normalised
   to sum = 1) we compute M = max(f∗f).  A linear programme finds a vector g
   of maximal ℓ₁‑norm under the linear constraints f∗g ≤ M (convolution is
   linear in g).  After normalising g to sum = 1 we replace f by a convex
   combination (1‑t) f + t g with a step size that shrinks with the iteration
   count.  Each accepted update strictly reduces the objective, and the loop
   runs while time permits (≈ 30 LP solves).

3. **Fine‑tuning** – a short Adam optimiser on the simplex (soft‑max
   parametrisation) with a smoothed max‑convolution objective finishes the
   search.

The best sequence is stored globally as ``height_sequence_2`` and returned
as a plain list of floats.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Basic objective (identical to evaluate_sequence but without the guards)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2).  Assumes seq is non‑negative."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = np.convolve(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# FFT utilities (used by the Adam stage)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _soft_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weighting of the entries of `conv` with inverse temperature `lam`."""
    max_c = conv.max()
    if lam <= 0.0:
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.where(active, 1.0 / np.count_nonzero(active), 0.0)
        return w
    rel = (conv - max_c) / max_c           # ≤ 0
    exps = np.exp(lam * rel)
    w = exps / exps.sum()
    return w


def _gradient_wrt_a(a: np.ndarray, lam: float) -> np.ndarray:
    """Gradient of the smoothed max‑convolution w.r.t. a (probability vector)."""
    n = a.size
    conv = _conv_fft(a, a)
    w = _soft_max_weights(conv, lam)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)            # length = n + (2n‑1) – 1 = 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimisation on the simplex (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 300_000,
                   lam_start: float = 5.0,
                   lam_factor: float = 1.001) -> tuple[np.ndarray, float]:
    """Run Adam on log‑probabilities, increasing the smoothing λ gradually."""
    eps = 1e-30
    logp = np.log(start + eps)                     # shape (n,)
    n = start.size
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = start.copy()
    best_val = _objective(start)

    t = 0
    lam = lam_start

    while t < max_iters and time.time() < deadline:
        t += 1
        # current probabilities via soft‑max
        logp_shift = logp - np.max(logp)
        a = np.exp(logp_shift)
        a /= a.sum()

        # gradient of the smoothed max‑convolution
        g_a = _gradient_wrt_a(a, lam)

        # chain rule to obtain gradient w.r.t. logits
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log * g_log)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.15 / (math.sqrt(t) + 1.0)
        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:7d}  best={cur_val:.6f}  λ={lam:.2f}")

        lam *= lam_factor

    return best_seq, best_val


# ----------------------------------------------------------------------
# Sparse convolution matrix for the LP (Toeplitz with constant diagonals)
# ----------------------------------------------------------------------
def _conv_matrix_sparse(a: np.ndarray) -> scipy.sparse.csr_matrix:
    """Return a sparse matrix A with shape (2n‑1, n) such that A @ x = a ∗ x."""
    n = a.shape[0]
    L = 2 * n - 1
    # each diagonal d (offset d) is a constant a[d]
    data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=int)
    A = scipy.sparse.diags(data, offsets, shape=(L, n), format='csr')
    return A


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step function that minimises `evaluate_sequence`.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # small safety margin
    rng = np.random.default_rng(seed + 12345)

    # ---------------------------------------------------------------
    # Problem size – a compromise between expressiveness and speed
    # ---------------------------------------------------------------
    n = 2048                     # power of two (helps FFT later)
    print(f"\n=== Optimising step function (length {n}) ===")

    # ---------------------------------------------------------------
    # Initialise a collection of seeds and pick the best one
    # ---------------------------------------------------------------
    seeds = []

    # try to reuse a previously stored sequence if it matches the size
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n:
                arr = np.maximum(arr, 0.0)
                if arr.sum() > 0:
                    seeds.append(arr / arr.sum())
        except Exception:
            pass

    # deterministic seeds
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))                     # uniform
    logits = rng.standard_normal(n)
    sm = np.exp(logits - np.max(logits))
    seeds.append(sm / sm.sum())                                            # random soft‑max
    rate = 0.001
    exp_decay = np.exp(-rate * np.arange(n))
    seeds.append(exp_decay / exp_decay.sum())                              # exponential
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    dec = inc[::-1] if n % 2 == 0 else np.arange(half, 0, -1, dtype=np.float64)
    tri = np.concatenate([inc, dec])
    seeds.append(tri / tri.sum())                                          # triangular

    # pick the best seed according to the exact objective
    best_seed = None
    best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seed = s.copy()
    print(f"[Init] best initial objective = {best_val:.6f}")

    # ---------------------------------------------------------------
    # Phase 1 – LP based expansion / improvement
    # ---------------------------------------------------------------
    f = best_seed
    best_seq = f.copy()
    best_val = _objective(f)

    max_lp_iters = 30
    lp_iter = 0
    while time.time() < deadline - 5.0 and lp_iter < max_lp_iters:
        # current max‑convolution value
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # Build the sparse Toeplitz matrix that encodes f∗x
        A = _conv_matrix_sparse(f)

        # Right‑hand side – allow a microscopic tolerance to keep feasibility
        b_ub = np.full(2 * n - 1, M * (1.0 + 1e-12), dtype=np.float64)

        # Linear programme: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = scipy.optimize.linprog(c,
                                     A_ub=A,
                                     b_ub=b_ub,
                                     bounds=bounds,
                                     method='highs',
                                     options={'presolve': True})

        # free the matrix as soon as possible
        del A

        if not res.success:
            print("[LP] linprog failed – terminating LP phase")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 1e-12:
            print("[LP] LP returned a zero vector – terminating")
            break

        # Normalise to unit sum (same normalisation used by the objective)
        g = g0 / sum_g0

        # Convex combination – step size shrinks with the iteration count
        step = 0.25 / math.sqrt(lp_iter + 1.0)   # between ~0.25 and ~0.01
        candidate = (1.0 - step) * f + step * g
        candidate = np.maximum(candidate, 0.0)
        candidate /= candidate.sum()
        cand_val = _objective(candidate)

        if cand_val < best_val:
            best_val = cand_val
            best_seq = candidate.copy()
            f = candidate                     # accept the move
            print(f"[LP] iter={lp_iter:2d}  best={cand_val:.6f}  step={step:.4f}")
        else:
            # no improvement – keep the old f
            pass

        lp_iter += 1

    # ---------------------------------------------------------------
    # Phase 2 – short Adam fine‑tuning (smooth max → true max)
    # ---------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("[Adam] final fine‑tuning")
        seq_adam, val_adam = _adam_optimize(best_seq,
                                            deadline,
                                            max_iters=200_000,
                                            lam_start=5.0,
                                            lam_factor=1.001)
        if val_adam < best_val:
            best_seq = seq_adam
            best_val = val_adam
            print(f"[Adam] improved to {val_adam:.6f}")

    # ---------------------------------------------------------------
    # Final normalisation and export
    # ---------------------------------------------------------------
    final_seq = np.maximum(best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = np.full(n, 1.0 / n, dtype=np.float64)
        total = final_seq.sum()
    final_seq /= total

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")
    globals()["height_sequence_2"] = final_seq.copy()
    return final_seq.tolist()
