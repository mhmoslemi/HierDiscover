from scipy.linalg import toeplitz
from scipy.optimize import linprog

def _lp_step(f):
    n = len(f)
    M = np.max(_conv_fft(f, f))
    first_col = np.concatenate([f, np.zeros(n-1)])
    first_row = np.concatenate([[f[0]], np.zeros(n-1)])
    A = toeplitz(first_col, first_row)
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError(res.message)
    b = res.x
    b = np.maximum(b, 0) # ensure nonneg
    b_sum = b.sum()
    if b_sum == 0:
        return f
    b = b / b_sum
    return b

"""
search for a step‑function (non‑negative list) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The optimisation is performed in four stages:

1. **cheap analytic seeds** – a handful of classic shapes (uniform,
   Hann, Beta‑pdf, etc.) are evaluated.
2. **LP refinement** – for a moderate length (2048) we iteratively
   solve a linear programme

        maximise   Σ b_j
        subject to 0 ≤ b,
                    conv(f, b) ≤ M   (M = max(conv(f,f)))

   where the convolution constraints are linear in `b`.  The solution
   `b` is interpolated into `f` (convex update) and normalised.
   This step is cheap (a few seconds per iteration) but often moves
   the distribution far towards a flat self‑convolution.

3. **size up‑scaling** – the best distribution from the LP phase is
   interpolated to the target length (default 4000) and normalised.

4. **final polishing** – smooth Adam optimisation on a log‑sum‑exp
   surrogate of the max‑convolution, followed by a projected
   sub‑gradient descent on the exact objective.

All helpers are defined at module level; no closures or lambdas are used.
The routine respects a total time budget (default 1000 s) and returns the
best sequence found.
"""

import time
import numpy as np
from math import pi, sqrt

# ----------------------------------------------------------------------
# FFT‑based convolution (fast for all lengths)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective (equivalent to evaluate_sequence for a normalised sequence)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  – assumes Σ seq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all returned normalised to Σ = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta‑pdf (shifted by 0.5 to avoid the exact ends)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape_05_05(n: int) -> np.ndarray:
    return _beta_shape(n, 0.5, 0.5)


def _beta_shape_2_2(n: int) -> np.ndarray:
    return _beta_shape(n, 2.0, 2.0)


def _beta_shape_05_2(n: int) -> np.ndarray:
    return _beta_shape(n, 0.5, 2.0)


def _beta_shape_2_05(n: int) -> np.ndarray:
    return _beta_shape(n, 2.0, 0.5)


# ----------------------------------------------------------------------
# Random Dirichlet sampling (used only for a quick warm‑start)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Adam optimisation on the smooth log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(start_seq: np.ndarray,
                          deadline: float,
                          beta_start: float = 5.0,
                          beta_end: float = 400.0,
                          max_iter: int = 8_000_000,
                          lr0: float = 0.03) -> tuple[np.ndarray, float]:
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        a = np.exp(logits)
        a /= a.sum()

        conv = _conv_fft(a, a)
        # gradually tighten the smooth‑max approximation
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()                     # softmax weights
        # gradient of the surrogate w.r.t. a
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if (it & 1023) == 0:                # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")
            if best_val < 1.0:               # theoretical lower bound
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _proj_subgradient_optimize(start_seq: np.ndarray,
                               deadline: float,
                               eps_rel_start: float = 1e-4,
                               eps_rel_end: float = 1e-7,
                               lr0: float = 0.12,
                               max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    n = len(a)

    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        # tolerance that shrinks linearly in the schedule
        eps_rel = eps_rel_start + (eps_rel_end - eps_rel_start) * min(1.0, it / (max_iter * 0.5))
        thresh = max_val - eps_rel * max_val
        active_idx = np.where(conv >= thresh)[0]

        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv)], dtype=int)

        w = np.zeros_like(conv)
        w[active_idx] = 1.0 / active_idx.size

        conv_ag = _conv_fft(w, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        lr = lr0 / sqrt(it)
        a_next = a - lr * g
        a_next = _project_simplex(a_next)

        if (it & 1023) == 0:
            val = _objective(a_next)
            if val < best_val:
                best_val = val
                best_seq = a_next.copy()
                print(f"[ProjSGD] it={it:7d} n={n:5d} val={val:.6f}")
            if best_val < 1.0:
                break

        a = a_next

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (non‑convex but cheap for moderate n)
# ----------------------------------------------------------------------
from scipy.linalg import toeplitz
from scipy.optimize import linprog


def _lp_refine_step(f: np.ndarray, step: float) -> np.ndarray:
    """
    One iteration of the LP refinement:
        maximise Σ b_j  subject to  b_j ≥ 0,  conv(f,b) ≤ M .
    Returns the updated distribution (convex combination of f and the
    normalised LP solution).
    """
    n = len(f)
    M = np.max(_conv_fft(f, f))

    # Build the dense Toeplitz matrix that encodes conv(f,b) ≤ M
    first_col = np.concatenate([f, np.zeros(n - 1)])
    first_row = np.concatenate([[f[0]], np.zeros(n - 1)])
    A = toeplitz(first_col, first_row)          # shape (2n‑1, n)

    # Linear programme: maximise 1ᵀb  ⇔ minimise -1ᵀb
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(2 * n - 1, M)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    del A  # free memory promptly
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    b = res.x
    b_sum = b.sum()
    if b_sum == 0:
        return f
    b = b / b_sum                      # normalise to Σ b = 1

    # convex update and re‑normalise
    new_f = (1.0 - step) * f + step * b
    new_f = np.maximum(new_f, 0.0)
    s = new_f.sum()
    return new_f / s if s > 0 else new_f

        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
