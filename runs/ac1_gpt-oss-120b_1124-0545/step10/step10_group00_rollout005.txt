def _mirror_subgradient_optimize(start_seq, n, deadline):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    lr0 = 0.2
    for it in range(1, 10**9):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # compute subgradient at max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end+1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)
        # renormalise
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = _uniform(n)
        else:
            a = a / s
    return best_seq, best_val

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(seq,seq)) / (sum(seq)²)

with a multi‑stage search:

1. Generate a diverse seed pool (analytic families, Dirichlet draws,
   low‑frequency curves, and a few handcrafted permutations).
2. Keep the best few seeds (according to a fast surrogate 2·n·max(conv)
   with the ℓ₁‑normalisation Σa = 1).
3. Run a long mirrored‑sub‑gradient descent (exponentiated‑gradient)
   on each seed – this directly minimises the true max‑convolution.
4. Polish the best result with a simulated‑annealing that only swaps
   entries (pure ordering optimisation).  The move set is cheap and
   often finds a lower max‑convolution for the same multiset of values.
5. Finally evaluate the official ``evaluate_sequence`` and store the
   result in the global name ``height_sequence_1`` for possible reuse.

All stages respect the global time budget (default 1000 s).  The
implementation is deliberately lightweight and uses only NumPy and
the provided ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective, identical to the numerator of the official
    evaluator when Σseq = 1:  2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Ordering helpers
# ----------------------------------------------------------------------
def _sorted_desc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)[::-1]
    return a / a.sum()


def _sorted_asc(vec: np.ndarray) -> np.ndarray:
    a = np.sort(vec)
    return a / a.sum()


def _alternate_order(vec: np.ndarray) -> np.ndarray:
    """
    Largest, smallest, 2nd‑largest, 2nd‑smallest, …
    This tends to keep the reversed pair of a large value small.
    """
    a = np.sort(vec)          # ascending
    out = np.empty_like(a)
    lo, hi = 0, len(a) - 1
    pos = 0
    while lo <= hi:
        if pos % 2 == 0:
            out[pos] = a[hi]
            hi -= 1
        else:
            out[pos] = a[lo]
            lo += 1
        pos += 1
    return out / out.sum()


def _interleaved_order(vec: np.ndarray, step: int) -> np.ndarray:
    """
    Place values in a permutation with a given step (must be coprime to length).
    """
    n = len(vec)
    perm = (np.arange(n) * step) % n
    out = np.empty_like(vec)
    out[perm] = vec
    return out / out.sum()


# ----------------------------------------------------------------------
# Random families & low‑frequency curves
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve by adding a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection (expanded)
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_gaussian(n, sigma_frac=0.40))

    # power‑law family (finer grid)
    for p in np.linspace(0.0, 4.0, 21):
        seeds.append(_power(n, p))

    # beta family (moderate grid)
    for a in np.linspace(0.3, 2.0, 5):
        for b in np.linspace(0.3, 4.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw order
            seeds.append(_sorted_desc(raw))
            seeds.append(_sorted_asc(raw))
            seeds.append(_alternate_order(raw))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform and low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Additional handcrafted permutations (interleaved with a coprime step)
    step = 1579  # a prime roughly sqrt(n), coprime to 4000
    for _ in range(3):
        vec = _random_dirichlet(n, 0.5, rng)
        seeds.append(_interleaved_order(vec, step))
        seeds.append(_alternate_order(vec))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _mirror_subgradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    lr0: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) sub‑gradient descent on the exact
    max‑convolution objective.  Returns the best sequence seen and its
    surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active term
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative exponentiated‑gradient update
        lr = lr0 / np.sqrt(it)
        a = a * np.exp(-lr * g)

        # re‑normalise onto the simplex (positivity already holds)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _simulated_annealing_swaps(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.05,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Very simple SA that only swaps two entries (pure ordering optimisation).
    The mass values stay unchanged – only the permutation varies.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    while time.time() < deadline:
        # geometric cooling
        T = max(T * 0.9997, min_temp)

        i, j = rng.choice(n, size=2, replace=False)
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats and stores it
    in the global name ``height_sequence_1``.
    """
    # ------------------------------------------------------------------
    # Basic setup
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for final evaluation & printing
    deadline = start_time + budget_s - 1.5

    n = 4000  # length that gives a good trade‑off between expressivity and speed

    # ------------------------------------------------------------------
    # Gather seeds (including any previously discovered solution)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    seeds = _collect_seeds(n, rng, previous)

    # ------------------------------------------------------------------
    # Quick surrogate evaluation and selection of the most promising seeds
    # ------------------------------------------------------------------
    evals = []
    for vec in seeds:
        val = _objective(vec)
        evals.append((vec, val))
    evals.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evals[:top_k]]

    best_seq = top_seeds[0].copy()
    best_val = _objective(best_seq)
    print(f"[Init] Best surrogate after seeding = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Mirror‑descent sub‑gradient optimisation on the top seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        # allocate a fair share of the remaining time to each seed
        remaining = deadline - time.time()
        per_seed = remaining / (len(top_seeds) - idx)
        sub_deadline = time.time() + per_seed * 0.95  # keep a tiny reserve

        seq, val = _mirror_subgradient_optimize(seed_vec, n, sub_deadline)
        if val < best_val:
            best_seq = seq.copy()
            best_val = val
            print(f"[Mirror] New best after seed {idx+1}/{top_k}: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Pure ordering optimisation via simulated annealing (swaps only)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[SA] Starting swap‑only simulated annealing")
        seq, val = _simulated_annealing_swaps(best_seq, n, deadline)
        if val < best_val:
            best_seq = seq.copy()
            best_val = val
            print(f"[SA] New best after SA: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[Eval] Official evaluate_sequence value = {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    # Store for possible reuse in a future call
    globals()["height_sequence_1"] = best_seq.tolist()
    print(
        f"=== Finished ===\nBest surrogate ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )
    return best_seq.tolist()
