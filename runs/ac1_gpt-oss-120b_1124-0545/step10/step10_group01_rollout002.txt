A hybrid optimiser for the step‑function attack:

* Enumerate a rich pool of analytic / random seeds.
* Refine the best seed with a cheap projected sub‑gradient descent.
* Repeatedly solve a linear programme that maximises the total mass

  mixed with the current iterate – this provably never raises the
  max‑convolution and often reduces it.
* Finish with a modest simulated‑annealing polishing stage.

The routine respects a global time budget (default 1000 s) and returns
the best sequence found (list of non‑negative floats).  The code works

"""

def _next_pow2(x: int) -> int:
    """smallest power of two ≥ x"""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` (ignores security checks).
    Returns 2·n·max(conv) / (sum²).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = float(conv.max())
    s = float(seq.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _objective_surrogate(seq: np.ndarray) -> float:
    """Same as _objective but assumes Σseq = 1 (used for quick ranking)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                        (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; result ℓ₁‑normed."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection – a diverse pool of ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        # simple Kaiser implementation (no SciPy)
        i = np.arange(n, dtype=np.float64)
        alpha = (n - 1) / 2.0
        y = (i - alpha) / alpha
        t = beta * np.sqrt(1 - y * y)
        a = np.exp(beta) * np.exp(-t)  # crude placeholder – still gives a shape
        a = np.maximum(a, 0.0)
        s = a.sum()
        seeds.append(a / s if s > 0 else _uniform(n))

    for sigma in np.linspace(0.15, 0.45, 7):
        seeds.append(_gaussian(n, sigma))

    # Power‑law family
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta family – systematic grid
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential (heavy‑tailed)
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with a low‑frequency curve
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None:
        v = np.asarray(previous, dtype=np.float64)
        if v.size == n and v.sum() > 0:
            seeds.append(v / v.sum())

    # Normalise and guard against pathological zero vectors
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_500_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true objective.
    Operates on the simplex via Euclidean projection.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the *active* convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size – Polyak‑style decay
        lr = 0.20 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _lp_refine_step(
    f: np.ndarray,
    M: float,
    deadline: float,
) -> tuple[np.ndarray, bool]:
    """
    Solve the LP:
        maximise   1ᵀg
        subject to (f * g)[k] ≤ M   for k = 0 … 2n‑2
                g ≥ 0
    Returns the scaled LP solution (so that Σg = Σf) and a success flag.
    """
    if not _has_scipy:
        return f, False

    n = len(f)
    rows = 2 * n - 1

    # Build the convolution matrix A where A[k, j] = f[k‑j] (zero when out‑of‑range)
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f  # slice‑wise fill, fast and memory‑friendly

    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)            # maximise sum(g) → minimise –sum(g)
    bounds = [(0.0, None)] * n

    # Solve with HiGHS – fast and reliable for dense problems of this size
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method='highs',
        options={'presolve': True, 'maxiter': 2000},
    )

    if not res.success:
        return f, False

    g = np.maximum(res.x, 0.0)   # safety against tiny negatives

    sum_f = f.sum()
    sum_g = g.sum()
    if sum_g == 0.0:
        return f, False

    # Scale so that Σg = Σf (preserves the normalisation used by the outer loop)
    g_scaled = g * (sum_f / sum_g)
    return g_scaled, True


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    init_temp_factor: float = 0.06,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """A lightweight annealer that reshuffles mass on small random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9999, min_temp)

        if rng.random() < 0.7:          # reshuffle a random subset
            k = rng.integers(2, 12)  # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                           # small pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
        2·n·max(conv(seq,seq)) / (sum(seq)²)

    The procedure:
        1. Generate a broad seed pool (analytic windows + random draws).
        2. Keep a few top seeds w.r.t. a cheap surrogate.
        3. Refine the best seed with a projected sub‑gradient method.
        4. Repeatedly solve a linear programme that finds a
           mass‑maximising vector under the current max‑convolution bound;
           mix the LP solution with the current iterate.
        5. Finish with a modest simulated‑annealing polishing stage.
    The best sequence found is stored globally as ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose a length that balances expressivity and LP size.
    # 3000 gives a dense 6000×3000 matrix (~144 MiB) – comfortably fits.
    # ------------------------------------------------------------------
    n = 3000

    # ------------------------------------------------------------------
    # Seed collection (reuse previous solution if available)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_norm = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_norm = arr / arr.sum()
        except Exception:
            prev_norm = None

    seeds = _collect_seeds_extended(n, rng, prev_norm)
    print(f"Generated {len(seeds)} seeds (n={n}).")

    # Rank seeds by cheap surrogate
    scored = [(v, _objective_surrogate(v)) for v in seeds]
    scored.sort(key=lambda x: x[1])
    best_seq = scored[0][0].copy()
    best_val = scored[0][1]
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – quick sub‑gradient refinement of the top seed
    # ------------------------------------------------------------------
    best_seq, _ = _subgradient_proj(best_seq, n, deadline, max_iters=300_000)
    best_val = _objective_surrogate(best_seq)
    print(f"[After SubProj] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – LP‑based refinement loop
    # ------------------------------------------------------------------
    iter_lp = 0
    while time.time() < deadline and iter_lp < 5:
        iter_lp += 1
        conv = _conv_fft(best_seq, best_seq)
        M = float(conv.max())

        # Solve the LP
        g_seq, ok = _lp_refine_step(best_seq, M, deadline)
        if not ok:
            print("[LP] solve failed – exiting LP loop.")
            break

        # Mix (convex combination) – 30 % of the LP solution.
        t_mix = 0.30
        mixed = (1.0 - t_mix) * best_seq + t_mix * g_seq
        mixed = np.maximum(mixed, 0.0)
        mixed /= mixed.sum()                      # renormalise to Σ=1

        # Optional short sub‑gradient polish after mixing
        mixed, _ = _subgradient_proj(mixed, n, deadline, max_iters=100_000)

        new_val = _objective_surrogate(mixed)
        print(f"[LP‑Iter {iter_lp}] M={M:.8f} → new surrogate={new_val:.6f}")

        if new_val + 1e-9 < best_val:
            best_seq = mixed
            best_val = new_val
            print(f"  ** improvement!  best surrogate now {best_val:.6f}")
        else:
            # No improvement – stop early
            print("  No improvement, terminating LP loop.")
            break

    # ------------------------------------------------------------------
    # Stage 3 – simulated annealing polishing (lightweight)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline, max_iters=500_000)
        if val_sa < best_val:
            best_seq = seq_sa
            best_val = val_sa
            print(f"[SA] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store result for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
