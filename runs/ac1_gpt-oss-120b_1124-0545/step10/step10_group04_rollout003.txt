def _sin_power_shape(n, p):
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n) ** p
    if w.sum() == 0:
        return _uniform(n)
    return w / w.sum()
```

def _poly_shape(n, p, q):
    i = np.arange(n, dtype=np.float64)
    w = (i+1) ** p * (n-i) ** q
    if w.sum() == 0: return _uniform(n)
    return w / w.sum()
```

As p and q increase, shape peaks near right side when p>q or left side when q>p, symmetry possible.

Define _exp_decay_shape:

```
def _exp_decay_shape(n, decay):
    i = np.arange(n, dtype=np.float64)
    centre = (n-1)/2.0
    # symmetric exponential decay from centre
    w = np.exp(-decay * np.abs(i - centre))
    if w.sum() == 0: return _uniform(n)
    return w / w.sum()
```

"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

The search proceeds in four stages under a 1000 s budget:

1. **Deterministic seeds & exhaustive param sweep** –  
   a collection of analytic windows (Hann, Blackman, etc.) together with
   many parameterised families (Beta, sin‑power, polynomial, exponential,
   edge‑spike).  All candidates are normalised to sum = 1 and evaluated
   with a fast FFT‑based convolution.

2. **Heavy random sampling** –  
   Dirichlet draws and random Beta shapes for the remaining time slice.

3. **Smooth Adam refinement** –  
   minimise a log‑sum‑exp surrogate of ``max(conv(a,a))`` starting from the
   current best candidate.  Several restarts are run, the schedule for the
   smoothing parameter ``β`` rises to 300 and the learning‑rate decays as
   ``1/√t``.

4. **Projected sub‑gradient polishing** –  
   exact max‑convolution optimisation using a sub‑gradient built from all
   entries whose value is within a shrinking relative tolerance of the
   current maximum.  The iterate is projected onto the probability simplex
   after each step.

If at any point the objective drops below the target (≈ 1.503) the search
terminates early.  The best normalised sequence found is returned as a
plain ``list[float]``; ``evaluate_sequence`` will re‑scale it automatically.
"""

import time
import numpy as np
from math import pi, sqrt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a *normalised* sequence
    (sum == 1).  Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0 - a1 * np.cos(2 * pi * i / (n - 1)) + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian_sigma015(n: int) -> np.ndarray:
    sigma = 0.15 * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = w.sum()
    return w / s if s else _uniform(n)


def _gaussian_sigma025(n: int) -> np.ndarray:
    sigma = 0.25 * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = w.sum()
    return w / s if s else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    s = w.sum()
    return w / s if s else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf shape on a discrete grid (shifted by 0.5 to avoid end‑points)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    s = w.sum()
    return w / s if s else _uniform(n)


# ----------------------------------------------------------------------
# Additional parametric families
# ----------------------------------------------------------------------


def _sin_power_shape(n: int, p: float) -> np.ndarray:
    """Bell‑shaped sin‑power window, normalised."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n) ** p
    s = w.sum()
    return w / s if s else _uniform(n)


def _poly_shape(n: int, p: float, q: float) -> np.ndarray:
    """Polynomial shape w_i ∝ (i+1)^p (n-i)^q."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1) ** p * (n - i) ** q
    s = w.sum()
    return w / s if s else _uniform(n)


def _exp_decay_shape(n: int, decay: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-decay * np.abs(i - centre))
    s = w.sum()
    return w / s if s else _uniform(n)


def _spike_end_shape(n: int, mass_end: float) -> np.ndarray:
    """
    Put a total mass `mass_end` at the two ends (half each) and spread
    the remaining mass uniformly over the interior.
    """
    if mass_end <= 0.0 or mass_end >= 1.0:
        return _uniform(n)
    interior = (1.0 - mass_end) / n
    w = np.full(n, interior, dtype=np.float64)
    w[0] = mass_end / 2.0
    w[-1] = mass_end / 2.0
    return w / w.sum()


# ----------------------------------------------------------------------
# Random utilities
# ----------------------------------------------------------------------


def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Sample a point on the (n‑dim) simplex with Dirichlet(alpha)."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s else _uniform(n)


# ----------------------------------------------------------------------
# Simplex projection (used by the sub‑gradient phase)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth Adam optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _smooth_adam_optimize(start_seq: np.ndarray,
                          deadline: float,
                          beta_start: float = 5.0,
                          beta_end: float = 300.0,
                          max_iter: int = 5_000_000,
                          lr0: float = 0.03) -> tuple[np.ndarray, float]:
    """Adam on a smoothed surrogate of the max‑convolution."""
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        a = np.exp(logits)
        a = a / a.sum()

        conv = _conv_fft(a, a)

        # linear schedule for smoothing parameter β
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()

        # gradient of the log‑sum‑exp surrogate
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if (it & 1023) == 0:          # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")
            if best_val < 1.503:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (exact max‑convolution)
# ----------------------------------------------------------------------


def _proj_subgradient_optimize(start_seq: np.ndarray,
                               deadline: float,
                               eps_rel_start: float = 1e-4,
                               eps_rel_end: float = 1e-6,
                               lr0: float = 0.12,
                               max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = start_seq.copy()
    n = len(a)

    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        eps_rel = eps_rel_start + (eps_rel_end - eps_rel_start) * min(1.0, it / (max_iter * 0.5))
        thresh = max_val - eps_rel * max_val
        active_idx = np.where(conv >= thresh)[0]

        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv)], dtype=int)

        w = np.zeros_like(conv)
        w[active_idx] = 1.0 / active_idx.size

        conv_ag = _conv_fft(w, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        lr = lr0 / sqrt(it)

        a_next = a - lr * g
        a_next = _project_simplex(a_next)

        if (it & 1023) == 0:
            val = _objective(a_next)
            if val < best_val:
                best_val = val
                best_seq = a_next.copy()
                print(f"[ProjSGD] it={it:7d} n={n:5d} val={val:.6f}")
            if best_val < 1.503:
                break

        a = a_next

    return best_seq, best_val

            a = 10 ** rng.uniform(-2, 0)   # 0.01 … 1
            b = 10 ** rng.uniform(-2, 0)
            seq = _beta_shape(n, a, b)
            _maybe_update(seq, n, "BetaRand")

        # Sin‑power sweep
        for p in np.linspace(0.5, 10.0, 10):
            seq = _sin_power_shape(n, p)
            _maybe_update(seq, n, "SinPow")

        # Polynomial sweep (small integer exponents)
        for p_int in range(0, 5):
            for q_int in range(0, 5):
                seq = _poly_shape(n, p_int, q_int)
                _maybe_update(seq, n, "Poly")

        # Exponential decay sweep
        for decay in np.linspace(0.0005, 0.02, 8):
            seq = _exp_decay_shape(n, decay)
            _maybe_update(seq, n, "ExpDecay")

        # reuse previous solution if lengths match
        if previous is not None and previous.shape == (n,):
            _maybe_update(previous, n, "Prev")

        # early exit if we already beat the target
        if best_val < 1.503:
            print("Target reached in Phase 1")
            print(f"Best val={best_val:.6f} (len={len(best_seq)})")
            evaluate_sequence(best_seq.tolist())  # sanity check
            return best_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 2 – heavy random sampling (Dirichlet + random Beta)
    # ------------------------------------------------------------------
    now = time.time()
    sampling_deadline = now + 0.30 * (deadline - now)   # ~30 % of the remaining budget

    alphas = [0.1, 0.3, 0.5, 1.0, 2.0, 5.0]
    beta_params = [(0.05, 0.05), (0.1, 0.1), (0.2, 0.2), (0.3, 0.3),
                   (0.4, 0.4), (0.5, 0.5), (1.0, 1.0), (2.0, 2.0)]

    while time.time() < sampling_deadline:
        n = int(rng.choice(lengths))
        # Dirichlet draw
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        _maybe_update(seq, n, "RandDir")

        # Random Beta shape
        a, b = rng.choice(beta_params)
        # jitter parameters a little
        a *= 10 ** rng.uniform(-0.3, 0.3)
        b *= 10 ** rng.uniform(-0.3, 0.3)
        seq = _beta_shape(n, a, b)
        _maybe_update(seq, n, "RandBeta")

        if best_val < 1.503:
            print("Target reached in Phase 2")
            evaluate_sequence(best_seq.tolist())
            return best_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 3 – smooth Adam optimisation (few restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            adam_deadline = time.time() + 0.35 * remaining
            # generate a few starting points: the current best + some random Dirichlet perturbations
            starts = [best_seq]
            top_len = len(best_seq)
            for _ in range(3):
                pert = _random_dirichlet(top_len, rng, alpha=rng.choice(alphas))
                starts.append(pert)

            for idx, start in enumerate(starts):
                seq_adam, val_adam = _smooth_adam_optimize(
                    start_seq=start,
                    deadline=adam_deadline,
                    beta_start=5.0,
                    beta_end=300.0,
                    max_iter=4_000_000,
                    lr0=0.03,
                )
                if val_adam < best_val:
                    best_val = val_adam
                    best_seq = seq_adam
                    print(f"[Adam‑Refine {idx}] new best {best_val:.6f}")
                if best_val < 1.503:
                    print("Target reached during Adam phase")
                    evaluate_sequence(best_seq.tolist())
                    return best_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 4 – aggressive projected sub‑gradient polishing
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            proj_deadline = time.time() + 0.45 * remaining
            seq_proj, val_proj = _proj_subgradient_optimize(
                start_seq=best_seq,
                deadline=proj_deadline,
                eps_rel_start=1e-4,
                eps_rel_end=1e-7,
                lr0=0.12,
                max_iter=8_000_000,
            )
            if val_proj < best_val:
                best_val = val_proj
                best_seq = seq_proj
                print(f"[Proj‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check & return
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
