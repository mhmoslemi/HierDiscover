def make_U_shape(n, edge_fraction_mass, edge_width_frac):
   k = max(1, int(edge_width_frac * n))
   # allocate mass each edge = edge_fraction_mass/2 per side (total = edge_fraction_mass)
   mass_per_edge_point = (edge_fraction_mass/2) / k
   # interior points count = n - 2*k
   interior_mass = 1 - edge_fraction_mass
   if interior_mass < 0: interior_mass = 0
   if n - 2*k > 0: mass_per_interior = interior_mass / (n - 2*k)
   else: mass_per_interior = 0
   a = np.empty(n)
   a[:k] = mass_per_edge_point
   a[-k:] = mass_per_edge_point
   a[k:n-k] = mass_per_interior
   return a

    edge_width = max(1, int(edge_width_frac * n))

def _soft_max_weights(conv, lam):

    if lam <= 0:

def _adam_optimize(start, deadline, max_iters=600_000,
                   lam_start=1.0, lam_factor=1.0005):
    eps = 1e-30
    logp = np.log(start + eps)
    n = start.size
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)
    beta1=0.9; beta2=0.999; eps_adam=1e-8
    best_seq = start.copy()
    best_val = _objective(start)
    t=0
    lam = lam_start
    while t < max_iters and time.time() < deadline:
        t+=1
        # softmax to get a
        logp_shift = logp - np.max(logp)
        a = np.exp(logp_shift)
        a /= a.sum()
        # gradient
        g_a = _gradient_wrt_a(a, lam)
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)
        # Adam update
        m = beta1 * m + (1-beta1) * g_log
        v = beta2 * v + (1-beta2) * (g_log * g_log)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = 0.15 / (math.sqrt(t) + 1.0)  # maybe adjust constant
        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # evaluate true objective
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:7d} best={cur_val:.6f} λ={lam:.2f}")
        lam *= lam_factor
        # occasional restart
        if t % 50000 == 0:
            # add noise
            noise = np.random.default_rng().normal(scale=0.01, size=n)
            logp += noise
    return best_seq, best_val

def _anneal_refine(start, deadline, max_iters=250_000):
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val
    n = cur.size
    it=0
    # initial temperature ~ best_val * 0.1 maybe
    T0 = max(1e-6, cur_val * 0.05)
    decay = 0.9997
    while it < max_iters and time.time() < deadline:
        it+=1
        T = T0 * (decay ** it)
        # propose small local move
        i, j = rng.integers(0, n, size=2)
        if i==j:
            continue
        delta = rng.random() * 0.03 * cur[i]   # up to 3% mass
        # ensure not exceed cur[i]
        delta = min(delta, cur[i])
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")
        # global perturb occasionally
        if it % 3000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0:
                new_seq /= total
            else:
                new_seq = _uniform(n)
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):

                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")
    return best, best_val

Propose a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three stages

1. **Exploratory seeds** – a collection of parametrised families
   (U‑shaped edge‑mass, symmetric exponential, symmetric power‑law

   convolution is approximated with a temperature λ that grows slowly
   (λ←λ·lam_factor).  Adam’s adaptive steps and a simple learning‑rate
   schedule allow many hundred‑thousand updates within the 1000 s budget.

3. **Final polishing** – a light Metropolis‑Hastings simulated‑annealing
   walk makes occasional local mass‑swaps and global Gaussian jumps,
   accepting moves that improve the true (hard) objective.

The routine respects the time budget, stores the best sequence in the
global variable ``height_sequence_1`` and returns it as a plain list of
floats.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""

def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b (full mode) using real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation used for internal comparison (identical to
    evaluate_sequence for a normalised sequence).
    """
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_simplex(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random point on the simplex (Dirichlet(1,…,1))."""
    a = rng.random(n)
    total = a.sum()
    if total == 0:
        return _uniform(n)
    return a / total


def _make_exponential(n: int, beta: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-β·i)+exp(-β·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-beta * i) + np.exp(-beta * (n - 1 - i))
    total = a.sum()
    if total == 0:
        return _uniform(n)
    return a / total


def _make_powerlaw(n: int, alpha: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^(-α)+(n‑i)^(-α)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-alpha) + (n - i) ** (-alpha)
    total = a.sum()
    if total == 0:
        return _uniform(n)
    return a / total


def _make_edge_piecewise(n: int, edge_mass: float, edge_width_frac: float) -> np.ndarray:
    """
    U‑shaped distribution.
    ``edge_mass`` – total fraction of the mass placed on the two edges.
    ``edge_width_frac`` – fraction of the index range occupied by each edge.
    """
    edge_mass = float(np.clip(edge_mass, 0.0, 1.0))
    edge_width = max(1, int(edge_width_frac * n))
    edge_width = min(edge_width, n // 2)                     # keep a non‑empty centre
    side_mass = edge_mass / 2.0
    per_edge = side_mass / edge_width if edge_width > 0 else 0.0
    interior_len = n - 2 * edge_width
    interior_mass = 1.0 - edge_mass
    per_interior = interior_mass / interior_len if interior_len > 0 else 0.0

    a = np.full(n, per_interior, dtype=np.float64)
    a[:edge_width] = per_edge
    a[-edge_width:] = per_edge
    return a


# ----------------------------------------------------------------------
# Soft‑max over convolution entries (used by Adam)
# ----------------------------------------------------------------------
def _soft_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Return a probability vector w (sum to 1) that is a soft‑max over the
    entries of ``conv``.  λ is an inverse temperature.
    """
    max_c = conv.max()
    if lam <= 0.0:
        active = np.isclose(conv, max_c, atol=1e-12)
        cnt = np.count_nonzero(active)
        return np.where(active, 1.0 / cnt, 0.0)

    # relative deviation, scaled to keep the exponent in a reasonable range
    rel = (conv - max_c) / max_c   # ≤ 0
    exps = np.exp(lam * rel)
    w = exps / exps.sum()
    return w


def _gradient_wrt_a(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution with respect to the
    probability vector ``a``.  Implements   g = 2·rev(a) ⊛ w,
    where w are the soft‑max weights over the convolution entries.
    """
    n = a.size
    conv = _conv_fft(a, a)
    w = _soft_max_weights(conv, lam)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                     # length = 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n]             # slice that matches ``a``
    return grad


# ----------------------------------------------------------------------
# Adam optimisation on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 600_000,
                   lam_start: float = 1.0,
                   lam_factor: float = 1.0005) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑probabilities (soft‑max) of ``start``.
    ``lam_start`` is the initial smoothing temperature for the soft‑max
    over the convolution entries; it is multiplied by ``lam_factor`` at
    each iteration.  Returns the best probability vector found and its
    true (hard‑max) objective value.
    """
    eps = 1e-30
    logp = np.log(start + eps)              # initialise log‑probabilities
    n = start.size
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = start.copy()
    best_val = _objective(start)

    t = 0
    lam = lam_start

    while t < max_iters and time.time() < deadline:
        t += 1

        # current simplex vector via soft‑max (numerically stable)
        logp_shift = logp - np.max(logp)
        a = np.exp(logp_shift)
        a /= a.sum()

        # gradient of the smoothed max‑convolution
        g_a = _gradient_wrt_a(a, lam)

        # chain‑rule to obtain gradient w.r.t. log‑probabilities
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)          # shape (n,)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log * g_log)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # simple step‑size schedule
        lr = 0.15 / (math.sqrt(t) + 1.0)

        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # evaluate the true (non‑smoothed) objective
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:7d}  best={cur_val:.6f}  λ={lam:.2f}")

        # increase smoothing (approach the hard max)
        lam *= lam_factor

        # occasional random kick to escape shallow plateaus
        if t % 50000 == 0:
            noise = np.random.default_rng().normal(scale=0.01, size=n)
            logp += noise

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very light simulated‑annealing neighbour search (final polishing)
# ----------------------------------------------------------------------
def _anneal_refine(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis–Hastings walk that swaps a small fraction of mass
    between random coordinates and occasionally performs a global
    Gaussian perturbation.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    n = cur.size
    it = 0
    # temperature starts as a small fraction of the current objective
    T0 = max(1e-6, cur_val * 0.05)
    decay = 0.9997

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a tiny fraction from i to j
        delta = rng.random() * 0.03 * cur[i]      # up to 3 % of the source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d}  best={new_val:.6f}")

        # every ~3000 steps try a global Gaussian jump
        if it % 3000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0.0:
                new_seq /= total
            else:
                new_seq = _uniform(n)
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects a time budget (default
    1000 s) and returns the best vector found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    rng = np.random.default_rng(seed + 12345)

    # ------------------------------------------------------------------
    # Problem size – a balance between expressiveness and speed
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising step function (length {n}) ===")

    # ------------------------------------------------------------------
    # Gather a diversified collection of initial seeds
    # ------------------------------------------------------------------
    seeds = []

    # include the previously‑found best sequence if it matches the size
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.shape == (n,):
            prev_arr = np.maximum(prev_arr, 0.0)
            if prev_arr.sum() > 0.0:
                seeds.append(prev_arr / prev_arr.sum())

    # deterministic baseline seeds
    seeds.append(_uniform(n))
    seeds.append(_random_simplex(n, rng))

    # parametrised families ------------------------------------------------
    # symmetric exponential decay
    for beta in np.linspace(0.0, 1.5, 16):
        seeds.append(_make_exponential(n, beta))

    # symmetric power‑law
    for alpha in np.linspace(0.0, 2.5, 26):
        seeds.append(_make_powerlaw(n, alpha))

    # U‑shaped edge‑mass families
    for edge_mass in np.linspace(0.0, 0.5, 11):
        for edge_width in np.linspace(0.01, 0.30, 10):
            seeds.append(_make_edge_piecewise(n, edge_mass, edge_width))

    # ------------------------------------------------------------------
    # Evaluate all seeds (quick) and keep the best one
    # ------------------------------------------------------------------
    best_seed = None
    best_val = float("inf")
    for idx, s in enumerate(seeds):
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seed = s.copy()
        if (idx + 1) % 50 == 0:
            print(f"[Seed] evaluated {idx + 1}/{len(seeds)}  best={best_val:.6f}")

    print(f"[Init] best initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation (most of the time budget)
    # ------------------------------------------------------------------
    phase1_deadline = deadline - 30.0   # keep a few seconds for polishing
    if time.time() < phase1_deadline:
        print("[Phase 1] Adam optimisation start")
        seq_adam, val_adam = _adam_optimize(best_seed, phase1_deadline,
                                            max_iters=600_000,
                                            lam_start=1.0,
                                            lam_factor=1.0005)
        print(f"[Phase 1] best after Adam = {val_adam:.6f}")
    else:
        seq_adam, val_adam = best_seed, best_val
        print("[Phase 1] skipped (no time)")

    # ------------------------------------------------------------------
    # Phase 2 – Light simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("[Phase 2] Simulated‑annealing polishing")
        seq_final, val_final = _anneal_refine(seq_adam, deadline,
                                              max_iters=250_000)
        if val_final < val_adam:
            seq_adam, val_adam = seq_final, val_final
            print(f"[Phase 2] improved to {val_final:.6f}")
        else:
            print("[Phase 2] no improvement")
    else:
        print("[Phase 2] no time left")

    # ------------------------------------------------------------------
    # Final normalisation and export
    # ------------------------------------------------------------------
    final_seq = np.maximum(seq_adam, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
        total = final_seq.sum()
    final_seq = final_seq / total

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")

    # store for possible reuse in a later call
    globals()["height_sequence_1"] = final_seq.copy()

    return final_seq.tolist()
