def _subgrad_descent(a, n, deadline, max_iters, init_lr):

    a = a.copy()

    best_seq = a.copy(); best_val = _objective(a).

    for it in range(1, max_iters+1):

        if time.time() > deadline: break

        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val; best_seq = a.copy()
        # compute subgradient:

        g = np.zeros_like(a)

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)

        idx_i = np.arange(i_start, i_end + 1, dtype=int)

        rev_idx = max_idx - idx_i

        g[idx_i] = 2.0 * a[rev_idx]

        lr = init_lr / np.sqrt(it)

        a = a - lr * g

        a = _proj_simplex(a)

    return best_seq, best_val

def _adam_optimize(start_vec, n, deadline, max_iters):

    v = np.log(start_vec + 1e-30) # logits

    m = np.zeros_like(v)

    v2 = np.zeros_like(v)

    beta1 = 0.9; beta2 = 0.999

    eps = 1e-12

    best_seq = start_vec.copy(); best_val = _objective(start_vec)

    beta_temp = 5.0; beta_max = 500.0; beta_mul = 1.003

    for it in range(1, max_iters+1):
        if time.time()>deadline: break

        a = _softmax(v)  # simplex

        conv = _conv_fft(a, a)

        shift = np.max(conv)

        w = np.exp(beta_temp * (conv - shift))  # weights for log-sum-exp

        w_sum = np.sum(w)

        if w_sum == 0: continue

        w = w / w_sum

        # gradient of sum w_i conv_i w.r.t a: using convolution of reverse a with w

        rev_a = a[::-1]

        full = _conv_fft(rev_a, w)  # length 3n-2 (since rev a length n, w length 2n-1? Actually w length 2n-1; we need to ensure lengths: conv of a and a yields length 2n-1, and w same length. So rev_a (n) convolve w (2n-1) yields length 3n-2. We want central n entries.

        grad_a = 2.0 * full[n-1:n-1+n]

        # convert to logit gradient

        dot = np.dot(grad_a, a)

        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1-beta1) * grad_v

        v2 = beta2 * v2 + (1-beta2) * (grad_v ** 2)

        m_hat = m / (1 - beta1**it)

        v2_hat = v2 / (1 - beta2**it)

        lr = 0.6 / np.sqrt(it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

        # record best
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val; best_seq = a.copy()

        # increase beta_temp
        beta_temp = min(beta_temp * beta_mul, beta_max)

    return best_seq, best_val

"""
Propose a non‑negative step function that (hopefully) improves the
evaluate_sequence bound.

The algorithm works in three phases:

1. **Seed generation** (n≈1024).  A large heterogeneous pool of analytic
   families (triangular, windows, beta, polynomial kernels, geometric,
   random Dirichlet/exponential, …) is created and normalised to
   ℓ₁‑norm 1.  The cheap surrogate 2·n·max(conv) is evaluated for all
   seeds and the best few are kept.

2. **Local optimisation** on the small problem.
   For each of the selected seeds we run a few hundred thousand
   iterations of an Adam optimiser on a smooth log‑sum‑exp surrogate of
   the max‑convolution, followed by a projected sub‑gradient descent on
   the true max‑convolution.  The best sequence found on the small scale
   is kept.

3. **Upscaling & final polishing**.
   The best small‑scale sequence is interpolated to a larger length
   (n≈8192) and a final short sub‑gradient descent is performed.
   The final sequence is returned as a list of floats; it is also stored
   in the global ``height_sequence_1`` variable for reuse.

All steps respect the overall time budget (default 1000 s) and avoid
any file/network I/O.  The implementation uses only NumPy and the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(a*a) for ℓ₁‑normalised sequences."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    """Decaying geometric series (0<r<1)."""
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (both ends)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.355 - 0.335 * np.cos(2 * np.pi * i / (n - 1))
         + 0.155 * np.cos(4 * np.pi * i / (n - 1))
         - 0.045 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _flattop_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.21557895 - 0.41663158 * np.cos(2 * np.pi * i / (n - 1))
         + 0.277263158 * np.cos(4 * np.pi * i / (n - 1))
         - 0.083578947 * np.cos(6 * np.pi * i / (n - 1))
         + 0.006947368 * np.cos(8 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _poly_kernel(n: int, power: float) -> np.ndarray:
    """Polynomial kernel (1 - ((i-centre)/centre)^2)^power, zero outside."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    u = (i - centre) / centre
    a = np.maximum(0.0, 1.0 - u ** 2) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _downsample(seq: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to target_n, keeping ℓ₁‑norm 1."""
    seq = np.asarray(seq, dtype=np.float64)
    n = len(seq)
    if n == target_n:
        return seq.copy()
    x = np.linspace(0.0, 1.0, n)
    xi = np.linspace(0.0, 1.0, target_n)
    interp = np.interp(xi, x, seq)
    interp = np.maximum(interp, 0.0)  # eliminate possible negatives
    s = interp.sum()
    if s > 0:
        interp /= s
    else:
        interp = _uniform(target_n)
    return interp


def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a rich pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for wf in np.linspace(0.05, 0.5, 10):
        seeds.append(_trapezoidal(n, width_fraction=wf))

    for gamma in np.linspace(0.0, 3.0, 13):
        seeds.append(_sine_window(n, gamma=gamma))

    for alpha in np.linspace(0.0, 0.9, 10):
        seeds.append(_tukey_window(n, alpha=alpha))

    r_vals = np.concatenate([
        np.linspace(0.2, 0.98, 12),
        np.linspace(0.98, 0.9999, 10)
    ])
    for r in r_vals:
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 16):
        seeds.append(_power(n, p))

    # Beta shape grid (moderately coarse)
    alphas = np.linspace(0.2, 4.0, 7)
    betas = np.linspace(0.2, 4.0, 7)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for sigma_frac in np.linspace(0.1, 0.5, 9):
        seeds.append(_gaussian(n, sigma_frac))

    # Additional windows
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    seeds.append(_flattop_window(n))

    # Polynomial kernels
    for power in np.linspace(0.5, 5.0, 10):
        seeds.append(_poly_kernel(n, power))

    # Random draws
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(10):
        seeds.append(_random_exponential(n, rng))

    # Previous solution (if any) – down‑sample to current length
    if previous is not None and previous.ndim == 1:
        prev_small = _downsample(previous, n)
        seeds.append(prev_small)

    return seeds

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 300_000,
                   beta_max: float = 800.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    Returns the best sequence found (ℓ₁‑normalised) and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.6

    beta_temp = 5.0
    beta_mul = 1.003

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        # temperature increase (smooth -> sharp)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)

        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgrad_optimize(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 1_000_000,
                      init_lr: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100000 == 0:
                print(f"[Subgrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak‑style diminishing step size
        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # attempt to load a previously found construction
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Phase 1 – seed generation on a modest size
    # ------------------------------------------------------------------
    n_small = 1024
    print(f"\n--- Phase 1: generating seeds (n={n_small}) ---")
    seeds = _collect_seeds(n_small, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])

    top_k = 10
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"\nTop {top_k} surrogate values (small n):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate (small n) = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – heavy local optimisation on the small problem
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break

        # allocate a modest budget for Adam based on remaining time
        remaining = deadline - time.time()
        max_iters_adam = int(min(300_000, remaining * 2000))
        if max_iters_adam < 5_000:
            continue
        print(f"\n--- Adam optimisation (seed {idx+1}/{len(best_seeds)}) "
              f"max_iters={max_iters_adam} ---")
        seq_adam, val_adam = _adam_optimize(
            seed_vec, n_small, deadline, max_iters=max_iters_adam
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Overall‑Adam] new global best = {val_adam:.6f}")

    # Sub‑gradient refinement on the small problem (bulk of remaining time)
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_sub = int(min(1_200_000, remaining * 1500))
        if max_iters_sub > 0:
            print(f"\n--- Sub‑gradient refinement (n={n_small}) "
                  f"max_iters={max_iters_sub} ---")
            seq_sub, val_sub = _subgrad_optimize(
                global_best_seq, n_small, deadline,
                max_iters=max_iters_sub, init_lr=0.30
            )
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[Overall‑Sub] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – up‑sample & final polishing on a larger scale
    # ------------------------------------------------------------------
    n_large = 8192
    print(f"\n--- Upsampling to length {n_large} ---")
    seq_large = _downsample(global_best_seq, n_large)

    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_large = int(min(400_000, remaining * 500))
        if max_iters_large > 0:
            print(f"\n--- Final sub‑gradient polishing (n={n_large}) "
                  f"max_iters={max_iters_large} ---")
            seq_large, _ = _subgrad_optimize(
                seq_large, n_large, deadline,
                max_iters=max_iters_large, init_lr=0.20
            )

    # ------------------------------------------------------------------
    # Final evaluation with the official function
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(seq_large.tolist())
        print(f"\nOfficial evaluate_sequence value = {official_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        official_val = float("inf")

    # Store result for potential reuse
    globals()["height_sequence_1"] = seq_large.tolist()
    return seq_large.tolist()
