def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N).real[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – the objective for a normalised (sum=1) seq."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed families (all return normalised vectors, sum ≈ 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = a0 - a1 * np.cos(2 * pi * i / (n - 1)) + a2 * np.cos(4 * pi * i / (n - 1))
    return w / w.sum()


def _gaussian_factory(sigma_factor: float):
    """Return a function that generates a Gaussian window with the given sigma factor."""
    def _gaussian(n: int) -> np.ndarray:
        sigma = sigma_factor * n
        i = np.arange(n, dtype=np.float64)
        centre = (n - 1) / 2.0
        w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
        s = w.sum()
        if s == 0.0:
            return _uniform(n)
        return w / s
    _gaussian.__name__ = f"_gaussian_{sigma_factor}"
    return _gaussian


def _parabolic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (i + 1) * (n - i)
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


def _lanczos_sq(n: int) -> np.ndarray:
    """Lanczos window squared (always non‑negative)."""
    i = np.arange(n, dtype=np.float64)
    x = 2 * i / (n - 1) - 1.0  # range [-1, 1]
    w = np.sinc(x)  # np.sinc(x) = sin(pi*x)/(pi*x)
    w = w ** 2
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


def _sin_shape(n: int) -> np.ndarray:
    """Half‑sine bump (zero at ends, peak in centre)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(pi * i / n)
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta pdf on {0,…,n‑1} (shifted by 0.5 to avoid the exact ends)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


# ----------------------------------------------------------------------
# Random generation utilities
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet‑like vector using Gamma or Exp depending on α."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


def _random_beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.uniform(0.1, 5.0)
    b = rng.uniform(0.1, 5.0)
    return _beta_shape(n, a, b)


def _random_mixture_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two independent Betas with a random mixing weight."""
    w = rng.random()
    a1, b1 = rng.uniform(0.1, 5.0), rng.uniform(0.1, 5.0)
    a2, b2 = rng.uniform(0.1, 5.0), rng.uniform(0.1, 5.0)
    s1 = _beta_shape(n, a1, b1)
    s2 = _beta_shape(n, a2, b2)
    mix = w * s1 + (1.0 - w) * s2
    mix /= mix.sum()
    return mix


def _random_sparse(n: int, rng: np.random.Generator, k_frac: float = 0.05) -> np.ndarray:
    """Create a sparse vector with ~k_frac·n non‑zero entries."""
    k = max(2, int(k_frac * n))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k) + 0.1
    seq = np.zeros(n, dtype=np.float64)
    seq[idx] = vals
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 1000.0,
    max_iter: int = 5_000_000,
    lr0: float = 0.03,
) -> tuple[np.ndarray, float]:
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    best_seq = start_seq.copy()
    best_val = _objective(best_seq)
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        a = np.exp(logits)
        a = a / a.sum()
        conv = _conv_fft(a, a)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)
        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")
            if best_val < 1.45:
                break
    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS refinement (smooth max with large β)
# ----------------------------------------------------------------------
def _lbfgs_refine(start_seq: np.ndarray, deadline: float, beta: float = 1000.0) -> tuple[np.ndarray, float]:
    if minimize is None:
        return start_seq, _objective(start_seq)

    n = len(start_seq)
    eps = 1e-30

    def fun_and_grad(logits: np.ndarray):
        a = np.exp(logits)
        a = a / a.sum()
        conv = _conv_fft(a, a)
        c = np.max(conv)
        exps = np.exp(beta * (conv - c))
        lse = c + (1.0 / beta) * np.log(exps.sum())
        val = 2.0 * n * lse
        p = exps / exps.sum()
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)
        return val, grad_logits.astype(np.float64)

    x0 = np.log(start_seq + eps)

    class TimeOut(RuntimeError):
        pass

    def callback(xk):
        if time.time() >= deadline:
            raise TimeOut()

    try:
        res = minimize(
            fun=lambda x: fun_and_grad(x)[0],
            x0=x0,
            jac=lambda x: fun_and_grad(x)[1],
            method='L-BFGS-B',
            options={'maxiter': 800, 'disp': False},
            callback=callback,
        )
    except TimeOut:
        best_logits = x0
    else:
        best_logits = res.x

    best_a = np.exp(best_logits)
    best_a /= best_a.sum()
    best_val = _objective(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive aggregated mirror‑descent polishing (symmetry enforced)
# ----------------------------------------------------------------------
def _aggregated_mirror_descent_sym(
    start_seq: np.ndarray,
    deadline: float,
    eps_rel_start: float = 0.02,
    eps_rel_end: float = 1e-8,
    lr_factor: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror‑descent where all entries whose convolution value lies within a
    relative tolerance of the current maximum are aggregated.  After each
    exponentiated‑gradient step the vector is symmetrised (a ← (a+a[::-1])/2)
    and re‑normalised.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    start_time = time.time()
    it = 0
    eps = 1e-30
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        # exponential decay of tolerance
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 0.0
        eps_rel = eps_rel_start * (eps_rel_end / eps_rel_start) ** frac
        thresh = max_val - eps_rel * max_val

        mask = (conv >= thresh).astype(np.float64)

        # aggregated sub‑gradient = 2 * (mask ⋆ a_rev)
        conv_ag = _conv_fft(mask, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        lr = lr_factor / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        a /= a.sum()
        # enforce symmetry (helps to keep the optimum in the symmetric sub‑space)
        a = (a + a[::-1]) * 0.5

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[AggMD‑sym] it={it:7d} n={n:5d} best={val:.6f}")
            if time.time() > deadline - 0.5:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Re‑use a previously computed solution if available
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            prev = np.asarray(prev, dtype=np.float64)
            if prev.size > 0 and np.all(prev >= 0):
                prev = prev / prev.sum()
        except Exception:
            prev = None
    else:
        prev = None

    # ------------------------------------------------------------------
    # Lengths to explore (FFT‑friendly choices)
    # ------------------------------------------------------------------
    lengths = [2000, 2500, 3000, 3500, 4000, 5000, 6000, 8000, 10000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic families
    # ------------------------------------------------------------------
    # families without parameters
    seed_families = [
        _uniform,
        _hann,
        _hamming,
        _blackman,
        _parabolic,
        _sin_shape,
        _lanczos_sq,
    ]

    # parameterised families (generated as concrete functions)
    seed_families += [
        _gaussian_factory(0.10),
        _gaussian_factory(0.15),
        _gaussian_factory(0.20),
        _gaussian_factory(0.25),
        _gaussian_factory(0.30),
        # beta pdfs with a few classic parameters
        (lambda n: _beta_shape(n, 0.5, 0.5)),
        (lambda n: _beta_shape(n, 2.0, 2.0)),
        (lambda n: _beta_shape(n, 0.5, 2.0)),
        (lambda n: _beta_shape(n, 2.0, 0.5)),
        (lambda n: _beta_shape(n, 5.0, 5.0)),
    ]

    # Note: the above lambdas are only used locally; they do not survive beyond this function.

    for n in lengths:
        for fn in seed_families:
            try:
                seq = fn(n)
            except Exception:
                continue
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

        if prev is not None and prev.shape == (n,):
            val = _objective(prev)
            if val < best_val:
                best_val = val
                best_seq = prev.copy()
                print(f"[Seed] n={n:5d} type=Prev val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – random sampling (≈30 % of total budget)
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.30
    sampling_deadline = now + time_for_sampling

    alphas = [0.05, 0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0]
    while time.time() < sampling_deadline:
        n = int(rng.choice(lengths))

        # Dirichlet with random concentration
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd‑Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        # Random Beta pdf
        seq = _random_beta_shape(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd‑Beta] n={n:5d} val={val:.6f}")

        # Random mixture of two Betas
        seq = _random_mixture_beta(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd‑Mix] n={n:5d} val={val:.6f}")

        # Random sparse vector (might discover exotic shapes)
        seq = _random_sparse(n, rng, k_frac=0.03)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd‑Sp] n={n:5d} val={val:.6f}")

        if best_val < 1.45:
            break

    # ------------------------------------------------------------------
    # Phase 3 – Smoothed Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        candidates = [best_seq]
        top_len = len(best_seq)

        # couple of fresh Dirichlet restarts, same length
        for _ in range(2):
            cand = _random_dirichlet(top_len, rng, alpha=rng.choice(alphas))
            candidates.append(cand)

        # a larger‑length restart (if any)
        larger = [L for L in lengths if L > top_len]
        if larger:
            L_big = int(rng.choice(larger))
            candidates.append(_random_dirichlet(L_big, rng, alpha=1.0))

        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.25   # ~25 % of what’s left

        for idx, cand in enumerate(candidates):
            seq_adam, val_adam = _smooth_adam_optimize(
                start_seq=cand,
                deadline=adam_deadline,
                beta_start=5.0,
                beta_end=1000.0,
                max_iter=5_000_000,
                lr0=0.03,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = seq_adam
                print(f"[Adam‑Refine {idx}] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – optional L‑BFGS refinement
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0 and minimize is not None:
            lbfgs_deadline = time.time() + remaining * 0.20
            seq_lb, val_lb = _lbfgs_refine(best_seq, lbfgs_deadline, beta=1000.0)
            if val_lb < best_val:
                best_val = val_lb
                best_seq = seq_lb
                print(f"[L‑BFGS] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Aggressive mirror‑descent polishing (symmetry enforced)
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _aggregated_mirror_descent_sym(
            start_seq=best_seq,
            deadline=deadline,
            eps_rel_start=0.02,
            eps_rel_end=1e-8,
            lr_factor=1.0,
        )
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[AggMD‑sym] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will re‑evaluate anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
