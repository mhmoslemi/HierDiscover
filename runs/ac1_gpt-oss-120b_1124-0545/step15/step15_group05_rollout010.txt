"""
Propose a step‑function (non‑negative vector) that minimises the objective

        2·n·max(conv(a,a)) / (sum(a)²)

using a *sparse Sidon‑like construction*.  The idea is to place a moderate
number of equal spikes (≈ 160‑200) on the length‑n grid such that the number
of ordered pairs (i,j) with the same sum i+j is bounded by a small constant
R (here 7).  For a uniform weight w=1/K the convolution at any index is
w²·c where c≤R, so the objective becomes 2·n·R/K².  With n=4000 and
R=7 we need only K≥⌈√(2·n·R/target)⌉≈164 to beat the target 1.503.  A greedy
randomised construction respects the bound R, then the resulting vector is
evaluated with the official ``evaluate_sequence`` routine.  If a better
candidate is found we optionally polish it with a short exponentiated‑gradient
refinement.

The whole search runs for the allocated time budget (≈1000 s) and returns the
best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: greedy construction of a set of positions with bounded
# ordered‑pair multiplicities.
# ----------------------------------------------------------------------
def _build_sparse_set(n: int,
                     K_target: int,
                     R: int,
                     deadline: float,
                     rng: np.random.Generator):
    """
    Try to build a set of K_target distinct positions in {0,…,n‑1}
    such that for every possible sum s the ordered count
    ``# {(i,j) : pos_i+pos_j = s}`` never exceeds R.
    Returns the list of positions (may be shorter than K_target) and the
    actual maximal ordered count observed.
    """
    max_len = 2 * n
    # ordered pair counts for each possible sum
    counts = np.zeros(max_len, dtype=np.int16)

    positions = []
    # iterate over a random permutation of the whole grid
    perm = rng.permutation(n)
    for cand in perm:
        if time.time() > deadline:
            break

        # compute the increments this candidate would introduce
        inc = {}                     # sum -> additional count
        s_self = cand * 2
        inc[s_self] = inc.get(s_self, 0) + 1

        for p in positions:
            s = cand + p
            inc[s] = inc.get(s, 0) + 2   # (cand,p) and (p,cand)

        # check the capacity constraint
        ok = True
        for s, delta in inc.items():
            if counts[s] + delta > R:
                ok = False
                break

        if ok:
            # accept the candidate
            positions.append(cand)
            for s, delta in inc.items():
                counts[s] += delta
            if len(positions) >= K_target:
                break

    max_factor = int(counts.max())
    return positions, max_factor


# ----------------------------------------------------------------------
# Optional fine‑tuning via exponentiated‑gradient on the exact max‑conv.
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 500_000):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact maximal
    convolution value.  ``start_seq`` must already be ℓ₁‑normalised.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(np.convolve(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = np.convolve(a, a)
        s = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, s - (n - 1))
        i_end = min(n - 1, s)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = s - idx
        g[idx] = 2.0 * a[rev]

        eta = 1.0 / np.sqrt(it)          # aggressive step‑size
        log_a = np.log(a + 1e-30) - eta * g
        a = np.exp(log_a)
        a = a / a.sum()

        cur_val = 2.0 * n * np.max(np.convolve(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq

    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    n = 4000
    R = 7                                      # max ordered count per sum
    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # If a previous solution exists, keep it as a baseline.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:

        except Exception:

    # ------------------------------------------------------------------
    # Repeated random greedy attempts.
    # ------------------------------------------------------------------
    attempts = 0
    while time.time() < deadline:
        attempts += 1

        K_target = rng.integers(150, 240)

        positions, max_factor = _build_sparse_set(n, K_target, R, deadline, rng)
        K = len(positions)

    # Store for possible reuse.
    globals()["height_sequence_1"] = best_seq

    elapsed = time.time() - start
    print(f"\n=== Search finished (elapsed {elapsed:.1f}s) ===")
    print(f"Best objective value discovered: {best_val:.6f}")

    return best_seq
