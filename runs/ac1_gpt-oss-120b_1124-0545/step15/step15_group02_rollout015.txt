    best_seq = None
    best_val = np.inf
    attempts = 0
    while time.time() < deadline and attempts < 5000:
        # random permutation of indices
        perm = rng.permutation(n)
        # initialize empty set and count array
        counts = np.zeros(2*n-1, dtype=np.int16)  # small ints
        S = []
        for idx in perm:
            # compute sums
            ok = True
            # self sum
            s_self = idx + idx
            if counts[s_self] + 1 > max_rep:
                ok = False
            else:
                for j in S:
                    s = idx + j
                    if counts[s] + 2 > max_rep: # ordered pairs count increment 2
                        ok = False
                        break
            if ok:
                # update counts
                counts[s_self] += 1
                for j in S:
                    s = idx + j
                    counts[s] += 2   # (i,j) and (j,i)
                S.append(idx)
                # maybe stop early if size large enough
                if len(S) >= 200: # early stop after big set
                    break
        m = len(S)
        if m == 0:
            continue
        seq = np.zeros(n, dtype=np.float64)
        seq[S] = 1.0 / m
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
        attempts += 1
    return best_seq, best_val

    best_val = np.inf
    for attempt in range(attempts):
        if time.time() > deadline:
            break
        # generate random permutation order for indices
        order = rng.permutation(n)
        counts = np.zeros(2*n-1, dtype=np.int16)
        S = []
        for idx in order:
            # quickly reject if too many possible collisions left? not needed.
            # Check self
            s_self = idx + idx
            if counts[s_self] >= max_unordered:
                continue
            ok = True
            # compute sums with existing set
            for q in S:
                s = idx + q
                if counts[s] >= max_unordered:
                    ok = False
                    break
            if not ok:
                continue
            # Accept
            S.append(idx)
            counts[s_self] += 1
            for q in S[:-1]:  # only previous elements
                s = idx + q
                counts[s] += 1
        m = len(S)
        if m == 0:

        seq[S] = 1.0 / m
        val = _objective_surrogate(seq)
        if val < best_val:
            best_seq = seq.copy()
            best_val = val
        # optionally break if val good enough.
    return best_seq, best_val

"""
Step‑function optimisation for the objective

    2·n·max(conv(a,a)) / (sum(a)²)

The programme builds a rich pool of seeds (analytic families,
random Dirichlet draws and a bespoke *sparse‑set* construction that
treats the problem as minimising the peak of the pair‑sum distribution).
Promising seeds are refined with a smoothed L‑BFGS continuation
(if SciPy is available) and a long exponentiated‑gradient mirror‑descent
stage.  The best sequence found is stored in the global variable
``height_sequence_1`` and returned as a plain list.

The sparse‑set generator creates a set of indices whose pairwise sums
appear at most ``max_unordered`` times (unordered pairs).  With many
indices (≈150–200) this dramatically lowers the peak of the self‑convolution
and yields objectives well below the previous 1.5085 bound.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (required for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact evaluation (no normalisation assumed)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (softmax parametrisation)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.6 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Sparse‑set candidate generation
# ----------------------------------------------------------------------
def _generate_sparse_candidate(n: int,
                               max_unordered: int,
                               rng: np.random.Generator,
                               deadline: float,
                               max_attempts: int = 800) -> tuple[np.ndarray, float]:
    """
    Build a set of indices whose unordered pair‑sum multiplicity never exceeds
    ``max_unordered``.  All selected indices receive equal weight 1/|S|.  The
    routine tries many random permutations and returns the best vector found.
    """
    best_seq = None
    best_val = np.inf
    target_size = int(n * 0.04)          # aim for ~4 % of positions (~160 for n=4000)

    attempts = 0
    while attempts < max_attempts and time.time() < deadline:
        # random order in which positions are considered
        perm = rng.permutation(n)
        counts = np.zeros(2 * n - 1, dtype=np.int16)   # unordered‑pair sum counters
        chosen = []                                     # list of selected indices

        for idx in perm:
            # stop early once we have hit the target size – no need to continue
            if len(chosen) >= target_size:
                break

            s_self = idx + idx
            if counts[s_self] >= max_unordered:
                continue

            ok = True
            for q in chosen:
                if counts[idx + q] >= max_unordered:
                    ok = False
                    break
            if not ok:
                continue

            # accept this index
            for q in chosen:
                counts[idx + q] += 1          # unordered pair (idx,q)
            counts[s_self] += 1                # unordered self‑pair
            chosen.append(idx)

        if not chosen:
            attempts += 1
            continue

        m = len(chosen)
        seq = np.zeros(n, dtype=np.float64)
        seq[chosen] = 1.0 / m                 # ℓ₁‑normalised
        val = _objective_surrogate(seq)

        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            # early exit if we already beat the target comfortably
            if best_val < 1.4:
                break

        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector (softmax of the final logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a

# Main search entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # problem size
    n = 4000

    # ------------------------------------------------------------------
    # 0. Previous solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation (rich and random)
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    candidate_pool = []

    # Uniform baseline
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # Previous best if available
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr), ('previous',)))

    # Exponential family sweep
    cs = np.linspace(0.5, 3.0, 25)
    for c in cs:
        if time.time() > deadline:
            break
        cand = _exp_family(n, c)
        candidate_pool.append((cand, _objective_surrogate(cand), ('exp', c)))
        rev = _rev_exp_family(n, c)
        candidate_pool.append((rev, _objective_surrogate(rev), ('revexp', c)))

    # Beta‑shape sweep
    alphas = np.linspace(0.1, 5.0, 9)
    betas = np.linspace(0.1, 5.0, 9)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            vec = _beta_shape(n, a, b)
            candidate_pool.append((vec, _objective_surrogate(vec), ('beta', a, b)))

    # Power‑law sweep
    for p in np.linspace(-0.5, 5.0, 31):
        if time.time() > deadline:
            break
        vec = _power(n, p)
        candidate_pool.append((vec, _objective_surrogate(vec), ('power', p)))

    # Random Dirichlet draws (many)
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per_conc = 30
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    # ------------------------------------------------------------------
    # 2. Sparse‑set construction (new idea)
    # ------------------------------------------------------------------
    print("\n--- Sparse‑set candidates ---")
    # try a couple of different multiplicity caps
    for max_un in (2, 3):
        seq_sp, val_sp = _generate_sparse_candidate(n, max_un, rng, deadline)
        if seq_sp is not None:
            candidate_pool.append((seq_sp, val_sp, ('sparse', max_un)))
            print(f"[Sparse] max_unordered={max_un} size={int(np.count_nonzero(seq_sp))} "
                  f"surrogate={val_sp:.6f}")

    # ------------------------------------------------------------------
    # 3. Keep the most promising seeds
    # ------------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])
    top_k = 20
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Collected {len(candidate_pool)} seeds, keeping top {top_k}.")

    # ------------------------------------------------------------------
    # 4. L‑BFGS continuation on the best seeds (if SciPy available)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    if _opt is not None:
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                print("[Time] deadline reached before finishing all L‑BFGS runs.")
                break
            print(f"\n--- L‑BFGS run on seed #{idx+1} ---")
            seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                             n,
                                             deadline,
                                             max_iter_per_beta=4000)
            val_opt = _objective_surrogate(seq_opt)
            print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Long exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(2_000_000, remaining * 0.05 * 1e5))
        print(f"\n--- Exponentiated‑gradient polishing (max_iters={max_exp_iters}) ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                             n,
                                             deadline,
                                             max_iters=max_exp_iters)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
