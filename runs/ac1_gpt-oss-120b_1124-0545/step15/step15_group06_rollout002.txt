"""
Propose a non‑negative step function that minimizes

        2·n·max(conv(a,a)) / (sum(a)²)

The implementation expands on the previous baseline by:

* Adding a richer seed pool (DPSS and Chebyshev windows, many extra
  β‑U‑shapes, sparsity, low‑concentration Dirichlet draws, and random
  mixtures).
* Using a simple Adam optimiser on the smooth log‑sum‑exp surrogate
  (instead of L‑BFGS) – it converges quickly with far fewer FFTs.
* Running several restarts of a mirror‑descent (exponentiated‑gradient)
  optimiser on the exact max‑convolution objective.
* A light‑weight perturb‑and‑polish stage (log‑softmax jitter + short
  Adam run) and a final high‑β Adam polishing.
* All stages respect the global 1000 s time budget and keep the best

The code is self‑contained, uses only NumPy / SciPy and prints progress

import time
import math

import scipy.optimize as _opt
from scipy.signal import windows as _wins

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on {1,…,n} (U‑shape when α,β<1)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Slepian (DPSS) window – good spectral concentration."""
    a = _wins.dpss(n, nw, return_ratios=False)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _chebwin_window(n: int, att: float) -> np.ndarray:
    a = _wins.chebwin(n, at=att)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_sparse(n: int, rng: np.random.Generator,
                   k_min: int = 30, k_max: int = 150) -> np.ndarray:
    """Sparse support: choose K random indices, give them random weights."""
    K = int(rng.integers(k_min, k_max + 1))
    idx = rng.choice(n, size=K, replace=False)
    a = np.zeros(n, dtype=np.float64)
    weights = rng.exponential(scale=1.0, size=K)
    a[idx] = weights
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α)+b·α then ℓ₁‑renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Surrogate (log‑sum‑exp) utilities
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Returns (value, gradient) of the smooth max‑convolution surrogate.
    Variable v are the logits (real vector); a = softmax(v) is the
    probability vector (ℓ₁‑normalised).
    """
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)              # full autocorrelation
    shift = np.max(conv)                # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_term / sum_exp               # softmax weights over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # convolution length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # Chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _adam_optimize(v0: np.ndarray, n: int, deadline: float,
                   beta_schedule, steps_per_beta: int,
                   lr: float = 0.05,
                   b1: float = 0.9, b2: float = 0.999,
                   eps: float = 1e-8) -> np.ndarray:
    """
    Adam optimiser on the smooth surrogate.  Operates directly on the
    logits v (unconstrained).  Returns the softmax of the final logits.
    """
    v = v0.copy()
    m = np.zeros_like(v)
    u = np.zeros_like(v)   # second moment
    t = 0
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                break
            t += 1
            obj, g = _smooth_objective(v, n, beta)
            # Adam updates
            m = b1 * m + (1 - b1) * g
            u = b2 * u + (1 - b2) * (g * g)
            m_hat = m / (1 - b1 ** t)
            u_hat = u / (1 - b2 ** t)
            v -= lr * m_hat / (np.sqrt(u_hat) + eps)
    return _softmax(v)


# ----------------------------------------------------------------------
# Exact objective utilities
# ----------------------------------------------------------------------
def _objective_raw(a: np.ndarray) -> float:
    """Exact evaluator 2·n·max(conv) / (sum²) (assumes sum(a)>0)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (based on the active lag).
    Returns g = ∂ max(conv) / ∂ a.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


def _exp_gradient_optimize(start_a: np.ndarray, n: int, deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with exponentiated‑gradient updates on the exact
    objective.  Returns the best vector encountered and its objective.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_raw(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        g = _subgradient_maxconv(a)
        eta = eta0 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 50_000 == 0:
            cur = _objective_raw(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
    # final check
    cur = _objective_raw(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Seed generation (larger and more diverse pool)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Generate a heterogeneous pool of ℓ₁‑normalised seeds."""
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12, 20):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.08, 0.45, 9):
        seeds.append(_gaussian(n, sigma))
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta families – include many U‑shapes
    for a in np.linspace(0.1, 1.5, 7):
        for b in np.linspace(0.1, 1.5, 7):
            seeds.append(_beta_shape(n, a, b))

    for al in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, al))

    # DPSS (Slepian) windows – very good spectral concentration
    for nw in (2.0, 2.5, 3.0, 4.0, 5.0):
        try:
            seeds.append(_dpss_window(n, nw))
        except Exception:
            pass

    # Chebyshev windows – control side‑lobe level
    for att in (30, 40, 50, 60, 80, 100):
        try:
            seeds.append(_chebwin_window(n, att))
        except Exception:
            pass

    # Random continuous draws
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sparse random supports
    for _ in range(30):
        seeds.append(_random_sparse(n, rng, k_min=30, k_max=120))

    # Mixtures of existing seeds
    base_idx = np.arange(len(seeds))
    for _ in range(500):
        i, j = rng.choice(base_idx, size=2, replace=False)
        alpha = rng.random()
        mix = _mix_shapes(seeds[i], seeds[j], alpha)
        seeds.append(mix)

    # Previous best (warm‑start)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise and enforce non‑negativity
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    ``evaluate_sequence`` under the given time budget.  Returns the best
    sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # reserve a few seconds for final eval

    # problem size (default 4000 – enough resolution while still fast)
    n = int(kwargs.get("n", 4000))
    print(f"\n=== Optimisation start (n={n}) ===")

    # ------------------------------------------------------------------
    # Stage 0 – seed pool
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # quickly evaluate the cheap surrogate (sum is 1 for all seeds)
    def _surrogate(v):
        return 2.0 * n * np.max(_conv_fft(v, v))

    surrogate_vals = [(vec, _surrogate(vec)) for vec in seeds]
    surrogate_vals.sort(key=lambda x: x[1])
    top_k = 10
    top_seeds = [vec for vec, _ in surrogate_vals[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _surrogate(best_seq)
    print(f"Initial best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation on the smooth surrogate (multiple starts)
    # ------------------------------------------------------------------
    beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0,
                     200.0, 500.0, 1000.0, 2000.0,
                     5000.0, 10000.0, 20000.0]

    # allocate a modest number of Adam steps per β (fast but effective)
    steps_per_beta = 500

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing Adam runs.")
            break
        print(f"\n--- Adam run #{idx+1} on seed (steps/β={steps_per_beta}) ---")
        logits0 = np.log(seed_vec + 1e-30)
        seq_opt = _adam_optimize(logits0, n, deadline,
                                 beta_schedule=beta_schedule,
                                 steps_per_beta=steps_per_beta,
                                 lr=0.05)
        val_opt = _surrogate(seq_opt)
        print(f"  Surrogate after Adam = {val_opt:.6f}")
        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"  ** New global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient refinement on the *exact* objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 0.3 * 1e5))
        if max_iters > 0:
            print("\n--- Exponentiated‑gradient refinement (exact objective) ---")
            # run a few restarts (including a jittered version of best_seq)
            restarts = 3
            for r in range(restarts):
                if time.time() > deadline:
                    break
                # jitter the logits of the current best
                logits = np.log(best_seq + 1e-30)
                jitter = rng.normal(scale=0.3 / (r + 1), size=n)
                start_logits = logits + jitter
                start_vec = _softmax(start_logits)

                seq_exp, val_exp = _exp_gradient_optimize(start_vec,
                                                          n,
                                                          deadline,
                                                          max_iters=max_iters,
                                                          eta0=1.0)
                print(f"  Restart {r+1}: exact obj = {val_exp:.6f}")
                if val_exp < best_val:
                    best_seq = seq_exp.copy()
                    best_val = val_exp
                    print(f"  ** Improved global exact obj = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – cheap perturb‑and‑polish (log‑softmax jitter + short Adam)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Perturb‑and‑polish phase ---")
        attempts = 20
        for a in range(attempts):
            if time.time() > deadline:
                break
            logits = np.log(best_seq + 1e-30)
            scale = max(0.07, 0.4 / (a + 1))
            jitter = rng.normal(scale=scale, size=n)
            start_logits = logits + jitter
            seq_cand = _adam_optimize(start_logits, n, deadline,
                                      beta_schedule=[1.0, 20.0, 200.0],
                                      steps_per_beta=300,
                                      lr=0.07)
            cand_val = _surrogate(seq_cand)
            if cand_val < best_val:
                best_seq = seq_cand.copy()
                best_val = cand_val
                print(f"  [Perturb] improvement after attempt {a+1}: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final high‑β polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        steps_final = int(min(2000, remaining * 0.05 * 1e5))
        if steps_final > 0:
            print("\n--- Final high‑β Adam polishing ---")
            seq_final = _adam_optimize(np.log(best_seq + 1e-30), n, deadline,
                                       beta_schedule=[5000.0, 10000.0, 20000.0, 50000.0],
                                       steps_per_beta=steps_final,
                                       lr=0.03)
            val_final = _surrogate(seq_final)
            print(f"  Final polish surrogate = {val_final:.6f}")
            if val_final < best_val:
                best_seq = seq_final.copy()
                best_val = val_final
                print(f"  ** New best after final polish = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
