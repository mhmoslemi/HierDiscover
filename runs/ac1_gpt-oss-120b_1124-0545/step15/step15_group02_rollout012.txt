def _mass_transfer_opt(a_vec, cur_max, time_deadline, max_iters=1_000_000):
    a = a_vec.copy()
    conv = np.convolve(a, a)  # full.
    cur_max_val = cur_max
    best_a = a.copy()
    best_val = 2.0 * len(a) * cur_max_val

    n = len(a)
    it = 0
    while time.time() < time_deadline and it < max_iters:
        it += 1
        # sample source proportional to a
        src = rng.choice(n, p=a)
        if a[src] < 1e-12:
            continue
        # choose target uniformly not src
        tgt = rng.integers(0, n-2)
        if tgt >= src:
            tgt += 1
        # step factor
        step_factor = 0.2 / math.sqrt(it)
        d = min(a[src] * step_factor, a[src] * 0.5)  # ensure not too large
        if d <= 0:
            continue
        # copy before
        a_before = a.copy()
        conv_before = conv.copy()
        # apply move
        a[src] -= d
        a[tgt] += d
        # update conv
        # Using a_before for contributions
        conv[src:src+n] -= 2.0 * d * a_before
        conv[tgt:tgt+n] += 2.0 * d * a_before
        conv[src+src] += d*d
        conv[tgt+tgt] += d*d
        conv[src+tgt] -= 2.0 * d*d

        new_max = conv.max()
        if new_max < cur_max_val:
            cur_max_val = new_max
            val = 2.0 * n * new_max
            if val < best_val:
                best_val = val
                best_a = a.copy()
                if it % 100000 == 0:
                    print(f"[MassTransfer] iter {it} best={best_val:.6f}")
        else:
            # revert
            a = a_before
            conv = conv_before
    return best_a, best_val

"""
Propose a step‑function (list of non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

within a 1000 s wall‑clock budget.

Algorithm
---------

1. **Initialisation** – start from the best sequence stored in the global
   variable ``height_sequence_1`` (if it has the required length).  If none
   is available fall back to a simple β‑shape distribution (smooth and
   already reasonably good).

2. **Smooth L‑BFGS** – optimise a log‑softmax parametrisation of the
   sequence using a log‑sum‑exp surrogate for the max‑convolution.  This
   quickly drives the objective close to the best known values.

3. **LP refinement** – for a few iterations solve the linear programme

        maximise   1ᵀb
        subject to  a∗b ≤ M,    b ≥ 0

   where ``M`` is the current maximal convolution value.
   The solution ``b`` gives a promising direction; a convex combination
   ``(1‑t)·a + t·b̂`` (with ``b̂`` the ℓ¹‑normalised ``b``) is taken,
   reducing the bound on the max‑convolution.

4. **Mass‑transfer hill‑climbing** – while time permits repeatedly move a
   small amount of mass from an index that heavily contributes to the
   current maximal convolution entry to the index with the smallest weight.
   The convolution is updated *incrementally* (O(n) per move) and a move
   is kept only if it lowers the maximal convolution.  The step size
   decays like ``1/√iteration`` so that the method can perform a large
   number of fine adjustments.

5. **Final evaluation** – the best sequence found is evaluated with the
   official ``evaluate_sequence`` function, stored in the global variable
   ``height_sequence_1`` and returned as a plain list.

All numbers are kept non‑negative, ℓ¹‑normalised (sum = 1) and the code
uses only NumPy, SciPy (for L‑BFGS and the tiny linear programmes) and the
standard library.  The implementation respects the 1000 s budget and
prints occasional progress messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# SciPy imports (may be absent on some runners – the code works without them)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
except Exception:  # pragma: no cover
    _opt = None
    _sp = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = exact objective for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 6000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector (softmax of the final logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0]

    v = np.log(start_vec + 1e-30)   # logits for softmax parametrisation

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Linear‑programme refinement (active‑set LP)
# ----------------------------------------------------------------------
def _lp_refinement(a_vec: np.ndarray,
                   cur_max: float,
                   n: int,
                   deadline: float,
                   max_iters: int = 6,
                   step: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based improvements.
    Returns the improved vector and its (surrogate) objective.
    """
    if _opt is None or _sp is None:
        return a_vec, 2.0 * n * cur_max

    a_curr = a_vec.copy()
    best_a = a_curr.copy()
    best_val = 2.0 * n * cur_max
    cur_M = cur_max

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # Build the Toeplitz convolution matrix A where A @ b = a * b
        # Only non‑negative offsets (0 … n‑1) are non‑zero.
        diag_data = [np.full(n, a_curr[d]) for d in range(n)]
        offsets = list(range(n))
        A = _sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format='csc')

        # Linear programme: maximise sum(b)  <=> minimise -sum(b)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        ub = np.full(2 * n - 1, cur_M, dtype=np.float64)

        # Give the solver a tiny time budget to keep overall runtime low.
        time_left = max(0.0, deadline - time.time())
        try:
            res = _opt.linprog(c,
                               A_ub=A,
                               b_ub=ub,
                               bounds=bounds,
                               method='highs',
                               options={'presolve': True,
                                        'time_limit': time_left})
        except Exception as e:  # pragma: no cover
            print(f"[LP] linprog failed: {e}")
            break

        if not res.success:
            # No progress – stop early.
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break
        # Normalise b to a probability vector (ℓ₁‑norm = 1)
        b_norm = b / sum_b

        # Convex combination with step size ``step``.
        a_new = (1.0 - step) * a_curr + step * b_norm
        conv_new = _conv_fft(a_new, a_new)
        M_new = np.max(conv_new)
        val_new = 2.0 * n * M_new

        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            a_curr = a_new.copy()
            cur_M = M_new
            print(f"[LP] iter {it+1:2d} improved surrogate = {best_val:.6f}")
        else:
            # No improvement – shrink the step size.
            step *= 0.5
            if step < 1e-4:
                break

    return best_a, best_val


# ----------------------------------------------------------------------
# Mass‑transfer hill‑climbing (incremental convolution update)
# ----------------------------------------------------------------------
def _mass_transfer_opt(a_start: np.ndarray,
                       cur_max: float,
                       n: int,
                       deadline: float,
                       max_iters: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Perform many tiny moves of mass from indices that heavily contribute
    to the current maximal convolution entry to the smallest entry.
    Returns the best vector seen and its surrogate objective.
    """
    rng = np.random.default_rng()          # local RNG
    a = a_start.copy()
    conv = _conv_fft(a, a)                # full convolution, length 2n‑1
    cur_M = cur_max
    best_a = a.copy()
    best_val = 2.0 * n * cur_M

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # --- locate the peak of the convolution -----------------------
        k = int(conv.argmax())

        # indices that actually participate in the peak
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
        idx_j = k - idx_i                     # counterpart indices

        # product a[i] * a[j] gives the contribution of each pair
        contrib = a[idx_i] * a[idx_j]
        src = idx_i[np.argmax(contrib)]       # source index (largest product)

        # target is the index with the smallest current weight
        tgt = int(np.argmin(a))
        if tgt == src:
            # pick the second smallest to avoid a zero‑move
            sorted_idx = np.argsort(a)
            for cand in sorted_idx:
                if cand != src:
                    tgt = int(cand)
                    break

        # step size – decays with the square‑root of the iteration count
        step_factor = 0.2 / math.sqrt(it)
        d = a[src] * step_factor
        # make sure we never remove more mass than available
        d = min(d, a[src] * 0.5)
        if d <= 0.0:
            continue

        # ----- incremental update ------------------------------------
        a_before = a.copy()
        conv_before = conv.copy()

        a[src] -= d
        a[tgt] += d

        # use the *old* a for the convolution update (a_before)
        conv[src:src + n] -= 2.0 * d * a_before
        conv[tgt:tgt + n] += 2.0 * d * a_before
        conv[src + src] += d * d
        conv[tgt + tgt] += d * d
        conv[src + tgt] -= 2.0 * d * d

        new_M = conv.max()
        if new_M < cur_M - 1e-15:               # accept only strict improvements
            cur_M = new_M
            val = 2.0 * n * cur_M
            if val < best_val:
                best_val = val
                best_a = a.copy()
                if it % 250_000 == 0:
                    print(f"[MassTransfer] it={it:7d} best={best_val:.6f}")

        else:
            # revert the move
            a = a_before
            conv = conv_before

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a small safety margin

    # Problem size – 4000 works well within the budget
    n = 4000

    # ------------------------------------------------------------------
    # 0. Load previous solution if it exists (and has the right size)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                a = arr / arr.sum()
                print("[Init] loaded previous solution")
            else:
                raise ValueError
        except Exception:
            a = _beta_shape(n, 2.0, 2.0)
            print("[Init] previous solution invalid – using β‑shape")
    else:
        a = _beta_shape(n, 2.0, 2.0)
        print("[Init] no previous solution – using β‑shape")

    # ------------------------------------------------------------------
    # 1. Smooth L‑BFGS optimisation
    # ------------------------------------------------------------------
    try:
        a_lb = _lbfgs_smooth_optimize(a, n, deadline,
                                      max_iter_per_beta=6000)
        conv_lb = _conv_fft(a_lb, a_lb)
        M_lb = np.max(conv_lb)
        val_lb = 2.0 * n * M_lb
        print(f"[L‑BFGS] surrogate = {val_lb:.6f}")

        if val_lb < _objective_surrogate(a):
            a = a_lb
    except Exception as exc:  # pragma: no cover
        print("[L‑BFGS] failed:", exc)

    # ------------------------------------------------------------------
    # 2. LP‑based refinement (a handful of iterations)
    # ------------------------------------------------------------------
    conv = _conv_fft(a, a)
    M = np.max(conv)
    if time.time() < deadline:
        a, val = _lp_refinement(a, M, n, deadline,
                                max_iters=6, step=0.2)
        conv = _conv_fft(a, a)
        M = np.max(conv)
        print(f"[After LP] surrogate = {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Mass‑transfer hill‑climbing (long polishing phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(3_000_000, remaining * 15000))  # scale with time left
        a_best, best_val = _mass_transfer_opt(a, M, n,
                                              deadline,
                                              max_iters=max_iters)
        print(f"[MassTransfer] final surrogate = {best_val:.6f}")
    else:
        a_best = a
        best_val = 2.0 * n * M

    # ------------------------------------------------------------------
    # 4. Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(a_best.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store globally for possible warm‑starts in subsequent runs
    globals()["height_sequence_1"] = a_best.tolist()
    return a_best.tolist()
