def _simplex_proj(v):
    # Project onto simplex {x >= 0, sum(x) = 1}
    # v is numpy array
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    cond = u - cssv / ind > 0
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _pgd_optimize(start_a, deadline, max_iters, eta0=0.5, eval_interval=5000):
    a = start_a.copy()
    best_a = a.copy()
    n = len(a)
    # compute current conv and val
    conv = _conv_fft(a, a)
    max_b = np.max(conv)
    best_val = 2.0 * n * max_b / (np.sum(a) ** 2)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # find idx of max
        idx = int(np.argmax(conv))
        # compute gradient
        n = len(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end+1)
        rev = idx - i_range
        g = np.zeros_like(a)
        g[i_range] = 2.0 * a[rev]
        # step size
        eta = eta0 / math.sqrt(it)
        a_tmp = a - eta * g
        a = _simplex_proj(a_tmp)
        # recompute conv for next iteration
        conv = _conv_fft(a, a)
        if it % eval_interval == 0:
            max_b = np.max(conv)
            val = 2.0 * n * max_b / (np.sum(a) ** 2)
            if val < best_val:
                best_val = val
                best_a = a.copy()
    # final check
    max_b = np.max(conv)
    val = 2.0 * n * max_b / (np.sum(a) ** 2)
    if val < best_val:
        best_val = val
        best_a = a.copy()
    return best_a, best_val

"""
Propose a non‑negative step function (length n) that minimizes

        2·n·max(conv(a,a)) / (sum(a)²)

The algorithm uses a mixture of deterministic analytic seeds (including a
Beta(0.5,0.5) “arcsine” shape), a simple projected‑gradient descent on the
exact max‑convolution objective, and a final high‑β L‑BFGS polishing step.
Projected‑gradient descent works on the probability simplex via Euclidean
projection, using the sub‑gradient of the active convolution lag.
The whole search respects the 1000 s time budget and stores the best
sequence in the global name ``height_sequence_1`` for warm‑starts.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Analytic shape families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on {1,…,n} (U‑shape when α,β<1)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, rng: np.random.Generator,
                   k_min: int = 30, k_max: int = 150) -> np.ndarray:
    """Sparse support: choose K random indices, give them random weights."""
    K = int(rng.integers(k_min, k_max + 1))
    idx = rng.choice(n, size=K, replace=False)
    a = np.zeros(n, dtype=np.float64)
    weights = rng.exponential(scale=1.0, size=K)
    a[idx] = weights
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α)+b·α then ℓ₁‑renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Exact objective utilities
# ----------------------------------------------------------------------
def _objective_raw(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) / (sum²)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (using the active lag).
    Returns g = ∂ max(conv) / ∂ a .
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    # indices that overlap for this lag
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g = np.zeros_like(a)
    g[i_range] = 2.0 * a[rev]
    return g


def _simplex_proj(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Algorithm from Duchi et al. (2008)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _pgd_optimize(start_a: np.ndarray,
                  deadline: float,
                  max_iters: int,
                  eta0: float = 0.5,
                  eval_interval: int = 5000) -> tuple[np.ndarray, float]:
    """
    Projected‑gradient descent on the exact objective.
    Uses step‑size η_t = η0 / sqrt(t) and Euclidean projection onto the simplex.
    Returns the best vector found and its objective value.
    """
    a = start_a.copy()
    n = len(a)
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv) / (np.sum(a) ** 2)
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        idx = int(np.argmax(conv))
        # build sub‑gradient for the active lag
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev = idx - i_range
        g = np.zeros_like(a)
        g[i_range] = 2.0 * a[rev]

        eta = eta0 / math.sqrt(it)
        a_tmp = a - eta * g
        a = _simplex_proj(a_tmp)

        # recompute convolution for the next iteration
        conv = _conv_fft(a, a)

        if it % eval_interval == 0:
            cur_val = 2.0 * n * np.max(conv) / (np.sum(a) ** 2)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[PGD] iter={it:,} new best={best_val:.6f}")

    # final check
    final_val = 2.0 * n * np.max(conv) / (np.sum(a) ** 2)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    return e / s


def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Return (value, gradient) for the smooth max‑convolution surrogate."""
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)              # full auto‑convolution
    shift = np.max(conv)                # numerical stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # weights for each lag
    w = exp_term / sum_exp
    # convolution of reversed a with the weights gives ∂obj/∂a
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 1500) -> np.ndarray:
    """
    L‑BFGS‑B optimisation of the smooth surrogate.
    Variable = logits of a softmax (unconstrained real vector).
    """
    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0,
                         5000.0, 10000.0, 20000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during beta={beta_val}")
            break
        except Exception as e:               # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)
