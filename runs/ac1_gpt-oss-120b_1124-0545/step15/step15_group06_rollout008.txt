def propose_candidate(seed=..., budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    n = int(kwargs.get('n', 4000))  # we could increase maybe 5000
    # generate seeds
    seeds = []
    # analytic families with fine grid
    seeds.append(_uniform(n))
    # ...
    # power p grid
    for p in np.arange(0.0, 5.01, 0.1):
        seeds.append(_power(n, float(p)))
    # sinus gamma grid
    for g in np.arange(0.1, 1.51, 0.1):
        seeds.append(_sine_window(n, gamma=float(g)))
    # Gaussian sigma
    for s_frac in np.arange(0.05, 0.51, 0.02):
        seeds.append(_gaussian(n, sigma_frac=float(s_frac)))
    # Beta shape grid
    for a_val in np.arange(0.1, 2.01, 0.15):
        for b_val in np.arange(0.1, 2.01, 0.15):
            seeds.append(_beta_shape(n, a_val, b_val))
    # Tukey
    for al in np.arange(0.1,0.91,0.1):
        seeds.append(_tukey_window(n, al))
    # random Dirichlet low conc
    for conc in [0.01, 0.02, 0.05]:
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # also include previous best if available
    previous = globals().get('height_sequence_1')
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape[0]==n:
                seeds.append(arr/arr.sum())
        except Exception:
            pass

    # Evaluate all seeds surrogate
    def surrogate(arr):
        conv = _conv_fft(arr, arr)
        maxb = np.max(conv)
        return 2.0 * n * maxb  # sum=1

    best_seq = None
    best_val = float('inf')
    for vec in seeds:
        if time.time()>deadline:
            break
        v_norm = np.asarray(vec, dtype=np.float64)
        v_norm = np.maximum(v_norm, 0)
        s = v_norm.sum()
        if s==0:

        v_norm = v_norm / s
        val = surrogate(v_norm)
        if val < best_val:
            best_val = val
            best_seq = v_norm
    print(f"Best seed surrogate: {best_val:.6f}")

    # Stage L-BFGS from best seed
    if best_seq is None:
        best_seq = _uniform(n)
    # L-BFGS schedule
    beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                     1000.0, 2000.0, 5000.0, 10000.0, 20000.0]
    seq_opt = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                     beta_schedule=beta_schedule,
                                     max_iter_per_beta=2000)
    val_opt = surrogate(seq_opt)
    print(f"L-BFGS refined surrogate: {val_opt:.6f}")
    if val_opt < best_val:
        best_val = val_opt
        best_seq = seq_opt

    # Stage Adam optimization
    # Use high beta e.g., 20000
    beta = 20000.0
    v = np.log(best_seq + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    lr = 0.2  # learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    while time.time() < deadline:

"""
search_optimiser.py

A self‑contained optimiser for the “step‑function” challenge.

The code explores a large family of analytic shapes (uniform, power‑law,
beta‑U, Gaussian, sine‑window, Tukey, triangular, arcsine) on a fine
grid of parameters, keeps the most promising candidates and refines them
with a continuation L‑BFGS optimiser on a smooth log‑sum‑exp surrogate.
Afterwards a short Adam minimise on the same surrogate and a few
perturb‑and‑polish attempts are performed.  The best vector found is
stored in the global name ``height_sequence_1`` and also returned as a
plain list of floats.

All constraints required by ``evaluate_sequence`` (non‑negative,
finite, normalised) are enforced.
"""

from __future__ import annotations
import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (O(n log n)).
    Returned length is len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax that maps ℝⁿ → the probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:                     # should never happen
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """All entries equal = 1/n."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p} (p≥0)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta distribution on {1,…,n}.  α,β>0.  U‑shapes when <1."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred at the centre, σ = sigma_frac·n."""
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    """Sin(π·(i+1)/(n+1)) raised to γ (γ>0)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Standard Tukey (cosine‑taper) window."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    return a / s


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular bump (peak in centre)."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete arcsine law a_i ∝ 1/sqrt((i+1)*(n-i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 1) * (n - i))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by L‑BFGS / Adam
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Returns (value, grad) of
        2·n·log‑sum‑exp_β(conv(a,a)),
    where a = softmax(v) (so a lies on the simplex).
    The gradient is w.r.t. the logits v.
    """
    a = _softmax(v)                     # probability vector
    conv = _conv_fft(a, a)              # full auto‑convolution
    shift = np.max(conv)                # shift for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # derivative of the log‑sum‑exp wrt a
    w = exp_term / sum_exp               # weights summing to 1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain‑rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    L‑BFGS‑B on the smooth surrogate.
    Variable = logits of a softmax (unconstrained).
    """
    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0,
                         5000.0, 10000.0, 20000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(_):
            # early stop if time is exceeded
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] time limit reached during beta={beta_val}")
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam optimiser (still on the smooth surrogate)
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   beta: float = 20000.0,
                   lr: float = 0.15,
                   max_iters: int = 100_000) -> np.ndarray:
    """
    Adam on the log‑sum‑exp surrogate.
    Returns the best vector encountered.
    """
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    best_a = _softmax(v)
    best_val = 2.0 * n * np.max(_conv_fft(best_a, best_a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        obj, grad = _smooth_objective(v, n, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

        if it % 250 == 0:          # occasional cheap check
            a = _softmax(v)
            cur = 2.0 * n * np.max(_conv_fft(a, a))
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[Adam] it={it:,} improved surrogate to {best_val:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Simple surrogate evaluator (exact max‑convolution because sum=1)
# ----------------------------------------------------------------------
def _surrogate_val(a: np.ndarray, n: int) -> float:
    """2·n·max(conv(a,a)) – works for vectors already ℓ₁‑normalised."""
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed construction (dense grid + randomised)
# ----------------------------------------------------------------------
def _collect_seed_pool(n: int, rng: np.random.Generator,
                       previous: np.ndarray | None) -> list[np.ndarray]:
    """Generate a diverse pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # Deterministic families (coarse grid)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_arcsine_shape(n))

    # Power‑law sweep
    for p in np.linspace(0.0, 5.0, 26):          # step ≈0.2
        seeds.append(_power(n, float(p)))

    # Sine‑window sweep
    for g in np.linspace(0.2, 1.8, 17):
        seeds.append(_sine_window(n, gamma=float(g)))

    # Gaussian sigma sweep
    for sf in np.linspace(0.05, 0.45, 17):
        seeds.append(_gaussian(n, sigma_frac=float(sf)))

    # Beta‑shape sweep (U‑shapes and moderate shapes)
    al_vals = np.linspace(0.1, 1.5, 15)
    be_vals = np.linspace(0.1, 1.5, 15)
    for a in al_vals:
        for b in be_vals:
            seeds.append(_beta_shape(n, float(a), float(b)))

    # Tukey sweep
    for al in np.linspace(0.1, 0.9, 17):
        seeds.append(_tukey_window(n, alpha=float(al)))

    # Random Dirichlet draws (very low concentration → sparse)
    for conc in (0.005, 0.01, 0.02, 0.05):
        for _ in range(6):
            v = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            seeds.append(v)

    # Random shuffled Dirichlet (breaks any ordering structure)
    for _ in range(6):
        base = rng.dirichlet(np.full(n, 0.1)).astype(np.float64)
        rng.shuffle(base)
        seeds.append(base)

    # Previous best (if it matches current length)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Clean‑up: enforce non‑negativity, ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

    # Problem size (may be overridden via kwargs)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 4000))
    print(f"\n=== Optimisation start (n={n}) ===")

    # ------------------------------------------------------------------
    # Stage 0 – seed generation & cheap screening
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seed_pool = _collect_seed_pool(n, rng, prev_arr)
    print(f"Generated {len(seed_pool)} seed vectors (including previous best).")

    best_seq = None
    best_val = float("inf")

    # cheap surrogate = 2·n·max(conv) (sum is 1 by construction)
    for idx, vec in enumerate(seed_pool):
        if time.time() > deadline:
            break
        val = _surrogate_val(vec, n)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
            # occasional progress report
            if idx % 100 == 0:
                print(f"[Seed] # {idx:04d} new best surrogate = {best_val:.6f}")

    print(f"\nBest seed surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on the smooth surrogate
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)

    beta_sched = [0.5, 1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1000.0, 2000.0, 5000.0, 10000.0, 20000.0]

    # give a generous per‑β budget (≈ 3000 iterations) while respecting time
    per_beta = max(1500,
                   int(min(5000,
                           (deadline - time.time()) * 0.03 * 1e5)))
    print(f"\n--- L‑BFGS continuation (per‑β max iter = {per_beta}) ---")
    seq_lbfgs = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                      beta_schedule=beta_sched,
                                      max_iter_per_beta=per_beta)
    val_lbfgs = _surrogate_val(seq_lbfgs, n)
    print(f"[L‑BFGS] surrogate = {val_lbfgs:.6f}")

    if val_lbfgs < best_val:
        best_val = val_lbfgs
        best_seq = seq_lbfgs.copy()
        print(f"[Update] new global best after L‑BFGS = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation on high‑β smooth surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam optimisation (high‑β surrogate) ---")
        # Adam works directly on the probability vector (via logits)
        seq_adam = _adam_optimize(best_seq, n, deadline,
                                  beta=20000.0, lr=0.12,
                                  max_iters=120_000)
        val_adam = _surrogate_val(seq_adam, n)
        print(f"[Adam] surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam.copy()
            print(f"[Update] new best after Adam = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – stochastic perturb‑and‑polish (short L‑BFGS runs)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        attempts = 60
        print("\n--- Perturb‑and‑polish phase ---")
        for a in range(attempts):
            if time.time() > deadline:
                break
            # jitter logits
            logits = np.log(best_seq + 1e-30)
            scale = max(0.02, 0.6 / (a + 1))   # rapidly decreasing jitter
            jitter = rng.normal(scale=scale, size=n)
            start_vec = _softmax(logits + jitter)

            # short L‑BFGS with a reduced beta schedule
            short_seq = _lbfgs_smooth_optimize(start_vec, n, deadline,
                                               beta_schedule=[1.0, 20.0, 200.0],
                                               max_iter_per_beta=1200)
            cand_val = _surrogate_val(short_seq, n)
            if cand_val < best_val:
                best_val = cand_val
                best_seq = short_seq.copy()
                print(f"[Perturb] improvement (attempt {a+1}) to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final high‑β L‑BFGS polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final high‑β L‑BFGS polishing ---")
        final_seq = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                           beta_schedule=[5000.0, 10000.0, 20000.0],
                                           max_iter_per_beta=3000)
        final_val = _surrogate_val(final_seq, n)
        print(f"[Final] surrogate = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq.copy()
            print(f"[Update] new best after final polish = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (for reporting purposes only)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:
        # never raise – the competition harness will handle the return value
        print("Warning: evaluate_sequence raised", exc)

    # store for warm‑start on the next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
