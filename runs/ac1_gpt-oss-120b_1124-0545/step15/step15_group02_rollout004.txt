"""
Optimiser for the step–function anti‑concentration problem.

The algorithm proceeds in several stages inside the 1000 s wall‑clock limit:

1. **Seed pool** – a small collection of hand‑crafted families
   (uniform, exponential, beta‑shapes) together with a few
   Dirichlet draws and the previous best solution (if present).

2. **Block‑wise differential evolution** – the vector is represented by
   a small number *k* (8…16) of block‑masses.  A cheap global optimiser
   (`scipy.optimize.differential_evolution`) searches the low‑dimensional
   space for a promising shape, which is then expanded to a full
   4000‑length sequence and normalised.

3. **Adam on a smooth surrogate** – the sequence is re‑parameterised
   by logits `v` (soft‑max → probabilities).  A smoothed version of the
   max–convolution (log‑sum‑exp with an increasing β) is minimised with
   Adam (first‑ and second‑moment estimates).  The β‑schedule starts
   small (≈1) and is increased geometrically, allowing the optimiser to
   follow a smooth landscape at first and then concentrate on the true
   objective.

4. **Exponentiated‑gradient polishing** – a classic EG (mirror descent)
   run on the exact objective finishes the search.

All stages stop as soon as the global deadline is reached.  The best
sequence found is stored in the global variable ``height_sequence_1`` and
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for differential evolution)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective for an ℓ₁‑normalised vector:
    2·n·max(conv(seq,seq)).
    The sequence must already be normalised to sum = 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_family(n: int, c: float) -> np.ndarray:
    """ a_i ∝ exp(-c·i/n) """
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """ a_i ∝ exp(-c·(n-1-i)/n) """
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Block‑wise representation utilities
# ----------------------------------------------------------------------
def _block_sizes(total_len: int, k: int) -> list[int]:
    """Return a list of block sizes for a given total length and block count."""
    base = total_len // k
    rem = total_len % k
    sizes = [base + 1] * rem + [base] * (k - rem)
    return sizes


def _build_seq_from_blocks(weights: np.ndarray, block_sizes: list[int]) -> np.ndarray:
    """
    Expand a weight vector (total mass per block) into a full length
    ℓ₁‑normalised sequence by giving each element in a block an equal share.
    """
    n = sum(block_sizes)
    seq = np.empty(n, dtype=np.float64)
    pos = 0
    for w, sz in zip(weights, block_sizes):
        # distribute the block mass uniformly across its elements
        seq[pos:pos + sz] = w / float(sz)
        pos += sz
    total = seq.sum()
    if total <= 0:
        return _uniform(n)
    return seq / total


# ----------------------------------------------------------------------
# Optimiser components
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              beta_schedule=None) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution surrogate.
    The parameterisation is `logits → softmax → probabilities`.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0,
                         200.0, 500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0]

    v = np.log(start_seq + 1e-30)          # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    cur_beta_idx = 0
    cur_beta = beta_schedule[cur_beta_idx]
    it_per_beta = max(10_000, int(0.02 * n))   # switch β after a few‑thousand steps

    while time.time() < deadline:
        t += 1
        if t % it_per_beta == 0 and cur_beta_idx + 1 < len(beta_schedule):
            cur_beta_idx += 1
            cur_beta = beta_schedule[cur_beta_idx]

        a = _softmax(v)                     # ℓ₁‑normalised probabilities
        conv = _conv_fft(a, a)
        shift = np.max(conv)

        # soft‑max over β·(conv – shift)
        exp_beta = np.exp(cur_beta * (conv - shift))
        w = exp_beta / np.sum(exp_beta)

        # gradient w.r.t. a  (see derivation in the L‑BFGS section)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 2n‑1
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        step = m_hat / (np.sqrt(v2_hat) + eps)
        lr = 0.5
        v = v - lr * step

        # keep track of the best surrogate value seen
        cur_seq = _softmax(v)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact objective.
    The sequence is kept on the probability simplex via the soft‑max
    of a log‑vector.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (same pattern as in the
        # earlier reference implementation)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.6 / math.sqrt(it)          # decreasing step‑size
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

                     budget_s: int = 1000,

    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within a 1000 s wall‑clock budget.
    The search combines cheap analytic seeds, a block‑wise global
    optimisation (DE), Adam on a smooth surrogate and a final
    exponentiated‑gradient polishing.

    rng = np.random.default_rng(seed)
    start_time = time.time()

    betas_ = np.linspace(0.1, 5.0, 7)
