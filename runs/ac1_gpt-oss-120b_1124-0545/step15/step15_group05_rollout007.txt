Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

within a 1000 s budget.

Algorithm
---------
1. **Problem size** – fix n=4000 (a good trade‑off between expressiveness
   and speed).  The sequence is kept ℓ₁‑normalised, so the denominator is 1
   and the objective reduces to `2·n·max(conv)`.
2. **Rich seed pool** – generate a large collection of ℓ₁‑normalised vectors
   from deterministic windows, wide families (Beta, power‑law, exponential,
   Gaussian, Kaiser, Blackman, Hamming, Tukey, triangular‑power, cosine‑power,
   log‑decay, random Dirichlet, low‑frequency Fourier curves) and a few
   random mixtures.  The previous best solution (if any) is also added.
3. **Cheap surrogate ranking** – evaluate the cheap surrogate
   `S(a)=2·n·max(conv(a,a))` with an FFT‑based convolution and keep the best
   `k_elite=5` seeds.
4. **Continuation Adam on elites** – for each elite seed run Adam on a
   *smooth‑max* surrogate.  The smooth‑max uses a schedule of β values that
   grows up to 200 000, making the surrogate arbitrarily close to the true
   max.  Many optimisation steps are performed (the exact number is
   allocated from the remaining time budget).
5. **Multi‑start exact sub‑gradient (exponentiated‑gradient)** – starting
   from the best Adam result (and optionally the other elites) run a
   long mirror‑descent on the *exact* max‑convolution.  The update uses the
   genuine sub‑gradient of the active constraints; a more aggressive step‑

   (the 5 largest convolution entries) is used to smooth the direction.
6. **Polishing** – a handful of short Adam runs from jittered versions of
   the current best vector are performed to escape shallow local minima.
7. **Return** – the best sequence discovered is stored in the global

shrinks the allocated work‑load accordingly, guaranteeing a result before

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (always ℓ₁‑normalised, non‑negative)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)).
    All candidate vectors are kept ℓ₁‑normalised, therefore this equals the
    true objective of ``evaluate_sequence``.
    """
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic shape families (all return ℓ₁‑normalised vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _triangular_pow(n: int, p: float) -> np.ndarray:
    """Triangular shape raised to a power p ≥ 0."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.power(np.maximum(a, 0.0), p)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _cosine_pow(n: int, p: float) -> np.ndarray:
    """Cosine‑raised shape a_i ∝ (cos(π·i/(n‑1)) + 1)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (np.cos(np.pi * i / (n - 1)) + 1.0) ** p
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1 / (i+1)^p  (p ≥ 0)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p  (p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _log_decay(n: int, base: float = 1.5, offset: float = 1.0) -> np.ndarray:
    """a_i ∝ 1 / (offset + i)^log_base."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / ((offset + i) ** np.log(base))
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window, β≥0 (β=0 yields a rectangular window)."""
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    w = i0(beta * np.sqrt(1.0 - t * t)) / i0(beta)
    return w / w.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.16
    w = (1 - a) / 2 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + a / 2 * np.cos(4 * np.pi * i / (n - 1))
    return w / w.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return w / w.sum()


def _tukey_window(n: int, alpha: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1)
    w = np.ones_like(i)
    mask1 = i < alpha / 2
    mask2 = i > 1 - alpha / 2
    w[mask1] = 0.5 * (1 + np.cos(np.pi * (2 * i[mask1] / alpha - 1)))
    w[mask2] = 0.5 * (1 + np.cos(np.pi * (2 * (i[mask2] - 1) / alpha + 1)))
    return w / w.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear interpolation between two shapes."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """A smooth random curve built from a few low‑frequency cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()

    seeds.append(_power(n, p=1.0))
    seeds.append(_power(n, p=2.0))

# Continuation Adam optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    steps_per_beta: int = 2000,
) -> np.ndarray:
    """
    Adam optimiser on the smoothed max‑convolution.
    Returns a ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        # very aggressive schedule, ending well beyond the earlier 20 000
        beta_schedule = [
            1.0,
            5.0,
            20.0,
            80.0,
            200.0,
            500.0,
            1000.0,
            3000.0,
            5000.0,
            10000.0,
            20000.0,
            50000.0,
            100000.0,
        ]

    eps = 1e-30
    logits = np.log(start_vec + eps)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.08
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    t = 0

    best_seq = None
    best_val = np.inf

    for beta in beta_schedule:
        beta_val = float(beta)
        max_iters = steps_per_beta
        for _ in range(max_iters):
            t += 1
            if time.time() > deadline:
                return _softmax(logits)

            a = _softmax(logits)                     # current distribution
            conv = _conv_fft(a, a)
            shift = np.max(conv)                     # numeric stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            w = exp_beta / exp_beta.sum()            # smooth‑max weights (size 2n‑1)

            # gradient of smooth‑max w.r.t. a:
            #   d/da  Σ w_k conv_k = Σ w_k * 2 * (rev_a * δ_k) convolution
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length 3n‑2
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            # back‑prop through the soft‑max
            dot = np.dot(grad_a, a)
            grad_logits = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_logits
            v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # occasional best‑track
            if t % 500 == 0:
                cur_val = _objective_surrogate(a, n)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
        # optional slight LR decay (better stability for later high β)
        lr *= 0.97

    return best_seq if best_seq is not None else _softmax(logits)

# Exact sub‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize_k(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
    top_k: int = 5,
    step_const: float = 2.0,
) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Uses an aggregation of the `top_k` largest convolution entries to smooth
    the sub‑gradient.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        # indices of the `top_k` largest entries (might contain ties)
        if top_k == 1:
            idxs = np.array([int(np.argmax(conv))], dtype=int)
        else:
            # np.argpartition is O(n); we then take the exact values later
            part = np.argpartition(conv, -top_k)[-top_k:]
            idxs = part

        # aggregate gradient over the selected indices
        g = np.zeros_like(a)
        for idx in idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = idx - idx_i
            # contribution of this active constraint
            g[idx_i] += 2.0 * a[rev_idx]

        # normalise the aggregated direction (avoid overly large steps)
        g = g / max(1.0, len(idxs))

        eta = step_const / np.sqrt(it)   # more aggressive than the default 1.0
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[ExpGradK] it={it:,} best≈{cur_val:.6f}")

    return best_seq

def _polish_with_jitter(
    base_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    attempts: int = 4,
) -> np.ndarray:
    """
    Perform a few short Adam runs starting from jittered logits.
    Keeps any improvement found.
    """
    best_seq = base_seq.copy()
    best_val = _objective_surrogate(best_seq, n)

    for k in range(attempts):
        if time.time() > deadline:
            break
        logits = np.log(best_seq + 1e-30)
        jitter = rng.normal(scale=0.025, size=n)   # modest noise
        logits_j = logits + jitter

        refined = _adam_smooth_optimize(
            start_vec=_softmax(logits_j),
            n=n,
            deadline=deadline,
            beta_schedule=[20000.0, 50000.0, 100000.0, 200000.0],
            steps_per_beta=1200,
        )
        cur_val = _objective_surrogate(refined, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = refined
            print(f"[Polish] improvement after jitter #{k+1}: {cur_val:.6f}")

    return best_seq

            beta_schedule=beta_sched,
            steps_per_beta=steps_per_beta,
        )
        val_opt = _objective_surrogate(seq_opt, n)
        print(f"[Adam] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – exact sub‑gradient refinement (multi‑start)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate ~70 % of the remaining budget to sub‑gradient optimisation
        max_total_iters = int(min(4_000_000, remaining * 0.7 * 1e4))
        # split the budget roughly equally among the top 3 elite seeds
        starts = [global_best_seq] + top_seeds[1:3]
        per_start_iters = max_total_iters // len(starts)
        print("\n--- Exact sub‑gradient (exp‑gradient) refinement ---")
        for i, start in enumerate(starts):
            print(f"[ExpGrad] start #{i+1} with {per_start_iters:,} iters")
            seq_exp = _exp_gradient_optimize_k(
                start_seq=start,
                n=n,
                deadline=deadline,
                max_iters=per_start_iters,
                top_k=5,
                step_const=2.0,
            )
            val_exp = _objective_surrogate(seq_exp, n)
            print(f"[ExpGrad] result surrogate = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] New global best after ExpGrad = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – random‑restart polishing (short Adam runs)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Polishing with jittered Adam runs ---")
        polished = _polish_with_jitter(
            global_best_seq, n, deadline, rng, attempts=5
        )
        val_polish = _objective_surrogate(polished, n)
        if val_polish < global_best_val:
            global_best_seq = polished.copy()
            global_best_val = val_polish
            print(f"[Polish] improved surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
