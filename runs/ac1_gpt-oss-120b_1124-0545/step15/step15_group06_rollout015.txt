# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation of the algorithm from
    # "Efficient Projections onto the l1‑Ball for Learning in High Dimensions"
    # (Duchi et al. 2008)
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on {1,…,n}.  α,β<1 give a U‑shape."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i) , ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, rng: np.random.Generator,
                   k_min: int = 30, k_max: int = 150) -> np.ndarray:
    """Sparse support: choose K random indices, give them random weights."""
    K = int(rng.integers(k_min, k_max + 1))
    idx = rng.choice(n, size=K, replace=False)
    a = np.zeros(n, dtype=np.float64)
    weights = rng.exponential(scale=1.0, size=K)
    a[idx] = weights
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _permute(v: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a random permutation of v (preserve ℓ₁‑norm)."""
    perm = rng.permutation(len(v))
    return v[perm]


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α)+b·α then ℓ₁‑renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Surrogate objective (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Return (value, gradient) for the smooth max‑convolution surrogate."""
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)              # full auto‑convolution
    shift = np.max(conv)                # numerical stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # weights for each lag
    w = exp_term / sum_exp
    # convolution of reversed a with the weights gives ∂obj/∂a
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    L‑BFGS‑B optimisation of the smooth surrogate.
    Variable = logits of a softmax (unconstrained real vector).
    """
    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0,
                         5000.0, 10000.0, 20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0, 1_000_000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:               # pragma: no cover
            break

    return _softmax(v)

def _objective_raw(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) / (sum²).  Assumes sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (using the active lag)."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


def _exp_gradient_optimize(start_a: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 500_000,
                          eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_raw(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        g = _subgradient_maxconv(a)
        eta = eta0 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 100_000 == 0:
            cur = _objective_raw(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_raw(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Build a diverse pool of ℓ₁‑normalised seed vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12, 20):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.08, 0.45, 9):
        seeds.append(_gaussian(n, sigma))

    # Power‑law – both decreasing and its reverse (increasing)
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, -p))  # reverse order via negative exponent

    # Beta families – include very sharp U‑shapes
    al_vals = (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    be_vals = (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    for a in al_vals:
        for b in be_vals:
            seeds.append(_beta_shape(n, a, b))

    # Exponential decays (different λ)
    for lam in (0.0002, 0.0004, 0.0006, 0.0008, 0.001, 0.002):
        seeds.append(_exp_decay(n, lam))

    # Random Dirichlet draws – very low concentration
    for conc in (0.001, 0.002, 0.005, 0.01, 0.02, 0.05):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Sparse random supports
    for _ in range(40):
        seeds.append(_random_sparse(n, rng, k_min=30, k_max=180))

    # Random mixtures of existing seeds
    base_idx = np.arange(len(seeds))
    for _ in range(300):
        i, j = rng.choice(base_idx, size=2, replace=False)
        alpha = rng.random()
        mix = _mix_shapes(seeds[i], seeds[j], alpha)
        seeds.append(mix)

    # Permutations of a subset of seeds (adds combinatorial diversity)
    perm_subset = seeds[:200]  # take a slice to keep work reasonable
    for vec in perm_subset:
        for _ in range(3):
            seeds.append(_permute(vec, rng))

    # Previous best (if any) ---------------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final clean‑up – ensure non‑negativity & ℓ₁‑normalisation -------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    ``evaluate_sequence`` under the allocated time budget.  Returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0        # safety margin

    n = int(kwargs.get("n", 4000))
    print(f"\n=== Optimisation start (n={n}) ===")

    # -----------------------------------------------------------------------
    # Stage 0 – generate a huge pool of seeds and keep the best few
    # -----------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors (including permutations).")

    # cheap exact objective (sum already = 1)
    evaluated = [(vec, _objective_raw(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12                      # a modest number – enough for later polishing
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best raw objective among seeds (top {top_k}):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # -----------------------------------------------------------------------
    # Stage 1 – L‑BFGS continuation on the smooth surrogate (high‑β schedule)
    # -----------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_raw(global_best_seq)
    print(f"\n[Init] exact objective = {global_best_val:.6f}")

    # aggressive β‑schedule ending at 1e6
    beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0,
                     200.0, 500.0, 1000.0, 2000.0,
                     5000.0, 10000.0, 20000.0, 50000.0,
                     100000.0, 200000.0, 500000.0, 1_000_000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS runs.")
            break

        # allocate a per‑β iteration budget based on remaining time
        remaining = deadline - time.time()
        per_beta_iters = max(1000,
                             int(min(6000,
                                     remaining * 0.05 * 1e5)))
        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter/beta={per_beta_iters}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=per_beta_iters)
        val_opt = _objective_raw(seq_opt)
        print(f"[L‑BFGS] exact objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best exact = {global_best_val:.6f}")

    # -----------------------------------------------------------------------
    # Stage 2 – mirror‑descent (exponentiated‑gradient) on the exact objective
    # -----------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(2_500_000,
                                remaining * 0.30 * 1e5))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient refinement (exact max) ---")
            seq_exp, val_exp = _exp_gradient_optimize(global_best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=max_exp_iters,
                                                      eta0=1.0)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved exact objective to {global_best_val:.6f}")

    # -----------------------------------------------------------------------
    # Stage 3 – stochastic perturb‑and‑polish (mini‑L‑BFGS runs)
    # -----------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Stochastic perturb‑and‑polish phase ---")
        attempts = 30
        for a in range(attempts):
            if time.time() > deadline:
                break
            logits = np.log(global_best_seq + 1e-30)
            scale = max(0.04, 0.8 / (a + 1))   # decreasing jitter
            jitter = rng.normal(scale=scale, size=n)
            start_logits = logits + jitter
            start_vec = _softmax(start_logits)

            seq_candidate = _lbfgs_smooth_optimize(start_vec,
                                                   n,
                                                   deadline,
                                                   beta_schedule=[1.0, 20.0, 200.0, 2000.0],
                                                   max_iter_per_beta=1200)
            cand_val = _objective_raw(seq_candidate)
            if cand_val < global_best_val:
                global_best_seq = seq_candidate.copy()
                global_best_val = cand_val
                print(f"[Perturb] improvement after attempt {a+1}: {global_best_val:.6f}")

    # -----------------------------------------------------------------------
    # Stage 4 – final high‑β polishing with generous iterations
    # -----------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta_iters = max(800,
                             int(min(5000,
                                     remaining * 0.10 * 1e5)))
        print("\n--- Final high‑β L‑BFGS polishing ---")
        seq_final = _lbfgs_smooth_optimize(global_best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=[200_000.0, 500_000.0, 1_000_000.0],
                                           max_iter_per_beta=per_beta_iters)
        val_final = _objective_raw(seq_final)
        print(f"[Final‑L‑BFGS] exact objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after polishing = {global_best_val:.6f}")

    # -----------------------------------------------------------------------
    # Final evaluation using the official evaluator
    # -----------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest exact ≈ {global_best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
