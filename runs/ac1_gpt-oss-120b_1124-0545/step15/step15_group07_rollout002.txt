"""
Search for a non‑negative step‑function (vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

using a multi‑stage optimisation:
1. generate a large, diverse pool of ℓ₁‑normalised seeds;
2. keep the best few by a cheap surrogate (sum=1 ⇒ objective = 2·n·max(conv));
3. refine each elite seed with a continuation L‑BFGS optimiser on a smooth
   log‑sum‑exp approximation of the max;
4. run a long exact exponentiated‑gradient (mirror‑descent) optimisation;
5. run a projected‑gradient descent on the exact non‑smooth objective;
6. finish with a simulated‑annealing hill‑climber;
7. return the best vector found (also stored as `height_sequence_1`).

The code respects the time budget (default 1000 s) and contains no
nested closures – all helper objects are top‑level classes/functions.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of `v` onto the probability simplex (non‑negative, sum=1)."""
    n = v.shape[0]
    if v.sum() == 1.0 and np.all(v >= 0):
        return v.copy()
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive => uniform vector
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid rounding errors
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full(n, 1.0 / n)
    return w / w_sum


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector.
    (If sum(seq)=1 then the true objective equals this value.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like polynomial shape on [0,1]."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotone exponential decay a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p, decreasing power law."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _linear_increasing_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ i^p, increasing power law."""
    i = np.arange(n, dtype=np.float64)
    a = i ** p
    a[0] = 1e-30                     # avoid exact zero before normalisation
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture of two ℓ₁‑normalised vectors."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()

def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None,
                   max_random: int = 2500) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # parametric random families
    for _ in range(max_random // 5):
        a = 10 ** rng.uniform(-3, 1)          # α ∈ [0.001,10]
        b = 10 ** rng.uniform(-3, 1)          # β ∈ [0.001,10]
        seeds.append(_beta_shape(n, a, b))

        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        lam = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_exp_decay(n, lam))

        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(n,
                             10 ** rng.uniform(-2, 1),
                             10 ** rng.uniform(-2, 1))
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final sanitisation – guarantee positivity and ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
class _SmoothMaxObj:
    """Callable object returning smooth max objective and its gradient."""

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = float(beta)

    def __call__(self, v: np.ndarray) -> tuple[float, np.ndarray]:
        a = _softmax(v)                     # stay on simplex
        conv = _conv_fft(a, a)
        shift = np.max(conv)                # numeric guard
        exp_beta = np.exp(self.beta * (conv - shift))
        sum_exp = exp_beta.sum()
        obj = 2.0 * self.n * (shift + np.log(sum_exp) / self.beta)

        w = exp_beta / sum_exp              # soft‑max weights over convolution entries
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]

        # back‑propagation through soft‑max (log‑sum‑exp) layer
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """
    Optimises a smooth surrogate of max‑convolution using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 3000.0, 8000.0, 20000.0]

    # initialise logits from the current vector
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        obj = _SmoothMaxObj(n, beta)

        def callback(xk, *args, **kwargs):   # simple timeout callback
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=obj,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-13,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] time limit reached (beta={beta})")
            break
        except Exception as exc:               # pragma: no cover
            print(f"[L‑BFGS] exception {exc} (beta={beta})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact sub‑gradient (mirror‑descent) optimiser – exponentiated gradient
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 1_500_000) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient at the active index
        e = np.zeros(2 * n - 1, dtype=np.float64)
        e[max_idx] = 1.0
        rev_a = a[::-1]
        full = _conv_fft(rev_a, e)
        g = 2.0 * full[n - 1:n - 1 + n]

        # step‑size ηₜ = 0.5 / √t
        eta = 0.5 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Exact projected‑gradient descent on the non‑smooth objective
# ----------------------------------------------------------------------
def _proj_gradient_optimize(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 600_000) -> np.ndarray:
    """
    Projected gradient descent on the exact max‑convolution objective.
    Uses the sub‑gradient at the current active index.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient at the active index (same construction as in exp‑grad)
        e = np.zeros(2 * n - 1, dtype=np.float64)
        e[max_idx] = 1.0
        rev_a = a[::-1]
        full = _conv_fft(rev_a, e)
        g = 2.0 * full[n - 1:n - 1 + n]

        # step size decays like 1/√t (tuned a little lower than exp‑grad)
        eta = 0.4 / math.sqrt(it)
        a_tmp = a - eta * g
        a = _proj_simplex(a_tmp)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 150_000 == 0:
                print(f"[ProjGrad] it={it:,} best≈{best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 600_000) -> np.ndarray:
    """
    Simulated‑annealing style hill‑climber.
    Randomly moves a small amount of mass from one index to another,
    accepting occasional uphill moves with a linearly decaying temperature.
    """
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(0xC0FFEE)
    total_iters = max_iters

    for it in range(total_iters):
        if time.time() > deadline:
            break

        progress = it / total_iters
        # temperature slowly shrinks from ~5 % of current value to near zero
        temp = (cur_val * 0.05) * (1.0 - progress) + (cur_val * 1e-6) * progress
        # step‑size similarly shrinks
        eps = 0.12 * (1.0 - progress) + 1e-6 * progress

        i = rng.integers(n)                     # donor (any index)
        if a[i] <= 1e-15:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1                               # ensure i ≠ j

        epsilon = eps * rng.random() * a[i]     # proportion of donor mass
        epsilon = min(epsilon, a[i])

        # apply move
        a[i] -= epsilon
        a[j] += epsilon

        new_val = _objective_surrogate(a)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / max(temp, 1e-15)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                if it % 100_000 == 0:
                    print(f"[HillClimb] it={it:,} best≈{best_val:.6f}")
        else:
            # revert
            a[j] -= epsilon
            a[i] += epsilon

    return best_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the return statement
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Problem size – keep it reasonably large but not too large
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting optimiser (n={n}) ===")

    # ------------------------------------------------------------------
    # Seed generation & cheap ranking
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=2500)
    print(f"Generated {len(seeds)} seed vectors.")

    # cheap surrogate ranking (ℓ₁‑normalised → surrogate = 2·n·max(conv))
    surrogate_vals = [_objective_surrogate(v) for v in seeds]
    order = np.argsort(surrogate_vals)
    top_k = min(8, len(seeds))
    top_seeds = [seeds[i] for i in order[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for idx, i in enumerate(order[:top_k], 1):
        print(f"  #{idx}: {surrogate_vals[i]:.6f}")

    # initialise global best
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on elite seeds
    # ------------------------------------------------------------------
    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1000.0, 3000.0, 8000.0, 20000.0]

    for seed_vec in top_seeds:
        if time.time() > deadline:
            break
        print("\n--- L‑BFGS refinement of an elite seed ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=5000)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_seq = seq_opt.copy()
            print(f"[Update] new global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient (mirror‑descent) on exact max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # heuristic: allocate most of the remaining time to this stage
        max_iters = int(min(3_000_000, remaining * 12000))
        print("\n--- Exponentiated‑gradient refinement (exact max) ---")
        seq_exp = _exp_gradient_optimize(global_best_seq, n, deadline,
                                         max_iters=max_iters)
        val_exp = _objective_surrogate(seq_exp)
        print(f"[ExpGrad] surrogate = {val_exp:.6f}")
        if val_exp < global_best_val:
            global_best_val = val_exp
            global_best_seq = seq_exp.copy()
            print(f"[Update] new global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – projected‑gradient descent on the non‑smooth objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(1_500_000, remaining * 8000))
        print("\n--- Projected‑gradient descent (exact max) ---")
        seq_proj = _proj_gradient_optimize(global_best_seq, n, deadline,
                                           max_iters=max_iters)
        val_proj = _objective_surrogate(seq_proj)
        print(f"[ProjGrad] surrogate = {val_proj:.6f}")
        if val_proj < global_best_val:
            global_best_val = val_proj
            global_best_seq = seq_proj.copy()
            print(f"[Update] new global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – hill‑climbing / simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(1_200_000, remaining * 15000))
        print("\n--- Hill‑climbing (simulated annealing) phase ---")
        seq_hc = _hill_climb(global_best_seq, n, deadline,
                             max_iters=max_iters)
        val_hc = _objective_surrogate(seq_hc)
        print(f"[HillClimb] surrogate = {val_hc:.6f}")
        if val_hc < global_best_val:
            global_best_val = val_hc
            global_best_seq = seq_hc.copy()
            print(f"[Update] new global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:      # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
