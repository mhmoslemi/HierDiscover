"""
propose_candidate
-----------------

Search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three stages:

1. **Initialisation** – reuse the previously best sequence if it exists,
   otherwise start from the uniform distribution (or a few Dirichlet draws).

2. **Smooth‑max optimisation** – minimise a smoothed version of the max‑norm.
   For a temperature α we use the soft‑max
        smaxα(x) = Σ w_i·x_i ,   w = softmax(α·x) .
   The gradient of 2·n·smaxα(conv(a,a)) can be written in closed form as
        grad = 2·n·(H + a) ,
   where H is the correlation of the weight vector w with a.
   We perform exponentiated‑gradient updates in the log‑softmax space
   (log‑parameters are updated with an Adam‑style schedule) and gradually
   increase α (1 → 4 → 10 → 25 → 60).  The updates automatically keep the
   vector on the probability simplex.

3. **Exact‑max polishing** – a short projected sub‑gradient descent on the
   true max‑convolution objective (the same routine as in the reference
   solution) to squeeze out the last bits of improvement.

All steps are time‑aware; the routine stops well before the 1000 s wall‑clock
budget and leaves the best sequence found in the global variable
``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n·log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc) and normalise to sum 1."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *exact* max‑convolution objective.
    Works directly on the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max exponentiated‑gradient optimiser
# ----------------------------------------------------------------------
def _smooth_max_expgrad(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       alpha: float,
                       max_iters: int) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max objective
        J(a) = 2·n·smaxα(conv(a,a))
    using exponentiated‑gradient (log‑softmax) updates together with an
    Adam‑style step‑size schedule.

    Returns the best sequence seen (according to the *exact* surrogate)
    and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    # Adam accumulators
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # 1) convolution a*a
        conv = _conv_fft(a, a)                     # length 2n‑1

        # 2) soft‑max weights w_i ∝ exp(α·conv_i)
        w = _softmax(alpha * conv)                 # shape (2n‑1,)

        # 3) correlation H_j = Σ_i w_i·a_{i‑j}
        a_rev = a[::-1]
        conv_w = _conv_fft(a_rev, w)               # length 3n‑2
        offset = len(w) - 1                        # = 2n‑2
        H = conv_w[offset: offset + n]             # length n

        # 4) gradient of J = 2·n·(H + a)
        grad = 2.0 * n * (H + a)

        # Adam update in log‑space
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size decays with √t and with α (larger α → sharper max → smaller steps)
        step = 0.30 / (math.sqrt(it) * (1.0 + alpha))

        log_a = log_a - step * (m_hat / (np.sqrt(v_hat) + eps))
        a = _softmax(log_a)                        # stays on simplex automatically

        # occasional exact surrogate check
        if (it % 20000 == 0) or (it == max_iters):
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[α={alpha:.1f}] it={it:6d} new best surrogate = {best_val:.6f}")
            # early exit if we already beat the target
            if best_val < 1.503:
                break

    return best_seq, best_val
