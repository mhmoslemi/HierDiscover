"""
Proposes a step‑function (non‑negative sequence) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

using an *alternating linear‑program / exponentiated‑gradient* scheme.
The algorithm:

1. Initialise a probability vector (L1‑normalised) – either the previous best
   sequence or a random Dirichlet draw.
2. Repeatedly solve a linear programme that maximises the L1‑norm of a new
   vector ``g`` while keeping the convolution ``f * g`` bounded by the
   current worst‑case value ``M``.  The solution is rescaled to unit L1‐norm
   and mixed with the current vector ``f``.
3. After the LP loop a short exponentiated‑gradient (mirror–descent) polish
   is performed on the exact surrogate ``2·n·max(conv)``.
4. The final sequence is evaluated with the official ``evaluate_sequence``
   function and returned.

The method stays within the 1000 s budget, uses only NumPy / SciPy and no
file‑system access.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
except Exception:  # pragma: no cover
    _opt = None
    _sp = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (surrogate works on L1‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for L1‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Linear‑program sub‑routine
# ----------------------------------------------------------------------
def _lp_max_sum(f: np.ndarray, M: float, max_val: float = 1000.0):
    """
    Solve: maximise sum(g)  subject to  f * g ≤ M   (componentwise),
    g ≥ 0, g ≤ max_val.
    Returns the optimal g (numpy array) or ``None`` on failure.
    """
    if _opt is None or _sp is None:   # pragma: no cover
        return None

    n = f.shape[0]
    # Build the convolution matrix A where (A g)[k] = (f * g)[k].
    # This is a Toeplitz matrix with constant diagonals.
    # Each diagonal i (0 ≤ i < n) contains the constant value f[i].
    data = [f.copy() for _ in range(n)]
    offsets = np.arange(n)                # 0 … n‑1
    A = _sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Linear programme in the form  min c·x  s.t. A·x ≤ b , 0 ≤ x ≤ max_val
    c = -np.ones(n, dtype=np.float64)     # minimise -sum(x)  → maximise sum(x)
    bounds = [(0.0, max_val) for _ in range(n)]

    res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                       method='highs', options={'disp': False})
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Exponentiated‑gradient polish (mirror descent on exact surrogate)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200_000) -> np.ndarray:
    """Short exponentiated‑gradient descent on the surrogate 2·n·max(conv)."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence found (list of floats).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # reserve a couple of seconds for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good balance between granularity and speed
    # ------------------------------------------------------------------
    n = 4000
    target_sum = 1.0                                   # we work on the simplex

    # ------------------------------------------------------------------
    # Initialise a feasible vector (L1‑normalised, non‑negative)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                init = arr.copy()
            else:
                raise ValueError
        except Exception:
            init = None
    else:
        init = None

    if init is None:
        # a random Dirichlet draw (moderate concentration) – already normalised
        init = rng.dirichlet(np.full(n, 0.5)).astype(np.float64)

    # Normalise to the simplex (sum = 1)
    init_sum = init.sum()
    if init_sum <= 0:
        init = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        init = init / init_sum

    f = init.copy()
    best_seq = f.copy()
    best_val = _objective_surrogate(f)

    print(f"\n--- Starting optimisation (n={n}) ---")
    print(f"Initial surrogate value: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop: alternating LP + mixing
    # ------------------------------------------------------------------
    iteration = 0
    t_mix = 0.2                      # mixing coefficient (1‑t is retained part)
    while time.time() < deadline - 5.0:
        iteration += 1
        conv_f = _conv_fft(f, f)
        M = float(np.max(conv_f))

        # Solve the LP to maximise L1‑norm under the current max‑conv bound
        g = _lp_max_sum(f, M, max_val=1000.0)
        if g is None:
            print("[LP] solver failed – terminating loop.")
            break

        sum_g = float(g.sum())
        if sum_g <= 0:
            print("[LP] got zero sum – terminating.")
            break

        # Rescale g to unit L1‑norm (the objective is scale‑invariant)
        sigma = target_sum / sum_g
        g = g * sigma

        # Mix the new candidate with the current vector
        f_new = (1.0 - t_mix) * f + t_mix * g

        # Re‑project onto the simplex (protect against clipping artefacts)
        f_new_sum = f_new.sum()
        if f_new_sum == 0.0:
            f_new = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            f_new = f_new / f_new_sum

        # Evaluate surrogate and keep the best
        cur_val = _objective_surrogate(f_new)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = f_new.copy()
            print(f"[Iter {iteration:3d}] improved surrogate = {best_val:.6f}")

        # Prepare for next iteration
        f = f_new

        # Slowly reduce mixing coefficient to stabilise as we converge
        t_mix = max(0.05, t_mix * 0.95)

        # Safety: stop after a moderate number of LP steps (memory bound)
        if iteration >= 25:
            break

    # ------------------------------------------------------------------
    # Final polishing with exponentiated‑gradient (fast, pure‑smooth)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 1e2))
        max_iters = max(20_000, max_iters)
        print(f"\n--- Exponentiated‑gradient polishing (max_iters={max_iters}) ---")
        best_seq = _exp_gradient_opt(best_seq, n, deadline, max_iters=max_iters)
        best_val = _objective_surrogate(best_seq)
        print(f"[Polish] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store the result for possible reuse by a later call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
