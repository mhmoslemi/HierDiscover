"""
Optimisation of a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where ``conv`` is the full linear convolution.  The objective is invariant
to scaling, therefore all candidates are kept ℓ₁‑normalised (i.e. “probability
vectors”).

The optimiser consists of three stages:

1. **Seed generation & short ranking** – a diverse collection of analytic
   families (uniform, beta‑shapes, power‑laws, discrete Gaussians, random
   Dirichlet draws and sparse k‑point spikes) is ℓ₁‑normalised and ranked by
   the exact surrogate `2·n·max(conv)`.  The best few seeds are retained.

2. **Smoothed mirror‑descent** – starting from each top seed we run a
   gradient descent on a *smooth* approximation of the `max` function:
   ```
       Fτ(a) = τ·log( Σk exp(conv[k]/τ) )
   ```
   which converges to the true maximum as τ→0.  The gradient of `Fτ`
   can be evaluated efficiently with two FFT‑based convolutions per
   iteration.  The iterate lives in log‑softmax space (`log_a`), guaranteeing
   non‑negativity and unit ℓ₁‑norm.  τ is driven exponentially from a
   moderate start (≈0.2) to a tiny final value (≈1e‑5) so that the optimiser
   first sees a smooth landscape and then focuses on the true peak.

3. **Projected sub‑gradient polishing** – the best result from the
   smoothed phase is refined by classic projected sub‑gradient descent on
   the exact `max` objective.

All stages are time‑aware; the total wall‑clock budget (default 1000 s) is
respected.  The final sequence is stored in the global name
``height_sequence_1`` and returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_k_uniform(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=min(k, n), replace=False)
    a[idx] = 1.0 / len(idx)
    return a


# ----------------------------------------------------------------------
# Smoothed mirror‑descent optimiser (log‑softmax space)
# ----------------------------------------------------------------------


def _smooth_mirror_descent(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int,
                           init_tau: float = 0.2,
                           final_tau: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth approximation
        Fτ(a) = τ·log Σk exp(conv[k]/τ)
    by mirror descent on the log‑softmax variables.

    Parameters
    ----------
    start_seq : np.ndarray
        ℓ₁‑normalised starting vector (length n).
    n : int
        Length of the sequence.
    deadline : float
        Unix time at which the routine must stop.
    max_iters : int
        Upper bound on iterations (soft, the real limit is the time‑deadline).
    init_tau, final_tau : float
        Temperature schedule – τ decays exponentially from init_tau to final_tau.

    Returns
    -------
    best_seq : np.ndarray
        Best vector seen (ℓ₁‑normalised).
    best_val : float
        Corresponding surrogate value 2·n·max(conv).
    """
    a = start_seq.copy()
    # initialise log‑probabilities
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # pre‑allocate buffer for reverse vector (updated each iteration)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current temperature (exponential decay)
        tau = init_tau * (final_tau / init_tau) ** (it / max_iters)

        # full self‑convolution
        conv = _conv_fft(a, a)                     # length 2n‑1
        max_conv = conv.max()

        # smooth weights approximating the arg‑max
        # use max for numerical stability (all exponents ≤ 0)
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:                         # extremely unlikely
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth objective:
        #   g_a = 2 * (w * a_rev)  (full convolution, then central slice)
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)            # length 3n‑2
        g_a = 2.0 * conv_wa[n - 1 : n - 1 + n]    # slice of length n

        # transform gradient to log‑space (mirror‑descent on simplex)
        avg = np.dot(g_a, a)                     # scalar = ⟨g_a, a⟩
        g_log = a * (g_a - avg)

        # step size – classic 0.5/√t schedule (works well in practice)
        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g_log

        # back to probability simplex
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional status output
            if it % 100_000 == 0:
                print(f"[SmoothMD] it={it:7d} best={best_val:.6f} τ={tau:.2e}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (same as in the reference code)
# ----------------------------------------------------------------------


def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Classical projected sub‑gradient descent on the exact max‑convolution
    objective.  Works directly on the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        # Euclidean projection onto the simplex
        # (the implementation works for any real vector)
        # – we use the standard O(n log n) algorithm
        u = np.sort(a)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(a.shape[0]) + 1
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond].max()
            theta = cssv[cond].max() / rho
            a = np.maximum(a - theta, 0.0)
        else:
            a = np.full_like(a, 1.0 / a.shape[0])

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main proposer
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the official ``evaluate_sequence`` metric.  The function respects the
    supplied time budget (default 1000 s) and returns the best vector found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    n = 4000                                    # problem dimension used so far

    # ------------------------------------------------------------------
    # 0. Load previous best (if any)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    previous_vec = None
    if isinstance(prev, list) and len(prev) == n:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.sum() > 0.0:
            previous_vec = arr / arr.sum()

    # ------------------------------------------------------------------
    # 1. Rich seed generation
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    if previous_vec is not None:
        seeds.append(previous_vec)

    # beta‑shape grid (coarse)
    for a in np.linspace(0.2, 5.0, 7):
        for b in np.linspace(0.2, 5.0, 7):
            seeds.append(_beta_shape(n, a, b))

    # beta‑shape random draws
    for _ in range(60):
        a = 10 ** rng.uniform(-2, 2)
        b = 10 ** rng.uniform(-2, 2)
        seeds.append(_beta_shape(n, a, b))

    # power‑law grid + random
    for p in np.linspace(0.0, 5.0, 9):
        seeds.append(_power(n, p))
    for _ in range(30):
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

    # Gaussian grid + random
    for sigma in np.linspace(0.12, 0.45, 6):
        seeds.append(_gaussian(n, sigma))
    for _ in range(20):
        sigma = rng.uniform(0.08, 0.55)
        seeds.append(_gaussian(n, sigma))

    # sparse k‑uniform
    for k in (10, 20, 50, 100):
        seeds.append(_random_k_uniform(n, k, rng))

    # random Dirichlet (several concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            if time.time() > deadline:
                break
            seeds.append(_random_dirichlet(n, conc, rng))

    # ------------------------------------------------------------------
    # Normalise seeds (guard against accidental zeros)
    # ------------------------------------------------------------------
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        total = s.sum()
        if total > 0.0:
            normed_seeds.append(s / total)

    print(f"\n--- Seed generation finished: {len(normed_seeds)} candidates ---")

    # ------------------------------------------------------------------
    # 2. Coarse ranking (exact surrogate)
    # ------------------------------------------------------------------
    scored = [(vec, _objective_surrogate(vec)) for vec in normed_seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Best raw surrogate = {scored[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed mirror‑descent optimisation starting from the top seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all smooth MD runs.")
            break

        # allocate iterations proportionally to remaining time
        remaining = max(0.0, deadline - time.time())
        # empirical 2000 iterations per second for the heavier routine
        est_iters = int(min(800_000, remaining * 2000))
        if est_iters < 20_000:
            est_iters = int(min(200_000, remaining * 2000))
        if est_iters <= 0:
            break

        print(f"\n--- Smoothed MD run #{idx + 1} (≈{est_iters} iters) ---")
        seq_opt, val_opt = _smooth_mirror_descent(
            seed_vec,
            n,
            deadline,
            max_iters=est_iters,
            init_tau=0.2,
            final_tau=1e-5,
        )

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Sub‑gradient polishing of the best candidate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(800_000, remaining * 0.5))
        if max_sub > 0:
            print("\n--- Sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq, n, deadline, max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved best to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final official evaluation and bookkeeping
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
