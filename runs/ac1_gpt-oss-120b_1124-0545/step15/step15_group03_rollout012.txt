    a = start_seq.copy()
    logits = np.log(a + 1e-30)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    conv_N = _next_pow2(2 * n - 1)
    corr_N = _next_pow2(3 * n - 2)
    start = (2 * n - 1) - 1  # L-1
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is the full linear convolution.  The search consists of:

1. Rich seed generation (uniform, beta‑shapes, powers, Gaussians,
   sparse k‑point, random Dirichlet).
2. Quick ranking of the seeds using the exact surrogate
   ``2·n·max(conv)`` (the denominator is 1 after ℓ₁‑normalisation).
3. For the best few seeds a *smooth Adam* optimisation is performed.
   The objective ``max(conv)`` is approximated by a smooth
   log‑sum‑exp (temperature → 0) which yields a differentiable loss.
   Gradients are obtained by FFT‑based correlation, the parameters are
   the logits of a soft‑max (so the simplex constraint is automatic).
   The temperature (inverse γ) is annealed from a very smooth regime to a
   very sharp one.
4. A short projected sub‑gradient polishing (as in the reference code)
   is applied to the current global best.
5. A brief simulated‑annealing run in logit space gives a final local
   push.

All steps are time‑aware and stop before the 1000 s budget.  The best
sequence is stored in the global variable ``height_sequence_1`` and
returned as a plain list of floats.
"""

import time

# ----------------------------------------------------------------------
# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector.
    The argument **seq** must sum to 1 (or any constant – the factor 2·n
    multiplies the max, the denominator is omitted here).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle; sigma = sigma_frac·n."""
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_k_uniform(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform distribution on a random subset of size k."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=min(k, n), replace=False)
    a[idx] = 1.0 / len(idx)
    return a

def _smooth_adam_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Minimises the exact ``max(conv)`` via a smooth log‑sum‑exp
    approximation and Adam updates on the logits (soft‑max space).
    Returns the best sequence seen and its surrogate value.
    """
    # initialise
    a = start_seq.copy()
    logits = np.log(a + 1e-30)               # logits give a = softmax(logits)
    m = np.zeros_like(logits)                # first moment
    v = np.zeros_like(logits)                # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # temperature schedule (inverse γ) – start very smooth, end very sharp
    gamma_start = 0.5
    gamma_end = 100_000.0

    # FFT length big enough for both convolution and correlation
    N = _next_pow2(3 * n - 2)                # >= (2n‑1)+(n‑1)
    conv_len = 2 * n - 1
    w_len = conv_len
    corr_start = w_len - 1                    # offset where lag 0 begins

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for t in range(1, max_iters + 1):
        # periodic time‑check
        if (t & 1023) == 0 and time.time() > deadline:   # every 1024 it.
            break

        # re‑compute a from logits (soft‑max ensures feasibility)
        a = _softmax(logits)

        # FFT of a (single transform, reused for convolution & correlation)
        fa = np.fft.rfft(a, n=N)

        # ----- convolution a*a (full) -----
        conv_full = np.fft.irfft(fa * fa, n=N)[:conv_len]
        max_conv = np.max(conv_full)

        # ----- smooth approximation of max -----
        gamma = gamma_start + (gamma_end - gamma_start) * (t / max_iters)
        # shift by max to avoid overflow: values ≤0, exp(…) ≤1
        w = np.exp(gamma * (conv_full - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w_len)
        else:
            w /= w_sum

        # ----- correlation w ⋆ a (to obtain gradient w.r.t. a) -----
        fw = np.fft.rfft(w, n=N)
        # correlation = ifft(conj(fw) * fa)
        corr_full = np.fft.irfft(np.conj(fw) * fa, n=N)[:w_len + n - 1]
        grad_a = 2.0 * corr_full[corr_start:corr_start + n]   # length n

        # ----- gradient in logit space (mirror descent) -----
        G = np.dot(grad_a, a)                     # scalar ⟨grad_a , a⟩
        grad_logits = a * (grad_a - G)            # element‑wise

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits * grad_logits)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size: 0.5 / sqrt(t) scaled by Adam’s adaptive term
        lr = (0.5 / (math.sqrt(t) + 1e-12)) / (np.sqrt(v_hat) + eps)
        logits = logits - lr * m_hat

        # occasional monitoring
        if (t % 50000) == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SmoothAdam] it={t:7d} best={best_val:.6f} gamma={gamma:.1f}")
            else:
                print(f"[SmoothAdam] it={t:7d} cur={cur_val:.6f} gamma={gamma:.1f}")

    # final check
    cur_val = _objective_surrogate(a)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = a.copy()
    return best_seq, best_val


def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Returns the best sequence seen and its surrogate.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if (it & 1023) == 0 and time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if (it % 100_000) == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple SA in log‑softmax (logit) space.
    """
    rng = np.random.default_rng(12345)
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if (it & 1023) == 0 and time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.06 * (1.0 - progress) + 1e-7
        sigma = 0.20 * (1.0 - progress) + 1e-5

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if (it % 100_000) == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep the size from the reference solution.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Retrieve previous solution if it exists.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    previous_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                previous_vec = arr / arr.sum()
        except Exception:
            previous_vec = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation.
    # ------------------------------------------------------------------
    seeds = []

    seeds.append(_uniform(n))
    if previous_vec is not None:
        seeds.append(previous_vec)

    for a in np.linspace(0.2, 5.0, 9):
        for b in np.linspace(0.2, 5.0, 9):
            seeds.append(_beta_shape(n, a, b))

    for _ in range(120):
        a = 10 ** rng.uniform(-2, 2)
        b = 10 ** rng.uniform(-2, 2)
        seeds.append(_beta_shape(n, a, b))

    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))
    for _ in range(50):
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    for _ in range(30):
        sigma = rng.uniform(0.05, 0.55)
        seeds.append(_gaussian(n, sigma))

    for k in (10, 20, 50, 100):
        seeds.append(_random_k_uniform(n, k, rng))

    for conc in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            if time.time() > deadline:
                break
            seeds.append(_random_dirichlet(n, conc, rng))

    # Normalise and discard degenerate vectors
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        total = s.sum()
        if total <= 0.0:
            continue
        normed_seeds.append(s / total)

    print(f"\n--- Seed generation complete: {len(normed_seeds)} candidates ---")

    # ------------------------------------------------------------------
    # 2. Coarse ranking – keep the best few raw seeds.
    # ------------------------------------------------------------------
    scored = []
    for vec in normed_seeds:
        val = _objective_surrogate(vec)
        scored.append((vec, val))
    scored.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Best surrogate among raw seeds = {scored[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 3. Smooth‑Adam optimisation on the top seeds.
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all smooth‑Adam runs.")
            break

        # allocate iterations based on remaining wall‑clock time
        remaining = max(0.0, deadline - time.time())
        # empirical 400 it/s for the smoother optimiser (conservative)
        est_iters = int(min(800_000, remaining * 400))
        if est_iters < 20_000:
            est_iters = int(min(200_000, remaining * 400))

        print(f"\n--- Smooth‑Adam run #{idx+1} (max_iters≈{est_iters}) ---")
        seq_opt, val_opt = _smooth_adam_opt(seed_vec, n, deadline, max_iters=est_iters)

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Sub‑gradient polishing of the current best.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(800_000, remaining * 0.5))
        if max_sub > 0:
            print("\n--- Sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq, n, deadline, max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved best to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final simulated‑annealing refinement (short).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(300_000, remaining * 0.3))
        if max_sa > 0:
            print("\n--- Simulated‑annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing(global_best_seq, n, deadline, max_iters=max_sa)
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[SA] final best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
