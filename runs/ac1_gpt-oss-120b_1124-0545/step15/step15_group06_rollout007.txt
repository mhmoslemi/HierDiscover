"""
Improved search for a non‑negative step function minimizing

        2·n·max(conv(a,a)) / (sum(a)²)

Algorithm
---------

* **Seed generation** – many analytic families (uniform, windows,
  Beta‑U‑shapes, exponential decay, etc.) plus random Dirichlet,
  sparse, low‑frequency curves and many random mixtures.
* **Coarse selection** – evaluate the exact max‑convolution of each seed,
  keep the best few.
* **Smooth‑max L‑BFGS continuation** – optimise the log‑sum‑exp smooth
  surrogate of the max using L‑BFGS‑B on the log‑softmax (unconstrained)
  variables.  A long β‑schedule (up to 1e5) plus a larger per‑β iteration
  budget expands the search surface.
* **Exact‑max exponentiated‑gradient refinement** – mirror‑descent on the
  true non‑smooth objective with a slowly decaying step‑size.
* **CMA‑ES‑style random‑search‑and‑polish** – repeatedly jitter the
  logits of the current best solution, run a short high‑β L‑BFGS polish
  and keep any improvement.  The jitter scale decays slowly, providing
  both exploration and fine‑grained exploitation.
* **Final high‑β polish** – a last L‑BFGS run with β≈2·10⁴.

All stages respect the global time budget (≈ 1000 s).  The best
sequence found is stored in the global name ``height_sequence_1`` for
warm‑starting the next call.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (uses zero‑padding)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector (ℓ₁‑normalised)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on {1,…,n}.  U‑shape when α,β<1."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_sparse(n: int, rng: np.random.Generator,
                   k_min: int = 30, k_max: int = 150) -> np.ndarray:
    """Sparse support: choose K random indices, give them random weights."""
    K = int(rng.integers(k_min, k_max + 1))
    idx = rng.choice(n, size=K, replace=False)
    a = np.zeros(n, dtype=np.float64)
    weights = rng.exponential(scale=1.0, size=K)
    a[idx] = weights
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-rate * i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α)+b·α then ℓ₁‑renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Surrogate objective (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Returns (value, gradient) for the smooth max‑convolution surrogate.
    v : logits (real vector) – softmax(v) yields a probability vector a.
    """
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)              # full auto‑convolution, length 2n‑1
    shift = np.max(conv)                # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient of obj wrt a (before the softmax chain rule)
    w = exp_term / sum_exp               # weights forming a convex combination
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2, central part needed
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """
    L‑BFGS‑B optimisation of the smooth max‑convolution surrogate.
    Variable = logits of a softmax (unconstrained real vector).
    """
    if beta_schedule is None:
        # a dense schedule ranging from very smooth to almost exact max
        beta_schedule = [0.1, 0.5, 1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0,
                         5000.0, 10000.0, 20000.0, 50000.0,
                         100000.0]

    # initialise logits from the start probabilities
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(local_v):
            return _smooth_objective(local_v, n, beta_val)

        def callback(local_v):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            # early stop – continue with the current logits
            break
        except Exception:                     # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact objective utilities
# ----------------------------------------------------------------------
def _objective_raw(a: np.ndarray) -> float:
    """Exact formulation 2·n·max(conv) / (sum²) (no clipping)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (using the active lag).
    Returns g = ∂ max(conv) / ∂ a .
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


def _exp_gradient_optimize(start_a: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 1_500_000,
                          eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Uses exponentiated‑gradient updates on logits.
    Returns (best_vector, its exact objective).
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_raw(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)          # sub‑gradient of the active lag
        eta = eta0 / math.sqrt(it)           # diminishing stepsize

        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 100_000 == 0:
            cur = _objective_raw(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
    # final check
    cur = _objective_raw(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val

def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised seed vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12, 20):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.08, 0.45, 9):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # beta families – include U‑shapes (α,β<1) as well as moderate
    for a in np.linspace(0.1, 1.5, 7):
        for b in np.linspace(0.1, 1.5, 7):
            seeds.append(_beta_shape(n, a, b))

    for al in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, al))

    # random continuous draws ---------------------------------------------------------
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sparse random supports -----------------------------------------------------------
    for _ in range(30):
        seeds.append(_random_sparse(n, rng, k_min=30, k_max=120))

    # New shape: monotone exponential decay (different rates)
    for rate in np.linspace(0.0005, 0.005, 10):
        seeds.append(_exponential_decay(n, rate))

    # Mixtures of already generated seeds ---------------------------------------------
    base_idx = np.arange(len(seeds))
    for _ in range(500):
        i, j = rng.choice(base_idx, size=2, replace=False)
        alpha = rng.random()
        mix = _mix_shapes(seeds[i], seeds[j], alpha)
        seeds.append(mix)

    # Previous best (if any) ---------------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final clean‑up – ensure non‑negativity & ℓ₁‑normalisation -------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

def _cma_style_refine(best_a: np.ndarray,
                      n: int,
                      rng: np.random.Generator,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a simple CMA‑ES‑style search:
    - jitter logits of the current best,
    - short high‑β L‑BFGS polish,
    - accept if the exact objective improves.
    """
    best_val = _objective_raw(best_a)
    sigma = 0.5                # initial logit‑space mutation strength
    min_sigma = 0.01
    decay = 0.9995

    # Pre‑compute the logit of the current best to avoid recomputing each loop
    logit_best = np.log(best_a + 1e-30)

    while time.time() < deadline:
        # jitter logits
        jitter = rng.normal(scale=sigma, size=n)
        start_logits = logit_best + jitter
        start_vec = _softmax(start_logits)

        # short high‑β polish (few iterations – fast)
        seq = _lbfgs_smooth_optimize(start_vec,
                                     n,
                                     deadline,
                                     beta_schedule=[5000.0, 20000.0, 50000.0],
                                     max_iter_per_beta=800)

        val = _objective_raw(seq)
        if val < best_val:
            best_a = seq
            best_val = val
            logit_best = np.log(best_a + 1e-30)
            print(f"[CMA‑ES] improvement: {best_val:.6f}")

        # decay mutation strength
        sigma = max(min_sigma, sigma * decay)

    return best_a, best_val

    print(f"Best surrogate values among seeds (top {top_k}):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # -------------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on the smooth surrogate
    # -------------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _maxconv(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    beta_schedule = [0.1, 0.5, 1.0, 5.0, 20.0, 80.0,
                     200.0, 500.0, 1000.0, 2000.0,
                     5000.0, 10000.0, 20000.0, 50000.0,
                     100000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all L‑BFGS seeds.")
            break

        remaining = deadline - time.time()
        # allocate a generous per‑β budget while remaining time permits
        per_beta_iters = max(4000,
                             int(min(12000,
                                     remaining * 0.07 * 1e5)))
        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter/beta={per_beta_iters}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=per_beta_iters)
        val_opt = _maxconv(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # -------------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient refinement on the exact max‑convolution
    # -------------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(2_500_000,
                                remaining * 0.30 * 1e5))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient refinement (exact max) ---")
            seq_exp, val_exp = _exp_gradient_optimize(global_best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=max_exp_iters,
                                                      eta0=1.0)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved exact objective to {global_best_val:.6f}")

    # -------------------------------------------------------------------------
    # Stage 3 – CMA‑ES‑style stochastic perturb‑and‑polish
    # -------------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- CMA‑ES‑style random‑search‑and‑polish phase ---")
        seq_cma, val_cma = _cma_style_refine(global_best_seq,
                                             n,
                                             rng,
                                             deadline)
        if val_cma < global_best_val:
            global_best_seq = seq_cma.copy()
            global_best_val = val_cma
            print(f"[CMA‑ES] new best after polishing = {global_best_val:.6f}")

    # -------------------------------------------------------------------------
    # Stage 4 – final polishing with a very high‑β L‑BFGS run
    # -------------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta_iters = max(800,
                             int(min(3000,
                                     remaining * 0.06 * 1e5)))
        print("\n--- Final high‑β L‑BFGS polishing ---")
        seq_final = _lbfgs_smooth_optimize(global_best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=[50000.0, 100000.0, 200000.0],
                                           max_iter_per_beta=per_beta_iters)
        val_final = _maxconv(seq_final)
        print(f"[Final‑L‑BFGS] surrogate = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polish = {global_best_val:.6f}")

    # -------------------------------------------------------------------------
    # Final evaluation with the official evaluator (for reporting)
    # -------------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible warm‑start on the next call
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
