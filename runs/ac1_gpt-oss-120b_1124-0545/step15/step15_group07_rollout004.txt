"""
A multi‑scale, multi‑restart optimiser for the step‑function problem.

The algorithm works as follows

0. Choose a moderate problem size (default n = 5 000) – larger than the
   reference 4 000 to allow a finer approximation of the continuous
   optimum, but still cheap enough for many FFT calls.

1. Build a very diverse pool of ℓ₁‑normalised candidates.
   In addition to the families used in the reference solution we also
   generate Hann / Black‑man windows, sparse random spikes and a few
   extra Dirichlet draws.  All candidates are sanitised to be non‑negative
   and sum to one.

2. Rank the seeds by the cheap surrogate   S(a)=2·n·max(conv(a,a)).
   Keep the best few (default 12).

3. Run a continuation L‑BFGS optimiser on the *top‑k* seeds
   (up to 4 of them).  The optimisation minimises a smoothed version
   of the max‑convolution (log‑sum‑exp) and uses a schedule of increasing
   β‑values ending at β=15 000.  The optimiser works directly on logits
   and therefore always stays on the simplex.

4. Refine the current best vector with an exact exponentiated‑gradient
   (mirror‑descent) on the true max‑convolution.  The step‑size is
   ηₜ=0.5/√t and the number of iterations is chosen dynamically from
   the remaining time budget.

5. A simulated‑annealing hill‑climber moves a small amount of mass from
   the currently largest entry towards a random other entry.
   Improvements are always accepted; worsening moves are accepted with
   a linearly decaying temperature.

6. If enough time remains, the solution is *upsampled* (linear
   interpolation) to twice the length and the whole refinement cycle
   (L‑BFGS → exponentiated‑gradient → hill‑climb) is repeated on the
   finer grid.

7. The best sequence found is returned (as a plain ``list[float]``) and
   stored in the global name ``height_sequence_1`` for reuse by later
   runs.

All heavy work is protected by a hard deadline (budget_s – 2 s) and
periodic time checks, guaranteeing termination before the 1 000 s
budget expires.
"""

import time
import numpy as np
import scipy.optimize as _opt


# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate: 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector.
    (If sum(seq)=1 then the true objective equals this value.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + 0.08 * np.cos(4 * np.pi * i / (n - 1))
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like (skewed polynomial) shape."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p  (decreasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _linear_increasing_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ i^p  (increasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = i ** p
    a[0] = 1e-30  # avoid exact zero before normalisation
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture, then ℓ₁‑normalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k‑spike vector (equal mass on k random positions)."""
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return _uniform(n)
    indices = rng.choice(n, size=min(k, n), replace=False)
    a[indices] = 1.0 / len(indices)
    return a


# ----------------------------------------------------------------------
# Seed generation – broad sweep over parametrised families
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None,
                   max_random: int = 3000) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_hann_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # random parametric families
    for _ in range(max_random // 6):
        a = 10 ** rng.uniform(-3, 1)      # alpha ∈ [0.001,10]
        b = 10 ** rng.uniform(-3, 1)      # beta  ∈ [0.001,10]
        seeds.append(_beta_shape(n, a, b))

        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        lam = rng.exponential(scale=0.002) + 0.0001
        seeds.append(_exp_decay(n, lam))

        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        # mixture of a uniform and a beta shape
        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(n,
                             10 ** rng.uniform(-2, 1),
                             10 ** rng.uniform(-2, 1))
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

        # sparse random spikes (k = 1..5)
        k_spike = rng.integers(1, 6)
        seeds.append(_sparse_random(n, k_spike, rng))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final sanitisation – guarantee positivity and ℓ₁‑normalisation
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = 1.0
        clean.append(v / s)
    return clean


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimises a smooth surrogate of max‑convolution using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 3000.0, 8000.0, 15000.0]

    # initialise logits safely
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                      # for numerical stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a  (soft‑max output)
            w = exp_beta / sum_exp                     # soft‑max weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]      # derivative of 2·n·max

            # back‑prop through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            # time limit reached inside the optimiser
            break
        except Exception as e:               # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 2_500_000) -> np.ndarray:
    """
    Mirror‑descent on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / np.sqrt(it)                # classic 1/√t schedule
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 800_000) -> np.ndarray:
    """
    Move a tiny amount of mass from the current largest entry to a random
    other entry.  Accept improvements always; accept worsening moves with a
    linearly decaying temperature.
    """
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(0xC0FFEE)          # deterministic rng for reproducibility
    total_iters = max_iters

    for it in range(total_iters):
        if time.time() > deadline:
            break

        progress = it / total_iters
        # temperature drops from ~5 % of current value to a tiny floor
        temp = (cur_val * 0.05) * (1.0 - progress) + (cur_val * 1e-6) * progress
        # step size shrinks from 10 % of the mass at the donor to ~1e‑6
        eps = 0.10 * (1.0 - progress) + 1e-6 * progress

        donor = int(np.argmax(a))
        if a[donor] <= 1e-15:
            continue

        # pick a different target uniformly
        target = rng.integers(n - 1)
        if target >= donor:
            target += 1

        epsilon = eps * rng.random()
        epsilon = min(epsilon, a[donor])          # keep non‑negative

        # apply move
        a[donor] -= epsilon
        a[target] += epsilon

        new_val = _objective_surrogate(a)

        # Metropolis criterion
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                if it % 100_000 == 0:
                    print(f"[HillClimb] it={it:,} best≈{best_val:.6f}")
        else:
            # revert
            a[target] -= epsilon
            a[donor] += epsilon

    return best_seq


# ----------------------------------------------------------------------
# Upsampling (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # 0. Problem size (start modest, upscale later)
    # ------------------------------------------------------------------
    n = 5000
    print(f"\n=== Starting multi‑scale search (n={n}) ===")

    # ------------------------------------------------------------------
    # 1. Seed generation & cheap ranking
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=3000)
    print(f"Generated {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best surrogate values (top {top_k} seeds):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 2. Continuation L‑BFGS on the elite seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] Current best surrogate = {global_best_val:.6f}")

    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1000.0, 3000.0, 8000.0, 15000.0]

    # run L‑BFGS on a few elite seeds (up to 4)
    for idx, seed_vec in enumerate(top_seeds[:4]):
        if time.time() >= deadline:
            break
        print(f"\n--- L‑BFGS refinement of elite seed #{idx+1} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=10000)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Exact exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(2_500_000, remaining * 25000))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient refinement (exact max) ---")
            seq_exp = _exp_gradient_optimize(global_best_seq, n, deadline,
                                            max_iters=max_exp_iters)
            val_exp = _objective_surrogate(seq_exp)
            print(f"[ExpGrad] surrogate = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Hill‑climbing (simulated annealing) phase
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc_iters = int(min(800_000, remaining * 15000))
        if max_hc_iters > 0:
            print("\n--- Hill‑climbing (simulated annealing) ---")
            seq_hc = _hill_climb(global_best_seq, n, deadline,
                                 max_iters=max_hc_iters)
            val_hc = _objective_surrogate(seq_hc)
            print(f"[HillClimb] surrogate = {val_hc:.6f}")
            if val_hc < global_best_val:
                global_best_seq = seq_hc.copy()
                global_best_val = val_hc
                print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Optional up‑sampling & fine‑tuning on a finer grid
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 30.0:
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} ---")
            up_seq = _upsample(global_best_seq, new_n)

            # a quick L‑BFGS on the finer grid
            up_opt = _lbfgs_smooth_optimize(up_seq, new_n, deadline,
                                            beta_schedule=beta_sched,
                                            max_iter_per_beta=8000)
            up_val = _objective_surrogate(up_opt)
            print(f"[Upsample‑L‑BFGS] surrogate = {up_val:.6f}")

            if up_val < global_best_val:
                global_best_seq = up_opt.copy()
                global_best_val = up_val
                n = new_n
                print(f"[Update] New best after up‑sampling = {global_best_val:.6f}")

            # final exponentiated‑gradient on the fine grid
            if time.time() < deadline:
                remaining = deadline - time.time()
                max_exp2 = int(min(2_500_000, remaining * 25000))
                if max_exp2 > 0:
                    print("\n--- Final exponentiated‑gradient on fine grid ---")
                    seq_final = _exp_gradient_optimize(global_best_seq, n,
                                                       deadline,
                                                       max_iters=max_exp2)
                    val_final = _objective_surrogate(seq_final)
                    print(f"[Final‑ExpGrad] surrogate = {val_final:.6f}")
                    if val_final < global_best_val:
                        global_best_seq = seq_final.copy()
                        global_best_val = val_final
                        print(f"[Update] New global best surrogate = {global_best_val:.6f}")

            # final hill‑climb (short)
            if time.time() < deadline:
                remaining = deadline - time.time()
                max_hc2 = int(min(400_000, remaining * 15000))
                if max_hc2 > 0:
                    print("\n--- Final hill‑climb on fine grid ---")
                    seq_hc2 = _hill_climb(global_best_seq, n, deadline,
                                          max_iters=max_hc2)
                    val_hc2 = _objective_surrogate(seq_hc2)
                    print(f"[Final‑HillClimb] surrogate = {val_hc2:.6f}")
                    if val_hc2 < global_best_val:
                        global_best_seq = seq_hc2.copy()
                        global_best_val = val_hc2
                        print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
