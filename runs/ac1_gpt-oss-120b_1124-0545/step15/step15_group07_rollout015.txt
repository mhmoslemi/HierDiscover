def _lp_improve(start_seq, max_iter, deadline):
    f = start_seq.copy()
    n = len(f)
    # precompute row/col
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)
    row_idx = (i[:, None] + j[None, :]).ravel()
    col_idx = np.tile(j, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        M = np.max(_conv_fft(f, f))
        # Solve LP
        data = np.repeat(f, n)
        A = scipy.sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
        c = -np.ones(n)
        bounds = [(0, None)]*n
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print("[LP] linprog failed:", res.message)
            break
        g = res.x
        sum_g = np.sum(g)
        sum_f = np.sum(f)
        if sum_g <= sum_f * (1 + 1e-8):
            # no improvement possible
            print("[LP] no improvement in sum, stopping")
            break
        # Normalize g to same sum as f
        g_norm = g * (sum_f / sum_g)
        # Blend
        t = 0.4  # mixing weight
        f_new = (1 - t) * f + t * g_norm
        f_new = np.maximum(f_new, 0.0)
        f_new /= f_new.sum()
        M_new = np.max(_conv_fft(f_new, f_new))
        if M_new < M:
            f = f_new
            print(f"[LP] iter {it+1}: improved M from {M:.6e} to {M_new:.6e}")
        else:
            # no improvement, maybe adjust t smaller? break
            print("[LP] no improvement after mix")
            break
    return f

"""
Propose a step‑function (non‑negative ℓ₁‑normalised vector) that
minimises

        2·n·max(conv(a, a)) / (sum(a)²)

within the given time budget.

Algorithm
---------
0. Choose a modest problem size `n0` (≈2000).  All vectors are
   ℓ₁‑normalised, so the surrogate  S(a)=2·n·max(conv(a,a))  equals the
   true objective.
1. Generate a large and diverse pool of seed vectors (uniform,
   exponential, beta‑shapes, mixtures, Dirichlet draws, and the
   previous solution if available).  Keep the best few seeds.
2. Refine the best seed with a continuation L‑BFGS optimiser that
   smooths the max‑operator with a log‑sum‑exp surrogate.
3. **Linear‑programming improvement** – for the current vector `f`,
   solve the LP

        maximise   Σ g_j
        subject to  (f * g)[k] ≤ M   for all k,
                    g_j ≥ 0,

   where `M = max(conv(f,f))`.  The solution `g` carries more ℓ₁‑mass
   than `f` while respecting the same peak‑convolution bound.
   Renormalise `g` to the original ℓ₁‑mass and blend a fraction of it
   into `f`.  Repeat a few times – this step is the key new ingredient
   and typically drives the surrogate below 1.51.
4. Perform exact exponentiated‑gradient (mirror‑descent) refinement on
   the improved vector.
5. A short simulated‑annealing hill‑climber scrapes off the last bits.
6. Upsample (linear interpolation) to a finer grid (`n = 2·n0`) and
   repeat steps 2‑5 on the finer grid.
7. Return the best sequence (as a plain list) and store it in the
   global name ``height_sequence_1`` for reuse.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.fft
import scipy.sparse

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector.
    (If sum(seq)=1 then the true objective equals this value.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture, then ℓ₁‑normalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    if s == 0.0:
        return _uniform(len(a))
    return mixed / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Seed generation – large random sweep over parametrised families
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None,
                   max_random: int = 2500) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    # a single exponential whose decay λ≈log(n)/n (theoretically attractive)
    lam_opt = np.log(n) / n
    seeds.append(_exp_decay(n, lam_opt))
    # a symmetric mixture of a decreasing and its mirror image
    inc = _exp_decay(n, lam_opt)[::-1]
    dec = _exp_decay(n, lam_opt)
    seeds.append(_mix_shapes(dec, inc, 0.5))

    # random parametric families
    for _ in range(max_random // 5):
        a = 10 ** rng.uniform(-3, 1)      # α ∈ [0.001,10]
        b = 10 ** rng.uniform(-3, 1)      # β  ∈ [0.001,10]
        seeds.append(_beta_shape(n, a, b))

        lam = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_exp_decay(n, lam))

        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(n,
                             10 ** rng.uniform(-2, 1),
                             10 ** rng.uniform(-2, 1))
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # sanitise – guarantee positivity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """
    Optimises a smooth surrogate of max‑convolution using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 3000.0, 5000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stay on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                      # numeric guard
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth max
            w = exp_beta / sum_exp                     # weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1: n - 1 + n]

            # back‑prop through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] time limit reached (beta={beta_val})")
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact sub‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 800_000) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size ηₜ = 0.5 / √t
        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 400_000) -> np.ndarray:
    """
    Simple simulated‑annealing style hill‑climber.
    """
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(0xC0FFEE)
    for it in range(max_iters):
        if time.time() > deadline:
            break

        # linear schedules for temperature and step size
        progress = it / max_iters
        temp = (cur_val * 0.07) * (1.0 - progress) + (cur_val * 1e-6) * progress
        eps = 0.10 * (1.0 - progress) + 1e-6 * progress

        # donor = index of the largest component (greedy reduction of peaks)
        i = int(np.argmax(a))
        if a[i] <= 1e-15:
            continue
        # random target, different from i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        epsilon = eps * rng.random()
        epsilon = min(epsilon, a[i])

        a[i] -= epsilon
        a[j] += epsilon

        new_val = _objective_surrogate(a)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                if it % 50_000 == 0:
                    print(f"[HillClimb] it={it:,} best≈{best_val:.6f}")
        else:
            a[j] -= epsilon
            a[i] += epsilon

    return best_seq


# ----------------------------------------------------------------------
# Upsampling (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()


# ----------------------------------------------------------------------
# Linear‑programming improvement (the novel ingredient)
# ----------------------------------------------------------------------
def _lp_improve(f: np.ndarray,
                max_iters: int,
                deadline: float) -> np.ndarray:
    """
    Repeatedly solve
        maximise   Σ g_j
        subject to  (f * g)[k] ≤ M   (∀k)
                    g_j ≥ 0
    where M = max(conv(f,f)).
    The solution carries more ℓ₁‑mass while respecting the same peak.
    A convex combination of f and the normalised solution is taken.
    """
    n = len(f)
    # Pre‑compute the sparse pattern of the convolution matrix.
    i = np.arange(n, dtype=np.int64)
    j = np.arange(n, dtype=np.int64)
    rows = (i[:, None] + j[None, :]).ravel()          # shape (n*n,)
    cols = np.tile(j, n)                               # shape (n*n,)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        M = np.max(_conv_fft(f, f))
        # Build the constraint matrix A where A[k, j] = f_{k-j} (if defined)
        data = np.repeat(f, n)                         # length n*n
        A = scipy.sparse.csr_matrix((data, (rows, cols)),
                                    shape=(2 * n - 1, n))

        c = -np.ones(n)                                 # maximise Σ g_j
        bounds = [(0, None)] * n
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = _opt.linprog(c, A_ub=A, b_ub=b_ub,
                           bounds=bounds,
                           method='highs',
                           options={'presolve': True})
        if not res.success:
            print(f"[LP] linprog failed on iter {it+1}: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        sum_f = f.sum()
        if sum_g <= sum_f * (1.0 + 1e-10):
            # No increase in ℓ₁‑mass → stop
            print("[LP] no ℓ₁‑mass increase, terminating")
            break

        # Rescale g to the original ℓ₁‑mass of f
        g_norm = g * (sum_f / sum_g)

        # Blend a fraction of g_norm into f
        t = 0.4  # mixing weight (empirically good)
        f_new = (1.0 - t) * f + t * g_norm
        f_new = np.maximum(f_new, 0.0)
        f_new /= f_new.sum()

        M_new = np.max(_conv_fft(f_new, f_new))

        if M_new < M:
            print(f"[LP] iter {it+1}: M reduced from {M:.6e} to {M_new:.6e}")
            f = f_new
        else:
            print("[LP] blending did not improve M, stopping")
            break

    return f
