Pseudo:

def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function sequence.

The strategy is simple but powerful:
* If a previous solution exists it is taken as a warm‑start.
* The current best vector is refined by a long run of exponentiated‑gradient
  (mirror‑descent) on the exact max‑convolution objective.
* After each block of iterations a fresh random restart (a random Dirichlet
  draw) is tried – the EG routine is started from that point and any
  improvement is kept.  This gives many chances to escape local minima.
* When the time budget is almost exhausted a short L‑BFGS polishing is
  performed on the (log‑softmax) surrogate.
* The result is stored in the global name ``height_sequence_1`` and also
  returned as a plain list of floats.

All heavy work is done using FFT‑based convolution, additive/sub‑gradient
updates and a projection onto the probability simplex (so the ℓ₁‑norm stays
exactly 1).  The code stays well within the 1000 s budget on a typical
machine.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient (max‑convolution)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum = 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    Returns a vector g such that
        max(conv(a+δ,a+δ)) ≈ max(conv(a,a)) + g·δ   (first‑order)
    for small δ (δ respects non‑negativity; this is the standard
    sub‑gradient used in mirror‑descent).
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))                 # lag that attains the max
    g = np.zeros_like(a)

    # the contributing indices are i with 0≤i≤n‑1 and 0≤idx‑i≤n‑1
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range                       # partner index
    g[i_range] = 2.0 * a[rev]                 # because derivative of a_i*a_j
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *exact* max‑convolution objective.
    Returns (best_a, best_obj).
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)                     # projection onto simplex

        if it % 100_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[ExpGrad] it={it:,}  new best = {cur:.6f}")

    # final check
    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS on a smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Log‑sum‑exp surrogate of the max‑convolution.
    Returns (obj, grad) where grad is the gradient w.r.t. the logits v.
    """
    a = _softmax(v)                          # probability vector (ℓ₁‑norm = 1)
    conv = _conv_fft(a, a)                    # full linear convolution
    shift = np.max(conv)                      # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_term)
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t a
    w = exp_term / sum_exp                    # soft‑max weights over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length = 3·n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]     # slice of length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 2000,
) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    The optimisation variable is the vector of logits (log‑softmax).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0]

    v = np.log(start_vec + 1e-30)            # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during β={beta_val}")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} (β={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Analytic seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    s = a.sum()
    return a / s


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, rng: np.random.Generator, max_m: int = 80) -> np.ndarray:
    """Pick a random support size m (2 ≤ m ≤ max_m) and spread mass uniformly."""
    m = rng.integers(2, max_m + 1)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step‑function within the given time budget.
    The algorithm consists of:
    1. Load the previous best solution (if any) and evaluate a few deterministic seeds.
    2. Perform a fast random‑parameter sweep (beta, exponential, power‑law,
       Dirichlet, sparse) and keep the best vector seen so far.
    3. Run a long exponentiated‑gradient (mirror‑descent) optimisation from the
       best seed; occasionally restart from a fresh random Dirichlet to escape
       local minima.
    4. Finish with a short L‑BFGS polishing on the smooth surrogate.
    5. Return the best sequence and store it globally.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting optimisation, n={n}, budget ≈ {budget_s}s ===")

    # ------------------------------------------------------------------
    # Initialise from previous run if possible
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            a0 = np.asarray(previous, dtype=np.float64)
            if a0.ndim == 1 and a0.size == n:
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum() if a0.sum() > 0 else 1.0
                print("[Info] Warm‑starting from previous solution.")
                best_a = a0.copy()
                best_val = _objective_exact(best_a)
                print(f"[Init] warm‑start objective = {best_val:.6f}")
            else:
                raise ValueError
        except Exception:
            a0 = None
    else:
        a0 = None

    # ------------------------------------------------------------------
    # Deterministic seed pool (very cheap)
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _triangular(n), _linear_decrease(n)]
    # add a few fixed beta‑shapes
    for alpha, beta in [(0.5, 2.0), (2.0, 0.5), (1.5, 1.5), (3.0, 1.0), (1.0, 3.0)]:
        seeds.append(_beta_shape(n, alpha, beta))
    # evaluate them
    best_a = best_a if a0 is not None else None
    best_val = best_val if a0 is not None else float("inf")
    for s in seeds:
        val = _objective_exact(s)
        if val < best_val:
            best_val = val
            best_a = s.copy()
    print(f"[Init] best deterministic seed = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fast random param sweep (uses remaining time aggressively)
    # ------------------------------------------------------------------
    param_start = time.time()
    sweep_counter = 0
    while time.time() < deadline - 20.0:      # leave ~20 s for the final phases
        sweep_counter += 1
        # choose family
        which = rng.integers(0, 5)
        if which == 0:                         # beta shape, log‑uniform params
            alpha = np.exp(rng.uniform(np.log(0.2), np.log(8.0)))
            beta = np.exp(rng.uniform(np.log(0.2), np.log(8.0)))
            cand = _beta_shape(n, alpha, beta)
        elif which == 1:                       # exponential decay
            lam = np.exp(rng.uniform(np.log(5e-4), np.log(0.02)))
            cand = _exponential_decay(n, lam)
        elif which == 2:                       # power‑law
            p = rng.uniform(0.0, 5.0)
            cand = _power_law(n, p)
        elif which == 3:                       # random Dirichlet
            conc = rng.uniform(0.05, 2.0)
            cand = _random_dirichlet(n, conc, rng)
        else:                                   # sparse uniform support
            cand = _random_sparse(n, rng, max_m=80)

        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            print(f"[Sweep] iter={sweep_counter:,} new best = {val:.6f}")

    print(f"[Sweep] completed {sweep_counter:,} random candidates")
    print(f"[Sweep] best after sweep = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long Exponentiated‑Gradient runs (restart‑friendly)
    # ------------------------------------------------------------------
    iters_per_restart = 250_000          # moderate block size
    while time.time() < deadline - 8.0:   # keep a few seconds for polishing
        remaining = deadline - time.time()
        max_iters = min(iters_per_restart, int(remaining * 2500))

        # occasional restart from a fresh random Dirichlet
        if rng.random() < 0.25:
            start_vec = _random_dirichlet(n, rng.uniform(0.05, 2.0), rng)
        else:
            start_vec = best_a.copy()

        seq, val = _exp_gradient_optimize(
            start_vec,
            n,
            deadline,
            max_iters=max_iters,
            eta0=1.2,
        )
        if val < best_val:
            best_val = val
            best_a = seq.copy()
            print(f"[ExpGrad] restart improvement → {val:.6f}")

    print(f"[ExpGrad] finished, best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final polishing with continuation L‑BFGS (high β)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(400, int(min(2500, remaining * 0.04 * 1e5)))
        print("\n--- Final L‑BFGS polishing (β schedule up to 1e5) ---")
        seq_final = _lbfgs_smooth_optimize(
            best_a,
            n,
            deadline,
            beta_schedule=[1e3, 5e3, 2e4, 1e5],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < best_val:
            best_val = val_final
            best_a = seq_final.copy()
            print(f"[Update] New best after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_a.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for warm‑starts in future runs
    globals()["height_sequence_1"] = best_a.tolist()
    return best_a.tolist()
