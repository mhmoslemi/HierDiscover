    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v_m = np.zeros_like(v) # not to be confused with variable name
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    for beta in beta_schedule:
        alpha_lr = 0.05  # base learning rate may need scaling depending on beta.
        # we can set lr = alpha_lr / np.sqrt(beta) maybe.

        # compute max iterations allowed within remaining time.
        remaining = deadline - time.time()
        if remaining <= 0: break
        # For rough, allocate a fixed count per beta: e.g., max_iter = min(5000, int(remaining*10)).
        max_iter = int(min(5000, max(500, remaining * 10))) # Hard to tune.

        for _ in range(max_iter):
            t += 1
            if time.time() > deadline:
                break

            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            # compute grad wrt a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]

            # chain rule to logits
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_v
            v_m = beta2 * v_m + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v_m / (1 - beta2 ** t)
            lr = alpha_lr / np.sqrt(beta)  # scaling? maybe fixed
            v = v - lr * m_hat / (np.sqrt(v_hat) + eps)

            # evaluate current surrogate
            cur_val = 2.0 * n * max_c  # surrogate exact for sum=1
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        # after each beta, we may want to set v to log of best_seq for fresh start
        v = np.log(best_seq + 1e-30)
    return best_seq, best_val

"""
Improved optimisation for the step‑function problem.

Algorithm
---------

1. **Initialisation** – Load the previous best sequence (``height_sequence_1``)
   if it exists and has the required length.  Otherwise generate a large
   collection of ℓ₁‑normalised candidates (uniform, power‑law, Gaussian,
   random Dirichlet draws, and many permutations) and keep the one with the
   smallest surrogate value 2·n·max(conv).

2. **Adam on a smooth max‑convolution** – Optimise the log‑softmax
   parameters (logits) of the sequence with the Adam optimiser.
   The smooth approximation of the max is
   ``smooth(β) = shift + log(sum exp(β·(conv‑shift)))/β``.
   A schedule of increasing β (1 → 5 → 20 → … → 5·10⁵) makes the surrogate
   converge to the true max.  For each β we perform many Adam steps,
   constantly projecting back onto the probability simplex via the softmax.

3. **Perturb‑restart** – After a full β‑schedule the logits are perturbed
   by a tiny Gaussian noise and the optimisation restarts a few times.
   This helps escape shallow local minima.

4. **Projected sub‑gradient polishing** – A final long sub‑gradient
   descent on the exact surrogate ``2·n·max(conv)`` (with projection onto
   the simplex) tightens the bound further.

5. **Evaluation** – The resulting sequence is handed to the official
   ``evaluate_sequence`` function, stored in ``height_sequence_1`` and
   returned.

All stages respect the global time‑budget (default 1000 s).  The code
uses only ``numpy`` and the standard library, runs on a single CPU and
produces a non‑negative ℓ₁‑normalised sequence (scaling is irrelevant for
the objective because it is homogeneous of degree 0).
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by the official evaluator."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Permutation helpers (enrich the seed pool)
# ----------------------------------------------------------------------
def _alternating_order_from_vec(vec: np.ndarray) -> np.ndarray:
    sorted_desc = np.sort(vec)[::-1]
    n = len(vec)
    out = np.empty_like(vec)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out


def _shuffle_by_step(sorted_desc: np.ndarray, step: int) -> np.ndarray:
    n = len(sorted_desc)
    out = np.empty(n, dtype=np.float64)
    pos = 0
    for val in sorted_desc:
        out[pos] = val
        pos = (pos + step) % n
    return out


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse set of ℓ₁‑normalised candidate vectors.
    """
    seeds: list[np.ndarray] = []

    # analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.12, 0.18, 0.25):
        seeds.append(_gaussian(n, sigma))
    for p in (0.0, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))

    # random Dirichlet draws + permutations
    conc_list = (0.05, 0.2, 0.5, 1.0, 2.0)
    step_list = [3, 7, 11, 13, 17, 19, 23, 29, 31, 37]
    for conc in conc_list:
        for _ in range(8):          # more draws than the original version
            raw = _random_dirichlet(n, conc, rng)

            # raw (already normalised)
            seeds.append(raw)

            # descending order
            desc = np.sort(raw)[::-1]
            seeds.append(desc)

            # alternating order
            seeds.append(_alternating_order_from_vec(raw))

            # modular step permutations
            for step in step_list[:5]:
                seeds.append(_shuffle_by_step(desc, step))

            # random permutation
            perm = rng.permutation(n)
            seeds.append(raw[perm])

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final normalisation and sanitisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Adam optimiser on smooth max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule: list[float]) -> tuple[np.ndarray, float]:
    """
    Optimise the log‑softmax parameters of ``start_seq`` with Adam.
    Uses a smooth log‑sum‑exp approximation of the max‑convolution.
    """
    # initialise logits
    v = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(v)
    v_m = np.zeros_like(v)   # second moment (named v_m to avoid clash)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # how many steps may we afford for this β?
        # a very rough estimate: ~200 iters per second of remaining wall‑time
        remaining = deadline - time.time()
        max_iter = int(min(30_000, max(2_000, remaining * 200)))
        # safety guard – never exceed the deadline inside the loop
        for _ in range(max_iter):
            t += 1
            if time.time() > deadline:
                break

            a = _softmax(v)                       # stays on the simplex
            conv = _conv_fft(a, a)                 # full convolution
            max_c = np.max(conv)

            # smooth‑max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp                 # soft weights for the gradient

            # gradient w.r.t. a (see derivation in the original code)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_m = beta2 * v_m + (1.0 - beta2) * (grad_v ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v_m / (1.0 - beta2 ** t)

            # learning‑rate scaling – larger β → sharper landscape → smaller step
            lr = 0.05 / (np.sqrt(beta) + 1e-12)
            v = v - lr * m_hat / (np.sqrt(v_hat) + eps)

            # current surrogate (exact for ℓ₁‑normalised a)
            cur_val = 2.0 * n * max_c
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                # occasional progress report
                if t % 5_000 == 0:
                    print(f"[Adam] it={t:6d} β={beta:.0f} best={best_val:.6f}")

        # after finishing this β restart from the best point found so far
        v = np.log(best_seq + 1e-30)
        m.fill(0.0)
        v_m.fill(0.0)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Long sub‑gradient descent on 2·n·max(conv) for ℓ₁‑normalised vectors."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)   # project back onto the simplex

    return best_seq, best_val

    scored.sort(key=lambda x: x[1])

    # --------------------------------------------------------------

    global_best_seq = best_seed.copy()
    global_best_val = best_val

    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,

    # multiple restart rounds (perturb‑and‑re‑optimise)

        if time.time() > deadline:

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()

        # small random perturbation of logits for the next restart

    return global_best_seq.tolist()
