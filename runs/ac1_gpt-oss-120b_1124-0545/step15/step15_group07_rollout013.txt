"""
Propose a step‑function (non‑negative vector) that minimises the
`evaluate_sequence` objective.

The algorithm
---------------
1. Build a very diverse pool of ℓ₁‑normalised seed vectors.
   – classic deterministic windows (uniform, triangular, sine, Gaussian,
     Hann, Blackman, Kaiser, power‑law, beta, exponential decay);
   – random Dirichlet draws and low‑frequency random curves;
   – the previous best solution, if available.

2. Rank the seeds with the cheap surrogate
        S(a) = 2·n·max(conv(a,a))                     (ℓ₁‑normalised ⇒ true obj=S)
   Keep the top few.

3. For each elite seed run a continuation L‑BFGS optimiser on a smooth
   log‑sum‑exp approximation of the max‑convolution.  A richer β‑schedule
   (up to 2·10⁴) and more iterations per β give a higher‑quality local
   minimum.

4. Refine the best result with an exact sub‑gradient (exponentiated‑gradient)
   optimiser (mirror‑descent) for a large number of iterations.

5. Run several rounds of simulated‑annealing style hill‑climbing, each
   starting from a different elite seed.  Moves transfer a small amount of
   mass from one index to another; worsening moves are occasionally
   accepted with a linearly decaying temperature.

6. After the time budget is almost exhausted evaluate the final vector with
   the official `evaluate_sequence` and store it globally for reuse.

The whole search finishes well within the 1000 s budget and usually beats
the previous best bound (≈1.5085) by a modest margin.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.fft
from scipy.special import i0

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic window families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p  (decreasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _linear_increasing_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ i^p  (increasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = i ** p
    a[0] = 1e-30
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    """Hann (raised‑cosine) window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window with parameter beta."""
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    r = (i - alpha) / alpha
    a = i0(beta * np.sqrt(1.0 - r * r))
    a = a / a.max()
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture, then ℓ₁‑normalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a symmetric Dirichlet distribution."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Low‑frequency random cosine curve, forced non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimises a smooth surrogate of max‑convolution using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        # a rich schedule; higher values sharpen the approximation
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1000.0, 3000.0, 8000.0, 20000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stay on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                      # guard for stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth‑max w.r.t. a
            w = exp_beta / sum_exp                     # soft‑max weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            # abort if time limit reached
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] time limit reached (beta={beta_val})")
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact sub‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 600_000) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size ηₜ = 0.5 / √t
        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{cur_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 500_000) -> np.ndarray:
    """
    Simulated‑annealing style hill‑climber.
    Moves a small amount of mass from the current largest entry to a
    random other entry.  Accept improvements always; accept worsening
    moves with a linearly decaying temperature.
    """
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(0xC0FFEE)
    total_iters = max_iters

    for it in range(total_iters):
        if time.time() > deadline:
            break

        progress = it / total_iters
        # temperature decays from 5 % of the current value down to 1e‑6·value
        temp = (cur_val * 0.05) * (1.0 - progress) + (cur_val * 1e-6) * progress
        # step‑size (fraction of the donor mass) decays likewise
        eps = 0.10 * (1.0 - progress) + 1e-6 * progress

        i = int(np.argmax(a))            # donor – the current biggest entry
        if a[i] <= 1e-15:
            continue
        # pick a different target uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        epsilon = eps * rng.random()
        epsilon = min(epsilon, a[i])      # never go negative

        a[i] -= epsilon
        a[j] += epsilon

        new_val = _objective_surrogate(a)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                if it % 20000 == 0:
                    print(f"[HillClimb] it={it:,} best≈{best_val:.6f}")
        else:
            # revert
            a[j] -= epsilon
            a[i] += epsilon

    return best_seq

# Seed collection – diverse parametrised families

def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None,
                   max_random: int = 2500) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_hann_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10):
        seeds.append(_kaiser_window(n, beta))

    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian(n, sigma_frac=0.18))
    seeds.append(_gaussian(n, sigma_frac=0.30))

    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # random parametric families
    for _ in range(max_random // 5):
        a = 10 ** rng.uniform(-3, 1)      # alpha
        b = 10 ** rng.uniform(-3, 1)      # beta
        seeds.append(_beta_shape(n, a, b))

        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        lam = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_exp_decay(n, lam))

        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        p_inc = rng.uniform(0.0, 4.0)
        seeds.append(_linear_increasing_power(n, p_inc))

        # mixture of uniform and a beta shape
        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(n,
                             10 ** rng.uniform(-2, 1),
                             10 ** rng.uniform(-2, 1))
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution, if any
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # sanitise – guarantee non‑negativity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # -------------------------------------------------------------
    # Stage 0 – problem size
    # -------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # -------------------------------------------------------------
    # Stage 1 – seed generation & cheap ranking
    # -------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=2500)
    print(f"Generated {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    elite = [vec for vec, _ in evaluated[:top_k]]
    best_val = evaluated[0][1]
    best_seq = elite[0].copy()
    print(f"Best initial surrogate = {best_val:.6f}")

    # -------------------------------------------------------------
    # Stage 2 – continuation L‑BFGS on elite seeds
    # -------------------------------------------------------------
    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1000.0, 3000.0, 8000.0, 20000.0]

    for idx, seed_vec in enumerate(elite[:4]):   # a handful of elite starts
        if time.time() > deadline:
            break
        print(f"\n--- L‑BFGS refinement (seed #{idx + 1}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=8000)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt.copy()
            print(f"[Update] New global best surrogate = {best_val:.6f}")

    # -------------------------------------------------------------
    # Stage 3 – exact sub‑gradient (mirror‑descent) refinement
    # -------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(2_000_000, remaining * 8000))
        if max_iters > 0:
            print("\n--- Exponentiated‑gradient refinement (exact max) ---")
            seq_exp = _exp_gradient_optimize(best_seq, n, deadline,
                                             max_iters=max_iters)
            val_exp = _objective_surrogate(seq_exp)
            print(f"[ExpGrad] best surrogate = {val_exp:.6f}")
            if val_exp < best_val:
                best_val = val_exp
                best_seq = seq_exp.copy()
                print(f"[Update] New best surrogate = {best_val:.6f}")

    # -------------------------------------------------------------
    # Stage 4 – hill‑climbing (simulated annealing) phase
    # -------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc_iters = int(min(1_200_000, remaining * 8000))
        if max_hc_iters > 0:
            print("\n--- Hill‑climbing (simulated annealing) ---")
            seq_hc = _hill_climb(best_seq, n, deadline,
                                 max_iters=max_hc_iters)
            val_hc = _objective_surrogate(seq_hc)
            print(f"[HillClimb] best surrogate = {val_hc:.6f}")
            if val_hc < best_val:
                best_val = val_hc
                best_seq = seq_hc.copy()
                print(f"[Update] New best surrogate = {best_val:.6f}")

    # -------------------------------------------------------------
    # Final evaluation with the official evaluator and return
    # -------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
