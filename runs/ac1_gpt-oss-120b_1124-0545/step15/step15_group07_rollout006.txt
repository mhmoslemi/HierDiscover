def _exp_gradient_adam(start_seq, n, deadline, max_iters):

   logits = np.log(start_seq + 1e-30)
   m = np.zeros_like(logits)
   v = np.zeros_like(logits)
   beta1=0.9
   beta2=0.999
   epsilon=1e-8
   lr = 0.2 # base learning rate
   best_seq = softmax(logits)
   best_val = _objective_surrogate(best_seq)
   t = 0

   while t<max_iters and time.time()<deadline:
       t+=1
       a = _softmax(logits)
       conv = _conv_fft(a, a)
       max_idx = int(np.argmax(conv))
       g = np.zeros_like(a)
       i_start = max(0, max_idx-(n-1))
       i_end = min(n-1, max_idx)
       idx = np.arange(i_start, i_end+1, dtype=int)
       rev_idx = max_idx - idx
       g[idx] = 2.0 * a[rev_idx]
       
       grad_a = g
       dot = np.dot(grad_a, a)
       grad_v = a * (grad_a - dot)
       
       m = beta1*m + (1-beta1)*grad_v
       v = beta2*v + (1-beta2)*(grad_v**2)
       
       m_hat = m / (1 - beta1**t)
       v_hat = v / (1 - beta2**t)
       
       step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
       
       logits = logits - step
       
       if t % 1000 == 0:
           cur_seq = _softmax(logits)
           cur_val = _objective_surrogate(cur_seq)
           if cur_val < best_val:
               best_val = cur_val
               best_seq = cur_seq.copy()
               # optionally print progress
       
   return best_seq

"""
Improved step‑function search for ``evaluate_sequence``.

Strategy
--------
1. **Rich parametric seed pool** – systematic sweep over a few low‑dimensional
   families (exponential decay, symmetric mixture of two exponentials,
   Gaussian windows, power‑law, Beta‑shapes, random Dirichlet draws, etc.).
   The surrogate  `S(a)=2·n·max(conv(a,a))` (identical to the true objective
   when Σa=1) is computed by a fast FFT convolution and the best ~15 seeds
   are kept.

2. **Continuation L‑BFGS on a smooth max** – the exact max is approximated
   by a log‑sum‑exp with a schedule of increasing inverse‑temperature
   ``beta``.  Each stage runs more iterations than the reference code
   (up to 12 000) and the schedule extends to ``beta=20000``.
   The result stays on the simplex via a soft‑max parametrisation.

3. **Exact mirror‑descent (Adam‑accelerated exponentiated‑gradient)** –
   optimisation of the true max‑convolution.  A Adam‑style adaptive step
   size replaces the hand‑crafted 0.5/√t schedule, allowing many more
   effective updates before the deadline.

4. **Local stochastic hill‑climbing** – small random mass transfers are
   repeatedly applied; uphill moves are accepted with a linearly decaying
   temperature.  The move set is richer: either move mass from a *large*
   entry to a random one *or* swap a tiny fragment between two random
   positions.

5. **Optional refinement round** – if time permits a second L‑BFGS /
   Adam pass is performed on the current best sequence.

The whole pipeline is time‑aware: every heavy loop checks the remaining
budget and aborts cleanly.  The best sequence found is stored in the
global name ``height_sequence_1`` for reuse by subsequent runs.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector
    (identical to the true objective when Σseq==1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _sym_exp_mix(n: int, lam: float, alpha: float) -> np.ndarray:
    """
    Convex mixture of a left‑hand exponential and its reversed copy.
    alpha∈[0,1] is the weight of the right‑hand component.
    """
    left = _exp_decay(n, lam)
    right = left[::-1]
    mixed = (1.0 - alpha) * left + alpha * right
    return mixed / mixed.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# L‑BFGS optimiser of the smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimises a smooth surrogate of ``max(conv)`` using L‑BFGS‑B.
    The optimisation lives in the space of logits (soft‑max) so the
    ℓ₁‑normalisation is enforced automatically.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0, 10000.0, 20000.0]

    logits = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v):
            a = _softmax(v)                         # stay on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                    # numeric guard
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)              # length = 2n‑1
            grad_a = 2.0 * full[n - 1:n - 1 + n]    # same length as a

            # back‑prop through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        # L‑BFGS‑B on the logits
        try:
            res = _opt.minimize(fun=fun,
                                x0=logits,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-13,
                                         'disp': False})
            logits = res.x
        except KeyboardInterrupt:
            # time limit reached
            break
        except Exception as e:
            # unexpected problem – abort this beta level
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(logits)


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser (Adam‑accelerated exponentiated gradient)
# ----------------------------------------------------------------------
def _exp_gradient_adam(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 2_000_000) -> np.ndarray:
    """
    Mirror‑descent on the exact ``max(conv)`` using an Adam‑style
    adaptive step for the logits.  Returns the best sequence seen.
    """
    # initialise logits so that softmax(logits) = start_seq
    logits = np.log(start_seq + 1e-30)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta0 = 0.2                      # base learning‑rate

    best_seq = _softmax(logits)
    best_val = _objective_surrogate(best_seq)

    it = 0
    while it < max_iters:
        if time.time() > deadline:
            break
        it += 1

        a = _softmax(logits)

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # gradient w.r.t. logits (mirrored‑gradient step)
        grad_a = g
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        logits -= step

        # periodic check for improvement
        if it % 2000 == 0:
            cur_seq = _softmax(logits)
            cur_val = _objective_surrogate(cur_seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[Adam‑EG] it={it:,} best≈{best_val:.6f}")

    # final evaluation
    final_seq = _softmax(logits)
    final_val = _objective_surrogate(final_seq)
    if final_val < best_val:
        best_seq = final_seq.copy()
        best_val = final_val
        print(f"[Adam‑EG] final improvement to {best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Hill‑climbing / Simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 400_000) -> np.ndarray:
    """
    Simple stochastic hill‑climber.  In each iteration a random amount of
    mass is transferred either *from* the largest entry OR *between* two
    random positions.  Improvements are always accepted; uphill moves
    follow a linearly decaying temperature schedule.
    """
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(0xC0FFEE)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        progress = it / max_iters
        temp = cur_val * (0.05 * (1.0 - progress) + 1e-6 * progress)
        eps = 0.10 * (1.0 - progress) + 1e-6 * progress

        # decide which move type to perform
        if rng.random() < 0.5:
            # move mass from the current largest entry to a random other entry
            i = int(np.argmax(a))
            if a[i] <= 1e-15:
                continue
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
        else:
            # swap a tiny fragment between two random indices
            i, j = rng.choice(n, size=2, replace=False)

        # amount to shift
        epsilon = eps * rng.random()
        epsilon = min(epsilon, a[i])            # never go negative

        # apply move
        a[i] -= epsilon
        a[j] += epsilon

        new_val = _objective_surrogate(a)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                if it % 20_000 == 0:
                    print(f"[HillClimb] it={it:,} best≈{best_val:.6f}")
        else:
            # revert
            a[j] -= epsilon
            a[i] += epsilon

    return best_seq

# Seed generation – systematic parametric sweep + random draws
# ----------------------------------------------------------------------
def _collect_rich_seeds(n: int,
                       rng: np.random.Generator,
                       previous: np.ndarray | None,
                       max_random: int = 1500) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic / classic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # systematic exponential‑decay sweep
    lam_grid = np.logspace(-5, -2.5, 30)               # 1e‑5 … ~3e‑3
    for lam in lam_grid:
        seeds.append(_exp_decay(n, lam))

        # symmetric mixture (two‑sided exponential)
        for alpha in np.linspace(0.1, 0.9, 9):
            seeds.append(_sym_exp_mix(n, lam, alpha))

    # Gaussian sweep
    for sigma in np.linspace(0.05, 0.35, 25):
        seeds.append(_gaussian(n, sigma_frac=sigma))

    # power‑law sweep
    for p in np.linspace(0.0, 4.5, 23):
        seeds.append(_power(n, p))

    # Beta‑shape sweep (log‑space for α,β)
    for log_alpha in np.linspace(-3, 1, 15):
        for log_beta in np.linspace(-3, 1, 15):
            alpha = 10 ** log_alpha
            beta = 10 ** log_beta
            seeds.append(_beta_shape(n, alpha, beta))

    # random Dirichlet draws with varied concentration
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final sanitisation – guarantee positivity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    # limit the total number of seeds to keep the cheap evaluation cheap
    if len(cleaned) > max_random:
        rng.shuffle(cleaned)
        cleaned = cleaned[:max_random]
    return cleaned

                val_final = _objective_surrogate(seq_final)
                print(f"[2nd‑Adam‑EG] surrogate = {val_final:.6f}")

                if val_final < global_best_val:
                    global_best_seq = seq_final.copy()
                    global_best_val = val_final
                    print(f"[Update] New best after 2nd Adam‑EG = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
