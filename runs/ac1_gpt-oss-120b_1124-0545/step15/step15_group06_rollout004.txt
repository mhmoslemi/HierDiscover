2. **Fast evaluation** – use a hand‑written FFT based convolution
   (`_objective_raw`) to obtain the exact objective.  The best few candidates
   are remembered while the random search runs (≈70 % of the time budget).

3. **Mirror‑descent refinement** – the current best vector is refined on the
   true non‑smooth objective with exponentiated‑gradient updates
   (`_exp_gradient_optimize`).  The step‑size is η₀/√t (η₀≈2).  The number of
   iterations is limited by the remaining time.

4. **High‑β L‑BFGS polishing** – finally run a few L‑BFGS passes on the
   smooth log‑sum‑exp surrogate with a very large β (up to 2·10⁴) to push the
   solution as close as possible to the true max‑convolution.

5. **Return** – the best sequence found is stored in the global name
   ``height_sequence_1`` (for warm‑starts) and returned as a plain Python list.
"""
import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors, length n)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on {1,…,n} (U‑shape when α,β<1)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-c * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_decay(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α)+b·α then ℓ₁‑renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Exact objective utilities
# ----------------------------------------------------------------------
def _objective_raw(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) / (sum²)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (using the active lag).
    Returns g = ∂ max(conv) / ∂ a .
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


def _exp_gradient_optimize(start_a: np.ndarray,
                          deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 2.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact objective.
    Returns the best vector seen and its objective value.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_raw(a)

    it = 0
    while it < max_iters:
        if time.time() > deadline:
            break
        it += 1
        g = _subgradient_maxconv(a)          # sub‑gradient of max(conv)
        eta = eta0 / math.sqrt(it)           # diminishing steps
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 100_000 == 0:
            cur = _objective_raw(a)
            print(f"[ExpGrad] it={it:,} obj={cur:.6f}")
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
    # final check
    cur = _objective_raw(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Smooth surrogate (log‑sum‑exp) for max‑convolution."""
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)              # full auto‑convolution
    shift = np.max(conv)                # numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # back‑prop through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    L‑BFGS‑B optimiser of the smooth surrogate.
    Variable = logits of a softmax (unconstrained real vector).
    """
    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0,
                         5000.0, 10000.0, 20000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during beta={beta_val}")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Randomised shape generators (used in the exploratory stage)
# ----------------------------------------------------------------------
def _rand_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta shape with log‑uniform α,β in [1e‑3, 2]."""
    log_a = rng.uniform(-3.0, np.log10(2.0))
    log_b = rng.uniform(-3.0, np.log10(2.0))
    alpha = 10 ** log_a
    beta = 10 ** log_b
    a = _beta_shape(n, alpha, beta)
    shift = rng.integers(0, n)
    return np.roll(a, shift)


def _rand_exp_decay(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-c·i) with c chosen log‑uniform."""
    c = rng.uniform(0.001, 0.02)          # decay rate
    a = _exponential_decay(n, c)
    shift = rng.integers(0, n)
    return np.roll(a, shift)


def _rand_power_decay(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law decay a_i ∝ 1/(i+1)^p with p in [0,5]."""
    p = rng.uniform(0.0, 5.0)
    a = _power_decay(n, p)
    shift = rng.integers(0, n)
    return np.roll(a, shift)


def _rand_mixed_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two independent beta shapes."""
    a1 = _rand_beta(n, rng)
    a2 = _rand_beta(n, rng)
    w = rng.random()
    mix = _mix_shapes(a1, a2, w)
    shift = rng.integers(0, n)
    return np.roll(mix, shift)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    ``evaluate_sequence`` under the given time budget.  Returns the best
    sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0     # safety margin

    n = int(kwargs.get("n", 4000))
    print(f"\n=== Optimisation start (n={n}) ===")

    # ------------------------------------------------------------------
    # Stage 0 – random parametric search (≈70 % of budget)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')

    # seeds from previous run (warm‑start) – give them a head start
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape == (n,) and prev_arr.sum() > 0:
                best_seq = prev_arr / prev_arr.sum()
                best_val = _objective_raw(best_seq)
                print(f"[Warm] previous value = {best_val:.6f}")
        except Exception:   # pragma: no cover
            pass

    # function list for random generation
    gen_fns = [_rand_beta, _rand_exp_decay, _rand_power_decay, _rand_mixed_beta]

    random_deadline = start_time + 0.70 * (deadline - start_time)
    eval_cnt = 0
    while time.time() < random_deadline:
        fn = rng.choice(gen_fns)
        cand = fn(n, rng)                     # already ℓ₁‑normalised
        # occasional permutation to break symmetry
        if rng.random() < 0.08:
            cand = rng.permutation(cand)
        val = _objective_raw(cand)
        eval_cnt += 1
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Random] new best #{eval_cnt:,} obj={best_val:.6f}")

    print(f"\nRandom search finished – best obj={best_val:.6f} after {eval_cnt:,} evaluations")

    # ------------------------------------------------------------------
    # Stage 1 – exponentiated‑gradient refinement (exact objective)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate iterations proportional to remaining time
        max_iters = int(min(3_000_000, remaining * 2e5))
        print(f"\n--- Exp‑gradient refinement (≤{max_iters:,} iters) ---")
        refined_seq, refined_val = _exp_gradient_optimize(best_seq,
                                                          deadline,
                                                          max_iters=max_iters,
                                                          eta0=2.0)
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"[ExpGrad] improvement → obj={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – high‑β L‑BFGS polishing (smooth surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta_iters = max(800, int(min(2500, remaining * 5e4)))
        beta_sched = [1.0, 20.0, 200.0, 1000.0, 5000.0, 10000.0, 20000.0]
        print("\n--- High‑β L‑BFGS polishing ---")
        polished = _lbfgs_smooth_optimize(best_seq,
                                          n,
                                          deadline,
                                          beta_schedule=beta_sched,
                                          max_iter_per_beta=per_beta_iters)
        polished_val = _objective_raw(polished)
        print(f"[L‑BFGS] polished obj={polished_val:.6f}")
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished.copy()
            print(f"[Update] new global best after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official `evaluate_sequence`
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = best_val

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
