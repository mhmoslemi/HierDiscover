"""
Propose a step‑function (non‑negative ℓ₁‑normalised vector) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

within a ~1000 s budget.

Approach
--------
1. Generate a large, diverse pool of candidate vectors (deterministic windows,
   random Dirichlet draws, low‑frequency curves, etc.) and rank them cheaply
   by the surrogate 2·n·max(conv) (no normalisation required).
2. Take the best few seeds and refine them with a *continuation* L‑BFGS
   optimiser.  The objective is a smooth approximation of the max‑convolution
   obtained via log‑sum‑exp with a temperature β that is increased stepwise.
   Parameters are represented by logits (soft‑max) so that non‑negativity and
   ℓ₁‑normalisation are enforced automatically.
3. Starting from the best L‑BFGS result run a *mirror‑descent* (exponentiated‑
   gradient) phase using a Polyak step‑size based on the known lower bound
   1/(2·n‑1).  This directly minimises the exact max‑convolution.
4. Finish with a few jittered L‑BFGS polishing runs.
5. Return the best sequence found and store it in the global variable
   ``height_sequence_1`` for reuse.

The combination of a powerful second‑order optimiser (L‑BFGS) for the smooth
approximation and a fast exact sub‑gradient phase yields a tighter bound than
the previous Adam‑only pipeline.
"""

import time
import numpy as np
import scipy.optimize as _opt
from scipy.special import i0  # Kaiser window helper


# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max (always ℓ₁‑normalised, non‑negative)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """Surrogate = 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector."""
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)ᵖ  (decreasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window, β≥0 (β=0 yields a rectangular window)."""
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    w = i0(beta * np.sqrt(1.0 - t * t)) / i0(beta)
    return w / w.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.16
    w = (1 - a) / 2 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + a / 2 * np.cos(4 * np.pi * i / (n - 1))
    return w / w.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return w / w.sum()


def _tukey_window(n: int, alpha: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1)
    w = np.ones_like(i)
    mask1 = i < alpha / 2
    mask2 = i > 1 - alpha / 2
    w[mask1] = 0.5 * (1 + np.cos(np.pi * (2 * i[mask1] / alpha - 1)))
    w[mask2] = 0.5 * (1 + np.cos(np.pi * (2 * (i[mask2] - 1) / alpha + 1)))
    return w / w.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()

    seeds: list[np.ndarray] = []

    seeds.append(_sine_window(n, gamma=0.5))

    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_kaiser_window(n, beta=4.0))

    seeds.append(_hamming_window(n))
    seeds.append(_tukey_window(n, alpha=0.4))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # random draws from parametric families
    for _ in range(max_random // 6):
        a = 10 ** rng.uniform(-3, 1)
        b = 10 ** rng.uniform(-3, 1)
        seeds.append(_beta_shape(n, a, b))

        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        lam = rng.exponential(scale=0.001) + 1e-4
        seeds.append(_exp_decay(n, lam))

        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        beta = rng.uniform(0.0, 12.0)
        seeds.append(_kaiser_window(n, beta))

        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(
            n,
            10 ** rng.uniform(-2, 1),
            10 ** rng.uniform(-2, 1),
        )
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # Dirichlet draws (different concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # sanitise – guarantee non‑negativity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)

# ----------------------------------------------------------------------
# Smooth‑max loss (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------
def _smooth_max_loss(logits: np.ndarray, beta: float, n: int):
    """
    Returns (loss, grad) where loss = smooth approximation of max(conv(a,a))
    with a = softmax(logits).  Gradient is w.r.t. the logits.
    """
    a = _softmax(logits)
    conv = _conv_fft(a, a)               # length 2n‑1
    shift = np.max(conv)                 # stability shift (max of conv)
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    # smooth max = shift + (log sum_exp) / beta
    loss = shift + np.log(sum_exp) / beta

    # weights for the smooth max
    w = exp_beta / sum_exp               # shape (2n‑1,)

    # gradient w.r.t. a : 2 * (rev_a * w) convolution
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]

    # back‑prop through the soft‑max
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return loss, grad_logits


# ----------------------------------------------------------------------
# Second‑order optimisation of the smooth‑max (L‑BFGS)
# ----------------------------------------------------------------------
def _l_bfgs_optimize(
    start_vec: np.ndarray,
    n: int,
    beta_schedule: list[float],
    deadline: float,
) -> np.ndarray:
    """
    Perform a continuation L‑BFGS optimisation on the smooth‑max loss.
    Returns the best ℓ₁‑normalised vector encountered.
    """
    logits = np.log(start_vec + 1e-30)   # initialise in log‑softmax space
    best_seq = None
    best_val = np.inf

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # allocate a modest number of L‑BFGS iterations; adapt to remaining time
        remaining = deadline - time.time()
        max_iters = max(200, min(3000, int(remaining * 150)))  # heuristic

        # minimise the smooth‑max loss for the current β
        res = _opt.minimize(
            fun=_smooth_max_loss,
            x0=logits,
            args=(beta, n),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iters, "disp": False},
        )
        logits = res.x
        seq = _softmax(logits)
        val = _objective_surrogate(seq, n)
        print(f"[L‑BFGS] β={beta:.0f} → surrogate={val:.6f} (iters={res.nit})")
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    return best_seq if best_seq is not None else _softmax(logits)


# ----------------------------------------------------------------------
# Exact max‑convolution mirror‑descent (exponentiated gradient)
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Uses a Polyak step‑size based on the trivial lower bound 1/(2n‑1).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)
    lower = 1.0 / (2 * n - 1)            # average of convolution entries

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step‑size (ensure positivity)
        norm_sq = np.dot(g, g)
        if norm_sq > 0.0:
            eta = max((max_val - lower) / norm_sq, 0.0)
        else:
            eta = 0.0

        # mirror‑descent update (log‑space, then softmax)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{cur_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Polishing via short L‑BFGS runs from jittered starting points
# ----------------------------------------------------------------------
def _polish_with_jitter(
    base_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    attempts: int = 4,
) -> np.ndarray:
    """
    Apply a few short L‑BFGS refinements starting from jittered logits.
    Keeps any improvement.
    """
    best_seq = base_seq.copy()
    best_val = _objective_surrogate(best_seq, n)

    for k in range(attempts):
        if time.time() > deadline:
            break
        logits = np.log(best_seq + 1e-30)
        jitter = rng.normal(scale=0.025, size=n)
        logits_j = logits + jitter

        refined = _l_bfgs_optimize(
            start_vec=_softmax(logits_j),
            n=n,
            beta_schedule=[5000.0, 10000.0, 20000.0],
            deadline=deadline,
        )
        cur_val = _objective_surrogate(refined, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = refined
            print(f"[Polish] improvement after jitter #{k+1}: {cur_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # Problem size (trade‑off between quality and speed)
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # --------------------------------------------------------------
    # Seed generation + cheap surrogate ranking
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=2500)
    print(f"Generated {len(seeds)} seed vectors.")

    # cheap evaluation (2·n·max(conv)) → keep top few
    evaluated = [(vec, _objective_surrogate(vec, n)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # --------------------------------------------------------------
    # Continuation L‑BFGS optimisation on elite seeds
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq, n)
    print(f"\n[Init] Best surrogate = {global_best_val:.6f}")

    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                  1000.0, 3000.0, 5000.0, 10000.0, 20000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all elite seeds.")
            break

        print(f"\n--- L‑BFGS on elite seed #{idx+1} ---")
        refined = _l_bfgs_optimize(
            start_vec=seed_vec,
            n=n,
            beta_schedule=beta_sched,
            deadline=deadline,
        )
        val_refined = _objective_surrogate(refined, n)
        print(f"[L‑BFGS] refined surrogate = {val_refined:.6f}")

        if val_refined < global_best_val:
            global_best_seq = refined.copy()
            global_best_val = val_refined
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Exact max‑convolution mirror‑descent refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(1_000_000, remaining * 2500))  # heuristic
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient (exact max) refinement ---")
            seq_exp = _exp_gradient_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_exp_iters,
            )
            val_exp = _objective_surrogate(seq_exp, n)
            print(f"[ExpGrad] refined surrogate = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] New global best after ExpGrad = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final polishing with jittered L‑BFGS runs
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Polishing phase (jitter + L‑BFGS) ---")
        polished = _polish_with_jitter(
            global_best_seq, n, deadline, rng, attempts=3
        )
        val_polish = _objective_surrogate(polished, n)
        if val_polish < global_best_val:
            global_best_seq = polished.copy()
            global_best_val = val_polish
            print(f"[Polish] improved surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
