"""
Propose a low‑max‑auto‑convolution step function.

The algorithm works in several stages:

1. Build a large, diverse pool of ℓ₁‑normalised candidates (analytic windows,
   Dirichlet / exponential draws, low‑frequency curves, the previous best and
   a finely tuned power‑law family).
2. Keep the best few seeds w.r.t. the exact surrogate
   2·n·max(conv(a,a)).
3. Run a continuation L‑BFGS optimisation (smooth‑max via log‑sum‑exp)
   on each top seed, allowing many iterations and a very fine β‑schedule.
4. Apply a deterministic linear‑programming step: with the current vector
   `f` and its max‑convolution `M = max(f*f)`, solve

        maximise   Σ g_i
        subject to conv(f,g) ≤ M   (element‑wise)
                    g_i ≥ 0

   The optimal `g` spreads the mass as evenly as possible while respecting the
   current bound.  After normalising `g` to ℓ₁‑norm 1 the new maximum
   convolution can only decrease.  The LP is solved with HiGHS (SciPy).
5. A short exponentiated‑gradient (mirror‑descent) polishing pass refines
   the final vector.
6. The best vector is returned as a plain Python list.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact surrogate (the quantity the official evaluator returns for ℓ₁‑normed vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – exact surrogate for a ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # Bessel I0 (NumPy provides it)
    return a / a.sum()


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    return a / a.sum()


def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a large pool of diverse ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    # Gaussians
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_gaussian(n, 0.15))

    # Power‑law (coarse grid – refined later)
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta‑family grid (coarse)
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Alternating‑extrema Dirichlet seed
    seeds.append(_alternating_dirichlet(n, rng))

    # Previous solution, if appropriate
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise everything robustly
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        cleaned.append(v / s if s > 0 else _uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 12000) -> np.ndarray:
    """Minimise a smooth approximation of the max‑convolution objective."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0, 10000.0,
                         20000.0, 50000.0, 100000.0]

    # logits (log‑softmax parameters) – stay on the simplex after softmax
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            mx = np.max(conv)
            # log‑sum‑exp smooth max
            shift = mx
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                     # soft weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print("[L‑BFGS] exception:", e)
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser on the exact max‑conv
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 250_000) -> np.ndarray:
    """Mirror‑descent (exponentiated gradient) on the exact max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)               # diminishing step size
        log_a = log_a - eta * g
        a = _softmax(log_a)

    return a


# ----------------------------------------------------------------------
# Linear‑programming bound‑reduction step
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               max_iters: int = 3,
               tol: float = 1e-12) -> np.ndarray:
    """
    Given a current vector f (ℓ₁‑norm 1) solve the LP

        maximise   Σ g_i
        subject to conv(f,g) ≤ M   (element‑wise)
                    g_i ≥ 0

    where M = max(conv(f,f)).  Return the normalised optimiser g_norm.
    """
    if _opt is None or _sp is None:
        # fall back – nothing to do
        return f

    rows = 2 * n - 1

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv_f = _conv_fft(f, f)
        M = float(np.max(conv_f))

        # Build sparse matrix A where A @ g = conv(f,g)
        # A_{k,i} = f_{k-i}  if 0 ≤ k-i < n else 0
        # This is a Toeplitz‑like matrix with constant diagonals.
        diag_data = []
        offsets = []
        # Offsets are negative (col = row - d) where d = row-col.
        for d in range(n):
            offset = -d
            # Length of this diagonal in a (rows x n) matrix
            diag_len = rows - d
            # First n entries of the diagonal are f[d], the rest are zeros.
            vals = np.empty(diag_len, dtype=np.float64)
            vals[:n] = f[d]
            if diag_len > n:
                vals[n:] = 0.0
            diag_data.append(vals)
            offsets.append(offset)

        A = _sp.diags(diagonal=diag_data, offsets=offsets,
                      shape=(rows, n), format='csr')

        # Linear programme: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(rows, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = _opt.linprog(c, A_ub=A, b_ub=b_ub,
                           bounds=bounds, method='highs',
                           options={'presolve': True, 'time_limit': max(1, int(deadline - time.time()))})
        if not res.success:
            # LP failed – abort further refinements
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        g_norm = g / sum_g

        # New max‑convolution value
        new_M = float(np.max(_conv_fft(g_norm, g_norm)))

        # Stop if improvement is negligible
        if M - new_M <= tol:
            break

        f = g_norm
    return f

# Main entry point expected by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. RNG & timing
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size – 4000 gives a good trade‑off between quality
    #    and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Build seed pool (includes a refined power‑law)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)

    # --------------------------------------------------------------
    # 2a. Refine the power‑law family (continuous search on the exponent)
    # --------------------------------------------------------------
    print("\n--- Power‑law fine search ---")
    best_pow_seq = None
    best_pow_val = float('inf')
    # coarse sweep
    coarse_ps = np.linspace(0.0, 3.0, 81)          # step ≈0.0375
    for p in coarse_ps:
        seq = _power(n, p)
        val = _objective_surrogate(seq)
        if val < best_pow_val:
            best_pow_val = val
            best_pow_seq = seq
            best_p = p

    # three refinement stages around the best exponent
    lo, hi = max(0.0, best_p - 0.1), best_p + 0.1
    for _ in range(3):
        fine_ps = np.linspace(lo, hi, 41)        # step ≈0.005
        for p in fine_ps:
            seq = _power(n, p)
            val = _objective_surrogate(seq)
            if val < best_pow_val:
                best_pow_val = val
                best_pow_seq = seq
                best_p = p
        lo, hi = max(0.0, best_p - (hi - lo) / 4), best_p + (hi - lo) / 4

    seeds.append(best_pow_seq)                     # guarantee its presence
    print(f"Best power‑law exponent ≈ {best_p:.6f}, surrogate = {best_pow_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Surrogate evaluation & select a handful of promising seeds
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 6
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop surrogate values among seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 4. Continuation L‑BFGS on the best seeds (more iterations than before)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                     500.0, 1000.0, 2000.0, 5000.0,
                     10000.0, 20000.0, 50000.0, 100000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached – stopping L‑BFGS runs.")
            break

        remaining = deadline - time.time()
        # give each seed a generous budget (around 10 % of remaining time)
        max_iter = int(min(24000, remaining * 0.1 * 1e5))
        max_iter = max(8000, max_iter)

        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter_per_beta={max_iter}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Deterministic LP refinement (tightens the bound)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- LP refinement (up to 3 iterations) ---")
        seq_lp = _lp_refine(global_best_seq, n, deadline, max_iters=3)
        val_lp = _objective_surrogate(seq_lp)
        print(f"[LP] surrogate = {val_lp:.6f}")
        if val_lp < global_best_val:
            global_best_seq = seq_lp
            global_best_val = val_lp
            print(f"[Update] after LP refinement: {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Exponentiated‑gradient polishing (short but many iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(400_000, remaining * 0.08 * 1e5))
        max_exp_iters = max(200_000, max_exp_iters)
        print(f"\n--- Exponentiated‑gradient polishing ({max_exp_iters} iters) ---")
        seq_exp = _exp_gradient_opt(global_best_seq, n, deadline, max_iters=max_exp_iters)
        val_exp = _objective_surrogate(seq_exp)
        print(f"[ExpGrad] surrogate = {val_exp:.6f}")
        if val_exp < global_best_val:
            global_best_seq = seq_exp
            global_best_val = val_exp
            print(f"[Update] after ExpGrad: {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
