def adam_optimize(seed_vec, n, deadline, beta_schedule, max_iters_per_beta):

    v = np.log(seed_vec + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1=0.9; beta2=0.999; eps=1e-8
    t = 0
    best_a = None; best_val = np.inf

    for beta in beta_schedule:
        beta_val = float(beta)
        for iter_local in range(max_iters_per_beta):
            t += 1
            if time.time() > deadline: break
            a = softmax(v)
            conv = conv_fft(a, a)
            shift = np.max(conv)
            # Compute exp term
            exp_term = np.exp(beta_val * (conv - shift))
            sum_exp = np.sum(exp_term)
            if sum_exp == 0.0:
                # If all zeros due to underflow, treat w as Dirac at max idx
                w = np.zeros_like(conv)
                w[int(np.argmax(conv))] = 1.0
            else:
                w = exp_term / sum_exp
            # gradient wrt a
            grad_a = grad_a_from_w(a, w)
            # chain rule for logits
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_v
            v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v2 / (1 - beta2 ** t)
            lr = 0.1 / (np.sqrt(v_hat) + eps)   # per-parameter adapt
            v = v - lr * m_hat

            # evaluate surrogate
            cur_val = 2.0 * n * np.max(conv)  # surrogate = 2 n max conv
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
    return best_a, best_val

"""
Optimal step‑function search (≈ 1000 s)

The programme searches the simplex (non‑negative entries, sum = 1) for a
vector *a* that minimises

    2·n·max(conv(a,a))

where *conv* is the ordinary linear convolution.  The optimal value is
asymptotically 1 (the average value of the convolution), while the
trivial uniform distribution yields 2.  The algorithm proceeds in three
phases:

1. **Rich seed generation** – analytic families (exponential,
   beta‑shape, power‑law, sinusoid, triangular, mixtures) and many
   random Dirichlet draws.  The 𝑘 best seeds (by the exact surrogate
   objective) are kept.

2. **Adam optimisation of a smooth max‑approximation** – we minimise

        Lβ(a) = shift + log∑exp(β·(conv – shift))/β ,

   with ``shift = max(conv)``.  The vector is parametrised by logits
   ``v`` (softmax → a) which automatically enforces the simplex
   constraints.  A schedule of increasing β is used; at large β the
   surrogate becomes the true max and Adam behaves like an accelerated
   sub‑gradient method.  Adam’s adaptive steps and momentum help escape
   non‑convex plateaus.

3. **Long exponentiated‑gradient polishing** – a mirror‑descent scheme
   on the exact max‑objective refines the result further.

All phases respect a wall‑clock deadline (default ≈ 1000 s).  The best
vector found is stored in the global ``height_sequence_1`` variable and
returned as a plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


# ----------------------------------------------------------------------
# Objective utilities (the sum of the vector is assumed to be 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for an ℓ₁‑normalised vector: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _grad_a_from_w(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ w_k * conv_k  w.r.t. a   (conv = a*a).
    The formula is 2·reverse(a) * w  (convolution, then slice).
    """
    rev = a[::-1]
    full = _conv_fft(rev, w)                     # length 3n‑2
    start = len(a) - 1
    return 2.0 * full[start:start + len(a)]      # size n


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i/n), 0 ≤ i < n."""
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    return a / a.sum()


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·(n‑1‑i)/n)."""
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sinusoid(n: int) -> np.ndarray:
    """Half‑sine wave, zero at both ends, positive inside."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _triangular(n: int, width: float = 1.0) -> np.ndarray:
    """
    Symmetric triangular shape.
    width ∈ (0,1] controls the support: width=1 ⇒ full triangle,
    smaller values truncate the base.
    """
    centre = (n - 1) / 2.0
    maxdist = width * centre
    i = np.arange(n, dtype=np.float64)
    dist = np.abs(i - centre)
    a = np.maximum(0.0, 1.0 - dist / maxdist)
    return a / a.sum()


def _mix_two_exponential(
    n: int, c1: float, c2: float, w1: float = 0.5
) -> np.ndarray:
    """
    Linear blend of a “forward” and a “reverse” exponential.
    w1 controls the weight of the forward component.
    """
    fwd = _exp_family(n, c1)
    rev = _rev_exp_family(n, c2)
    a = w1 * fwd + (1.0 - w1) * rev
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return a / a.sum()


# ----------------------------------------------------------------------
# Adam optimisation of the smoothed max‑objective
# ----------------------------------------------------------------------
def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule,
    max_iter_per_beta: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp surrogate.
    Returns the best ℓ₁‑normalised vector seen and its surrogate value.
    """
    # initialise logits (softmax parametrisation)
    v = np.log(start_vec + 1e-30)

    m = np.zeros_like(v)          # first moment
    v2 = np.zeros_like(v)         # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0

    best_a = None
    best_val = np.inf

    for beta in beta_schedule:
        beta_val = float(beta)

        for _ in range(max_iter_per_beta):
            t += 1
            if time.time() > deadline:
                break

            a = _softmax(v)
            conv = _conv_fft(a, a)

            # ----- smooth max (log‑sum‑exp) -----
            shift = np.max(conv)
            # exponentials are never > 1 because conv ≤ shift
            ex = np.exp(beta_val * (conv - shift))
            sum_ex = ex.sum()
            # guard against complete underflow – fall back to a Dirac at the max index
            if sum_ex == 0.0 or not np.isfinite(sum_ex):
                w = np.zeros_like(conv)
                w[int(np.argmax(conv))] = 1.0
            else:
                w = ex / sum_ex

            # gradient w.r.t. a
            grad_a = _grad_a_from_w(a, w)

            # chain rule through the softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update (per‑parameter)
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            step = 0.12 / (np.sqrt(v2_hat) + eps)   # modest learning‑rate
            v -= step * m_hat

            # surrogate (exact) objective for monitoring
            cur_val = 2.0 * n * np.max(conv)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

        if time.time() > deadline:
            break

    # If nothing improved (should not happen) fall back to the start vector
    if best_a is None:
        best_a = start_vec.copy()
        best_val = _objective_surrogate(best_a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑objective
# ----------------------------------------------------------------------
def _exp_gradient_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent using the exact sub‑gradient (the max‑index)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.6 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val
