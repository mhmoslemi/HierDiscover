def _adam_optimize(start_vec, n, deadline, beta_schedule, iters_per_beta):
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    t = 0
    beta1=0.9; beta2=0.999; eps=1e-8
    best_seq = None; best_val = np.inf

    for beta in beta_schedule:
        for it in range(iters_per_beta):
            t += 1
            if time.time()>deadline: break outer loops
            a = _softmax(v)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            # compute log-sum-exp:
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            # gradient w.r.t a:
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]  # size n
            # compute gradient w.r.t v:
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # update Adam:
            m = beta1 * m + (1 - beta1) * grad_v
            v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1 ** t)
            v2_hat = v2 / (1 - beta2 ** t)
            lr = 0.5 / (np.sqrt(t))  # or constant step size? maybe 0.1
            v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

            # evaluate surrogate occasionally
            if it % 2000 == 0:
                cur_val = _objective_surrogate(_softmax(v))
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = _softmax(v).copy()
    return best_seq or _softmax(v)

Propose a non‑negative step‑function that (almost) minimises

    2·n·max(conv(a, a)) / (sum(a)²)

The routine follows a three‑phase strategy inside the 1000 s budget:

1. **Massive seed generation** – many analytically‑motivated families

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by the official evaluator (no normalisation)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p) if p != 0 else np.ones_like(idx)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Permutation helpers
# ----------------------------------------------------------------------
def _alternating_order_from_vec(vec: np.ndarray) -> np.ndarray:
    """Place sorted values alternating from the left and right ends."""
    sorted_desc = np.sort(vec)[::-1]
    n = len(vec)
    out = np.empty_like(vec)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out


def _shuffle_by_step(sorted_desc: np.ndarray, step: int) -> np.ndarray:
    """
    Distribute the sorted values over the index set using a modular step.
    ``step`` should be coprime with n to obtain a permutation.
    """
    n = len(sorted_desc)
    out = np.empty(n, dtype=np.float64)
    pos = 0
    for val in sorted_desc:
        out[pos] = val
        pos = (pos + step) % n
    return out


# ----------------------------------------------------------------------
# Seed collection (rich pool)
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    top_k: int = 8,
) -> List[np.ndarray]:
    """
    Build a diverse collection of ℓ₁‑normalised candidate vectors.
    Returns the *top_k* candidates according to the exact surrogate.
    """
    # --- Analytic families -------------------------------------------------
    seeds = [
        _uniform(n),
        _triangular(n),
    ]

    # many Gaussian widths
    for sigma in (0.07, 0.10, 0.13, 0.16, 0.20, 0.25, 0.30, 0.35, 0.40):
        seeds.append(_gaussian(n, sigma))

    # power‑law exponents (including negative ones)
    for p in (-0.5, -0.25, 0.0, 0.2, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))

    # --- Random Dirichlet draws + enriched permutations --------------------
    conc_list = (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    step_list = [3, 7, 11, 13, 17, 19, 23, 29, 31, 37]  # a handful of coprime steps
    for conc in conc_list:
        for _ in range(8):      # eight independent draws per concentration
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                           # raw order
            desc = np.sort(raw)[::-1]
            seeds.append(desc)                           # descending
            seeds.append(_alternating_order_from_vec(raw))
            for step in step_list[:3]:
                seeds.append(_shuffle_by_step(desc, step))
            # a random full permutation
            perm = rng.permutation(n)
            seeds.append(raw[perm])

    # --- Previous solution -------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise (guard against NaNs)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))

    # Keep only the best ``top_k`` seeds (cheap surrogate is exact for ℓ₁‑normed vecs)
    heap: List[Tuple[float, np.ndarray]] = []
    for vec in cleaned:
        val = _objective_surrogate(vec)
        if len(heap) < top_k:
            heapq.heappush(heap, (-val, vec))
        else:
            if -val > heap[0][0]:          # we have a better (smaller) value
                heapq.heapreplace(heap, (-val, vec))
    # Return in ascending order of surrogate value
    best = sorted([(-score, vec) for score, vec in heap], key=lambda x: x[0])
    return [vec for _, vec in best]


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: List[float] | None = None,
    max_iter_per_beta: int = 15000,
) -> np.ndarray:
    """
    Refine ``start_vec`` (already ℓ₁‑normalised) with L‑BFGS‑B
    on a smooth approximation of the max‑convolution.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0,
            5.0,
            20.0,
            80.0,
            200.0,
            500.0,
            1000.0,
            2000.0,
            5000.0,
            10000.0,
            20000.0,
            50000.0,
            100000.0,
            200000.0,
            500000.0,
            1_000_000.0,
        ]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on the simplex
            conv = _conv_fft(a, a)                     # full linear conv
            shift = np.max(conv)
            # smooth max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # soft‑weights for the gradient
            w = exp_beta / sum_exp
            # gradient w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]      # slice the middle n entries

            # map to gradient w.r.t. logits (softmax Jacobian)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> Tuple[np.ndarray, float]:
    """Long sub‑gradient descent on 2·n·max(conv) for ℓ₁‑normalised vectors."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.30 / math.sqrt(it)   # a slightly more aggressive schedule
        a -= lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing style refinement (logit‑space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
) -> Tuple[np.ndarray, float]:
    """Random walk in logit space with a simple cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Entry point called by the evaluation harness.
    Returns a list of non‑negative heights that (hopefully) yields a lower
    bound than the previous best.
    """
    # ------------------------------------------------------------------
    # 0. Initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 4.0

    # ------------------------------------------------------------------
    # 1. Problem size – 4000 is a good trade‑off between granularity and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Assemble a rich pool of seeds and keep the very best few
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n or prev_arr.sum() <= 0.0:
                prev_arr = None
        except Exception:
            prev_arr = None

    top_seeds = _collect_seeds(n, rng, prev_arr, top_k=8)
    print(f"Collected {len(top_seeds)} top seeds (after cheap surrogate ranking).")
    for i, seq in enumerate(top_seeds[:4], 1):
        print(f"  Seed #{i} surrogate = {_objective_surrogate(seq):.6f}")

    # ------------------------------------------------------------------
    # 3. Heavy continuous optimisation (L‑BFGS) on the best seed(s)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    # aggressive β‑schedule – the same for every seed
    beta_schedule = [
        1.0,
        5.0,
        20.0,
        80.0,
        200.0,
        500.0,
        1000.0,
        2000.0,
        5000.0,
        10000.0,
        20000.0,
        50000.0,
        100000.0,
        200000.0,
        500000.0,
        1_000_000.0,
    ]

    # We will only devote the first ~50 % of the budget to L‑BFGS.
    l_bfgs_deadline = start_time + budget_s * 0.48
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > l_bfgs_deadline:
            print("[Time] L‑BFGS phase finished by deadline.")
            break

        remaining = l_bfgs_deadline - time.time()
        # allocate at most ~15000 iterations per β (more if time still abundant)
        max_iter_beta = int(min(15000, max(5000, remaining * 0.04 * 1e5)))
        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter/β = {max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline=l_bfgs_deadline,
            beta_schedule=beta_schedule,
            max_iter_per_beta=max_iter_beta,
        )
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Sub‑gradient polishing (the most time‑consuming phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate ~35 % of the total budget to sub‑gradient descent
        max_sub_iters = int(min(2_500_000, remaining * 0.35 * 1e5))
        if max_sub_iters > 0:
            print("\n--- Long sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(
                global_best_seq,
                n,
                deadline,
                max_iters=max_sub_iters,
            )
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Simulated‑annealing refinement (the final exploratory phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # use the rest of the time for SA (≈15 % of the budget)
        max_sa_iters = int(min(1_200_000, remaining * 0.3 * 1e5))
        if max_sa_iters > 0:
            print("\n--- Simulated‑annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing_opt(
                global_best_seq,
                n,
                deadline,
                max_iters=max_sa_iters,
            )
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[SA] final best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
