def _adam_lowdim_optimize(basis, conv_pair, n, deadline, max_iters=20000):
    m = len(basis)
    # Initialize logits
    logits = np.zeros(m)  # uniform w
    # Adam params
    beta1=0.9; beta2=0.999; lr=0.05
    m_ad = np.zeros_like(logits)
    v_ad = np.zeros_like(logits)
    t = 0
    best_w = None
    best_val = np.inf
    L = conv_pair.shape[2]
    while time.time() < deadline and t < max_iters:
        t += 1
        w = _softmax(logits)
        # compute conv_all
        tmp = np.tensordot(conv_pair, w, axes=([1],[0]))  # shape (m, L)
        conv_all = np.dot(w, tmp)  # shape (L,)
        idx = int(np.argmax(conv_all))
        A = conv_pair[:, :, idx]  # (m,m)
        grad_w = 2.0 * A.dot(w)  # shape (m,)
        # backprop through softmax
        dot = np.dot(grad_w, w)
        grad_logits = w * (grad_w - dot)

        # Adam update
        m_ad = beta1 * m_ad + (1 - beta1) * grad_logits
        v_ad = beta2 * v_ad + (1 - beta2) * (grad_logits ** 2)
        m_hat = m_ad / (1 - beta1 ** t)
        v_hat = v_ad / (1 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        if t % 500 == 0:
            cur_val = 2.0 * n * conv_all[idx]   # surrogate
            if cur_val < best_val:
                best_val = cur_val

"""
Propose a non‑negative step function (vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

within the given time budget (≈1000 s).

Algorithm
---------
1. **Rich seed pool** – generate many ℓ₁‑normalised candidates from
   deterministic windows and random parametric families (beta,
   power‑law, exponential, Dirichlet, low‑frequency sine series, …).
2. **Select a compact basis** – keep the 30 seeds with the smallest
   surrogate 2·n·max(conv) (the surrogate equals the true objective for
   ℓ₁‑normalised vectors).  The current best solution (`height_sequence_1`)
   is always added to the basis.
3. **Low‑dimensional optimisation** – any convex combination of the basis
   vectors is again ℓ₁‑normalised, so we optimise the weight vector `w`
   (simplex of size *m* ≈ 30).  For a fixed weight vector the convolution
   is a quadratic form

        C(w) = Σ_{i,j} w_i w_j  conv(b_i,b_j) .

   The maximum entry of `C(w)` is the objective.  We run an Adam optimiser
   on the *smooth* maximum in the low‑dimensional space using the exact
   gradient (2·Aₖ w where Aₖ is the pairwise‑convolution matrix at the
   active index `k`).  The optimisation stays completely in the simplex
   by working in the log‑softmax (exponentiated‑gradient) parametrisation.
4. **Re‑assemble the step function** – `a = Σ_i w_i · basis_i`
   (still ℓ₁‑normalised).
5. **Fine‑grained polish** – a short continuation‑Adam run on the full‑
   dimensional vector further reduces the smooth‑max objective.
6. Return the best vector found and store it globally as `height_sequence_1`.

The low‑dimensional stage is very cheap (≈ 30 variables) and allows
tens of thousands of iterations, dramatically improving the bound while
leaving ample time for a final high‑dimensional polish.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities (same as in the original code)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always ℓ₁‑normalised, non‑negative)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """Surrogate = 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector."""
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p  (decreasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window, β≥0 (β=0 yields a rectangular window)."""
    from scipy.special import i0

    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    w = i0(beta * np.sqrt(1.0 - t * t)) / i0(beta)
    return w / w.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.16
    w = (
        (1 - a) / 2
        - 0.5 * np.cos(2 * np.pi * i / (n - 1))
        + a / 2 * np.cos(4 * np.pi * i / (n - 1))
    )
    return w / w.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return w / w.sum()


def _tukey_window(n: int, alpha: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1)
    w = np.ones_like(i)
    mask1 = i < alpha / 2
    mask2 = i > 1 - alpha / 2
    w[mask1] = 0.5 * (1 + np.cos(np.pi * (2 * i[mask1] / alpha - 1)))
    w[mask2] = 0.5 * (1 + np.cos(np.pi * (2 * (i[mask2] - 1) / alpha + 1)))
    return w / w.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed generation (large, diverse pool)
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    max_random: int = 2500,
) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_kaiser_window(n, beta=4.0))
    seeds.append(_kaiser_window(n, beta=8.0))
    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_tukey_window(n, alpha=0.4))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # random parametric families
    for _ in range(max_random // 6):
        a = 10 ** rng.uniform(-3, 1)
        b = 10 ** rng.uniform(-3, 1)
        seeds.append(_beta_shape(n, a, b))

        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        lam = rng.exponential(scale=0.001) + 1e-4
        seeds.append(_exp_decay(n, lam))

        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        beta = rng.uniform(0.0, 12.0)
        seeds.append(_kaiser_window(n, beta))

        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(
            n,
            10 ** rng.uniform(-2, 1),
            10 ** rng.uniform(-2, 1),
        )
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # sanitise
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Low‑dimensional Adam optimiser (weights on a basis)
# ----------------------------------------------------------------------
def _adam_lowdim_optimize(
    basis: list[np.ndarray],
    conv_pair: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the low‑dimensional weight vector `w` (simplex),
    using the exact max‑convolution gradient.
    Returns the best weight vector found and its surrogate value.
    """
    m = len(basis)
    logits = np.zeros(m)                     # uniform start → w = softmax(logits)
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.07
    m_ad = np.zeros_like(logits)
    v_ad = np.zeros_like(logits)
    t = 0

    best_w = None
    best_val = np.inf
    L = conv_pair.shape[2]

    while time.time() < deadline and t < max_iters:
        t += 1
        w = _softmax(logits)                 # current weights (sum to 1)

        # conv_all = Σ_i Σ_j w_i w_j conv(b_i,b_j)    (length L)
        tmp = np.tensordot(conv_pair, w, axes=([1], [0]))   # shape (m, L)
        conv_all = np.dot(w, tmp)            # shape (L,)

        idx = int(np.argmax(conv_all))        # active shift
        A = conv_pair[:, :, idx]              # (m, m) matrix of that shift

        # gradient w.r.t. w: 2 * A @ w
        grad_w = 2.0 * A.dot(w)               # shape (m,)

        # back‑prop through softmax
        dot = np.dot(grad_w, w)
        grad_logits = w * (grad_w - dot)

        # Adam update on logits
        m_ad = beta1 * m_ad + (1.0 - beta1) * grad_logits
        v_ad = beta2 * v_ad + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m_ad / (1.0 - beta1 ** t)
        v_hat = v_ad / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional best‑track
        if t % 500 == 0:
            cur_val = 2.0 * n * conv_all[idx]   # surrogate (ℓ₁‑normalised)
            if cur_val < best_val:
                best_val = cur_val
                best_w = w.copy()

    if best_w is None:
        best_w = _softmax(logits)
        best_val = 2.0 * n * np.max(np.dot(best_w, np.tensordot(conv_pair, best_w, axes=([1], [0]))))
    return best_w, best_val


# ----------------------------------------------------------------------
# Fine‑grained high‑dimensional polishing (reuse the original Adam)
# ----------------------------------------------------------------------
def _adam_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule=None,
    steps_per_beta: int = 2000,
) -> np.ndarray:
    """
    Adam optimiser on the smoothed max‑convolution (as in the original code).
    Returns a ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1000.0, 3000.0, 5000.0, 10000.0, 20000.0]

    eps = 1e-30
    logits = np.log(start_vec + eps)

    beta1 = 0.9
    beta2 = 0.999
    lr = 0.07
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    t = 0

    best_seq = None
    best_val = np.inf

    for beta in beta_schedule:
        beta_val = float(beta)
        max_iters = steps_per_beta
        for _ in range(max_iters):
            t += 1
            if time.time() > deadline:
                return _softmax(logits)

            a = _softmax(logits)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp

            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_logits = a * (grad_a - dot)

            m = beta1 * m + (1.0 - beta1) * grad_logits
            v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            if t % 500 == 0:
                cur_val = _objective_surrogate(a, n)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    return best_seq if best_seq is not None else _softmax(logits)

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # --------------------------------------------------------------
    # Seed pool -> compact basis
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    all_seeds = _collect_seeds(n, rng, prev_arr, max_random=3000)
    print(f"Generated {len(all_seeds)} candidate seeds.")

    # cheap surrogate ranking, keep the best few as basis
    surrogate_vals = [_objective_surrogate(vec, n) for vec in all_seeds]
    sorted_idx = np.argsort(surrogate_vals)
    basis_size = 30
    basis = [all_seeds[i] for i in sorted_idx[:basis_size]]
    print(f"Selected {len(basis)} basis vectors (top surrogate).")

    # ensure the previous best (if any) is present in the basis
    if prev_arr is not None and prev_arr.size == n:
        # normalize just in case
        prev_norm = np.maximum(prev_arr, 0.0)
        s = prev_norm.sum()
        if s > 0:
            prev_norm = prev_norm / s
            # check if already in basis (within tolerance)
            already = any(np.allclose(prev_norm, b, atol=1e-12) for b in basis)
            if not already and len(basis) < basis_size:
                basis.append(prev_norm)
                print("Added previous best to the basis.")
            elif not already:
                # replace the worst basis element
                worst_idx = sorted_idx[basis_size]  # first element outside the top‑k
                basis[-1] = all_seeds[worst_idx]
                print("Replaced a basis element with previous best.")

    m = len(basis)
    L = 2 * n - 1

    # --------------------------------------------------------------
    # Pre‑compute all pairwise convolutions (size m × m × L)
    # --------------------------------------------------------------
    print("Computing pairwise convolutions …")
    conv_pair = np.empty((m, m, L), dtype=np.float64)
    for i in range(m):
        conv_pair[i, i, :] = _conv_fft(basis[i], basis[i])
        for j in range(i + 1, m):
            conv_ij = _conv_fft(basis[i], basis[j])
            conv_pair[i, j, :] = conv_ij
            conv_pair[j, i, :] = conv_ij
    print("Pairwise convolutions ready.")

    # --------------------------------------------------------------
    # Low‑dimensional optimisation on the simplex of weights
    # --------------------------------------------------------------
    print("\n--- Optimising weights on the basis (low‑dim Adam) ---")
    w_opt, lowdim_val = _adam_lowdim_optimize(
        basis, conv_pair, n, deadline, max_iters=20000
    )
    print(f"Low‑dim surrogate after optimisation: {lowdim_val:.6f}")

    # --------------------------------------------------------------
    # Assemble the full‑dimensional candidate
    # --------------------------------------------------------------
    a = np.zeros(n, dtype=np.float64)
    for wi, bi in zip(w_opt, basis):
        a += wi * bi
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = _uniform(n)
    else:
        a /= s

    # --------------------------------------------------------------
    # Final polishing (high‑dim Adam with a strong smooth‑max schedule)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print("\n--- Final high‑dim polishing (smooth‑max Adam) ---")
        # allocate a modest number of steps per β (depends on remaining time)
        steps_per_beta = max(800, int(min(5000, remaining * 0.02 * 1e5)))
        a = _adam_smooth_optimize(
            start_vec=a,
            n=n,
            deadline=deadline,
            beta_schedule=[2000.0, 5000.0, 10000.0, 20000.0, 50000.0],
            steps_per_beta=steps_per_beta,
        )
        final_surrogate = _objective_surrogate(a, n)
        print(f"Surrogate after polishing: {final_surrogate:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(a.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {_objective_surrogate(a, n):.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for future runs
    globals()["height_sequence_1"] = a.tolist()
    return a.tolist()
