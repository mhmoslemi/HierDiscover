"""
A multi‑stage optimizer for the step‑function evaluation problem.

Stages
------
1. **Diverse seed generation** – analytic families (uniform, triangular,
   sine, Gaussian, β‑shapes, Tukey, Kaiser …), many low‑concentration Dirichlet
   draws, random sparse supports and random mixtures.  All candidates are
   ℓ₁‑normalised.

2. **Coarse selection** – exact objective (2·n·max(conv)/sum²) is evaluated
   for every seed and the best few are kept.

3. **Mirror‑descent on the exact max‑convolution** – exponentiated‑gradient
   updates (a_i ← a_i·exp(‑η·g_i) , a normalised) using the sub‑gradient of
   the active lag.  Multiple restarts from the selected seeds explore the
   landscape thoroughly.

4. **Perturb‑and‑polish** – the current best log‑weights are jittered and a
   short L‑BFGS optimisation of a smooth log‑sum‑exp surrogate is run.
   Several attempts are performed with decreasing jitter.

5. **Final high‑β polishing** – a longer L‑BFGS run with a very sharp β‑
   schedule (up to 10⁶) finishes the search.

The algorithm respects a global time‑budget (≈ 1000 s) and stores the best
sequence in the global name ``height_sequence_1`` for warm‑starts in later
calls.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on {1,…,n} (U‑shape when α,β<1)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random primitives
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_sparse(n: int, rng: np.random.Generator,
                   k_min: int = 30, k_max: int = 150) -> np.ndarray:
    """Sparse support: choose K random indices, give them random weights."""
    K = int(rng.integers(k_min, k_max + 1))
    idx = rng.choice(n, size=K, replace=False)
    a = np.zeros(n, dtype=np.float64)
    weights = rng.exponential(scale=1.0, size=K)
    a[idx] = weights
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α)+b·α then ℓ₁‑renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Exact objective utilities
# ----------------------------------------------------------------------
def _objective_raw(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) / (sum²)."""
    n = len(a)
    conv = _conv_fft(a, a)               # full linear convolution
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a for the active lag.
    Returns g = ∂ max(conv) / ∂ a .
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


def _exp_gradient_optimize(start_a: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best vector encountered and its exact objective.
    """
    a = np.maximum(start_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective_raw(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_maxconv(a)          # sub‑gradient of the max lag
        eta = eta0 / math.sqrt(it)           # diminishing stepsize

        # multiplicative update in log‑space
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)                  # projection onto simplex

        if it % 100_000 == 0:
            cur = _objective_raw(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    cur = _objective_raw(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and L‑BFGS optimisation
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Value and gradient of the smooth max‑convolution surrogate."""
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)              # full auto‑convolution
    shift = np.max(conv)                # shift for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient wrt a
    w = exp_term / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    L‑BFGS‑B optimisation of the smooth surrogate.
    Variable = logits of a softmax (unconstrained real vector).
    """
    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0,
                         5000.0, 10000.0, 20000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            # early‑stop if overall deadline is exceeded
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline during beta={beta_val}")
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)

def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Build a rich pool of ℓ₁‑normalised seed vectors."""
    seeds: list[np.ndarray] = []

    # simple analytic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in np.linspace(0.2, 5.0, 16):
        seeds.append(_sine_window(n, gamma=g))
    for sigma in np.linspace(0.06, 0.45, 10):
        seeds.append(_gaussian(n, sigma))
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))
    for a in np.linspace(0.2, 1.5, 14):
        for b in np.linspace(0.2, 1.5, 14):
            seeds.append(_beta_shape(n, a, b))
    for al in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, al))
    for beta in np.linspace(0.5, 14.0, 12):
        seeds.append(_kaiser_window(n, beta))

    # random low‑concentration Dirichlet draws ---------------------------------
    for conc in np.logspace(-3, 0, 7):          # 0.001 … 1
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials ------------------------------------------------------
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # sparse random supports ---------------------------------------------------
    for _ in range(30):
        K = int(rng.integers(20, 200))
        seeds.append(_random_sparse(n, rng, k_min=K, k_max=K))

    # mixtures of existing seeds -----------------------------------------------
    base_idx = np.arange(len(seeds))
    for _ in range(500):
        i, j = rng.choice(base_idx, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # previous best (warm start) ------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final clean‑up – guarantee non‑negative and ℓ₁‑normalised ------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` under the given time budget.  Returns the best
    sequence found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    n = int(kwargs.get("n", 4000))
    print(f"\n=== Optimisation start (n={n}) ===")

    # -------------------------------------------------------------------
    # Load possible warm‑start
    # -------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:  # pragma: no cover
            prev_arr = None

    # -------------------------------------------------------------------
    # Stage 0 – generate a broad pool of seeds
    # -------------------------------------------------------------------
    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # cheap exact objective for coarse ranking
    evaluated = [(vec, _objective_raw(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best raw objective among seeds (top {top_k}):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # -------------------------------------------------------------------
    # Stage 1 – exponentiated‑gradient refinement from the top seeds
    # -------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_raw(best_seq)
    print(f"\n[Init] best raw objective = {best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing EG runs.")
            break
        # allocate a modest number of EG iterations depending on remaining time
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 12))   # about 12 iters / second
        if max_iters < 10_000:
            max_iters = int(remaining * 20) + 1
        print(f"\n--- EG run #{idx+1} (max_iters={max_iters}) ---")
        seq_opt, val_opt = _exp_gradient_optimize(seed_vec, n,
                                                  deadline,
                                                  max_iters=max_iters,
                                                  eta0=0.8)
        print(f"[EG] result raw objective = {val_opt:.6f}")
        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] New best after EG = {best_val:.6f}")

    # -------------------------------------------------------------------
    # Stage 2 – stochastic perturb‑and‑polish with short L‑BFGS runs
    # -------------------------------------------------------------------
    if time.time() < deadline:
        attempts = 60
        print("\n--- Perturb‑and‑polish phase ---")
        for a in range(attempts):
            if time.time() > deadline:
                break
            logits = np.log(best_seq + 1e-30)
            scale = max(0.05, 0.5 / (a + 1))   # decreasing jitter
            jitter = rng.normal(scale=scale, size=n)
            start_vec = _softmax(logits + jitter)

            # a quick L‑BFGS optimisation with a modest β‑schedule
            seq_candidate = _lbfgs_smooth_optimize(start_vec,
                                                   n,
                                                   deadline,
                                                   beta_schedule=[1.0, 20.0, 200.0],
                                                   max_iter_per_beta=1200)
            cand_val = _objective_raw(seq_candidate)
            if cand_val < best_val:
                best_seq = seq_candidate.copy()
                best_val = cand_val
                print(f"[Perturb] improvement after attempt {a+1}: {best_val:.6f}")

    # -------------------------------------------------------------------
    # Stage 3 – final high‑β L‑BFGS polishing
    # -------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final high‑β polishing ---")
        seq_final = _lbfgs_smooth_optimize(best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=[5e3, 2e4, 1e5, 5e5, 1e6],
                                           max_iter_per_beta=2500)
        val_final = _objective_raw(seq_final)
        if val_final < best_val:
            best_seq = seq_final.copy()
            best_val = val_final
            print(f"[Final] improved raw objective = {best_val:.6f}")

    # -------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # -------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest raw objective ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible warm‑start on the next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
