"""
Hybrid search for a non‑negative step function that minimises

    2·n·max(conv(a, a)) / (sum(a)²)

The method combines several ingredients:

1. **Rich seed pool**
   • Analytic families (uniform, triangular, Gaussian, power‑law).
   • New “U‑shaped” families that concentrate mass near the ends:
       a_i ∝ (min(i, n‑1‑i)+1)^{-p}.
   • Simple 2‑parameter mixtures of the U‑shape with the uniform vector.
   • Random Dirichlet draws (various concentrations) and many permutations.
   • The best seed found by a tiny exhaustive grid‑search over the
     mixture parameters (weight w, exponent p).

2. **Continuation L‑BFGS**
   The best few seeds are refined with L‑BFGS‑B on a smooth
   log‑sum‑exp approximation of the max‑convolution.
   The schedule of the smoothing parameter β is aggressive and the
   number of inner iterations is increased compared with the previous
   version.

3. **Projected sub‑gradient polishing**
   A long run of a Euclidean projected sub‑gradient method (exact
   surrogate 2·n·max(conv)) is performed on the current global best.

4. **Final evaluation**
   The resulting sequence is passed to the official ``evaluate_sequence``.
   It is also stored in the global variable ``height_sequence_1`` for
   warm‑starting the next run.

All steps respect the overall 1000 s budget and stop early if the
deadline is reached.

import time

# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (surrogate = exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by the evaluator (no normalisation)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# New “U‑shaped” families (edge‑heavy)
# ----------------------------------------------------------------------
def _edge_heavy(n: int, p: float) -> np.ndarray:
    """Mass concentrated near the ends: a_i ∝ (dist+1)^{-p}."""
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)  # distance to nearer border
    a = (dist + 1.0) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_edge_uniform(n: int, w: float, p: float,
                      uniform: np.ndarray, edge: np.ndarray) -> np.ndarray:
    """Convex mixture of uniform and an edge‑heavy shape (both already ℓ₁‑normalised)."""
    return (1.0 - w) * uniform + w * edge


# ----------------------------------------------------------------------
# Permutation helpers (to enrich the seed pool)
# ----------------------------------------------------------------------
def _alternating_order_from_vec(vec: np.ndarray) -> np.ndarray:
    """Place sorted values alternating from left and right ends."""
    sorted_desc = np.sort(vec)[::-1]
    n = len(vec)
    out = np.empty_like(vec)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out


def _shuffle_by_step(sorted_desc: np.ndarray, step: int) -> np.ndarray:
    """
    Distribute the sorted values over the index set using a modular step.
    ``step`` should be coprime with n to obtain a permutation.
    """
    n = len(sorted_desc)
    out = np.empty(n, dtype=np.float64)
    pos = 0
    for val in sorted_desc:
        out[pos] = val
        pos = (pos + step) % n
    return out


# ----------------------------------------------------------------------
# Seed pool construction (adds many shuffled variants)
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse collection of ℓ₁‑normalised candidate vectors.
    """
    seeds: list[np.ndarray] = []

    # ---- Analytic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.12, 0.18, 0.25):
        seeds.append(_gaussian(n, sigma))

    for p in (0.0, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))

    # ---- Edge‑heavy families (U‑shaped) ------------------------------------
    for p in np.linspace(0.0, 4.0, 9):          # 0,0.5,…,4
        edge = _edge_heavy(n, p)
        seeds.append(edge)                      # pure edge‑heavy
        # mixture variants (different mixing weights)
        for w in (0.2, 0.4, 0.6, 0.8):
            mix = _mix_edge_uniform(n, w, p,
                                    _uniform(n), edge)
            seeds.append(mix)

    # ---- Random Dirichlet draws + shuffles ---------------------------------
    conc_list = (0.05, 0.2, 0.5, 1.0, 2.0)
    step_list = [3, 7, 11, 13, 17, 19, 23, 29, 31, 37]      # coprime to 4000
    for conc in conc_list:
        for _ in range(4):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                                 # as‑drawn
            desc = np.sort(raw)[::-1]
            seeds.append(desc)                                # descending
            seeds.append(_alternating_order_from_vec(raw))    # alt. order
            for step in step_list[:5]:
                seeds.append(_shuffle_by_step(desc, step))    # modular steps
            perm = rng.permutation(n)
            seeds.append(raw[perm])                           # random perm.

    # ---- Previous solution -------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise robustly (protect against NaNs)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Small grid search over mixture of uniform + edge‑heavy
# ----------------------------------------------------------------------
def _grid_search_edge_mixture(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exhaustively scan w∈[0,1] and p∈[0,4] for the mixture
    (1‑w)*U + w*Edge(p). Returns the best sequence and its surrogate value.
    """
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)

    best_val = math.inf
    best_seq = None

    # coarse grids – fast to evaluate
    w_grid = np.linspace(0.0, 1.0, 21)       # 0,0.05,…,1
    p_grid = np.linspace(0.0, 4.0, 17)       # 0,0.25,…,4

    for p in p_grid:
        if time.time() > deadline:
            break
        edge = (dist + 1.0) ** (-p)
        edge_sum = edge.sum()
        if edge_sum == 0.0:
            continue
        edge /= edge_sum                      # now ℓ₁‑normalised
        for w in w_grid:
            seq = (1.0 - w) * uniform + w * edge   # already ℓ₁‑normalised
            max_conv = np.max(_conv_fft(seq, seq))
            val = 2.0 * n * max_conv
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
    if best_seq is None:
        # fall back to uniform – should never happen
        best_seq = uniform
        best_val = _objective_surrogate(best_seq)
    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smooth max‑conv."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0]

    v = np.log(start_vec + 1e-30)          # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                # stays on the simplex
            conv = _conv_fft(a, a)                # full conv
            max_c = np.max(conv)
            shift = max_c                         # numerical stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth max (see comment in original code)
            w = exp_beta / sum_exp                # soft‑weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """Long sub‑gradient descent on 2·n·max(conv) for ℓ₁‑normalised vectors."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]      # surrogate = exact for ℓ₁‑normed a

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry (same pattern as earlier code)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin for final call

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between granularity and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Seed pool construction
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds (analytic + Dirichlet).")

    # ------------------------------------------------------------------
    # 2. Quick grid search over edge‑heavy + uniform mixtures
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Grid search: uniform ↔ edge‑heavy mixture ---")
        best_mix_seq, best_mix_val = _grid_search_edge_mixture(n, deadline)
        seeds.append(best_mix_seq)
        print(f"Best mixture surrogate = {best_mix_val:.6f}")

    # cheap surrogate (exact for ℓ₁‑normalised vectors)
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS on the top seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                     500.0, 1000.0, 2000.0, 5000.0,
                     10000.0, 20000.0, 50000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] Deadline reached before finishing all L‑BFGS runs.")
            break

        remaining = deadline - time.time()
        # allocate a generous slice per seed (but never more than a few seconds)
        max_iter_beta = int(min(8000, remaining * 0.03 * 1e5))
        max_iter_beta = max(3000, max_iter_beta)

        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter/β = {max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long sub‑gradient polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(1_200_000, remaining * 0.04 * 1e5))
        if max_sub > 0:
            print("\n--- Long sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
