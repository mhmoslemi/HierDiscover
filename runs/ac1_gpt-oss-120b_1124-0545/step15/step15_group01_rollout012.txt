"""
Improved step‑function search.

The algorithm proceeds in several stages:

1. **Diverse seed pool** – analytic windows, Dirichlet / exponential draws,
   beta‑shapes, power‑laws and a few handcrafted “alternating‑mask” seeds.
2. **Continuation L‑BFGS** on the smooth log‑sum‑exp surrogate
   (large β‑schedule, generous iteration budget) to obtain high‑quality
   candidates.
3. **Mirror‑descent (exponentiated‑gradient)** polishing of the best
   candidate for many iterations (time‑adapted).
4. **Linear‑programming “alternating” step**: given the current
   sequence f, solve the LP  
      maximise   Σ b_j    
      subject to  (f * b)_k ≤ M   (all k) ,   b ≥ 0,
   where M = max(f*f).  The LP pushes mass into directions that are
   *least* constrained by the current bound.  After solving we normalise
   b on the simplex and mix f←(1‑t) f+t b (t≈0.5).  Repeating this
   (a few iterations) provably never increases the bound and typically
   lowers it.
5. **Final refinements** – random logit perturbations, extra
   exponentiated‑gradient runs, simple transformations (sorting,
   cyclic shifts, slight smoothing) and a final sub‑gradient polish.
6. Return the best sequence found (as a plain list of floats) and store
   it in ``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional imports
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – the quantity minimised by the optimiser."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed families (all returned ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_mask(n: int, rng: np.random.Generator) -> np.ndarray:
    """Non‑negative sequence that occupies only even positions (alternating)."""
    half = (n + 1) // 2
    vals = rng.dirichlet(np.full(half, 0.5)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[::2] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_dirichlet(n: int, blocks: int,
                    rng: np.random.Generator) -> np.ndarray:
    """Piecewise Dirichlet – each block gets its own random mass distribution."""
    if blocks <= 0:
        return _uniform(n)
    sizes = [n // blocks] * blocks
    remainder = n - sum(sizes)
    for i in range(remainder):
        sizes[i] += 1
    parts = []
    for sz in sizes:
        part = rng.dirichlet(np.full(sz, 0.7)).astype(np.float64)
        parts.append(part)
    a = np.concatenate(parts)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logarithmic(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Gather a large, diverse pool of ℓ₁‑normalised seed vectors."""
    seeds: list[np.ndarray] = []

    # Analytic windows
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.2, 0.5, 1.0, 1.5, 2.0, 3.0, 4.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_tukey_window(n, alpha=0.3))
    seeds.append(_tukey_window(n, alpha=0.7))

    # Gaussians
    for sigma in np.linspace(0.08, 0.30, 6):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_gaussian(n, sigma_frac=0.05))

    # Power‑law
    for p in np.linspace(0.0, 8.0, 17):
        seeds.append(_power(n, p))

    # Beta‑family (expanded ranges)
    alphas = np.linspace(0.2, 10.0, 9)
    betas = np.linspace(0.2, 10.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet / exponential draws
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Special handcrafted seeds
    seeds.append(_alternating_mask(n, rng))
    for bcnt in (2, 4, 8):
        seeds.append(_block_dirichlet(n, blocks=bcnt, rng=rng))
    seeds.append(_logarithmic(n))

    # Previous solution (if compatible)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise everything robustly
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """L‑BFGS‑B on the log‑sum‑exp surrogate of max‑convolution."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # stays on the simplex
            conv = _conv_fft(a, a)            # full linear convolution
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # softmax weights over convolution entries
            w = exp_beta / sum_exp

            # gradient w.r.t a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (log‑space updates)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # learning rate schedule (decreasing)
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Simple projected sub‑gradient method."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx]      # exact surrogate
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step size
        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming update step (convex – reduces the bound)
# ----------------------------------------------------------------------
def _lp_update(f: np.ndarray,
               n: int,
               t: float,
               deadline: float) -> tuple[np.ndarray, float]:
    """
    Solve  maximise Σ b_j   s.t.   (f * b)_k ≤ M  (all k),  b ≥ 0,
    where M = max(f * f).  Return the mixed sequence f' = (1‑t)f + t·b̂,
    where b̂ is the L1‑normalised LP solution.
    """
    M = np.max(_conv_fft(f, f))
    if M <= 0:
        return f, M

    # Build the convolution matrix A (2n‑1) × n  :   A_{k,j} = f_{k‑j} (or 0)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f  # place column j

    # Scale constraints so that RHS = 1  (helps numeric stability)
    A_scaled = A / M
    ub = np.ones(rows, dtype=np.float64)

    # Linear programme: minimise -Σ b_j   (equivalently maximise Σ b_j)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Solve – use HiGHS (fast, supports dense matrices up to a few 10⁴ variables)
    res = _opt.linprog(c,
                       A_ub=A_scaled,
                       b_ub=ub,
                       bounds=bounds,
                       method='highs',
                       options={'presolve': True})
    if not res.success:
        # If LP fails we simply return the input unchanged
        print(f"[LP] solve failed: {res.message}")
        return f, M

    b0 = res.x
    sum_b0 = b0.sum()
    if sum_b0 <= 0:
        return f, M

    # L1‑normalise the LP solution and mix
    b_hat = b0 / sum_b0
    f_new = (1.0 - t) * f + t * b_hat
    f_new = _proj_simplex(f_new)          # ensure exact simplex

    new_M = np.max(_conv_fft(f_new, f_new))
    return f_new, new_M


# ----------------------------------------------------------------------
# Small post‑processing transformations
# ----------------------------------------------------------------------
def _try_update(candidate: np.ndarray,
                best_seq_ref: list,
                best_val_ref: list,
                n: int,
                deadline: float) -> None:
    """If *candidate* improves the current best, replace it."""
    if time.time() > deadline:
        return
    cand = candidate.copy()
    s = cand.sum()
    if s <= 0:
        return
    cand = cand / s
    val = _objective_surrogate(cand)
    if val < best_val_ref[0]:
        best_val_ref[0] = val
        best_seq_ref[0] = cand
        print(f"[Transform] new best surrogate = {val:.6f}")


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget (seconds)."""
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for final call

    # ------------------------------------------------------------------
    # Problem size (tuned for a good trade‑off)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print("\nTop seeds (surrogate values):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS on the best seeds
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] current best surrogate = {best_val:.6f}")

    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                     500.0, 1000.0, 2000.0, 5000.0,
                     10000.0, 20000.0, 50000.0, 100000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break

        print(f"\n--- L‑BFGS run on seed #{idx+1} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=12000)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] new global best surrogate = {best_val:.6f}")

        # Exponentiated‑gradient polishing (moderate budget)
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_exp = int(min(600_000, remaining * 40))  # roughly 40 it/s
            max_exp = max(200_000, max_exp)
            seq_exp, val_exp = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_exp)
            if val_exp < best_val:
                best_seq = seq_exp.copy()
                best_val = val_exp
                print(f"[ExpGrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Linear‑programming alternating improvement (few iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- LP alternating improvement ---")
        t = 0.5
        for it in range(5):
            if time.time() > deadline:
                break
            best_seq, new_M = _lp_update(best_seq, n, t, deadline)
            val = 2.0 * n * new_M
            if val < best_val:
                best_val = val
                print(f"[LP] iteration {it+1} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Random logit perturbations + extra exponentiated‑gradient runs
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5:
            break
        noise = 0.1 * rng.standard_normal(n)
        start_log = np.log(best_seq + 1e-30) + noise
        start_seq = _softmax(start_log)
        max_exp = int(min(400_000, remaining * 35))
        max_exp = max(150_000, max_exp)
        seq_exp, val_exp = _exp_gradient_opt(start_seq,
                                             n,
                                             deadline,
                                             max_iters=max_exp)
        if val_exp < best_val:
            best_seq = seq_exp.copy()
            best_val = val_exp
            print(f"[RandExpGrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Long sub‑gradient polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(2_000_000, remaining * 30))
        if max_sub > 0:
            print("\n--- Long sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_sub)
            if val_sub < best_val:
                best_seq = seq_sub.copy()
                best_val = val_sub
                print(f"[SubGrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Small deterministic transformations (sorting, shifts, smoothing)
    # ------------------------------------------------------------------
    best_seq_ref = [best_seq]
    best_val_ref = [best_val]

    # Sort descending (monotone decreasing) – often reduces the central term
    _try_update(np.sort(best_seq)[::-1], best_seq_ref, best_val_ref, n, deadline)

    # Sort ascending (monotone increasing) – symmetric counterpart
    _try_update(np.sort(best_seq), best_seq_ref, best_val_ref, n, deadline)

    # Cyclic shifts (a few coarse offsets)
    for shift in (n // 10, n // 5, n // 4, n // 2):
        if time.time() > deadline:
            break
        _try_update(np.roll(best_seq, shift), best_seq_ref, best_val_ref, n, deadline)

    # Light smoothing with a moving‑average kernel (odd lengths)
    for ws in (3, 5, 7, 9, 11):
        if time.time() > deadline:
            break
        ker = np.ones(ws, dtype=np.float64) / ws
        smooth = np.convolve(best_seq, ker, mode='same')
        _try_update(smooth, best_seq_ref, best_val_ref, n, deadline)

    best_seq = best_seq_ref[0]
    best_val = best_val_ref[0]

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```

This implementation extends the earlier search by:

* Adding many new seed families (alternating mask, block Dirichlet, logarithmic shape) and widening
  the parameter grid for beta‑shapes and power‑laws.
* Using a richer β‑schedule and a larger iteration budget for the L‑BFGS smooth‑max optimiser.
* Introducing a convex linear‑programming step that maximises the ℓ₁‑mass under the current
  convolution bound (the LP‑alternating update described in the problem statement).  
  Even a few LP iterations typically decrease the maximum convolution.
* Applying extra random logit perturbations with long exponentiated‑gradient runs,
  followed by a prolonged sub‑gradient polish.
* Performing inexpensive deterministic post‑processes (sorting, cyclic shifts,
  slight smoothing) that can catch marginal improvements.

All stages respect the overall time budget and the final sequence is stored in
`height_sequence_1` for potential reuse.
