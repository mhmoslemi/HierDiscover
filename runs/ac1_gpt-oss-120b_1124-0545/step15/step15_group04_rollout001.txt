"""
Propose a step‑function (non‑negative heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a))**2

with n=4000 and a sum fixed to 1.
The algorithm combines:

* a very diverse seed pool (analytic families, fine‑gridded parameters,
  random Dirichlet / exponential draws, low‑frequency curves and random
  β‑shape mixes);
* a continuation L‑BFGS run on a smooth log‑sum‑exp surrogate for the
  max‑convolution (β is increased gradually up to 1e6);
* an exponentiated‑gradient (mirror‑descent) optimisation on the exact
  non‑smooth objective (sub‑gradient = max‑convolution index);
* a long stochastic local‑search that moves a tiny fraction of mass
  between random indices (hill‑climbing);
* a final high‑β L‑BFGS polish.

All steps respect a 1000 s wall‑clock budget, keep the best vector in the
global name ``height_sequence_1`` and finally return it as a plain list of
floats.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors, sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)  # n, n‑1, …, 1
    s = a.sum()
    return a / s


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)  (i = 0 … n‑1)"""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)      # Bessel I0 from NumPy
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i = 1/(i+1)^p  (i = 0 … n‑1)"""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α) + b·α, re‑normalised to sum‑1."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Surrogate (smooth max) helpers for continuation L‑BFGS
# ----------------------------------------------------------------------


def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (obj, grad) where grad is w.r.t. the logits v.
    """
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)               # full convolution
    shift = np.max(conv)                 # numeric stabiliser
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # soft‑max weights on lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3·n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]  # slice of length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    The optimisation variable is the log‑softmax (logits) vector.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            1e4, 2e4, 5e4, 1e5, 2e5, 5e5, 1e6,
        ]

    # initialise logits safely
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            """Return (value, gradient) for scipy.minimize."""
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            # deadline reached – exit early
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)   # final probability vector (sum = 1)


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient (max‑convolution)
# ----------------------------------------------------------------------


def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²).  Sum(a) is assumed = 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a (using a single maximal lag).
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))          # the lag attaining the maximal value

    # derivative of conv[idx] = Σ a_j a_{idx‑j}
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = idx - idx_i
    g[idx_i] = 2.0 * a[rev_idx]
    return g


def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_500_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the *exact* max‑convolution
    objective.  Returns the best vector seen and its objective.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 200_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    # final check
    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny stochastic local‑search (mass transfers)
# ----------------------------------------------------------------------


def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 300_000,
    step_frac: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a small fraction of mass from a random
    donor index to a random receiver index.  Keep any improving move.
    """
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456)

    for it in range(iters):
        if time.time() > deadline:
            break

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue

        delta = step_frac * a[i]
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand     # continue from the improved point

    return best_a, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function of length 4000 that minimises the official
    evaluation function.  The routine respects the given time budget,
    stores the best vector globally as ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    n = 4000
    print(f"\n=== start search (n={n}) – budget ≈ {budget_s}s ===")

    # ------------------------------------------------------------------
    # Recover previous solution (if any) for warm‑starting
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(previous, list):
        arr = np.asarray(previous, dtype=np.float64)
        if arr.shape == (n,) and arr.sum() > 0.0:
            prev_vec = arr / arr.sum()

    # ------------------------------------------------------------------
    # Build a rich seed pool (all vectors are ℓ₁‑normalised)
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_linear_decrease(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        seeds.append(_sine_window(n, gamma=beta))
    for kbeta in (2, 4, 6, 8, 10, 12, 16):
        seeds.append(_kaiser_window(n, beta=kbeta))

    # exponential decay – finer grid
    for lam in np.linspace(0.0005, 0.0030, 20):
        seeds.append(_exponential_decay(n, lam))

    # power‑law – extended range
    for p in np.linspace(0.0, 6.0, 31):
        seeds.append(_power_law(n, p))

    # Gaussian – more sigma values
    for sigma in np.linspace(0.10, 0.45, 15):
        seeds.append(_gaussian(n, sigma))

    # Beta shape – random samples covering a wide range
    for _ in range(200):
        a = rng.uniform(0.05, 10.0)
        b = rng.uniform(0.05, 10.0)
        seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of existing seeds (pairwise)
    base = np.arange(len(seeds))
    for _ in range(300):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Add the previous solution if available
    if prev_vec is not None:
        seeds.append(prev_vec)

    # Clean, enforce non‑negativity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            continue
        cleaned.append(v / s)
    seeds = cleaned
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # Quick ranking of seeds using the exact objective (sum = 1)
    # ------------------------------------------------------------------
    def _obj(x: np.ndarray) -> float:
        return _objective_exact(x)

    evaluated = [(vec, _obj(vec)) for vec in seeds]
    evaluated.sort(key=lambda tup: tup[1])
    top_k = min(15, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = evaluated[0][1]
    print(f"[Init] best seed objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on the smooth surrogate
    # ------------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1e3, 2e3, 5e3,
        1e4, 2e4, 5e4, 1e5,
        2e5, 5e5, 1e6,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS runs.")
            break

        # allocate a modest per‑beta budget – enough for progress, cheap enough
        per_beta = 800
        print(f"\n--- L‑BFGS run #{idx+1} (max_iter/beta={per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_opt = _obj(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")

        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] new global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient on the exact objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # aim for roughly 1e-4 s per iteration → allocate proportionally
        max_exp_iters = int(min(2_500_000, remaining * 2000))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient (exact max) ---")
            seq_exp, val_exp = _exp_gradient_optimize(
                best_seq,
                n,
                deadline,
                max_iters=max_exp_iters,
                eta0=1.0,
            )
            print(f"[ExpGrad] best objective = {val_exp:.6f}")
            if val_exp < best_val:
                best_seq = seq_exp.copy()
                best_val = val_exp
                print("[Update] improved after EG")

    # ------------------------------------------------------------------
    # Stage 3 – long stochastic local search (mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # aim for a few hundred thousand cheap moves
        iters = int(min(600_000, remaining * 500))
        if iters > 0:
            print("\n--- Stochastic local search (mass transfers) ---")
            seq_loc, val_loc = _local_search(
                best_seq,
                n,
                deadline,
                iters=iters,
                step_frac=0.02,
            )
            print(f"[LocalSearch] objective = {val_loc:.6f}")
            if val_loc < best_val:
                best_seq = seq_loc.copy()
                best_val = val_loc
                print("[Update] improved after local search")

    # ------------------------------------------------------------------
    # Stage 4 – final high‑β polishing L‑BFGS
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing L‑BFGS (high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            best_seq,
            n,
            deadline,
            beta_schedule=[5e4, 1e5, 2e5, 5e5, 1e6],
            max_iter_per_beta=800,
        )
        val_final = _obj(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < best_val:
            best_seq = seq_final.copy()
            best_val = val_final
            print("[Update] improved after final polishing")

    # ------------------------------------------------------------------
    # Log the official evaluation (for information only)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for possible warm‑starts
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
