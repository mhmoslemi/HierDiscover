def _smooth_max_and_grad(p, beta):
    conv = _conv_fft(p, p)
    # compute smooth weights w:
    # compute max of beta*conv for stability

    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)

    # correlation of w with p

def _adam_opt(start_seq, n, deadline, total_iters, beta_start=5.0, beta_end=200.0, lr=0.05):
    logits = np.log(start_seq + 1e-30)  # shape (n,)
    m = np.zeros_like(logits)
    v2 = np.zeros_like(logits)

    best_seq = None
    best_val = np.inf # surrogate
    for t in range(1, total_iters+1):
        if time.time() > deadline:
            break
        p = _softmax(logits)
        beta = beta_start + (beta_end - beta_start) * (t / total_iters)
        conv, w, g_p = _smooth_max_and_grad(p, beta)

        # compute surrogate max rather than smooth? We'll compute real max for tracking
        max_conv = np.max(conv)
        val = 2 * n * max_conv
        if val < best_val:
            best_val = val
            best_seq = p.copy()
            # optional print
            if t % 20000 == 0:
                print(f"[Adam] iter {t} val {best_val:.6f} beta {beta:.2f}")

        # gradient wrt logits:
        sum_gp = np.dot(p, g_p)
        grad_logits = p * (g_p - sum_gp)

        # Adam update
        m = 0.9 * m + 0.1 * grad_logits  # beta1 = 0.9

    return best_seq, best_val

1.  **Seed generation** – a small, diverse set of ℓ₁‑normalised vectors:
    uniform, the previous best (if any) and a few random Dirichlet draws.

2.  **Coarse ranking** – keep the 3 best seeds according to the exact
    surrogate 2·n·max(conv) (the denominator is 1 after normalisation).

3.  **Deterministic‑annealing Adam optimisation** – for every top seed
    run a mirror‑descent (Adam) on the *smooth* max‑convolution
    𝓁(β) = (1/β)·log Σᵢ exp(β·(p∗p)ᵢ).  The temperature β grows from a
    small value (β≈5) to a large one (β≈200), i.e. a continuation method.
    The variables are stored in the log‑softmax (logit) space, guaranteeing
    non‑negativity and the ℓ₁‑normalisation automatically.

4.  **Projected sub‑gradient polishing** – the best vector found by Adam
    is refined with a large number of projected sub‑gradient steps on the
    *true* max‑convolution (the non‑smoothed objective).

5.  **Final evaluation** – the best vector is passed to the official
    ``evaluate_sequence`` routine.  The resulting list of floats is stored
    in the global variable ``height_sequence_1`` and returned.

All steps respect the 1000 s wall‑clock budget (a 2 s safety margin is
kept).  The optimiser is deliberately lightweight and uses only NumPy,
so it works on the restricted execution environment.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (real → real)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smooth‑max and its gradient (used by Adam)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(p: np.ndarray, beta: float):
    """
    Returns (conv, softmax‑weights w, gradient wrt p) for the smooth max:
        L(p) = (1/beta) log Σ_i exp(beta * (p∗p)_i)

    Gradient wrt p is   g = 2 * correlate(w, p)   (valid‑mode correlation)
    """
    conv = _conv_fft(p, p)                     # length = 2n-1
    # softmax over beta * conv  (numerically stable)
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()          # length = 2n-1, sums to 1

    # correlation w ⋆ p (valid) = Σ_i w_{i+k} p_i , k = 0..n-1
    corr_full = _conv_fft(w, p[::-1])          # length = 3n-2
    start = len(p) - 1                         # n-1
    g_p = 2.0 * corr_full[start:start + len(p)]   # length = n
    return conv, w, g_p


# ----------------------------------------------------------------------
# Optimiser – Adam on the smooth‑max objective (logit space)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 300_000,
              beta_start: float = 5.0,
              beta_end: float = 200.0,
              base_lr: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser in logit (softmax) space.
    Returns the best vector seen (ℓ₁‑normalised) and its surrogate value.
    """
    # initialise logits from the start sequence
    logits = np.log(start_seq + 1e-30)

    m = np.zeros_like(logits)
    v2 = np.zeros_like(logits)

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    beta_range = beta_end - beta_start
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current probability vector
        p = _softmax(logits)

        # schedule beta (deterministic annealing)
        beta = beta_start + beta_range * (it / max_iters)

        # smooth‑max value and gradient wrt p
        conv, w, g_p = _smooth_max_and_grad(p, beta)

        # surrogate using the true max (for bookkeeping)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = p.copy()
            if it % 20000 == 0 or it == max_iters:
                print(f"[Adam] iter={it:7d} best={best_val:.6f} beta={beta:.2f}")

        # gradient wrt logits (chain rule through softmax)
        sum_gp = np.dot(p, g_p)                       # scalar = Σ_j p_j g_pj
        grad_logits = p * (g_p - sum_gp)              # shape (n,)

        # Adam updates (bias‑corrected)
        m = 0.9 * m + 0.1 * grad_logits
        v2 = 0.999 * v2 + 0.001 * (grad_logits ** 2)
        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        step = base_lr / (np.sqrt(v2_hat) + 1e-8)
        logits = logits - step * m_hat

    return best_seq, best_val


# ----------------------------------------------------------------------
# Classic projected sub‑gradient optimiser (true max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *exact* max‑convolution objective.
    Returns the best vector seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0 or it == max_iters:
                print(f"[SubGrad] iter={it:7d} best={best_val:.6f}")

        # decreasing step‑size and Euclidean projection onto the simplex
        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for the best step‑function within the given time budget.
    The returned list is stored globally as `height_sequence_1`.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – a power‑of‑two length is convenient for the FFT.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 0. Retrieve the previous solution (if present) and adapt its size.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    previous_vec = None
    if isinstance(prev, list) and len(prev) > 0:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.sum() > 0.0:
                if arr.size == n:
                    previous_vec = arr / arr.sum()
                elif arr.size > n:
                    previous_vec = arr[:n]
                    previous_vec = previous_vec / previous_vec.sum()
                else:   # pad with zeros
                    tmp = np.zeros(n, dtype=np.float64)
                    tmp[:arr.size] = arr
                    previous_vec = tmp / tmp.sum()
        except Exception:
            previous_vec = None

    # --------------------------------------------------------------
    # 1. Rich seed generation.
    # --------------------------------------------------------------
    seeds = [_uniform(n)]
    if previous_vec is not None:
        seeds.append(previous_vec)

    # a few random Dirichlet draws with different concentrations
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            if time.time() > deadline:
                break
            seeds.append(_random_dirichlet(n, conc, rng))

    # Normalise (just in case) and discard degenerate vectors
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        total = s.sum()
        if total > 0.0:
            normed_seeds.append(s / total)

    print(f"\n--- Seed generation: {len(normed_seeds)} candidates ---")

    # --------------------------------------------------------------
    # 2. Coarse ranking – keep the best few seeds.
    # --------------------------------------------------------------
    scored = [(vec, _objective_surrogate(vec)) for vec in normed_seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 3
    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Best surrogate among raw seeds = {scored[0][1]:.6f}")

    # --------------------------------------------------------------
    # 3. Deterministic‑annealing Adam optimisation.
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all Adam runs.")
            break

        remaining = max(0.0, deadline - time.time())
        # rough estimate: ~3000 iterations per second on this machine
        est_iters = int(min(600_000, remaining * 3000))
        if est_iters < 20_000:
            est_iters = int(min(200_000, remaining * 3000))
        if est_iters <= 0:
            break

        print(f"\n--- Adam optimisation run #{idx + 1} (max_iters≈{est_iters}) ---")
        seq_opt, val_opt = _adam_opt(seed_vec, n, deadline,
                                     max_iters=est_iters,
                                     beta_start=5.0,
                                     beta_end=200.0,
                                     base_lr=0.5)

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Projected sub‑gradient polishing of the current best.
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(800_000, remaining * 0.5))
        if max_sub > 0:
            print("\n--- Sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq, n,
                                                deadline,
                                                max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[Polish] improved best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final evaluation with the official evaluator.
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
