"""
search_step_function.py

A self‑contained optimizer for the evaluation function

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in several stages:

1. **Seed pool** – a large, diverse collection of ℓ₁‑normalised
   non‑negative vectors (analytic windows, Dirichlet draws, sparse
   supports, random cosine mixtures, β‑shapes, …).  The previous best
   sequence (global ``height_sequence_1``) is also injected.

2. **Screening** – each seed is scored with a cheap surrogate
   ``max(conv)`` (the sum is fixed to 1, therefore the surrogate
   equals the exact objective up to the factor ``2·n``).  The best few
   (``top_k``) seeds survive.

3. **Smooth optimisation** – for every surviving seed we run a
   continuation L‑BFGS on the log‑sum‑exp smooth approximation of the
   max.  The schedule climbs to a very large β (10⁵) so that the
   surrogate is virtually the true max.  The best vector of this
   stage is kept.

4. **Exact refinement** – mirror‑descent with an exponentiated‑gradient
   update on the *exact* max‑convolution objective.  The sub‑gradient
   is averaged over all lags attaining the current maximum, which
   yields a more stable direction.  The iteration count is scaled to
   the remaining time budget.

5. **Perturb‑and‑polish** – a short stochastic phase: the logits of the
   current best vector are jittered, a brief high‑β L‑BFGS run is applied,
   and any improvement is kept.

All stages respect a global 1000 s budget (the caller supplies the
budget via ``budget_s``).  The final best vector is stored in the global
name ``height_sequence_1`` for warm‑starts in later calls and is
returned as a plain list of Python ``float`` objects.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on {1,…,n}.  U‑shape when α,β<1."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                           # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for _ in range(num_freq):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        freq = rng.integers(1, n // 2)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_sparse(n: int, rng: np.random.Generator,
                   k_min: int = 30, k_max: int = 150) -> np.ndarray:
    """Sparse support: choose K random indices, give them random weights."""
    K = int(rng.integers(k_min, k_max + 1))
    idx = rng.choice(n, size=K, replace=False)
    a = np.zeros(n, dtype=np.float64)
    weights = rng.exponential(scale=1.0, size=K)
    a[idx] = weights
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_cosine_mix(n: int, rng: np.random.Generator,
                      num_terms: int = 6) -> np.ndarray:
    """Sum of a few random cosine waves, clipped to non‑negative."""
    i = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for _ in range(num_terms):
        freq = rng.uniform(0.0, 0.5)               # up to Nyquist
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * freq * i + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α)+b·α then ℓ₁‑renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Surrogate objective (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """(value, gradient) for the smooth max‑convolution surrogate."""
    a = _softmax(v)                     # lie on the simplex
    conv = _conv_fft(a, a)              # full auto‑convolution
    shift = np.max(conv)                # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # weights for each lag in the soft‑max
    w = exp_term / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    L‑BFGS‑B optimisation of the smooth surrogate.
    ``start_vec`` is a probability vector (ℓ₁‑normalised).
    The optimisation variable is the logit vector.
    """
    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 2000.0,
                         5000.0, 20000.0, 100000.0]

    # initialise logits safely
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:               # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact objective utilities
# ----------------------------------------------------------------------
def _objective_raw(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) / (sum²).  ``a`` need not sum to 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. ``a``.
    All lags attaining the current maximum are averaged,
    which yields a more stable direction.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    max_b = np.max(conv)
    idxs = np.where(conv >= max_b - 1e-12)[0]   # all active lags

    g = np.zeros_like(a)
    for idx in idxs:
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev = idx - i_range
        g[i_range] += 2.0 * a[rev]
    if len(idxs) > 0:
        g /= float(len(idxs))
    return g


def _exp_gradient_optimize(start_a: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 500_000,
                          eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *exact* max‑convolution objective.
    Returns the best vector encountered and its objective value.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_raw(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_maxconv(a)          # sub‑gradient of max(conv)
        eta = eta0 / math.sqrt(it)           # classic diminishing stepsize

        # exponentiated‑gradient update on logits
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        # occasional exact evaluation
        if it % 100_000 == 0:
            cur = _objective_raw(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[ExpGrad] it={it:,} obj={cur:.6f}")

    # final check
    cur = _objective_raw(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised seed vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12, 20):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.08, 0.45, 9):
        seeds.append(_gaussian(n, sigma))
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))
    for a in np.linspace(0.1, 1.5, 7):
        for b in np.linspace(0.1, 1.5, 7):
            seeds.append(_beta_shape(n, a, b))
    for al in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, al))

    # random continuous draws ---------------------------------------------------------
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # sparse random supports -----------------------------------------------------------
    for _ in range(30):
        seeds.append(_random_sparse(n, rng, k_min=30, k_max=120))

    # random cosine mixtures ---------------------------------------------------------
    for _ in range(40):
        seeds.append(_random_cosine_mix(n, rng, num_terms=rng.integers(3, 9)))

    # mixtures of already generated seeds ---------------------------------------------
    base_idx = np.arange(len(seeds))
    for _ in range(500):
        i, j = rng.choice(base_idx, size=2, replace=False)
        alpha = rng.random()
        mix = _mix_shapes(seeds[i], seeds[j], alpha)
        seeds.append(mix)

    # previous best --------------------------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final clean‑up – non‑negative + ℓ₁‑normalise --------------------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0        # safety margin

    n = int(kwargs.get("n", 4000))
    print(f"\n=== Optimisation start (n={n}) ===")

    # ----------------------------------------------------------------------
    # Stage 0 – seed collection and cheap screening
    # ----------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    def _surrogate(v):
        """2·n·max(conv) – the exact objective for ℓ₁‑normalised vectors."""
        return 2.0 * n * np.max(_conv_fft(v, v))

    # evaluate all seeds quickly (exact surrogate, since sum=1)
    evaluated = [(vec, _surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _surrogate(best_seq)
    print(f"Best surrogate among seeds: {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Stage 1 – high‑β L‑BFGS continuation on the smooth surrogate
    # ----------------------------------------------------------------------
    beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0,
                     200.0, 500.0, 2000.0,
                     5000.0, 20000.0, 100000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS seeds.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx+1}/{top_k} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=2000)
        val_opt = _surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt.copy()
            print(f"[Update] New global best surrogate = {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Stage 2 – exact mirror‑descent (exponentiated gradient)
    # ----------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(4_500_000, remaining * 1e5))
        if max_iters > 0:
            print("\n--- Exponentiated‑gradient refinement (exact max) ---")
            seq_exp, val_exp = _exp_gradient_optimize(best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=max_iters,
                                                      eta0=1.0)
            if val_exp < best_val:
                best_val = val_exp
                best_seq = seq_exp.copy()
                print(f"[ExpGrad] improved exact objective to {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Stage 3 – stochastic perturb‑and‑polish
    # ----------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Stochastic perturb‑and‑polish phase ---")
        attempts = 30
        for a in range(attempts):
            if time.time() > deadline:
                break
            logits = np.log(best_seq + 1e-30)
            scale = max(0.07, 0.7 / (a + 1))   # decreasing jitter amplitude
            jitter = rng.normal(scale=scale, size=n)
            start_vec = _softmax(logits + jitter)

            seq_cand = _lbfgs_smooth_optimize(start_vec,
                                              n,
                                              deadline,
                                              beta_schedule=[1.0, 20.0, 200.0, 2000.0],
                                              max_iter_per_beta=800)
            cand_val = _surrogate(seq_cand)
            if cand_val < best_val:
                best_val = cand_val
                best_seq = seq_cand.copy()
                print(f"[Perturb] improvement after attempt {a+1}: {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Stage 4 – final high‑β polishing
    # ----------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta_iters = max(500,
                             int(min(2500,
                                     remaining * 0.06 * 1e5)))
        print("\n--- Final high‑β L‑BFGS polishing ---")
        seq_final = _lbfgs_smooth_optimize(best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=[50000.0, 100000.0, 200000.0],
                                           max_iter_per_beta=per_beta_iters)
        val_final = _surrogate(seq_final)
        print(f"[Final‑L‑BFGS] surrogate = {val_final:.6f}")
        if val_final < best_val:
            best_val = val_final
            best_seq = seq_final.copy()
            print(f"[Update] New best after polishing = {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Final evaluation with the official evaluator (for reporting)
    # ----------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:          # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
          f"evaluate_sequence = {true_val}")

    # store for warm‑start on the next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
