The goal is to minimise  

\[
F(a)=\frac{2\,n\; \max (a*a)}{(\sum a_i)^2},
\qquad a_i\ge 0,
\]

where `*` denotes the full linear convolution.
The function is invariant under a global scaling of `a`, therefore we can
always work with a **probability vector** (`∑a_i = 1`).  
The problem is non‑convex; we use a multi‑stage heuristic:

1. **Rich seed pool** – many analytically‑defined families (uniform,
   triangular, exponentials, beta‑distributions, power‑laws, random
   Dirichlet draws, low‑frequency curves, …) together with many random
   mixtures.  The pool also contains the previous best solution (warm‑start).

2. **Pre‑screening** – compute the *exact* objective for every seed
   (fast convolution via FFT) and keep the best few (`top_k`).

3. **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp surrogate of
   `max(a*a)`.  We start from every seed in `top_k`, walk through an
   increasing schedule of the temperature parameter `β` and use a generous
   number of L‑BFGS iterations.  The result of each run is remembered.

4. **Mirror‑descent on the exact objective** – exponentiated‑gradient (EG)
   on the *true* non‑smooth function.  EG works on the logits (`log a`)
   and uses the exact sub‑gradient of the active lag.  The step‑size is
   `η_t = η0 / √t` (with a slightly larger `η0` than before) and the
   number of iterations is limited only by the remaining time.

5. **Stochastic annealing** – a cheap hill‑climber that moves a small
   fraction of mass between two random positions.  Worse moves are kept
   with a temperature that slowly cools down, allowing the search to jump
   out of local minima.

6. **Final polishing** – one more short L‑BFGS run at very high `β`
   (≈ 10⁶) to tighten the solution.

All heavy work (convolution, gradients) is performed with FFTs
(`np.fft.rfft`) which is `O(n log n)` and fast enough for `n = 4000`.
The whole routine respects the 1000 s budget, constantly checks the deadline
and stops as soon as time runs out.  The best sequence found is stored in
the global name `height_sequence_1` and returned.

```python
"""search for a step‑function that minimises
   2·n·max(conv(a,a)) / (sum(a)**2).

The algorithm:
  * builds a large, diverse seed pool (analytic families + random Dirichlet)
  * keeps the best few seeds according to the exact objective
  * runs continuation L‑BFGS on a smooth log‑sum‑exp surrogate
  * refines with exponentiated‑gradient mirror descent on the exact objective
  * applies a simulated‑annealing hill‑climber
  * finishes with a high‑β L‑BFGS polishing step
All work is done under the 1000 s budget; the best vector is stored in the
global variable ``height_sequence_1``.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for n≈4000)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Numerical helpers -----------------------------------------------------
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    return e / s


# ----------------------------------------------------------------------
# Analytic families (all returned ℓ₁‑normalised, sum = 1) -------------
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _linear_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)  (i = 0 … n‑1)"""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) on {0,…,n‑1} (discrete) – works for α,β > 0."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)          # Bessel I0 (NumPy)
    return a / a.sum()


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i = 1/(i+1)^p  (i = 0 … n‑1)"""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mix a·(1‑α) + b·α, rebalance to sum‑1."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


# ----------------------------------------------------------------------
# Seed pool construction -------------------------------------------------
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_linear_decrease(n))

    for lam in (0.0002, 0.0004, 0.0005, 0.001, 0.002, 0.005):
        seeds.append(_exponential_decay(n, lam))

    for g in (0.3, 0.5, 0.8, 1.0, 2.0):
        seeds.append(_sine_window(n, gamma=g))

    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.08, 0.45, 8):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 4.0, 21):
        seeds.append(_power_law(n, p))

    # Beta family – we also explore α,β < 1 (U‑shaped)
    for a in (0.15, 0.2, 0.3, 0.5, 0.8, 1.0):
        for b in (0.15, 0.2, 0.3, 0.5, 0.8, 1.0):
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet – very sparse and very flat concentrations
    for conc in (0.001, 0.005, 0.01, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency curves
    for nf in (2, 3, 4, 5):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mix existing seeds (pairwise) – produce many hybrids
    base = np.arange(len(seeds))
    for _ in range(400):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Add the previous solution (warm‑start) if we have one
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Normalise and clip tiny numerical artefacts
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient (max‑convolution) ----------------------
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))  (∑a = 1)"""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the *active* lag of the max‑convolution.
    Returns a vector g with g_i = 2·a_{idx-i} on the active support.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = idx - idx_i
    g[idx_i] = 2.0 * a[rev_idx]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) ---------------------------------------
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Log‑sum‑exp surrogate of max(conv).  Returns (value, grad_v)
    where the variable is the log‑softmax logits.
    """
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)               # full convolution
    shift = np.max(conv)                 # numerically stable shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp                # weights over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)            # length = 3·n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]  # slice of length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 2500,
) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    The optimisation variable is the log‑softmax logits.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits (avoid log(0))

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta,
                         "ftol": 1e-12,
                         "disp": False}
            )
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during β={beta_val:.0f}")
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} (β={beta_val})")
            break

    return _softmax(v)   # final probability vector


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective -----
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 3_500_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *exact* max‑convolution objective.
    Returns (best_a, best_obj).
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        # occasional bookkeeping
        if it % 200_000 == 0:
            cur = _objective_exact(a)
            print(f"[ExpGrad] it={it:,} obj={cur:.6f}")
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    # final check
    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing (mass‑transfer) ---------------------------------
# ----------------------------------------------------------------------
def _simulated_annealing(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_steps: int = 150_000,
    start_temp: float = 0.02,
    end_temp: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Very cheap SA: move a small fraction of probability mass.
    Temperature decreases linearly from start_temp to end_temp.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    rng = np.random.default_rng(777)

    for step in range(1, max_steps + 1):
        if time.time() > deadline:
            break
        # linear temperature schedule
        temp = start_temp + (end_temp - start_temp) * (step / max_steps)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue

        # propose moving a random fraction of a[i]
        delta = (rng.random() * 0.03 + 0.01) * a[i]   # 1–4 % of donor mass
        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta

        cur = _objective_exact(a_cand)
        diff = cur - best_val
        if diff < 0 or rng.random() < np.exp(-diff / temp):
            # accept
            a = a_cand
            if cur < best_val:
                best_val = cur
                best_a = a_cand.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny stochastic local‑search (deterministic hill‑climbing) ----------
# ----------------------------------------------------------------------
def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 6_000,
    step_frac: float = 0.02,
) -> tuple[np.ndarray, float]:
    """Move a fixed fraction of mass from i to j; keep improving moves."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(12345)

    for it in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue

        delta = step_frac * a[i]
        if delta <= 0.0:
            continue

        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta

        val = _objective_exact(a_cand)
        if val < best_val:
            best_val = val
            best_a = a_cand.copy()
            a = a_cand
    return best_a, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a better step‑function within the given time budget.
    Returns the best sequence (list of floats) and stores it globally.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search, n={n}, budget≈{budget_s}s ===")

    # --------------------------------------------------------------
    # Seed generation
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # --------------------------------------------------------------
    # Quick exact‑objective ranking (keep the best few)
    # --------------------------------------------------------------
    evals = [(vec, _objective_exact(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = min(18, len(evals))
    top_seeds = [vec for vec, _ in evals[:top_k]]

    print(f"Top {top_k} seed exact values:")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on smooth surrogate
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_exact(global_best_seq)
    print(f"\n[Init] exact = {global_best_val:.6f}")

    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        10000.0, 20000.0, 50000.0, 100000.0,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline before finishing L‑BFGS pool.")
            break
        # give each run a modest per‑beta budget
        per_beta = max(600, int(min(3000,
                                    (deadline - time.time()) * 0.05 * 1e5)))
        print(f"\n--- L‑BFGS run #{idx+1} (max_iter/β={per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] exact = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient (exact max‑convolution)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(3_500_000, remaining * 0.30 * 1e5))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient (exact) ---")
            seq_exp, val_exp = _exp_gradient_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_exp_iters,
                eta0=1.5,
            )
            print(f"[ExpGrad] best = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] New best after EG = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – simulated annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        steps = int(min(200_000, remaining * 80))  # cheap, time‑aware
        if steps > 0:
            print("\n--- Simulated annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing(
                global_best_seq,
                n,
                deadline,
                max_steps=steps,
                start_temp=0.03,
                end_temp=0.001,
            )
            print(f"[SA] best = {val_sa:.6f}")
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[Update] New best after SA = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – tiny deterministic local search
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(8_000, remaining * 25))
        if iters > 0:
            print("\n--- Local search (deterministic) ---")
            seq_loc, val_loc = _local_search(
                global_best_seq, n, deadline, iters=iters, step_frac=0.02
            )
            print(f"[Local] best = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] New best after local = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – final high‑β polishing L‑BFGS
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(500, int(min(2500,
                                    remaining * 0.07 * 1e5)))
        print("\n--- Final high‑β L‑BFGS polishing ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[200000.0, 500000.0, 1_000_000.0],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] exact = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polish = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final official evaluation (only for logging)
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest exact ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for warm‑starts on the next call
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```
