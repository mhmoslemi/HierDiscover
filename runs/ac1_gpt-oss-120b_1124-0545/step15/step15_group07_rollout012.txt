import scipy.sparse as sp

def _lp_step(a: np.ndarray, n: int, deadline: float) -> np.ndarray | None:
    M = np.max(_conv_fft(a, a))
    if time.time() > deadline:
        return None

    # Build sparse matrix A_ub
    # Offsets for each column (col index j) is -j.
    offsets = -np.arange(n, dtype=int)
    # For each diagonal, create constant vector of length n (a[j])
    # Use sp.diags with a list of scalars and offsets.
    # According to SciPy docs: if a diagonal is scalar, it's broadcasted.
    # Let's try: diags([a[j] for j in range(n)], offsets, shape=(2*n-1, n), format='csr')
    # This should create each diagonal of constant a[j].
    try:
        A_ub = sp.diags([float(val) for val in a], offsets, shape=(2*n-1, n), format='csr')
    except Exception as e:
        # Fallback to building full matrix
        rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
        cols = np.tile(np.arange(n), n)
        # Might exceed shape though for rows > 2n-2. Actually rows = i+j; range 0..2n-2 fine.
        # But we need to restrict rows < 2n-1.
        rows = rows[rows < 2*n-1]
        cols = cols[rows < 2*n-1]
        data = np.take(a, rows - cols)  # a[i] where i = row-col
        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Build RHS
    b_ub = np.full(2*n-1, M, dtype=np.float64)

    # Objective: minimize -sum(b) (i.e., maximize sum)
    c = -np.ones(n, dtype=np.float64)

    # Solve LP
    # time for linprog can be limited indirectly by passing max_iter options, but we rely on time limit otherwise.

    # Using SciPy's 'highs' method:
    result = sp.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n,
                                 method='highs', options={'presolve': True})
    if not result.success:
        print("[LP] optimization failed:", result.message)
        return None

    b = result.x
    sum_b = b.sum()
    if sum_b <= 1.0 + 1e-12:
        # No improvement
        return None

    b_norm = b / sum_b
    return b_norm

"""
Propose a step‑function that minimizes

        2·n·max(conv(a, a)) / (sum(a)²)

using a diversified seed pool, L‑BFGS on a smooth surrogate,
exact mirror‑descent (exponentiated‑gradient), aggressive hill‑climbing,
and a single linear‑programming refinement (Frank‑Wolfe–style update).
The LP step often yields a modest reduction of the max‑convolution peak
by finding a feasible direction that increases the ℓ₁‑mass while keeping
the current peak bound.  The whole pipeline respects the 1000 s budget
and stores the best sequence in the global variable ``height_sequence_1``.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sparse

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector.
    (If sum(seq)=1 then the true objective equals this value.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p  (decreasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _linear_increasing_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ i^p  (increasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = i ** p
    a[0] = 1e-30
    return a / a.sum()


def _V_shape(n: int, p: float) -> np.ndarray:
    """U‑shaped distribution – high at the ends, low in the centre."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = (np.abs(i - c) + 1e-6) ** p
    return a / a.sum()


def _cos_shape(n: int, p: float) -> np.ndarray:
    """Cosine‑based shape, peaks at the borders (|cos|^p)."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture, then ℓ₁‑normalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed generation – diverse random sweep over parametrised families
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None,
                   max_random: int = 2500) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))
    seeds.append(_V_shape(n, p=1.0))
    seeds.append(_V_shape(n, p=2.0))
    seeds.append(_cos_shape(n, p=1.0))
    seeds.append(_cos_shape(n, p=2.0))

    # random parametric families
    for _ in range(max_random // 5):
        a = 10 ** rng.uniform(-3, 1)      # alpha ∈ [0.001,10]
        b = 10 ** rng.uniform(-3, 1)      # beta  ∈ [0.001,10]
        seeds.append(_beta_shape(n, a, b))

        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        lam = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_exp_decay(n, lam))

        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        # V‑shaped random powers
        pV = rng.uniform(0.5, 3.0)
        seeds.append(_V_shape(n, pV))

        # cosine‑shaped random powers
        pC = rng.uniform(0.5, 3.0)
        seeds.append(_cos_shape(n, pC))

        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(
            n,
            10 ** rng.uniform(-2, 1),
            10 ** rng.uniform(-2, 1)
        )
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # sanitisation – guarantee positivity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimises a smooth surrogate of max‑convolution using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 3000.0, 5000.0, 8000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stay on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                      # numeric guard
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                     # soft‑max weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length (2n‑1)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = opt.minimize(fun=fun,
                               x0=v,
                               method='L-BFGS-B',
                               jac=True,
                               callback=callback,
                               options={'maxiter': max_iter_per_beta,
                                        'ftol': 1e-13,
                                        'disp': False})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] time limit reached (beta={beta_val})")
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact sub‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 600_000) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size ηₜ = 0.5 / √t
        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{cur_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 500_000) -> np.ndarray:
    """
    Simple simulated‑annealing style hill‑climber.
    Moves a small amount of mass from the current largest entry to a
    random other entry, accepting improvements always and uphill moves
    with a linearly decaying temperature.
    """
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(0xC0FFEE)
    total_iters = max_iters

    for it in range(total_iters):
        if time.time() > deadline:
            break

        # Linear schedules
        progress = it / total_iters
        temp = (cur_val * 0.05) * (1.0 - progress) + (cur_val * 1e-6) * progress
        eps = 0.10 * (1.0 - progress) + 1e-6 * progress

        # donor = index of the largest component
        i = int(np.argmax(a))
        if a[i] <= 1e-15:
            continue
        # random target, different from i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        epsilon = eps * rng.random()
        epsilon = min(epsilon, a[i])           # never go negative

        a[i] -= epsilon
        a[j] += epsilon

        new_val = _objective_surrogate(a)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                if it % 20000 == 0:
                    print(f"[HillClimb] it={it:,} best≈{best_val:.6f}")
        else:
            a[j] -= epsilon
            a[i] += epsilon

    return best_seq


# ----------------------------------------------------------------------
# Linear‑programming refinement (Frank‑Wolfe style)
# ----------------------------------------------------------------------
def _lp_refine(current: np.ndarray,
               n: int,
               deadline: float,
               max_iters: int = 5) -> tuple[np.ndarray, float]:
    """
    Solve the LP:
        max  Σ b_j
        s.t. conv(current, b) ≤ M   (M = current max‑convolution)
             b_j ≥ 0
    Return a scaled feasible point if it yields a lower surrogate.
    """
    best_seq = current.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        M = np.max(_conv_fft(best_seq, best_seq))

        # Build the constraint matrix A_ub where
        #   A_ub[k, j] = best_seq[i]  with i = k - j (if in range)
        # This is a Toeplitz matrix; we construct it via sparse diagonals.
        offsets = -np.arange(n, dtype=int)           # column - row = -i
        diag_vals = best_seq.tolist()                # constant on each diagonal
        A_ub = sparse.diags(diag_vals, offsets,
                            shape=(2 * n - 1, n), format='csr')
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        c = -np.ones(n, dtype=np.float64)           # minimise -Σb ⇒ maximise Σb
        bounds = [(0.0, None)] * n

        # Solve LP (highs is efficient for moderate size problems)
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs')

        if not res.success:
            print(f"[LP] iteration {it + 1} failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-9:
            # No ℓ₁‑mass increase ⇒ cannot improve the peak bound
            break

        # Scale to unit ℓ₁‑mass
        g = b / sum_b
        new_val = _objective_surrogate(g)

        if new_val + 1e-12 < best_val:
            best_seq = g
            best_val = new_val
            print(f"[LP] iteration {it + 1} improved surrogate to {best_val:.6f}")
        else:
            # No improvement – stop
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Upsampling (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence as a plain list of floats and stores it
    in the global name ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # --------------------------------------------------------------
    # Seed generation & cheap ranking
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=2500)
    print(f"Generated {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best surrogate values (top {top_k} seeds):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1}: {val:.6f}")

    # --------------------------------------------------------------
    # L‑BFGS refinement on the current best seed
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] Current best surrogate = {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- L‑BFGS refinement of the best seed ---")
        seq_opt = _lbfgs_smooth_optimize(global_best_seq, n, deadline,
                                         beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                        200.0, 500.0, 1000.0,
                                                        3000.0, 5000.0, 8000.0],
                                         max_iter_per_beta=8000)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Exact exponentiated‑gradient refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(2_500_000, remaining * 0.25 * 1e5))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient refinement (exact max) ---")
            seq_exp = _exp_gradient_optimize(global_best_seq, n, deadline,
                                            max_iters=max_exp_iters)
            val_exp = _objective_surrogate(seq_exp)
            print(f"[ExpGrad] surrogate = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Hill‑climbing (simulated annealing) phase
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc_iters = int(min(800_000, remaining * 0.30 * 1e5))
        if max_hc_iters > 0:
            print("\n--- Hill‑climbing (simulated annealing) ---")
            seq_hc = _hill_climb(global_best_seq, n, deadline,
                                 max_iters=max_hc_iters)
            val_hc = _objective_surrogate(seq_hc)
            print(f"[HillClimb] surrogate = {val_hc:.6f}")
            if val_hc < global_best_val:
                global_best_seq = seq_hc.copy()
                global_best_val = val_hc
                print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Linear‑programming refinement (Frank‑Wolfe style)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Linear‑programming refinement ---")
        seq_lp, val_lp = _lp_refine(global_best_seq, n, deadline, max_iters=5)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[Update] After LP refinement surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Optional up‑sampling & fine‑tuning on a finer grid
    # --------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 30.0:
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} ---")
            up_seq = _upsample(global_best_seq, new_n)

            # short L‑BFGS on the finer grid
            up_opt = _lbfgs_smooth_optimize(up_seq, new_n, deadline,
                                            beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                           200.0, 500.0, 1000.0],
                                            max_iter_per_beta=4000)
            up_val = _objective_surrogate(up_opt)
            print(f"[Upsample‑L‑BFGS] surrogate = {up_val:.6f}")

            if up_val < global_best_val:
                global_best_seq = up_opt.copy()
                global_best_val = up_val
                n = new_n
                print(f"[Update] New best after up‑sampling = {global_best_val:.6f}")

            # final exponentiated‑gradient on fine grid
            if time.time() < deadline:
                rem = deadline - time.time()
                max_exp2 = int(min(800_000, rem * 0.10 * 1e5))
                if max_exp2 > 0:
                    print("\n--- Final exponentiated‑gradient on fine grid ---")
                    seq_final = _exp_gradient_optimize(global_best_seq, n,
                                                       deadline,
                                                       max_iters=max_exp2)
                    val_final = _objective_surrogate(seq_final)
                    if val_final < global_best_val:
                        global_best_seq = seq_final.copy()
                        global_best_val = val_final
                        print(f"[Final‑ExpGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
