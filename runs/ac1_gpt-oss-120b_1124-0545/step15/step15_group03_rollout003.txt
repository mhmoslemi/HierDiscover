def _smooth_max_opt(start_seq, n, deadline, alpha_schedule, steps_per_stage):
   a = start_seq.copy()
   best_seq = a.copy()
   best_val = _objective_surrogate(a)  # actual max conv
   for stage_idx, (alpha, iters) in enumerate(alpha_schedule):
        for it in range(1, iters+1):
            if time.time() > deadline: break outer
            # compute conv a*a
            conv = _conv_fft(a, a)
            # compute softmax weights
            w = np.exp(alpha * conv - np.max(alpha * conv))  # for stability
            w_sum = np.sum(w)
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w = w / w_sum
            # compute gradient
            grad = 2 * np.convolve(w, a[::-1])[n-1:n-1+n]
            # step size schedule
            step = 0.2 / math.sqrt(it)   # or maybe constant small after some stage
            a = a - step * grad
            # projection onto simplex
            a = _proj_simplex(a)
            # evaluate actual max conv occasionally
            if it % 5000 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
   return best_seq, best_val

"""
Propose a step‑function (list of non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is full linear convolution.  The algorithm improves on the
reference solution by:

* generating a richer set of analytic seeds;
* refining each seed with a *smooth‑max* (log‑sum‑exp) gradient descent
  that uses the exact convolution gradients (two FFTs per iteration);
  the smoothness parameter α is increased gradually (annealed) so that the
  optimisation first shapes the whole spectrum and then focuses on the true
  max‑convolution value;
* polishing the best candidate with a projected sub‑gradient method;
* a final short simulated‑annealing pass in the log‑softmax space.

All steps respect the 1000 s wall‑clock budget and the sequence is kept on
the probability simplex (non‑negative, sum = 1) via Euclidean projection.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(m log m))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle; sigma = sigma_frac·n."""
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _inverse_tent(n: int, p: float = 1.0) -> np.ndarray:
    """U‑shaped “tent” – high at the edges, low in the centre."""
    i = np.arange(n, dtype=np.float64)
    d = np.abs(i - (n - 1) / 2.0) + 1e-12
    a = d ** p
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_k_uniform(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=min(k, n), replace=False)
    a[idx] = 1.0 / len(idx)
    return a

def _smooth_max_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    total_iters: int,
                    alphas: list[float]) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smooth approximation of max(conv).
    The smooth max is  (1/α)·log Σ exp(α·conv).
    ``alphas`` is a list of smoothing parameters – we anneal from small
    to large values, spending equal effort for each α.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # split iterations evenly among the α‑stages
    stage_iters = [total_iters // len(alphas)] * len(alphas)
    for i in range(total_iters - sum(stage_iters)):
        stage_iters[i] += 1

    t = 0                      # global iteration counter
    eta0 = 0.5                 # base step‑size (will decay as 1/√t)

    for alpha, iters in zip(alphas, stage_iters):
        if time.time() > deadline:
            break
        for _ in range(iters):
            if time.time() > deadline:
                break
            t += 1
            # 1) current self‑convolution
            conv = _conv_fft(a, a)                       # length 2n‑1
            # 2) smooth‑max weights (softmax over α·conv)
            scaled = alpha * conv
            scaled -= np.max(scaled)                     # for numerical stability
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w = w / w_sum

            # 3) gradient of the smooth max (2·(w * a) convolution)
            grad = 2.0 * _conv_fft(w, a[::-1])[n - 1:n - 1 + n]

            # 4) simple diminishing step‑size
            eta = eta0 / math.sqrt(t)
            a = a - eta * grad

            # 5) project back onto the simplex
            a = _proj_simplex(a)

            # occasional exact evaluation (every 10 k iterations)
            if t % 10000 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
    return best_seq, best_val


def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Simple SA in the log‑softmax (logit) space."""
    rng = np.random.default_rng(12345)
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.06 * (1.0 - progress) + 1e-7
        sigma = 0.20 * (1.0 - progress) + 1e-5

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = np.exp(proposal_logits - np.max(proposal_logits))
        proposal_seq /= proposal_seq.sum()
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0           # small safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Re‑use a previous solution if present.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    previous_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                previous_vec = arr / arr.sum()
        except Exception:
            previous_vec = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation.
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # simple analytic families
    seeds.append(_uniform(n))
    seeds.append(_inverse_tent(n, p=1.0))
    seeds.append(_inverse_tent(n, p=2.0))

    if previous_vec is not None:
        seeds.append(previous_vec)

    # beta‑shape grid (coarse)
    for a in np.linspace(0.2, 5.0, 7):
        for b in np.linspace(0.2, 5.0, 7):
            seeds.append(_beta_shape(n, a, b))

    # random beta‑shape draws (log‑uniform parameters)
    for _ in range(80):
        a = 10 ** rng.uniform(-2, 2)   # 0.01 … 100
        b = 10 ** rng.uniform(-2, 2)
        seeds.append(_beta_shape(n, a, b))

    # power‑law family
    for p in np.linspace(0.0, 5.0, 9):
        seeds.append(_power(n, p))
    for _ in range(30):
        seeds.append(_power(n, rng.uniform(0.0, 5.0)))

    # Gaussian family
    for sigma in np.linspace(0.10, 0.45, 5):
        seeds.append(_gaussian(n, sigma))
    for _ in range(20):
        seeds.append(_gaussian(n, rng.uniform(0.05, 0.55)))

    # Sparse k‑point uniform
    for k in (10, 20, 50, 100):
        seeds.append(_random_k_uniform(n, k, rng))

    # Random Dirichlet draws (several concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            if time.time() > deadline:
                break
            seeds.append(_random_dirichlet(n, conc, rng))

    # Normalise and discard pathological vectors
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        total = s.sum()
        if total > 0.0:
            normed_seeds.append(s / total)

    print(f"\n--- Seed generation: {len(normed_seeds)} candidates ---")

    # ------------------------------------------------------------------
    # 2. Rank seeds using the exact surrogate.
    # ------------------------------------------------------------------
    scored = []
    for vec in normed_seeds:
        val = _objective_surrogate(vec)
        scored.append((vec, val))
    scored.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Best surrogate among raw seeds = {scored[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 3. Smooth‑max gradient descent on the top seeds.
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # allocate a large chunk of the time budget to the smooth‑max optimisation
    while time.time() < deadline and top_seeds:
        remaining = deadline - time.time()
        # roughly 70 % of the remaining budget for this phase
        phase_budget = remaining * 0.70
        # decide a realistic number of iterations (≈3000 it/s on a typical CPU)
        est_iters = int(min(800_000, phase_budget * 3000))
        if est_iters < 20_000:
            est_iters = int(min(200_000, phase_budget * 3000))
        if est_iters <= 0:
            break

        seed_vec = top_seeds.pop(0)   # take the next best seed
        print(f"\n--- Smooth‑max optimisation run (≈{est_iters} iters) ---")
        alphas = [1.0, 5.0, 15.0, 40.0]          # increasing smoothness
        seq_opt, val_opt = _smooth_max_opt(seed_vec, n, deadline,
                                           total_iters=est_iters,
                                           alphas=alphas)

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Sub‑gradient polishing (focus on the true max).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(800_000, remaining * 0.5))
        if max_sub > 0:
            print("\n--- Sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq, n,
                                                deadline, max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved best to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Simulated‑annealing refinement (small final push).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(300_000, remaining * 0.3))
        if max_sa > 0:
            print("\n--- Simulated‑annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing(global_best_seq, n,
                                                  deadline, max_iters=max_sa)
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[SA] final best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
