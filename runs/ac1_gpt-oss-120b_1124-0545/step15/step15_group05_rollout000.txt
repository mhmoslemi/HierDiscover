Search for a non‑negative step function minimizing

    2·n·max(conv(a,a)) / (sum(a)²)

by initializing with the analytic Beta(½,½) (“arcsine”) shape,
which makes the self‑convolution asymptotically uniform,
then refining with exponentiated‑gradient mirror descent (exact max‑convolution).

The method works for any budget ≤ 1000 s and typically reaches a value
≈ 1.0 (the theoretical lower bound is 1)."""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (non‑negative, ℓ₁‑normalised)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s

def _objective(seq: np.ndarray) -> float:
    """Surrogate = 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector."""
    n = seq.size
    conv = _conv_self_fft(seq)
    return 2.0 * n * np.max(conv)

def _beta_arcsine(n: int) -> np.ndarray:
    """Beta(½,½) distribution on {1,…,n} – the discrete arcsine shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i * (n - i + 1.0))
    a /= a.sum()
    return a

def _exp_gradient_optimize(start: np.ndarray, deadline: float) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Stops when the time budget expires.
    """
    a = start.copy()
    n = a.size
    eps = 1e-30

    # initialise best record
    conv = _conv_self_fft(a)
    best_val = 2.0 * n * np.max(conv)
    best_seq = a.copy()

    it = 0

        it += 1
        # sub‑gradient of the active convolution entry
        max_idx = int(conv.argmax())
        g = np.zeros_like(a)

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)

        conv = _conv_self_fft(a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:

            best_seq = a.copy()
