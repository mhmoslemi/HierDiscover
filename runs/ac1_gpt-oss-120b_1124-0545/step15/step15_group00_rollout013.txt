def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet distribution with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Surrogate / exact objective helpers
# ----------------------------------------------------------------------
def _surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by the official evaluator."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Build a diversified set of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # analytic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.10, 0.18, 0.30):
        seeds.append(_gaussian(n, sigma))

    for p in (0.0, 0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))

    # Dirichlet draws --------------------------------------------------
    concs = (0.05, 0.2, 0.5, 1.0, 2.0, 5.0)
    for conc in concs:
        for _ in range(6):
            d = _random_dirichlet(n, conc, rng)
            seeds.append(d)                     # raw
            seeds.append(np.sort(d)[::-1])      # descending
            # shuffled variants
            rng.shuffle(d)
            seeds.append(d.copy())

    # previous best ----------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final normalisation (just in case)
    out = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            out.append(v / s)
        else:
            out.append(_uniform(n))
    return out


# ----------------------------------------------------------------------
# L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth(start_vec: np.ndarray,
                  n: int,
                  deadline: float,
                  beta_schedule: list[float] | None = None,
                  max_iter_per_beta: int = 3000) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) using L‑BFGS‑B on a smooth max‑conv."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                # stays on the simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # log‑sum‑exp smooth approximation of max
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth max part
            w = exp_beta / sum_exp                # soft weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # derivative w.r.t. a

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # bail‑out if time ran out
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] Exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (objective = 2·n·max(conv)/(sum²))
# ----------------------------------------------------------------------
def _exact_subgrad(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 600_000) -> np.ndarray:
    """Projected sub‑gradient descent on the true objective."""
    a = start_vec.copy()
    best = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k] / (np.sum(a) ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if it % 100_000 == 0:
                print(f"[Sub‑Grad] it={it:,d} best={best_val:.6f}")

        # sub‑gradient of max(conv) at index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # derivative of denominator part
        S = np.sum(a)
        denom_grad = -4.0 * n * conv[k] / (S ** 3) * np.ones_like(a)

        # total sub‑gradient of the objective
        subgrad = (2.0 * n / (S ** 2)) * g + denom_grad

        # step size (standard diminishing schedule)
        lr = 0.5 / math.sqrt(it)
        a -= lr * subgrad
        a = _proj_simplex(a)            # keep a on the simplex (∑a=1, a≥0)

    return best


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (log‑it space)
# ----------------------------------------------------------------------
def _sim_anneal(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 300_000) -> np.ndarray:
    """Simple SA walk in log‑it space."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.08 * (1.0 - progress) + 1e-6   # temperature schedule
        sigma = 0.12 * (1.0 - progress) + 1e-4

        proposal = logits + sigma * rng.normal(size=n)
        cand_seq = _softmax(proposal)
        cand_val = _objective(cand_seq)

        if cand_val < cur_val or math.exp(-(cand_val - cur_val) / T) > rng.random():
            logits = proposal
            cur_seq = cand_seq
            cur_val = cand_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:,d} best={best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence found as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # Basic initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # small safety margin

    # ------------------------------------------------------------------
    # Problem size – a value around 4000 gives a good trade‑off
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Seed pool (with possibility of using the previous best)
    # ------------------------------------------------------------------
    print(f"\n=== Seed construction (n={n}) ===")
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape[0] != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    raw_seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(raw_seeds)} raw seeds.")

    # cheap surrogate ranking (exact for ℓ₁‑normalised vectors)
    ranked = sorted(((s, _surrogate(s)) for s in raw_seeds), key=lambda x: x[1])
    top_k = 20
    top_seeds = [s for s, _ in ranked[:top_k]]
    print("\nBest surrogate values among raw seeds:")
    for i, (_, v) in enumerate(ranked[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS on the top seeds
    # ------------------------------------------------------------------
    global_best = top_seeds[0].copy()
    global_best_val = _surrogate(global_best)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1000.0, 2000.0, 5000.0,
                  10000.0, 20000.0, 50000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline hit before finishing L‑BFGS passes.")
            break

        remaining = deadline - time.time()
        # allow at most a few hundred thousand LBFGS iterations in total
        max_iter_beta = int(min(4000, remaining * 0.07 * 1e5))
        max_iter_beta = max(1800, max_iter_beta)

        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter/β = {max_iter_beta}) ---")
        refined = _lbfgs_smooth(seed_vec, n, deadline,
                                beta_schedule=beta_sched,
                                max_iter_per_beta=max_iter_beta)
        val = _surrogate(refined)
        print(f"[L‑BFGS] surrogate = {val:.6f}")

        if val < global_best_val:
            global_best = refined.copy()
            global_best_val = val
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Exact sub‑gradient descent (focus on true objective)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(1_200_000, remaining * 0.07 * 1e5))
        print("\n--- Exact sub‑gradient descent ---")
        sub_opt = _exact_subgrad(global_best, n, deadline, max_iters=max_sub)
        sub_val = _objective(sub_opt)
        print(f"[Sub‑Grad] true objective = {sub_val:.6f}")
        if sub_val < _objective(global_best):
            global_best = sub_opt.copy()

    # ------------------------------------------------------------------
    # Simulated‑annealing final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(800_000, remaining * 0.06 * 1e5))
        print("\n--- Simulated‑annealing refinement ---")
        sa_opt = _sim_anneal(global_best, n, deadline, max_iters=max_sa)
        sa_val = _objective(sa_opt)
        print(f"[SA] true objective = {sa_val:.6f}")
        if sa_val < _objective(global_best):
            global_best = sa_opt.copy()

    # ------------------------------------------------------------------
    # Final evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"true objective = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best.tolist()
    return global_best.tolist()
