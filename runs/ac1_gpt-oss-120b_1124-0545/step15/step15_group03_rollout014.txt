def _smooth_mirror_descent(start_seq, n, deadline, total_iters, initial_tau, tau_decay_rate):

   log_a = np.log(start_seq + 1e-30)

   best_seq = None, best_val = inf

   m = np.zeros_like(log_a) ; v = np.zeros_like(log_a) ; beta1=0.9 ; beta2=0.999; epsilon = 1e-8

   for t in range(1, total_iters+1):

        if time.time()>deadline: break

        a = _softmax(log_a)   # ensures sum=1

        # compute convolution
        conv = _conv_fft(a, a)
        max_conv = np.max(conv)
        # surrogate objective
        val = 2.0 * n * max_conv
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        # compute gradient
        # Use tau schedule: tau = max(initial_tau * (tau_decay_rate ** (t/total_iters)), min_tau)
        tau = max(initial_tau * (tau_decay_rate ** (t/total_iters)), 1e-8) # decays exponential
        w = np.exp(conv / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0/len(w))
        else:
            w /= w_sum

        # compute gradient: g = 2 * conv_rev[n-1:n-1+n] where conv_rev = conv(w, a_rev)
        a_rev = a[::-1]
        conv_rev = _conv_fft(w, a_rev)
        g = 2.0 * conv_rev[n-1:n-1+n]

        # Adam update on log_a
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        eta = 0.5 / math.sqrt(t)  # or constant?
        log_a -= eta * m_hat / (np.sqrt(v_hat) + epsilon)

   return best_seq, best_val

def _smooth_mirror_descent(...):
    log_a = np.log(start_seq + eps)
    best_seq = None
    best_val = inf
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1=0.9; beta2=0.999; eps_adam=1e-8

    for it in range(1, max_iters+1):
        if time.time()>deadline:break
        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        val = 2.0*n*max_conv
        if val<best_val:
            best_val=val; best_seq=a.copy()
        # compute dynamic tau: tau = max_tau * (0.5**(it/phase_len))
        # use schedule: tau = max( 0.1*max_conv, 0.001 * (max_iters-it)/max_iters * max_conv )
        tau = 0.1 * max_conv * max(1.0 - it/max_iters, 0.01)
        # compute w
        shift = conv - max_conv  # <=0
        w = np.exp(shift / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0/len(w))
        else:
            w /= w_sum
        # compute gradient
        a_rev = a[::-1]
        conv_rev = _conv_fft(w, a_rev)
        g = 2.0 * conv_rev[n-1:n-1+n]
        # Adam update
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        eta = 0.5 / math.sqrt(it)
        log_a -= eta * m_hat / (np.sqrt(v_hat) + eps_adam)

    return best_seq, best_val

def _hill_climb(seq, n, deadline, max_moves=20000):
    a = seq.copy()
    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0*n*conv.max()
    for move in range(max_moves):
        if time.time()>deadline: break
        # Choose two random indices i,j distinct
        i,j = rng.choice(n, 2, replace=False)
        # Determine max delta we can move from i to j without negative
        delta_max = a[i]
        if delta_max <= 0: continue
        # Choose delta maybe small fraction of a[i] i.e., uniform random between 0 and some fraction
        delta = delta_max * rng.random() * 0.1  # up to 10% of i weight
        if delta <= 0: continue
        # Apply move
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # compute conv and val
        conv_new = _conv_fft(a_new, a_new)
        val_new = 2.0*n*conv_new.max()
        if val_new < best_val - 1e-12:  # improvement
            a = a_new
            best_seq = a_new.copy()
            best_val = val_new
            conv = conv_new
        # else reject
    return best_seq, best_val

"""
A faster, more aggressive optimiser for the self‑convolution problem.

The aim is to minimise   2·n·max(conv(a,a))/(sum a)² .
Because the objective is homogeneous we can enforce sum(a)=1; then the
objective reduces to 2·n·max(conv).  The optimiser works in three
phases:

1️⃣  **Rich seeding** – many analytical families (uniform, beta‑shape,
    power‑law, Gaussian, sinusoid, random Dirichlet, sparse k‑point) are
    generated and ℓ₁‑normalised.

2️⃣  **Refinement** – the best seed is refined with three complementary
    methods (all respect the simplex constraint):
       • *Exponentiated‑gradient* (mirror‑descent) with a classic
         0.5/√t schedule (fast, cheap).
       • *Smooth mirror‑descent* that minimises a soft‑max
         approximation of the ℓ∞‑norm (gradient uses the full
         convolution but stays cheap thanks to FFT).
       • *Projected sub‑gradient* polishing (many cheap steps).

3️⃣  **Local hill‑climbing** – random “mass‑transfer” moves
    (i→j) are tried; each move is accepted only if it reduces the
    surrogate value.  This cheap stochastic search often finds the last
    few percent of improvement.

All work is time‑aware; the function stops before the 1000 s budget
and returns the best sequence found.  The final sequence is stored in
the global ``height_sequence_1`` for reuse by future runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _sin_shape(n: int, power: float = 1.0) -> np.ndarray:
    """A half‑sine shape, optionally sharpened with a power."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** power
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_k_uniform(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=min(k, n), replace=False)
    a[idx] = 1.0 / len(idx)
    return a


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑conv."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered‑pair version)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _smooth_mirror_descent(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 120_000,
                           tau0_factor: float = 0.5,
                           tau_decay: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on a smooth approximation of max(conv):
    F(a)=τ·log Σ exp(conv/τ).  τ is annealed geometrically.
    Uses Adam‑style adaptive steps in log‑space.
    """
    eps = 1e-30
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + eps)

    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)                 # simplex projection
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # annealed temperature, never smaller than 1e-12
        tau = max(tau0_factor * max_conv * (tau_decay ** (it / max_iters)), 1e-12)

        # soft‑max weights (stable)
        shift = conv - max_conv                 # ≤0
        w = np.exp(shift / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / len(w))
        else:
            w /= w_sum

        # gradient = 2·corr(w, a)  (extract the middle n entries)
        a_rev = a[::-1]
        conv_rev = _conv_fft(w, a_rev)          # length 3n‑2
        g = 2.0 * conv_rev[n - 1:n - 1 + n]

        # Adam update in log‑space
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = 0.6 / math.sqrt(it)               # a bit larger than EG step‑size
        log_a -= eta * m_hat / (np.sqrt(v_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑conv."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_moves: int = 15_000,
                rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: repeatedly move a small amount of mass from
    one index to another and keep the move if the surrogate improves.
    """
    if rng is None:
        rng = np.random.default_rng()
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * conv.max()
    best_seq = a.copy()

    for mv in range(max_moves):
        if time.time() > deadline:
            break

        i, j = rng.choice(n, 2, replace=False)
        if a[i] <= 0.0:
            continue
        # move up to 10 % of the source mass
        delta = a[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        conv_new = _conv_fft(a_new, a_new)
        val_new = 2.0 * n * conv_new.max()
        if val_new < best_val:
            a = a_new
            conv = conv_new
            best_val = val_new
            best_seq = a_new.copy()

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 works well for the current benchmark.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Re‑use the previous solution if available.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    previous_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                previous_vec = arr / arr.sum()
        except Exception:
            previous_vec = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation.
    # ------------------------------------------------------------------
    seeds = []

    # baseline uniform
    seeds.append(_uniform(n))

    # previous best (if any)
    if previous_vec is not None:
        seeds.append(previous_vec)

    # beta‑shape grid
    for a in np.linspace(0.2, 5.0, 9):
        for b in np.linspace(0.2, 5.0, 9):
            seeds.append(_beta_shape(n, a, b))

    # beta‑shape random draws
    for _ in range(80):
        a = 10 ** rng.uniform(-2, 2)
        b = 10 ** rng.uniform(-2, 2)
        seeds.append(_beta_shape(n, a, b))

    # power‑law grid + random
    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))
    for _ in range(40):
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

    # Gaussian grid + random sigma
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    for _ in range(25):
        sigma = rng.uniform(0.05, 0.55)
        seeds.append(_gaussian(n, sigma))

    # Sinusoid family (different powers)
    for pw in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_sin_shape(n, power=pw))

    # Sparse k‑point uniform
    for k in (10, 20, 50, 100, 200):
        seeds.append(_random_k_uniform(n, k, rng))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            if time.time() > deadline:
                break
            seeds.append(_random_dirichlet(n, conc, rng))

    # Normalise and discard degenerate vectors
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        total = s.sum()
        if total > 0.0:
            normed_seeds.append(s / total)

    print(f"\n--- Seed generation complete: {len(normed_seeds)} candidates ---")

    # ------------------------------------------------------------------
    # 2. Evaluate all seeds (exact surrogate) and keep the best few.
    # ------------------------------------------------------------------
    scored = [(vec, _objective_surrogate(vec)) for vec in normed_seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 7
    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Best surrogate among raw seeds = {scored[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation on the very best seed.
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ----- a) Exponentiated‑gradient (fast) ---------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(400_000, remaining * 3000))
        print(f"\n--- Exponentiated‑gradient (max_iters≈{max_iters}) ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq, n, deadline, max_iters=max_iters)
        if val_exp < global_best_val:
            global_best_seq, global_best_val = seq_exp, val_exp
            print(f"[Update] after EG: {global_best_val:.6f}")

    # ----- b) Smooth mirror‑descent (soft‑max) -------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(140_000, remaining * 2500))
        print(f"\n--- Smooth mirror‑descent (max_iters≈{max_iters}) ---")
        seq_smooth, val_smooth = _smooth_mirror_descent(
            global_best_seq, n, deadline, max_iters=max_iters,
            tau0_factor=0.6, tau_decay=0.45)
        if val_smooth < global_best_val:
            global_best_seq, global_best_val = seq_smooth, val_smooth
            print(f"[Update] after smooth MD: {global_best_val:.6f}")

    # ----- c) Sub‑gradient polishing (many cheap steps) --------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(800_000, remaining * 0.55))
        print("\n--- Sub‑gradient polishing ---")
        seq_sub, val_sub = _subgradient_opt(global_best_seq, n, deadline, max_iters=max_sub)
        if val_sub < global_best_val:
            global_best_seq, global_best_val = seq_sub, val_sub
            print(f"[Update] after sub‑grad: {global_best_val:.6f}")

    # ----- d) Local hill‑climbing (random mass transfers) -------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_moves = int(min(30_000, remaining * 30))
        print("\n--- Random hill‑climbing ---")
        seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline, max_moves=max_moves, rng=rng)
        if val_hc < global_best_val:
            global_best_seq, global_best_val = seq_hc, val_hc
            print(f"[Update] after hill‑climb: {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
