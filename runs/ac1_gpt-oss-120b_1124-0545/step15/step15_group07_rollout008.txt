"""
A fast multi‑stage optimizer for the step‑function objective.

The goal is to minimise
    2·n·max(conv(a,a)) / (sum(a)**2)

which, after normalising the ℓ₁‑norm to 1, is exactly
    2·n·max(conv(a,a)).
The algorithm proceeds as follows

0. Build a large, diverse pool of ℓ₁‑normalised seeds (uniform,
   triangular, sines, Gaussians, power‑laws, Beta‑shapes,
   exponential decay, Dirichlet draws and low‑frequency random
   Fourier series).

1. Keep the K best seeds according to a cheap surrogate
   (the exact objective, but without the final division).

2. Refine each elite seed with a smooth‑max L‑BFGS optimiser
   (log‑sum‑exp surrogate).  The best result is kept.

3. Run a custom Adam optimiser on the logits of the best vector,
   using the *exact* sub‑gradient of the max‑convolution.

4. Apply a targeted hill‑climber:
   pick the index that contributes most to the current maximal
   convolution entry and move a small amount of mass to an index
   that contributes least.  Accept only improving moves (with a
   tiny simulated‑annealing tail).

5. If time permits, up‑sample the current best vector (linear
   interpolation to twice the length) and repeat steps 2–4 on the
   finer grid.

The final vector (as a plain list) is stored in the global name
``height_sequence_1`` for reuse.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.fft

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq))  (seq is ℓ₁‑normalised, sum=1).
    This is exactly the evaluation score used in the competition.
    """
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Parametric shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like polynomial shape."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for _ in range(num_freq):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * (rng.integers(1, n)) * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed generation – large random sweep over parametrised families
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    max_random: int = 2000,
) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_power_law(n, 0.0))
    seeds.append(_power_law(n, 1.0))
    seeds.append(_power_law(n, 2.0))

    # random parametric families
    for _ in range(max_random // 5):
        a = 10 ** rng.uniform(-2, 1)  # alpha ∈ [0.01,10]
        b = 10 ** rng.uniform(-2, 1)  # beta  ∈ [0.01,10]
        seeds.append(_beta_shape(n, a, b))

        p = rng.uniform(0.0, 4.0)
        seeds.append(_power_law(n, p))

        lam = rng.exponential(scale=0.001) + 1e-4
        seeds.append(_exp_decay(n, lam))

        # mixture of uniform and a beta shape
        mix_alpha = rng.uniform(0.0, 1.0)
        beta_vec = _beta_shape(n,
                               10 ** rng.uniform(-2, 1),
                               10 ** rng.uniform(-2, 1))
        mixed = (1.0 - mix_alpha) * _uniform(n) + mix_alpha * beta_vec
        seeds.append(mixed / mixed.sum())

    # Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # sanitise: non‑negative and ℓ₁‑normalised
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule=None,
    max_iter_per_beta: int = 3000,
) -> np.ndarray:
    """
    Optimises a smooth surrogate of max‑convolution using L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector (soft‑max of logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1000.0, 3000.0, 8000.0, 15000.0]

    v = np.log(start_vec + 1e-30)   # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)               # numeric guard
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # soft‑max weights for the active index k
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            # convolution of rev_a with w gives gradient of each a_i
            full = _conv_fft(rev_a, w)[:2 * n - 1]
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline hit (beta={beta_val})")
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam optimiser on logits (exact sub‑gradient of max‑conv)
# ----------------------------------------------------------------------
def _adam_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 120_000,
    lr_init: float = 0.3,
) -> np.ndarray:
    """
    Mirror‑descent / Adam on logits using the *exact* sub‑gradient of
    max(conv).  Returns the best sequence seen.
    """
    a = start_seq.copy()
    v = np.log(a + 1e-30)                # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # softmax
        a = _softmax(v)

        # exact sub‑gradient of the (non‑smoothed) max‐conv
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        grad_a = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx
        grad_a[idx] = 2.0 * a[rev_idx]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        # learning‑rate schedule (slow decay)
        lr = lr_init / np.sqrt(t)

        v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

        # keep track of the best solution
        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress report
        if t % 20_000 == 0:
            print(f"[Adam] iter={t:,} best≈{best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Targeted hill‑climbing (move mass from the worst contributor to the best)
# ----------------------------------------------------------------------
def _targeted_hill_climb(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> np.ndarray:
    """
    Greedy hill‑climbing: at each iteration move a tiny amount of mass
    from the index that contributes most to the current maximal
    convolution entry to an index that contributes least.
    """
    a = start_seq.copy()
    cur_val = _objective_surrogate(a, n)
    best_seq = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(0xDEADBEEF)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx
        partner = np.zeros_like(a)
        partner[idx] = a[rev_idx]                     # w_i = a_{k‑i}

        # source = index with largest partner weight (largest impact on max)
        src = int(idx[np.argmax(partner[idx])])
        # target = index with smallest partner weight (ideally 0)
        tgt_candidates = np.where(partner == partner.min())[0]
        # ensure we do not pick the source again
        tgt_candidates = tgt_candidates[tgt_candidates != src]
        if tgt_candidates.size == 0:
            tgt = (src + rng.integers(1, n - 1)) % n
        else:
            tgt = int(rng.choice(tgt_candidates))

        # how much mass to move?
        epsilon = min(0.02 * a[src], 1e-3)   # modest step
        if epsilon <= 0.0:
            continue

        a[src] -= epsilon
        a[tgt] += epsilon

        new_val = _objective_surrogate(a, n)

        if new_val < cur_val:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                if it % 50_000 == 0:
                    print(f"[HillClimb] it={it:,} best≈{best_val:.6f}")
        else:
            # revert
            a[tgt] -= epsilon
            a[src] += epsilon

    return best_seq


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and re‑normalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()

            val_hc = _objective_surrogate(seq_hc, n)
            if val_hc < global_best_val:
                global_best_seq = seq_hc.copy()
                global_best_val = val_hc
                print(f"[HillClimb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling + fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 30.0:
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} ---")
            up_seq = _upsample(global_best_seq, new_n)

            # L‑BFGS on finer grid (short schedule)
            up_opt = _lbfgs_smooth_optimize(
                up_seq,
                new_n,
                deadline,
                beta_schedule=[1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0],
                max_iter_per_beta=2000,
            )
            up_val = _objective_surrogate(up_opt, new_n)
            print(f"[Upsample‑L‑BFGS] surrogate = {up_val:.6f}")

            if up_val < global_best_val:
                global_best_seq = up_opt.copy()
                global_best_val = up_val
                n = new_n
                print(f"[Update] new best after up‑sampling = {global_best_val:.6f}")

            # Adam on fine grid
            if time.time() < deadline:
                rem = deadline - time.time()
                iters = int(min(80_000, rem * 60))
                if iters > 0:
                    print("\n--- Adam on up‑sampled grid ---")
                    seq_fine = _adam_optimize(global_best_seq, n,
                                              deadline, max_iters=iters)
                    val_fine = _objective_surrogate(seq_fine, n)
                    if val_fine < global_best_val:
                        global_best_seq = seq_fine.copy()
                        global_best_val = val_fine
                        print(f"[Fine‑Adam] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
