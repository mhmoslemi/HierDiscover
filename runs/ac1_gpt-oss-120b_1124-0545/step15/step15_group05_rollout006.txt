def _adam_exact_optimize(start_vec, n, deadline, steps=??, lr=0.05):
    logits = np.log(start_vec + eps)
    m=v=0.
    best_seq = start_vec; best_val = _objective_surrogate(start_vec, n)
    for t in range(1, steps+1):
        if time.time() > deadline: break
        a = _softmax(logits)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # compute gradient using max index only:
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # gradient for logits
        dot = np.dot(g, a)
        grad_logits = a * (g - dot)
        # Adam update (beta1,beta2 default)
        m = beta1*m + (1-beta1)*grad_logits
        v = beta2*v + (1-beta2)*(grad_logits**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        # track best
        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val; best_seq = a.copy()
    return best_seq
```

"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

within a 1000 s budget.

Algorithm
---------
0. Fix a problem size n (4000) – a good trade‑off between quality and speed.
1. Initialise the search from the previously best sequence (if available)
   otherwise from the uniform distribution.
2. Run several *Adam‑exact* refinements:
   – variables are stored as logits (log‑softmax) to stay on the simplex.
   – each iteration uses the true sub‑gradient of the max‑convolution
     (i.e. the active shift only) and Adam’s adaptive learning rates.
   – the best observed vector is kept.
3. Finish with a long *exponentiated‑gradient* (mirror‑descent) phase
   on the exact max‑convolution, again with a diminishing step‑size.
4. A few short jittered Adam runs are performed as polishing.
5. Return the best sequence and store it in the global name
   ``height_sequence_1`` for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – always ℓ₁‑normalised, non‑negative."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector."""
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _uniform(n: int) -> np.ndarray:
    """Uniform ℓ₁‑normalised vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Adam optimiser that uses the *exact* max‑convolution sub‑gradient
# ----------------------------------------------------------------------
def _adam_exact_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    steps: int = 500_000,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Adam optimiser on the *exact* max‑convolution.
    Returns a simplex‑normalised vector (soft‑max of logits).
    """
    eps = 1e-30
    logits = np.log(start_vec + eps)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    best_seq = start_vec.copy()
    best_val = _objective_surrogate(best_seq, n)

    t = 0
    for _ in range(steps):
        t += 1
        if time.time() > deadline:
            break

        a = _softmax(logits)                     # current probability vector
        conv = _conv_fft(a, a)                    # linear convolution
        max_idx = int(np.argmax(conv))

        # --- sub‑gradient of the active convolution entry -----------------
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        # 2·a[rev_idx] because derivative of a_i·a_j + a_j·a_i = 2·a_j
        g[idx_i] = 2.0 * a[rev_idx]

        # back‑prop through softmax
        dot = np.dot(g, a)
        grad_logits = a * (g - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # keep logits numerically sane
        logits = np.clip(logits, -50, 50)

        # occasional best‑track
        if t % 500 == 0:
            cur_val = _objective_surrogate(a, n)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    return best_seq


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 3_000_000,
) -> np.ndarray:
    """
    Mirror‑descent (exponentiated gradient) on the *exact* max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient at the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size ηₜ = 1/√t  (more aggressive than the original 0.5)
        eta = 1.0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 250_000 == 0:
            cur_val = _objective_surrogate(a, n)
            print(f"[ExpGrad] it={it:,} surrogate≈{cur_val:.6f}")
        # keep the best observed vector
        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq


# ----------------------------------------------------------------------
# Polishing helper – short Adam runs from jittered logits
# ----------------------------------------------------------------------
def _polish_with_jitter(
    base_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    attempts: int = 3,
) -> np.ndarray:
    """
    Perform a few short Adam runs starting from jittered logits.
    Keep any improvement.
    """
    best_seq = base_seq.copy()
    best_val = _objective_surrogate(best_seq, n)

    for k in range(attempts):
        if time.time() > deadline:
            break
        logits = np.log(best_seq + 1e-30)
        jitter = rng.normal(scale=0.02, size=n)  # modest noise
        logits_j = logits + jitter

        refined = _adam_exact_optimize(
            start_vec=_softmax(logits_j),
            n=n,
            deadline=deadline,
            steps=200_000,
            lr=0.07,
        )
        cur_val = _objective_surrogate(refined, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = refined
            print(f"[Polish] improvement after jitter #{k+1}: {cur_val:.6f}")

    return best_seq
