"""
Step‑function optimiser.

The algorithm proceeds in several stages (each respecting the global
time‑budget):

1. **Seed generation** – a diverse pool of ℓ₁‑normalised candidates:
   uniform, the previous best (if any), exponential / reversed‑exponential
   families, beta‑shapes, power‑laws and many Dirichlet draws.

2. **L‑BFGS on a smooth max‑convolution surrogate** – the max is
   approximated by a log‑sum‑exp with an increasing temperature β.
   The optimiser works on logits (soft‑max parametrisation) and uses a
   long schedule of β values and up to several thousand L‑BFGS iterations
   per β.

3. **Adam on the same surrogate** – a first‑order method with momentum
   that finishes the optimisation for a very large β (≈1e6) and many
   cheap iterations.

4. **Exponentiated‑gradient polishing** – mirrors the “mirror‑descent”
   step used previously but with a larger iteration budget.

5. **Simulated‑annealing local search** – random mass transfers between
   two positions while keeping the ℓ₁‑norm fixed; uphill moves are accepted
   with a temperature schedule, giving a chance to escape shallow local
   minima.

All intermediate vectors stay on the probability simplex via the soft‑max
or an exact sum‑preserving move, so the final sequence is automatically
non‑negative and normalised.  The best sequence found is stored in the
global variable `height_sequence_1` and returned as a plain list.
"""

# Low-level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective utilities (surrogate = 2·n·max(conv) because ℓ₁‑norm = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a ℓ₁‑normalised vector (sum==1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i/n)  for i=0,…,n‑1."""
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·(n‑1‑i)/n)."""
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc`."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector (softmax of the final logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [0.1, 0.5, 1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 2000.0, 10000.0,
                         50000.0, 1e5, 5e5, 1e6]

    # Initialise logits (log‑softmax parametrisation)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stays on the simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            # log‑sum‑exp smoothing
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a

def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   beta: float = 1e6,
                   max_iters: int = 2_000_000,
                   lr: float = 0.05,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the (large‑β) smoothed max‑convolution.
    Returns (best_vector, best_surrogate_value).
    """
    v = np.log(start_vec + 1e-30)          # logits
    m = np.zeros_like(v)
    vexp = np.zeros_like(v)

    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        # smoothing; for very large beta the exponent may under/overflow,
        # shift protects it.
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = exp_beta.sum()
        w = exp_beta / sum_exp

        # gradient wrt a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_v
        vexp = beta2 * vexp + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = vexp / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep track of best surrogate value
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # occasional progress report
        if it % 250_000 == 0:
            print(f"[Adam] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

# Exponentiated‑gradient (mirror descent) optimiser on the exact max‑conv.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (softmax parametrisation)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.6 / math.sqrt(it)          # step‑size schedule
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

# Simulated‑annealing local search
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             rng: np.random.Generator,
                             max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves a random amount of mass from one coordinate to another.
    Keeps the vector ℓ₁‑normalised exactly.
    Returns (best_seq, best_surrogate).
    """
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    # initialise temperature as a small fraction of the current objective
    T = cur_val * 0.01
    decay = 0.99995  # slow cooling

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        if a[i] == 0.0:
            continue

        # transfer a random fraction of mass from i to j
        frac = rng.random() * 0.2          # up to 20 % of a[i]
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate new surrogate
        new_val = _objective_surrogate(a_new)

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # metropolis rule
            prob = math.exp(-(new_val - cur_val) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a.copy()
                # occasional progress line
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} new best={best_val:.6f}")

        # temperature schedule
        T *= decay
        if T < 1e-12:
            T = 1e-12

    return best_seq, best_val

    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_adam = int(min(2_000_000, remaining * 0.05 * 1e5))
        print(f"\n--- Adam polishing (max_iters={max_adam}) ---")
        seq_adam, val_adam = _adam_optimize(global_best_seq,
                                            n,
                                            deadline,
                                            beta=1e6,
                                            max_iters=max_adam,
                                            lr=0.05)
        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp = int(min(2_000_000, remaining * 0.07 * 1e5))
        print(f"\n--- Exponentiated‑gradient polishing (max_iters={max_exp}) ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                             n,
                                             deadline,
                                             max_iters=max_exp)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(300_000, remaining * 0.05 * 1e5))
        if max_sa > 0:
            print(f"\n--- Simulated‑annealing polishing (max_iters={max_sa}) ---")
            seq_sa, val_sa = _simulated_annealing_opt(global_best_seq,
                                                      n,
                                                      deadline,
                                                      rng,
                                                      max_iters=max_sa)
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
