"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

using a multi‑stage local‑search pipeline:

1.  Generate a large, diverse pool of ℓ₁‑normalised seeds (uniform,
    triangular, sine‑windows, Gaussians, power‑laws, Beta‑shapes,
    exponential decays, random Dirichlet draws, low‑frequency curves
    and the previous best if any).

2.  Keep the few best seeds according to the cheap surrogate
    ``S(a)=2·n·max(conv(a,a))`` (sum(a)=1).

3.  Refine the elite seed with a continuation L‑BFGS optimiser that
    minimises a smooth approximation of the max (log‑sum‑exp) and
    gradually increases the smoothing parameter β up to 20000.

4.  Perform a long exact exponentiated‑gradient (mirror‑descent) run
    on the refined vector.  The step‑size ηₜ=γ/√t with γ≈0.7 works well
    in practice; the number of iterations is chosen from the remaining
    time budget.

5.  Apply a focused hill‑climbing / simulated‑annealing procedure.
    The move always takes mass from an index that contributes most to
    the current maximal convolution entry and deposits it at a random
    low‑contributing index.  Metropolis acceptance with a linearly
    decaying temperature allows the method to escape shallow plateaus.

6.  If enough time is left, up‑sample the current best vector (double
    the grid size), repeat a short L‑BFGS + exponentiated‑gradient +
    hill‑climb refinement on the finer grid and keep the better of the
    two lengths.

7.  Finally evaluate the best candidate with the official ``evaluate_sequence``
    function, save it in the global name ``height_sequence_1`` and return
    it as a plain list.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector.
    (If sum(seq)=1 then the true objective equals this value.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p  (decreasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _linear_increasing_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ i^p  (increasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = i ** p
    a[0] = 1e-30  # avoid exact zero before normalisation
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture, then ℓ₁‑normalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()

# Seed generation – large random sweep over parametrised families
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None,
                   max_random: int = 4000) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # random parametric families
    for _ in range(max_random // 5):
        a = 10 ** rng.uniform(-3, 1)      # alpha ∈ [0.001,10]
        b = 10 ** rng.uniform(-3, 1)      # beta  ∈ [0.001,10]
        seeds.append(_beta_shape(n, a, b))

        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        lam = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_exp_decay(n, lam))

        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(n,
                             10 ** rng.uniform(-2, 1),
                             10 ** rng.uniform(-2, 1))
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final sanitisation – guarantee positivity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 600) -> np.ndarray:
    """
    Optimises a smooth surrogate of max‑convolution using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 3000.0, 8000.0,
                         20000.0]

    # initialise logits (log‑softmax representation)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stay on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                       # stability shift
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a for the smooth max (soft‑max weights)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # same length as a

            # back‑prop through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-13,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] time limit reached (beta={beta_val})")
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)

# Exact sub‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 800_000,
                           gamma: float = 0.7) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = gamma / np.sqrt(it)                     # step‑size schedule
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Focused hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _active_hill_climb(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 300_000) -> np.ndarray:
    """
    Move mass from indices that contribute most to the current maximal
    convolution entry to random low‑contributing indices.  Accept improvements
    always, uphill moves with a linearly decaying temperature.
    """
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(0xC0FFEE)
    total_iters = max_iters

    for it in range(total_iters):
        if time.time() > deadline:
            break

        progress = it / total_iters
        # temperature decays from ~5 % of current value to near‑zero
        temp = cur_val * (0.05 * (1.0 - progress) + 1e-6 * progress)

        conv = _conv_fft(a, a)
        k_max = int(np.argmax(conv))

        # active index range for the max entry
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_max - idx_i
        contrib = a[idx_i] * a[rev_idx]               # contribution of each term

        # donor = index with largest contribution (break ties randomly)
        if contrib.size == 0:
            continue
        donor = idx_i[int(np.argmax(contrib))]
        # target = a random index different from donor (prefer low contribution)
        target = rng.integers(n - 1)
        if target >= donor:
            target += 1

        # step size: up to a few percent of donor mass
        eps_frac = 0.02 * rng.random() + 1e-6
        epsilon = eps_frac * a[donor]
        if epsilon <= 0.0:
            continue

        # apply move
        a[donor] -= epsilon
        a[target] += epsilon

        new_val = _objective_surrogate(a)

        # Metropolis criterion
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                if it % 50000 == 0:
                    print(f"[HillClimb] it={it:,} best≈{best_val:.6f}")
        else:
            # revert
            a[target] -= epsilon
            a[donor] += epsilon

    return best_seq


# ----------------------------------------------------------------------
# Upsampling (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Stage 0 – problem size
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # ------------------------------------------------------------------
    # Stage 1 – seed generation & cheap ranking
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, (list, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=4000)
    print(f"Generated {len(seeds)} seed vectors.")

    # cheap surrogate ranking (sum == 1 ⇒ surrogate = true objective)
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best surrogate values (top {top_k} seeds):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – continuation L‑BFGS refinement of the elite seed(s)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] Current best surrogate = {global_best_val:.6f}")

    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1000.0, 3000.0, 8000.0, 20000.0]

    # we spend the bulk of the early budget on the single best seed
    if time.time() < deadline:
        print("\n--- L‑BFGS refinement (high‑β schedule) ---")
        seq_opt = _lbfgs_smooth_optimize(global_best_seq, n, deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=800)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long exact exponentiated‑gradient run
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate roughly 40 % of remaining time to this phase
        max_iters = int(min(1_200_000, remaining * 0.4 * 1e4))
        print(f"\n--- Exponentiated‑gradient (max_iters={max_iters}) ---")
        seq_exp = _exp_gradient_optimize(global_best_seq, n, deadline,
                                         max_iters=max_iters, gamma=0.7)
        val_exp = _objective_surrogate(seq_exp)
        print(f"[ExpGrad] surrogate = {val_exp:.6f}")
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – focused hill‑climbing / simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc_iters = int(min(800_000, remaining * 0.3 * 1e4))
        print(f"\n--- Focused hill‑climbing (max_iters={max_hc_iters}) ---")
        seq_hc = _active_hill_climb(global_best_seq, n, deadline,
                                    max_iters=max_hc_iters)
        val_hc = _objective_surrogate(seq_hc)
        print(f"[HillClimb] surrogate = {val_hc:.6f}")
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – optional up‑sampling & fine‑tuning on a finer grid
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 35.0:
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} ---")
            up_seq = _upsample(global_best_seq, new_n)

            # short L‑BFGS on the finer grid
            up_opt = _lbfgs_smooth_optimize(up_seq, new_n, deadline,
                                            beta_schedule=beta_sched[:7],
                                            max_iter_per_beta=500)
            up_val = _objective_surrogate(up_opt)
            print(f"[Upsample‑L‑BFGS] surrogate = {up_val:.6f}")

            if up_val < global_best_val:
                global_best_seq = up_opt.copy()
                global_best_val = up_val
                n = new_n
                print(f"[Update] New best after up‑sampling = {global_best_val:.6f}")

            # final exponentiated‑gradient on the fine grid
            if time.time() < deadline:
                rem = deadline - time.time()
                max_exp_fine = int(min(500_000, rem * 0.2 * 1e4))
                print("\n--- Final exponentiated‑gradient on fine grid ---")
                seq_final = _exp_gradient_optimize(global_best_seq, n,
                                                   deadline,
                                                   max_iters=max_exp_fine,
                                                   gamma=0.7)
                val_final = _objective_surrogate(seq_final)
                print(f"[Final‑ExpGrad] surrogate = {val_final:.6f}")
                if val_final < global_best_val:
                    global_best_seq = seq_final.copy()
                    global_best_val = val_final
                    print(f"[Update] New global best after final fine‑tune = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator and return
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
