def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors, sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i) (i = 0 … n‑1)"""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s


def _edge_exponential(n: int, lam: float) -> np.ndarray:
    """U‑shaped: a_i ∝ exp(-lam * min(i, n-1-i))"""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    s = a.sum()
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape on the discrete set {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i = 1/(i+1)^p  (i = 0 … n‑1)"""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α) + b·α, renormalised to sum‑1."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed‑pool construction
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families (including many extreme β‑shapes)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Exponential families (decay from left / right)
    for lam in (0.001, 0.002, 0.005, 0.01, 0.02):
        seeds.append(_exponential_decay(n, lam))
        seeds.append(_edge_exponential(n, lam))

    # Beta families – we sweep very small shape parameters to get
    # pronounced U‑shapes.
    alphas = (0.05, 0.10, 0.20, 0.30, 0.50, 0.80, 1.0, 1.5, 2.0)
    for a in alphas:
        for b in alphas:
            seeds.append(_beta_shape(n, a, b))

    # Symmetric edge‑biased mixtures of two exponentials
    for lam in (0.005, 0.01, 0.02):
        left = _exponential_decay(n, lam)
        right = left[::-1]
        seeds.append(_mix_shapes(left, right, 0.5))

    # Gaussian bumps (different widths)
    for sigma in np.linspace(0.08, 0.30, 7):
        seeds.append(_gaussian(n, sigma))

    # Power‑law families – can generate strong edge bias
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        seeds.append(_power_law(n, p))

    # Random Dirichlet draws – wide range of concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Pairwise mixes of already‑generated seeds (adds variability)
    base = np.arange(len(seeds))
    for _ in range(250):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Warm start – previous solution if supplied
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up – enforce non‑negativity & ℓ₁ normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Exact evaluation helpers
# ----------------------------------------------------------------------
def _exact_objective(a: np.ndarray) -> float:
    """Exact value of the official objective (sum is assumed 1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) with respect to a."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    # g_i = 2 * a_{idx-i} for all i that contribute to this lag
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = idx - idx_i
    g[idx_i] = 2.0 * a[rev_idx]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Log‑sum‑exp surrogate for max‑convolution.
    Returns (obj, grad) where grad is w.r.t. the logits v.
    """
    a = _softmax(v)               # probabilities on the simplex
    conv = _conv_fft(a, a)
    shift = np.max(conv)          # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # soft‑max weights over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3·n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 3000,
) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    The variable is the vector of logits (log‑softmax).
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 100000.0,
        ]

    v = np.log(start_vec + 1e-30)   # initialise logits safely

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            # early abort if we have overrun the deadline
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            # deadline reached in the middle of an L‑BFGS run
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)   # final probability vector


# ----------------------------------------------------------------------
# Mirror‑descent on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    Returns (best_a, best_obj) found before the deadline.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _exact_objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 100_000 == 0:
            cur = _exact_objective(a)
            print(f"[ExpGrad] iter={it:,} obj={cur:.6f}")
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _exact_objective(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny stochastic local‑search (mass transfers)
# ----------------------------------------------------------------------
def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 6000,
    step_frac: float = 0.025,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: move a small fraction of mass from a random
    index i to a random index j and keep improvements.
    """
    a = start_a.copy()
    best_val = _exact_objective(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456)

    for it in range(iters):
        if time.time() > deadline:
            break

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue

        delta = step_frac * a[i]
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        val = _exact_objective(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand   # continue from the improved point

    return best_a, best_val


# ----------------------------------------------------------------------
# Final polishing – a short re‑run after small random perturbations
# ----------------------------------------------------------------------
def _perturb_and_refine(
    base_a: np.ndarray,
    n: int,
    deadline: float,
    num_trials: int = 3,
) -> np.ndarray:
    """Add tiny Gaussian noise to the logits and run a brief L‑BFGS."""
    best = base_a.copy()
    best_val = _exact_objective(best)

    rng = np.random.default_rng(20230827)

    for trial in range(num_trials):
        if time.time() > deadline:
            break

        # perturb logits (i.e. the log‑softmax parameters)
        logits = np.log(best + 1e-30)
        noise = rng.normal(scale=0.05, size=logits.shape)
        perturbed = np.exp(logits + noise)
        perturbed /= perturbed.sum()

        # a short run – few iterations, high β
        seq = _lbfgs_smooth_optimize(
            perturbed,
            n,
            deadline,
            beta_schedule=[20000.0, 50000.0, 100000.0],
            max_iter_per_beta=1200,
        )
        val = _exact_objective(seq)
        if val < best_val:
            best_val = val
            best = seq.copy()
            print(f"[Perturb] trial {trial+1} improved to {best_val:.6f}")

    return best


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a lower evaluation score.
    The function obeys the 1000 s time limit and stores the best sequence
    in the global variable ``height_sequence_1`` for warm‑starting.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== start search (n={n}, budget≈{budget_s}s) ===")

    # --------------------------------------------------------------
    # Seed generation + quick ranking
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"generated {len(seeds)} diverse seeds")

    # Evaluate the exact surrogate (2·n·max(conv)) – this is the true score
    evals = [(vec, _exact_objective(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = min(12, len(evals))
    top_seeds = [vec for vec, _ in evals[:top_k]]

    print("top seed scores (exact objective):")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on the smooth surrogate
    # --------------------------------------------------------------
    global_best = top_seeds[0].copy()
    global_best_val = _exact_objective(global_best)
    print(f"\n[init] best exact objective = {global_best_val:.6f}")

    # betas – a balanced schedule (moderate number of stages)
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        10000.0, 20000.0, 50000.0, 100000.0,
    ]

    # Run L‑BFGS on the very best seeds (at most 5, time‑aware)
    max_lbfgs_runs = 5
    runs_done = 0
    while runs_done < max_lbfgs_runs and time.time() < deadline:
        idx = runs_done
        seed_vec = top_seeds[idx]
        remaining = deadline - time.time()
        per_beta = max(1000, int(min(4000, remaining * 0.07)))  # adapt to remaining time

        print(f"\n--- L‑BFGS run #{idx+1} (max_iter/beta={per_beta}) ---")
        seq = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val = _exact_objective(seq)
        print(f"[L‑BFGS] exact objective = {val:.6f}")

        if val < global_best_val:
            global_best = seq.copy()
            global_best_val = val
            print(f"[update] new global best = {global_best_val:.6f}")

        runs_done += 1

    # --------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient (exact max‑convolution)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 0.25 * 1e5))
        if max_iters > 0:
            print("\n--- Exponentiated‑gradient (exact) ---")
            seq_eg, val_eg = _exp_gradient_optimize(
                global_best,
                n,
                deadline,
                max_iters=max_iters,
                eta0=1.2,
            )
            print(f"[ExpGrad] best objective = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best = seq_eg.copy()
                global_best_val = val_eg
                print(f"[update] after EG: {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – stochastic local‑search refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(10_000, remaining * 40))   # cheap but many moves
        if iters > 0:
            print("\n--- Local‑search (mass transfers) ---")
            seq_loc, val_loc = _local_search(
                global_best, n, deadline, iters=iters, step_frac=0.025
            )
            print(f"[Local] objective = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best = seq_loc.copy()
                global_best_val = val_loc
                print(f"[update] after local search = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – final perturb‑and‑refine (small noise + short L‑BFGS)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final perturb‑and‑refine ---")
        global_best = _perturb_and_refine(global_best, n, deadline, num_trials=4)
        global_best_val = _exact_objective(global_best)
        print(f"[final] objective after perturb = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final check with the official evaluator (for logging only)
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best.tolist())
        print(
            f"\n=== finished ===\nbest exact surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for warm‑starts in the next round
    globals()["height_sequence_1"] = global_best.tolist()
    return global_best.tolist()
