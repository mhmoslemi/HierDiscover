   Each candidate is normalised to ℓ₁‑norm 1 and scored with the exact
   surrogate 2·n·max(conv) (this equals the true objective after the
   usual normalisation).  The best few seeds are kept.

2. **Intensive smooth‑max optimisation** – starting from each of the
   selected seeds we run a continuation L‑BFGS optimiser on the
   log‑sum‑exp surrogate of the maximum convolution.  The β‑schedule
   is extended far beyond the previous run (up to 10⁵) and we allocate a
   larger iteration budget per β.  The optimiser works in log‑softmax
   space, stays on the probability simplex and therefore respects the
   non‑negativity and normalisation constraints.

3. **Final polishing** – a short exponentiated‑gradient (mirror‑descent)
   run refines the best sequence using the exact max‑convolution
   sub‑gradient, followed by one more L‑BFGS pass with an even larger
   β‑schedule.

All work respects the 1000 s time budget and the final sequence is
returned as a plain list of floats (also stored in ``height_sequence_1``).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# optional SciPy import – required for L‑BFGS
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability‑simplex utilities (log‑softmax and projection)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback = uniform
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact (ℓ₁‑normalised) surrogate: 2·n·max(conv)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a probability vector (ℓ₁‑norm = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (always return a *probability* vector)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int,
                     sigma_frac: float = 0.15,
                     shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                # Bessel I0 (numpy provides it)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 15000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns a probability vector (ℓ₁‑normalised).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        # extended schedule – goes up to 1e5
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0, 10000.0,
            20000.0, 50000.0, 100000.0
        ]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                       # stays on the simplex
            conv = _conv_fft(a, a)                       # full convolution
            max_c = np.max(conv)
            # log‑sum‑exp (smooth max)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # softmax weights over the convolution entries
            w = exp_beta / sum_exp

            # gradient w.r.t a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 250_000) -> np.ndarray:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Returns the refined probability vector.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # learning rate schedule (decreasing)
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        # occasional progress report
        if it % 100_000 == 0:
            print(f"[ExpGrad] it={it} surrogate={_objective_surrogate(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Random seed pool builder
# ----------------------------------------------------------------------
def _collect_random_seeds(n: int,
                          rng: np.random.Generator,
                          limit: int,
                          previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce up to *limit* random vectors from a diverse collection of families.
    Every vector is ℓ₁‑normalised.
    """
    seeds: list[np.ndarray] = []

    # a few analytic families (coarse)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.4, 0.8, 1.2, 2.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    # parametric families with random parameters
    for _ in range(4):
        sigma = rng.uniform(0.08, 0.45)
        seeds.append(_gaussian(n, sigma))

    for _ in range(3):
        p = rng.uniform(0.0, 4.0)
        seeds.append(_power(n, p))

    for _ in range(4):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        seeds.append(_beta_shape(n, a, b))

    for _ in range(4):
        beta = rng.uniform(2.0, 12.0)
        seeds.append(_kaiser_window(n, beta))

    for _ in range(3):
        sigma = rng.uniform(0.06, 0.20)
        shift = rng.uniform(0.15, 0.35)
        seeds.append(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))

    for nf in (2, 3, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Dirichlet / exponential draws
    for conc in (0.05, 0.2, 0.5, 1.5):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(4):
        seeds.append(_random_exponential(n, rng))

    # previous best (if any)
    if previous is not None and previous.shape[0] == n:
        seeds.append(previous / previous.sum())

    # ensure everything is a proper probability vector
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    # truncate to the requested limit
    return clean[:limit]


# ----------------------------------------------------------------------
# Main entry point required by the judge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. initialise RNG and timing
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. problem size (chosen to give a good trade‑off)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. generate a large random pool of seeds
    # ------------------------------------------------------------------
    print(f"\n--- Random seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr
        except Exception:
            pass

    # we ask for a fairly large pool – the exact size is limited by the time budget
    seed_pool = _collect_random_seeds(n, rng, limit=150, previous=prev_arr)
    print(f"Collected {len(seed_pool)} random seeds.")

    # quick surrogate evaluation (exact for ℓ₁‑normalised vectors)
    evaluated = [(v, _objective_surrogate(v)) for v in seed_pool]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [v for v, _ in evaluated[:top_k]]

    print(f"\nTop {top_k} surrogate values among random seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. heavy L‑BFGS optimisation from the best seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    # a more aggressive β‑schedule
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        10000.0, 20000.0, 50000.0, 100000.0
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS runs.")
            break

        # allocate a slice of remaining time
        remaining = deadline - time.time()
        # give each run roughly the same share (but at least a few thousand iters)
        max_iter = int(min(18000, remaining * 0.07 * 1e5))
        max_iter = max(8000, max_iter)

        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter_per_beta={max_iter}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

        # --------------------------------------------------------------
        # 3a. short exponentiated‑gradient polishing after each L‑BFGS run
        # --------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_exp = int(min(400_000, remaining * 0.07 * 1e5))
            max_exp = max(150_000, max_exp)
            seq_exp = _exp_gradient_opt(global_best_seq,
                                        n,
                                        deadline,
                                        max_iters=max_exp)
            val_exp = _objective_surrogate(seq_exp)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. final polishing – one more long L‑BFGS on the survivor
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS polishing on current best ---")
        # give it as much remaining time as possible
        final_seq = _lbfgs_smooth_optimize(global_best_seq,
                                            n,
                                            deadline,
                                            beta_schedule=beta_schedule,
                                            max_iter_per_beta=25000)
        final_val = _objective_surrogate(final_seq)
        if final_val < global_best_val:
            global_best_seq = final_seq.copy()
            global_best_val = final_val
            print(f"[Final L‑BFGS] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
