"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a, a)) / (sum(a)²)

The algorithm searches for a *sparse* support set S ⊂ {0,…,n‑1}
with |S| = k, assigns equal weight 1/k to each index of S
and tries to make the distribution of pair‑sums as flat as possible.
A fast simulated‑annealing (SA) on the binary mask explores the
combinatorial space; the surrogate objective (exact for ℓ₁‑normalised
vectors) is

    obj(mask) = 2·n·max(conv(mask)) / k²

because a_i = mask_i/k and ∑a_i = 1.

Multiple values of k are tried; each run performs a bounded number of
swap moves (remove one index from the current support and insert a new
one).  After the SA phase a short exponentiated‑gradient polish (mirror
descent on the softmax parametrisation) is applied to fine‑tune the
weights.  The best sequence found before the deadline is returned and
stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective helpers (surrogate for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_mask(mask: np.ndarray, k: int, n: int) -> float:
    """
    Exact objective for a binary mask with |mask| = k and a_i = mask_i/k.
    Returns 2·n·max(conv(mask))/k².
    """
    conv = _conv_fft(mask.astype(np.float64), mask.astype(np.float64))
    max_int = float(np.max(conv))
    return 2.0 * n * max_int / (k * k)


# ----------------------------------------------------------------------
# Simulated‑annealing on a binary mask
# ----------------------------------------------------------------------
def _sa_sparse_mask(k: int,
                    n: int,
                    deadline: float,
                    max_iters: int,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on a mask with exactly k ones.
    Returns the best mask seen and its surrogate objective.
    """
    # initialise a random k‑subset
    mask = np.zeros(n, dtype=np.int8)
    init_idx = rng.choice(n, size=k, replace=False)
    mask[init_idx] = 1
    support = list(init_idx)                # list of indices with mask[i]=1
    support_set = set(support)

    # current objective
    cur_obj = _objective_mask(mask, k, n)
    cur_max = None  # not used directly, kept for possible debugging

    best_mask = mask.copy()
    best_obj = cur_obj

    # temperature schedule parameters
    T_start = 5.0
    T_end = 1e-4

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear cooling
        progress = it / max_iters
        T = T_start * (1.0 - progress) + T_end * progress

        # propose a swap: remove i_out ∈ support, insert i_in ∉ support
        i_out = support[rng.integers(len(support))]
        i_in = rng.integers(0, n)
        while i_in in support_set:
            i_in = rng.integers(0, n)

        # perform swap
        mask[i_out] = 0
        mask[i_in] = 1
        support_set.remove(i_out)
        support_set.add(i_in)

        # replace i_out in the list with i_in (O(k) search is fine for k~1000)
        pos = support.index(i_out)
        support[pos] = i_in

        # evaluate new objective (exact surrogate)
        new_obj = _objective_mask(mask, k, n)

        delta = new_obj - cur_obj
        accept = delta <= 0.0 or rng.random() < math.exp(-delta / T)

        if accept:
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_mask = mask.copy()
        else:
            # revert swap
            mask[i_in] = 0
            mask[i_out] = 1
            support_set.remove(i_in)
            support_set.add(i_out)
            support[pos] = i_out

    return best_mask, best_obj


# ----------------------------------------------------------------------
# Exponentiated‑gradient polish (mirror descent on the softmax)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Short exponentiated‑gradient polish on the exact surrogate.
    Works directly with strictly positive entries via a log‑softmax.
    """
    a = start_seq.copy()
    best_a = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered‑pair count)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()          # explicit normalisation

        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a, best_val

            best_seq = seq_exp
            best_obj = val_exp
            print(f"[ExpGrad] improved surrogate = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_obj:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
