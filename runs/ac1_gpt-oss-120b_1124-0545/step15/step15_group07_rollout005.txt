"""
Propose a step‑function (non‑negative vector) minimizing

        2·n·max(conv(a, a)) / (sum(a)²)

within a 1000 s budget.

Algorithm
---------
1. Create a diverse pool of ℓ₁‑normalised seed vectors (uniform,
   triangular, sine‑window, Gaussian, power‑law, beta‑shapes,
   exponential decay, double‑ended exponential, random Dirichlet
   draws and the previous best if any).  Keep the few best according
   to the cheap surrogate 2·n·max(conv(a,a)) (sum=1 ⇒ true value).

2. Refine the elite seed with a continuation L‑BFGS optimiser that
   smoothes the *max* via a log‑sum‑exp surrogate.  Use a long
   schedule of β‑values (up to 20 000) to obtain a sharp local
   optimum.

3. Perform an exact mirror‑descent (exponentiated‑gradient) run on
   the *true* max‑convolution using a diminishing step‑size
   ηₜ=0.5/√t.  Run as many iterations as the remaining budget
   permits (up to a few million).

4. **Alternating LP refinement** – down‑sample the current shape to
   n_small≈n/2, then repeatedly solve the linear programme

        maximise   Σ g_j
        subject to  (a⋆g)_k ≤ M   ∀k ,   g ≥ 0

   where a is the current shape, M= max(a⋆a) and “⋆’’ denotes linear
   convolution.  After solving, normalise g to unit ℓ₁‑norm,
   convex‑combine a←(1‑t)a+t g (t≈0.2) and repeat a few times.
   The LP is built as a dense Toeplitz matrix (size ≈2 n_small×n_small).

5. Upsample the refined low‑resolution vector back to the original
   length, re‑normalise and run a final short L‑BFGS + exponentiated‑
   gradient fine‑tuning.

6. Return the best vector (as a plain list) and store it in the
   global name ``height_sequence_1`` for reuse.
"""

import time
import numpy as np

import scipy.fft

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector.
    If sum(seq)==1 the surrogate equals the true objective.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _double_exp_decay(n: int, lam: float) -> np.ndarray:
    """Large mass at both ends, decaying towards the centre."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p  (decreasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _linear_increasing_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ i^p  (increasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = i ** p
    a[0] = 1e-30  # avoid exact zero before normalisation
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture of two ℓ₁‑normalised vectors."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Down‑/up‑sampling utilities
# ----------------------------------------------------------------------
def _downsample(seq: np.ndarray) -> np.ndarray:
    """Simple average‑pair down‑sampling (factor ≈2) preserving ℓ₁‑norm."""
    n = len(seq)
    # if n is odd, drop the last element before pairing
    if n % 2 == 1:
        seq = seq[:-1]
    new = (seq[0::2] + seq[1::2]) * 0.5
    if new.sum() == 0.0:
        return _uniform(len(new))
    return new / new.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    if new_seq.sum() == 0.0:
        return _uniform(new_n)
    return new_seq / new_seq.sum()

    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))
    seeds.append(_exp_decay(n, lam=0.001))
    seeds.append(_double_exp_decay(n, lam=0.001))

    # random parametric families
    for _ in range(max_random // 5):
        a = 10 ** rng.uniform(-3, 1)      # alpha ∈ [0.001,10]
        b = 10 ** rng.uniform(-3, 1)      # beta  ∈ [0.001,10]
        seeds.append(_beta_shape(n, a, b))

        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        lam = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_exp_decay(n, lam))

        lam2 = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_double_exp_decay(n, lam2))

        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(n,
                             10 ** rng.uniform(-2, 1),
                             10 ** rng.uniform(-2, 1))
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final sanitisation – guarantee non‑negativity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """
    Optimises a smooth surrogate of max‑convolution using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 3_000.0, 5_000.0,
                         10_000.0, 20_000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stay on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                      # numeric guard
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of smoothed max
            w = exp_beta / sum_exp                     # soft‑max weights over conv entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2, but central n entries needed
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-13,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] time limit reached (beta={beta_val})")
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact sub‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 2_500_000) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size ηₜ = 0.5 / √t
        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 800_000) -> np.ndarray:
    """
    Simple simulated‑annealing style hill‑climber.
    Moves a small amount of mass from the current largest entry to a
    random other entry.
    """
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(0xC0FFEE)
    total_iters = max_iters

    for it in range(total_iters):
        if time.time() > deadline:
            break

        progress = it / total_iters
        temp = (cur_val * 0.05) * (1.0 - progress) + (cur_val * 1e-6) * progress
        eps = 0.10 * (1.0 - progress) + 1e-6 * progress

        # donor = index of the largest component
        i = int(np.argmax(a))
        if a[i] <= 1e-15:
            continue
        # random target, different from i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        epsilon = eps * rng.random()
        epsilon = min(epsilon, a[i])

        a[i] -= epsilon
        a[j] += epsilon

        new_val = _objective_surrogate(a)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                if it % 100_000 == 0:
                    print(f"[HillClimb] it={it:,} best≈{best_val:.6f}")
        else:
            a[j] -= epsilon
            a[i] += epsilon

    return best_seq


# ----------------------------------------------------------------------
# Alternating Linear‑Program refinement (low‑resolution)
# ----------------------------------------------------------------------
def _lp_refine(start_seq: np.ndarray,
               n_small: int,
               deadline: float,
               max_iters: int = 8,
               t: float = 0.2) -> np.ndarray:
    """
    Repeatedly solve the LP

        maximise Σ g_j
        s.t.   (a⋆g)_k ≤ M   ∀k
               g ≥ 0

    where a is the current shape, M = max(a⋆a).  After solving,
    normalise g to unit ℓ₁‑norm and replace a ← (1‑t)a + t g.
    Returns the refined low‑resolution vector.
    """
    a = start_seq.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current bound
        conv_aa = _conv_fft(a, a)
        M = np.max(conv_aa)

        # build Toeplitz matrix A where A[k, j] = a_{k-j} if valid else 0
        rows = 2 * n_small - 1
        A = np.zeros((rows, n_small), dtype=np.float64)
        for j in range(n_small):
            A[j:j + n_small, j] = a   # vectorised slice assignment

        b = np.full(rows, M, dtype=np.float64)
        c = -np.ones(n_small, dtype=np.float64)          # maximise Σ g ⇒ minimise -Σ g
        bounds = [(0.0, None)] * n_small

        # Solve LP (highs is fast and robust)
        try:
            res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                               method='highs', options={'presolve': True})
        except Exception as e:
            print(f"[LP] solver error: {e}")
            break

        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            # no improvement possible
            print("[LP] no increase in ℓ₁‑norm, terminating.")
            break

        # normalise and convex‑combine
        g_norm = g / sum_g
        a = (1.0 - t) * a + t * g_norm

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            print(f"[LP] iter {it}: surrogate improved to {cur_val:.6f}")

    return a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ---------------------------------------------------------------
    # Stage 0 – problem size
    # ---------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # ---------------------------------------------------------------
    # Stage 1 – seed generation & cheap ranking
    # ---------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=2500)
    print(f"Generated {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 6
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best surrogate values (top {top_k} seeds):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ---------------------------------------------------------------
    # Stage 2 – L‑BFGS refinement of the best seed
    # ---------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] Current best surrogate = {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- L‑BFGS refinement ---")
        seq_opt = _lbfgs_smooth_optimize(global_best_seq, n, deadline,
                                         beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                        200.0, 500.0, 1_000.0,
                                                        3_000.0, 5_000.0,
                                                        10_000.0, 20_000.0],
                                         max_iter_per_beta=6000)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ---------------------------------------------------------------
    # Stage 3 – exact exponentiated‑gradient refinement
    # ---------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(2_500_000, remaining * 0.30 * 1e5))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient refinement ---")
            seq_exp = _exp_gradient_optimize(global_best_seq, n, deadline,
                                             max_iters=max_exp_iters)
            val_exp = _objective_surrogate(seq_exp)
            print(f"[ExpGrad] surrogate = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ---------------------------------------------------------------
    # Stage 4 – hill‑climbing (simulated annealing)
    # ---------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc_iters = int(min(800_000, remaining * 0.25 * 1e5))
        if max_hc_iters > 0:
            print("\n--- Hill‑climbing (simulated annealing) ---")
            seq_hc = _hill_climb(global_best_seq, n, deadline,
                                 max_iters=max_hc_iters)
            val_hc = _objective_surrogate(seq_hc)
            print(f"[HillClimb] surrogate = {val_hc:.6f}")
            if val_hc < global_best_val:
                global_best_seq = seq_hc.copy()
                global_best_val = val_hc
                print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ---------------------------------------------------------------
    # Stage 5 – alternating LP refinement on a coarser grid
    # ---------------------------------------------------------------
    if time.time() < deadline:
        # down‑sample to roughly half the size
        n_small = n // 2
        low_res = _downsample(global_best_seq)
        print(f"\n--- LP refinement on low‑resolution (n={n_small}) ---")
        it_deadline = deadline - 5.0  # tiny safety margin
        refined_low = _lp_refine(low_res, n_small, it_deadline,
                                 max_iters=12, t=0.2)
        # upsample back to original size
        up_seq = _upsample(refined_low, n)
        # re‑normalise
        up_seq = up_seq / up_seq.sum()
        global_best_seq = up_seq.copy()
        global_best_val = _objective_surrogate(global_best_seq)
        print(f"[LP‑Upsample] surrogate after upsampling = {global_best_val:.6f}")

    # ---------------------------------------------------------------
    # Stage 6 – final fine‑tuning (short L‑BFGS + a few EG steps)
    # ---------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # a brief L‑BFGS pass with an aggressive β‑schedule
        print("\n--- Final L‑BFGS fine‑tuning ---")
        seq_fine = _lbfgs_smooth_optimize(global_best_seq, n, time.time() + remaining * 0.5,
                                          beta_schedule=[5_000.0, 10_000.0, 20_000.0],
                                          max_iter_per_beta=3000)
        val_fine = _objective_surrogate(seq_fine)
        print(f"[Final L‑BFGS] surrogate = {val_fine:.6f}")
        if val_fine < global_best_val:
            global_best_seq = seq_fine.copy()
            global_best_val = val_fine
            print(f"[Update] New best after final L‑BFGS = {global_best_val:.6f}")

        # a short exponentiated‑gradient polishing
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_final_eg = int(min(600_000, remaining * 0.5 * 1e5))
            if max_final_eg > 0:
                print("\n--- Final exponentiated‑gradient polishing ---")
                seq_polish = _exp_gradient_optimize(global_best_seq, n,
                                                    time.time() + remaining,
                                                    max_iters=max_final_eg)
                val_polish = _objective_surrogate(seq_polish)
                print(f"[Polish EG] surrogate = {val_polish:.6f}")
                if val_polish < global_best_val:
                    global_best_seq = seq_polish.copy()
                    global_best_val = val_polish
                    print(f"[Update] New best after polishing = {global_best_val:.6f}")

    # ---------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ---------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
