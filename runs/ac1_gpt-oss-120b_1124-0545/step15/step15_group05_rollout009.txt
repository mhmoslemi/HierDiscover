"""
Improved search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm follows the same overall scheme as the previous state‑of‑the‑art
but allocates the majority of the time budget to a *deep* local optimisation
of the best seed.  The optimisation consists of:

1.  A long Adam run on a smooth approximation of the max‑convolution.
    The smoothing parameter β grows geometrically up to 1e5 and the number
    of Adam steps per β is increased (≈ 8000) – this gives many‑hundreds of
    thousands of gradient evaluations.

2.  An exact exponentiated‑gradient (mirror‑descent) phase on the true
    max‑convolution with a more aggressive step‑size ηₜ = C/√t
    (C=2.0).  Up to a few million sub‑gradient steps are performed.

3.  A few short jittered‑Adam polish runs to escape possible plateaus.

All vectors are ℓ₁‑normalised by using a soft‑max parametrisation, thus the
search space is the probability simplex.  The code respects the 1000 s
budget (a small safety margin) and returns the best candidate it found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always ℓ₁‑normalised, non‑negative)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector."""
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    from scipy.special import i0  # lazily import inside the function
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    w = i0(beta * np.sqrt(1.0 - t * t)) / i0(beta)
    return w / w.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.16
    w = (1 - a) / 2 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + a / 2 * np.cos(4 * np.pi * i / (n - 1))
    return w / w.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return w / w.sum()


def _tukey_window(n: int, alpha: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1)
    w = np.ones_like(i)
    mask1 = i < alpha / 2
    mask2 = i > 1 - alpha / 2
    w[mask1] = 0.5 * (1 + np.cos(np.pi * (2 * i[mask1] / alpha - 1)))
    w[mask2] = 0.5 * (1 + np.cos(np.pi * (2 * (i[mask2] - 1) / alpha + 1)))
    return w / w.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()

def _collect_seeds(

    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_tukey_window(n, alpha=0.4))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # random draws from parametric families
    for _ in range(max_random // 6):
        a = 10 ** rng.uniform(-3, 1)
        b = 10 ** rng.uniform(-3, 1)
        seeds.append(_beta_shape(n, a, b))

        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        lam = rng.exponential(scale=0.001) + 1e-4
        seeds.append(_exp_decay(n, lam))

        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        beta = rng.uniform(0.0, 12.0)
        seeds.append(_kaiser_window(n, beta))

        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(
            n,

        )
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # sanitise – non‑negative and ℓ₁‑normalised
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)

    v = np.zeros_like(logits)

    best_seq = None
    best_val = np.inf
    t = 0

    for beta in beta_vals:
        for _ in range(steps_per_beta):
            t += 1
            if time.time() > deadline:
                return best_seq if best_seq is not None else _softmax(logits)

            a = _softmax(logits)                     # current probability vector
            conv = _conv_fft(a, a)                    # linear convolution, length 2n‑1
            shift = np.max(conv)                      # for numerical stability
            exp_beta = np.exp(beta * (conv - shift))
            w = exp_beta / exp_beta.sum()             # smooth‑max weights (size 2n‑1)

            # gradient w.r.t. a:
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1 : n - 1 + n]    # slice of length n

            # back‑prop through the soft‑max
            dot = np.dot(grad_a, a)
            grad_logits = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_logits
            v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # periodic tracking of the best‑seen surrogate
            if t % 1000 == 0:
                cur_val = _objective_surrogate(a, n)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
    # return the best observed (or final) vector
    return best_seq if best_seq is not None else _softmax(logits)


# ----------------------------------------------------------------------
# Exact sub‑gradient (exponentiated‑gradient / mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_500_000,
    C: float = 2.0,
) -> np.ndarray:
    """
    Mirror‑descent on the *exact* max‑convolution.
    The step size is ηₜ = C / √t.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = C / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a, n)
        if cur_val + 1e-12 < best_val:   # slight tolerance to avoid noise
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Polishing with jittered Adam runs
# ----------------------------------------------------------------------
def _polish_with_jitter(
    base_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    attempts: int = 4,
) -> np.ndarray:
    """
    Perform a few short Adam runs from jittered logits.
    Keeps any improvement found.
    """
    best_seq = base_seq.copy()
    best_val = _objective_surrogate(best_seq, n)

    for k in range(attempts):
        if time.time() > deadline:
            break
        logits = np.log(best_seq + 1e-30)
        jitter = rng.normal(scale=0.015, size=n)
        logits_j = logits + jitter

        # aggressive schedule for polishing
        beta_vals = np.geomspace(2e3, 5e5, num=9)
        seq_refined = _adam_smooth_optimize(
            start_vec=_softmax(logits_j),
            n=n,
            deadline=deadline,
            beta_vals=beta_vals,
            steps_per_beta=1200,
        )
        cur_val = _objective_surrogate(seq_refined, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = seq_refined
            print(f"[Polish] improvement after jitter #{k+1}: {cur_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # --------------------------------------------------------------
    # Seed generation & cheap surrogate ranking
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=2000)
    print(f"Generated {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective_surrogate(vec, n)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 3
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best surrogate values (top {top_k} seeds):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # --------------------------------------------------------------
    # Deep optimisation of the best seed
    # --------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq, n)
    print(f"\n[Init] Current best surrogate = {best_val:.6f}")

    # allocate most of the remaining time to a single, deep Adam run
    remaining = deadline - time.time()
    # Decide a comfortable number of steps (≈ 250 000)
    total_adam_steps = int(min(250_000, remaining * 0.4 * 1e5))
    steps_per_beta = max(3000, total_adam_steps // 9)  # 9 beta values in the schedule
    beta_vals = np.geomspace(1.0, 1e5, num=9)          # 1 → 10⁵ geometric schedule

    print("\n--- Long Adam optimisation (smooth max) ---")
    deep_adam_seq = _adam_smooth_optimize(
        start_vec=best_seq,
        n=n,
        deadline=deadline,
        beta_vals=beta_vals,
        steps_per_beta=steps_per_beta,
    )
    deep_adam_val = _objective_surrogate(deep_adam_seq, n)
    print(f"[Adam] surrogate = {deep_adam_val:.6f}")

    if deep_adam_val < best_val:
        best_seq = deep_adam_seq
        best_val = deep_adam_val
        print(f"[Update] New best after deep Adam = {best_val:.6f}")

    # --------------------------------------------------------------
    # Exact mirror‑descent refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(2_500_000, remaining * 0.4 * 1e5))
        print("\n--- Exact exponentiated‑gradient refinement (max‑conv) ---")
        exp_seq = _exp_gradient_optimize(
            best_seq,
            n,
            deadline,
            max_iters=max_iters,
            C=2.5,                 # a slightly larger constant than 1.0
        )
        exp_val = _objective_surrogate(exp_seq, n)
        print(f"[ExpGrad] surrogate = {exp_val:.6f}")
        if exp_val < best_val:
            best_seq = exp_seq
            best_val = exp_val
            print(f"[Update] New best after ExpGrad = {best_val:.6f}")

    # --------------------------------------------------------------
    # Polishing with jittered Adam runs
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Polishing with jittered Adam runs ---")
        polished = _polish_with_jitter(
            best_seq, n, deadline, rng, attempts=4
        )
        polished_val = _objective_surrogate(polished, n)
        if polished_val < best_val:
            best_seq = polished
            best_val = polished_val
            print(f"[Polish] improved surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
