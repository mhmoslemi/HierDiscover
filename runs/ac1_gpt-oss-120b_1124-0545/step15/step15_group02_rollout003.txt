    dest = rng.integers(0, len(a)-1)

Improved step‑function search for the “max‑convolution” objective.

The algorithm follows a multi‑stage scheme:

1. **Rich seeding** – uniform, previous result, analytic families
   (beta, exponential, mixtures) and many Dirichlet draws.
2. **L‑BFGS continuation** on a smooth log‑sum‑exp surrogate.

4. **A final L‑BFGS refinement** starting from the best EG result.
5. **Random local search** – tiny stochastic moves on the simplex

   basin of attraction.

All convolutions are performed via a real FFT (`_conv_fft`) for speed.
The sequence is always kept ℓ₁‑normalised (sum = 1); the final

# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Objective helpers (the sequence is assumed to be ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (no normalisation)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate used during the search – assumes sum(seq) == 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_exponential(n: int, c1: float, c2: float, w: float) -> np.ndarray:
    """Mixture w·exp(c1)+(1‑w)·exp(c2). Returned vector ℓ₁‑normalised."""
    a = w * _exp_family(n, c1) + (1.0 - w) * _exp_family(n, c2)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc`."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s

# Smoothed L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """Optimise the log‑sum‑exp surrogate via L‑BFGS‑B."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS‑B.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    # initialise logits (log‑softmax param)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                     # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 3_000_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent using the exact max‑convolution sub‑gradient."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (could be many, we use the active one)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # decreasing step‑size (slightly larger than previous version)
        eta = 0.6 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[ExpGrad] it={it:,d} best={best_val:.6f}")

    return best_seq, best_val

# Random local search (tiny perturbations on the simplex)
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         init_sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climbing on the simplex."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    best = cur.copy()
    best_val = _objective_surrogate(best)

    it = 0
    sigma = init_sigma
    # schedule: exponential decay of sigma
    while time.time() < deadline:
        it += 1
        # propose Gaussian perturbation, keep non‑negative, re‑normalise
        perturb = rng.normal(scale=sigma, size=n)
        cand = cur + perturb
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s <= 0.0:
            continue
        cand /= s

        cand_val = _objective_surrogate(cand)
        if cand_val < best_val:
            best_val = cand_val
            best = cand.copy()
            cur = cand.copy()
        else:
            # occasional acceptance of non‑improving move (simulated annealing flavour)
            T = max(1e-6, 0.1 / math.sqrt(it))
            if rng.random() < math.exp(-(cand_val - best_val) / T):
                cur = cand.copy()

        # slowly shrink sigma
        if it % 200_000 == 0:
            sigma *= 0.7
            print(f"[LocalSearch] it={it:,d} best={best_val:.6f} sigma={sigma:.6f}")

    return best, best_val

def _grid_search_exponential(n: int, deadline: float):
    vals = []
    cs = np.linspace(0.5, 4.0, 31)        # finer sweep than before
    for c in cs:
        if time.time() > deadline:
            break
        vec = _exp_family(n, c)
        vals.append((vec, _objective_surrogate(vec), ('exp', c)))
        rev = _rev_exp_family(n, c)
        vals.append((rev, _objective_surrogate(rev), ('revexp', c)))
    return vals


def _grid_search_beta(n: int, deadline: float):
    vals = []
    alphas = np.geomspace(0.05, 5.0, 25)   # geometric grid – good for small α
    betas = np.geomspace(0.05, 5.0, 25)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                return vals
            vec = _beta_shape(n, a, b)
            vals.append((vec, _objective_surrogate(vec), ('beta', a, b)))
    return vals


def _grid_search_mixexp(n: int, deadline: float):
    """Mixture of two exponentials with random weights."""
    vals = []
    rng = np.random.default_rng(12345)
    for _ in range(60):
        if time.time() > deadline:
            break
        c1 = rng.uniform(0.5, 4.0)
        c2 = rng.uniform(0.5, 4.0)
        w = rng.random()
        vec = _mix_exponential(n, c1, c2, w)
        vals.append((vec, _objective_surrogate(vec), ('mixexp', c1, c2, w)))
    return vals

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

    within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # Problem size (kept the same as the previous best)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Retrieve any previous best (if present)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation (rich family sweep + random draws)
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    candidate_pool = []

    # uniform baseline
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # previous best
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr), ('previous',)))

    # analytic sweeps
    candidate_pool.extend(_grid_search_exponential(n, deadline))
    candidate_pool.extend(_grid_search_beta(n, deadline))
    candidate_pool.extend(_grid_search_mixexp(n, deadline))

    # random Dirichlet draws (broad concentration range)
    concs = [0.005, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per_conc = 40
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    # random exponential draws (no normalisation needed – already normalised)
    for _ in range(80):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        candidate_pool.append((vec, _objective_surrogate(vec), ('randexp',)))

    print(f"Generated {len(candidate_pool)} raw seeds.")

    # ------------------------------------------------------------------
    # 2. Keep the most promising seeds
    # ------------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])
    top_k = 15
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Keeping top {top_k} seeds (best surrogate ≈ {candidate_pool[0][1]:.6f}).")

    # ------------------------------------------------------------------
    # 3. L‑BFGS refinement on the best seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx + 1} ---")
        try:
            seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                             n,
                                             deadline,
                                             max_iter_per_beta=8000)
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] failed on seed #{idx + 1}: {e}")
            continue
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate the bulk of remaining time to EG
        max_exp = int(min(3_000_000, remaining * 0.07 * 1e5))
        print(f"\n--- Exponentiated‑gradient polishing (max_iters={max_exp:,d}) ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                             n,
                                             deadline,
                                             max_iters=max_exp)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final L‑BFGS refinement (high β, many iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # give a short final L‑BFGS run with an even larger β‑schedule
        print("\n--- Final L‑BFGS refinement ---")
        try:
            seq_final = _lbfgs_smooth_optimize(global_best_seq,
                                               n,
                                               deadline,
                                               beta_schedule=[1e5, 5e5, 1e6, 2e6, 5e6],
                                               max_iter_per_beta=12_000)
            val_final = _objective_surrogate(seq_final)
            if val_final < global_best_val:
                global_best_seq = seq_final.copy()
                global_best_val = val_final
                print(f"[Final L‑BFGS] improved to {global_best_val:.6f}")
        except Exception as e:  # pragma: no cover
            print(f"[Final L‑BFGS] failed: {e}")

    # ------------------------------------------------------------------
    # 6. Random local search for the last few seconds
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 0.5:   # need at least a fraction of a second
            print("\n--- Random local search (final polishing) ---")
            seq_local, val_local = _local_random_search(global_best_seq,
                                                        n,
                                                        deadline,
                                                        init_sigma=0.01)
            if val_local < global_best_val:
                global_best_seq = seq_local.copy()
                global_best_val = val_local
                print(f"[LocalSearch] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for potential reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
