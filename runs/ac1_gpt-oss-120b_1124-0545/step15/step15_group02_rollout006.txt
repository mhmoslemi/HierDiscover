def exp_gradient_adam(a, n, deadline, max_iters):
    # a is normalized (sum=1)
    # using Adam: m_t, v_t arrays same shape.
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    t = 0
    while True:
        if time.time() > deadline:
            break
        if t >= max_iters:
            break
        t+=1
        # compute a via softmax of log_a (ensuring positivity)
        a = _softmax(log_a)
        # compute convolution
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # compute subgradient w.r.t a
        grad_a = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx_i
        grad_a[idx_i] = 2.0 * a[rev_idx]
        # convert to gradient w.r.t log_a via softmax chain rule:
        # grad_log = a * (grad_a - np.dot(a, grad_a))
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)
        # Adam update
        m = beta1*m + (1-beta1)*grad_log
        v = beta2*v + (1-beta2)*(grad_log**2)
        m_hat = m/(1 - beta1**t)
        v_hat = v/(1 - beta2**t)
        step = 0.1 / (np.sqrt(v_hat) + eps)  # learning rate maybe 0.1
        log_a = log_a - step * m_hat
        # evaluate surrogate
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def _adam_optimize(start_seq, n, deadline, max_iters=500000):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    t = 0
    while True:
        if time.time() > deadline:
            break
        if t >= max_iters:
            break
        t += 1
        # compute a via softmax
        a = _softmax(log_a)
        # conv
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # gradient w.r.t a
        grad_a = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        grad_a[idx_i] = 2.0 * a[rev_idx]
        # grad w.r.t log_a
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.05 / (np.sqrt(v_hat) + eps)  # base_lr 0.05
        log_a = log_a - step * m_hat
        # evaluate
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

"""
Propose a step‑function that (hopefully) improves the best known bound.
Algorithm
---------

* **Rich seed generation** – a very large pool of ℓ₁‑normalised candidates
  (uniform, linear, exponential families, beta‑shapes, Dirichlet draws,
  random power‑laws).  All are evaluated with the exact surrogate
  `2·n·max(conv)` and the top‑K seeds are kept.

* **Adam‑based polishing** – each seed is refined on the exact
  non‑smooth objective by an exponentiated‑gradient (soft‑max)
  parametrisation and Adam’s adaptive moments.  The update uses the
  true sub‑gradient of the active convolution entry, i.e.
  `∂/∂a_i max(conv)=2·a_{k−i}` for the shift `k` where the maximum is
  attained.  The iterate stays on the probability simplex via the
  soft‑max transform.

* **Stochastic hill‑climbing** – starting from the best polished
  sequence, many tiny “mass‑transfer’’ moves (take a random ε‑fraction
  from one index and add it to another) are tried.  Whenever the move
  reduces the surrogate objective it is kept.  This cheap local search
  can break out of shallow plateaux left by the gradient phase.

* **Final check** – the best sequence found is passed to the official
  `evaluate_sequence` function and stored globally as `height_sequence_1`.

The whole routine respects the 1000 s wall‑clock limit, stops early if
time runs out and always returns a non‑negative ℓ₁‑normalised list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective (surrogate) – exact because we keep ℓ₁‑norm = 1
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate  = 2·n·max(conv)  (seq must sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i/n), i = 0,…,n‑1."""
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·(n‑1‑i)/n)."""
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_descending(n: int) -> np.ndarray:
    """a_i ∝ (n‑i)."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _linear_ascending(n: int) -> np.ndarray:
    """a_i ∝ (i+1)."""
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc`."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Log‑uniform random α,β ∈ [1e‑2, 2·10] → beta‑shaped vector."""
    low, high = -4.605170185988091, 3.0  # log(0.01) … log(20)
    alpha = math.exp(rng.uniform(low, high))
    beta = math.exp(rng.uniform(low, high))
    return _beta_shape(n, alpha, beta)


def _random_exp_family(n: int, rng: np.random.Generator) -> np.ndarray:
    """Log‑uniform random c ∈ [1e‑2, 20] → exponential family."""
    low, high = -4.605170185988091, 3.0  # log(0.01) … log(20)
    c = math.exp(rng.uniform(low, high))
    return _exp_family(n, c)


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Refine `start_seq` (already ℓ₁‑normalised) using Adam."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)            # soft‑max parameters
    m = np.zeros_like(a)                # first moment
    v = np.zeros_like(a)                # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    t = 0

    while True:
        if time.time() > deadline or t >= max_iters:
            break
        t += 1

        # project back onto simplex via softmax
        a = _softmax(log_a)

        # convolution & most active shift
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry w.r.t. a
        grad_a = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        grad_a[idx_i] = 2.0 * a[rev_idx]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = 0.05 / (np.sqrt(v_hat) + eps)   # base learning rate
        log_a -= step * m_hat

        # evaluate
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Random pairwise mass transfers that accept only improving moves."""
    rng = np.random.default_rng()
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0

    while True:
        if time.time() > deadline or it >= max_iters:
            break
        it += 1

        i = rng.integers(n)
        if a[i] < 1e-12:          # nothing to move
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction (up to 10 %) from i to j
        delta = a[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur_val = _objective_surrogate(new_a)
        if cur_val < best_val:
            a = new_a
            best_seq = new_a.copy()
            best_val = cur_val
    return best_seq, best_val

# ----------------------------------------------------------------------
