"""
Propose a step‑function (non‑negative heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

under the evaluate_sequence() security checks.

The algorithm consists of four stages:

1. **Massive diverse seed generation** – deterministic analytic families,
   random Dirichlet draws, low‑frequency curves and mixtures.  All vectors are
   ℓ₁‑normalised (sum = 1) so that the objective equals 2·n·max(conv).

2. **Continuation L‑BFGS on a log‑sum‑exp smooth surrogate** – a high‑β schedule
   (up to 5·10⁵) with many iterations per β.  The optimisation works in the
   log‑softmax (log‑probability) space so that the simplex constraint is built‑in.

3. **Mirror‑descent (exponentiated‑gradient) on the exact objective** – a long
   run with a decreasing step‑size.  This directly minimises the true max‑convolution.

4. **Simulated‑annealing style local refinement** in log‑space, moving a few
   masses at a time with a cooling schedule.

All stages respect a 1000 s wall‑clock budget, keep the best sequence found,
store it globally as ``height_sequence_1`` and finally return it as a plain
list of floats.
"""

import time
import numpy as np
import scipy.optimize as _opt
from scipy.special import i0

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector, sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families – all return ℓ₁‑normalised vectors (sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _linear_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = a0 - a1 * np.cos(2 * np.pi * i / (n - 1)) + a2 * np.cos(4 * np.pi * i / (n - 1))
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = i0(t)
    return a / a.sum()


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


# ----------------------------------------------------------------------
# Exact objective and its sub‑gradient (for the exact max‑convolution)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_conv = np.max(conv)
    sum_seq = np.sum(seq)
    if sum_seq < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_seq * sum_seq)


def _subgradient_maxconv(seq: np.ndarray) -> np.ndarray:
    """A sub‑gradient of max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    idx = int(np.argmax(conv))

    g = np.zeros_like(seq)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = idx - idx_i
    g[idx_i] = 2.0 * seq[rev_idx]
    return g

# Surrogate (smooth max) helpers for continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Log‑sum‑exp surrogate of the max‑convolution.
    Returns (obj, grad) where grad is w.r.t. the logits v.
    """
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)               # full linear convolution
    shift = np.max(conv)                 # for numeric stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t a
    w = exp_term / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3·n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]   # slice length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 2500) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    Variable = logits (log‑softmax).  Returns a probability vector (sum = 1).
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 100000.0,
            200000.0, 500000.0,
        ]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method="L-BFGS-B",
                                jac=True,
                                callback=callback,
                                options={"maxiter": max_iter_per_beta,
                                         "disp": False,
                                         "ftol": 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during beta={beta_val:.0f}")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact mirror‑descent (exponentiated‑gradient) on the true objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 3_000_000,
                           eta0: float = 1.2) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Returns (best_a, best_obj).
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 150_000 == 0:
            cur = _objective(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
            # optional progress print
            print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    # final check
    cur = _objective(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny stochastic local‑search (mass transfers) in log‑space
# ----------------------------------------------------------------------
def _simulated_annealing(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple SA / hill‑climbing on the exact objective.
    Perturbs a few log‑probabilities each step and accepts according to temperature.
    """
    rng = np.random.default_rng(42)
    a = start_a.copy()
    best = a.copy()
    best_val = _objective(a)
    T0 = 1.0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T0 * (1.0 - it / max_iters)   # linear cooling

        # propose a perturbation in log‑space
        log_a = np.log(a + 1e-30)
        idxs = rng.choice(n, size=5, replace=False)
        log_a[idxs] += rng.normal(scale=0.05 * max(T, 1e-8), size=5)
        cand = _softmax(log_a)

        cand_val = _objective(cand)

        # accept if better or with Metropolis probability
        if cand_val < best_val or rng.random() < np.exp((best_val - cand_val) / max(T, 1e-12)):
            a = cand
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()

        if it % 20_000 == 0:
            print(f"[SA] iter {it:,}/{max_iters} best={best_val:.6f}")

    return best, best_val

# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a better step‑function within the given time budget.
    Returns the best sequence (list of floats) and stores it globally.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search, n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Seed pool construction (large and diverse)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds: list[np.ndarray] = []

    # Deterministic families (dense parameter sweeps)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_linear_decrease(n))

    for lam in np.logspace(-4, -1, 12):
        seeds.append(_exponential_decay(n, lam))

    for sigma in np.linspace(0.07, 0.45, 15):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 4.0, 25):
        seeds.append(_power_law(n, p))

    for g in np.linspace(0.2, 2.0, 12):
        seeds.append(_sine_window(n, gamma=g))

    for beta in np.linspace(1.0, 15.0, 15):
        seeds.append(_kaiser_window(n, beta))

    for a in np.linspace(0.5, 4.0, 10):
        for b in np.linspace(0.5, 4.0, 10):
            seeds.append(_beta_shape(n, a, b))

    # Randomised families
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(10):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(20):
        seeds.append(_random_exponential(n, rng))

    for nf in (2, 3, 4, 5):
        for _ in range(20):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Double‑Gaussian mixtures (two bumps at random locations)
    for _ in range(30):
        centres = rng.uniform(0, n, size=2)
        sigma = rng.uniform(0.05, 0.25) * n
        a = np.zeros(n, dtype=np.float64)
        i = np.arange(n, dtype=np.float64)
        for c in centres:
            a += np.exp(-((i - c) ** 2) / (2.0 * sigma ** 2))
        a = a / a.sum()
        seeds.append(a)

    # Pairwise mixtures of already‑generated seeds
    base = np.arange(len(seeds))
    for _ in range(200):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Add the previous solution if it matches the size
    if prev_arr is not None and prev_arr.size == n and prev_arr.sum() > 0.0:
        seeds.append(prev_arr / prev_arr.sum())

    # Clean noisy entries and enforce ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            continue
        cleaned.append(v / s)
    seeds = cleaned
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # Quick exact‑objective ranking – keep the very best seeds
    # ------------------------------------------------------------------
    scored = [( _objective(vec), vec) for vec in seeds]
    scored.sort(key=lambda x: x[0])
    top_k = min(12, len(scored))
    top_seeds = [vec for _, vec in scored[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = scored[0][0]
    print(f"Initial best objective from seeds: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Continuation L‑BFGS on the smooth surrogate
    # ------------------------------------------------------------------
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        10000.0, 20000.0, 50000.0, 100000.0,
        200000.0, 500000.0,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS runs.")
            break

        remaining = deadline - time.time()
        per_beta = int(min(8000, max(2500, remaining * 0.015 * 1e5)))

        print(f"\n--- L‑BFGS run #{idx+1}/{len(top_seeds)} (max_iter/beta={per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=per_beta)
        val_opt = _objective(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")

        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print("[Update] New global best after L‑BFGS.")

    # ------------------------------------------------------------------
    # Stage 2 – Exponentiated‑gradient on the exact objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(5_000_000, remaining * 0.25 * 1e5))
        if max_iters > 0:
            print("\n--- Exponentiated‑gradient (exact max) ---")
            seq_exp, val_exp = _exp_gradient_optimize(best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=max_iters,
                                                      eta0=1.2)
            print(f"[ExpGrad] final best objective = {val_exp:.6f}")
            if val_exp < best_val:
                best_seq = seq_exp.copy()
                best_val = val_exp
                print("[Update] New best after EG.")

    # ------------------------------------------------------------------
    # Stage 3 – Simulated‑annealing style local refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(200_000, remaining * 200))
        if iters > 0:
            print("\n--- Simulated Annealing (log‑space) ---")
            seq_sa, val_sa = _simulated_annealing(best_seq,
                                                  n,
                                                  deadline,
                                                  max_iters=iters)
            print(f"[SA] best objective = {val_sa:.6f}")
            if val_sa < best_val:
                best_seq = seq_sa.copy()
                best_val = val_sa
                print("[Update] New best after SA.")

    # ------------------------------------------------------------------
    # Stage 4 – Final polishing L‑BFGS with ultra‑high β
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = int(min(5000, max(1500, remaining * 0.015 * 1e5)))
        print("\n--- Final polishing L‑BFGS (ultra‑high β) ---")
        seq_final = _lbfgs_smooth_optimize(best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=[200000.0, 500000.0, 1_000_000.0],
                                           max_iter_per_beta=per_beta)
        val_final = _objective(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < best_val:
            best_seq = seq_final.copy()
            best_val = val_final
            print("[Update] New best after final polishing.")

    # ------------------------------------------------------------------
    # Final evaluation with the provided evaluator (only for logging)
    # ------------------------------------------------------------------
    try:
        eval_score = evaluate_sequence(best_seq.tolist())
        print("\n=== Finished ===")
        print(f"Best surrogate ≈ {best_val:.6f}, evaluate_sequence = {eval_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally for warm‑starts in future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
