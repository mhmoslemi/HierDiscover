def solve_lp_max_sum(f, M):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # i index for f
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # for each i between i_start,i_end: j = k-i
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n) # minimize -sum(b)
    bounds = [(0, None)]*n
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        # fallback: return f perhaps
        return f

def _lp_refine(start_seq, n, deadline, max_iter=12, t0=0.2, t_decay=0.9):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    t = t0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        M = np.max(conv)
        b = _solve_lp_max_sum(a, M)
        if b is None or b.sum() <= 0:
            break
        g = b / b.sum()
        # symmetrize g to enforce mirror symmetry (optional)
        g = (g + g[::-1]) / 2.0
        g /= g.sum()
        # convex combination
        a = (1 - t) * a + t * g
        a = np.maximum(a, 0.0)
        a /= a.sum()
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[LP refine] iter {it+1} val {best_val:.6f}")
        t *= t_decay
    return best_seq, best_val

"""
Propose a non‑negative step‑function (vector) that minimizes

        2·n·max(conv(a, a)) / (sum(a)²)

within a 1000 s budget.

Algorithm
---------
0. Choose a length `n` (4000) and a RNG seed.
1. Build a diverse pool of ℓ₁‑normalised seed vectors (uniform,
   triangular, sine‑windows, Gaussians, power‑laws, Beta‑shapes,
   exponential decays, Dirichlet draws and the previous best).
2. Keep the few best seeds (cheap surrogate = 2·n·max(conv) because
   sum=1).  The current best seed is used for the heavy work.
3. **Linear‑programming refinement** – solve, for the current vector
   `f`, the LP

        maximise   Σ b_j
        subject to  (f * b)_k ≤ M   (M = max(f*f))
                    b_j ≥ 0

   The solution `b` is normalised to a probability vector `g`.  We
   update `f ← (1‑t)·f + t·g` (with a decaying step size `t`) and
   repeat a few dozen times.  This is exactly the Frank‑Wolfe style
   iteration described in the reference and provably never increases
   the objective.
4. A continuation L‑BFGS optimiser smooths the max‑operator (log‑sum‑exp)
   and performs a high‑precision local optimisation.
5. Exact exponentiated‑gradient (mirror‑descent) on the true max‑convolution.
6. Simulated‑annealing hill‑climber (move a small amount of mass from the
   current largest entry to a random other entry) for final tweaking.
7. Return the best sequence; store it as ``height_sequence_1`` for reuse.
"""

import time
import numpy as np
import scipy.fft
import scipy.optimize as _opt
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector.
    (If sum(seq)=1 then the true objective equals this value.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p  (decreasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


# ----------------------------------------------------------------------
# Seed generation – large random sweep over parametrised families
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None,
                   max_random: int = 2500) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # random parametric families
    for _ in range(max_random // 5):
        a = 10 ** rng.uniform(-3, 1)
        b = 10 ** rng.uniform(-3, 1)
        seeds.append(_beta_shape(n, a, b))

        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        lam = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_exp_decay(n, lam))

        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(n,
                             10 ** rng.uniform(-2, 1),
                             10 ** rng.uniform(-2, 1))
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final sanitisation – guarantee positivity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimises a smooth surrogate of max‑convolution using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 3000.0, 5000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stay on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                      # numeric guard
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                     # soft‑max weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-13,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] time limit reached (beta={beta_val})")
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact sub‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 600_000) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size ηₜ = 0.5 / √t
        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{cur_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 500_000) -> np.ndarray:
    """
    Simple simulated‑annealing style hill‑climber.
    Moves a small amount of mass from the current largest entry to a
    random other entry, accepting improvements always and uphill moves
    with a linearly decaying temperature.
    """
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(0xC0FFEE)
    total_iters = max_iters

    for it in range(total_iters):
        if time.time() > deadline:
            break

        # linear schedules for temperature and step size
        progress = it / total_iters
        temp = (cur_val * 0.05) * (1.0 - progress) + (cur_val * 1e-6) * progress
        eps = 0.10 * (1.0 - progress) + 1e-6 * progress

        # donor = index of the largest component
        i = int(np.argmax(a))
        if a[i] <= 1e-15:
            continue
        # random target, different from i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        epsilon = eps * rng.random()
        epsilon = min(epsilon, a[i])

        a[i] -= epsilon
        a[j] += epsilon

        new_val = _objective_surrogate(a)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                if it % 20000 == 0:
                    print(f"[HillClimb] it={it:,} best≈{best_val:.6f}")
        else:
            a[j] -= epsilon
            a[i] += epsilon

    return best_seq


# ----------------------------------------------------------------------
# Linear‑programming refinement (Frank‑Wolfe style)
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve
        maximise   Σ b_j
        subject to  (f * b)_k ≤ M   for all k = 0..2n‑2
                    b_j ≥ 0
    Returns the optimal vector b (may be any feasible vector if the LP fails).
    """
    n = len(f)
    # Build the convolution matrix A where A @ b = f * b   (full linear conv)
    offsets = np.arange(n, dtype=int)                 # 0 .. n‑1
    # Each diagonal contains the whole vector f (zero‑padded beyond its length)
    A = _sp.diags([f] * n, offsets, shape=(2 * n - 1, n), format='csr')
    c = -np.ones(n)                                   # maximise Σ b_j  →  minimise –Σ b_j
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = _opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                           method='highs', options={'presolve': True})
        if res.success:
            return res.x
    except Exception as e:
        print("[LP] linprog failed:", e)
    # fallback – return the current vector (feasible)
    return f.copy()


def _lp_refine(start_seq: np.ndarray,
               n: int,
               deadline: float,
               max_iter: int = 20,
               t0: float = 0.20,
               t_decay: float = 0.93) -> tuple[np.ndarray, float]:
    """
    Run a few Frank‑Wolfe / LP iterations starting from ``start_seq``.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    t = t0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        M = np.max(conv)

        b = _solve_lp_max_sum(a, M)
        if b.sum() <= 0.0:
            break
        g = b / b.sum()
        # enforce symmetry (empirically helpful)
        g = (g + g[::-1]) / 2.0
        g /= g.sum()

        # convex combination
        a = (1.0 - t) * a + t * g
        a = np.maximum(a, 0.0)
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LP] iter {it+1:2d}  surrogate={cur_val:.6f}")

        t *= t_decay

    return best_seq, best_val


# ----------------------------------------------------------------------
# Upsampling (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()

                                          max_iter_per_beta=8000)
        val_lb = _objective_surrogate(seq_lbfgs)
        print(f"[L‑BFGS] surrogate = {val_lb:.6f}")
        if val_lb < best_val:
            best_seq, best_val = seq_lbfgs, val_lb
            print(f"[Update] after L‑BFGS: surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Exact exponentiated‑gradient (mirror‑descent)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(400_000, remaining * 0.35 * 1e5))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient refinement ---")
            seq_exp = _exp_gradient_optimize(best_seq, n, deadline,
                                            max_iters=max_exp_iters)
            val_exp = _objective_surrogate(seq_exp)
            print(f"[ExpGrad] surrogate = {val_exp:.6f}")
            if val_exp < best_val:
                best_seq, best_val = seq_exp, val_exp
                print(f"[Update] after ExpGrad: surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Hill‑climbing / simulated‑annealing phase
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc_iters = int(min(700_000, remaining * 0.30 * 1e5))
        if max_hc_iters > 0:
            print("\n--- Hill‑climbing (simulated annealing) ---")
            seq_hc = _hill_climb(best_seq, n, deadline,
                                 max_iters=max_hc_iters)
            val_hc = _objective_surrogate(seq_hc)
            print(f"[HillClimb] surrogate = {val_hc:.6f}")
            if val_hc < best_val:
                best_seq, best_val = seq_hc, val_hc
                print(f"[Update] after HillClimb: surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Optional up‑sampling (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 30.0:
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} ---")
            up_seq = _upsample(best_seq, new_n)

            up_opt = _lbfgs_smooth_optimize(up_seq, new_n, deadline,
                                            beta_schedule=[1, 5, 20, 80, 200,
                                                           500, 1000, 3000, 5000],
                                            max_iter_per_beta=4000)
            up_val = _objective_surrogate(up_opt)
            print(f"[Upsample‑L‑BFGS] surrogate = {up_val:.6f}")

            if up_val < best_val:
                best_seq = up_opt.copy()
                best_val = up_val
                n = new_n
                print(f"[Update] after up‑sampling: surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
