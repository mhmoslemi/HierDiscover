propose_candidate
=================

Search for a non‑negative step‑function (vector) that minimises the
official objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The evaluator is scale‑invariant: we keep the ℓ₁‑norm equal to 1
and minimise 2·n·max(conv(a,a)).  The algorithm works in three
stages inside the allotted 1000 s:

1. **Broad seeding** – create a large, diverse set of ℓ₁‑normalised
   candidates (uniform, triangular, sine‑windows, Gaussian,
   power‑laws, several Beta‑shapes – in particular the exact
   Beta(0.5, 0.5) “arcsine”, exponential decays, random Dirichlet draws,
   low‑frequency random curves, plus the previous best if available).
   Rank them by the cheap surrogate `surrogate(a)=2·n·max(conv(a,a))`.

2. **Quick refinement** – for the best 12 seeds run a fast
   exponentiated‑gradient (mirror‑descent) optimiser for a few‑ten
   thousand iterations.  Keep the best refined vector.

3. **Deep polishing** – on the overall best vector run a long

   simulated‑annealing hill‑climber that moves a small amount of mass

All steps constantly monitor the wall‑clock and stop before the

``height_sequence_1`` for reuse).  The code uses only NumPy (no file or

# ----------------------------------------------------------------------
# Low‑level FFT utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return np.max(_conv_fft(a, a))


def _surrogate(a: np.ndarray, n: int) -> float:
    """Surrogate = 2·n·max(conv(a,a)) for a ℓ₁‑normalised vector."""
    return 2.0 * n * _max_conv(a)


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape, α,β>0 ; α=β=0.5 gives the classic arcsine law."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum() if a.sum() > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed generation (large diverse pool, ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None,
                   max_random: int = 2500) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families (including the exact arcsine)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))
    seeds.append(_beta_shape(n, 0.5, 0.5))      # exact arcsine

    # random parametric families
    for _ in range(max_random // 6):
        a = 10 ** rng.uniform(-3, 1)          # α ∈ [10⁻³,10]
        b = 10 ** rng.uniform(-3, 1)          # β ∈ [10⁻³,10]
        seeds.append(_beta_shape(n, a, b))

        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        lam = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_exp_decay(n, lam))

        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        # a random mixture of uniform and a beta shape
        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(n,
                             10 ** rng.uniform(-2, 1),
                             10 ** rng.uniform(-2, 1))
        mixed = (1.0 - mix_alpha) * _uniform(n) + mix_alpha * a_beta
        seeds.append(mixed / mixed.sum())

    # Dirichlet draws with various concentrations
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # sanitise: non‑negative and ℓ₁‑normalised
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        total = v.sum()
        if total <= 0.0:
            v = _uniform(n)
        else:
            v = v / total
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Exact sub‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        # diminishing step‑size η_t = c / √t
        eta = 0.3 / np.sqrt(it)          # a slightly larger constant than 0.5
        log_a = np.log(a + 1e-30) - eta * g
        a = np.exp(log_a)                # soft‑max without explicit normalisation
        a = a / a.sum()                  # enforce ℓ₁‑norm = 1

        cur_val = _surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Simple SA hill‑climber: move a tiny amount of mass from the current
    largest entry to a random destination.
    """
    a = start_seq.copy()
    cur_val = _surrogate(a, n)
    best_seq = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(0xC0FFEE)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # linearly cooling temperature and step size
        progress = it / max_iters
        temp = cur_val * (0.05 * (1 - progress) + 1e-6 * progress)
        eps = 0.10 * (1 - progress) + 1e-6 * progress

        i = int(np.argmax(a))                     # donor (current peak)
        if a[i] <= 1e-15:
            continue
        j = rng.integers(n - 1)                    # random target ≠ i
        if j >= i:
            j += 1

        epsilon = eps * rng.random()
        epsilon = min(epsilon, a[i])                # keep non‑negative

        a[i] -= epsilon
        a[j] += epsilon

        new_val = _surrogate(a, n)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                if it % 100_000 == 0:
                    print(f"[HillClimb] it={it:,} best≈{best_val:.6f}")
        else:
            # revert
            a[j] -= epsilon
            a[i] += epsilon

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) optimiser – L‑BFGS on a soft‑max parametrisation
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """
    Minimises a smooth approximation of max‑conv using L‑BFGS‑B.
    The vector of logits `v` is transformed to a probability vector via
    the soft‑max, guaranteeing the ℓ₁‑norm = 1.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 3000.0, 8000.0, 15000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta = float(beta)

        def fun(v_local):
            a = np.exp(v_local - np.max(v_local))      # soft‑max (normalised)
            a = a / a.sum()
            conv = _conv_fft(a, a)
            m = np.max(conv)
            # smooth max via log‑sum‑exp
            exp_term = np.exp(beta * (conv - m))
            smooth = m + np.log(np.sum(exp_term)) / beta
            obj = 2.0 * n * smooth

            # gradient of smooth max w.r.t. a
            w = exp_term / np.sum(exp_term)            # soft‑weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # d obj / d a (before soft‑max)

            # back‑prop through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-13,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} (beta={beta})")
            break

    # final soft‑max
    a = np.exp(v - np.max(v))
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    official evaluator.  Returns the best sequence as a plain list of
    floats and stores it in the global name ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # ------------------------------------------------------------------
    # Problem size (keep it moderate – 4000 works well)
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # ------------------------------------------------------------------
    # Stage 0 – seed generation & cheap ranking
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=2500)
    print(f"Generated {len(seeds)} raw seeds.")

    # cheap surrogate ranking (surrogate == true objective because sum=1)
    scored = [(vec, _surrogate(vec, n)) for vec in seeds]
    scored.sort(key=lambda kv: kv[1])
    top_k = 12
    elite = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, v) in enumerate(scored[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – quick exponentiated‑gradient refinements on the elite set
    # ------------------------------------------------------------------
    global_best_seq = elite[0].copy()
    global_best_val = _surrogate(global_best_seq, n)

    # quick refinement parameters
    quick_iters = 80_000          # modest number of iterations per seed
    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            break
        print(f"\n--- Quick Exp‑Grad on elite #{idx + 1} ---")
        refined_seq, refined_val = _exp_gradient_optimize(seed_vec, n,
                                                          deadline,
                                                          max_iters=quick_iters)
        # keep only the best seen during the run
        if refined_val < global_best_val:
            global_best_val = refined_val
            global_best_seq = refined_seq.copy()
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deeper polishing of the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate half of the remaining time to L‑BFGS, half to Exp‑Grad
        lbfgs_time = remaining * 0.45
        expgrad_time = remaining * 0.45
        sa_time = remaining * 0.10

        # -------- L‑BFGS (smooth max) --------
        lbfgs_deadline = time.time() + lbfgs_time
        print("\n--- L‑BFGS smooth‑max refinement ---")
        seq_lbfgs = _lbfgs_smooth_optimize(global_best_seq, n,
                                           lbfgs_deadline,
                                           max_iter_per_beta=4000)
        val_lbfgs = _surrogate(seq_lbfgs, n)
        if val_lbfgs < global_best_val:
            global_best_val = val_lbfgs
            global_best_seq = seq_lbfgs.copy()
            print(f"[L‑BFGS] improved to {global_best_val:.6f}")

        # -------- Exact exponentiated gradient (long) --------
        exp_deadline = time.time() + expgrad_time
        print("\n--- Long Exp‑Grad polishing ---")
        seq_exp, val_exp = _exp_gradient_optimize(global_best_seq, n,
                                                  exp_deadline,
                                                  max_iters=2_500_000)
        if val_exp < global_best_val:
            global_best_val = val_exp
            global_best_seq = seq_exp.copy()
            print(f"[Long Exp‑Grad] improved to {global_best_val:.6f}")

        # -------- Simulated‑annealing hill‑climb --------
        if time.time() < deadline:
            sa_deadline = time.time() + sa_time
            print("\n--- Hill‑climbing (simulated annealing) ---")
            seq_hc, val_hc = _hill_climb(global_best_seq, n,
                                         sa_deadline,
                                         max_iters=600_000)
            if val_hc < global_best_val:
                global_best_val = val_hc
                global_best_seq = seq_hc.copy()
                print(f"[HillClimb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
