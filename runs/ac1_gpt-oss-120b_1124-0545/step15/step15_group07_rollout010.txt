def _exp_gradient_optimize(start_seq, n, deadline, max_iters):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = np.argmax(conv)
        # gradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        eta = 0.5/np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq

def propose_candidate(seed=123, budget_s=1000):
    import time
    import numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    n = 4000
    print(f"Starting search, n={n}")

    # generate seeds
    seeds = _generate_seeds(n, rng)
    print(f"Generated {len(seeds)} seeds")
    # evaluate surrogate and keep top_k
    top_k = 10
    top = []  # list of (val, vec)
    for vec in seeds:
        val = _objective_surrogate(vec)
        if len(top) < top_k:
            top.append((val, vec))
            top.sort(key=lambda x: x[0])
        else:
            if val < top[-1][0]:
                top[-1] = (val, vec)
                top.sort(key=lambda x: x[0])
    # Extract top vectors
    top_seeds = [vec for _, vec in top]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"Initial best surrogate {best_val:.6f}")

    # refinement for each top seed
    for i, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"Refining seed #{i+1}")
        # exponentiated gradient
        seq_opt = _exp_gradient_optimize(seed_vec, n, deadline, max_iters=200_000)
        val_opt = _objective_surrogate(seq_opt)
        print(f"  after exp grad surrogate {val_opt:.6f}")
        if val_opt < best_val:
            best_seq = seq_opt
            best_val = val_opt
            print(f"  new global best {best_val:.6f}")

    # final hill climbing
    if time.time() < deadline:
        iters = int(min(500_000, (deadline - time.time()) * 5000))  # rough estimate for iterations per second
        print(f"Final hill climbing with up to {iters} iterations")
        seq_hc = _hill_climb(best_seq, n, deadline, max_iters=iters)
        val_hc = _objective_surrogate(seq_hc)
        print(f"  hill climb surrogate {val_hc:.6f}")
        if val_hc < best_val:
            best_seq = seq_hc
            best_val = val_hc
            print(f"  new global best after hill climb {best_val:.6f}")

    # final evaluation
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"Finished: surrogate {best_val:.6f}, true value {true_val:.6f}")
    except Exception as e:
        print(f"evaluate_sequence raised {e}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

def _exp_gradient_optimize(start_seq, n, deadline, max_iters=200000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline:

        max_idx = int(np.argmax(conv))
        g = np.zeros_like(a)

   low‑frequency random curves and sparse random masks).  All vectors are
   forced to be strictly positive (tiny epsilon) and ℓ₁‑normalised.

2. **Surrogate ranking + local refinement** – the cheap surrogate
   `2·n·max(conv(a,a))` (which equals the true objective when `sum(a)=1`)
   is evaluated for every seed; the best `top_k` seeds are kept.
   Each of those seeds is refined with a mirror‑descent (exponentiated‑gradient)
   optimiser that works directly on the exact max‑convolution.  The best
   refined vector becomes the current global optimum.

3. **Hill‑climbing (simulated annealing)** – starting from the global optimum
   a fast annealing routine repeatedly moves a small amount of mass from the
   currently largest coefficient to a random other coefficient, accepting
   improvements always and uphill moves with a linearly decaying temperature.

The routine respects the overall time budget (default 1000 s), stores the
best found sequence in the global variable `height_sequence_1` and returns it
as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector.
    (If sum(seq)=1 then the true objective equals this value.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _sanitize_vec(v: np.ndarray) -> np.ndarray:
    """Make all entries strictly positive and ℓ₁‑normalise."""
    eps = 1e-30
    v = np.maximum(v, 0.0) + eps
    return v / v.sum()


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _reversed_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.abs(i - c) / c
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a


def _exp_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a


def _sym_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric double‑exponential (mass at both ends)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    return a


def _sym_power_sum(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} + (n-i)^{-p} (U‑shaped power law)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = (i ** (-p)) + ((n - i + 1) ** (-p))
    return a


def _inverse_product(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))^{-p} – very small in the centre."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = ((i) * (n - i + 1)) ** (-p)
    return a


def _dual_block(n: int, frac: float) -> np.ndarray:
    """
    Mass concentrated in two blocks at the ends.
    `frac` is the total fraction of positions occupied (0<frac≤1).
    """
    block_len = max(1, int(frac * n / 2))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = 0.5 / block_len
    a[-block_len:] = 0.5 / block_len
    return a  # already sums to 1


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return a


# ----------------------------------------------------------------------
# Seed generation – diverse pool of ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    seeds = []

    # deterministic families
    seeds.append(_sanitize_vec(_uniform(n)))
    seeds.append(_sanitize_vec(_triangular(n)))
    seeds.append(_sanitize_vec(_reversed_triangular(n)))
    seeds.append(_sanitize_vec(_sine_window(n, gamma=0.5)))
    seeds.append(_sanitize_vec(_sine_window(n, gamma=1.0)))
    seeds.append(_sanitize_vec(_sine_window(n, gamma=2.0)))
    seeds.append(_sanitize_vec(_gaussian(n, sigma_frac=0.15)))
    seeds.append(_sanitize_vec(_gaussian(n, sigma_frac=0.30)))

    # power‑law family
    for p in [0.0, 0.5, 1.0, 2.0, 3.0]:
        seeds.append(_sanitize_vec(_power(n, p)))

    # Beta‑shapes
    for _ in range(12):
        a = 10 ** rng.uniform(-3, 1)
        b = 10 ** rng.uniform(-3, 1)
        seeds.append(_sanitize_vec(_beta_shape(n, a, b)))

    # exponential decays (single‑sided)
    for lam in np.logspace(-5, -1, 8):
        seeds.append(_sanitize_vec(_exp_decay(n, lam)))

    # symmetric exponential (double‑sided)
    for lam in np.logspace(-5, -1, 8):
        seeds.append(_sanitize_vec(_sym_exponential(n, lam)))

    # symmetric power‑sum (U‑shaped)
    for p in np.linspace(0.1, 5.0, 15):
        seeds.append(_sanitize_vec(_sym_power_sum(n, p)))

    # inverse‑product family
    for p in np.linspace(0.2, 3.0, 10):
        seeds.append(_sanitize_vec(_inverse_product(n, p)))

    # dual‑block (mass at ends)
    for frac in np.linspace(0.02, 0.5, 8):
        seeds.append(_sanitize_vec(_dual_block(n, frac)))

    # random Dirichlet draws (different concentrations)
    for conc in [0.01, 0.05, 0.1, 0.5, 1.0, 2.0]:
        for _ in range(3):
            seeds.append(_sanitize_vec(_random_dirichlet(n, conc, rng)))

    # low‑frequency random curves
    for nf in [2, 3, 4, 5]:
        for _ in range(3):
            seeds.append(_sanitize_vec(_lowfreq_curve(n, nf, rng)))

    # random sparse masks
    for _ in range(200):
        s = rng.integers(10, n // 3)
        idx = rng.choice(n, size=s, replace=False)
        vals = rng.exponential(scale=1.0, size=s)
        v = np.zeros(n, dtype=np.float64)
        v[idx] = vals
        seeds.append(_sanitize_vec(v))

    # previous best (if present)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.shape == (n,) and arr.sum() > 0.0:
            seeds.append(_sanitize_vec(arr))

    return seeds


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 200_000) -> np.ndarray:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns the best sequence seen during the run.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size ηₜ = 0.5 / √t
        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climber
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 500_000) -> np.ndarray:
    """
    Moves a small amount of mass from the current largest entry to a random
    other entry, accepting improvements always and uphill moves with a linearly
    decaying temperature.
    """
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(0xC0FFEE)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        progress = it / max_iters
        temp = (cur_val * 0.05) * (1.0 - progress) + (cur_val * 1e-6) * progress
        eps = 0.10 * (1.0 - progress) + 1e-6 * progress

        i = int(np.argmax(a))          # donor – largest component
        if a[i] <= 1e-15:
            continue
        j = rng.integers(n - 1)        # target – any other index
        if j >= i:
            j += 1

        epsilon = eps * rng.random()
        epsilon = min(epsilon, a[i])

        a[i] -= epsilon
        a[j] += epsilon

        new_val = _objective_surrogate(a)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        else:
            a[j] -= epsilon
            a[i] += epsilon

    return best_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    n = 4000
    print(f"\n=== Starting optimisation (n={n}) ===")

    # ------------------------------------------------------------------
    # Phase 1 – generate a large pool of seeds & keep the elite few
    # ------------------------------------------------------------------
    seeds = _generate_seeds(n, rng)
    print(f"Generated {len(seeds)} seeds (raw)")

    top_k = 10
    top = []   # list of (value, vector)

    for vec in seeds:
        val = _objective_surrogate(vec)
        if len(top) < top_k:
            top.append((val, vec))
            top.sort(key=lambda x: x[0])
        else:
            if val < top[-1][0]:
                top[-1] = (val, vec)
                top.sort(key=lambda x: x[0])

    top_seeds = [vec for (_, vec) in top]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] Best surrogate after seeding = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact exponentiated‑gradient refinement of the elite seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Refining seed #{idx + 1}/{len(top_seeds)} ---")
        refined = _exp_gradient_optimize(seed_vec, n, deadline, max_iters=200_000)
        val_ref = _objective_surrogate(refined)
        print(f"[ExpGrad] surrogate = {val_ref:.6f}")
        if val_ref < best_val:
            best_seq = refined.copy()
            best_val = val_ref
            print(f"[Update] New global best = {best_val:.6f}")

        # early exit if we already beat the target
        if best_val < 1.503:
            print("Target threshold reached – skipping remaining seeds.")
            break

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climbing on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # Roughly estimate how many iterations we can afford (≈5000 it/s)
        max_hc_iters = int(min(500_000, remaining * 5000))
        if max_hc_iters > 0:
            print(f"\n--- Hill‑climbing (up to {max_hc_iters} iters) ---")
            hc_seq = _hill_climb(best_seq, n, deadline, max_iters=max_hc_iters)
            hc_val = _objective_surrogate(hc_seq)
            print(f"[HillClimb] surrogate = {hc_val:.6f}")
            if hc_val < best_val:
                best_seq = hc_seq.copy()
                best_val = hc_val
                print(f"[Update] New global best after hill‑climb = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
