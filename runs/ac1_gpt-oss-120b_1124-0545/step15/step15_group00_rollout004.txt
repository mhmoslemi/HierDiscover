def exp_multi_gradient(start_seq, n, deadline, max_iters, epsilon=1e-9, topk=None):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        conv = _conv_fft(a, a)
        M = conv.max()
        # Determine active indices
        if topk is None:
            # Use epsilon relative threshold
            idxs = np.where(conv >= M - epsilon)[0]
        else:
            # get topk indices
            if topk < len(conv):
                idx_part = np.argpartition(-conv, topk-1)[:topk]
                idxs = idx_part
            else:
                idxs = np.arange(len(conv))
        # Compute gradient a
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            idx_i = np.arange(i_start, i_end+1)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        # If multiple idx, we are summing gradient contributions; maybe average
        # We'll average:
        if len(idxs) > 1:
            g /= len(idxs)   # optional
        # Mirror descent update
        dot = np.dot(a, g)
        grad_log = a * (g - dot)
        eta = 0.4 / math.sqrt(it)
        log_a -= eta * grad_log
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # if improvement, maybe print
            if it % 50000 ==0:
                print(f"[ExpMult] it={it} best={best_val:.6f}")
    return best_seq, best_val

def _exp_gradient_multi(start_seq, n, deadline, max_iters, topk=5):
    a = start_seq.copy()
    # ensure non-zero entries to compute logs
    a = np.maximum(a, 1e-30)
    log_a = np.log(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        # find topk indices
        if topk >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(-conv, topk-1)[:topk]  # unsorted topk
        # optionally sort descending for readability
        # compute gradient
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        # average across active peaks to keep magnitude similar
        g /= len(idxs)
        # compute mirror descent grad for log_a
        dot = np.dot(a, g)
        grad_log = a * (g - dot)
        eta = 0.4 / math.sqrt(it)
        log_a -= eta * grad_log
        a = _softmax(log_a)
        # track best
        cur_val = _objective_surrogate(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            # print occasional improvement
            if it % 200000 == 0:
                print(f"[ExpMulti] it={it} new best={best_val:.6f}")
    return best_seq, best_val

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a, a)) / (sum(a)²)

using a long mirror‑descent optimisation on the exact max‑convolution.
The method starts from the best known vector (or a uniform start) and
performs many iterations of exponentiated‑gradient updates that use the
sub‑gradient of *all* currently maximal convolution entries (not only a
single one).  A very large number of cheap iterations (≈10⁶) is
performed within the 1000 s budget, after which the final vector is
evaluated with the official ``evaluate_sequence`` function.

The optimisation works in the log‑softmax (probability‑simplex) space,
so the ℓ₁‑normalisation is automatic and the objective is invariant to
scaling.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser using *all* maximal peaks
# ----------------------------------------------------------------------
def _exp_multi_gradient(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        topk: int = 5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution.

    Parameters
    ----------
    start_seq : np.ndarray
        ℓ₁‑normalised starting vector (sum = 1, all entries ≥ 0).
    n : int
        Length of the vector.
    deadline : float
        Absolute Unix time after which the loop must stop.
    topk : int
        Number of largest convolution entries used to build the sub‑gradient.

    Returns
    -------
    best_seq, best_val
        The best sequence seen and its surrogate value (2·n·max(conv)).
    """
    a = np.copy(start_seq)
    # keep a tiny floor to avoid log(0)
    a = np.maximum(a, 1e-30)
    log_a = np.log(a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        it += 1
        # --- convolution and (approximate) active set -----------------
        conv = _conv_fft(a, a)           # length 2n‑1
        # pick the top‑k entries (they may include ties)
        if topk >= conv.size:
            idxs = np.arange(conv.size)
        else:
            # argpartition is O(N) and cheap for our sizes
            idxs = np.argpartition(-conv, topk - 1)[:topk]

        # --- build sub‑gradient of the (max) objective ----------------
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            # derivative of conv_k w.r.t a_i is 2·a_{k‑i}
            g[idx_i] += 2.0 * a[rev_idx]

        # average over the selected peaks to keep the step size stable
        if len(idxs) > 1:
            g /= float(len(idxs))

        # --- mirror‑descent update in log‑softmax space ---------------
        dot = np.dot(a, g)
        grad_log = a * (g - dot)                 # ∇_{log} f
        eta = 0.5 / math.sqrt(it)                # diminishing stepsize
        log_a -= eta * grad_log
        a = _softmax(log_a)                      # stays on the simplex

        # occasional tiny random perturbation to escape shallow plateaux
        if it % 500_000 == 0:
            log_a += rng.normal(scale=1e-4, size=n)
            a = _softmax(log_a)

        # --- keep best ------------------------------------------------
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # light progress reporting
            if it % 200_000 == 0:
                print(f"[ExpMulti] it={it:,} best surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # reserve a couple of seconds for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – the previous runs used n=4000 and it is a good trade‑off
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Initialise from the previous best if available, otherwise uniform
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and len(previous) == n:
        seq0 = np.asarray(previous, dtype=np.float64)
        if seq0.sum() > 0.0:
            seq0 = np.maximum(seq0, 0.0)
            seq0 = seq0 / seq0.sum()
        else:
            seq0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # fall back to a simple analytic seed (alternating order of a power law)
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / idx ** 0.7
        a = np.sort(a)[::-1]                     # descending
        a = np.maximum(a, 0.0)
        seq0 = a / a.sum()

    print(f"[Init] starting surrogate = {_objective_surrogate(seq0):.6f}")

    # ------------------------------------------------------------------
    # Long mirror‑descent optimisation (many cheap iterations)
    # ------------------------------------------------------------------
    best_seq, best_surrogate = _exp_multi_gradient(seq0, n, deadline, topk=7)

    print(f"[Result] best surrogate after mirror‑descent = {best_surrogate:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Final] evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
