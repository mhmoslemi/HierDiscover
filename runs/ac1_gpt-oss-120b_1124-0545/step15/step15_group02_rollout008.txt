"""
Optimisation of a non‑negative step‑function for the objective

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm combines:

* a rich set of analytic families (exponential, power‑law, beta, etc.) and
  a dense “mixed” family (uniform + exponential + power‑law);
* a coarse pre‑selection of the most promising seeds;
* L‑BFGS optimisation of a log‑softmax parametrisation using a
  log‑sum‑exp surrogate for the max‑convolution;
* a short mirror‑descent (exponentiated‑gradient) phase;
* an intensive random‑pair‑wise local‑search (simulated‑annealing style)
  that repeatedly moves a tiny amount of mass from one coordinate to another
  and accepts improvements (or occasional uphill moves).

All phases respect the global time‑budget (default 1000 s).  The best
sequence is stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (required for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact / surrogate objective
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for an ℓ₁‑normalised vector (sum(seq)=1)."""
    # seq is already normalised to sum 1
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i/n) , i = 0,…,n‑1."""
    r = math.exp(-c / n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """Reverse exponential – a_i ∝ exp(-c·(n‑1‑i)/n)."""
    r = math.exp(-c / n)
    a = r ** (n - 1 - np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) (p can be negative)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _combined_exp_power(n: int, p: float, c: float) -> np.ndarray:
    """a_i ∝ i^{-p}·exp(-c·i/n) , i=1,…,n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) * np.exp(-c * (i - 1) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mixed_uniform_exp(n: int, lam: float, c: float) -> np.ndarray:
    """Convex combination of uniform and exponential."""
    uniform = _uniform(n)
    expo = _exp_family(n, c)
    a = lam * uniform + (1.0 - lam) * expo
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc`."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Smoothed L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Optimise the smoothed max‑convolution objective using L‑BFGS‑B."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0,
                         100000.0, 500000.0, 1e6]

    # initialise logits (log‑softmax parametrisation)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stays on the simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                        # max value
            # smooth approximation of max using log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp                     # soft‑max over convolution entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # gradient of the smoothed max

            # chain rule through the soft‑max (log‑softmax) layer
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (softmax parametrisation)."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size – slowly decreasing
        eta = 0.6 / math.sqrt(it + 1.0)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Random pair‑wise local search (simulated annealing style)
# ----------------------------------------------------------------------
def _random_local_search(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass between two coordinates; accept improvements
    and occasional uphill moves with a temperature that cools down."""
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    # total wall‑clock budget remaining for this phase
    remaining_time = deadline - time.time()
    # rough estimate of iterations we can afford
    max_iters = min(max_iters, int(remaining_time * 4000))  # ≈4000 it/s on modest hardware

    start_time = time.time()
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # choose two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        if cur[i] == 0.0:
            continue

        # fraction of mass to move (log‑uniform between 0.1% and 5%)
        frac = 10.0 ** rng.uniform(-3.0, -1.3)   # between ~0.001 and 0.05
        delta = cur[i] * frac
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluate surrogate (fast, uses FFT)
        cand_val = _objective_surrogate(cand)

        # accept if better
        if cand_val < cur_val:
            cur = cand
            cur_val = cand_val
            if cand_val < best_val:
                best = cand.copy()
                best_val = cand_val
                print(f"[LocalSearch] it={it:7d} new best={best_val:.6f}")
        else:
            # simulated‑annealing style acceptance
            # temperature linearly decays with remaining wall‑clock time
            time_left = deadline - time.time()
            total = deadline - start_time
            T = max(1e-9, time_left / total) * 0.02   # max temperature ≈0.02
            if rng.random() < math.exp(-(cand_val - cur_val) / T):
                cur = cand
                cur_val = cand_val

    return best, best_val

def _grid_search_exponential(n: int, deadline: float) -> list[tuple[np.ndarray, float, tuple]]:
    """Sweep exponential family over c in [0.0, 5.0]."""
    vals = []
    cs = np.linspace(0.0, 5.0, 31)
    for c in cs:
        if time.time() > deadline:
            break
        vec = _exp_family(n, c)
        vals.append((vec, _objective_surrogate(vec), ('exp', c)))
        rev = _rev_exp_family(n, c)
        vals.append((rev, _objective_surrogate(rev), ('revexp', c)))
    return vals


def _grid_search_power(n: int, deadline: float) -> list[tuple[np.ndarray, float, tuple]]:
    """Power‑law sweep p ∈ [-0.5,5] (41 points)."""
    vals = []
    for p in np.linspace(-0.5, 5.0, 41):
        if time.time() > deadline:
            break
        vec = _power(n, p)
        vals.append((vec, _objective_surrogate(vec), ('power', p)))
    return vals


def _grid_search_beta(n: int, deadline: float) -> list[tuple[np.ndarray, float, tuple]]:
    """Coarse beta‑shape search (α,β ∈ [0.1,5] with 9 points each)."""
    vals = []
    alphas = np.linspace(0.1, 5.0, 9)
    betas = np.linspace(0.1, 5.0, 9)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                return vals
            vec = _beta_shape(n, a, b)
            vals.append((vec, _objective_surrogate(vec), ('beta', a, b)))
    return vals


def _grid_search_combined(n: int, deadline: float) -> list[tuple[np.ndarray, float, tuple]]:
    """Dense search of the combined exponential‑power family."""
    vals = []
    ps = np.linspace(0.0, 5.0, 21)      # 21 values
    cs = np.linspace(0.0, 5.0, 21)      # 21 values
    for p in ps:
        for c in cs:
            if time.time() > deadline:
                return vals
            vec = _combined_exp_power(n, p, c)
            vals.append((vec, _objective_surrogate(vec), ('comb', p, c)))
    return vals


def _grid_search_mixed(n: int, deadline: float) -> list[tuple[np.ndarray, float, tuple]]:
    """Mixed uniform–exponential family."""
    vals = []
    lams = np.linspace(0.0, 1.0, 11)    # 11 values
    cs = np.linspace(0.0, 5.0, 21)      # 21 values
    for lam in lams:
        for c in cs:
            if time.time() > deadline:
                return vals
            vec = _mixed_uniform_exp(n, lam, c)
            vals.append((vec, _objective_surrogate(vec), ('mix', lam, c)))
    return vals

# Main search entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – we keep the length used by the previous solution.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load any previously stored solution.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation (rich mix of analytic families and random draws)
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    candidate_pool = []

    # baseline uniform
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # previous solution if any
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr), ('previous',)))

    # sweep a variety of families
    candidate_pool.extend(_grid_search_exponential(n, deadline))
    candidate_pool.extend(_grid_search_power(n, deadline))
    candidate_pool.extend(_grid_search_beta(n, deadline))
    candidate_pool.extend(_grid_search_combined(n, deadline))
    candidate_pool.extend(_grid_search_mixed(n, deadline))

    # random Dirichlet draws (broad range of concentrations)
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per_conc = 80
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    # ------------------------------------------------------------------
    # 2. Keep the most promising seeds
    # ------------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])
    top_k = 20
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Collected {len(candidate_pool)} seed vectors, keeping top {top_k}.")

    # ------------------------------------------------------------------
    # 3. L‑BFGS continuation on the best seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx+1} ---")
        try:
            seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                             n,
                                             deadline,
                                             max_iter_per_beta=2000)
            val_opt = _objective_surrogate(seq_opt)
            print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[Update] new global best surrogate = {global_best_val:.6f}")
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] failed on seed #{idx+1} with error: {e}")

    # ------------------------------------------------------------------
    # 4. Exponentiated‑gradient polishing (exact max‑convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(800_000, remaining * 3000))  # heuristic for iterations
        print(f"\n--- Exponentiated‑gradient polishing (max_iters={max_exp_iters}) ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                             n,
                                             deadline,
                                             max_iters=max_exp_iters)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Random pair‑wise local search (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_local_iters = int(min(2_000_000, remaining * 4000))
        if max_local_iters > 0:
            print("\n--- Random local search (simulated annealing) ---")
            seq_loc, val_loc = _random_local_search(global_best_seq,
                                                    n,
                                                    deadline,
                                                    rng,
                                                    max_iters=max_local_iters)
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[LocalSearch] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
