"""
Enhanced search for a non‑negative step‑function that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

The algorithm works in three phases:

1️⃣  **Rich seed generation** – deterministic windows (uniform,
    triangular, sine, Gaussian, Kaiser, Black‑Man, Hamming) together
    with a few hand‑crafted beta‑shapes (especially the U‑shaped
    α=β≈0.5 arcsine family).  All seeds are ℓ₁‑normalised.

2️⃣  **Smooth‑max Adam** – each elite seed is optimised with Adam on the
    smooth approximation `log‑sum‑exp(β·conv)`; the β‑schedule grows
    up to 4·10⁴ so that the surrogate converges to the true max.
    More than 40 k Adam steps per seed are performed (limited only by the
    time budget).

3️⃣  **Exact exponentiated‑gradient (mirror‑descent)** – starting from the
    best Adam result we run a long sub‑gradient descent on the exact
    max‑convolution using a step‑size ηₜ = 1.5/√t.  This aggressively
    reduces the max‑convolution while keeping the vector on the simplex.
    A final “jitter‑Adam” polishing stage tries a few short Adam runs
    from perturbed logits to escape shallow local minima.

The whole search respects the 1000 s wall‑clock limit; the best vector
found is stored as ``height_sequence_1`` (a plain list of floats) and is
returned.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from scipy.special import i0  # Kaiser window

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – yields a non‑negative vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """2·n·max(conv(seq, seq)) – the exact objective when sum(seq)=1."""
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape on the discrete grid 1…n (α,β>0)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window (β≥0)."""
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    w = i0(beta * np.sqrt(1.0 - t * t)) / i0(beta)
    return w / w.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.16
    w = (1 - a) / 2 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + a / 2 * np.cos(
        4 * np.pi * i / (n - 1)
    )
    return w / w.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return w / w.sum()


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    max_random: int = 2000,
) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_kaiser_window(n, beta=4.0))
    seeds.append(_kaiser_window(n, beta=8.0))
    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))

    # hand‑picked beta shapes (U‑shaped / arcsine)
    for a in (0.5, 0.3, 0.2):
        for b in (0.5, 0.3, 0.2):
            seeds.append(_beta_shape(n, a, b))

    # random Beta / power‑law families
    for _ in range(max_random // 5):
        a = 10 ** rng.uniform(-3, 1)
        b = 10 ** rng.uniform(-3, 1)
        seeds.append(_beta_shape(n, a, b))

        p = rng.uniform(0.0, 5.0)
        i = np.arange(1, n + 1, dtype=np.float64)
        a_pow = (i ** (-p))
        seeds.append(a_pow / a_pow.sum())

        lam = rng.exponential(scale=0.001) + 1e-4
        i = np.arange(n, dtype=np.float64)
        exp = np.exp(-lam * i)
        seeds.append(exp / exp.sum())

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # sanitise – guarantee non‑negativity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Continuation Adam optimiser (smooth‑max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    steps_per_beta: int = 12000,
) -> np.ndarray:
    """
    Adam optimiser on the smoothed max‑convolution.
    Returns a simplex‑constrained vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1000.0, 3000.0, 8000.0, 15000.0, 30000.0, 40000.0]

    eps = 1e-30
    logits = np.log(start_vec + eps)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.10                 # a tad larger than the original
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    t = 0

    best_seq = None
    best_val = np.inf

    for beta in beta_schedule:
        beta_val = float(beta)
        # dynamic step count: respect remaining time
        max_iters = steps_per_beta
        for _ in range(max_iters):
            t += 1
            if time.time() > deadline:
                return _softmax(logits)

            a = _softmax(logits)               # current simplex point
            conv = _conv_fft(a, a)              # length 2n‑1
            shift = np.max(conv)                # numeric shift for stability
            exp_beta = np.exp(beta_val * (conv - shift))
            w = exp_beta / exp_beta.sum()       # smooth‑max weights, shape (2n‑1,)

            # gradient wrt a: 2 * (rev_a * w) convolution
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_logits = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_logits
            v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # occasional best‑track (every few hundred iters)
            if t % 500 == 0:
                cur_val = _objective_surrogate(a, n)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
    return best_seq if best_seq is not None else _softmax(logits)

# Exact exponentiated‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 3_000_000,
) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (2·a_{k*-i})
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size (more aggressive than the original 1.0)
        eta = 1.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        # track best surrogate
        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 250_000 == 0:
                print(f"[ExpGrad] it={it:,}  best≈{cur_val:.6f}")

    return best_seq

# Small random‑restart polishing via Adam
# ----------------------------------------------------------------------
def _polish_with_jitter(
    base_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    attempts: int = 4,
) -> np.ndarray:
    """
    Perform a few short Adam runs starting from jittered logits.
    Keeps any improvement found.
    """
    best_seq = base_seq.copy()
    best_val = _objective_surrogate(best_seq, n)

    for k in range(attempts):
        if time.time() > deadline:
            break
        logits = np.log(best_seq + 1e-30)
        jitter = rng.normal(scale=0.025, size=n)  # modest noise
        logits_j = logits + jitter

        refined = _adam_smooth_optimize(
            start_vec=_softmax(logits_j),
            n=n,
            deadline=deadline,
            beta_schedule=[20000.0, 30000.0, 40000.0],
            steps_per_beta=1200,
        )
        cur_val = _objective_surrogate(refined, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = refined
            print(f"[Polish] improvement after jitter #{k+1}: {cur_val:.6f}")

    return best_seq
