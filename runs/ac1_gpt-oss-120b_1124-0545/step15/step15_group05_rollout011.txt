def _proj_simplex(v):
    """Project onto simplex {x >=0, sum=1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1  # index of last positive
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Improved search for a non‑negative step‑function that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

The algorithm is a three‑stage optimiser:

1. **Seed sweep** – generate a diverse pool of ℓ₁‑normalised vectors from many
   analytic families (uniform, triangular, Kaiser, Beta, Dirichlet, low‑freq
   random curves, …) and keep the best few according to the cheap surrogate
   S(a)=2·n·max(conv(a,a)).

2. **Highly‑smoothed Adam** – optimise the smooth‑max surrogate
   `log‑sum‑exp(β·conv)` with a very long Adam run.
   The β‑schedule climbs up to 5·10⁵, so the optimisation finally targets the
   true max with sub‑gradient accuracy.  The optimisation works on the log‑soft‑max
   (exponential‑family) parametrisation, guaranteeing non‑negativity and
   ℓ₁‑normalisation at every step.

3. **Exact max‑optimisation with Adam on the sub‑gradient** – after the smooth
   phase we switch to the genuine max‑convolution objective.  The gradient of the
   active convolution entry is computed analytically, back‑propagated through the
   soft‑max, and Adam is applied to the logits.  This converges far faster than
   the plain exponentiated‑gradient used previously.

Finally a few jitter‑restart polishing runs are performed and the best sequence
found is returned (and stored globally as ``height_sequence_1`` for the next
invocation).  The whole search respects the 1000 s wall‑time and prints

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for n≈4000)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always ℓ₁‑normalised, non‑negative)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """Cheap surrogate = 2·n·max(conv(seq,seq)) for ℓ₁‑normalised vectors."""
    return 2.0 * n * np.max(_conv_fft(seq, seq))

# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    from scipy.special import i0
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    w = i0(beta * np.sqrt(1.0 - t * t)) / i0(beta)
    return w / w.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.16
    w = (1 - a) / 2 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + a / 2 * np.cos(4 * np.pi * i / (n - 1))
    return w / w.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return w / w.sum()


def _tukey_window(n: int, alpha: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1)
    w = np.ones_like(i)
    mask1 = i < alpha / 2
    mask2 = i > 1 - alpha / 2
    w[mask1] = 0.5 * (1 + np.cos(np.pi * (2 * i[mask1] / alpha - 1)))
    w[mask2] = 0.5 * (1 + np.cos(np.pi * (2 * (i[mask2] - 1) / alpha + 1)))
    return w / w.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    max_random: int = 2500,
) -> list[np.ndarray]:
    """Create a wide variety of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.4))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.10))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_kaiser_window(n, beta=3.0))
    seeds.append(_kaiser_window(n, beta=9.0))
    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_tukey_window(n, alpha=0.45))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # random parametric families
    for _ in range(max_random // 7):
        # Random Beta shapes – very spiky at the edges are possible
        a = 10 ** rng.uniform(-4, 0)          # 1e‑4 … 1
        b = 10 ** rng.uniform(-4, 0)
        seeds.append(_beta_shape(n, a, b))

        # Random power‑law
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        # Random exponential decay
        lam = rng.exponential(scale=0.001) + 1e-5
        seeds.append(_exp_decay(n, lam))

        # Random linear‑power (decreasing)
        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        # Random Kaiser
        beta = rng.uniform(0.0, 12.0)
        seeds.append(_kaiser_window(n, beta))

        # Random Dirichlet (different concentrations)
        conc = 10 ** rng.uniform(-2, 0)   # 0.01 … 1
        seeds.append(_random_dirichlet(n, conc, rng))

        # Low‑frequency random curves
        nf = rng.integers(2, 6)
        seeds.append(_lowfreq_curve(n, nf, rng))

        # Mix a Beta shape with the uniform distribution
        mix = rng.uniform(0.0, 1.0)
        beta_mix = _beta_shape(
            n,
            10 ** rng.uniform(-3, 0),
            10 ** rng.uniform(-3, 0),
        )
        seeds.append(_mix_shapes(_uniform(n), beta_mix, mix))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # sanitise
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float],
    steps_per_beta: int,
) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns a ℓ₁‑normalised vector (soft‑max of the logits).
    """
    eps = 1e-30
    logits = np.log(start_vec + eps)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.08          # a bit larger – we have many iterations
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    t = 0

    best_seq = None
    best_val = np.inf

    for beta in beta_schedule:
        beta_val = float(beta)
        for _ in range(steps_per_beta):
            t += 1
            if time.time() > deadline:
                return _softmax(logits)

            a = _softmax(logits)  # current candidate (ℓ₁‑normed)
            conv = _conv_fft(a, a)               # full linear convolution
            shift = np.max(conv)                  # numerical stability
            # smooth‑max weights via log‑sum‑exp trick
            exp_beta = np.exp(beta_val * (conv - shift))
            w = exp_beta / exp_beta.sum()         # shape (2n‑1,)

            # gradient w.r.t. a : 2 * rev(a) * w  (convolution)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            # back‑prop through softmax (logits → a)
            dot = np.dot(grad_a, a)
            grad_logits = a * (grad_a - dot)

            # Adam update on logits
            m = beta1 * m + (1.0 - beta1) * grad_logits
            v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # keep best (every ~1000 iters to limit overhead)
            if t % 1000 == 0:
                cur_val = _objective_surrogate(_softmax(logits), n)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = _softmax(logits).copy()
        # (optional) shrink learning rate a little after each β
        lr *= 0.97
    return best_seq if best_seq is not None else _softmax(logits)


# ----------------------------------------------------------------------
# Adam optimiser on the *exact* max‑convolution (sub‑gradient version)
# ----------------------------------------------------------------------
def _adam_exact_max_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
) -> np.ndarray:
    """
    Adam on the exact max‑convolution (sub‑gradient with soft‑max parametrisation).
    """
    eps = 1e-30
    # initialise logits from the start vector
    logits = np.log(start_seq + eps)

    # Adam hyper‑parameters (similar to above)
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.04
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    t = 0

    a = _softmax(logits)          # initial feasible point
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exact convolution and active index
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry:
        #   g_i = 2 * a_{max_idx - i}  (if the index lies in range)
        idx = np.arange(n)
        j = max_idx - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # back‑prop through softmax
        dot = np.dot(g, a)
        grad_logits = a * (g - dot)

        # Adam update on logits
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # re‑project to simplex via soft‑max
        a = _softmax(logits)

        # occasional best‑track
        if it % 2000 == 0:
            cur_val = _objective_surrogate(a, n)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[AdamExact] it={it:,} new best ≈{cur_val:.6f}")

    # final check
    cur_val = _objective_surrogate(a, n)
    if cur_val < best_val:
        best_seq = a.copy()
        best_val = cur_val
    return best_seq


# ----------------------------------------------------------------------
# Small jitter‑restart polishing (short Adam runs)
# ----------------------------------------------------------------------
def _polish_with_jitter(
    base_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    attempts: int = 3,
) -> np.ndarray:
    """A few short Adam runs from jittered logits."""
    best_seq = base_seq.copy()
    best_val = _objective_surrogate(best_seq, n)

    for k in range(attempts):
        if time.time() > deadline:
            break
        logits = np.log(best_seq + 1e-30) + rng.normal(scale=0.015, size=n)

        refined = _adam_smooth_optimize(
            start_vec=_softmax(logits),
            n=n,
            deadline=deadline,
            beta_schedule=[1e4, 5e4, 2e5, 5e5],
            steps_per_beta=1200,
        )
        cur_val = _objective_surrogate(refined, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = refined
            print(f"[Polish] improvement after jitter #{k+1}: {cur_val:.6f}")
    return best_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for the best non‑negative step‑function under the given time budget.
    Returns the sequence as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size (n=4000 is a good balance)
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting improved search (n={n}) ===")

    # --------------------------------------------------------------
    # 1️⃣  Seed generation & cheap surrogate ranking
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=3000)
    print(f"Generated {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective_surrogate(vec, n)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best surrogate values among seeds (top {top_k}):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Heavy‑smooth Adam from the best seed
    # --------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq, n)
    print(f"\n[Init] Surrogate = {best_val:.6f}")

    # Very aggressive β‑schedule, long inner loops
    beta_sched = [1.0, 10.0, 100.0, 1_000.0, 10_000.0, 50_000.0, 200_000.0, 500_000.0]
    # allocate most of the remaining time to this stage
    remaining = deadline - time.time()
    steps_per_beta = max(4000, int(min(12_000, remaining * 0.004 * 1e5)))
    print(f"Running smooth‑Adam: steps_per_beta={steps_per_beta}")

    seq_smooth = _adam_smooth_optimize(
        start_vec=best_seq,
        n=n,
        deadline=deadline,
        beta_schedule=beta_sched,
        steps_per_beta=steps_per_beta,
    )
    val_smooth = _objective_surrogate(seq_smooth, n)
    print(f"[SmoothAdam] surrogate = {val_smooth:.6f}")

    if val_smooth < best_val:
        best_seq = seq_smooth
        best_val = val_smooth
        print(f"[Update] New best after smooth‑Adam = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Exact max‑optimisation with Adam (sub‑gradient)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(2_500_000, remaining * 0.25 * 1e5))
        if max_iters > 0:
            print("\n--- Exact max optimisation with Adam ---")
            seq_exact = _adam_exact_max_optimize(
                start_seq=best_seq,
                n=n,
                deadline=deadline,
                max_iters=max_iters,
            )
            val_exact = _objective_surrogate(seq_exact, n)
            if val_exact < best_val:
                best_seq = seq_exact
                best_val = val_exact
                print(f"[ExactAdam] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Polishing via jittered short runs
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Polishing with jittered Adam runs ---")
        polished = _polish_with_jitter(best_seq, n, deadline, rng, attempts=4)
        val_polish = _objective_surrogate(polished, n)
        if val_polish < best_val:
            best_seq = polished
            best_val = val_polish
            print(f"[Polish] final surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
