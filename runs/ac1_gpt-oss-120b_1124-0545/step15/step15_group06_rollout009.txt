"""
Search for a non‑negative step function minimizing 2·n·max(conv(a,a))/(sum(a)²).

The algorithm builds a large pool of ℓ₁‑normalised “seed” shapes, including
very “U‑shaped” beta distributions, edge‑heavy power‑law shapes, low‑conc
Dirichlet draws and many random mixtures.  The best seeds are refined with
a continuation L‑BFGS optimiser on a smooth log‑sum‑exp surrogate of the
objective, followed by an exponentiated‑gradient (mirror‑descent) phase on
the exact max‑convolution objective.  Finally a stochastic perturb‑and‑polish
stage plus a high‑β L‑BFGS polishing run are performed.  All stages respect
the 1000 s time budget and keep the best sequence found.
"""
import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on {1,…,n} (U‑shape when α,β<1)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _edge_power_shape(n: int, p: float) -> np.ndarray:
    """Edge‑heavy shape a_i ∝ (i+1)^(-p) + (n‑i)^(-p)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) + (n - i) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _edge_trough_shape(n: int, q: float) -> np.ndarray:
    """Shape low in the centre, high at edges: 1‑|i‑c|^q."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - (np.abs(i - c) / c) ** q
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_sparse(n: int, rng: np.random.Generator,
                  k_min: int = 30, k_max: int = 150) -> np.ndarray:
    """Sparse support: choose K random indices, give them random weights."""
    K = int(rng.integers(k_min, k_max + 1))
    idx = rng.choice(n, size=K, replace=False)
    a = np.zeros(n, dtype=np.float64)
    weights = rng.exponential(scale=1.0, size=K)
    a[idx] = weights
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α)+b·α then ℓ₁‑renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Surrogate objective (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Return (value, gradient) for the smooth max‑convolution surrogate."""
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)              # full auto‑convolution
    shift = np.max(conv)                # numerical stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # weights for each lag
    w = exp_term / sum_exp
    # convolution of reversed a with the weights gives ∂obj/∂a
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    L‑BFGS‑B optimisation of the smooth surrogate.
    Variable = logits of a softmax (unconstrained real vector).
    """
    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0,
                         5000.0, 10000.0, 20000.0, 50000.0, 100000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)

# Exact objective utilities
# ----------------------------------------------------------------------
def _objective_raw(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) / (sum²)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (using the active lag).
    Returns g = ∂ max(conv) / ∂ a .
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


def _exp_gradient_optimize(start_a: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 500_000,
                          eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Uses exponentiated‑gradient updates on logits.
    Returns (best_vector, its objective).
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_raw(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_maxconv(a)          # sub‑gradient of max
        eta = eta0 / math.sqrt(it)           # diminishing stepsize

        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 100_000 == 0:
            cur = _objective_raw(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_raw(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val

def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Build a diverse pool of ℓ₁‑normalised seed vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic analytic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    # Edge‑heavy power‑law shapes
    for p in (0.2, 0.5, 1.0, 2.0):
        seeds.append(_edge_power_shape(n, p))
    # Edge‑trough (deep centre dip)
    for q in (2.0, 3.0, 4.0, 5.0):
        seeds.append(_edge_trough_shape(n, q))

    # Gausian / window families
    for sigma in np.linspace(0.08, 0.45, 9):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta families – explore very small α,β as well as moderate values
    betas = [0.01, 0.02, 0.05, 0.08, 0.1, 0.2, 0.5, 0.8, 1.0, 1.5]
    for a in betas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Low‑conc Dirichlet draws (heavier tails)
    for conc in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Exponential draws
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sparse random supports with a broader size range
    for _ in range(30):
        k_min = rng.integers(20, 50)
        k_max = rng.integers(100, 200)
        seeds.append(_random_sparse(n, rng, k_min=int(k_min), k_max=int(k_max)))

    # Mixtures of already generated seeds – create a large pool of hybrids
    base_idx = np.arange(len(seeds))
    for _ in range(400):
        i, j = rng.choice(base_idx, size=2, replace=False)
        alpha = rng.random()
        mix = _mix_shapes(seeds[i], seeds[j], alpha)
        seeds.append(mix)

    # Warm‑start from previously discovered best
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Clean‑up – ensure non‑negativity & ℓ₁‑normalisation -------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

        print(f"[{n}] evaluate_sequence raised an exception: {exc}")

"""
Search for a non‑negative step function that minimises
    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm builds a large pool of ℓ₁‑normalised seed shapes
(including very “U‑shaped” beta distributions, edge‑heavy
power‑law shapes, low‑conc Dirichlet draws and many random
mixtures).  The most promising seeds are refined with a
continuation L‑BFGS optimiser on a smooth log‑sum‑exp surrogate
of the objective, followed by an exponentiated‑gradient (mirror‑descent)
phase on the exact max‑convolution objective.  Finally a stochastic
perturb‑and‑polish stage plus a high‑β L‑BFGS polishing run are
performed.  All stages respect the 1000 s budget and keep the best
sequence found in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on {1,…,n} (U‑shape when α,β<1)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _edge_power_shape(n: int, p: float) -> np.ndarray:
    """Edge‑heavy shape a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) + (n - i) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _edge_trough_shape(n: int, q: float) -> np.ndarray:
    """Low centre, high edges: 1‑|i‑c|^q (clamped)."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - (np.abs(i - c) / c) ** q
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_sparse(n: int,
                   rng: np.random.Generator,
                   k_min: int = 30,
                   k_max: int = 150) -> np.ndarray:
    """Sparse support: pick K random indices and assign exponential weights."""
    K = int(rng.integers(k_min, k_max + 1))
    idx = rng.choice(n, size=K, replace=False)
    a = np.zeros(n, dtype=np.float64)
    weights = rng.exponential(scale=1.0, size=K)
    a[idx] = weights
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α)+b·α then ℓ₁‑renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Surrogate objective (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Value and gradient of the smooth surrogate."""
    a = _softmax(v)                     # lie on the probability simplex
    conv = _conv_fft(a, a)              # full auto‑convolution
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0,
                         5000.0, 10000.0, 20000.0,
                         50000.0, 100000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break
    return _softmax(v)


# ----------------------------------------------------------------------
# Exact objective utilities
# ----------------------------------------------------------------------
def _objective_raw(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv)/(sum²)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a using the active lag.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


def _exp_gradient_optimize(start_a: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 500_000,
                          eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Returns (best_vector, best_value).
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_raw(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        g = _subgradient_maxconv(a)
        eta = eta0 / math.sqrt(it)   # diminishing step size
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 100_000 == 0:
            cur = _objective_raw(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
    cur = _objective_raw(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Seed generation (extended)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Construct a diverse pool of ℓ₁‑normalised seed vectors."""
    seeds: list[np.ndarray] = []

    # analytic families -----------------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))

    # edge‑heavy shapes -----------------------------------------------------------
    for p in (0.2, 0.5, 1.0, 2.0):
        seeds.append(_edge_power_shape(n, p))
    for q in (2.0, 3.0, 4.0, 5.0):
        seeds.append(_edge_trough_shape(n, q))

    # Gaussian / power families ----------------------------------------------------
    for sigma in np.linspace(0.08, 0.45, 9):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta distributions – explore very small parameters as well as medium ones
    beta_vals = [0.01, 0.02, 0.05, 0.08, 0.1, 0.2, 0.5, 0.8, 1.0, 1.5]
    for a in beta_vals:
        for b in beta_vals:
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws with low concentration -------------------------------
    for conc in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Exponential draws ------------------------------------------------------------
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves --------------------------------------------------
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sparse random supports, broader size range ----------------------------------
    for _ in range(30):
        k_min = int(rng.integers(20, 50))
        k_max = int(rng.integers(100, 200))
        seeds.append(_random_sparse(n, rng, k_min=k_min, k_max=k_max))

    # Many random mixtures ---------------------------------------------------------
    base_idx = np.arange(len(seeds))
    for _ in range(400):
        i, j = rng.choice(base_idx, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Warm‑start from previous best (if compatible size) -------------------------
    if previous is not None and previous.shape == (n,):
        if previous.sum() > 0:
            seeds.append(previous / previous.sum())

    # final clean‑up ---------------------------------------------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Search driver for a single problem size
# ----------------------------------------------------------------------
def _search_for_n(n: int,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Run the full optimisation pipeline for a concrete length ``n``."""
    # -------------------------------------------------------------------
    # Gather seeds, possibly using the previous best as a warm‑start
    # -------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            pa = np.asarray(previous, dtype=np.float64)
            if pa.shape == (n,) and pa.sum() > 0:
                prev_arr = pa / pa.sum()
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"[{n}] generated {len(seeds)} seed vectors.")

    # surrogate: 2·n·max(conv) (all seeds are ℓ₁‑normalised)
    def surrogate(v):
        return 2.0 * n * np.max(_conv_fft(v, v))

    scored = [(vec, surrogate(vec)) for vec in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 7
    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"[{n}] best surrogate among seeds: {scored[0][1]:.6f}")

    best_seq = top_seeds[0].copy()
    best_val = surrogate(best_seq)

    # -------------------------------------------------------------------
    # Stage 1 – L‑BFGS continuation on the smooth surrogate
    # -------------------------------------------------------------------
    beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0,
                     200.0, 500.0, 1000.0, 2000.0,
                     5000.0, 10000.0, 20000.0,
                     50000.0, 100000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        per_beta = max(600,
                       int(min(3000,
                               (deadline - time.time()) * 0.05 * 1e5)))
        seq = _lbfgs_smooth_optimize(seed_vec,
                                      n,
                                      deadline,
                                      beta_schedule=beta_schedule,
                                      max_iter_per_beta=per_beta)
        val = surrogate(seq)
        if val < best_val:
            best_seq = seq.copy()
            best_val = val
            print(f"[{n}] L‑BFGS seed #{idx+1} improved to {best_val:.6f}")

    # -------------------------------------------------------------------
    # Stage 2 – Exact‑objective exponentiated‑gradient refinement
    # -------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(2_500_000, remaining * 0.3 * 1e5))
        if max_iters > 0:
            seq, val = _exp_gradient_optimize(best_seq,
                                              n,
                                              deadline,
                                              max_iters=max_iters,
                                              eta0=0.9)
            if val < best_val:
                best_seq = seq.copy()
                best_val = val
                print(f"[{n}] Exp‑Grad refined to {best_val:.6f}")

    # -------------------------------------------------------------------
    # Stage 3 – Stochastic perturb‑and‑polish
    # -------------------------------------------------------------------
    if time.time() < deadline:
        attempts = 40
        for a in range(attempts):
            if time.time() > deadline:
                break
            logits = np.log(best_seq + 1e-30)
            scale = max(0.02, 0.6 / (a + 1))
            jitter = rng.normal(scale=scale, size=n)
            start_vec = _softmax(logits + jitter)

            seq_cand = _lbfgs_smooth_optimize(start_vec,
                                              n,
                                              deadline,
                                              beta_schedule=[1.0, 20.0, 200.0,
                                                            2000.0],
                                              max_iter_per_beta=700)
            cand_val = surrogate(seq_cand)
            if cand_val < best_val:
                best_seq = seq_cand.copy()
                best_val = cand_val
                print(f"[{n}] Perturb‑&‑polish #{a+1} improved to {best_val:.6f}")

    # -------------------------------------------------------------------
    # Stage 4 – Final high‑β polishing
    # -------------------------------------------------------------------
    if time.time() < deadline:
        per_beta = max(500,
                       int(min(2500,
                               (deadline - time.time()) * 0.04 * 1e5)))
        seq_final = _lbfgs_smooth_optimize(best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=[50000.0, 100000.0,
                                                          200000.0],
                                           max_iter_per_beta=per_beta)
        val = surrogate(seq_final)
        if val < best_val:
            best_seq = seq_final.copy()
            best_val = val
            print(f"[{n}] Final high‑β polishing gave {best_val:.6f}")

    # -------------------------------------------------------------------
    # Exact evaluation (official)
    # -------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[{n}] Exact evaluate_sequence = {true_val:.6f}")
        best_val = true_val
    except Exception as exc:   # pragma: no cover
        print(f"[{n}] evaluate_sequence raised: {exc}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step function that minimises ``evaluate_sequence``.
    The search respects the given time budget (default 1000 s) and
    stores the best sequence in the global ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 1.0   # safety margin

    # Primary problem size (default 4000).  A slightly smaller and larger size
    # are also tried when time permits – this sometimes bypasses discretisation
    # artefacts.
    n_default = int(kwargs.get("n", 4000))
    n_candidates = [n_default]
    if budget_s > 300:
        n_candidates.append(max(2000, int(0.9 * n_default)))
        n_candidates.append(min(8000, int(1.1 * n_default)))

    best_overall_seq = None
    best_overall_val = float('inf')

    for n in n_candidates:
        if time.time() > deadline:
            break
        # split remaining time equally among remaining n‑values
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining / max(1, len(n_candidates) - n_candidates.index(n))
        seq, val = _search_for_n(n, sub_deadline, rng)
        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq
            print(f"[Overall] New best for n={n}: {best_overall_val:.6f}")

    # Fallback in the extremely unlikely case that nothing was produced
    if best_overall_seq is None:
        best_overall_seq = _uniform(10)

    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
