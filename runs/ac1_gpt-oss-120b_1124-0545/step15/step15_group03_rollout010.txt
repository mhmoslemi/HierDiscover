def _smooth_max_opt(start_seq, n, deadline, max_iters=300_000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    # scheduling
    T0 = 0.1
    T_min = 1e-5
    eta0 = 0.5

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # temperature schedule: exponential decay over iterations
        T = max(T_min, T0 * (0.9995 ** it))
        a = _softmax(log_a)   # ensure a is fresh after each log update

        # compute conv
        conv = _conv_fft(a, a)
        max_conv = np.max(conv)
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # maybe print progress occasionally
            if it % 50000 == 0:
                print(f"[SmoothOpt] it={it} best={best_val:.6f} T={T:.6g}")

        # compute softmax weights for max
        shift = conv - max_conv  # shift for numeric stability
        w_raw = np.exp(shift / T)
        w_sum = np.sum(w_raw)
        if w_sum == 0.0:
            # fallback to uniform weights
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_raw / w_sum

        # gradient wrt a
        # g = 2 * np.correlate(w, a, mode='valid')
        g = 2.0 * np.correlate(w, a, mode='valid')
        # convert to gradient in log space (mirror descent)
        dot_ag = np.dot(a, g)
        grad_log = a * (g - dot_ag)

        # step size
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log

    return best_seq, best_val

"""
Hybrid minimiser for the step‑function objective

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in five stages:

1. **Rich seed pool** – a large collection of analytical families
   (uniform, beta, power‑law, Gaussian, triangular, random Dirichlet,
   sparse‑k‑point) is generated and ℓ₁‑normalised.

2. **Coarse pre‑selection** – every seed is scored with the exact
   surrogate ``2·n·max(conv)`` (the denominator is 1 after normalisation)
   and the best ``top_k`` seeds are kept.

3. **Smooth‑max mirror descent** – for each top seed we run a
   temperature‑annealed optimisation in the log‑softmax space.
   The objective is the *soft‑max* of the convolution values,
   i.e.  ``T·log Σ exp(conv/T)``.  Its gradient can be written in closed
   form using a correlation, allowing thousands of cheap iterations.
   The schedule gradually lowers the temperature, driving the smooth
   approximation to the true max.

4. **Projected sub‑gradient polishing** – a classic Euclidean projected
   sub‑gradient descent is applied to the current global best for a
   large number of iterations (still on the simplex).

5. **Short simulated annealing** – a final exploration in the logit
   space refines the solution a little further.

All stages are time‑aware; the total wall‑clock budget is ``budget_s``.
The best sequence found is stored in the global variable
``height_sequence_1`` and returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangle peaked in the centre."""
    a = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(a, n + 1 - a)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_k_uniform(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=min(k, n), replace=False)
    a[idx] = 1.0 / len(idx)
    return a

def _smooth_max_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Temperature‑annealed mirror‑descent on the *soft‑max* of the convolution.
    The surrogate used is 2·n·max(conv); we minimize a smooth approximation
    ``T·log Σ exp(conv/T)`` which converges to the true max as ``T→0``.
    Returns the best sequence observed and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in the log‑softmax (logit) space – guarantees simplex feasibility
    log_a = np.log(a + 1e-30)

    # temperature schedule (linear decay) and basic step‑size rule
    T0 = 0.12          # start temperature
    T_min = 1e-5
    eta0 = 0.6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current probability vector
        a = _softmax(log_a)

        # full convolution
        conv = _conv_fft(a, a)
        max_conv = np.max(conv)
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50_000 == 0:
                print(f"[Smooth] it={it:7d} best={best_val:.6f}")

        # ----- smooth‑max weights -----
        progress = it / max_iters
        T = max(T_min, T0 * (1.0 - progress))   # linear cooling

        # shift for numerical stability (max conv becomes zero)
        shift = conv - max_conv
        # safe exp – under‑flow is fine (gives 0)
        w_raw = np.exp(shift / T)
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_raw / w_sum

        # gradient of Σ w_k · conv_k w.r.t. a:
        # g = 2 * corr(w, a)   (valid mode → length n)
        g = 2.0 * np.correlate(w, a, mode='valid')

        # convert Euclidean gradient to gradient in log‑space
        dot_ag = np.dot(a, g)
        grad_log = a * (g - dot_ag)

        # step size (mirror descent)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient (mirror‑descent) that follows the *active*
    max‑convolution index only.  Kept as a fast fallback.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Euclidean projected sub‑gradient descent on the true max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple SA in the log‑softmax space, useful for a final local push.
    """
    rng = np.random.default_rng(12345)
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.06 * (1.0 - progress) + 1e-7
        sigma = 0.20 * (1.0 - progress) + 1e-5

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Best surrogate among raw seeds = {scored[0][1]:.6f}")

    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all smooth runs.")
            break

        # allocate iteration budget proportionally to remaining time
        remaining = deadline - time.time()
        est_iters = int(min(800_000, remaining * 2500))   # rough 2500 it/s
        if est_iters < 30_000:
            est_iters = int(min(200_000, remaining * 2500))

        print(f"\n--- Smooth‑max run #{idx+1} (max_iters≈{est_iters}) ---")
        seq_opt, val_opt = _smooth_max_opt(seed_vec, n, deadline, max_iters=est_iters)

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. (Optional) short exponentiated‑gradient refinement for any seeds
    #    that still have time left – sometimes it pushes the value a little.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        est_iters = int(min(250_000, remaining * 1200))
        if est_iters > 0:
            print("\n--- Exponentiated‑gradient polishing (optional) ---")
            seq_opt, val_opt = _exp_gradient_opt(global_best_seq, n, deadline, max_iters=est_iters)
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[ExpGrad] improved best to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Sub‑gradient polishing (long run) – pushes the hard max further down.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(1_200_000, remaining * 0.6))
        if max_sub > 0:
            print("\n--- Sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq, n, deadline, max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved best to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing refinement (short).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(350_000, remaining * 0.35))
        if max_sa > 0:
            print("\n--- Simulated‑annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing(global_best_seq, n, deadline, max_iters=max_sa)
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[SA] final best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
