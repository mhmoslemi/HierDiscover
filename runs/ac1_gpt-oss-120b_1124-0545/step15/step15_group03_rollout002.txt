"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is the full linear convolution.  The optimiser is a
projected Adam with a smooth “soft‑max’’ approximation of the max
operator.  A small schedule of ε‑values is used (large ε → smooth,
small ε → close to the true max).  Several analytical seeds are tried,
the best few are refined with the Adam optimiser, a short projected
sub‑gradient polish finishes the run and the final sequence is stored
in the global ``height_sequence_1`` variable.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all returned ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed gradient of the surrogate (max) objective
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, eps_smooth: float) -> np.ndarray:
    """
    Gradient of  f_eps(a) = 2·n·smooth_max(conv(a,a))
    where smooth_max(x) = eps*log Σ exp(x/eps).
    Returns a vector of length n.
    """
    n = a.shape[0]

    # 1) convolution a*a (full length 2n-1)
    conv = _conv_fft(a, a)                     # shape (2n-1,)

    # 2) soft‑max weights over the convolution entries
    m = np.max(conv)
    # stable softmax
    exp_vals = np.exp((conv - m) / eps_smooth)
    w = exp_vals / np.sum(exp_vals)            # Σ w = 1

    # 3) cross‑correlation w ⋆ a = Σ_k w_k a_{k‑j}
    a_rev = a[::-1]
    corr_full = _conv_fft(w, a_rev)            # length = (2n‑1)+(n‑1) = 3n‑2
    start = n - 1
    corr = corr_full[start:start + n]          # length n  (g_j = Σ_k w_k a_{k‑j})

    # 4) gradient of 2·n·smooth_max = 4·n·corr
    grad = 4.0 * n * corr
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser (smooth max)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              deadline: float,
              max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best (lowest surrogate) sequence seen.
    """
    a = start_seq.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.10                     # base learning rate

    # ε‑schedule for the smooth max (coarse → fine)
    # (fraction of total iterations, ε value)
    eps_schedule = [(0.25, 0.02), (0.55, 0.01), (0.80, 0.005), (1.0, 0.001)]

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- pick smoothing ε according to schedule -----
        frac = it / max_iters
        for thr, eps_val in eps_schedule:
            if frac <= thr:
                eps_smooth = eps_val
                break

        # ----- gradient of the smoothed objective -----
        grad = _grad_smooth_max(a, eps_smooth)

        # ----- Adam update (projected onto simplex) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = a - step
        a = _proj_simplex(a)

        # ----- track best surrogate -----
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[Adam] it={it:7d} best_surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient polish (exact max, not smoothed)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the true max‑convolution.
    Returns the best sequence seen.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # simple diminishing step size
        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best_surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 2.0

    n = 4000  # problem size – the same as in the reference run

    # ------------------------------------------------------------------
    # 0. Retrieve previous best (if it exists)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    previous_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                previous_vec = arr / arr.sum()
        except Exception:
            previous_vec = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation (all are ℓ₁‑normalised)
    # ------------------------------------------------------------------
    seeds = []

    seeds.append(_uniform(n))
    if previous_vec is not None:
        seeds.append(previous_vec)

    # a few Beta‑shape seeds (including strong U‑shapes)
    for a in (0.4, 0.6, 1.0, 2.0, 3.0):
        for b in (0.4, 0.6, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))

    # random Beta shapes with log‑uniform parameters
    for _ in range(60):
        a = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
        b = 10 ** rng.uniform(-2, 1)
        seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (a few concentrations)
    for conc in (0.2, 0.5, 1.0, 2.0):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # normalise everything and filter degenerate vectors
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        tot = s.sum()
        if tot > 0.0:
            normed_seeds.append(s / tot)

    print(f"\n--- Seed generation complete: {len(normed_seeds)} candidates ---")

    # ------------------------------------------------------------------
    # 2. Coarse ranking – keep the best few (by surrogate)
    # ------------------------------------------------------------------
    scored = []
    for vec in normed_seeds:
        val = _objective_surrogate(vec)
        scored.append((vec, val))
    scored.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Best surrogate among raw seeds = {scored[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 3. Long Adam optimisation on the top seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all Adam runs.")
            break

        # Allocate a chunk of the remaining budget for this seed
        remaining = max(0.0, deadline - time.time())
        # Rough estimate: aim for ~1500 iterations per second with Adam
        est_iters = int(min(400_000, remaining * 1500))
        if est_iters < 30_000:
            # Not enough time for a meaningful run – skip
            continue

        print(f"\n--- Adam optimisation run #{idx + 1} (max_iters≈{est_iters}) ---")
        seq_opt, val_opt = _adam_opt(seed_vec, deadline, max_iters=est_iters)

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Short projected sub‑gradient polish (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(350_000, remaining * 0.5))
        if max_sub > 0:
            print("\n--- Sub‑gradient polishing (exact max) ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq, deadline, max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[Polish] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

This implementation builds a diverse collection of analytic seeds, ranks them quickly using the exact surrogate,
optimises the most promising candidates with a projected Adam that follows a smooth approximation of the
max‑convolution (allowing gradients from *all* convolution entries), finishes with a short exact sub‑gradient
polish, and finally reports the true evaluation.  The whole search respects the 1000 s wall‑clock budget
and stores the best sequence in the global variable `height_sequence_1`.
