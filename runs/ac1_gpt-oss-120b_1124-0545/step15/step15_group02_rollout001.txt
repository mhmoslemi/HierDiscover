"""
Propose the best non‑negative step function (list of heights) within a
1000 s wall‑clock budget.

Algorithm (high level)
----------------------
1. Gather a diverse pool of seed vectors (uniform, exponential families,
   beta‑shapes, random Dirichlet draws, the previous best if any).
2. Keep the most promising seeds (lowest exact surrogate 2·n·max(conv)).
3. Run a smoothed L‑BFGS‐B optimisation on each seed.  The max is
   approximated by a log‑sum‑exp with a schedule of β that ends at 1e7,
   giving a tighter approximation than the previous 1e5 schedule.
4. Polishing with exponentiated‑gradient (mirror descent on the exact
   max‑convolution) for many iterations.
5. **New step** – LP refinement:
   For the current best vector `a`, solve the linear program
        maximise   Σ g_j
        subject to (a * g)_k ≤ M   (∀k)   and   g ≥ 0,
   where M = max(a*a).  The optimal `g` points in a direction that reduces
   the peak of the auto‑convolution.  Mix the current vector with `g`
   (mixing weight 0.3) and repeat a few times, each time followed by a
   short EG polish.
6. Return the best vector found (ℓ₁‑normalised).  The official
   ``evaluate_sequence`` is called once at the end for the true score.

All helper functions are top‑level; the code respects the 2‑CPU and
1000 s limits and prints progress information.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (no normalisation assumed)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = exact objective for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i/n) for i=0..n‑1."""
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·(n‑1‑i)/n) – reversed exponential."""
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc`."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector (softmax of the final logits).
    """
    if beta_schedule is None:
        # More aggressive schedule ending at 1e7
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 2000.0, 5000.0, 20000.0,
                         50000.0, 200000.0, 500000.0, 1e6,
                         5e6, 1e7]

    # Initialise logits (log‑softmax parametrisation)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stays on the simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # Gradient w.r.t. a
            w = exp_beta / sum_exp                     # soft‑max over conv entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # extract the n‑length block

            # Chain rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (softmax parametrisation)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)   # slightly more conservative than before
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

# Linear‑programming refinement (new component)
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray,
               deadline: float,
               mix: float = 0.30,
               max_lp_time: float = 30.0) -> tuple[np.ndarray, bool]:
    """
    Solve   maximise Σ g_j   s.t.   (a * g)_k ≤ M   ∀k , g ≥ 0.
    Returns the mixed vector (1‑mix)*a + mix*g_normalised and a flag
    indicating success.  The LP is solved with SciPy's HiGHS solver.
    """
    n = a.shape[0]
    M = np.max(_conv_fft(a, a))

    # Build the Toeplitz constraint matrix (2n‑1) × n.
    # Row k, column j has coefficient a_{k−j} if 0≤k−j<n, else 0.
    offsets = -np.arange(n)                     # -0, -1, …, -(n‑1)
    # Each diagonal contains a constant a[d]
    diag_data = [np.full(n, a[d]) for d in range(n)]
    A = _sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format='csr')

    b = np.full(2 * n - 1, M)

    c = -np.ones(n)   # minimise -Σg  → maximise Σg

    # time budget for the LP itself
    lp_deadline = time.time() + min(max_lp_time, deadline - time.time())

    try:
        res = _opt.linprog(c,
                           A_ub=A,
                           b_ub=b,
                           bounds=[(0, None)] * n,
                           method='highs',
                           options={'presolve': True,
                                    'time_limit': max(lp_deadline - time.time(), 0.0)})
    except Exception as e:
        print(f"[LP] solver exception: {e}")
        return a, False

    if not res.success:
        # fall back – no improvement possible
        return a, False

    g = res.x
    if g.sum() == 0:
        return a, False
    g = g / g.sum()

    # convex combination
    a_new = (1.0 - mix) * a + mix * g
    return a_new, True

def _grid_search_exponential(n: int, deadline: float):
    """Sweep exponential family over c ∈ [0.5, 3.0] (25 points)."""
    vals = []
    cs = np.linspace(0.5, 3.0, 25)
    for c in cs:
        if time.time() > deadline:
            break
        vec = _exp_family(n, c)
        vals.append((vec, _objective_surrogate(vec), ('exp', c)))
        rev = _rev_exp_family(n, c)
        vals.append((rev, _objective_surrogate(rev), ('revexp', c)))
    return vals


def _grid_search_beta(n: int, deadline: float):
    """Coarse beta‑shape sweep (α,β ∈ [0.1,5] with 9 points each)."""
    vals = []
    alphas = np.linspace(0.1, 5.0, 9)
    betas = np.linspace(0.1, 5.0, 9)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                return vals
            vec = _beta_shape(n, a, b)
            vals.append((vec, _objective_surrogate(vec), ('beta', a, b)))
    return vals


def _grid_search_power(n: int, deadline: float):
