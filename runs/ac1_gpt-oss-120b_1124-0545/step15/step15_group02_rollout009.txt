"""
search for a step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The optimiser works in three stages:

1.  Load the best sequence from the previous run (if any) and generate a few
    cheap random Dirichlet candidates.  Keep the best of these as a start.
2.  Perform a fast “mirror‑descent on the exact objective”.  A move consists
    of transferring a small amount of mass ε from a relatively large entry
    i to a relatively small entry j; the convolution is updated analytically
    in O(n) using slice operations so that many iterations fit into the 1000 s
    budget.
3.  Return the best sequence found (renormalised to Σa = √(2n)) and store it in
    the global variable `height_sequence_1` for the next run.

The algorithm does not rely on external solvers and stays well within the
time limit.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Hill‑climbing utilities (exact max‑convolution update)
# ----------------------------------------------------------------------
def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    i: int,
                    j: int,
                    eps: float) -> np.ndarray:
    """
    Return a copy of ``conv`` that corresponds to moving ``eps`` mass from
    index ``i`` to index ``j`` in the vector ``a`` (the array ``a`` itself
    is **not** modified here).

    The update uses the exact algebraic formula

        conv' = conv - 2·eps·shift_i + 2·eps·shift_j + d*d

    where ``shift_i`` and ``shift_j`` are the pieces of ``a`` that appear in
    the convolution with the moved mass.
    """
    n = a.shape[0]
    conv_new = conv.copy()

    # linear part: a*d + d*a
    conv_new[i:i + n] -= 2.0 * eps * a
    conv_new[j:j + n] += 2.0 * eps * a

    # quadratic part d*d
    conv_new[2 * i] += eps * eps
    conv_new[2 * j] += eps * eps
    conv_new[i + j] -= 2.0 * eps * eps

    return conv_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated wall‑clock budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # a tiny safety margin

    # ------------------------------------------------------------------
    # Problem size and normalisation
    # ------------------------------------------------------------------
    n = 4000
    target_sum = math.sqrt(2.0 * n)                # Σa = √(2n) makes the objective = max(conv)

    # ------------------------------------------------------------------
    # 0. Load previous solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and len(previous) == n:
        a0 = np.asarray(previous, dtype=np.float64)
        # protect against accidental zero‑sum
        if a0.sum() > 0.0:
            a0 = a0 * (target_sum / a0.sum())
        else:
            a0 = None
    else:
        a0 = None

    # ------------------------------------------------------------------
    # 1. Seed generation: uniform + a few random Dirichlet draws
    # ------------------------------------------------------------------
    if a0 is not None:
        best_a = a0.copy()
        conv = _conv_fft(best_a, best_a)
        best_val = conv.max()
    else:
        best_a = np.full(n, target_sum / n, dtype=np.float64)   # uniform start
        conv = _conv_fft(best_a, best_a)
        best_val = conv.max()

    print(f"[Init] start value = {best_val:.6f}")

    # a few cheap random Dirichlet candidates
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(12):
            if time.time() > deadline:
                break
            cand = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            cand *= target_sum / cand.sum()
            cur_conv = _conv_fft(cand, cand)
            cur_val = cur_conv.max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = cand
                conv = cur_conv
                print(f"[Random] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Exact sub‑gradient / mirror‑descent polishing
    # ------------------------------------------------------------------
    a = best_a.copy()               # current working vector
    conv = _conv_fft(a, a)          # current convolution (full)
    cur_max = conv.max()
    best_seq = a.copy()
    best_score = cur_max

    # hyper‑parameters
    move_scale = 0.30               # initial relative step size (fraction of source entry)
    temperature = cur_max * 0.02 + 1e-9   # Simulated‑annealing temperature
    cooling = 0.99995               # geometric cooling per iteration
    last_improve = 0
    it = 0

    # helper for fast sum (constant)
    const_sum = target_sum

    print("[Hill‑climb] start polishing")
    while time.time() < deadline:
        it += 1

        # --- choose a source (large entry) and a destination (small entry) ---
        # source index sampled proportionally to its current mass
        prob = a / const_sum
        i = rng.choice(n, p=prob)

        # destination uniformly, but avoid picking the same index
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # proposed transfer amount (ensure non‑negativity)
        eps = a[i] * move_scale * rng.random()
        if eps <= 0.0:
            continue

        # --- compute the new convolution (exact update) ---
        conv_candidate = _apply_transfer(a, conv, i, j, eps)
        new_max = conv_candidate.max()

        # --- acceptance test (simple hill‑climb with occasional uphill moves) ---
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # simulated annealing
            delta = new_max - cur_max
            if rng.random() < math.exp(-delta / temperature):
                accept = True

        if accept:
            # apply the move
            a[i] -= eps
            a[j] += eps
            conv = conv_candidate
            cur_max = new_max

            if cur_max < best_score:
                best_score = cur_max
                best_seq = a.copy()
                last_improve = it
                print(f"[Iter {it:8d}] new best = {best_score:.6f}")

        # cooling schedule
        temperature *= cooling
        if temperature < 1e-12:
            temperature = 1e-12

        # slowly shrink the step size
        if it % 15000 == 0:
            move_scale *= 0.7
            if move_scale < 1e-5:
                move_scale = 1e-5

        # occasional full recomputation to avoid drift from rounding errors
        if it % 5000 == 0:
            conv = _conv_fft(a, a)
            cur_max = conv.max()

        # if we have been stuck for a long while, perform a small random shake
        if it - last_improve > 150000:
            # randomly redistribute the mass of a tiny random subset
            subset = rng.choice(n, size=5, replace=False)
            mass = a[subset].sum()
            a[subset] = rng.dirichlet(np.ones(5)) * mass
            conv = _conv_fft(a, a)
            cur_max = conv.max()
            last_improve = it
            print(f"[Shake] after {it} iters, max = {cur_max:.6f}")

    # ------------------------------------------------------------------
    # 3. Final normalisation and evaluation
    # ------------------------------------------------------------------
    # re‑normalise exactly (tiny drift may have accumulated)
    best_seq = best_seq * (target_sum / best_seq.sum())

    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== finished ===\n surrogate max(conv) = {best_score:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
