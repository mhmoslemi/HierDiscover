def _antidiagonal_random(n, rng, k_min=30, k_max=200):
    K = int(rng.integers(k_min, k_max+1))
    idx = rng.choice(n//2, size=K, replace=False)  # pick from first half
    weights = rng.exponential(size=K)
    a = np.zeros(n)
    a[idx] = weights
    # The other half stay zero.

"""
Search for a non‑negative step function that minimizes

        2·n·max(conv(a,a)) / (sum(a)²)

The algorithm
----------------
1.  Build a large, diverse pool of ℓ₁‑normalised “seed” vectors
    (analytic windows, Dirichlet draws, sparse random supports, …).
    Extra seeds emphasise mass near the edges (strongly U‑shaped) and
    a novel “edge‑heavy” family a_i ∝ ((i+1)(n‑i))⁻ᵖ.

2.  Keep the best few seeds (according to the exact objective, which for a
    normalised vector is 2·n·max(conv)) and run a continuation L‑BFGS
    on a smooth surrogate (log‑sum‑exp of the convolution).  The β‑schedule
    goes from 0.5 up to 2·10⁴, with a generous per‑β iteration budget.

3.  Refine the result with a mirror‑descent (exponentiated‑gradient)
    optimiser that uses *all* near‑maximal lags (not just a single arg‑max)
    to form a richer sub‑gradient.  The step‑size decays as ηₜ∝1/√t.

4.  A cheap stochastic “perturb‑and‑polish’’ loop (log‑softmax jitter

5.  Finally a very cheap hill‑climbing phase moves small amounts of mass
    between random coordinates and accepts any improvement.  This often
    squeezes a few extra digits out of the objective.

All stages respect a global time budget (≈ 1000 s) and the best sequence
found is stored in the global name ``height_sequence_1`` for warm‑starts.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:                     # safety – should never happen
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on {1,…,n} (U‑shape when α,β<1)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)

    s = a.sum()

def _random_sparse(n: int, rng: np.random.Generator,
                   k_min: int = 30, k_max: int = 150) -> np.ndarray:
    """Sparse support: choose K random indices, give them random weights."""
    K = int(rng.integers(k_min, k_max + 1))
    idx = rng.choice(n, size=K, replace=False)
    a = np.zeros(n, dtype=np.float64)
    weights = rng.exponential(scale=1.0, size=K)
    a[idx] = weights
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α)+b·α then ℓ₁‑renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Surrogate objective (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Value & gradient for the smooth max‑convolution surrogate."""
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)              # full auto‑convolution
    shift = np.max(conv)                # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # weights for each lag – same as soft‑max over conv entries
    w = exp_term / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """L‑BFGS‑B optimisation of the smooth surrogate (log‑sum‑exp)."""
    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during beta={beta_val}")
            break
        except Exception as e:               # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)

def _objective_raw(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) / (sum²).  a is assumed ℓ₁‑normalised."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_b = np.max(conv)
    return 2.0 * n * max_b


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (using a *single* active lag).
    Returns g = ∂ max(conv) / ∂ a .
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


def _exp_gradient_multi_subgrad(start_a: np.ndarray,
                               n: int,
                               deadline: float,
                               max_iters: int = 500_000,
                               eta0: float = 1.0,
                               rel_tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *exact* max‑convolution objective.
    Uses the *average* sub‑gradient over all lags whose
    convolution value is within ``rel_tol`` of the current maximum.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_raw(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)

        # active set = all indices with value >= max‑rel_tol
        active = np.where(conv >= max_val - rel_tol)[0]
        # construct weight vector over lags (uniform over the active set)
        w = np.zeros_like(conv)
        if len(active) == 0:               # shouldn't happen
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / len(active)

        # gradient of the smooth surrogate with this discrete weight
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)         # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # mirror‑descent (exponentiated‑gradient) update
        eta = eta0 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * grad_a
        a = _softmax(log_a)

        if it % 100_000 == 0:
            cur = _objective_raw(a)
            print(f"[Exp‑Grad‑Multi] it={it:,} obj={cur:.6f}")
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_raw(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


def _hill_climb(seq: np.ndarray,
                n: int,
                deadline: float,
                max_moves: int = 150_000,
                step_scale: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing: move a small amount of probability
    mass from a random source index to a random target index and keep the
    change if the exact objective improves.
    """
    rng = np.random.default_rng()
    a = seq.copy()
    best_val = _objective_raw(a)

    for m in range(max_moves):
        if time.time() > deadline:
            break

        # pick a source with non‑trivial mass
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        # pick a different target
        tgt = rng.integers(n - 1)
        if tgt >= src:
            tgt += 1

        # propose moving a random fraction of the source mass
        delta = step_scale * a[src] * rng.random()
        if delta <= 0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # sanity – source cannot become negative (numerical guard)
        if cand[src] < -1e-15:
            continue
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()          # renormalise to stay on the simplex

        cur = _objective_raw(cand)
        if cur < best_val:
            best_val = cur
            a = cand
    return a, best_val


# ----------------------------------------------------------------------
# Seed generation (now enriched with extra families)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Build a diverse pool of ℓ₁‑normalised seed vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12, 20):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.08, 0.45, 9):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    for a in np.linspace(0.1, 1.5, 7):
        for b in np.linspace(0.1, 1.5, 7):
            seeds.append(_beta_shape(n, a, b))

    for al in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, al))

    # New families ------------------------------------------------------
    # Alternating “high‑low’’ patterns
    for amp in (0.6, 0.8, 0.95):
        seeds.append(_alternating(n, amp))

    # Edge‑heavy U‑shapes (strong mass at the two ends)
    for power in (0.3, 0.5, 0.8, 1.0, 1.5):
        seeds.append(_edge_heavy(n, power))

    # Random continuous draws -------------------------------------------
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sparse random supports --------------------------------------------
    for _ in range(20):
        seeds.append(_random_sparse(n, rng, k_min=30, k_max=120))

    # Mixtures of already generated seeds -------------------------------
    base_idx = np.arange(len(seeds))
    for _ in range(300):
        i, j = rng.choice(base_idx, size=2, replace=False)
        alpha = rng.random()
        mix = _mix_shapes(seeds[i], seeds[j], alpha)
        seeds.append(mix)

    # Previous best (if any) --------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Final clean‑up – ensure non‑negativity & ℓ₁‑normalisation ---------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    ``evaluate_sequence`` under the given time budget.  Returns the best
    sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    # -------------------------------------------------------------------
    # Problem size -------------------------------------------------------
    n = int(kwargs.get("n", 4000))
    print(f"\n=== Optimisation start (n={n}) ===")

    # -------------------------------------------------------------------
    # Stage 0 – generate seeds and keep the most promising few ------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # cheap exact objective for screening (sum == 1 → 2·n·max(conv))
    def exact_obj(v):
        return 2.0 * n * np.max(_conv_fft(v, v))

    evaluated = [(vec, exact_obj(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 7
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best exact objective among seeds (top {top_k}):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # -------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on the smooth surrogate ---------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = exact_obj(global_best_seq)
    print(f"\n[Init] exact = {global_best_val:.6f}")

    beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0,
                     200.0, 500.0, 1000.0, 2000.0,
                     5000.0, 10000.0, 20000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all L‑BFGS seeds.")
            break

        per_beta_iters = max(800,
                             int(min(3500,
                                     (deadline - time.time()) * 0.05 * 1e5)))
        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter/beta={per_beta_iters}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=per_beta_iters)
        val_opt = exact_obj(seq_opt)
        print(f"[L‑BFGS] exact = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best exact = {global_best_val:.6f}")

    # -------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient with multi‑active set ---------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(2_200_000,
                                remaining * 0.30 * 1e5))
        if max_exp_iters > 0:
            print("\n--- Multi‑active exponentiated‑gradient refinement ---")
            seq_exp, val_exp = _exp_gradient_multi_subgrad(global_best_seq,
                                                          n,
                                                          deadline,
                                                          max_iters=max_exp_iters,
                                                          eta0=1.0,
                                                          rel_tol=1e-12)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Exp‑Grad‑Multi] improved exact objective to {global_best_val:.6f}")

    # -------------------------------------------------------------------
    # Stage 3 – stochastic perturb‑and‑polish (log‑softmax jitter) -------
    if time.time() < deadline:
        print("\n--- Stochastic perturb‑and‑polish phase ---")
        attempts = 30
        for a in range(attempts):
            if time.time() > deadline:
                break
            logits = np.log(global_best_seq + 1e-30)
            scale = max(0.03, 0.6 / (a + 1))
            jitter = rng.normal(scale=scale, size=n)
            start_logits = logits + jitter
            start_vec = _softmax(start_logits)

            seq_candidate = _lbfgs_smooth_optimize(start_vec,
                                                   n,
                                                   deadline,
                                                   beta_schedule=[1.0, 20.0, 200.0],
                                                   max_iter_per_beta=1200)
            cand_val = exact_obj(seq_candidate)
            if cand_val < global_best_val:
                global_best_seq = seq_candidate.copy()
                global_best_val = cand_val
                print(f"[Perturb] improvement after attempt {a+1}: {global_best_val:.6f}")

    # -------------------------------------------------------------------
    # Stage 4 – cheap hill‑climbing local search -------------------------
    if time.time() < deadline:
        print("\n--- Hill‑climbing local search ---")
        seq_hc, val_hc = _hill_climb(global_best_seq,
                                     n,
                                     deadline,
                                     max_moves=180_000,
                                     step_scale=0.015)
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[Hill‑Climb] improved exact objective to {global_best_val:.6f}")

    # -------------------------------------------------------------------
    # Stage 5 – final high‑β L‑BFGS polishing --------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta_iters = max(600,
                             int(min(2500,
                                     remaining * 0.04 * 1e5)))
        print("\n--- Final high‑β L‑BFGS polishing ---")
        seq_final = _lbfgs_smooth_optimize(global_best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=[5000.0, 10000.0, 20000.0],
                                           max_iter_per_beta=per_beta_iters)
        val_final = exact_obj(seq_final)
        print(f"[Final‑L‑BFGS] exact = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after polishing = {global_best_val:.6f}")

    # -------------------------------------------------------------------
    # Final evaluation with the official evaluator (for reporting)
    # -------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest exact ≈ {global_best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible warm‑starts on the next call
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
