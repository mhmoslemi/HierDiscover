def refine_by_pairwise(seq, max_iters, deadline):
    a = seq.copy()
    for it in range(max_iters):
        if time.time()>deadline: break
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        # compute contributions
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1)
        contrib = a[idx_i] * a[k - idx_i]
        # Pick i with max contrib
        i = idx_i[np.argmax(contrib)]
        # Choose j randomly not equal i
        j = rng.integers(0, n)
        if i == j: continue
        # transfer epsilon = min(a[i], delta)
        delta = min(a[i], 0.01 * a[i])  # 1% of mass at i
        if delta <= 0: continue
        a[i] -= delta
        a[j] += delta
        # re-normalize? sum unchanged automatically.
        # compute new objective; keep if improved else revert
        new_val = _objective_surrogate(a)
        # keep current best if better else revert
        if new_val < best_val:
            best_val, best_seq = new_val, a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta
```

def _pairwise_hill_climb(seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    rng = np.random.default_rng()
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        max_val = conv[k]

        # compute contributions
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx_i
        contrib = a[idx_i] * a[rev]

        # choose index i with probability proportional to contrib
        total_contrib = contrib.sum()
        if total_contrib <= 0:
            i = rng.integers(0, n)
        else:
            probs = contrib / total_contrib
            i = rng.choice(idx_i, p=probs)

        # choose another index j uniformly at random (different)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # amount to transfer
        delta = min(a[i], max(1e-9, a[i] * 0.03))  # 3% of mass at i
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _objective_surrogate(a_new)

        if new_val < best_val - 1e-9:  # significant improvement
            best_val = new_val
            best_seq = a_new.copy()
            a = a_new
        # else keep current a (reject)
        # optional: accept if equal?

        # optionally print progress
        if it % 100_000 == 0:
            print(f"[HillClimb] it={it} best={best_val:.6f}")
    return best_seq, best_val
```

"""
Propose a non‑negative step‑function (list of heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The search proceeds in several stages:

1.  Seed generation – a rich pool of analytically‑parameterised families
    (uniform, exponential, beta, power‑law, Gaussian, triangular) together
    with random Dirichlet draws.  All seeds are ℓ₁‑normalised.

2.  Keep the most promising seeds (≈10) and improve each of them with
    L‑BFGS on a smooth log‑sum‑exp surrogate of the max‑convolution.
    The temperature (β) schedule is aggressive and reaches very large
    values (up to 1e7).

3.  Mix the two best seeds linearly – many convex combinations are tested
    – to capture possible benefits of complementary shapes.

4.  Exponentiated‑gradient (mirror‑descent) polishing on the exact
    max‑convolution for a large number of iterations (the step‑size
    ηₜ≈0.6/√t).

5.  A cheap L‑BFGS pass again (starting from the EG result) with an
    even higher β‑schedule.

6.  Final “pairwise hill‑climbing’’ refinement: repeatedly move a
    small amount of mass from an index that contributes heavily to the
    current maximal convolution entry to a random other index.  Moves are
    accepted only if they strictly lower the surrogate value.  This
    local search further pushes the objective below the target.

All stages respect the global time budget (default 1000 s) and the
sequence length n=4000.  The best sequence found is stored in the global
variable ``height_sequence_1`` and returned as a plain list.
"""

import time
import math
import numpy as np

except Exception:  # pragma: no cover

# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via a real FFT (O(n log n)).
    The result length is len(x) + len(y) - 1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv)/sum² (no normalisation assumed).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective for ℓ₁‑normalised vectors:
    f(seq) = 2·n·max(conv(seq,seq))   (since sum(seq)=1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i/n) for i = 0,…,n‑1."""
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """Reverse exponential – a_i ∝ exp(-c·(n‑1‑i)/n)."""
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_family(n: int, sigma: float) -> np.ndarray:
    """
    Discrete Gaussian centred in the middle of the vector.
    a_i ∝ exp(-0.5*((i-(n‑1)/2)/sigma)²).
    """
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _triangular(n: int) -> np.ndarray:
    """Symmetric discrete triangle (peak in the centre)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(i, n - i + 1)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc`."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution (log‑sum‑exp) using L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector (softmax of the final logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         1e4, 2e4, 5e4, 1e5, 2e5, 5e5,
                         1e6, 2e6, 5e6, 1e7]

    # initialise logits (log‑softmax parametrisation)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stays on the simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                       # max_i conv_i
            # log‑sum‑exp smoothing
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (derived from the smoothed max)
            w = exp_beta / sum_exp                     # soft‑weights
            # convolution of reversed a with w gives gradient components
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution (sub‑gradient of the max).
    Works in the log‑space (softmax parametrisation).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (same pattern as in the original code)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.6 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        if it % 250_000 == 0:
            print(f"[ExpGrad] iter={it:,}  best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Pairwise hill‑climbing refinement (local search)
# ----------------------------------------------------------------------
def _pairwise_hill_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Cheap O(n log n) local search: repeatedly move a small amount of mass
    from an index that heavily contributes to the current maximal
    convolution entry to a random other index, accepting only improving
    moves.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # contribution of each pair (i, k-i) to the current max entry
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx_i
        contrib = a[idx_i] * a[rev]

        # pick an index i proportional to its contribution
        total = contrib.sum()
        if total <= 0:
            i = rng.integers(0, n)
        else:
            probs = contrib / total
            i = rng.choice(idx_i, p=probs)

        # pick a distinct random destination j
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # amount to transfer (3 % of the source mass, bounded below)
        delta = min(a[i], max(1e-12, 0.03 * a[i]))
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        val_new = _objective_surrogate(a_new)

        if val_new + 1e-12 < best_val:      # strict improvement
            best_val = val_new
            best_seq = a_new.copy()
            a = a_new
        # else keep the old a

        if it % 100_000 == 0:
            print(f"[HillClimb] iter={it:,}  best={best_val:.6f}")

        # re‑normalise every few thousand steps to avoid drift
        if it % 25_000 == 0:
            a = a / a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Mixing of two seed vectors (simple convex combination)
# ----------------------------------------------------------------------
def _mix_two_vectors(v1: np.ndarray, v2: np.ndarray,
                     n: int, steps: int = 21) -> tuple[np.ndarray, float]:
    """Test convex combinations (t·v1 + (1‑t)·v2) and keep the best."""
    best_seq = v1
    best_val = _objective_surrogate(v1)
    for s in range(steps):
        t = s / (steps - 1)
        mix = (1 - t) * v1 + t * v2
        mix = mix / mix.sum()
        val = _objective_surrogate(mix)
        if val < best_val:
            best_val = val
            best_seq = mix.copy()
    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    The function returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------
    # 0. Warm‑start from a previously stored solution, if any
    # ------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and np.sum(arr) > 0:
                prev_arr = arr / np.sum(arr)   # ℓ₁‑normalise
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------
    # 1. Seed generation
    # ------------------------------------------------------------
    print("\n--- Seed generation ---")
    candidate_pool = []

    # Baselines
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)),
                           ('uniform',)))
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr),
                               ('previous',)))

    # Exponential families (both decreasing and increasing)
    for c in np.linspace(0.5, 3.0, 27):
        if time.time() > deadline:
            break
        vec = _exp_family(n, c)
        candidate_pool.append((vec, _objective_surrogate(vec), ('exp', c)))
        rev = _rev_exp_family(n, c)
        candidate_pool.append((rev, _objective_surrogate(rev), ('revexp', c)))

    # Beta‑shape sweep (coarse then finer around the best)
    alphas = np.linspace(0.5, 3.5, 13)
    betas = np.linspace(0.5, 3.5, 13)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            vec = _beta_shape(n, a, b)
            candidate_pool.append((vec, _objective_surrogate(vec),
                                   ('beta', a, b)))

    # Power‑law sweep
    for p in np.linspace(-0.5, 5.0, 31):
        if time.time() > deadline:
            break
        vec = _power(n, p)
        candidate_pool.append((vec, _objective_surrogate(vec),
                               ('power', p)))

    # Gaussian family (several sigmas)
    sigma_vals = [n / 8.0, n / 6.0, n / 5.0, n / 4.0, n / 3.0, n / 2.5]
    for sigma in sigma_vals:
        if time.time() > deadline:
            break
        vec = _gaussian_family(n, sigma)
        candidate_pool.append((vec, _objective_surrogate(vec),
                               ('gauss', sigma)))

    # Triangular shape
    tri = _triangular(n)
    candidate_pool.append((tri, _objective_surrogate(tri), ('triangular',)))

    # Random Dirichlet draws (various concentrations)
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per_conc = 25
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec),
                                   ('dirichlet', conc)))

    # ------------------------------------------------------------
    # 2. Keep the most promising seeds
    # ------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])               # lower is better
    top_k = 12
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Generated {len(candidate_pool)} seeds, keeping top {top_k}.")

    # ------------------------------------------------------------
    # 3. L‑BFGS continuation on the top seeds
    # ------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx+1} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         max_iter_per_beta=2500)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------
    # 4. Linear mixing of the two best seeds (if we have at least two)
    # ------------------------------------------------------------
    if len(top_seeds) >= 2 and time.time() < deadline:
        print("\n--- Mixing best two seeds ---")
        seq_mix, val_mix = _mix_two_vectors(top_seeds[0], top_seeds[1],
                                            n, steps=31)
        print(f"[Mix] surrogate = {val_mix:.6f}")
        if val_mix < global_best_val:
            global_best_seq = seq_mix.copy()
            global_best_val = val_mix
            print(f"[Update] after mixing: best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------
    # 5. Exponentiated‑gradient polishing (long run)
    # ------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # very generous allocation – roughly 0.5 % of remaining sec per 10⁴ it.
        max_exp = int(min(6_000_000, remaining * 0.008 * 1e5))
        print(f"\n--- Exponentiated‑gradient polishing (max_iters={max_exp}) ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                             n,
                                             deadline,
                                             max_iters=max_exp)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------
    # 6. Second L‑BFGS pass with an even higher β‑schedule
    # ------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Second L‑BFGS pass (high β) ---")
        high_beta_sched = [1e5, 5e5, 1e6, 5e6, 1e7, 2e7]
        seq_opt2 = _lbfgs_smooth_optimize(global_best_seq,
                                          n,
                                          deadline,
                                          beta_schedule=high_beta_sched,
                                          max_iter_per_beta=2000)
        val_opt2 = _objective_surrogate(seq_opt2)
        print(f"[L‑BFGS‑2] surrogate = {val_opt2:.6f}")
        if val_opt2 < global_best_val:
            global_best_seq = seq_opt2.copy()
            global_best_val = val_opt2
            print(f"[Update] new best after 2nd L‑BFGS = {global_best_val:.6f}")

    # ------------------------------------------------------------
    # 7. Pairwise hill‑climbing refinement
    # ------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc = int(min(400_000, remaining * 0.007 * 1e5))
        print(f"\n--- Pairwise hill‑climbing (max_iters={max_hc}) ---")
        seq_hc, val_hc = _pairwise_hill_climb(global_best_seq,
                                              n,
                                              deadline,
                                              max_iters=max_hc)
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[HillClimb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------
    # 8. Final evaluation with the official evaluator
    # ------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
