"""
Search for a non‑negative step function (list of heights) that
minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The strategy improves on the previous implementation by

1. Starting from the best known vector (or a fresh Dirichlet sample).
2. Optimising with an Adam‑style mirror‑descent on a *smoothed* max‑convolution
   objective.  The smoothing uses a soft‑max over the few largest convolution
   entries, which yields a richer sub‑gradient than the vanilla “single‑active‑index”
   gradient.
3. A short but aggressive sub‑gradient polishing phase.
4. A longer simulated‑annealing refinement in log‑softmax space.
5. All phases respect the 1000 s wall‑clock budget.

The optimiser works directly on log‑probabilities (logits); the soft‑max
ensures the ℓ₁‑normalisation constraint automatically.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities (FFT convolution, soft‑max, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        # fallback – uniform
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for a vector with ℓ₁‑norm = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _adam_smooth_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     beta0: float = 0.5,
                     beta_max: float = 30.0,
                     top_k: int = 5,
                     rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent on the *smooth* max‑convolution objective.
    The smoothing blends the largest `top_k` convolution entries with a
    soft‑max weight (controlled by a schedule from `beta0` to `beta_max`).
    Returns the best sequence seen and its (non‑smoothed) surrogate value.
    """
    if rng is None:
        rng = np.random.default_rng()
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)               # work in log‑space
    m = np.zeros_like(log_a)                # first‑moment accumulator
    v = np.zeros_like(log_a)                # second‑moment accumulator
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    beta1_pow, beta2_pow = 1.0, 1.0

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current probabilities
        a = _softmax(log_a)

        # full convolution and current surrogate
        conv = _conv_fft(a, a)
        max_conv = np.max(conv)
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 100_000 == 0:
                print(f"[Adam] it={t:7d} best={best_val:.6f}")

        # ----- smoothing schedule -------------------------------------------------
        # linear interpolation of the temperature-like parameter β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / max_iters)

        # pick the `top_k` largest convolution entries (unsorted is fine)
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
        # compute soft‑max weights (normalised)
        # we work with the *relative* values conv/ max_conv to obtain a sensible scale
        rel = conv[idxs] / max_conv   # in (0,1]
        w_raw = np.exp(beta * (rel - 1.0))   # max entry gets weight 1
        w = w_raw / (w_raw.sum() + 1e-30)

        # ----- gradient of the (smoothed) maximum ---------------------------------
        g = np.zeros_like(a)                     # gradient w.r.t. a
        for idx, wi in zip(idxs, w):
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            # vector of indices in the overlapping interval
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = idx - idx_i
            g[idx_i] += 2.0 * wi * a[rev_idx]

        # gradient w.r.t. log‑probabilities (chain rule)
        grad_log = g * a

        # ----- Adam update ---------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # diminishing step size (similar to classical schedule)
        eta = 0.3 / math.sqrt(t)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        log_a = log_a - step

    return best_seq, best_val


def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Simple SA in the log‑softmax (logit) space."""
    rng = np.random.default_rng(12345)
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.06 * (1.0 - progress) + 1e-7
        sigma = 0.20 * (1.0 - progress) + 1e-5

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence`` within
    the allocated time budget.  Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # problem size – kept identical to the reference solution
    n = 4000

    # ------------------------------------------------------------------
    # 0. Retrieve the previous best (if any) and normalise it
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    base = None
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.size == n and arr.sum() > 0.0:
            base = arr / arr.sum()
    if base is None:
        base = np.full(n, 1.0 / n, dtype=np.float64)   # uniform fallback

    # ------------------------------------------------------------------
    # 1. Tiny seed pool (previous best + a few Dirichlet draws)
    # ------------------------------------------------------------------
    seeds = [base]
    for _ in range(4):
        # Dirichlet with moderate concentration – gives diverse shapes
        conc = rng.uniform(0.2, 2.0)
        seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # evaluate seeds (surrogate = 2·n·max(conv) because ℓ₁‑norm = 1)
    best_val = float('inf')
    best_seq = None
    for s in seeds:
        val = _objective_surrogate(s)
        if val < best_val:
            best_val = val
            best_seq = s.copy()
    print(f"\n--- Seed stage complete: best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam‑smooth optimisation (multiple restarts)
    # ------------------------------------------------------------------
    global_best_seq = best_seq.copy()
    global_best_val = best_val
    run = 0
    max_restarts = 5

    while time.time() < deadline and run < max_restarts:
        # perturb the current best a little for the later restarts
        if run == 0:
            start_vec = global_best_seq.copy()
        else:
            noise = rng.normal(scale=0.02, size=n)
            logits = np.log(global_best_seq + 1e-30) + noise
            start_vec = _softmax(logits)

        remaining = max(0.0, deadline - time.time())
        # a conservative estimate: ~2500 iterations per second for this optimiser
        max_iters = int(min(600_000, remaining * 2500))
        if max_iters <= 0:
            break

        print(f"\n--- Adam‑smooth run #{run+1} (max_iters≈{max_iters}) ---")
        seq_opt, val_opt = _adam_smooth_opt(
            start_vec,
            n,
            deadline,
            max_iters=max_iters,
            beta0=0.4,
            beta_max=30.0,
            top_k=7,
            rng=rng,
        )

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

        run += 1

    # ------------------------------------------------------------------
    # 3. Sub‑gradient polishing (exact max‑convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(800_000, remaining * 0.6))
        if max_sub > 0:
            print("\n--- Sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq, n, deadline, max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved best to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing refinement (longer than before)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(600_000, remaining * 0.5))
        if max_sa > 0:
            print("\n--- Simulated‑annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing(global_best_seq, n, deadline, max_iters=max_sa)
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[SA] final best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
