"""
Light‑weight but aggressive optimiser for the max‑auto‑convolution objective.

The pipeline
-------------
1. **Rich seeding** – many Dirichlet draws, analytic families and a few
   simple deterministic shapes (uniform, V‑shaped).  All candidates are
   ℓ₁‑normalised (∑a = 1) and evaluated with a cheap surrogate
   2·n·max(conv)  (exact max of the linear convolution).

2. **Continuation on a smooth surrogate** – the non‑smooth max is replaced
   by a log‑sum‑exp with an increasing temperature β.  For each β we run
   L‑BFGS‑B on the soft‑max parametrisation v (a = softmax(v)).  The
   gradient of the smoothed objective is obtained analytically by a single
   FFT‑based convolution.

3. **Adam‑based polishing** – after the continuation a high β (≈10⁶) is
   kept and the same gradient is used inside an Adam optimiser.  This
   converges the solution much closer to the true non‑smooth optimum.

4. **Random hill‑climbing** – tiny mass transfers between random
   coordinates improve the exact max‑convolution further.

All steps respect a 1000 s wall‑clock budget and keep the sequence
non‑negative.  The best sequence is stored in the global variable
``height_sequence_1`` and also returned as a plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (required for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Numerical utilities (FFT‑based convolution, soft‑max, simplex projection)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _v_shape(n: int) -> np.ndarray:
    """U‑shaped (large at the ends, small in the centre)."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(i, n - 1 - i) + 1.0
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangle (peak in the middle)."""
    i = np.arange(n, dtype=np.float64)
    a = np.minimum(i + 1, n - i)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Objective helpers (fast surrogate – assumes ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _surrogate_val(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised seq."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _max_conv(seq: np.ndarray) -> float:
    """Exact max‑convolution (no normalisation)."""
    return np.max(_conv_fft(seq, seq))


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient (used by L‑BFGS & Adam)
# ----------------------------------------------------------------------


def _smooth_obj_grad(v: np.ndarray, n: int, beta: float):
    """
    Smooth surrogate of the max‑convolution.
    Input:
        v    – log‑it vector (softmax(v) = a)
        n    – length of a
        beta – temperature (larger ⇒ closer to true max)
    Returns:
        obj  – scalar (smooth max)
        grad – gradient w.r.t. v (same size as v)
    """
    a = _softmax(v)               # stays on the simplex, sum(a)=1
    conv = _conv_fft(a, a)        # length 2n‑1
    shift = np.max(conv)          # subtract max for numerical stability
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_beta)
    # smooth max = shift + log(sum_exp)/beta
    obj = shift + np.log(sum_exp) / beta

    w = exp_beta / sum_exp        # softmax weights over the 2n‑1 entries

    # gradient of smooth max w.r.t a:
    #   g_i = 2 * Σ_j a_j w_{i+j}
    conv_rev = _conv_fft(a, w[::-1])                     # length 3n‑2
    idx = (len(w) - 1) - np.arange(n)                    # slice giving Σ_j a_j w_{i+j}
    grad_a = 2.0 * conv_rev[idx]

    # chain rule through soft‑max: ∂obj/∂v = a * (grad_a - <grad_a,a>)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return float(obj), grad_v.astype(np.float64)


# ----------------------------------------------------------------------
# L‑BFGS continuation on the smooth surrogate
# ----------------------------------------------------------------------


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    L‑BFGS‑B optimisation of the smoothed objective.
    Returns an ℓ₁‑normalised vector (soft‑max of the final logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            0.1, 0.5, 2.0, 10.0, 50.0, 200.0, 1_000.0,
            5_000.0, 20_000.0, 100_000.0,
            500_000.0, 1_000_000.0, 5_000_000.0, 10_000_000.0,
        ]

    # Initialise logits (log‑softmax parametrisation)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def func(v_local, n_local, beta_local):
            return _smooth_obj_grad(v_local, n_local, beta_local)

        # SciPy's `minimize` can receive extra arguments via the `args` tuple
        try:
            res = _opt.minimize(
                fun=func,
                x0=v,
                args=(n, beta),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': max_iter_per_beta,
                         'ftol': 1e-12,
                         'disp': False},
                callback=lambda *_: (_ for _ in ()).throw(KeyboardInterrupt())
                if time.time() > deadline else None,
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta}: {e}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate (high β → true max)
# ----------------------------------------------------------------------


def _adam_smooth_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta: float = 1_000_000.0,
                     lr: float = 0.05,
                     max_iters: int = 1_500_000) -> np.ndarray:
    """
    Adam‑style optimisation of the log‑sum‑exp surrogate.
    Works directly on the probability vector (via log‑variables).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    m = np.zeros_like(a)
    v_vec = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    best_a = a.copy()
    best_val = _surrogate_val(best_a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        exp_beta = np.exp(beta * (conv - shift))
        w = exp_beta / exp_beta.sum()

        conv_rev = _conv_fft(a, w[::-1])
        idx = (len(w) - 1) - np.arange(n)
        grad_a = 2.0 * conv_rev[idx]

        g = a * (grad_a - np.dot(grad_a, a))

        # Adam update on log‑variables
        m = beta1 * m + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v_vec / (1 - beta2 ** it)

        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        cur_val = _surrogate_val(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[Adam] it={it:,d} best surrogate={best_val:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Simple random hill‑climbing (exact max‑convolution)
# ----------------------------------------------------------------------


def _hill_climb(seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_trials: int = 200_000) -> np.ndarray:
    """
    Very cheap random local search: move a tiny amount of mass between
    two random positions and keep the change if the exact max‑convolution
    improves.
    """
    a = seq.copy()
    best_a = a.copy()
    best_val = _max_conv(best_a)

    for t in range(max_trials):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a small fraction (up to 5 % of the source mass)
        delta = rng.random() * 0.05 * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta

        cur = _max_conv(cand)
        if cur < best_val:
            best_val = cur
            best_a = cand
            # occasional status output
            if t % 50_000 == 0:
                print(f"[HillClimb] trial={t:,d} best max_conv={best_val:.6f}")

    return best_a

def _grid_search_exponential(n: int, deadline: float):
    vals = []
    cs = np.linspace(0.5, 3.0, 25)
    for c in cs:
        if time.time() > deadline:
            break
        vec = _exp_family(n, c)
        vals.append((vec, _surrogate_val(vec), ('exp', c)))
        rev = _rev_exp_family(n, c)
        vals.append((rev, _surrogate_val(rev), ('revexp', c)))
    return vals


def _grid_search_beta(n: int, deadline: float):
    vals = []
    alphas = np.linspace(0.1, 5.0, 9)
    betas = np.linspace(0.1, 5.0, 9)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                return vals
            vec = _beta_shape(n, a, b)
            vals.append((vec, _surrogate_val(vec), ('beta', a, b)))
    return vals


def _grid_search_power(n: int, deadline: float):
    vals = []
    for p in np.linspace(-0.5, 5.0, 31):
        if time.time() > deadline:
            return vals
        vec = _power(n, p)
        vals.append((vec, _surrogate_val(vec), ('power', p)))
    return vals


def _perturb_vec(seq: np.ndarray, rng: np.random.Generator, scale: float = 0.02):
    """Add a small Gaussian perturbation and re‑project onto the simplex."""
    noise = rng.normal(scale=scale, size=seq.shape)
    pert = seq + noise
    pert = np.maximum(pert, 0.0)
    return _proj_simplex(pert)

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size (kept modest for speed, same as the reference)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous best if present
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1. Seed generation (rich and random)
    # --------------------------------------------------------------
    print("\n--- Seed generation ---")
    candidate_pool = []

    # Baselines
    candidate_pool.append((_uniform(n), _surrogate_val(_uniform(n)), ('uniform',)))
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _surrogate_val(prev_arr), ('previous',)))

    # Analytic families
    candidate_pool.extend(_grid_search_exponential(n, deadline))
    candidate_pool.extend(_grid_search_beta(n, deadline))
    candidate_pool.extend(_grid_search_power(n, deadline))

    # Simple V‑shaped and triangular families
    candidate_pool.append((_v_shape(n), _surrogate_val(_v_shape(n)), ('vshape',)))
    candidate_pool.append((_triangular(n), _surrogate_val(_triangular(n)), ('triang',)))

    # Random Dirichlet draws (broad concentration sweep)
    concs = [0.001, 0.005, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(60):            # 60 draws per concentration (fast)
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _surrogate_val(vec), ('dirichlet', conc)))

    # Small random perturbations of the best few seeds (helps seed diversity)
    candidate_pool.sort(key=lambda x: x[1])
    for base_vec, _, _ in candidate_pool[:8]:
        pert = _perturb_vec(base_vec, rng, scale=0.05)
        candidate_pool.append((pert, _surrogate_val(pert), ('perturb',)))

    # --------------------------------------------------------------
    # 2. Keep most promising seeds
    # --------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])
    top_k = 20                     # a little larger than before
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Generated {len(candidate_pool)} seeds, keeping top {top_k} for L‑BFGS.")

    # --------------------------------------------------------------
    # 3. L‑BFGS continuation on the best seeds
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _surrogate_val(global_best_seq)
    print(f"[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx + 1} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                          n,
                                          deadline,
                                          max_iter_per_beta=8000)
        val_opt = _surrogate_val(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Adam‑based polishing on a very high β
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_adam_iters = int(min(1_200_000, remaining * 150))   # rough scaling
        print(f"\n--- Adam polishing (β=1e6, max_iters={max_adam_iters}) ---")
        seq_adam = _adam_smooth_opt(global_best_seq,
                                    n,
                                    deadline,
                                    beta=1_000_000.0,
                                    lr=0.05,
                                    max_iters=max_adam_iters)
        val_adam = _surrogate_val(seq_adam)
        print(f"[Adam] surrogate = {val_adam:.6f}")
        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Update] after Adam best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Random hill‑climbing with the exact max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_trials = int(min(300_000, remaining * 200))   # cheap trials
        print(f"\n--- Hill‑climbing (max_trials={max_trials}) ---")
        seq_hc = _hill_climb(global_best_seq,
                             n,
                             deadline,
                             rng,
                             max_trials=max_trials)
        val_hc = _surrogate_val(seq_hc)
        print(f"[HillClimb] surrogate = {val_hc:.6f}")
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[Update] after hill‑climb best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
