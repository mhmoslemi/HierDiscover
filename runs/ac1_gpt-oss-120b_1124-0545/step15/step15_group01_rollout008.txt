"""
Propose a step‑function (list of non‑negative heights) that minimizes
`evaluate_sequence`.  The algorithm builds a diverse set of ℓ1‑normalised
seeds, runs a continuation L‑BFGS optimiser on a smooth approximation of
the max‑convolution objective, repeatedly restarts the optimiser from
noisy perturbations of the current best, and finally polishes the result
with a long exponentiated‑gradient mirror‑descent.  The search runs for the
full time budget (≈1000 s) and returns the best sequence found.
"""

from __future__ import annotations
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric tools
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for the sizes we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – always returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # extremely unlikely, but keep a uniform fallback
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate (exact) objective for ℓ1‑normalised vectors
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray, n: int) -> float:
    """Exact value of 2·n·max(conv(a,a)) for a probability vector a."""
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic/Random families of ℓ₁‑normalised seeds
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density sampled on the grid {1,…,n} and ℓ₁‑normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    # a U‑shaped Beta distribution (α,β)<1 tends to flatten the
    # convolution – include a few such shapes
    for a in (0.4, 0.6, 0.8):
        for b in (0.4, 0.6, 0.8):
            seeds.append(_beta_shape(n, a, b))
    # Gaussian‑like blobs
    for sigma in np.linspace(0.12, 0.30, 5):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws (various concentrations)
    for conc in (0.1, 0.3, 0.7, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # alternating Dirichlet (often gives a low max‑conv shape)
    seeds.append(_alternating_dirichlet(n, rng))

    # previous best, if compatible
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final normalisation safeguard
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser on the smooth max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _lbfgs_smooth_opt(start_a: np.ndarray,
                      n: int,
                      deadline: float,
                      beta_schedule: list[float] | None = None,
                      max_iter_per_beta: int = 20000) -> np.ndarray:
    """Run L‑BFGS‑B on the smooth max‑convolution objective.

    Returns a probability vector (ℓ₁‑normalised).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimisation package is required.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 100000.0
        ]

    # initialise logits – softmax of logits = start_a
    v = np.log(start_a + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta = float(beta)

        def fun(v_local: np.ndarray):
            a = _softmax(v_local)                # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp (stable)
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            smooth = shift + np.log(sum_exp) / beta
            obj = 2.0 * n * smooth

            # gradient wrt a
            w = exp_beta / sum_exp               # soft‑max weights over conv entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # central block (size n)

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # early abort – L‑BFGS does not expose a stop flag,
            # so we raise KeyboardInterrupt.
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta, 'ftol': 1e-12, 'disp': False}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            # any numerical hiccup – just continue with current v
            print(f"[L‑BFGS] warning: {e} (beta={beta})")
            continue

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser on the exact max‑conv
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_a: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent optimizer that directly minimises the exact max‐conv."""
    a = start_a.copy()
    best = a.copy()
    best_val = _objective_surrogate(a, n)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (all indices attaining the max receive the same sub‑gradient)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # decreasing step size
        eta = 0.7 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()

    return best, best_val
