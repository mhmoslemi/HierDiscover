def _project_onto_simplex(v):
    # Project onto the probability simplex (sum = 1, v >= 0)
    # Using algorithm from Duchi et al.
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1  # zero-based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Multi‑stage non‑negative step‑function optimizer.

The algorithm proceeds as follows:

1. **Seed generation** – Produce a rich pool of ℓ₁‑normalised vectors
   (uniform, triangular, exponential, window, random Dirichlet, etc.).
   The previous best solution (if any) is injected into the pool.

2. **Coarse ranking** – Evaluate all seeds with the exact objective
   2·n·max(conv(a,a)) (the sum is normalised to 1).  Keep the K best
   seeds for further optimisation.

3. **Continuation L‑BFGS** – Optimise each of the top seeds on the
   smooth log‑sum‑exp surrogate of the max‑convolution.  An increasing
   β‑schedule is used and the optimisation runs until the time budget
   is almost exhausted.

4. **Exact exponentiated‑gradient descent** – Starting from the best
   L‑BFGS result, perform a long mirror‑descent run on the *exact*
   max‑convolution objective with a diminishing step size
   (ηₜ = η₀/√t).  This further reduces the peak of the convolution.

5. **Intensive stochastic local search** – Perform many cheap
   “mass‑transfer’’ moves: move a tiny fraction of probability mass from
   the index contributing most to the current peak to an index that
   contributes least.  Accept only improving moves (a tiny simulated‑annealing
   acceptance is added for robustness).

All stages respect a global deadline (≈ 1000 s).  The final vector is
stored in the global name `height_sequence_1` and returned as a plain
list of floats.
"""

# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    return e / s if s != 0 else np.full_like(v, 1.0 / len(v))


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    Implements the algorithm from Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_monotone(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random decreasing vector (sorted random numbers)."""
    v = rng.random(n)
    v = np.sort(v)[::-1]
    s = v.sum()
    return v / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for lam in (0.0005, 0.001, 0.002, 0.005, 0.01):
        seeds.append(_exponential_decay(n, lam))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0):
        seeds.append(_sine_window(n, gamma=g))
    for sigma in np.linspace(0.10, 0.45, 5):
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random monotone draws
    for _ in range(12):
        seeds.append(_random_monotone(n, rng))

    # mixtures of random pairs
    base = np.arange(len(seeds))
    for _ in range(200):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # sanitise: enforce non‑negativity and ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Surrogate helpers for continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Log‑sum‑exp surrogate of the max‑convolution."""
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)               # full convolution
    shift = np.max(conv)                 # numeric stabiliser
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t a
    w = exp_term / sum_exp                # soft‑max weights over all lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)            # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]  # slice length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 2500,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 100000.0,
            200000.0, 500000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)  # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached (beta={beta_val})")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)   # final probability vector


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient (max‑convolution)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (sum(a) = 1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    Returns 2·a_{idx‑i} on the overlapping support of the maximal lag idx.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i
    g[i] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Operates on the log‑probability simplex (exponentiated gradient).
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)                 # sub‑gradient of max(conv)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        # occasional bookkeeping
        if it % 200_000 == 0 or time.time() > deadline:
            cur = _objective_exact(a)
            print(f"[ExpGrad] it={it:,} obj={cur:.6f}")
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    # final check
    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive stochastic local search (mass transfers)
# ----------------------------------------------------------------------
def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 200_000,
    step_frac: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing: move a small fraction of probability mass from a
    “worst’’ index (largest contribution to the current peak) to a
    “best’’ index (smallest contribution).  Accept only improving moves;
    a tiny simulated‑annealing acceptance rule is added for robustness.
    """
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(12345)

    # temperature schedule for simulated annealing
    T0 = 1e-4
    T_decay = 0.99995

    for it in range(iters):
        if time.time() > deadline:
            break

        # compute current convolution and the maximising lag(s)
        conv = _conv_fft(a, a)
        max_conv = np.max(conv)
        idx = int(np.argmax(conv))

        # contribution of each index to that lag
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev = idx - i_range
        contrib = a[i_range] * a[rev]          # a_i * a_{idx-i}
        # donor: largest contribution
        donor = i_range[np.argmax(contrib)]
        # receiver: smallest contribution (prefer a zero‑pair)
        receiver = i_range[np.argmin(contrib)]

        if donor == receiver or a[donor] <= 0.0:
            # fall back to a random transfer if nothing useful found
            donor = rng.integers(0, n)
            receiver = rng.integers(0, n)
            if donor == receiver or a[donor] <= 0.0:
                continue

        delta = step_frac * a[donor]
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[donor] -= delta
        cand[receiver] += delta

        cur_val = max_conv * 2 * n / (np.sum(cand) ** 2)   # exact objective (sum stays =1)
        if cur_val < best_val:
            best_val = cur_val
            best_a = cand.copy()
            a = cand
        else:
            # simulated‑annealing acceptance
            T = max(T0 * (T_decay ** it), 1e-12)
            if rng.random() < np.exp(-(cur_val - best_val) / T):
                a = cand
        # periodic progress report
        if (it + 1) % 40_000 == 0:
            print(f"[LocalSearch] it={it+1:,} best={best_val:.6f}")

    return best_a, best_val

# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    Returns the best sequence found (list of floats) and stores it in
    the global name `height_sequence_1`.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    n = 4000
    print(f"\n=== Optimisation start: n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Seed pool
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors (ℓ₁‑normalised).")

    # ------------------------------------------------------------------
    # Quick exact ranking of seeds
    # ------------------------------------------------------------------
    exact_vals = [(vec, _objective_exact(vec)) for vec in seeds]
    exact_vals.sort(key=lambda x: x[1])
    top_k = min(12, len(exact_vals))
    top_seeds = [vec for vec, _ in exact_vals[:top_k]]

    print(f"Top {top_k} seed exact values (objective):")
    for i, (_, val) in enumerate(exact_vals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on smooth surrogate
    # ------------------------------------------------------------------
    global_best = top_seeds[0].copy()
    global_best_val = _objective_exact(global_best)
    print(f"\n[Init] best exact objective = {global_best_val:.6f}")

    # L‑BFGS per seed (few best seeds)
    per_seed = min(len(top_seeds), 6)
    for idx in range(per_seed):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS stage.")
            break
        seed_vec = top_seeds[idx]
        remaining = deadline - time.time()
        # give each L‑BFGS run a modest budget (≈ 6 % of remaining time)
        per_beta_iters = max(1200, int(min(4000, remaining * 0.06 * 1e5)))
        print(f"\n--- L‑BFGS run #{idx+1} (max_iter/beta={per_beta_iters}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=[
                1.0, 5.0, 20.0, 80.0, 200.0,
                500.0, 1000.0, 2000.0, 5000.0,
                10000.0, 20000.0, 50000.0, 100000.0,
                200000.0, 500000.0, 1_000_000.0,
            ],
            max_iter_per_beta=per_beta_iters,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] exact objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best (L‑BFGS) = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient (exact max‑convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(6_000_000, remaining * 0.30 * 1e5))
        if max_exp_iters > 0:
            print("\n--- Exhaustive exponentiated‑gradient phase ---")
            seq_exp, val_exp = _exp_gradient_optimize(
                global_best,
                n,
                deadline,
                max_iters=max_exp_iters,
                eta0=1.0,
            )
            print(f"[ExpGrad] best objective = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] New best after EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – intensive stochastic local refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        local_iters = int(min(400_000, remaining * 80))  # a few hundred‑thousands moves
        if local_iters > 0:
            print("\n--- Intensive stochastic local search ---")
            seq_loc, val_loc = _local_search(
                global_best, n, deadline, iters=local_iters, step_frac=0.015
            )
            print(f"[LocalSearch] best objective = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] New best after local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best.tolist())
        print(
            f"\n=== Finished ===\nBest exact objective ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally for warm‑starts
    globals()["height_sequence_1"] = global_best.tolist()
    return global_best.tolist()
