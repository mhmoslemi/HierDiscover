"""
Optimiser for the step‑function bound

Goal: find a non‑negative vector a (length n) minimising
    2·n·max(conv(a,a)) / (sum(a)**2)

Because the ratio is scale‑invariant we keep the iterates ℓ₁‑normalised.
The optimiser consists of three stages:

1.  Massive low‑dimensional seed search (deterministic windows + dense
    sweeps of a few parametric families).  All seeds are ℓ₁‑normalised and
    the 10‑best (according to the exact surrogate 2·n·max(conv)) are kept.

2.  Continuation Adam on the *smooth* max (log‑sum‑exp) using a soft‑max
    parametrisation.  A schedule of β‑values drives the surrogate towards
    the true max while Adam’s adaptive moments accelerate convergence.
    Each elite seed receives a few‑thousand optimisation steps.

3.  Exact mirror‑descent (exponentiated‑gradient) on the *exact* max‑convolution.
    The update  a ← softmax(log a – η·g)  with η=2/√t quickly drives the
    maximum convolution down.  The best vector from stage 2 is refined for up
    to a few million iterations (or until the time budget expires).

4.  Optional short Adam “polish” runs from jittered copies of the current
    best solution.

The whole pipeline respects the 1000 s budget and returns the best vector
found as a plain Python list.
"""

import time
import numpy as np
import heapq
from scipy.special import i0  # Kaiser window


# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – always ℓ₁‑normalised, non‑negative."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector.
    (Exactly the same as evaluate_sequence for sum(seq)=1.)
    """
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    w = i0(beta * np.sqrt(1.0 - t * t)) / i0(beta)
    return w / w.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.16
    w = (1 - a) / 2 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + a / 2 * np.cos(4 * np.pi * i / (n - 1))
    return w / w.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return w / w.sum()


def _tukey_window(n: int, alpha: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1)
    w = np.ones_like(i)
    mask1 = i < alpha / 2
    mask2 = i > 1 - alpha / 2
    w[mask1] = 0.5 * (1 + np.cos(np.pi * (2 * i[mask1] / alpha - 1)))
    w[mask2] = 0.5 * (1 + np.cos(np.pi * (2 * (i[mask2] - 1) / alpha + 1)))
    return w / w.sum()


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p  (decreasing, p≥0)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like (skewed polynomial) shape."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p  (decreasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed generation – large parametric sweep
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    max_random: int = 2500,
) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_kaiser_window(n, beta=4.0))
    seeds.append(_kaiser_window(n, beta=8.0))
    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_tukey_window(n, alpha=0.4))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # systematic sweeps ----------------------------------------------------
    # linear‑power (decreasing) – dense grid
    for p in np.linspace(0.1, 4.0, 30):
        seeds.append(_linear_power(n, p))

    # exponential decay – log‑space grid
    for lam in np.logspace(-5, -0.5, 30):
        seeds.append(_exp_decay(n, lam))

    # power‑law (decreasing) – dense grid
    for p in np.linspace(0.0, 5.0, 30):
        seeds.append(_power(n, p))

    # beta‑shapes – modest cartesian product
    alphas = [0.1, 0.5, 1.0, 2.0, 5.0, 10.0]
    betas = [0.1, 0.5, 1.0, 2.0, 5.0, 10.0]
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    # random Dirichlet draws ------------------------------------------------
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves -------------------------------------------
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # random mixtures with uniform -------------------------------------------
    for _ in range(200):
        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(
            n,
            10 ** rng.uniform(-2, 1),
            10 ** rng.uniform(-2, 1),
        )
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # previous solution (if any) --------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # ----------------------------------------------------------------------
    # Sanitise – ensure non‑negativity and ℓ₁‑normalisation
    # ----------------------------------------------------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Continuation Adam optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule=None,
    steps_per_beta: int = 2000,
) -> np.ndarray:
    """
    Adam optimiser on the smoothed max‑convolution.
    Returns a ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1000.0, 3000.0, 5000.0, 10000.0,
                         20000.0, 50000.0, 100000.0]

    eps = 1e-30
    logits = np.log(start_vec + eps)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.07
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    t = 0

    best_seq = None
    best_val = np.inf

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)
        for _ in range(steps_per_beta):
            t += 1
            if time.time() > deadline:
                break

            a = _softmax(logits)          # current probability vector
            conv = _conv_fft(a, a)        # full conv, length 2n‑1
            shift = np.max(conv)          # numeric stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            w = exp_beta / exp_beta.sum()  # smooth‑max weights, shape (2n‑1,)

            # gradient w.r.t. a: 2 * (rev_a * w) convolution
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)    # length 3n‑2
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_logits = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_logits
            v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # occasional best‑track
            if t % 500 == 0:
                cur_val = _objective_surrogate(a, n)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
        # optional learning‑rate decay (kept simple)
    return best_seq if best_seq is not None else _softmax(logits)


# ----------------------------------------------------------------------
# Exact sub‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_500_000,
) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size ηₜ = 2.0 / √t  (more aggressive than 1.0)
        eta = 2.0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[ExpGrad] iter={it:,} best≈{cur_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Small random‑restart polishing via Adam
# ----------------------------------------------------------------------
def _polish_with_jitter(
    base_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    attempts: int = 3,
) -> np.ndarray:
    """
    Perform a few short Adam runs starting from jittered logits.
    Keeps any improvement found.
    """
    best_seq = base_seq.copy()
    best_val = _objective_surrogate(best_seq, n)

    for k in range(attempts):
        if time.time() > deadline:
            break
        logits = np.log(best_seq + 1e-30)
        jitter = rng.normal(scale=0.015, size=n)  # modest noise
        logits_j = logits + jitter

        refined = _adam_smooth_optimize(
            start_vec=_softmax(logits_j),
            n=n,
            deadline=deadline,
            beta_schedule=[2000.0, 5000.0, 10000.0, 20000.0, 50000.0],
            steps_per_beta=800,
        )
        cur_val = _objective_surrogate(refined, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = refined
            print(f"[Polish] improvement after jitter #{k+1}: {cur_val:.6f}")

    return best_seq

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimisation start (n={n}) ===")

    # ------------------------------------------------------------------
    # Seed generation & rough ranking
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=2500)
    print(f"Generated {len(seeds)} seed vectors.")

    # keep the 15 best according to the exact surrogate (cheap)
    top_k = 15
    heap: list[tuple[float, np.ndarray]] = []  # max‑heap of size top_k
    for vec in seeds:
        val = _objective_surrogate(vec, n)
        if len(heap) < top_k:
            heapq.heappush(heap, (-val, vec))
        else:
            if -val > heap[0][0]:
                heapq.heapreplace(heap, (-val, vec))
    # retrieve in ascending order
    top_seeds = [item[1] for item in sorted(heap, key=lambda x: -x[0])]
    top_vals = [-item[0] for item in sorted(heap, key=lambda x: -x[0])]
    print(f"Top {top_k} surrogate values (lower is better):")
    for i, v in enumerate(top_vals):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam on elite seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq, n)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                  1000.0, 3000.0, 5000.0, 10000.0,
                  20000.0, 50000.0, 100000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all seeds.")
            break

        # allocate a modest number of steps – adapt to remaining time
        remaining = deadline - time.time()
        steps_per_beta = max(800,
                             int(min(4000, remaining * 0.006 * 1e5)))
        print(
            f"\n--- Adam optimisation on seed #{idx+1} "
            f"(steps_per_beta={steps_per_beta}) ---"
        )
        seq_opt = _adam_smooth_optimize(
            start_vec=seed_vec,
            n=n,
            deadline=deadline,
            beta_schedule=beta_sched,
            steps_per_beta=steps_per_beta,
        )
        val_opt = _objective_surrogate(seq_opt, n)
        print(f"[Adam] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact mirror‑descent refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(4_000_000, remaining * 0.28 * 1e5))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient refinement (exact max) ---")
            seq_exp = _exp_gradient_optimize(
                global_best_seq, n, deadline, max_iters=max_exp_iters
            )
            val_exp = _objective_surrogate(seq_exp, n)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – polishing with jittered short Adam runs
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Polishing with jittered Adam runs ---")
        polished = _polish_with_jitter(
            global_best_seq, n, deadline, rng, attempts=4
        )
        val_polish = _objective_surrogate(polished, n)
        if val_polish < global_best_val:
            global_best_seq = polished.copy()
            global_best_val = val_polish
            print(f"[Polish] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
