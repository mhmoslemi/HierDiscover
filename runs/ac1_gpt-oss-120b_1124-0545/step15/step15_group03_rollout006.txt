        f_β(a) = (1/β)·log Σ_i exp(β·(a*a)[i]).
   The temperature β is increased during the run (coarse → fine).
   Gradients are obtained in O(n log n) via FFTs and the update is
   performed in log‑space, automatically preserving the simplex.
4. the best result is polished with a few projected sub‑gradient steps;
5. finally the official ``evaluate_sequence`` function is called to get
   the true score.

The whole search is time‑aware (budget ≈ 1000 s) and returns the
best sequence found.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray):
    """
    Full linear convolution a*a using real FFT.
    Returns (conv, Fa) where conv = a*a and Fa = rfft(a) with length N.
    """
    n = a.shape[0]
    L = 2 * n - 1
    N = _next_pow2(L)
    Fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(Fa * Fa, n=N)[:L]
    return conv, Fa


def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max objective:
        g_j = 2 * Σ_i w_i * a_{i-j}
    for a vector a (size n) and weight vector w (size 2n‑1).
    Computed via FFT in O(n log n).
    """
    n = a.shape[0]
    L = w.shape[0]                # = 2n‑1
    w_rev = w[::-1]              # reverse
    L2 = n + L - 1                # = 3n‑2
    N = _next_pow2(L2)
    Fa = np.fft.rfft(a, n=N)
    Fw = np.fft.rfft(w_rev, n=N)
    conv = np.fft.irfft(Fa * Fw, n=N)[:L2]   # length = 3n‑2
    start = L - 1                # index that corresponds to j = 0
    idx = start - np.arange(n)   # decreasing slice
    g = 2.0 * conv[idx]
    return g


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = a.shape[0]
    conv, _ = _conv_self_fft(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_k_uniform(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=min(k, n), replace=False)
    a[idx] = 1.0 / len(idx)
    return a


def _window_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _window_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _window_blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _window_kaiser(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    x = (i - alpha) / alpha
    from scipy.special import i0  # type: ignore

    a = i0(beta * np.sqrt(1.0 - x * x))
    return a / a.sum()


# ----------------------------------------------------------------------
# Optimiser: smooth‑max exponentiated‑gradient (log‑space)
# ----------------------------------------------------------------------
def _smooth_expgrad_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth‑max surrogate.
    log_a is updated by ``log_a ← log_a – η_t·g`` where ``g`` is the
    gradient of the soft‑max of the convolution and η_t = η₀ / √t.
    A schedule of increasing temperature β (coarse → fine) is used.
    Returns the best sequence seen and its exact surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)  # keep positivity
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # temperature schedule (β values) and corresponding iteration cut‑offs
    # (the thresholds are fractions of max_iters)
    beta_schedule = [(0.15, 20.0), (0.35, 80.0), (0.65, 300.0), (1.0, 800.0)]
    # convert fractions to iteration numbers
    thresholds = [int(frac * max_iters) for frac, _ in beta_schedule]
    betas = [beta for _, beta in beta_schedule]

    eta0 = 0.5

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # pick β according to the current iteration
        if it <= thresholds[0]:
            beta = betas[0]
        elif it <= thresholds[1]:
            beta = betas[1]
        elif it <= thresholds[2]:
            beta = betas[2]
        else:
            beta = betas[3]

        # full self‑convolution
        conv, _ = _conv_self_fft(a)

        # soft‑max weights over the convolution entries
        cmax = np.max(conv)
        # stability: shift by the max before exponentiation
        exp_term = np.exp(beta * (conv - cmax))
        w = exp_term / exp_term.sum()

        # gradient w.r.t. a (smooth version)
        g = _grad_from_weights(a, w)

        # step‑size (classic √t schedule)
        eta = eta0 / math.sqrt(it)

        # mirror‑descent update in log‑space
        log_a = log_a - eta * g
        a = _softmax(log_a)

        # evaluate surrogate (exact max‑conv) – cheap because conv already computed
        cur_val = float(2.0 * n * np.max(conv))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(
                    f"[SmoothEG] iter={it:7d} best={best_val:.6f} β={beta:.1f}"
                )

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (optional)
# ----------------------------------------------------------------------
def _subgradient_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Simple projected sub‑gradient descent on the exact max‑conv."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv, _ = _conv_self_fft(a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        cur_val = 2.0 * n * np.max(_conv_self_fft(a)[0])
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] iter={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Problem size: use the same length as the reference solution.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Recover previous solution if present.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    previous_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                previous_vec = arr / arr.sum()
        except Exception:
            previous_vec = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation (all ℓ₁‑normalised)
    # ------------------------------------------------------------------
    seeds = []

    # baseline uniform and previous best
    seeds.append(_uniform(n))
    if previous_vec is not None:
        seeds.append(previous_vec)

    # classic window families
    seeds.append(_window_hann(n))
    seeds.append(_window_hamming(n))
    seeds.append(_window_blackman(n))
    for beta in (2.0, 4.0, 6.0):
        seeds.append(_window_kaiser(n, beta))

    # beta‑shape grid (coarse)
    for a in np.linspace(0.2, 5.0, 9):
        for b in np.linspace(0.2, 5.0, 9):
            seeds.append(_beta_shape(n, a, b))

    # beta‑shape random draws
    for _ in range(80):
        a = 10 ** rng.uniform(-2, 2)
        b = 10 ** rng.uniform(-2, 2)
        seeds.append(_beta_shape(n, a, b))

    # power‑law grid + random
    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))
    for _ in range(40):
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

    # Gaussian grid + random
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    for _ in range(25):
        sigma = rng.uniform(0.05, 0.55)
        seeds.append(_gaussian(n, sigma))

    # Sparse uniform on random subsets
    for k in (10, 20, 50, 100):
        seeds.append(_random_k_uniform(n, k, rng))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            if time.time() > deadline:
                break
            seeds.append(_random_dirichlet(n, conc, rng))

    # Normalise everything and discard degenerate vectors
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        tot = s.sum()
        if tot > 0.0:
            normed_seeds.append(s / tot)

    print(f"\n--- Seed generation complete: {len(normed_seeds)} candidates ---")

    # ------------------------------------------------------------------
    # 2. Coarse ranking – keep the best few seeds.
    # ------------------------------------------------------------------
    scored = [(vec, _objective_surrogate(vec)) for vec in normed_seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Best surrogate among raw seeds = {scored[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 3. Long smooth‑max exponentiated‑gradient optimisation.
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all runs.")
            break

        # allocate iterations based on remaining time
        remaining = max(0.0, deadline - time.time())
        # Rough benchmark: ~400 iterations per second for the heavy loop
        est_iters = int(min(300_000, remaining * 400))
        if est_iters < 20_000:
            est_iters = int(min(100_000, remaining * 400))
        if est_iters <= 0:
            break

        print(
            f"\n--- Smooth‑ExpGrad run #{idx+1} (max_iters≈{est_iters}) ---"
        )
        seq_opt, val_opt = _smooth_expgrad_opt(
            seed_vec, n, deadline, max_iters=est_iters
        )

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Sub‑gradient polishing (optional, cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(150_000, remaining * 0.5))
        if max_sub > 0:
            print("\n--- Sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(
                global_best_seq, n, deadline, max_iters=max_sub
            )
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved best to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
