def _adam_optimize(start_seq, n, deadline, beta_schedule, max_iter_per_beta=1000, lr=0.05):
    """
    Optimize surrogate using Adam on logits (softmax) with a schedule of beta values.
    Returns the probabilities a (sum=1).
    """
    a = start_seq.copy()
    # logits from softmax inverse
    eps = 1e-30
    logits = np.log(a + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    t = 0

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)
        for it in range(max_iter_per_beta):
            if time.time() > deadline:
                return None
            # compute a
            a = _softmax(logits)
            # convolution
            conv = _conv_fft(a, a)
            # softmax weights for log-sum-exp
            # Using stable shift approach
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            weights = exp_beta / sum_exp

            # gradient w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, weights)
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # shape n

            # backprop to logits
            dot = np.dot(grad_a, a)
            grad_logits = a * (grad_a - dot)

            # Adam update
            t += 1
            m = beta1 * m + (1 - beta1) * grad_logits
            v = beta2 * v + (1 - beta2) * (grad_logits ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

    # final a
    a = _softmax(logits)
    return a

"""
Propose a step‑function (non‑negative vector) minimizing

        2·n·max(conv(a, a)) / (sum(a)²)

within a limited time budget.

Algorithm Overview
------------------
0.  Choose a problem size `n` (default 4000) and a RNG seed.
1.  Generate a large, diverse pool of ℓ₁‑normalised seed vectors:
    * deterministic families (uniform, triangular, sine‑window, Gaussian,
      power‑law, beta‑shapes, exponential decay, linear‑power, mixtures)
    * many random Dirichlet draws (including very sparse ones)
    * sparse “spike” vectors with a few equal‑weight delta‑functions
    * low‑frequency random curves
    * the previous best solution, if any.
2.  Evaluate a cheap surrogate   S(a)=2·n·max(conv(a,a))   (the true
    objective when the vector is ℓ₁‑normalised) and keep the top‑k seeds.
3.  For each elite seed run a *continuous* optimisation based on **Adam**
    applied to the logits of a soft‑max parametrisation.  The max‑operator
    is approximated by a log‑sum‑exp with a schedule of increasing
    temperature‑parameter `β`.  Adam gives a robust, fast local optimiser
    that differs from the L‑BFGS used previously.
4.  Take the best result and improve it with the exact sub‑gradient /
    exponentiated‑gradient (mirror‑descent) method – this works directly
    on the true max‑convolution and can lower the objective further.
5.  Finish with a **simulated‑annealing hill‑climber** that moves a tiny
    amount of mass from the currently largest component to a random one.
    The temperature decays linearly to zero.
6.  If time permits, up‑sample the current best sequence (linear
    interpolation) to double the grid size and repeat steps 3‑5 on the
    finer grid.
7.  Return the best sequence found (as a plain list) and store it in the
    global name ``height_sequence_1`` for reuse.
"""

import time

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector.
    (If sum(seq)=1 then the true objective equals this value.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p  (decreasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _linear_increasing_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ i^p  (increasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = i ** p
    a[0] = 1e-30
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture, then ℓ₁‑normalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int,
                   rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()

    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_power(n, 0.0))

                             10 ** rng.uniform(-2, 1),

        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final sanitisation – guarantee positivity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule,
                   max_iter_per_beta: int = 1200,
                   lr: float = 0.05) -> np.ndarray | None:
    """
    Adam optimisation on logits (soft‑max) of a probability vector.
    The max‑operator is replaced by a log‑sum‑exp with temperature β.
    Returns an ℓ₁‑normalised vector or ``None`` if the deadline is hit.
    """
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    t = 0

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        for _ in range(max_iter_per_beta):
            if time.time() > deadline:
                return None

            a = _softmax(logits)                     # stay on simplex
            conv = _conv_fft(a, a)

            # log‑sum‑exp with a numerical shift
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            if sum_exp == 0.0:                      # safety
                weights = np.full_like(conv, 1.0 / conv.size)
            else:
                weights = exp_beta / sum_exp

            # gradient w.r.t. a (same formula as in the L‑BFGS part)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, weights)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through soft‑max
            dot = np.dot(grad_a, a)
            grad_logits = a * (grad_a - dot)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_logits
            v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

    # final probabilities
    a = _softmax(logits)
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient (mirror‑descent) optimiser (same as reference)
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 600_000) -> np.ndarray:
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size ηₜ = 0.5 / √t
        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq


# ----------------------------------------------------------------------
# Hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 500_000) -> np.ndarray:
    """
    Simulated‑annealing style hill‑climber.
    Moves a small amount of mass from the current largest entry to
    a random other entry. Accepts improvements always; uphill moves
    probabilistically with a linearly decaying temperature.
    """
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(0xC0FFEE)
    total_iters = max_iters

    for it in range(total_iters):
        if time.time() > deadline:
            break

        progress = it / total_iters
        temp = (cur_val * 0.05) * (1.0 - progress) + (cur_val * 1e-6) * progress
        eps = 0.10 * (1.0 - progress) + 1e-6 * progress

        i = int(np.argmax(a))
        if a[i] <= 1e-15:
            continue

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        epsilon = eps * rng.random()
        epsilon = min(epsilon, a[i])

        a[i] -= epsilon
        a[j] += epsilon

        new_val = _objective_surrogate(a)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        else:
            a[j] -= epsilon
            a[i] += epsilon

    return best_seq


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # safety margin

    # ------------------------------------------------------------------
    # Stage 0 – problem size
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # ------------------------------------------------------------------
    # Stage 1 – seed generation & cheap ranking
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,) and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=3000)
    print(f"Generated {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    elite = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best surrogate among seeds: {evaluated[0][1]:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam refinement of elite seeds
    # ------------------------------------------------------------------
    global_best_seq = elite[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] Global best surrogate = {global_best_val:.6f}")

    beta_sched = [2.0, 5.0, 15.0, 50.0, 200.0, 800.0, 3000.0, 8000.0]
    max_iter_beta = 1500

    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            break
        print(f"\n--- Adam optimisation of elite #{idx + 1}/{len(elite)} ---")
        opt_seq = _adam_optimize(seed_vec, n, deadline,
                                 beta_schedule=beta_sched,
                                 max_iter_per_beta=max_iter_beta,
                                 lr=0.05)
        if opt_seq is None:
            print("[Adam] timeout during optimisation")
            continue
        val = _objective_surrogate(opt_seq)
        print(f"[Adam] surrogate = {val:.6f}")
        if val < global_best_val:
            global_best_val = val
            global_best_seq = opt_seq.copy()
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact sub‑gradient (mirror‑descent) refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(1_200_000, remaining * 0.35 * 1e5))
        if max_iters > 0:
            print("\n--- Exponentiated‑gradient refinement of global best ---")
            exp_seq = _exp_gradient_optimize(global_best_seq, n, deadline,
                                            max_iters=max_iters)
            exp_val = _objective_surrogate(exp_seq)
            print(f"[ExpGrad] surrogate = {exp_val:.6f}")
            if exp_val < global_best_val:
                global_best_val = exp_val
                global_best_seq = exp_seq.copy()
                print(f"[Update] New global best after ExpGrad = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 0.30 * 1e5))
        if max_iters > 0:
            print("\n--- Hill‑climbing final phase ---")
            hc_seq = _hill_climb(global_best_seq, n, deadline, max_iters=max_iters)
            hc_val = _objective_surrogate(hc_seq)
            print(f"[HillClimb] surrogate = {hc_val:.6f}")
            if hc_val < global_best_val:
                global_best_val = hc_val
                global_best_seq = hc_seq.copy()
                print(f"[Update] New global best after HillClimb = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – optional up‑sampling & fine‑tuning on a finer grid
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 30.0:
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} for final refinement ---")
            up_seq = _upsample(global_best_seq, new_n)

            # Adam on the finer grid
            up_opt = _adam_optimize(up_seq, new_n, deadline,
                                    beta_schedule=beta_sched,
                                    max_iter_per_beta=1200,
                                    lr=0.05)
            if up_opt is not None:
                up_val = _objective_surrogate(up_opt)
                print(f"[Upsample‑Adam] surrogate = {up_val:.6f}")
                if up_val < global_best_val:
                    global_best_seq = up_opt.copy()
                    global_best_val = up_val
                    n = new_n
                    print(f"[Update] New best after up‑sampling = {global_best_val:.6f}")

            # one more exact mirror‑descent pass on the fine grid
            if time.time() < deadline:
                remaining = deadline - time.time()
                max_iters = int(min(600_000, remaining * 0.15 * 1e5))
                if max_iters > 0:
                    print("\n--- Final exponentiated‑gradient on fine grid ---")
                    final_seq = _exp_gradient_optimize(global_best_seq, n,
                                                       deadline,
                                                       max_iters=max_iters)
                    final_val = _objective_surrogate(final_seq)
                    print(f"[Final‑ExpGrad] surrogate = {final_val:.6f}")
                    if final_val < global_best_val:
                        global_best_seq = final_seq.copy()
                        global_best_val = final_val
                        print(f"[Update] New best after final ExpGrad = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"official evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
