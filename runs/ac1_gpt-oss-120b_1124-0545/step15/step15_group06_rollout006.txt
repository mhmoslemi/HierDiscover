def project_simplex(v):
    # v is a 1D np array.
    n = v.shape[0]
    # ensure not negative? It can be any.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

            if time.time() > deadline:
                break
            logits = np.log(global_best_seq + 1e-30)
            jitter_scale = max(0.05, 0.5/(attempt+1))
            jitter = rng.normal(scale=jitter_scale, size=n)
            start_vec = _softmax(logits + jitter)
            seq_cand = _lbfgs_smooth_optimize(start_vec,
                                              n,
                                              deadline,
                                              beta_schedule=[1.0, 20.0, 200.0],
                                              max_iter_per_beta=800)
            val_cand = _objective_raw(seq_cand)
            if val_cand < global_best_val:
                global_best_seq = seq_cand.copy()
                global_best_val = val_cand
                print(f"[Perturb] improvement at attempt {attempt+1}: {global_best_val:.6f}")

    # final polishing
    if time.time() < deadline:
        print("\n--- Final L-BFGS high beta polish ---")
        seq_final = _lbfgs_smooth_optimize(global_best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=[5000.0, 10000.0, 20000.0],
                                           max_iter_per_beta=2500)
        val_final = _objective_raw(seq_final)
        print(f"[Final polish] objective {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final

    # final evaluation
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nFinished. final objective = {true_val:.6f}")
    except Exception as e:
        print("Evaluation error:", e)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()

"""
A fast multi‑stage optimiser for the step‑function problem.

The core ideas that differ from the previous baseline are:

1. **Projected sub‑gradient descent on the exact max‑convolution objective**.
   The iterate lives on the 𝓁₁‑simplex (using the classic Euclidean projection).
   A simple diminishing step‑size + adaptive back‑off yields a strong
   non‑convex descent that often escapes the local minima that mirror‑descent
   (exponentiated‑gradient) gets stuck in.

2. **Mixture‑based warm‑starts** – a large, diverse seed pool (analytic windows,
   β‑shapes, random Dirichlet, sparse draws, and many random convex mixtures) is
   generated and the best few are taken as starting points for the sub‑gradient
   stage.

3. **Fine polishing** – after the sub‑gradient phase we run the earlier smooth
   L‑BFGS continuation (log‑sum‑exp surrogate) and a cheap stochastic
   perturb‑and‑polish phase.  This combination gives a more thorough search while
   respecting the 1000 s budget.

The function `propose_candidate` follows the required signature; it returns a
list of non‑negative floats (the best sequence found) and stores it in the
global name ``height_sequence_1`` for warm‑starting future runs.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector (ℓ₁‑normalised)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on {1,…,n} (U‑shape when α,β<1)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                           # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_sparse(n: int, rng: np.random.Generator,
                   k_min: int = 30, k_max: int = 150) -> np.ndarray:
    """Sparse support: choose K random indices and assign random weights."""
    K = int(rng.integers(k_min, k_max + 1))
    idx = rng.choice(n, size=K, replace=False)
    a = np.zeros(n, dtype=np.float64)
    weights = rng.exponential(scale=1.0, size=K)
    a[idx] = weights
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α)+b·α then ℓ₁‑renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# ----------------------------------------------------------------------
# Exact objective utilities
# ----------------------------------------------------------------------
def _objective_raw(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_b = np.max(conv)
    return 2.0 * n * max_b

def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (using the active lag).
    Returns g = ∂ max(conv) / ∂ a .
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g

# ----------------------------------------------------------------------
# Projection onto the probability simplex (ℓ₁ = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – return uniform
        return _uniform(n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Smooth surrogate used by L‑BFGS (same as in the reference code)
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Return (value, gradient) for the smooth max‑convolution surrogate."""
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)              # full auto‑convolution
    shift = np.max(conv)                # numerical stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # weights for each lag
    w = exp_term / sum_exp
    # convolution of reversed a with the weights gives ∂obj/∂a
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain‑rule through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v

def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """L‑BFGS‑B optimisation of the smooth surrogate (log‑sum‑exp)."""
    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0,
                         5000.0, 10000.0, 20000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during beta={beta_val}")
            break
        except Exception as e:               # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)

# ----------------------------------------------------------------------
# Seed generation (unchanged except for a few more mixtures)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Build a diverse pool of ℓ₁‑normalised seed vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12, 20):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.08, 0.45, 9):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta families – include U‑shapes (α,β<1) as well as moderate
    for a in np.linspace(0.1, 1.5, 7):
        for b in np.linspace(0.1, 1.5, 7):
            seeds.append(_beta_shape(n, a, b))

    for al in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, al))

    # Random continuous draws ---------------------------------------------------------
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sparse random supports -----------------------------------------------------------
    for _ in range(30):
        seeds.append(_random_sparse(n, rng, k_min=30, k_max=120))

    # Additional mixtures (more aggressive) -------------------------------------------
    base_idx = np.arange(len(seeds))
    for _ in range(500):
        i, j = rng.choice(base_idx, size=2, replace=False)
        alpha = rng.random()
        mix = _mix_shapes(seeds[i], seeds[j], alpha)
        seeds.append(mix)

    # Previous best (if any) ---------------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final clean‑up – ensure non‑negativity & ℓ₁‑normalisation -------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation on the exact objective
# ----------------------------------------------------------------------
def _projected_subgradient_optimize(start_a: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   max_iters: int = 300_000,
                                   init_step: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on 2·n·max(conv) (exact objective).
    Operates directly on the probability simplex via Euclidean projection.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_raw(a)

    step = init_step
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # sub‑gradient of the active lag
        g = _subgradient_maxconv(a)                # shape (n,)

        # diminishing step‑size (classic 1/√t schedule)
        eta = step / math.sqrt(it)

        a_new = a - eta * g
        a_new = _project_simplex(a_new)

        cur_val = _objective_raw(a_new)

        # simple acceptance scheme
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # every few iterations shrink the step if we are stuck
            if no_improve % 2000 == 0:
                step *= 0.9

        a = a_new

        if it % 20_000 == 0:
            print(f"[PGD] iter={it:,} best={best_val:.6f}")

    return best_a, best_val

# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises the
    provided ``evaluate_sequence`` under a time budget.
    Returns the best sequence found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0        # safety margin

    # -----------------------------------------------------------------------
    # Problem size
    # -----------------------------------------------------------------------
    n = int(kwargs.get("n", 4000))
    print(f"\n=== Optimisation start (n={n}) ===")

    # -----------------------------------------------------------------------
    # Warm‑start with a pool of diverse seeds
    # -----------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # exact evaluation (objective = 2·n·max(conv) because sums are normalised to 1)
    evaluated = [(vec, _objective_raw(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 7
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    global_best_seq = top_seeds[0].copy()
    global_best_val = evaluated[0][1]
    print("\nBest seeds (exact objective):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # -----------------------------------------------------------------------
    # Stage 1 – Projected sub‑gradient descent on the exact max‑convolution
    # -----------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before completing PGD on all seeds.")
            break

        remaining = deadline - time.time()
        # Allocate a generous but bounded iteration budget per seed
        max_iters = int(min(300_000, remaining * 25))   # ≈ 25 it/s on a typical core
        print(f"\n--- PGD run on seed #{idx+1} (max_iters={max_iters}) ---")
        seq_opt, val_opt = _projected_subgradient_optimize(seed_vec,
                                                            n,
                                                            deadline,
                                                            max_iters=max_iters,
                                                            init_step=0.5)
        print(f"[PGD] best value from this start = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best = {global_best_val:.6f}")

    # -----------------------------------------------------------------------
    # Stage 2 – Continuation L‑BFGS on the smooth log‑sum‑exp surrogate
    # -----------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- L‑BFGS refinement (smooth surrogate) ---")
        beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0,
                         5000.0, 10000.0, 20000.0]
        per_beta_iters = 3000
        seq_opt = _lbfgs_smooth_optimize(global_best_seq,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=per_beta_iters)
        val_opt = _objective_raw(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] improved global best after L‑BFGS = {global_best_val:.6f}")

    # -----------------------------------------------------------------------
    # Stage 3 – Stochastic perturb‑and‑polish
    # -----------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Stochastic perturb‑and‑polish phase ---")
        attempts = 30
        for a in range(attempts):
            if time.time() > deadline:
                break
            logits = np.log(global_best_seq + 1e-30)
            scale = max(0.05, 0.5 / (a + 1))   # diminishing jitter magnitude
            jitter = rng.normal(scale=scale, size=n)
            start_vec = _softmax(logits + jitter)

            seq_candidate = _lbfgs_smooth_optimize(start_vec,
                                                   n,
                                                   deadline,
                                                   beta_schedule=[1.0, 20.0, 200.0],
                                                   max_iter_per_beta=800)
            cand_val = _objective_raw(seq_candidate)
            if cand_val < global_best_val:
                global_best_seq = seq_candidate.copy()
                global_best_val = cand_val
                print(f"[Perturb] improvement after attempt {a+1}: {global_best_val:.6f}")

    # -----------------------------------------------------------------------
    # Stage 4 – Final high‑β polishing
    # -----------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta_iters = max(500, int(min(2500, remaining * 0.06 * 1e5)))
        print("\n--- Final high‑β L‑BFGS polishing ---")
        seq_final = _lbfgs_smooth_optimize(global_best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=[5000.0, 10000.0, 20000.0],
                                           max_iter_per_beta=per_beta_iters)
        val_final = _objective_raw(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] new best after final polishing = {global_best_val:.6f}")

    # -----------------------------------------------------------------------
    # Final official evaluation (for reporting)
    # -----------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest exact objective ≈ {global_best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for warm‑start on the next call
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
