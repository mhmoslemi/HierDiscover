"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

within the given time budget.

Algorithm – main ideas
-----------------------
* Generate a very diverse set of ℓ₁‑normalised seeds:
  - classical families (uniform, triangular, sine‑window, Gaussian …)
  - *U‑shaped* families (β‑shape with α,β<1, convex power, min‑distance)
  - mixtures of the above and random Dirichlet draws.
* Rank the seeds cheaply with the surrogate S(a)=2·n·max(conv(a,a))
  (valid because the ℓ₁‑norm is forced to 1).
* Run a short L‑BFGS‑B refinement on the best **k** seeds
  (smooth max via log‑sum‑exp).  The best refined seed becomes the
  starting point for the remaining phases.
* Exact mirror‑descent (exponentiated‑gradient) on the *true* max‑conv
  for a few‑hundred‑thousand iterations.
* A simulated‑annealing hill‑climber that moves a small amount of mass
  between *any* two positions (not only from the current maximum).
* If time permits, double the grid size, repeat the L‑BFGS+mirror‑descent
  pipeline and keep the superior result.
* Finally scale the vector to the required ℓ₁‑norm √(2 n) and return it.

All heavy work is guarded by the global deadline (budget – 2 s safety
margin).  The routine stores the best sequence in the global name
``height_sequence_1`` for reuse by later runs.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.fft

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector.
    If sum(seq)=1 the true objective equals this value.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ i^{α−1} (n−i+1)^{β−1}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-λ i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p  (decreasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _convex_u_shape(n: int, p: float) -> np.ndarray:
    """
    U‑shaped (convex) family: a_i ∝ (min(i, n-1-i)+ε)^p.
    ε prevents a zero‑center that would break the log‑sum‑exp.
    """
    eps = 1e-12
    d = np.minimum(np.arange(n, dtype=np.float64),
                    (n - 1) - np.arange(n, dtype=np.float64))
    a = (d + eps) ** p
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture of two probability vectors."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float,
                     rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int,
                   rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed generation – large random sweep over parametrised families
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None,
                   max_random: int = 3000) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # Convex (U‑shaped) family – several exponents
    for p in (0.2, 0.5, 1.0, 2.0, 4.0):
        seeds.append(_convex_u_shape(n, p))

    # Random β‑shapes covering both “peak” and “U” regimes
    for _ in range(max_random // 6):
        # α,β > 0 – we deliberately draw also values < 1 for U‑shapes
        a = 10 ** rng.uniform(-3, 1)      # α ∈ [0.001, 10]
        b = 10 ** rng.uniform(-3, 1)      # β ∈ [0.001, 10]
        seeds.append(_beta_shape(n, a, b))

        # power‑law tails
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        # exponential decay
        lam = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_exp_decay(n, lam))

        # linear power
        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        # mixture of uniform and a β‑shape
        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(n,
                             10 ** rng.uniform(-2, 1),
                             10 ** rng.uniform(-2, 1))
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

        # low‑frequency random cosine curves
        nf = rng.integers(2, 6)   # 2…5 frequencies
        seeds.append(_lowfreq_curve(n, nf, rng))

    # Dirichlet draws – sparse and dense regimes
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Final sanitisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """
    Optimises a smooth surrogate of max‑convolution using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 3000.0, 5000.0, 8000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits (free variables)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stay on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                      # numeric guard
            # smoothed max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smoothed max w.r.t. a
            w = exp_beta / sum_exp                     # soft‑max weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-13,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] time limit reached (beta={beta_val})")
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact sub‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 300_000) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size ηₜ = 0.5 / √t  (standard decreasing schedule)
        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        if it % 100_000 == 0:
            print(f"[ExpGrad] it={it:,} best≈{best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 600_000) -> np.ndarray:
    """
    Simulated‑annealing style hill‑climber.
    Mass is moved from a random donor to a random receiver.
    Improvements are always taken; uphill moves are accepted with a
    temperature that decays linearly to zero.
    """
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(0xC0FFEE)
    total_iters = max_iters

    for it in range(total_iters):
        if time.time() > deadline:
            break

        progress = it / total_iters
        # temperature linearly shrinks – a tiny fraction of the current value
        temp = (cur_val * 0.05) * (1.0 - progress) + (cur_val * 1e-6) * progress
        # step‑size (fraction of donor mass) also decays
        eps = 0.10 * (1.0 - progress) + 1e-6 * progress

        # pick donor (any index with non‑zero mass)
        i = rng.integers(n)
        while a[i] <= 1e-15:
            i = rng.integers(n)

        # pick a different receiver
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        epsilon = eps * rng.random()
        epsilon = min(epsilon, a[i])

        a[i] -= epsilon
        a[j] += epsilon

        new_val = _objective_surrogate(a)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                if it % 100_000 == 0:
                    print(f"[HillClimb] it={it:,} best≈{best_val:.6f}")
        else:
            # revert
            a[j] -= epsilon
            a[i] += epsilon

    return best_seq


# ----------------------------------------------------------------------
# Upsampling (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # -------------------------------------------------------------
    # Stage 0 – problem size
    # -------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # -------------------------------------------------------------
    # Stage 1 – seed generation & cheap ranking
    # -------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=3000)
    print(f"Generated {len(seeds)} seed vectors.")

    # cheap surrogate evaluation
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best surrogate values among seeds (top {top_k}):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # -------------------------------------------------------------
    # Stage 2 – multi‑start L‑BFGS refinement
    # -------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] Current best surrogate = {global_best_val:.6f}")

    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1000.0, 3000.0, 5000.0, 8000.0]

    # run L‑BFGS on the few elite seeds (more breadth than the previous script)
    lbgfs_max_iter = 5000
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- L‑BFGS refinement on elite seed #{idx+1} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline,
                                          beta_schedule=beta_sched,
                                          max_iter_per_beta=lbgfs_max_iter)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS #{idx+1}] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # -------------------------------------------------------------
    # Stage 3 – exact sub‑gradient (mirror‑descent) refinement
    # -------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate roughly 30 % of remaining time to this phase
        max_exp_iters = int(min(800_000, remaining * 0.30 / 0.0005))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient (exact max) refinement ---")
            seq_exp = _exp_gradient_optimize(global_best_seq, n, deadline,
                                            max_iters=max_exp_iters)
            val_exp = _objective_surrogate(seq_exp)
            print(f"[ExpGrad] surrogate = {val_exp:.6f}")

            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # -------------------------------------------------------------
    # Stage 4 – hill‑climbing / simulated‑annealing
    # -------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc_iters = int(min(1_200_000, remaining * 0.30 / 0.0005))
        if max_hc_iters > 0:
            print("\n--- Hill‑climbing (simulated annealing) phase ---")
            seq_hc = _hill_climb(global_best_seq, n, deadline,
                                 max_iters=max_hc_iters)
            val_hc = _objective_surrogate(seq_hc)
            print(f"[HillClimb] surrogate = {val_hc:.6f}")

            if val_hc < global_best_val:
                global_best_seq = seq_hc.copy()
                global_best_val = val_hc
                print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # -------------------------------------------------------------
    # Stage 5 – optional up‑sampling & fine‑tuning
    # -------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 30.0:
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} and refinement ---")
            up_seq = _upsample(global_best_seq, new_n)

            # L‑BFGS on the finer grid
            up_opt = _lbfgs_smooth_optimize(up_seq, new_n, deadline,
                                            beta_schedule=beta_sched,
                                            max_iter_per_beta=3000)
            up_val = _objective_surrogate(up_opt)
            print(f"[Upsample‑L‑BFGS] surrogate = {up_val:.6f}")

            if up_val < global_best_val:
                global_best_seq = up_opt.copy()
                global_best_val = up_val
                n = new_n
                print(f"[Update] new best after up‑sampling = {global_best_val:.6f}")

            # final exp‑gradient on fine grid (shorter run)
            if time.time() < deadline:
                rem = deadline - time.time()
                max_final_exp = int(min(400_000, rem * 0.20 / 0.0005))
                if max_final_exp > 0:
                    print("\n--- Final exponentiated‑gradient on fine grid ---")
                    seq_final = _exp_gradient_optimize(global_best_seq, n,
                                                       deadline,
                                                       max_iters=max_final_exp)
                    val_final = _objective_surrogate(seq_final)
                    print(f"[Final‑ExpGrad] surrogate = {val_final:.6f}")

                    if val_final < global_best_val:
                        global_best_seq = seq_final.copy()
                        global_best_val = val_final
                        print(f"[Update] final best surrogate = {global_best_val:.6f}")

    # -------------------------------------------------------------
    # Scale to the required ℓ₁‑norm and final evaluation
    # -------------------------------------------------------------
    sum_before = global_best_seq.sum()
    scale = np.sqrt(2.0 * n) / (sum_before if sum_before > 0 else 1.0)
    result_seq = (global_best_seq * scale).astype(np.float64)

    true_val = None
    try:
        true_val = evaluate_sequence(result_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = result_seq.tolist()
    return result_seq.tolist()
