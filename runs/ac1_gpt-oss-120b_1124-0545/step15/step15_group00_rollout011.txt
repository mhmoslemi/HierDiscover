def _lbfgs_smooth_optimize(start_vec, n, deadline, beta_schedule=None, max_iter_per_beta=5000):
    if _opt is None: raise ...
    v = np.log(start_vec + 1e-30)
    for beta in beta_schedule:
          if time.time()>deadline: break
          beta_val=beta
          def fun(v_local):
                 a = _softmax(v_local)
                 conv = _conv_fft(a, a)
                 shift = np.max(conv) # shift = max_conv
                 exp_beta = np.exp(beta_val * (conv - shift))
                 sum_exp = exp_beta.sum()
                 obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)
                 # compute gradient:
                 w = exp_beta / sum_exp
                 rev_a = a[::-1]
                 # conv of rev_a and w yields length 3n-2; we need central segment
                 full = _conv_fft(rev_a, w)
                 grad_a = 2.0 * full[n-1:n-1+n]
                 # compute grad_v = a * (grad_a - dot(grad_a, a))
                 dot = np.dot(grad_a, a)
                 grad_v = a * (grad_a - dot)
                 return obj, grad_v
          def callback(xk):
                 if time.time()>deadline:
                     raise KeyboardInterrupt()
          try:
                res = _opt.minimize(fun=fun, x0=v, method='L-BFGS-B', jac=True,
                                    callback=callback,
                                    options={'maxiter': max_iter_per_beta, 'disp': False, 'ftol': 1e-12})
                v = res.x
          except KeyboardInterrupt:
                break
          except Exception as e:
                print(f"[L‑BFGS] exception: {e}")
                break
    return _softmax(v)

def _subgradient_opt(start_seq, n, deadline, max_iters):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        conv = _conv_fft(a,a)
        idx = int(np.argmax(conv))
        # gradient at idx
        g = np.zeros_like(a)
        starti = max(0, idx - (n-1))
        endi = min(n-1, idx)
        # compute contributions:
        i_range = np.arange(starti, endi+1)
        rev = idx - i_range
        g[i_range] = 2.0 * a[rev]
        # step size:
        lr = 0.5 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

"""
search for a non‑negative step‑function that minimises the objective

    2·n·max(conv(a, a)) / (sum(a)²)

The algorithm follows a multi‑stage scheme that fits into the 1000 s budget:

1. **Seed pool** – a large collection of ℓ₁‑normalised vectors built from analytic
   families (uniform, triangular, Gaussian, power‑law), many Dirichlet draws
   with different concentrations and a few permutations, plus the previous
   best solution (if any).  The cheap surrogate 2·n·max(conv) is used to keep
   the top‑K seeds.

2. **Continuation L‑BFGS** – each top seed is refined in the log‑softmax
   (simplex) parametrisation.  A log‑sum‑exp smoothed version of the max‑conv
   is minimised with a long β‑schedule (β grows up to 1e6) and many interior
   iterations.  This gives a very accurate approximation of the true maximum.

3. **Random restarts** – the current best logits are perturbed with a small
   Gaussian noise and the L‑BFGS refinement is repeated a few times.

4. **Projected sub‑gradient descent** – a long run on the exact max‑conv
   surrogate further lowers the peak.

5. **Simulated‑annealing walk** – a final stochastic walk in log‑space explores
   the neighbourhood and may find tiny improvements.

All phases constantly check the global deadline; the best sequence found is
stored in the global variable ``height_sequence_1`` and returned as a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy optimiser
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for ℓ₁‑normalised vectors:
    2·n·max(conv), i.e. the quantity returned by the evaluator when sum(seq)=1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by the official evaluator (no normalisation)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Seed pool construction (adds many shuffled variants)
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Return a rich set of ℓ₁‑normalised candidate vectors.
    """
    seeds: list[np.ndarray] = []

    # analytic families ----------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.10, 0.15, 0.20, 0.25):
        seeds.append(_gaussian(n, sigma))

    for p in (0.0, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        seeds.append(_power(n, p))

    # random Dirichlet draws + shuffles ------------------------------------
    conc_list = (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    step_list = [3, 7, 11, 13, 17, 19, 23, 29, 31, 37]  # coprime to 4000
    for conc in conc_list:
        for _ in range(6):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                               # as‑drawn
            desc = np.sort(raw)[::-1]
            seeds.append(desc)                              # descending
            # alternating large–small order
            sorted_desc = desc
            left, right = 0, n - 1
            alt = np.empty_like(desc)
            for k, val in enumerate(sorted_desc):
                if k % 2 == 0:
                    alt[left] = val
                    left += 1
                else:
                    alt[right] = val
                    right -= 1
            seeds.append(alt)
            for step in step_list[:5]:
                pos = 0
                out = np.empty_like(desc)
                for val in desc:
                    out[pos] = val
                    pos = (pos + step) % n
                seeds.append(out)
            seeds.append(rng.permutation(raw))               # random permutation

    # previous best ---------------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # safe normalisation ----------------------------------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 6000) -> np.ndarray:
    """
    Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a log‑sum‑exp smooth
    approximation of the max‑convolution.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        ]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            # probabilities on the simplex
            a = _softmax(v_local)
            conv = _conv_fft(a, a)                     # full conv
            shift = np.max(conv)                       # max(conv)
            # smooth max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth max
            w = exp_beta / sum_exp                     # soft weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # slice the central n entries

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)                # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (sub‑gradient on exact max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Short exponentiated‑gradient polish on the exact max‑conv surrogate.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Long sub‑gradient descent on 2·n·max(conv) for ℓ₁‑normalised vectors.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]  # exact surrogate

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 300_000 == 0:
                print(f"[SubGrad] it={it:,} best={best_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.4 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing style refinement (logit‑space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Random walk in log‑softmax space with a simple cooling schedule.
    """
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Basic initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Problem size – 4000 is a good trade‑off
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Seed pool construction
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            tmp = np.asarray(previous, dtype=np.float64)
            if tmp.size == n and tmp.sum() > 0.0:
                prev_arr = tmp
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # cheap surrogate (exact for ℓ₁‑normalised vectors)
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nBest surrogate values among raw seeds:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS refinement on the top seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] Deadline reached before finishing all L‑BFGS runs.")
            break

        remaining = deadline - time.time()
        # allocate a few thousand iterations per β, limited by remaining time
        max_iter_beta = int(min(10_000, max(2_000, remaining * 0.015 * 1e5)))

        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter/β = {max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

        # short exponentiated‑gradient polish
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_exp = int(min(300_000, max(80_000, remaining * 0.008 * 1e5)))
            seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_exp)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Random restarts from the current best (jitter + L‑BFGS)
    # ------------------------------------------------------------------
    restarts = 5
    for r in range(restarts):
        if time.time() > deadline:
            break
        sigma = 0.02 * (1.0 - r / restarts) + 1e-4
        jitter = sigma * rng.normal(size=n)
        logits = np.log(global_best_seq + 1e-30) + jitter
        start_vec = _softmax(logits)

        remaining = deadline - time.time()
        max_iter_beta = int(min(8_000, max(2_000, remaining * 0.012 * 1e5)))
        print(f"\n--- Restart #{r+1} L‑BFGS (max_iter/β = {max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(start_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter_beta)
        val_opt = _objective_surrogate(seq_opt)
        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"[Restart {r+1}] improvement to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Long sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(2_000_000, max(300_000, remaining * 0.03 * 1e5)))
        if max_sub > 0:
            print(f"\n--- Long sub‑gradient polishing (up to {max_sub:,} iters) ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub
                global_best_val = val_sub
                print(f"[SubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing refinement (final phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(800_000, max(150_000, remaining * 0.025 * 1e5)))
        if max_sa > 0:
            print(f"\n--- Simulated‑annealing refinement (up to {max_sa:,} iters) ---")
            seq_sa, val_sa = _simulated_annealing_opt(global_best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=max_sa)
            if val_sa < global_best_val:
                global_best_seq = seq_sa
                global_best_val = val_sa
                print(f"[SA] final best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
