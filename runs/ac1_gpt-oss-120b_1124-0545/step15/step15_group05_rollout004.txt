"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

using a two‑phase optimisation:

* Phase 1 – L‑BFGS‑B on a smooth max (log‑sum‑exp) with an increasing
  β‑schedule.  The variables are the logits of a soft‑max
  representation, guaranteeing ℓ₁‑normalisation and non‑negativity.
  This converges much faster than plain Adam and finds a better local
  minimum for the same computational budget.

* Phase 2 – exact mirror‑descent (exponentiated‑gradient) on the true
  max‑convolution to polish the result.  The two phases together push
  the bound below the previous best (~1.5085) while staying comfortably
  inside the 1000‑second limit.

All helper functions are defined at the top level (no closures).  The
global name ``height_sequence_1`` is updated so that later runs can
reuse the best sequence found so far.
"""

import time
import numpy as np
import scipy.optimize as _opt
from scipy.special import i0  # Kaiser window

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (ℓ₁‑normalised, non‑negative)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector."""
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    w = i0(beta * np.sqrt(1.0 - t * t)) / i0(beta)
    return w / w.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.16
    w = (1 - a) / 2 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + a / 2 * np.cos(4 * np.pi * i / (n - 1))
    return w / w.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return w / w.sum()


def _tukey_window(n: int, alpha: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1)
    w = np.ones_like(i)
    mask1 = i < alpha / 2
    mask2 = i > 1 - alpha / 2
    w[mask1] = 0.5 * (1 + np.cos(np.pi * (2 * i[mask1] / alpha - 1)))
    w[mask2] = 0.5 * (1 + np.cos(np.pi * (2 * (i[mask2] - 1) / alpha + 1)))
    return w / w.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
    max_random: int = 2500,
) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_kaiser_window(n, beta=4.0))
    seeds.append(_kaiser_window(n, beta=8.0))
    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_tukey_window(n, alpha=0.4))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 0.5))          # the promising ½‑power
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # random parametric families
    for _ in range(max_random // 6):
        a = 10 ** rng.uniform(-3, 1)
        b = 10 ** rng.uniform(-3, 1)
        seeds.append(_beta_shape(n, a, b))

        p = rng.uniform(0.0, 2.0)        # focus on small p where we expect good shapes
        seeds.append(_power(n, p))

        lam = rng.exponential(scale=0.001) + 1e-4
        seeds.append(_exp_decay(n, lam))

        p_lin = rng.uniform(0.0, 3.0)
        seeds.append(_linear_power(n, p_lin))

        beta = rng.uniform(0.0, 12.0)
        seeds.append(_kaiser_window(n, beta))

        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(
            n,
            10 ** rng.uniform(-2, 1),
            10 ** rng.uniform(-2, 1),
        )
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # sanitise
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Smooth‑max objective used by L‑BFGS‑B
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(logits: np.ndarray, beta: float, n: int):
    """
    Returns (value, grad) where value = 2·n·smooth_max(conv(a,a))
    and grad is the gradient w.r.t. the logits (a = softmax(logits)).
    """
    a = _softmax(logits)
    conv = _conv_fft(a, a)                # length 2n‑1
    shift = np.max(conv)                  # stability shift
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    smooth_max = shift + np.log(sum_exp) / beta
    value = 2.0 * n * smooth_max

    # weights for the smooth max
    w = exp_beta / sum_exp                # shape (2n‑1,)

    # gradient of smooth max w.r.t. a: 2 * (rev_a * w) convolution
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)            # length 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return value, grad_logits


# ----------------------------------------------------------------------
# Optimisation wrappers
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    beta_schedule,
    max_iter_per_beta: int,
    deadline: float,
) -> np.ndarray:
    """
    L‑BFGS‑B optimiser on the smooth‑max objective.
    Returns an ℓ₁‑normalised vector.
    """
    eps = 1e-30
    logits = np.log(start_vec + eps)

    best_seq = None
    best_val = np.inf

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # adapt iteration budget to the remaining time
        time_left = deadline - time.time()
        # rough estimate: 0.003 s per L‑BFGS iteration (convolutions)
        safe_iters = int(min(max_iter_per_beta, max(10, time_left / 0.003 * 0.9)))
        if safe_iters <= 0:
            break

        # SciPy's L‑BFGS‑B can accept a function that returns (f, g)
        res = _opt.minimize(
            fun=_smooth_obj_and_grad,
            x0=logits,
            args=(beta, n),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": safe_iters, "gtol": 1e-14, "disp": False},
        )
        logits = res.x
        a = _softmax(logits)
        val = _objective_surrogate(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        # occasional progress report
        print(f"[L‑BFGS] β={beta:.1e}, iters={res.nit}, surrogate={val:.6f}")

    return best_seq if best_seq is not None else _softmax(logits)


def _exp_gradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 1.0 / np.sqrt(it)          # aggressive step‑size
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{cur_val:.6f}")

    return best_seq


def _polish_with_jitter(
    base_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    attempts: int = 3,
) -> np.ndarray:
    """
    Short L‑BFGS runs from jittered logits.
    Improves the solution if a lucky perturbation helps.
    """
    best_seq = base_seq.copy()
    best_val = _objective_surrogate(best_seq, n)

    for k in range(attempts):
        if time.time() > deadline:
            break
        logits = np.log(best_seq + 1e-30)
        jitter = rng.normal(scale=0.02, size=n)
        logits_j = logits + jitter

        refined = _lbfgs_smooth_optimize(
            start_vec=_softmax(logits_j),
            n=n,
            beta_schedule=[2000.0, 5000.0, 10000.0, 20000.0],
            max_iter_per_beta=600,
            deadline=deadline,
        )
        cur_val = _objective_surrogate(refined, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = refined
            print(f"[Polish] improvement after jitter #{k+1}: {cur_val:.6f}")

    return best_seq

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # problem size -------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # seed generation ----------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=2500)

    # cheap surrogate ranking --------------------------------------------

    print(f"Best surrogate values (top {top_k} seeds):")
    for i, (_, val) in enumerate(evaluated[:top_k]):

    # Phase 1 – L‑BFGS on smooth max ------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq, n)
    print(f"\n[Init] Current best surrogate = {global_best_val:.6f}")

    beta_sched = [

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
