def mass_transfer_mutation(a, step_frac, rng):
    a = a.copy()
    n = len(a)
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i==j or a[i]==0:
        return a
    delta = step_frac * a[i]
    a[i] -= delta
    a[j] += delta
    return a

def _genetic_optimize(initial_seq, n, deadline, pop_size=30, elite_size=5, step_frac=0.02):
    rng = np.random.default_rng(0) # maybe fixed seed?
    # initialize population
    population = [initial_seq.copy()]
    # fill remaining with random Dirichlet draws (conc=0.5 maybe)
    for _ in range(pop_size-1):
        v = rng.dirichlet(np.ones(n) * 0.5)  # conc parameter
        population.append(v)
    # evaluate
    obj_vals = [_objective_exact(v) for v in population]
    best_idx = int(np.argmin(obj_vals))
    best_seq = population[best_idx].copy()
    best_val = obj_vals[best_idx]
    gen = 0
    while time.time() < deadline:
        # sort population
        idx_order = np.argsort(obj_vals)
        elite = [population[i] for i in idx_order[:elite_size]]
        elite_vals = [obj_vals[i] for i in idx_order[:elite_size]]
        # generate new offsprings
        new_pop = elite.copy()  # keep elites
        # generate pop_size - elite_size new
        for _ in range(pop_size - elite_size):
            # pick two parents weighted by fitness (lower obj better)
            # simple random from elite
            p1, p2 = rng.choice(elite, size=2, replace=False)
            alpha = rng.random()
            child = (1 - alpha) * p1 + alpha * p2
            # mutation
            if rng.random() < 0.9:
                child = _mass_transfer_mutation(child, step_frac, rng)
            # optionally small random noise
            if rng.random() < 0.2:
                noise = rng.normal(scale=0.01, size=n)
                child = np.maximum(child + noise, 0.0)
            # renormalize to sum=1
            s = child.sum()
            if s <= 0:
                child = _uniform(n)
                s = child.sum()
            child = child / s
            new_pop.append(child)
        # evaluate new population
        population = new_pop
        obj_vals = [_objective_exact(v) for v in population]
        # update best
        cur_best_idx = int(np.argmin(obj_vals))
        cur_best_val = obj_vals[cur_best_idx]
        if cur_best_val < best_val:
            best_val = cur_best_val
            best_seq = population[cur_best_idx].copy()
            print(f"[Genetic] new best {best_val:.6f} at gen {gen}")
        gen += 1
    return best_seq, best_val

"""
Propose candidate step‑function using a hybrid strategy:
1. Diverse analytic seeds → quick surrogate ranking.
2. Continuation L‑BFGS on a log‑sum‑exp smooth max‑convolution.
3. Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
4. Population‑based genetic search with cheap mass‑transfer mutations.
5. Final high‑β L‑BFGS polish.

The code respects the 1000 s budget, keeps the best vector in the global
name ``height_sequence_1`` and returns it as a plain list of floats.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)  # n, n‑1, …, 1
    s = a.sum()
    return a / s


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)  (i = 0 … n‑1)"""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)          # Bessel I0 (NumPy)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i = 1/(i+1)^p  (i = 0 … n‑1)"""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _chebyshev_window(n: int, attenuation: float = 100.0) -> np.ndarray:
    try:
        from scipy.signal import chebwin
        a = chebwin(n, at=attenuation)
    except Exception:
        a = np.ones(n, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _flattop_window(n: int) -> np.ndarray:
    try:
        from scipy.signal import get_window
        a = get_window('flattop', n, fftbins=False)
    except Exception:
        a = np.ones(n, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α) + b·α, renormalised to sum‑1."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_linear_decrease(n))

    for lam in (0.0005, 0.001, 0.002, 0.005, 0.01):
        seeds.append(_exponential_decay(n, lam))

    for g in (0.3, 0.5, 0.8, 1.0, 2.0):
        seeds.append(_sine_window(n, gamma=g))

    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian / double‑Gaussian
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))

    # Power‑law family
    for p in np.linspace(0.0, 4.0, 21):
        seeds.append(_power_law(n, p))

    # Beta family
    for a in np.linspace(0.5, 3.0, 7):
        for b in np.linspace(0.5, 3.0, 7):
            seeds.append(_beta_shape(n, a, b))

    # Window families with good sidelobe suppression
    seeds.append(_chebyshev_window(n, attenuation=100.0))
    seeds.append(_chebyshev_window(n, attenuation=80.0))
    seeds.append(_flattop_window(n))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Pairwise mixtures of already‑generated seeds
    base = np.arange(len(seeds))
    for _ in range(300):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Add previous solution if available
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Clean any numerical noise and enforce ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) helpers for continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Log‑sum‑exp surrogate of the max‑convolution.
    Returns (obj, grad) where grad is w.r.t. the logits v.
    """
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)               # full linear convolution
    shift = np.max(conv)                 # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t a
    w = exp_term / sum_exp               # soft‑max weights over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3·n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]   # slice of length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 2000,
) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    The variable is the vector of logits (log‑softmax).
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 100000.0,
        ]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during beta={beta_val:.0f}")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)   # final probability vector


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient (max‑convolution)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    Implements the derivative of the *single* lag attaining the maximum.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    # g_i = 2 * a_{idx-i}  for all i that contribute to this lag
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = idx - idx_i
    g[idx_i] = 2.0 * a[rev_idx]
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *exact* max‑convolution objective.
    Returns (best_a, best_obj).
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while it < max_iters:
        if time.time() > deadline:
            break
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 150_000 == 0:
            cur = _objective_exact(a)
            print(f"[ExpGrad] it={it:,} obj={cur:.6f}")
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
    # final check
    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple mass‑transfer mutation (used by the GA)
# ----------------------------------------------------------------------
def _mass_transfer_mutation(a: np.ndarray, step_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Move a small fraction of mass from a random source index to a random target."""
    a = a.copy()
    n = len(a)
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j or a[i] <= 0.0:
        return a
    delta = step_frac * a[i]
    a[i] -= delta
    a[j] += delta
    return a


# ----------------------------------------------------------------------
# Genetic optimisation (population‑based search)
# ----------------------------------------------------------------------
def _genetic_optimize(
    initial_seq: np.ndarray,
    n: int,
    deadline: float,
    pop_size: int = 30,
    elite_size: int = 5,
    step_frac: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Very cheap genetic algorithm:
    - keep elite members,
    - generate children as convex mixtures of elites,
    - mutate via mass‑transfer,
    - optional small Gaussian jitter.
    """
    rng = np.random.default_rng(12345)

    # initialise population
    population = [initial_seq.copy()]
    for _ in range(pop_size - 1):
        vec = rng.dirichlet(np.ones(n) * 0.5).astype(np.float64)
        population.append(vec)

    # evaluate
    obj_vals = [_objective_exact(v) for v in population]
    best_idx = int(np.argmin(obj_vals))
    best_seq = population[best_idx].copy()
    best_val = obj_vals[best_idx]

    generation = 0
    while time.time() < deadline:
        # sort and keep elites
        order = np.argsort(obj_vals)
        elites = [population[i] for i in order[:elite_size]]

        # produce offspring
        new_pop = elites.copy()
        for _ in range(pop_size - elite_size):
            p1, p2 = rng.choice(elites, size=2, replace=False)
            alpha = rng.random()
            child = (1.0 - alpha) * p1 + alpha * p2

            # mutation
            if rng.random() < 0.9:
                child = _mass_transfer_mutation(child, step_frac, rng)

            # occasional Gaussian jitter (keeps diversity)
            if rng.random() < 0.2:
                jitter = rng.normal(scale=0.01, size=n)
                child = np.maximum(child + jitter, 0.0)

            # renormalise to sum‑1
            s = child.sum()
            if s <= 0.0:
                child = _uniform(n)
                s = child.sum()
            child = child / s

            new_pop.append(child)

        population = new_pop
        obj_vals = [_objective_exact(v) for v in population]

        # track global best
        cur_best_idx = int(np.argmin(obj_vals))
        cur_best_val = obj_vals[cur_best_idx]
        if cur_best_val < best_val:
            best_val = cur_best_val
            best_seq = population[cur_best_idx].copy()
            print(f"[Genetic] new best {best_val:.6f} at generation {generation}")

        generation += 1

    return best_seq, best_val

            global_best_seq = seq_opt.copy()

            )
            print(f"[ExpGrad] best objective = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] New best after EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – genetic population search (cheap hill‑climbing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Genetic optimisation ---")
        seq_ga, val_ga = _genetic_optimize(
            global_best_seq,
            n,
            deadline,
            pop_size=30,

        )
        print(f"[Genetic] best objective = {val_ga:.6f}")
        if val_ga < global_best_val:

    # ------------------------------------------------------------------

            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polish = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest internal objective ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally for future warm‑starts
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
