"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a, a)) / (sum(a)²)

The algorithm works directly on the probability simplex (∑a_i = 1) using a
log‑softmax (logit) parametrisation.  A smooth approximation of the max
operator (log‑sum‑exp) is annealed from low to high inverse‑temperature β.
Gradient steps are taken with the Adam optimiser; the gradient of the
smoothed objective can be computed efficiently by one extra FFT per
iteration.  The procedure runs until the time budget expires, keeping the
best sequence seen (as measured by the exact surrogate 2·n·max(conv)).
Finally a short projected sub‑gradient polishing phase is applied.

The function returns the best sequence as a plain list of floats ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective helpers (surrogate works for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact surrogate) – used for polishing
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact surrogate."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1
        # compute convolution and locate the active index
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for that index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step‑size
        lr = 0.25 / math.sqrt(t)
        a -= lr * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if t % 200_000 == 0:
                print(f"[SubGrad] it={t:7d} best={best_val:.6f}")

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the given time budget.
    Returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()

    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Problem size (keep the original 4000 – good trade‑off)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Initialise from the previous best if available, otherwise uniform
    # --------------------------------------------------------------

        if np.sum(init_arr) > 0:

    logits = np.log(a + 1e-30)

    # Adam optimiser state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # keep the best sequence seen

    best_surrogate = _objective_surrogate(a)

    print(f"\n[Init] surrogate = {best_surrogate:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,

    total_iters = 0

    while time.time() < deadline and total_iters < max_total_iters:
        # decide current β
        beta_idx = min(total_iters // iters_per_beta,

        # 1) current probabilities

        #    w_i = exp(β·(conv_i – max_conv))  (already normalised)

        else:

        conv_rev = _conv_fft(w, a_rev)               # length 3n‑2

        # 5) gradient w.r.t. logits (Jacobian of softmax)
        grad_logits = a * (d - np.dot(d, a))

        # 6) Adam update on logits
        total_iters += 1
        t = total_iters
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # 7) evaluate surrogate (exact max‑conv) and keep best
        cur_surrogate = 2.0 * n * max_conv
        if cur_surrogate < best_surrogate:
            best_surrogate = cur_surrogate
            best_seq = a.copy()
            # occasional reporting
            if total_iters % 5000 == 0:
                print(f"[Adam] it={total_iters:7d} β={beta:.0f} "
                      f"surrogate={best_surrogate:.6f}")

        # stop early if we have exhausted the per‑β budget
        if (total_iters % iters_per_beta) == 0:
            # optional: shrink learning rate a little as β grows
            lr *= 0.95

    print(f"\n[Adam] finished after {total_iters} iterations "
          f"best surrogate = {best_surrogate:.6f}")

    # --------------------------------------------------------------
    # Final projected sub‑gradient polishing (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # estimate a safe number of sub‑gradient steps (≈200 iters/sec)
        max_sub_iters = int(min(800_000,
                                max(0, remaining * 200.0)))
        if max_sub_iters > 0:
            print("\n--- Sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_sub_iters)
            if val_sub < best_surrogate:
                best_surrogate = val_sub
                best_seq = seq_sub.copy()
                print(f"[SubGrad] improved to {best_surrogate:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_surrogate:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for potential reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
