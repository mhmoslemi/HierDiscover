assistantfinal**Solution Overview**

The score is  

\[
\text{score}= \frac{2\,n\;\max_k(a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

with a non‑negative vector `a`.  
The function is *scale invariant*, therefore we may always keep `∑a = 1`.  
The problem is to minimise the peak of the auto‑convolution `a*a` under the
simplex constraint.

The program follows three phases  

1. **Diverse seed generation** – analytic families (uniform, triangular,
   exponential, beta, Gaussian, windows …) and a few random draws.
   All seeds are ℓ₁‑normalised.

2. **Continuation L‑BFGS on a smooth surrogate** – the non‑smooth “max’’ is
   approximated by a log‑sum‑exp with an increasing temperature `β`.  The
   optimiser works in the log‑softmax (logit) space, guaranteeing
   non‑negativity and simplex normalisation.  We run a long schedule that
   reaches `β = 1e6`.

3. **Fine tuning** –  
   * exponentiated‑gradient (mirror descent) on the exact objective, and  
   * a cheap *incremental* local search that moves a small fraction of mass
     from one position to another while updating the convolution in O(n)
     time.  The move is accepted only if it improves the objective.
   This step massively explores the neighbourhood of the L‑BFGS solution
   and is able to squeeze out the last few hundredths of the score.

All steps respect the 1000 s time budget, regularly check the deadline and
stop early if needed.  The best vector found is stored globally as
`height_sequence_1` and returned as a plain list of floats.

```python
"""
search for a step‑function (non‑negative vector) that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm:
1. generate many analytic and random seeds (ℓ₁‑normalised)
2. pick the best seed (exact objective)
3. continuation L‑BFGS on a log‑sum‑exp surrogate with a long β‑schedule
4. mirror‑descent (exponentiated gradient) on the exact max‑convolution
5. cheap incremental local search that moves a tiny mass between two indices
   while updating the convolution in O(n) time.

All phases stop before the deadline.  The final vector is stored in the
global name ``height_sequence_1`` and returned as a plain list.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector on the simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families – all return ℓ₁‑normalised vectors (sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _linear_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)          # Bessel I0
    return a / a.sum()


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α) + b·α, re‑normalised to sum‑1."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_linear_decrease(n))
    for lam in (0.0005, 0.001, 0.002, 0.005, 0.01):
        seeds.append(_exponential_decay(n, lam))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian family
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))

    # power‑law family
    for p in np.linspace(0.0, 4.0, 21):
        seeds.append(_power_law(n, p))

    # beta family
    for a in np.linspace(0.5, 3.0, 7):
        for b in np.linspace(0.5, 3.0, 7):
            seeds.append(_beta_shape(n, a, b))

    # random draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # pairwise mixtures of the already generated seeds
    base = np.arange(len(seeds))
    for _ in range(300):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # add the previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # clean numerical noise, enforce ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.maximum(v, 0.0)
        s = v.sum()
        cleaned.append(v / s if s > 0 else _uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – objective + gradient for L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Returns (obj, grad) where grad is w.r.t. the log‑softmax variables v.
    The objective is a smooth approximation of 2·n·max(conv(a,a)).
    """
    a = _softmax(v)                      # a lives on the simplex
    conv = _conv_fft(a, a)                # full convolution
    shift = np.max(conv)                  # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    # surrogate = 2·n·(shift + log(sum_exp)/beta)
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t a (see derivation in the description)
    w = exp_term / sum_exp               # soft‑max weights over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3·n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice of length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 3000,
) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    Operates on the log‑softmax variables (unconstrained).
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # logits, never exactly zero

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during β={beta_val:.0f}")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} (β={beta_val})")
            break

    return _softmax(v)   # final probability vector (sum = 1)


# ----------------------------------------------------------------------
# Exact objective and its sub‑gradient (max‑convolution)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    Returns a vector g such that g_i = 2·a_{k-i} where k is the lag that
    attains the maximum (any one of the maximisers is fine).
    """
    n = len(a)
    conv = _conv_fft(a, a)
    k = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    rev = k - idx
    g[idx] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)                 # stays on the simplex

        if it % 250_000 == 0:
            cur = _objective_exact(a)
            print(f"[ExpGrad] it={it:,} obj={cur:.6f}")
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Incremental local search – O(n) update of the convolution
# ----------------------------------------------------------------------
def _apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float) -> None:
    """
    Move a mass ``delta`` from index i to index j (i ≠ j, delta > 0).
    The vector ``a`` and its full convolution ``conv`` are updated *in‑place*.
    ``conv`` has length 2·n‑1.
    """
    n = len(a)
    # cross‑terms (linear part)
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # quadratic part (delta with itself)
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    conv[i + j] -= 2.0 * delta * delta

    # finally move the mass
    a[i] -= delta
    a[j] += delta


def _incremental_local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    total_iters: int = 500_000,
    init_step: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing that moves a small fraction of mass between two
    random positions.  Updates the convolution in O(n) time, so millions of
    steps are affordable.
    """
    a = start_a.copy()
    conv = _conv_fft(a, a)          # initialise exact convolution
    sum_a = a.sum()
    best_obj = 2.0 * n * np.max(conv) / (sum_a * sum_a)
    best_a = a.copy()

    rng = np.random.default_rng(12345)
    step = init_step

    for it in range(1, total_iters + 1):
        if time.time() > deadline:
            break

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue

        # random fraction up to current step size
        delta = step * a[i] * rng.random()
        if delta <= 0.0:
            continue

        # apply candidate move
        _apply_move(a, conv, i, j, delta)

        cur_obj = 2.0 * n * np.max(conv) / (sum_a * sum_a)

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            # occasional progress report
            if it % 100_000 == 0:
                print(f"[Local] iter={it:,} best={best_obj:.6f}")
        else:
            # revert – the reverse move exactly undoes the change
            _apply_move(a, conv, j, i, delta)

        # slowly shrink the step size so we can fine‑tune at the end
        if it % 150_000 == 0:
            step = max(step * 0.5, 0.001)

        # every 200 k iterations recompute the convolution from scratch
        # to wipe out accumulated rounding error
        if it % 200_000 == 0:
            conv = _conv_fft(a, a)

    return best_a, best_obj


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative vector) that minimises the
    evaluation function.  The best vector found is stored globally as
    ``height_sequence_1`` and returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    n = 4000
    print(f"\n=== start search, n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # previous solution (warm‑start)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # seed generation & quick exact evaluation
    # ------------------------------------------------------------------
    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"generated {len(seeds)} seed vectors")

    evaluated = [(vec, _objective_exact(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    best_vec, best_val = evaluated[0]
    print(f"initial best objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – deep continuation L‑BFGS on the smooth surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # use a fairly long β‑schedule; allocate most of the remaining time here
        remaining = deadline - time.time()
        per_beta = int(min(4000, remaining * 0.3 / 16))   # 16 betas in schedule
        per_beta = max(800, per_beta)
        print(f"[Phase 1] L‑BFGS with per‑β iters ≈ {per_beta}")

        seq_lbfgs = _lbfgs_smooth_optimize(
            best_vec,
            n,
            deadline,
            beta_schedule=[
                1.0, 5.0, 20.0, 80.0, 200.0,
                500.0, 1_000.0, 2_000.0, 5_000.0,
                10_000.0, 20_000.0, 50_000.0,
                100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            ],
            max_iter_per_beta=per_beta,
        )
        val_lbfgs = _objective_exact(seq_lbfgs)
        print(f"[Phase 1] L‑BFGS objective = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:
            best_vec, best_val = seq_lbfgs, val_lbfgs

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient on the exact objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(8_000_000, remaining * 5_000))   # rough 5 k iters per sec
        print(f"[Phase 2] Exp‑gradient with up to {max_iters:,} iters")
        seq_exp, val_exp = _exp_gradient_optimize(
            best_vec,
            n,
            deadline,
            max_iters=max_iters,
            eta0=1.0,
        )
        print(f"[Phase 2] Exp‑grad objective = {val_exp:.6f}")
        if val_exp < best_val:
            best_vec, best_val = seq_exp, val_exp

    # ------------------------------------------------------------------
    # Phase 3 – massive incremental local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(2_000_000, remaining * 30_000))  # a few million cheap steps
        print(f"[Phase 3] Incremental local search – {iters:,} iterations")
        seq_loc, val_loc = _incremental_local_search(
            best_vec,
            n,
            deadline,
            total_iters=iters,
            init_step=0.02,
        )
        print(f"[Phase 3] Local‑search objective = {val_loc:.6f}")
        if val_loc < best_val:
            best_vec, best_val = seq_loc, val_loc

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_vec.tolist())
        print(f"\n=== finished ===\nobjective (surrogate) = {best_val:.6f}")
        print(f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for warm‑starts in later runs
    globals()["height_sequence_1"] = best_vec.tolist()
    return best_vec.tolist()
```
