"""
Search for a non‑negative step‑function that minimises the
evaluate_sequence objective.

The algorithm:
1. Generate a diverse pool of ℓ₁‑normalised seeds (analytic families,
   Dirichlet draws, permutations) and keep the best few according to
   the exact surrogate 2·n·max(conv).
2. Initialise the optimisation with the best seed (or the previous
   global solution if present) and run a long mirror‑descent (exponentiated‑gradient)
   on the exact max‑convolution.  The update is
        a ← a ⊙ exp(−η_t·g)   with η_t = η₀ / √t,
   where g is the sub‑gradient of max(conv) at the current active index.
   The iterate is renormalised to stay on the probability simplex.
   The loop runs until the time budget expires, constantly tracking the
   best surrogate value.
3. A final call to ``evaluate_sequence`` returns the true objective.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Helper to build a rich seed pool
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.12, 0.18, 0.25):
        seeds.append(_gaussian(n, sigma))

    for p in (0.0, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))

    # Random Dirichlet draws
    concs = (0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    for conc in concs:
        for _ in range(5):
            v = _random_dirichlet(n, conc, rng)
            seeds.append(v)
            seeds.append(np.sort(v)[::-1])                # descending
            # random permutation
            seeds.append(v[rng.permutation(n)])

    # Previous solution, if any, re‑normalised to ℓ₁
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Normalise robustly, guard against NaNs / zeros
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0 and np.all(np.isfinite(v)):
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised sequence."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑conv
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Run a long exponentiated‑gradient descent on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_a = a.copy()
    best_val = _surrogate(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()
    # parameters for stagnation handling
    stagnation_limit = 200_000   # iterations without improvement before noise

    while True:
        if time.time() > deadline:
            break
        it += 1

        # compute full convolution
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # build sub‑gradient for the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size schedule η_t = η₀ / √t
        eta = eta0 / math.sqrt(it)

        # mirror‑descent update in the probability simplex
        a = a * np.exp(-eta * g)
        total = a.sum()
        if total == 0.0 or not np.isfinite(total):
            # fallback to uniform distribution
            a = _uniform(n)
        else:
            a = a / total

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = it
            # occasional progress output
            if it % 100_000 == 0:
                print(f"[ExpGrad] iter={it:,} new best={best_val:.6f}")

        # occasional stagnation escape
        if it - last_improve >= stagnation_limit:
            # inject a small random perturbation around the current best
            noise = rng.normal(scale=0.02, size=n)  # 2 % Gaussian noise
            a = best_a * (1.0 + noise)
            a = np.clip(a, 0.0, None)
            s = a.sum()
            a = a / s if s > 0 else _uniform(n)
            last_improve = it
            # reset step‑size schedule by pretending we are at start again
            it = 0
            print(f"[ExpGrad] stagnation → perturbation; restart counter")

        # lightweight progress display
        if it % 250_000 == 0:
            print(f"[ExpGrad] iter={it:,} best={best_val:.6f}")

    return best_a, best_val

    # ------------------------------------------------------------------

    rng = np.random.default_rng(seed)

    # Choose a problem size – 3000 gives a good trade‑off between
    # resolution and per‑iteration speed.
    # ------------------------------------------------------------------
    n = 3000

    # Assemble a rich collection of ℓ₁‑normalised seeds
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} raw seeds.")

    # Evaluate the cheap surrogate (exact for ℓ₁‑normalised vectors)
    evaluated = [(vec, _surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Initialise optimisation with the best seed (or with the previous run)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _surrogate(global_best_seq)

    # If a previous solution exists and is better, prefer it
    if prev_arr is not None:
        prev_val = _surrogate(prev_arr / prev_arr.sum())
        if prev_val < global_best_val:

"""
Search for a non‑negative step‑function that minimises the
evaluate_sequence objective.

The algorithm:
1. Build a large pool of ℓ₁‑normalised seeds (uniform, triangular,
   Gaussian, power‑law, Dirichlet draws, permutations and a possible
   previous solution).  The exact surrogate 2·n·max(conv) is cheap
   to compute; the best few seeds are retained.
2. Starting from the best seed (or the previous run) a long
   exponentiated‑gradient (mirror‑descent) is performed on the exact
   max‑convolution.  The update is
        a ← a ⊙ exp(−ηₜ·g)   with ηₜ = η₀/√t,
   where g is the sub‑gradient of the active max entry.
   The iterate is renormalised after each step, stays on the probability
   simplex and is kept if it improves the surrogate.  A simple
   stagnation detector injects a small random perturbation when no
   improvement occurs for many iterations.
3. As soon as the time budget expires (or a target surrogate ≤ 1.503 is
   reached) the best sequence seen is evaluated with the official
   ``evaluate_sequence`` function and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.12, 0.18, 0.25):
        seeds.append(_gaussian(n, sigma))
    for p in (0.0, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))

    # random Dirichlet draws + simple permutations
    concs = (0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    for conc in concs:
        for _ in range(5):
            v = _random_dirichlet(n, conc, rng)
            seeds.append(v)                                    # raw
            seeds.append(np.sort(v)[::-1])                     # descending
            seeds.append(v[rng.permutation(n)])                # random perm

    # previous solution, if any
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # robust normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0 and np.all(np.isfinite(v)):
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised sequence."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 0.5,
                      target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Long mirror‑descent (exponentiated gradient) on the exact max‑conv.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_a = a.copy()
    best_val = _surrogate(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()
    stagnation_limit = 200_000          # steps without improvement → perturb
    while True:
        if time.time() > deadline:
            break
        it += 1

        # full convolution and active index
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent step in the simplex (exponentiated‑gradient)
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * g)
        total = a.sum()
        if total == 0.0 or not np.isfinite(total):
            a = _uniform(n)
        else:
            a = a / total

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = it
            if best_val <= target:
                print(f"[ExpGrad] target reached (surrogate ≤ {target})")
                break
            if it % 100_000 == 0:
                print(f"[ExpGrad] iter={it:,} new best={best_val:.6f}")

        # stagnation handling – small random perturbation around the current best
        if it - last_improve >= stagnation_limit:
            noise = rng.normal(scale=0.02, size=n)          # ≈2 % Gaussian noise
            a = best_a * (1.0 + noise)
            a = np.clip(a, 0.0, None)
            s = a.sum()
            a = a / s if s > 0 else _uniform(n)
            it = 0                     # reset the step‑size schedule
            last_improve = 0
            print("[ExpGrad] stagnation – perturbation injected, restart counter")

        if it % 250_000 == 0:
            print(f"[ExpGrad] iter={it:,} best={best_val:.6f}")

    return best_a, best_val
