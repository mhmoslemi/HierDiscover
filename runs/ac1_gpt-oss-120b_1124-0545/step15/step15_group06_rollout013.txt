assistantfinal**Solution Overview**

The function `propose_candidate` builds a step‑function (a non‑negative vector that sums to 1) whose
evaluation value  

\[
\frac{2\,n\;\max (a*a)}{(\sum a)^2}=2n\;\max (a*a)
\]

is as small as possible.  
The optimisation proceeds in three phases  

1. **Diversified seeding** – a large pool of analytic families (uniform, β‑U‑shapes,
   Dirichlet draws, etc.) is generated and the best few are kept.  
2. **Mirror‑descent with Adam** – each seed is refined by an Adam‑style
   exponentiated‑gradient (mirrored on the probability simplex).  
   The sub‑gradient of the *exact* max‑convolution is employed, so the
   optimisation works directly on the true objective.  
   Restarts and random perturbations keep the search from getting stuck.  
3. **High‑β L‑BFGS polishing** – a short L‑BFGS run on the smooth
   log‑sum‑exp surrogate (very large β) removes any remaining
   irregularities.

All work respects the 1000 s budget, uses only two CPU cores and never
calls `evaluate_sequence` inside the inner loops (a fast FFT‑based
implementation is used instead).  The best vector found is stored in the
global name `height_sequence_1` for warm‑starts on subsequent calls.

```python
"""Optimise a non‑negative step function to minimise
   2·n·max(conv(a,a)) / (sum(a)²).

   Strategy
   ----------
   1. Build a large diverse pool of ℓ₁‑normalised seeds
      (uniform, β‑U‑shapes, Dirichlet draws, etc.).
   2. Refine each seed with a mirror‑descent / Adam optimiser that
      works on the *exact* max‑convolution objective (sub‑gradient of the
      active lag, exponentiated‑gradient on the softmax logits).
   3. Occasionally restart from the current best, add jitter and repeat.
   4. Finish with a high‑β L‑BFGS run on the smooth log‑sum‑exp surrogate.
   5. Return the best sequence found; store it globally for warm‑starts.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on {1,…,n} (U‑shape for α,β<1)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exact objective utilities (fast, no security checks)
# ----------------------------------------------------------------------
def _objective_fast(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv) for a already ℓ₁‑normalised."""
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (using the active lag).
    Returns g = ∂ max(conv) / ∂ a .
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    g[i_range] = 2.0 * a[rev]
    return g


# ----------------------------------------------------------------------
# Adam style mirror‑descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _adam_optimize(start_a: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iter: int = 200_000,
                   lr: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam on logits (v = log a).  Uses the exact
    sub‑gradient of the max‑convolution.  Returns the best vector seen.
    """
    eps = 1e-30
    a = start_a.copy()
    v = np.log(a + eps)                     # initialise logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    best_a = a.copy()
    best_val = _objective_fast(a, n)

    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        # sub‑gradient of the exact max‑convolution
        g_a = _subgradient_maxconv(a)                     # ∂max/∂a
        # convert to gradient w.r.t. logits (entropy mirror map)
        grad_v = a * (g_a - np.dot(g_a, a))

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

        a = _softmax(v)

        # occasional check for improvement
        if t % 2000 == 0:
            cur = _objective_fast(a, n)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[Adam] it={t:,} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth surrogate & L‑BFGS (high‑β polishing)
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Value and gradient of the log‑sum‑exp surrogate."""
    a = _softmax(v)                                 # probability simplex
    conv = _conv_fft(a, a)                          # full auto‑convolution
    shift = np.max(conv)
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # weights for each lag
    w = exp_term / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                      # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    L‑BFGS‑B optimisation of the smooth surrogate.
    Variable = logits of a softmax (unconstrained real vector).
    """
    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0,
                         5000.0, 10000.0, 20000.0,
                         50000.0, 100000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during beta={beta_val}")
            break
        except Exception as e:               # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Seed generation (wide variety)
# ----------------------------------------------------------------------
def _collect_initial_seeds(n: int,
                          rng: np.random.Generator,
                          previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised seed vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families -------------------------------------------------
    seeds.append(_uniform(n))

    # β‑U‑shapes – dense grid in (0, 2] × (0, 2]
    alphas = np.linspace(0.05, 2.0, 9)
    betas = np.linspace(0.05, 2.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws with many concentrations
    for conc in (0.01, 0.02, 0.05, 0.1, 0.2, 0.5):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Previous best (warm‑start)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Clean‑up – enforce non‑negativity & ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best list of non‑negative floats found before the time
    budget expires.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    n = int(kwargs.get("n", 4000))
    print(f"\n=== Optimisation start (n={n}) ===")

    # ------------------------------------------------------------------
    # 0. Seeding
    # ------------------------------------------------------------------
    prev_global = globals().get("height_sequence_1")
    prev_arr = None
    if prev_global is not None:
        try:
            prev_arr = np.asarray(prev_global, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_initial_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors (incl. warm‑start).")

    # Evaluate seeds with the *fast* objective (exact same value as the
    # official evaluator for ℓ₁‑normalised vectors).
    best_seq = None
    best_val = float('inf')
    for s in seeds:
        val = _objective_fast(s, n)
        if val < best_val:
            best_val = val
            best_seq = s.copy()
    print(f"[Seed] best objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Adam‑mirrored descent with restarts
    # ------------------------------------------------------------------
    restart_counter = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        # decide whether to start from a fresh Dirichlet or perturb the current best
        if rng.random() < 0.6:
            conc = rng.uniform(0.01, 0.5)
            a0 = _random_dirichlet(n, conc, rng)
        else:
            # jitter the logits of the current best
            logit = np.log(best_seq + 1e-30)
            jitter = rng.normal(scale=0.2, size=n)
            a0 = _softmax(logit + jitter)

        # run the Adam optimiser
        seq_opt, val_opt = _adam_optimize(a0, n, deadline,
                                          max_iter=200_000,
                                          lr=1.0)

        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt.copy()
            print(f"[Restart #{restart_counter}] new best = {best_val:.6f}")

        restart_counter += 1
        if remaining < 5.0:               # leave a tiny buffer for polishing
            break

    # ------------------------------------------------------------------
    # 2. Final high‑β L‑BFGS polishing (smooth surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- High‑β L‑BFGS polishing ---")
        try:
            polished = _lbfgs_smooth_optimize(best_seq,
                                              n,
                                              deadline,
                                              beta_schedule=[5000.0, 10000.0,
                                                             20000.0, 50000.0,
                                                             100000.0],
                                              max_iter_per_beta=3000)
            polished_val = _objective_fast(polished, n)
            if polished_val < best_val:
                best_val = polished_val
                best_seq = polished.copy()
                print(f"[Polish] improved to {best_val:.6f}")
        except Exception as exc:          # pragma: no cover
            print("Polish raised an exception:", exc)

    # ------------------------------------------------------------------
    # 3. Final official evaluation & store result
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest fast objective ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:              # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```
