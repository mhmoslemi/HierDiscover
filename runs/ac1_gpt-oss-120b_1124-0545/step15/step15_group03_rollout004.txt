def _smooth_max_opt(start_seq, n, deadline, max_iters=200_000, alpha_start=5.0, alpha_end=200.0):

   a = start_seq.copy()
   log_a = np.log(a + 1e-30)  # initial logits
   best_seq = a.copy()
   best_val = _objective_surrogate(a) # actual surrogate

   for it in range(1, max_iters+1):
        if time.time() > deadline: break

        # linear schedule for alpha
        alpha = alpha_start + (alpha_end - alpha_start) * (it-1) / (max_iters-1)

        conv = _conv_fft(a, a)
        # compute p = softmax(alpha * conv)
        scaled = alpha * conv
        m = np.max(scaled)
        exp_scaled = np.exp(scaled - m)
        sum_exp = np.sum(exp_scaled)
        p = exp_scaled / sum_exp

        # compute gradient: g = 2 * conv(p, a_rev)[n-1:n-1+n]
        a_rev = a[::-1]
        grad_full = _conv_fft(p, a_rev)
        g = 2.0 * grad_full[n-1:n-1+n]  # slice length n

        # update log_a via EG
        eta = 0.1 / math.sqrt(it)  # step size parameter; maybe 0.2
        log_a = log_a - eta * g
        a = _softmax(log_a)

        # track best based on true surrogate
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # optionally print progress
            if it % 5000 ==0:
                print(f"[Smooth] it={it:6d} alpha={alpha:.1f} best={best_val:.6f}")

   return best_seq, best_val

"""Candidate proposer for step-function minimisation.

This implementation builds on the earlier exponentiated‑gradient method,
but adds a *smooth‑max* refinement phase.  The smooth max replaces the
hard ``max(conv)`` by a log‑sum‑exp approximation, yielding a differentiable
objective whose gradient is cheap to compute via two FFT‑based
convolutions.  The algorithm proceeds as

1. Generate a large, diverse set of ℓ₁‑normalised seeds (uniform,
   beta‑shapes, power‑law, Gaussian, random Dirichlet, sparse combs,
   two‑level mixtures).

2. Keep the best few seeds w.r.t. the exact surrogate ``2·n·max(conv)``.

3. Run a fast exponentiated‑gradient optimiser on each top seed
   (the classic sub‑gradient of the active convolution entry).

4. Refine the current global best with the *smooth‑max* exponentiated‑gradient
   optimiser, gradually sharpening the approximation (α from 5 to 200).

5. Optional sub‑gradient polishing and a short simulated‑annealing run.

All steps are time‑aware and respect the 1000 s wall‑clock budget.
The final sequence is stored in ``height_sequence_1`` and returned as a
plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_k_uniform(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=min(k, n), replace=False)
    a[idx] = 1.0 / len(idx)
    return a


def _two_level_mixture(n: int, high_frac: float, high_val: float, rng: np.random.Generator) -> np.ndarray:
    """
    Construct a mixture where a fraction `high_frac` of indices receive a
    larger weight `high_val`, the remaining indices share the rest uniformly.
    The vector is renormalised to ℓ₁ = 1.
    """
    a = np.full(n, (1.0 - high_frac) / (n - int(high_frac * n)), dtype=np.float64)
    # pick high_frac proportion of positions
    k = max(1, int(high_frac * n))
    idx = rng.choice(n, size=k, replace=False)
    a[idx] = high_val
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient (mirror‑descent) using the true max‑convolution sub‑gradient.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _smooth_max_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 80_000,
                    alpha_start: float = 5.0,
                    alpha_end: float = 200.0,
                    eta0: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimiser on a smooth approximation of the max‑convolution.
    The smooth objective is   2·n·log‑sum‑exp(α·conv) / α .
    α is increased linearly from ``alpha_start`` to ``alpha_end``.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linearly schedule α
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / (max_iters - 1)

        conv = _conv_fft(a, a)                # length 2n‑1
        # softmax over α·conv (stable)
        scaled = alpha * conv
        m = np.max(scaled)
        exp_scaled = np.exp(scaled - m)
        sum_exp = np.sum(exp_scaled)
        p = exp_scaled / sum_exp              # probability vector, sums to 1

        # gradient = 2 * (p * a) convolution
        a_rev = a[::-1]
        grad_full = _conv_fft(p, a_rev)      # length 3n‑2
        g = 2.0 * grad_full[n - 1:n - 1 + n]  # slice of length n

        # EG update in log‑space
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Smooth] it={it:5d} α={alpha:.1f} best={best_val:.6f}")

    return best_seq, best_val


def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Simple SA in log‑softmax (logit) space, short duration.
    """
    rng = np.random.default_rng(12345)
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.06 * (1.0 - progress) + 1e-7
        sigma = 0.20 * (1.0 - progress) + 1e-5

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 30_000 == 0:
            print(f"[SA] it={it:6d} best={best_val:.6f}")

    return best_seq, best_val

    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle; sigma = sigma_frac·n."""
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_k_uniform(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform distribution on a random subset of size k."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=min(k, n), replace=False)
    a[idx] = 1.0 / len(idx)
    return a


def _two_level_mixture(n: int, high_frac: float, high_val: float,
                       rng: np.random.Generator) -> np.ndarray:
    """
    Mixture: a fraction `high_frac` of indices receive a larger weight `high_val`,
    the remainder share the rest uniformly.  The vector is ℓ₁‑normalised.
    """
    a = np.full(n, (1.0 - high_frac) / (n - int(high_frac * n)), dtype=np.float64)
    k = max(1, int(high_frac * n))
    idx = rng.choice(n, size=k, replace=False)
    a[idx] = high_val
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Optimisers

def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient (mirror‑descent) optimiser using the true
    max‑convolution sub‑gradient.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _smooth_max_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 80_000,
                    alpha_start: float = 5.0,
                    alpha_end: float = 200.0,
                    eta0: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimiser on the smooth approximation

        2·n·log‑sum‑exp(α·conv) / α

    where α is increased linearly from ``alpha_start`` to ``alpha_end``.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear schedule for α
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / (max_iters - 1)

        conv = _conv_fft(a, a)               # length 2n‑1
        # stable softmax over α·conv
        scaled = alpha * conv
        m = np.max(scaled)
        exp_scaled = np.exp(scaled - m)
        sum_exp = np.sum(exp_scaled)
        p = exp_scaled / sum_exp            # probability vector, sum(p)=1

        # gradient = 2 * (p convolve a_rev)
        a_rev = a[::-1]
        grad_full = _conv_fft(p, a_rev)     # length 3n‑2
        g = 2.0 * grad_full[n - 1:n - 1 + n]  # slice of length n

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Smooth] it={it:5d} α={alpha:.1f} best={best_val:.6f}")

    return best_seq, best_val


def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Simple SA in log‑softmax (logit) space, short duration.
    """
    rng = np.random.default_rng(12345)
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.06 * (1.0 - progress) + 1e-7
        sigma = 0.20 * (1.0 - progress) + 1e-5

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 30_000 == 0:
            print(f"[SA] it={it:6d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best ℓ₁‑normalised vector (list of floats) and stores it
    in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a balance between quality and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Recover previous solution if available.
    # ------------------------------------------------------------------
    previous_vec = None
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                previous_vec = arr / arr.sum()
        except Exception:
            previous_vec = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation.
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    if previous_vec is not None:
        seeds.append(previous_vec)

    # beta‑shape grid (coarse)
    for a in np.linspace(0.2, 5.0, 9):
        for b in np.linspace(0.2, 5.0, 9):
            seeds.append(_beta_shape(n, a, b))

    # random beta‑shape draws
    for _ in range(80):
        a = 10 ** rng.uniform(-2, 2)
        b = 10 ** rng.uniform(-2, 2)
        seeds.append(_beta_shape(n, a, b))

    # power‑law grid + random
    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))
    for _ in range(40):
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

    # Gaussian family
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    for _ in range(25):
        sigma = rng.uniform(0.05, 0.55)
        seeds.append(_gaussian(n, sigma))

    # Sparse k‑point uniform
    for k in (10, 20, 50, 100):
        seeds.append(_random_k_uniform(n, k, rng))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            if time.time() > deadline:
                break
            seeds.append(_random_dirichlet(n, conc, rng))

    # Two‑level mixtures (few high‑weight spikes)
    for high_frac in (0.01, 0.02, 0.05):
        for high_val in (0.2, 0.3, 0.5):
            seeds.append(_two_level_mixture(n, high_frac, high_val, rng))

    # Normalise (safe) and drop degenerate vectors
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        total = s.sum()
        if total <= 0.0:
            continue
        normed_seeds.append(s / total)

    print(f"\n--- Seed generation complete: {len(normed_seeds)} candidates ---")

    # ------------------------------------------------------------------
    # 2. Coarse ranking – keep the best few seeds.
    # ------------------------------------------------------------------
    scored = []
    for vec in normed_seeds:
        val = _objective_surrogate(vec)
        scored.append((vec, val))
    scored.sort(key=lambda x: x[1])
    top_k = 7
    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Best surrogate among raw seeds = {scored[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 3. Exponentiated‑gradient optimisation on the top seeds.
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing EG runs.")
            break

        remaining_sec = max(0.0, deadline - time.time())
        est_iters = int(min(550_000, remaining_sec * 2000))
        if est_iters < 30_000:
            est_iters = int(min(150_000, remaining_sec * 2000))
        if est_iters <= 0:
            break

        print(f"\n--- Exponentiated‑gradient run #{idx+1} (max_iters≈{est_iters}) ---")
        seq_opt, val_opt = _exp_gradient_opt(seed_vec, n, deadline,
                                             max_iters=est_iters)

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth‑max refinement (the new ingredient).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining_sec = deadline - time.time()
        max_smooth_iters = int(min(120_000, remaining_sec * 500))
        if max_smooth_iters > 10_000:
            print("\n--- Smooth‑max refinement ---")
            seq_smooth, val_smooth = _smooth_max_opt(global_best_seq, n,
                                                     deadline,
                                                     max_iters=max_smooth_iters,
                                                     alpha_start=5.0,
                                                     alpha_end=200.0,
                                                     eta0=0.15)
            if val_smooth < global_best_val:
                global_best_seq = seq_smooth.copy()
                global_best_val = val_smooth
                print(f"[Smooth] improved global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Sub‑gradient polishing (optional).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining_sec = deadline - time.time()
        max_sub = int(min(600_000, remaining_sec * 0.5))
        if max_sub > 0:
            print("\n--- Sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq, n,
                                                deadline, max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved best to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing push (short).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining_sec = deadline - time.time()
        max_sa = int(min(200_000, remaining_sec * 0.3))
        if max_sa > 0:
            print("\n--- Simulated‑annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing(global_best_seq, n,
                                                  deadline, max_iters=max_sa)
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[SA] final best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
