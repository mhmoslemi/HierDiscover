def _solve_lp_max_sum_given_conv_bound(f, M):
    n = len(f)
    # Build A_ub and b_ub for constraints conv_f_g <= M
    # Use sparse matrix via scipy.sparse.coo_matrix
    rows = []
    cols = []
    data = []
    for i in range(n):
        fi = f[i]
        # g index j from 0..n-1 contributes to constraint at row k = i + j
        # row indices range i ... i + n-1
        # For each j, row = i + j
        j_range = np.arange(n, dtype=np.int64)
        rows.extend(i + j_range)
        cols.extend(j_range)
        data.extend([fi] * n)
    # Convert to arrays

    cols = np.array(cols, dtype=np.int32)
    data = np.array(data, dtype=np.float64)

def _lp_max_sum_bound(f, M):
    n = len(f)
    import scipy.optimize as opt
    import scipy.sparse as sp
    offsets = np.arange(n)
    data = np.vstack([np.full(n, val) for val in f])   # shape (n, n)
    A = sp.diags(data, offsets=offsets, shape=(2*n-1, n), format='csr')
    b = np.full(2*n-1, M)
    c = -np.ones(n)   # maximize sum => minimize negative sum
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    else:
        print("LP not solved:", res.message)
        return None
```

"""
Propose a better step‑function.

Algorithm
---------
1. Initialise a candidate (uniform, previous solution, or a random Dirichlet draw)
   with ℓ₁‑normalisation (∑a_i = 1).
2. Apply a few iterations of smooth‑max L‑BFGS (the “log‑sum‑exp” surrogate) to
   reduce the peak of the self‑convolution.
3. Perform a Frank‑Wolfe–style update:
   * compute M = max(f * f)
   * solve a linear programme  max ∑g_i   s.t.   (f * g)_k ≤ M  (∀k) ,  g_i ≥ 0
   * scale the LP‑solution to a probability vector and move a fraction t towards
     it:  f ← (1‑t) f + t ĝ .
   * repeat while time permits.
4. One final short L‑BFGS polish.
5. Return the best sequence (list of floats).  The sequence is stored in the
   global variable ``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (required for L‑BFGS and LP)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
except Exception:  # pragma: no cover
    _opt = None
    _sp = None

# ----------------------------------------------------------------------
# Numeric utilities (FFT‑based convolution, softmax, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helper (exact surrogate for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for an ℓ₁‑normalised vector: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """Run continuation L‑BFGS on the log‑sum‑exp surrogate."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0, 10000.0, 20000.0]

    v = np.log(start_vec + 1e-30)                     # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on simplex
            conv = _conv_fft(a, a)                     # full linear convolution
            max_c = np.max(conv)
            # stabilize exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                      # weights over conv entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Linear‑programming sub‑step (maximise ∑g_i subject to f∗g ≤ M)
# ----------------------------------------------------------------------
def _lp_max_sum_bound(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max ∑ g_i   s.t.   (f * g)_k ≤ M   for all k,
    g_i ≥ 0.
    Returns the optimal g (may be non‑normalised) or None on failure.
    """
    if _opt is None or _sp is None:
        return None

    n = len(f)

    # Build the Toeplitz‑style constraint matrix A where
    #   (A @ g)_k = sum_i f_i * g_{k-i}  (with out‑of‑range indices ignored)
    offsets = np.arange(n, dtype=np.int64)
    data = [_sp.csr_matrix(np.full((n, ), val, dtype=np.float64))
            for val in f]                     # one constant diagonal per f_i
    A = _sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    b = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)          # maximise sum → minimise negative sum
    bounds = [(0.0, None)] * n

    # The HiGHS solver accepts sparse matrices.
    try:
        res = _opt.linprog(c, A_ub=A, b_ub=b,
                           bounds=bounds, method='highs',
                           options={'presolve': True})
    except Exception as e:  # pragma: no cover
        print("[LP] Linprog exception:", e)
        return None

    if res.success:
        return res.x
    else:
        print("[LP] failed:", res.message)
        return None


# ----------------------------------------------------------------------
# Helper for adjusting a previous solution to a new length
# ----------------------------------------------------------------------
def _adjust_length(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Resample a 1‑D array to a new length using linear interpolation."""
    if arr.size == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, arr.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    return np.interp(x_new, x_old, arr)

    while time.time() < deadline and lp_iters < max_lp_iters:
        M = np.max(_conv_fft(f, f))
        g = _lp_max_sum_bound(f, M)
        if g is None:
            print("[LP] no solution – breaking.")
            break
        sum_g = g.sum()
        if sum_g <= 0:
            print("[LP] got zero sum – breaking.")
            break

        # Scale to a probability vector (ℓ₁‑normalised)
        g_norm = g / sum_g

        # Convex blend – a modest step size works well
        t_step = 0.3
        f_new = (1.0 - t_step) * f + t_step * g_norm
        f_new = f_new / f_new.sum()               # re‑normalise (should stay ≈1)

        new_val = _objective_surrogate(f_new)
        print(f"[LP iter {lp_iters + 1}] surrogate = {new_val:.6f}")

        if new_val < best_val:
            f = f_new
            best_val = new_val
            print(f"[Update] new best surrogate = {best_val:.6f}")
        else:
            # no improvement – reduce step size
            t_step *= 0.5
            if t_step < 1e-4:
                print("[LP] step size too small – terminating LP loop.")
                break
        lp_iters += 1

    # ------------------------------------------------------------------
    # 5. Final L‑BFGS refinement (short, high‑β schedule)
    # ------------------------------------------------------------------
    if time.time() < deadline and _opt is not None:
        f = _lbfgs_smooth_optimize(f,
                                   n,
                                   deadline,
                                   beta_schedule=[1.0, 5.0, 20.0, 80.0, 200.0,
                                                  500.0, 1000.0, 2000.0,
                                                  5000.0, 10000.0, 20000.0],
                                   max_iter_per_beta=6000)
        best_val = _objective_surrogate(f)
        print(f"[Final L‑BFGS] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. True evaluation and return
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(f.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
          f"true evaluate_sequence = {true_val:.6f}")

    globals()["height_sequence_1"] = f.tolist()
    return f.tolist()
