def _maxconv_and_gradient(a, beta):
    n = len(a)
    conv = _conv_fft(a, a)
    maxcov = conv.max()
    # compute soft weights
    # shift conv for numeric stabilization
    base = maxcov
    diff = conv - base
    # If beta large, diff may be large negative leading to underflow to 0; but that okay.
    # compute exponentials
    exp = np.exp(beta * diff)
    sumexp = exp.sum()
    w = exp / sumexp

    # compute gradient
    a_rev = a[::-1]
    conv_w = _conv_fft(a_rev, w)
    # conv_w length = len(a_rev) + len(w) - 1 = n + (2n-1) -1 = 3n-2
    # slice
    start = n - 1
    grad = 2.0 * conv_w[start : start + n]
    return maxcov, grad

def _smooth_max_opt(start_seq, n, deadline, total_iters, beta_start, beta_end):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2 * n * np.max(_conv_fft(a, a)) # exact objective

    # schedule for beta: geometric progression per iteration? Could linearly increase after certain steps.
    # For simplicity, compute per iteration: beta = beta_start * (beta_end / beta_start) ** (it / total_iters)
    beta_start_f = float(beta_start)
    beta_end_f = float(beta_end)
    log_ratio = math.log(beta_end_f / beta_start_f) if beta_start_f > 0 else 0

    # using Adam hyperparams
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for it in range(1, total_iters + 1):
        if time.time() > deadline:
            break

        beta = beta_start_f * math.exp(log_ratio * (it - 1) / total_iters)  # geometric increase

        # compute gradient via smooth max
        maxcov, g = _maxconv_and_gradient(a, beta)

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        # step size schedule: start lr maybe 0.5, scaled by 1/sqrt(it)
        lr = 0.5 / math.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _proj_simplex(a)

        cur_max = maxcov  # but using previous iteration's max; we can recompute exact for best check
        cur_val = 2 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # printing progress
            if it % 50000 == 0:
                print(f"[SmoothOpt] it={it} beta={beta:.1f} best={best_val:.6f}")

    return best_seq, best_val

"""
Propose a candidate step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm improves on the previous solution by:
* Using a *smooth* approximation of the max‑convolution (log‑sum‑exp)
  together with a continuation on the smoothing parameter β.
* Performing Adam‑style projected gradient descent in the simplex
  (ℓ₁‑normalisation) – the gradient of the smooth max can be computed
  efficiently with two FFT‑based convolutions.
* Starting from a diverse set of analytical seeds, selecting the best few,
  and then refining each with the smooth optimiser.
* A final short sub‑gradient polishing pass (as in the reference code) to
  squeeze out any remaining improvement.

The whole search is time‑aware and stops before the 1000 s budget.
The best sequence found is stored in the global variable
``height_sequence_1`` and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (O(N log N)).
    Returns a real array of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w | w_i ≥ 0, sum w_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate = 2·n·max(conv) for a sequence already ℓ₁‑normalised.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle; sigma = sigma_frac·n."""
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_k_uniform(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform distribution on a random subset of size k."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=min(k, n), replace=False)
    a[idx] = 1.0 / len(idx)
    return a


# ----------------------------------------------------------------------
# Smooth‑max utilities
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Returns (max_conv, gradient) where the objective is the
    smoothed max: f(a) = (1/β)·log Σ exp(β·conv_k).
    Gradient ∂f/∂a = 2·(w * a_rev)[center : center+n],
    where w are the softmax weights of β·conv.
    """
    n = a.shape[0]

    # 1. Linear convolution a * a
    conv = _conv_fft(a, a)                     # length 2n-1
    max_conv = float(conv.max())

    # 2. Softmax weights for smooth max
    #    w_k = exp(β·(conv_k - max_conv)) / Σ exp(...)
    diff = conv - max_conv                    # max at 0, others ≤ 0
    # Numerical stability:
    exp_term = np.exp(beta * diff)
    sum_exp = exp_term.sum()
    if sum_exp == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = exp_term / sum_exp

    # 3. Gradient: g_i = 2 * Σ_k w_k * a_{k-i}
    a_rev = a[::-1]
    conv_w = _conv_fft(a_rev, w)              # length 3n-2
    start = n - 1
    grad = 2.0 * conv_w[start:start + n]

    return max_conv, grad


def _smooth_max_opt(start_seq: np.ndarray,
                    deadline: float,
                    max_iters: int = 300_000,
                    beta_start: float = 5.0,
                    beta_end: float = 2000.0):
    """
    Adam‑style projected gradient descent on the smooth‑max objective.
    Returns the best sequence (ℓ₁‑normalised) seen and its exact surrogate
    value 2·n·max(conv).
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # geometric schedule for β
    log_ratio = math.log(beta_end / beta_start) if beta_start > 0 else 0.0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        beta = beta_start * math.exp(log_ratio * (it - 1) / max_iters)

        # smooth max and gradient
        max_conv, grad = _smooth_max_and_grad(a, beta)

        # Adam update (step size decreasing like 1/√t)
        lr = 0.5 / math.sqrt(it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project back onto simplex (keeps sum=1, non‑negative)
        a = _proj_simplex(a)

        # Track exact surrogate for reporting
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SmoothOpt] it={it:7d} β={beta:6.1f} best={best_val:.6f}")

    return best_seq, best_val

# Sub‑gradient polishing (as in the reference solution)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Returns best sequence seen and its exact surrogate value.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size schedule
        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val
