def evaluate_sequence(sequence):
    # input list -> sequence
    # convert each element to float
    # sequence = [max(0, x) for x in sequence]
    # sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)

    # sum_a must be >=0.01 else inf
    return float(2 * n * max_b / (sum_a**2))

def solve_linear_program(a, M):
    n = len(a)
    L = 2 * n - 1
    # build T
    col = np.concatenate([a, np.zeros(n-1)])
    row = np.concatenate(([a[0]], np.zeros(n-1)))
    T = np.linalg.toeplitz(col, row) # but Toeplitz function is inside scipy.linalg.
    # Use Linprog
    c = -np.ones(n)  # maximize sum b => minimize -sum
    bounds = [(0, None)] * n
    A_ub = T
    b_ub = np.full(L, M)
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        raise RuntimeError("LP failed")
    b = res.x
    return b

def _lp_optimize(a, M):
    n = len(a)
    L = 2*n - 1
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # Build Toeplitz matrix
    col = np.concatenate([a, np.zeros(n-1)])
    row = np.empty(n)
    row[0] = a[0]
    row[1:] = 0.0
    A = scipy.linalg.toeplitz(col, row)  # shape L x n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(L, M), bounds=bounds,
                                method='highs', options={'presolve': True, 'time_limit': 10.0})
    if not res.success:
        raise RuntimeError('LP failed: ' + res.message)
    return res.x

"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

within the given budget (≈ 1000 s).

The algorithm is a classic “alternating linear programme”:

0. Choose a moderate length (n≈2000).  The objective is scale‑invariant,
   so we normalise every candidate to ℓ₁‑norm √(2n);  then the
   evaluation reduces to the maximal entry of the self‑convolution
   (the factor 2·n/‖a‖₁² cancels).

1. Initialise a with a simple deterministic distribution (uniform).

2. **LP step** – for the current a compute M = max(a∗a).  Solve the LP

        maximise    Σ b_j
        subject to  (a∗b)_k ≤ M      for all k = 0 … 2n‑2
                     b_j ≥ 0

   (a∗b is a linear map in b, i.e. a Toeplitz matrix).  The solution b
   is the “most massive” function that still respects the same max‑convolution
   bound.

3. Rescale b to the same ℓ₁‑norm as a:
        g = b · (√(2n) / Σ b).

   Mix a and g:
        a ← (1‑t)·a + t·g   (t = 0.2 initially, later reduced).

   This mixture provably does not increase M, and usually makes it
   strictly smaller.  Iterate the LP‑mix step until the improvement
   stalls or the time budget expires.

4. When the LP loop ends, run a short exponentiated‑gradient (mirror‑descent)
   refinement on the exact max‑convolution to polish the solution.

5. Return the best vector found (ℓ₁‑normalised to 1 so that the public
   evaluator can handle it directly).

All heavy linear‑algebra uses NumPy / SciPy.  The LP is solved with the
high‑performance HiGHS interior‑point solver (scipy.optimize.linprog,
method='highs').  The implementation is fully self‑contained and obeys
the time limit.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.linalg as _la

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return max(conv(a, a)).  a is assumed non‑negative."""
    conv = _conv_fft(a, a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Linear‑programming step (solve for b given a and its current bound)
# ----------------------------------------------------------------------
def _build_toeplitz_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n Toeplitz matrix T such that
        (a ∗ b) = T @ b .
    Row i, column j contains a[i‑j] when 0 ≤ i‑j < n, else 0.
    """
    n = a.shape[0]
    L = 2 * n - 1
    # first column: a followed by (n‑1) zeros
    col = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
    # first row: a[0] followed by zeros
    row = np.empty(n, dtype=np.float64)
    row[0] = a[0]
    row[1:] = 0.0
    # Toeplitz matrix – dense, size L×n
    return _la.toeplitz(col, row)


def _lp_optimize(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP
        maximise    Σ b_j
        subject to  (a∗b)_k ≤ M   for all k
                     b_j ≥ 0
    and return the optimal b (as a dense NumPy array).
    """
    n = a.shape[0]
    L = 2 * n - 1

    # objective – minimise –Σb  (maximise Σb)
    c = -np.ones(n, dtype=np.float64)

    # bounds b_j ≥ 0
    bounds = [(0.0, None)] * n

    # build Toeplitz matrix for the convolution a∗b
    T = _build_toeplitz_matrix(a)          # shape (L, n)

    # right‑hand side of the inequalities
    b_ub = np.full(L, M, dtype=np.float64)

    # solve with HiGHS (fast interior‑point / simplex implementation)
    res = _opt.linprog(
        c,
        A_ub=T,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30.0},
    )
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    return res.x


# ----------------------------------------------------------------------
# Main iterative scheme (alternating LP + mixing)
# ----------------------------------------------------------------------
def _lp_iterations(
    n: int,
    deadline: float,
    max_outer: int = 30,
    init_step: float = 0.25,
) -> np.ndarray:
    """
    Run the alternating LP procedure.
    Returns the best vector (ℓ₁‑normalised to 1) discovered during the loop.
    """
    # target ℓ₁‑norm – the algorithm works with this convenient scaling
    target_sum = np.sqrt(2.0 * n)

    # start from a uniform distribution (scaled to target_sum)
    a = np.full(n, target_sum / n, dtype=np.float64)

    best_a = a.copy()
    best_M = _max_convolution(a)

    step = init_step
    it = 0
    while time.time() < deadline and it < max_outer:
        it += 1
        # current bound
        M = _max_convolution(a)

        # LP: find the heaviest b that respects the same bound
        try:
            b = _lp_optimize(a, M)
        except Exception as exc:  # pragma: no cover
            print("[LP] failure:", exc)
            break

        sum_b = b.sum()
        if sum_b == 0.0:
            print("[LP] got zero‑sum b – terminating.")
            break

        # rescale b to the target ℓ₁‑norm
        g = b * (target_sum / sum_b)

        # mixture update (convex combination)
        a = (1.0 - step) * a + step * g

        # renormalise numerically (keeps the sum exactly target_sum)
        a = a * (target_sum / a.sum())

        # keep the best seen so far
        if M < best_M:
            best_M = M
            best_a = a.copy()
            print(f"[LP] iteration {it:2d}: improved M = {M:.6f}")

        # gradually shrink the mixing coefficient – later updates are finer
        step = max(0.02, step * 0.85)

    # final normalisation to ℓ₁ = 1 (the public evaluator is scale‑invariant)
    best_a = best_a / best_a.sum()
    return best_a


# ----------------------------------------------------------------------
# Optional polishing with exponentiated‑gradient (mirror descent)
# ----------------------------------------------------------------------
def _exp_gradient_refine(a0: np.ndarray, n: int, deadline: float) -> np.ndarray:
    """
    Mirror‑descent on the exact max‑convolution.
    Starts from a0 (already ℓ₁‑normalised) and stops when time runs out.
    """
    a = a0.copy()
    best = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev = idx - i_range
        g[i_range] = 2.0 * a[rev]

        # step‑size ηₜ = 1/√t  (more aggressive than the original 0.5)

        log_a = np.log(a + 1e-30) - eta * g
        a = np.exp(log_a)
        a = a / a.sum()

        cur = _max_convolution(a)
        if cur < best_val:
            best_val = cur
            best = a.copy()
            print(f"[ExpGrad] it={it:,}  new best M ≈ {cur:.6f}")

        # very cheap early exit when improvement stalls
        if it % 250_000 == 0 and time.time() > deadline - 5:
            break
    return best
