def smooth_obj_and_grad(v, alpha):
    a = _softmax(v)
    conv = _conv_fft(a, a)   # length 2n-1
    # Use stable log-sum-exp: subtract max
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    sum_exp = np.sum(exp_term)
    smax = maxc + np.log(sum_exp) / alpha
    obj = 2 * n * smax   # approximate objective
    w = exp_term / sum_exp   # normalized weights
    # compute grad_a = 2 * Σ_k w_k * a_{k-i} = 2 * (w * a_rev)[i+(n-1)]
    a_rev = a[::-1]
    w_conv = _conv_fft(w, a_rev)  # length 2n-1 + n -1 = 3n-2? Actually w length 2n-1, a_rev length n
    # after convolution the result length L = (2n-1) + n -1 = 3n-2. We need indices offset = n-1 to n-1 + n-1 = 2n-2.
    start = n-1
    grad_a = 2 * w_conv[start:start+n]   # shape n
    # compute gradient wrt v (softmax logits)
    dot = np.dot(grad_a, a)
    grad_v = (grad_a - dot) * a
    return obj, grad_v

"""
Propose a candidate step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is the full linear convolution.

The search combines:
  * a large, diverse set of analytic and random seeds,
  * a smooth‑max Adam optimiser working in log‑softmax space,
  * a short projected sub‑gradient polishing,
  * a final fast pair‑wise hill‑climbing (local) refinement.

All steps respect the 1000 s wall‑clock budget.
The best sequence found is stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_k_uniform(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=min(k, n), replace=False)
    a[idx] = 1.0 / len(idx)
    return a


def _multimodal_gaussian(n: int, rng: np.random.Generator, modes: int = 3) -> np.ndarray:
    """Mixture of ``modes`` discrete Gaussians with random centres/widths."""
    x = np.zeros(n, dtype=np.float64)
    for _ in range(modes):
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.02, 0.15) * n
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-((i - centre) ** 2) / (2.0 * sigma ** 2))
        weight = rng.random()
        x += weight * a
    s = x.sum()
    if s <= 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(logits: np.ndarray, n: int, alpha: float):
    """
    Returns
        obj   – smooth approximation of 2·n·max(conv)
        grad  – gradient w.r.t. ``logits`` (log‑softmax parameters)
    """
    a = _softmax(logits)                     # a_i ≥0, sum=1
    conv = _conv_fft(a, a)                   # length 2n‑1
    # stable softmax over alpha*conv
    m = np.max(conv)
    exp_term = np.exp(alpha * (conv - m))
    sum_exp = np.sum(exp_term)
    smax = m + np.log(sum_exp) / alpha       # smooth max
    obj = 2.0 * n * smax

    # weights for the smooth max
    w = exp_term / sum_exp                    # length 2n‑1, sum=1

    # gradient wrt a : 2 * Σ_k w_k * a_{k‑i}
    a_rev = a[::-1]
    w_conv = _conv_fft(w, a_rev)             # length 3n‑2
    start = n - 1
    grad_a = 2.0 * w_conv[start:start + n]    # shape (n,)

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_logits = (grad_a - dot) * a
    return obj, grad_logits


def _adam_smooth_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best ℓ₁‑normalised sequence seen and its *exact* surrogate value.
    """
    # initialise log‑softmax parameters
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linearly increase the smoothing parameter
        alpha = 5.0 + 195.0 * (t / max_iters)          # from 5 → 200

        # smooth objective + gradient
        _, grad = _smooth_obj_and_grad(v, n, alpha)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v2 = beta2 * v2 + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v2_hat = v2 / (1 - beta2 ** t)

        lr = 0.08                                      # a safe learning rate
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

        # evaluate exact surrogate every few iterations
        if t % 5000 == 0 or t == max_iters:
            a = _softmax(v)
            val = _objective_surrogate(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={t:7d} surrogate={best_val:.6f}")

    return best_seq, best_val

def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size
        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def _local_pairwise_search(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly transfer a small amount of mass from one index to another.
    Accept only improvements (optional tiny probability for worsening could be added).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    rng = np.random.default_rng(987654321)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # pick donor i (with probability proportional to current mass)
        i = rng.choice(n, p=a)
        # pick a random recipient j != i
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # amount to move: a small random fraction of donor's mass
        if a[i] < 1e-12:
            continue
        frac = rng.uniform(0.001, 0.03)   # 0.1 %–3 % of donor
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        # evaluate – exact surrogate is cheap enough
        val = _objective_surrogate(a2)
        if val < best_val:
            a = a2
            best_seq = a2.copy()
            best_val = val
            # occasional progress report
            if it % 50000 == 0:
                print(f"[Local] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    seeds = []

    # uniform baseline
    seeds.append(_uniform(n))

    # analytic families
    for a in np.linspace(0.2, 5.0, 9):
        for b in np.linspace(0.2, 5.0, 9):
            seeds.append(_beta_shape(n, a, b))

    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))

    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))

    # sparse uniform on a random subset
    for k in (10, 20, 50, 100):
        seeds.append(_random_k_uniform(n, k, rng))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # multimodal Gaussian mixtures (highly diverse)
    for _ in range(100):
        modes = rng.integers(2, 6)    # 2‑5 modes
        seeds.append(_multimodal_gaussian(n, rng, modes=modes))

    # tiny jittered versions of the best known solution (if any)
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.sum() > 0.0:
            arr = arr / arr.sum()
            for _ in range(10):
                # add little Gaussian noise then renormalise
                noisy = arr + rng.normal(scale=0.001, size=n)
                noisy = np.clip(noisy, 0, None)
                if noisy.sum() > 0:
                    seeds.append(noisy / noisy.sum())

    # final housekeeping – ensure non‑negative ℓ₁‑normalised vectors
    uniq = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        tot = s.sum()
        if tot <= 0.0:
            continue
        uniq.append(s / tot)
    return uniq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep the size used in the reference solution.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Generate a rich pool of seeds.
    # ------------------------------------------------------------------
    seeds = _generate_seeds(n, rng)
    print(f"\n--- Generated {len(seeds)} seed candidates ---")

    # ------------------------------------------------------------------
    # 2. Coarse ranking – keep the 3 best seeds (by exact surrogate).
    # ------------------------------------------------------------------
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 3
    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Best surrogate among raw seeds: {scored[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam smooth‑max optimisation on the top seeds.
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all Adam runs.")
            break

        # allocate a number of iterations proportional to remaining time
        remaining = max(0.0, deadline - time.time())
        est_iters = int(min(150_000, remaining * 2500))   # rough 2500 it/s
        if est_iters < 20_000:
            est_iters = int(min(80_000, remaining * 2500))
        print(f"\n--- Adam optimisation #{idx + 1} (≈{est_iters} iters) ---")
        seq_opt, val_opt = _adam_smooth_opt(seed_vec, n, deadline, max_iters=est_iters)

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Sub‑gradient polishing (projected) – short but effective.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(250_000, remaining * 0.5))
        if max_sub > 0:
            print("\n--- Sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq, n, deadline, max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved best to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final local pair‑wise hill‑climbing refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_local = int(min(300_000, remaining * 0.4))
        if max_local > 0:
            print("\n--- Local pair‑wise refinement ---")
            seq_loc, val_loc = _local_pairwise_search(global_best_seq, n, deadline, max_iters=max_local)
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Local] final best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
