"""
Simulated‑annealing / evolutionary search for a non‑negative step function.

The algorithm works in three stages:

1. **Seed collection** – generate a diverse set of ℓ₁‑normalised vectors
   (analytic windows, extreme Beta‑U shapes, low‑concentration Dirichlet,
   sparse supports and a warm‑start from ``height_sequence_1`` if present).
   The best few seeds are kept as starting points.

2. **Annealed random walk in log‑softmax space** – the current vector is
   represented by logits `v` (so `a = softmax(v)`).  At each iteration a
   Gaussian perturbation is added to the logits; the new candidate is
   accepted if it improves the objective or with probability
   `exp(-(Δ)/T)` where `T` is a linearly decreasing temperature.
   The temperature and the perturbation scale shrink as the deadline
   approaches.  The best vector seen so far is stored globally as
   ``height_sequence_1`` for warm‑starting later runs.

3. **Optional polishing** – a short exponentiated‑gradient (mirror‑descent)
   phase using the exact max‑convolution sub‑gradient further refines the
   solution.

All convolutions are computed by a real FFT (O(n log n)) to keep each
objective evaluation cheap.  The final list of floats is returned and also
saved in the module global ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on {1,…,n}.  For α,β<1 this is a U‑shape."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Low‑concentration Dirichlet → very sparse support."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_sparse(n: int, rng: np.random.Generator,
                   k_min: int = 30, k_max: int = 120) -> np.ndarray:
    """Sparse support: choose K random positions with exponential weights."""
    K = int(rng.integers(k_min, k_max + 1))
    idx = rng.choice(n, size=K, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = rng.exponential(scale=1.0, size=K)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Objective utilities (exact, not the smoothed surrogate)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  (sum(a) is 1 by construction)."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (active lag only)."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))          # the lag attaining the max
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    rev = idx - i_range
    # contribution is 2*a[rev] for each i in the active range
    g[i_range] = 2.0 * a[rev]
    return g


def _exp_gradient_optimize(start_a: np.ndarray,
                          deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated gradient) on the exact max‑conv objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        g = _subgradient_maxconv(a)
        eta = eta0 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 10_000 == 0:
            cur = _objective(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
    # final check
    cur = _objective(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val

def _collect_seeds(n: int,

    seeds: list[np.ndarray] = []

    seeds.append(_triangular(n))

        seeds.append(warm_start / warm_start.sum())

    cleaned = []

        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    ``evaluate_sequence``.  The method builds a wide pool of analytic and
    random seeds, selects the best as a start, and then performs a
    simulated‑annealing walk in the log‑softmax space with an adaptive
    Gaussian perturbation.  A short exponentiated‑gradient refinement follows.
    The best sequence found before the deadline is returned and stored in the
    global name ``height_sequence_1`` for warm‑starting future calls.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin so that we can always return something
    deadline = start_time + max(0.0, budget_s - 1.0)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 4000))
    print(f"\n=== Optimisation start (n={n}) ===")

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    warm_start = None
    if isinstance(previous, list):
        try:
            warm_arr = np.asarray(previous, dtype=np.float64)
            if warm_arr.shape == (n,):
                warm_start = warm_arr
        except Exception:
            warm_start = None

    # ------------------------------------------------------------------
    # Stage 0 – generate seeds and keep the best one
    # ------------------------------------------------------------------
    seeds = _collect_seeds(n, rng, warm_start)
    print(f"Generated {len(seeds)} seed vectors.")

    # evaluate seed pool (exact objective)
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    best_seed, best_val = seed_vals[0]
    print(f"[Seed] best seed objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – annealed random walk in logits (softmax) space
    # ------------------------------------------------------------------
    # initialise logits from the best seed
    cur_logits = np.log(best_seed + 1e-30)
    cur_obj = best_val
    best_logits = cur_logits.copy()
    best_obj = cur_obj

    # hyper‑parameters for the annealing schedule
    sigma0 = 2.0                     # initial standard deviation of perturbations
    sigma_min = 0.02                 # final perturbation scale
    temp0 = 0.15                     # initial temperature (controls acceptance)
    temp_min = 1e-5                  # final temperature

    iteration = 0
    no_improve = 0
    # we will run until the deadline expires
    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s - 1.0)               # progress 0 → 1
        # linear decay of sigma and temperature
        sigma = sigma0 * (1.0 - frac) + sigma_min * frac
        temp = temp0 * (1.0 - frac) + temp_min * frac

        # propose a perturbation in logit space
        delta = rng.normal(scale=sigma, size=n)
        cand_logits = cur_logits + delta
        cand_a = _softmax(cand_logits)
        cand_obj = _objective(cand_a)

        # acceptance test (Metropolis)
        accept = False
        if cand_obj < cur_obj:
            accept = True
        else:
            # probability of accepting a worse move
            prob = math.exp(-(cand_obj - cur_obj) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_logits = cand_logits
            cur_obj = cand_obj
            no_improve = 0
        else:
            no_improve += 1

        # keep global best
        if cand_obj < best_obj:
            best_obj = cand_obj
            best_logits = cand_logits.copy()
            print(f"[Iter {iteration:>6}] new best objective = {best_obj:.6f}")

        # occasional restart from the current best if we have been stuck
        if no_improve >= 5000:
            jitter = rng.normal(scale=sigma0 * 0.5, size=n)
            cur_logits = best_logits + jitter
            cur_obj = _objective(_softmax(cur_logits))
            no_improve = 0

        # limit the loop overhead (check deadline every few iterations)
        if iteration % 1000 == 0 and time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Stage 2 – short exponentiated‑gradient refinement (exact max‑conv)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining_time = deadline - time.time()
        max_iters = int(min(200_000, remaining_time * 200))   # ~200 iters per ms
        if max_iters > 0:
            print("\n--- Exponentiated‑gradient refinement (exact max) ---")
            start_vec = _softmax(best_logits)
            refined_vec, refined_val = _exp_gradient_optimize(start_vec,
                                                             deadline,
                                                             max_iters=max_iters,
                                                             eta0=1.0)
            if refined_val < best_obj:
                best_obj = refined_val
                best_logits = np.log(refined_vec + 1e-30)
                print(f"[Refine] improved objective to {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final conversion & reporting
    # ------------------------------------------------------------------
    best_seq = _softmax(best_logits).tolist()
    try:
        true_val = evaluate_sequence(best_seq)
        print(f"\n=== Finished ===\nBest objective (exact) = {best_obj:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible warm‑starting on the next call
    globals()["height_sequence_1"] = best_seq
    return best_seq
