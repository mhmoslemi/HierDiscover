def _exp_gradient_opt_until_deadline(a, n, deadline, base_eta=0.5):
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    while True:
        if time.time() > deadline:
            break
        it += 1
        # compute conv
        conv = _conv_fft(np.exp(log_a), np.exp(log_a))
        # Or compute a from softmax each iteration? Actually we can compute a = _softmax(log_a) once.
        a_current = _softmax(log_a)   # ensures positivity and sum=1
        conv = _conv_fft(a_current, a_current)
        max_idx = int(np.argmax(conv))
        # compute subgradient g
        g = np.zeros_like(a_current)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a_current[rev_idx]
        # update log_a with step size decreasing sqrt
        eta = base_eta / math.sqrt(it)
        log_a = log_a - eta * g
        # compute surrogate
        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()
            # maybe print occasionally
            if it % 200000 == 0:
                print(f"[ExpGradLong] it={it} best={best_val:.6f}")
    return best_seq, best_val

"""
Propose a high‑quality non‑negative step‑function sequence.

Algorithm
---------
1. **Seed pool** – Generate a diverse set of ℓ₁‑normalised vectors:
   * analytic families (uniform, triangular, Gaussian, power‑law)
   * Dirichlet draws with several concentration parameters,
   * a few shuffled / alternating variants,
   * the previous solution (if any).

2. **Initial selection** – Rank seeds by the cheap surrogate
   `2·n·max(conv(a,a))` (exact for ℓ₁‑normalised vectors) and keep the
   best few for deep optimisation.

3. **Continuous optimisation**
   * optionally run a few short L‑BFGS‑B refinements on the smooth
     log‑sum‑exp surrogate of the max‑convolution,
   * a long mirror‑descent (exponentiated‑gradient) run that works
     directly on the true max‑convolution sub‑gradient.  It runs until
     a configurable fraction of the remaining time has been spent.

4. **Permutation refinement** – With the shape obtained above, we
   explore the space of permutations:
   * many random shuffles are tried,
   * a simple hill‑climber performs a few random swaps,
   all evaluated by the surrogate.

5. **Final evaluation** – The best candidate is finally passed to the
   official `evaluate_sequence` function and stored in the global
   variable ``height_sequence_1`` for the next round.

The whole routine respects a 1000 s budget, exiting early if the time
limit is reached.  All helper functions are top‑level, there are no
closures or file I/O.  The code works with or without SciPy
(`L‑BFGS` part is skipped if SciPy is unavailable).
"""

# ----------------------------------------------------------------------

import time
import math
import numpy as np

# SciPy is optional – only needed for the L‑BFGS stage

except Exception:  # pragma: no cover

# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Permutation helpers for richer seed pool
# ----------------------------------------------------------------------
def _alternating_order_from_vec(vec: np.ndarray) -> np.ndarray:
    """Place sorted values alternating from left and right ends."""
    sorted_desc = np.sort(vec)[::-1]
    n = len(vec)
    out = np.empty_like(vec)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out


def _shuffle_by_step(sorted_desc: np.ndarray, step: int) -> np.ndarray:
    """Modular step permutation (step coprime with n)."""
    n = len(sorted_desc)
    out = np.empty(n, dtype=np.float64)
    pos = 0
    for val in sorted_desc:
        out[pos] = val
        pos = (pos + step) % n
    return out


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diversified list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families ----------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.10, 0.18, 0.25):
        seeds.append(_gaussian(n, sigma))

    for p in (0.0, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))

    # Dirichlet draws + permutations ----------------------------------------
    conc_list = (0.05, 0.2, 0.5, 1.0, 2.0, 5.0)
    step_list = [3, 7, 11, 13, 17, 19, 23, 29, 31, 37]  # many coprime steps
    for conc in conc_list:
        for _ in range(3):
            raw = _random_dirichlet(n, conc, rng)

            # raw itself
            seeds.append(raw)

            # descending order
            desc = np.sort(raw)[::-1]
            seeds.append(desc)

            # alternating order
            seeds.append(_alternating_order_from_vec(raw))

            # a handful of modular shuffles
            for step in step_list[:4]:
                seeds.append(_shuffle_by_step(desc, step))

            # a random permutation
            perm = rng.permutation(n)
            seeds.append(raw[perm])

    # Include previous solution (if any) ------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Robust normalisation ---------------------------------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smooth max‑conv."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for the L‑BFGS stage.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0]

    # Initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                # stays on the simplex
            conv = _conv_fft(a, a)                # full convolution
            max_c = np.max(conv)
            shift = max_c
            # smooth approximation of max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                # weights for gradient
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # gradient of the smooth max
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule for log‑softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser (exact max‑conv)
# ----------------------------------------------------------------------
def _exp_gradient_until_deadline(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 base_eta: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Long mirror‑descent on the true max‑conv surrogate.
    Runs until ``deadline`` is hit.
    Returns the best sequence found and its surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)
    best_seq = _softmax(log_a)        # current simplex point
    best_val = _objective_surrogate(best_seq)
    it = 0

    while time.time() < deadline:
        # current point on simplex
        a = _softmax(log_a)

        # convolution and location of the active entry
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (see discussion)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size (sqrt schedule)
        it += 1
        eta = base_eta / math.sqrt(it)
        log_a = log_a - eta * g

        # evaluate new point (after the update)
        a_new = _softmax(log_a)
        cur_val = _objective_surrogate(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()
            if it % 200_000 == 0:
                print(f"[ExpGradLong] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based local refinement
# ----------------------------------------------------------------------
def _permutation_refinement(seq: np.ndarray,
                            n: int,
                            deadline: float,
                            rng: np.random.Generator,
                            attempts: int = 250,
                            swaps: int = 4000) -> tuple[np.ndarray, float]:
    """
    Random shuffles followed by a few random‑swap hill climbs.
    Works on the surrogate objective only.
    """
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)

    # ---- random shuffles -------------------------------------------------
    for _ in range(attempts):
        if time.time() > deadline:
            break
        perm = rng.permutation(n)
        cand = best_seq[perm]
        cand_val = _objective_surrogate(cand)
        if cand_val < best_val:
            best_seq = cand
            best_val = cand_val

    # ---- simple swap hill‑climb -----------------------------------------
    for _ in range(swaps):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        cand = best_seq.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_val = _objective_surrogate(cand)
        if cand_val < best_val:
            best_seq = cand
            best_val = cand_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Basic initialisation
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Problem size – 4000 gives a good trade‑off
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Build seed pool (include previous solution if available)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n:
                prev_arr = arr
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"[Seed] Collected {len(seeds)} raw seeds.")

    # cheap surrogate (exact for ℓ₁‑normalised vectors)
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print("[Seed] Best surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise best candidate
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] Current best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional short L‑BFGS on a few elite seeds
    # ------------------------------------------------------------------
    if _opt is not None:
        # run only on the first three seeds to keep the budget
        for idx, seed_vec in enumerate(top_seeds[:3]):
            if time.time() > deadline:
                break
            try:
                seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                                 n,
                                                 deadline,
                                                 max_iter_per_beta=1500)
                val_opt = _objective_surrogate(seq_opt)
                print(f"[L‑BFGS seed {idx}] surrogate = {val_opt:.6f}")
                if val_opt < best_val:
                    best_seq = seq_opt.copy()
                    best_val = val_opt
                    print(f"[Update] new best surrogate = {best_val:.6f}")
            except Exception as e:  # pragma: no cover
                print(f"[L‑BFGS] failed on seed {idx}: {e}")

    # ------------------------------------------------------------------
    # 5. Long exponentiated‑gradient (mirror descent) run
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # reserve ~70 % of the remaining budget for this stage
        remaining = deadline - time.time()
        exp_deadline = time.time() + remaining * 0.70
        print("[ExpGradLong] Running mirror‑descent until deadline …")
        seq_opt, val_opt = _exp_gradient_until_deadline(best_seq,
                                                        n,
                                                        exp_deadline)
        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[ExpGradLong] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Permutation‑based refinement (random shuffles + swaps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Permutation] Refining ordering …")
        seq_opt, val_opt = _permutation_refinement(best_seq,
                                                   n,
                                                   deadline,
                                                   rng,
                                                   attempts=300,
                                                   swaps=5000)
        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Permutation] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next round
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
