"""
Improved search for a non‑negative step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm builds a large and diverse pool of seed shapes (including
more extreme Beta and exponential families), evaluates the exact
objective for each seed, and then refines the best candidates with three
different optimisation stages:

1. Continuation L‑BFGS on a smooth log‑sum‑exp surrogate (many β steps,
   many L‑BFGS iterations per β).
2. Adam optimisation on the same surrogate (momentum helps escape
   shallow local minima).
3. Mirror‑descent (exponentiated‑gradient) on the *exact* max‑convolution
   objective for a final fine‑tune.

A short stochastic “mass‑transfer” hill‑climb finishes the search.
All stages respect the 1000 s deadline, and the best sequence found is
stored in the global variable ``height_sequence_1`` and returned as a
plain list of floats.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors, sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)  # n, n‑1, …, 1
    s = a.sum()
    return a / s


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)  (i = 0 … n‑1)"""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) on {1,…,n} (U‑shaped when α,β<1)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)          # Bessel I0 (NumPy)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i = 1/(i+1)^p  (i = 0 … n‑1)"""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α) + b·α, renormalised to sum‑1."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed pool construction (diverse, ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # Deterministic families – extended parameter ranges
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_linear_decrease(n))

    # exponential decay (wide λ range)
    for lam in (1e-4, 5e-4, 1e-3, 2e-3, 5e-3, 1e-2, 2e-2):
        seeds.append(_exponential_decay(n, lam))

    # sine window with various gammas
    for g in (0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        seeds.append(_sine_window(n, gamma=g))

    # Classic windows
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12, 15, 20):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian windows
    for sigma in np.linspace(0.08, 0.45, 9):
        seeds.append(_gaussian(n, sigma))

    # Power‑law family (including large exponents)
    for p in np.linspace(0.0, 6.0, 25):
        seeds.append(_power_law(n, p))

    # Beta family – include very spiky U‑shapes (α,β < 1)
    for a in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0):
        for b in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0):
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws (wide concentration range)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of existing seeds (pairwise)
    base = np.arange(len(seeds))
    for _ in range(400):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Add previous solution if available
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Clean any numerical noise and enforce ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Surrogate (smooth max) helpers for continuation optimisation
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Log‑sum‑exp surrogate of the max‑convolution.
    Returns (obj, grad) where grad is w.r.t. the logits v.
    """
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)               # full linear convolution
    shift = np.max(conv)                 # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t a
    w = exp_term / sum_exp               # soft‑max weights over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3·n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]   # slice of length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 3000,
) -> np.ndarray:
    """
    Continuation L‑BFGS on the smooth surrogate.
    The optimisation variable is the log‑softmax vector.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 100000.0,
            200000.0, 500000.0,
        ]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            # early abort if we have overrun the deadline
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during beta={beta_val:.0f}")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)   # final probability vector


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient (max‑convolution)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    Takes the gradient of the *single* lag attaining the maximum.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    # g_i = 2 * a_{idx-i} for all i that contribute to this lag
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = idx - idx_i
    g[idx_i] = 2.0 * a[rev_idx]
    return g


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------
def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 2000,
    lr: float = 0.3,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp surrogate.
    Returns (best_a, best_obj) where best_obj is the *exact* objective.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0,
                         200000.0]

    v = np.log(start_vec + 1e-30)   # logits
    best_a = _softmax(v)
    best_val = _objective_exact(best_a)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        b = float(beta)

        m = np.zeros_like(v)
        v2 = np.zeros_like(v)

        for it in range(1, max_iter_per_beta + 1):
            if time.time() > deadline:
                break

            obj, grad = _smooth_objective(v, n, b)

            # Adam moment updates
            m = beta1 * m + (1.0 - beta1) * grad
            v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** it)
            v2_hat = v2 / (1.0 - beta2 ** it)

            v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

            # occasional exact evaluation
            if it % 400 == 0:
                a_cur = _softmax(v)
                cur_val = _objective_exact(a_cur)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a_cur.copy()

    # final check
    a_final = _softmax(v)
    final_val = _objective_exact(a_final)
    if final_val < best_val:
        best_val = final_val
        best_a = a_cur.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 3_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *exact* max‑convolution objective.
    Returns (best_a, best_obj).
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 200_000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny stochastic local‑search (mass transfers)
# ----------------------------------------------------------------------
def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 5000,
    step_frac: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a small fraction of mass from a random
    index i to a random index j.  Keep changes that improve the exact
    objective.
    """
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(12345)

    for it in range(iters):
        if time.time() > deadline:
            break

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue

        delta = step_frac * a[i]
        if delta <= 0.0:
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        val = _objective_exact(a_candidate)
        if val < best_val:
            best_val = val
            best_a = a_candidate.copy()
            a = a_candidate   # continue from improved point

    return best_a, best_val

# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a better step‑function within the given time budget.
    Returns the best sequence (list of floats) and stores it globally.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search, n={n}, budget≈{budget_s}s ===")

    # --------------------------------------------------------------
    # Load previous solution if present
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # Seed generation (large and diverse)
    # --------------------------------------------------------------
    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # --------------------------------------------------------------
    # Quick exact evaluation of all seeds
    # --------------------------------------------------------------
    exact_vals = [_objective_exact(vec) for vec in seeds]
    ordered = sorted(zip(seeds, exact_vals), key=lambda x: x[1])
    top_k = min(20, len(ordered))
    top_seeds = [vec for vec, _ in ordered[:top_k]]
    print(f"Top {top_k} seed exact objectives:")
    for i, (_, val) in enumerate(ordered[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – heavy L‑BFGS on the smooth surrogate (multiple starts)
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_exact(global_best_seq)
    print(f"\n[Init] Exact objective = {global_best_val:.6f}")

    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        10000.0, 20000.0, 50000.0, 100000.0,
        200000.0, 500000.0,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS runs.")
            break

        # allocate per‑seed budget adaptively (≈ 30 s or remaining * 0.07)
        remaining = deadline - time.time()
        per_seed_budget = min(30.0, remaining * 0.07)
        per_beta = max(500, int(min(3000, per_seed_budget * 30)))  # rough mapping

        print(f"\n--- L‑BFGS run #{idx+1} (max_iter/beta≈{per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=per_beta,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] exact objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – Adam optimisation on the smooth surrogate (high β)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_per_beta = int(min(4000, remaining * 40))
        print("\n--- Adam optimisation (smooth surrogate) ---")
        seq_adam, val_adam = _adam_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=max_iters_per_beta,
            lr=0.5,
        )
        print(f"[Adam] exact objective = {val_adam:.6f}")
        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Update] New global best after Adam = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – exponentiated‑gradient on the *exact* objective
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(2_500_000, remaining * 0.25 * 1e5))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient (exact max) ---")
            seq_exp, val_exp = _exp_gradient_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_exp_iters,
                eta0=1.0,
            )
            print(f"[ExpGrad] final exact objective = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] New best after EG = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – final polishing L‑BFGS with ultra‑large β
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta_final = max(600, int(min(2500, remaining * 0.07 * 1e5)))
        print("\n--- Final polishing L‑BFGS (β≈5e5) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[200000.0, 500000.0],
            max_iter_per_beta=per_beta_final,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] exact objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – tiny stochastic local refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters_loc = int(min(8000, remaining * 25))
        if iters_loc > 0:
            print("\n--- Stochastic local search (mass transfers) ---")
            seq_loc, val_loc = _local_search(
                global_best_seq, n, deadline, iters=iters_loc, step_frac=0.015
            )
            print(f"[LocalSearch] exact objective = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] New best after local search = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final official evaluation (logging only)
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest exact ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for possible warm‑starts
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
