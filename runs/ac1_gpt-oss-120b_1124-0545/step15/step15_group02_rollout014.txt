def solve_lp_max_sum(f, M, time_limit):
    n = len(f)
    m = 2*n - 1
    # Build sparse A_ub.
    rows = []
    cols = []
    data = []
    for k in range(m):
        j_start = max(0, k-(n-1))
        j_end = min(k, n-1)
        # columns range j_start..j_end inclusive.
        for j in range(j_start, j_end+1):
            rows.append(k)
            cols.append(j)
            data.append(f[k-j])

    ub = np.full(m, M)

import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

def _conv_full(a, b):
    return np.convolve(a, b)

def _lp_optimize(f, M, method='highs'):
    n = len(f)
    m = 2*n - 1
    # Build A
    # Data for each diagonal offset i = 0..n-1: constant f[i]
    diags = [np.full(n, f[i]) for i in range(n)]
    offsets = np.arange(n)
    A = sp.diags(diagonals=diags, offsets=offsets, shape=(m, n), format='csr')
    ub = np.full(m, M)
    c = -np.ones(n)  # maximize sum -> minimize -sum
    res = opt.linprog(c, A_ub=A, b_ub=ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        raise RuntimeError("LP failed: " + res.message)
    b = res.x
    # Normalize to sum 1
    if b.sum() == 0:
        return np.full(n, 1.0/n)
    return b / b.sum()
```

def _iterative_flat_conv(seq, iterations, L):
    a = seq.copy()
    for it in range(iterations):
        # pad simple zeros to length L
        a_padded = np.zeros(L, dtype=np.complex128)
        a_padded[:len(a)] = a
        A = np.fft.fft(a_padded)
        # desired magnitude:
        power = np.mean(np.abs(A)**2)  # average power across frequencies
        target_mag = np.sqrt(power)
        # construct new A with target magnitude and original phase
        A_new = target_mag * np.exp(1j * np.angle(A))
        a_new_full = np.fft.ifft(A_new).real
        # take first n entries (others are near zero due to zero padding)
        a_new = a_new_full[:len(a)]
        # enforce nonnegativity and simplex
        a_new = np.maximum(a_new, 0)
        sum_a = a_new.sum()
        if sum_a == 0:
            a_new = np.full_like(a_new, 1.0/len(a_new))
        else:
            a_new = a_new / sum_a
        a = a_new
    return a
```

"""
Propose a step‑function (list of non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three stages:

1. **Initial seed** – reuse the previous best construction if available,
   otherwise start from a random Dirichlet vector and run a few L‑BFGS
   iterations on the smooth log‑sum‑exp surrogate of the max‑convolution.

2. **Greedy local search** – repeatedly move a small fraction of mass from
   the index that contributes most to the current peak of the convolution
   to the index with smallest weight.  The convolution is updated incrementally
   (O(n) per move) and the move is accepted only if the surrogate
   (2·n·max(conv)) improves.  The step size is adapted automatically.

3. **Final polishing** – a short L‑BFGS run on the refined vector to
   settle any remaining smooth optimisation.

All vectors are kept ℓ₁‑normalised (sum = 1) so the evaluation reduces to the
surrogate 2·n·max(conv).  The routine respects the 1000 s wall‑clock budget,
stores the best sequence into the global variable ``height_sequence_1`` and
returns it as a plain list.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective for an ℓ₁‑normalised vector:
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc`."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """Optimises the log‑sum‑exp surrogate of the max‑convolution."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0]

    # initialise logits (log‑softmax parametrisation)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stays on the simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a
            w = exp_beta / sum_exp                      # soft‑max on the convolution
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Greedy local search (mass‑shifting)
# ----------------------------------------------------------------------
def _greedy_local_search(start_seq: np.ndarray,
                         start_surrogate: float,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple coordinate‑wise hill‑climbing:
    move a fraction of the weight from the index that most contributes to
    the current maximal convolution entry towards the currently lightest index.
    Accept only improving moves.
    """
    a = start_seq.copy()
    n = len(a)
    m = 2 * n - 1
    conv = _conv_fft(a, a)
    best_seq = a.copy()
    best_val = start_surrogate

    # step size is a fraction of the mass taken from the “high” index
    step_frac = 0.2
    rng = np.random.default_rng(0)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        max_idx = int(np.argmax(conv))
        # indices that actively participate in the peak
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)

        # sub‑gradient of the peak (2·a_{k−i})
        grads = 2.0 * a[max_idx - idx_i]

        # pick a “high” index – large gradient *and* non‑negligible mass
        mask = a[idx_i] > 1e-12
        if not np.any(mask):
            break
        candidates_high = idx_i[mask]
        grads_high = grads[mask]
        i_high = candidates_high[np.argmax(grads_high)]

        # pick a “low” index – smallest current weight (any index is admissible)
        i_low = int(np.argmin(a))
        if i_low == i_high:
            # choose the second smallest to avoid zero move
            i_low = int(np.argsort(a)[1])

        delta = step_frac * a[i_high]
        if delta < 1e-14:
            # nothing worthwhile to move
            continue

        # store the vector before the move (needed for the update formula)
        a_old = a.copy()

        # compute the effect on the convolution (linear in the moved mass)
        delta_c = np.zeros(m, dtype=np.float64)
        delta_c[i_low:i_low + n] = a_old
        delta_c[i_high:i_high + n] -= a_old
        delta_c *= 2.0 * delta
        conv_new = conv + delta_c
        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        if new_val + 1e-12 < best_val:
            # accept the move
            a[i_high] -= delta
            a[i_low] += delta
            conv = conv_new
            best_val = new_val
            best_seq = a.copy()
            # slowly increase step size when we are successful
            step_frac = min(0.5, step_frac * 1.02)
            if it % 2500 == 0:
                print(f"[Greedy] it={it:6d} surrogate={best_val:.6f}")
        else:
            # reject – shrink step size
            step_frac *= 0.95
            if step_frac < 1e-4:
                step_frac = 0.2
                # occasional random perturbation to escape tiny plateaus
                if it % 2000 == 0:
                    pert = rng.normal(scale=0.005, size=n)
                    a = np.maximum(a + pert, 0.0)
                    a /= a.sum()
                    conv = _conv_fft(a, a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 provides a good trade‑off between expressiveness
    # and speed for the simple greedy moves.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Starting point (previous best if available)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                start_vec = arr / arr.sum()
            else:
                raise ValueError
        except Exception:
            start_vec = None
    else:
        start_vec = None

    # ------------------------------------------------------------------
    # 1. If no previous solution, generate a random Dirichlet seed and
    #    perform a short L‑BFGS run to get a decent starting sequence.
    # ------------------------------------------------------------------
    if start_vec is None:
        print("\n--- Seed generation (no previous solution) ---")
        start_vec = _random_dirichlet(n, conc=0.2, rng=rng)
        init_surrogate = _objective_surrogate(start_vec)
        print(f"[Init] surrogate = {init_surrogate:.6f}")

        try:
            start_vec = _lbfgs_smooth_optimize(start_vec,
                                                n,
                                                deadline,
                                                beta_schedule=[1, 5, 20, 80, 200, 500, 1000],
                                                max_iter_per_beta=2000)
            init_surrogate = _objective_surrogate(start_vec)
            print(f"[L‑BFGS init] surrogate = {init_surrogate:.6f}")
        except Exception as e:
            print("[L‑BFGS init] exception", e)

    # ------------------------------------------------------------------
    # 2. Greedy local search – main improvement phase
    # ------------------------------------------------------------------
    print("\n--- Greedy local search ---")
    init_surrogate = _objective_surrogate(start_vec)
    best_seq, best_surrogate = _greedy_local_search(start_vec,
                                                    init_surrogate,
                                                    deadline,
                                                    max_iters=250_000)

    print(f"[Greedy] final surrogate = {best_surrogate:.6f}")

    # ------------------------------------------------------------------
    # 3. Final short L‑BFGS polishing (smooth surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS polishing ---")
        try:
            polished = _lbfgs_smooth_optimize(best_seq,
                                               n,
                                               deadline,
                                               beta_schedule=[1, 5, 20, 80, 200, 500, 1000],
                                               max_iter_per_beta=2000)
            polished_val = _objective_surrogate(polished)
            if polished_val < best_surrogate:
                best_seq, best_surrogate = polished, polished_val
                print(f"[Polish] improved to {best_surrogate:.6f}")
        except Exception as e:
            print("[Polish] exception", e)

    # ------------------------------------------------------------------
    # 4. Real evaluation with the provided evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_surrogate:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store globally for possible future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
