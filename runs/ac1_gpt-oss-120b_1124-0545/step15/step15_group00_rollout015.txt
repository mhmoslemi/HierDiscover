"""
A fast, reproducible optimizer for the step‑function objective

    2·n·max(conv(a, a)) / (sum(a)²)

The strategy is simple and well‑behaved under the 1000 s budget:

1. **Massive seed pool** – analytic families (uniform, triangular, Gaussian,
   power‑law, geometric), many Dirichlet draws (several concentrations) and
   permutation tricks.  All seeds are ℓ₁‑normalised, so the cheap surrogate
   ``2·n·max(conv)`` equals the true objective.
2. **Select the best few seeds** – evaluate the surrogate with an FFT‑based
   convolution (≈ O(n log n)) and keep the top‑K candidates.
3. **Long projected sub‑gradient descent** – start from each top seed and
   perform thousands of sub‑gradient steps on the exact max‑convolution
   (the sub‑gradient is known analytically).  The iterate is projected onto
   the probability simplex after every step.
4. **Optional simulated‑annealing polish** – a short random walk in log‑softmax
   space explores the neighbourhood of the best solution.
5. **Final evaluation** – the best vector found is evaluated with the official
   ``evaluate_sequence`` function and stored in ``height_sequence_1``.
"""

# ----------------------------------------------------------------------
import time
import math
import numpy as np

# SciPy is optional – L‑BFGS is only used if it is available.
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (surrogate = exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all returned ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geometric(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i  (0 < r < 1)."""
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Permutation helpers (used to enrich the seed pool)
# ----------------------------------------------------------------------
def _alternating_order_from_vec(vec: np.ndarray) -> np.ndarray:
    """Place sorted values alternating from the left and right ends."""
    sorted_desc = np.sort(vec)[::-1]
    n = len(vec)
    out = np.empty_like(vec)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out


def _shuffle_by_step(sorted_desc: np.ndarray, step: int) -> np.ndarray:
    """Modular step permutation (step coprime to n)."""
    n = len(sorted_desc)
    out = np.empty(n, dtype=np.float64)
    pos = 0
    for val in sorted_desc:
        out[pos] = val
        pos = (pos + step) % n
    return out


# ----------------------------------------------------------------------
# Seed construction (large, diverse pool)
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Fixed analytic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.12, 0.18, 0.25):
        seeds.append(_gaussian(n, sigma))
    for p in (0.0, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))

    # Geometric family (dense sampling of the decay factor) -----------------
    # r close to 1 is the interesting region – we use a fine grid there.
    r_grid = np.concatenate([
        np.linspace(0.90, 0.98, 10, endpoint=False),
        np.linspace(0.98, 0.999, 30, endpoint=False),
        np.linspace(0.999, 0.9999, 30, endpoint=False)
    ])
    for r in r_grid:
        seeds.append(_geometric(n, r))

    # Random Dirichlet draws + permutation tricks ---------------------------
    conc_list = (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    step_list = [3, 7, 11, 13, 17, 19, 23, 29]          # coprime to most n
    for conc in conc_list:
        for _ in range(8):                      # 8 draws per concentration
            raw = _random_dirichlet(n, conc, rng)

            # keep raw (already normalised)
            seeds.append(raw)

            # descending order (helps spreading large values)
            desc = np.sort(raw)[::-1]
            seeds.append(desc)

            # alternating order (large–small–large–small)
            seeds.append(_alternating_order_from_vec(raw))

            # modular step permutations (few steps)
            for step in step_list[:4]:
                seeds.append(_shuffle_by_step(desc, step))

            # a random permutation (extra diversity)
            perm = rng.permutation(n)
            seeds.append(raw[perm])

    # Include the previous solution if it matches the dimensionality ----------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise robustly (protect against zero sum)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑conv surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Long projected sub‑gradient descent on 2·n·max(conv) for ℓ₁‑normalised vectors.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]

        # learning rate diminishing like 1/√t
        lr = 0.25 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[SubGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

# Simulated‑annealing refinement (logit‑space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Random walk in logit space with a simple cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        # temperature linearly decreasing from 0.07 to ~0
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_seq, best_val

        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} raw seeds.")

    # cheap surrogate (exact for ℓ₁‑normalised vectors)
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12                       # number of seeds to be refined heavily
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nBest surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient descent on each top seed (no L‑BFGS to save time)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] Deadline reached before finishing sub‑gradient runs.")
            break

        # remaining fraction of the total time – we split it evenly among the seeds
        remaining = deadline - time.time()
        # allow at most 150 k iterations per seed (adjusted by remaining time)
        max_iters = int(min(200_000, remaining * 0.08 * 1e5))
        max_iters = max(30_000, max_iters)

        print(f"\n--- Sub‑gradient run #{idx+1} (max_iters = {max_iters}) ---")
        seq_opt, val_opt = _subgradient_opt(seed_vec,
                                            n,
                                            deadline,
                                            max_iters=max_iters)

        print(f"[SubGrad] best surrogate from this run = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long sub‑gradient polishing on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(2_000_000, remaining * 0.12 * 1e5))
        if max_iters > 0:
            print("\n--- Extended sub‑gradient polishing on global best ---")
            seq_ext, val_ext = _subgradient_opt(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_iters)
            if val_ext < global_best_val:
                global_best_seq = seq_ext.copy()
                global_best_val = val_ext
                print(f"[ExtSubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Simulated‑annealing refinement (quick final push)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(800_000, remaining * 0.05 * 1e5))
        if max_sa > 0:
            print("\n--- Simulated‑annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing_opt(global_best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=max_sa)
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[SA] final best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
