--------
1. Initialise from the previous best sequence (``height_sequence_1``) or from

2. Run a continuation L‑BFGS optimiser on a smooth‑max surrogate

   local optimum.

   the exact objective 2·n·max(conv(a,a)).  Gradients are obtained from the
   active convolution entry; the optimisation lives in log‑softmax space,
   guaranteeing the ℓ₁‑normalisation and non‑negativity constraints.
4. If the Adam phase makes no progress for a long stretch, a small random
   perturbation (restart) is applied to escape possible plateaus.
5. The best sequence found is returned and stored in the global variable
   ``height_sequence_1`` for reuse.
"""

import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------

    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate when ``seq`` is ℓ₁‑normalised (sum = 1):
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Continuation L‑BFGS optimisation of a smooth‑max surrogate
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 20000) -> np.ndarray:
    """
    Minimise a smoothed version of 2·n·max(conv) using L‑BFGS‑B.
    Optimisation is performed in log‑softmax space (the simplex is automatic).

    Parameters
    ----------
    start_vec: ℓ₁‑normalised start vector.
    n: length of the vector.
    deadline: Unix time after which the optimisation must stop.
    beta_schedule: list of β values for the log‑sum‑exp smooth‑max.
    max_iter_per_beta: maximum L‑BFGS iterations per β.
    """
    if _opt is None:
        raise RuntimeError("SciPy is required for L‑BFGS phase.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0]

    # initialise logits – stays on the simplex after a soft‑max
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                # ℓ₁‑normalised
            conv = _conv_fft(a, a)                # linear convolution
            max_c = np.max(conv)

            # smooth‑max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # soft‑max weights (derivative of smooth‑max)
            w = exp_beta / sum_exp

            # gradient w.r.t. a: 2·conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

            if time.time() > deadline:
                raise KeyboardInterrupt()

                                x0=v,
                                method='L-BFGS-B',

                                         'disp': False,
