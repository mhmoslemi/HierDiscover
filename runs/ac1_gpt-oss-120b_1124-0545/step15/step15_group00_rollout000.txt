def _adam_optimize(start_seq, n, deadline, beta_schedule, iter_per_beta, lr=0.5):
    v = np.log(start_seq + 1e-30) # logits
    m = np.zeros_like(v)  # first moment
    v2 = np.zeros_like(v) # second moment
    t = 0
    for beta in beta_schedule:
        for itr in range(iter_per_beta):
            if time.time()>deadline:
                return _softmax(v)
            # compute a
            a = _softmax(v)
            # conv
            conv = _conv_fft(a, a)
            # compute p = softmax(beta*conv)
            # may have overflow. Use stable: mval = np.max(beta*conv); exp = np.exp(beta*conv - mval)
            scaled = beta * conv
            max_scaled = np.max(scaled)
            exp_scaled = np.exp(scaled - max_scaled)
            sum_exp = np.sum(exp_scaled)
            p = exp_scaled / sum_exp
            # grad wrt a: 2 * conv_rev * p
            rev_a = a[::-1]
            # convolve rev_a and p using FFT
            # need convolution of rev_a with p (full length). Using _conv_fft.
            grad_a = 2.0 * _conv_fft(rev_a, p)[n-1:n-1+n]
            # gradient wrt logits: a * (grad_a - dot(grad_a, a))
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            t += 1
            m = 0.9 * m + (1 - 0.9) * grad_v
            v2 = 0.999 * v2 + (1 - 0.999) * (grad_v ** 2)
            m_hat = m / (1 - 0.9 ** t)
            v2_hat = v2 / (1 - 0.999 ** t)
            # learning rate schedule: base_lr / sqrt(t) maybe.
            # Use base_lr = 0.1 or adapt.
            effective_lr = lr / (np.sqrt(t) + 1e-8)
            v = v - effective_lr * m_hat / (np.sqrt(v2_hat) + 1e-8)
        # end for itr
    return _softmax(v)

"""
Propose a non‑negative step function (list of heights) that lowers the
objective 2·n·max(conv(a,a)) /(sum(a)**2).

The search works in three phases:

1. **Seed pool** – a diverse collection of ℓ₁‑normalised vectors is built
   from analytic families (uniform, triangular, Gaussian, sine),
   Dirichlet draws with several concentrations and a few shuffled variants.
   The cheap surrogate 2·n·max(conv) (exact for ℓ₁‑normalised vectors) ranks
   the seeds and the best few are kept.

2. **Refinement** – each top seed is first polished by an L‑BFGS‑B optimiser
   on a smooth log‑sum‑exp approximation of the maximum convolution.
   Afterwards a custom Adam optimiser (log‑softmax parameters) runs with a
   schedule of increasing “temperature’’ β, quickly driving the distribution
   towards a flatter pair‑sum histogram.

3. **Fine‑grained polishing** – the best vector from phase 2 is fed to a
   long projected sub‑gradient descent on the exact max‑convolution.
   The sub‑gradient descent keeps the iterate on the probability simplex,
   shrinking the step‑size as 1/√t.

All phases constantly respect the global deadline (default 1000 s).  The
best sequence found is stored in the global variable ``height_sequence_1``
and returned as a plain list of floats.
"""

# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via a real FFT (O(n log n)).
    Returns an array of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic / handcrafted families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_halfwave(n: int) -> np.ndarray:
    """A smooth half‑sine bump, zero at the ends, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Permutation helpers for richer seed pool
# ----------------------------------------------------------------------
def _alternating_order(vec: np.ndarray) -> np.ndarray:
    """Place sorted values alternating from the left and right ends."""
    sorted_desc = np.sort(vec)[::-1]
    n = len(vec)
    out = np.empty_like(vec)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out


def _shuffle_by_step(sorted_desc: np.ndarray, step: int) -> np.ndarray:
    """Modular step permutation (step must be coprime to n)."""
    n = len(sorted_desc)
    out = np.empty(n, dtype=np.float64)
    pos = 0
    for val in sorted_desc:
        out[pos] = val
        pos = (pos + step) % n
    return out

# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diversified list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.12, 0.18, 0.25):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_sine_halfwave(n))

    # Random Dirichlet draws with several concentrations, plus shuffles
    conc_list = (0.05, 0.2, 0.5, 1.0, 2.0)
    step_list = [3, 7, 11, 13, 17, 19, 23, 29, 31, 37]  # all coprime to 4000
    for conc in conc_list:
        for _ in range(3):
            raw = _random_dirichlet(n, conc, rng)

            seeds.append(raw)                                    # raw
            seeds.append(np.sort(raw)[::-1])                     # descending
            seeds.append(_alternating_order(raw))                # alternating
            for step in step_list[:3]:                           # a few modular shuffles
                seeds.append(_shuffle_by_step(np.sort(raw)[::-1], step))
            # a random permutation
            perm = rng.permutation(n)
            seeds.append(raw[perm])

    # Insert the previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise robustly (protect against NaNs)
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0 and not np.isnan(s):
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (log‑sum‑exp smooth max)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smooth max‑conv."""
    if _opt is None:  # pragma: no cover
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0]

    # Work in log‑softmax space – this automatically keeps the iterate on the simplex
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            """Smooth‐max objective and its gradient w.r.t. logits."""
            a = _softmax(v_local)                     # stays on simplex
            conv = _conv_fft(a, a)                    # full conv (len 2n‑1)
            # smooth max = (1/β) log Σ exp(β·conv)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp                    # soft weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 2n‑1
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax (logits)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule=None,
                   iters_per_beta: int = 6000,
                   base_lr: float = 0.02) -> np.ndarray:
    """
    Adam optimiser on log‑softmax parameters.
    Uses a schedule of increasing β (sharpens the smooth max).
    Returns the final ℓ₁‑normalised probability vector.
    """
    if beta_schedule is None:
        beta_schedule = [5.0, 20.0, 100.0, 500.0, 2000.0,
                         10000.0, 50000.0, 200000.0]

    # Initialise logits
    logits = np.log(start_vec + 1e-30)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    t = 0
    best_seq = _softmax(logits)
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break

            a = _softmax(logits)                     # probability vector
            conv = _conv_fft(a, a)                    # full convolution

            # soft‑max over β·conv (stable)
            scaled = beta_val * conv
            m_scaled = np.max(scaled)
            exp_scaled = np.exp(scaled - m_scaled)
            sum_exp = exp_scaled.sum()
            w = exp_scaled / sum_exp                  # weights for gradient

            # gradient w.r.t. a
            rev_a = a[::-1]
            grad_a = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_logits = a * (grad_a - dot)

            # Adam update
            t += 1
            m = 0.9 * m + 0.1 * grad_logits
            v = 0.999 * v + 0.001 * (grad_logits ** 2)
            m_hat = m / (1.0 - 0.9 ** t)
            v_hat = v / (1.0 - 0.999 ** t)

            lr_t = base_lr / (math.sqrt(t) + 1e-8)
            logits -= lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)

            # occasionally record best surrogate value
            if t % 5000 == 0:
                cur = _objective_surrogate(_softmax(logits))
                if cur < best_val:
                    best_val = cur
                    best_seq = _softmax(logits).copy()

    # Return the best encountered (sometimes the final logits are slightly worse)
    return best_seq.copy()


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000) -> np.ndarray:
    """
    Long projected sub‑gradient descent on the exact max‑convolution.
    Works directly on the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]        # exact surrogate for ℓ₁‑normed a

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry (ordered pair contribution)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step‑size
        lr = 0.35 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

    return best_seq

        if time.time() < deadline:
            remaining = deadline - time.time()
            iters_per_beta = int(min(8000, remaining * 0.03 * 1e5))
            iters_per_beta = max(2500, iters_per_beta)
            seq_adam = _adam_optimize(global_best_seq,
                                      n,
                                      deadline,
                                      beta_schedule=[5.0, 20.0, 100.0, 500.0,
                                                     2000.0, 10000.0, 50000.0],
                                      iters_per_beta=iters_per_beta,
                                      base_lr=0.02)
            val_adam = _objective_surrogate(seq_adam)
            print(f"[Adam] surrogate after polishing = {val_adam:.6f}")
            if val_adam < global_best_val:
                global_best_seq = seq_adam.copy()
                global_best_val = val_adam
                print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long projected sub‑gradient descent (exact surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub_iters = int(min(800_000, remaining * 0.04 * 1e5))
        if max_sub_iters > 0:
            print("\n--- Long sub‑gradient descent (exact surrogate) ---")
            seq_sub = _subgradient_opt(global_best_seq,
                                       n,
                                       deadline,
                                       max_iters=max_sub_iters)
            val_sub = _objective_surrogate(seq_sub)
            print(f"[SubGrad] final surrogate = {val_sub:.6f}")
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
