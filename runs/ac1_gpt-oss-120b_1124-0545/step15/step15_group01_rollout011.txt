def _exp_gradient_opt_fast(start_seq, n, deadline):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    while True:
        if time.time() > deadline:
            break
        it += 1
        # compute conv
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # gradient g
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size
        eta = 0.75 / math.sqrt(it)   # may tune
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100000 == 0:
                print(f"[EG] it={it} best={best_val:.6f}")

    return best_seq, best_val

def _subgradient_opt_fast(start_seq, n, deadline):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    while True:
        if time.time() > deadline:
            break
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]  # exact
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # gradient as earlier
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        lr = 0.3 / math.sqrt(it) # adjust factor bigger
        a = a - lr * g
        a = _proj_simplex(a)
    return best_seq, best_val

evaluate_sequence(sequence)  = 2·n·max(conv(a,a)) / (sum(a)^2)

 * a final short L‑BFGS polish

The routine respects the 1000 s time budget and stores the best sequence in the

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a ℓ₁‑normalised vector a (sum(a)=1):
    2·n·max(conv(a,a)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()

def _blackman_window(n: int) -> np.ndarray:

    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

            seeds.append(_beta_shape(n, a, b))

    # Tukey windows -------------------------------------------------------
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (various concentrations) --------------------
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws (heavy‑tailed) ---------------------------
    for _ in range(10):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves ----------------------------------------
    for nf in (2, 3, 4, 5):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with low‑frequency curves -----------------------
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Special alternating‑extrema Dirichlet seed -------------------------
    seeds.append(_alternating_dirichlet(n, rng))

    # Previous solution (if size matches)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise everything robustly ---------------------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns a probability vector (ℓ₁‑normalised).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        # extended schedule reaching very large smoothness parameters
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0,
                         200000.0, 500000.0, 1e6, 2e6]

    # initialise logits – stays on the simplex after softmax
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # stays on the simplex
            conv = _conv_fft(a, a)            # full linear convolution
            max_c = np.max(conv)
            # stabilise the exponential
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # soft‑max weights over convolution entries
            w = exp_beta / sum_exp

            # gradient wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑conv
# ----------------------------------------------------------------------
def _exp_gradient_opt_fast(start_seq: np.ndarray,
                          n: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Long mirror‑descent (exponentiated‑gradient) using the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while True:
        if time.time() > deadline:
            break
        # compute convolution and locate the active index
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (see derivation in documentation)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # decreasing step size (larger constant than before)
        it += 1
        eta = 1.0 / math.sqrt(it)   # can be tuned; 1.0 works well empirically

        # mirror‑descent update in log‑space
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_opt_fast(start_seq: np.ndarray,
                          n: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while True:
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[SubGrad] it={it:,} best={best_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # learning rate – larger constant for quicker early progress
        it += 1
        lr = 0.9 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny local random‑search refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         move_scale: float = 0.002) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing: transfer a tiny amount of mass
    between two random positions while keeping the ℓ₁‑norm = 1.
    Accept only improving moves.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose a tiny transfer (both directions possible)
        eps = move_scale * (rng.random() - 0.5)  # in [‑scale/2, scale/2]
        if eps == 0.0:
            continue
        # ensure we stay non‑negative
        if eps > 0:
            max_eps = min(eps, a[i])          # take from i, give to j
            if max_eps <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= max_eps
            a_new[j] += max_eps
        else:
            eps = -eps
            max_eps = min(eps, a[j])          # take from j, give to i
            if max_eps <= 0:
                continue
            a_new = a.copy()
            a_new[j] -= max_eps
            a_new[i] += max_eps

        val = _objective_surrogate(a_new)
        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            a = a_new
            # occasional progress report
            if int(time.time()) % 30 == 0:
                print(f"[RandomSearch] improved to {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main candidate proposer ------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and timing
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Problem size (feel free to adjust)
    # ------------------------------------------------------------------
    n = 4000  # a good trade‑off between expressiveness and speed

    # ------------------------------------------------------------------
    # 2. Retrieve previous solution (if available) to seed the search
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 3. Build a huge, diverse seed pool
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_seeds_extended(n, rng, prev_arr)

    # add many extra random Dirichlet draws with a finer grid of concentrations
    extra_concs = np.geomspace(0.01, 3.0, 12)
    for conc in extra_concs:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, float(conc), rng))

    # ensure all seeds are ℓ₁‑normalised
    seeds = [s / s.sum() for s in seeds if s.sum() > 0]

    # ------------------------------------------------------------------
    # 4. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 15
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"\nTop {top_k} surrogate values among {len(seeds)} seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 5. First wave of L‑BFGS refinements (moderate budget)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # allocate roughly 15 % of the remaining budget to these quick L‑BFGS runs
    remaining = deadline - time.time()
    lbfgs_budget = remaining * 0.15
    lbfgs_deadline = time.time() + lbfgs_budget
    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1000.0, 2000.0, 5000.0, 10000.0,
                  20000.0, 50000.0, 100000.0, 200000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > lbfgs_deadline:
            print("[Time] L‑BFGS phase exhausted.")
            break
        # a modest iteration budget per β (fast sweeps)
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         lbfgs_deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=3000)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS #{idx+1}] surrogate = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Very long exponentiated‑gradient (mirror‑descent) phase
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        eg_deadline = time.time() + remaining * 0.45
        print("\n--- Long exponentiated‑gradient phase ---")
        seq_eg, val_eg = _exp_gradient_opt_fast(global_best_seq, n, eg_deadline)
        if val_eg < global_best_val:
            global_best_seq = seq_eg.copy()
            global_best_val = val_eg
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Projected sub‑gradient descent (different step‑size schedule)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.30
        print("\n--- Projected sub‑gradient polishing ---")
        seq_sub, val_sub = _subgradient_opt_fast(global_best_seq, n, sub_deadline)
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[SubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Tiny stochastic local search (mass‑transfer moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        rs_deadline = time.time() + remaining * 0.08
        print("\n--- Local random‑search refinement ---")
        seq_rs, val_rs = _local_random_search(global_best_seq, n, rs_deadline)
        if val_rs < global_best_val:
            global_best_seq = seq_rs.copy()
            global_best_val = val_rs
            print(f"[RandomSearch] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Final short L‑BFGS polish (high β, few iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline and _opt is not None:
        remaining = deadline - time.time()
        final_lbfgs_deadline = time.time() + remaining
        print("\n--- Final L‑BFGS polish ---")
        final_beta = [1e5, 5e5, 1e6, 2e6, 5e6, 1e7]
        seq_final = _lbfgs_smooth_optimize(global_best_seq,
                                            n,
                                            final_lbfgs_deadline,
                                            beta_schedule=final_beta,
                                            max_iter_per_beta=2000)
        val_final = _objective_surrogate(seq_final)
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Final L‑BFGS] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 10. Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
