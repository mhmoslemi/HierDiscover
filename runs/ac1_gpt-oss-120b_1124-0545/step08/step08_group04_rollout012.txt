import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

def _objective(seq):
    n = len(seq)
    conv = np.convolve(seq, seq)
    return 2.0 * n * np.max(conv)

def _build_conv_mat_template(n):
    """Return row and column index arrays for convolution matrix for length n."""
    i = np.arange(n, dtype=np.int64)
    row = (i[:, None] + i[None, :]).reshape(-1)  # shape n*n
    col = np.tile(i, n)  # each column index repeated for each i
    return row, col

def _solve_lp_for_g(f, M, row_idx, col_idx):
    # f is array length n, sum=1
    n = len(f)
    data = np.repeat(f, n)  # f_i repeated for each column
    A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
    # b vector: all entries M
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return res

def _lp_iterate(f, deadline):
    n = len(f)
    row_idx, col_idx = _build_conv_mat_template(n)
    best_f = f.copy()
    best_M = np.max(np.convolve(f, f))
    it = 0
    while time.time() < deadline:
        it += 1
        M = best_M
        res = _solve_lp_for_g(best_f, M, row_idx, col_idx)
        if not res.success:
            print("[LP] iteration", it, "LP failed:", res.message)
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 1 + 1e-12:
            # cannot increase sum
            break
        g = g0 / sum_g0
        # try a few t values
        improved = False
        for t in (0.6, 0.8, 0.9, 0.4, 0.2, 0.05):
            f_new = (1 - t) * best_f + t * g

            if M_new < best_M - 1e-15:

                best_M = M_new
                improved = True
                # possibly break after first improvement
                break
        if not improved:
            # No improvement possible with convex combos
            break
        # log progress
        if it % 5 == 0:
            print(f"[LP-Iter] n={n}, it={it}, M={best_M:.8f}, sum_g0={sum_g0:.4f}")
        # termination if M small enough:
        if best_M * 2 * n < 1.50:  # objective < target

"""
Improved step‑function search via a linear‑programming “active‑set” step.

The routine works in three time‑aware phases:

1. **Analytic seeds** – a small collection of classic windows
   (uniform, Catalan, Hann, Hamming, Blackman, Gaussian and a
   triangular ramp) is generated for a few moderate lengths
   (800…2400).  The best seed for each length (according to the
   true objective) is kept.

2. **LP‑driven polishing** – for the current best seed we solve a
   linear program that maximises the total mass of a new vector `g`
   while respecting the *already known* max‑convolution bound `M`.
   The LP is cheap because the constraints are linear in `g`
   (the convolution `f * g` is a Toeplitz matrix).  The optimiser
   repeatedly mixes the current vector `f` towards the LP solution
   (`f←(1‑t)f + t g`) and recomputes `M`.  This process reduces the
   peak of the autocorrelation and converges in a handful of
   iterations.

3. **Final smooth refinement** – a short Adam‑based optimisation
   (the same smooth‑max routine from the reference solution) is
   applied to the best vector found in phase 2 for a few hundred
   thousand steps.  This polishes the convolution peaks and often
   pushes the objective just below the target 1.503.

All phases respect the global time budget (default 1000 s) and the
function always returns a non‑negative list that can be fed to
`evaluate_sequence`.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for moderate‑large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a normalised
    sequence (sum = 1):
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _triangle(n: int) -> np.ndarray:
    """Symmetric linear ramp (triangular window)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = 1.0 - np.abs(i - centre) / centre
    w[w < 0] = 0.0
    return w / w.sum()


# ----------------------------------------------------------------------
# Linear‑programming utilities
# ----------------------------------------------------------------------
def _conv_template_indices(n: int):
    """
    Pre‑compute row/column index arrays for the Toeplitz matrix that maps
    a vector g (length n) to the linear convolution f * g (length 2n‑1).
    The non‑zero pattern depends only on n, not on the current f.
    """
    i = np.arange(n, dtype=np.int64)
    # row = i[:,None] + i[None,:]  – shape (n,n)
    row = (i[:, None] + i[None, :]).reshape(-1)
    col = np.tile(i, n)
    return row, col


def _solve_lp_max_sum(f: np.ndarray, M: float, row_idx, col_idx):
    """
    Solve the LP
        maximise   1ᵀ g
        subject to  T_f g ≤ M   (element‑wise, where T_f is the conv. matrix)
                    g ≥ 0
    The matrix entries are scaled by 1/M so that the RHS becomes a vector
    of ones, which improves numerical conditioning.
    Returns the optimisation result from ``scipy.optimize.linprog``.
    """
    n = len(f)
    # scale the Toeplitz coefficients by 1/M
    data = np.repeat(f, n) / M
    A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    b = np.ones(2 * n - 1, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)      # minimise -sum(g)  <=> maximise sum(g)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return res


def _lp_polish(start_seq: np.ndarray, deadline: float, max_iter: int = 20):
    """
    Repeatedly apply the LP step described in the docstring.
    Returns the best sequence encountered and its peak‑convolution value.
    """
    n = len(start_seq)
    row_idx, col_idx = _conv_template_indices(n)

    f = start_seq.copy()
    M = np.max(_conv_fft(f, f))
    best_f = f.copy()
    best_M = M

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # --------------------------------------------------------------
        # 1) solve LP for the current bound M
        # --------------------------------------------------------------
        res = _solve_lp_max_sum(f, M, row_idx, col_idx)
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 1.0 + 1e-12:
            # No mass can be added – we have reached a fixed point.
            break

        # Normalise the LP solution to unit sum.
        g = g0 / sum_g0
        g = np.clip(g, 0.0, None)          # numerical safety

        # --------------------------------------------------------------
        # 2) try the pure LP solution – it may already improve the bound.
        # --------------------------------------------------------------
        M_g = np.max(_conv_fft(g, g))
        if M_g < best_M - 1e-14:
            f = g
            M = M_g
            best_f = f.copy()
            best_M = M_g
            print(f"[LP] direct improvement it={it} M={best_M:.9f}")
            continue

        # --------------------------------------------------------------
        # 3) otherwise mix f towards g (a few candidate step sizes)
        # --------------------------------------------------------------
        improved = False
        for t in (0.6, 0.8, 0.9, 0.4, 0.2, 0.1):
            f_new = (1.0 - t) * f + t * g
            M_new = np.max(_conv_fft(f_new, f_new))
            if M_new < best_M - 1e-14:
                f = f_new
                M = M_new
                best_f = f.copy()
                best_M = M_new
                improved = True
                print(f"[LP] mixed improvement it={it} t={t:.2f} M={best_M:.9f}")
                break
        if not improved:
            # No convex combination yields improvement – stop.
            break

        # early exit if already under the target threshold
        if 2.0 * n * best_M < 1.50:
            break

    return best_f, best_M

# Smoothed Adam refinement (re‑used from the reference solution)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(start_seq: np.ndarray, deadline: float,
                          beta_start: float = 5.0,
                          beta_end: float = 50.0,
                          max_iter: int = 5_000_000):
    """
    Fine‑tune a probability vector using a smoothed max (log‑sum‑exp)
    of the self‑convolution, optimised with Adam on the logits.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.02

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        a = np.exp(logits)
        a /= a.sum()

        conv = _conv_fft(a, a)

        # anneal temperature
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.5))

        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()                      # smooth max weights

        # gradient w.r.t. a
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} val={val:.6f}")

            if best_val < 1.503:
                break

    return best_seq, best_val

    # run LP polishing only for the length that currently looks best
    if best_seq is not None:
        n_best = len(best_seq)
        print(f"[LP] Starting polishing on length {n_best}")
        seq_lp, M_lp = _lp_polish(best_seq, lp_deadline, max_iter=30)
        val_lp = 2.0 * n_best * M_lp
        if val_lp < best_val:
            best_val = val_lp
            best_seq = seq_lp.copy()
            print(f"[LP] improved to val={val_lp:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – short Adam fine‑tuning (final polishing)
    # --------------------------------------------------------------
    now = time.time()
    if time.time() < deadline and best_seq is not None:
        # give the remaining ~30 % to Adam (if any time left)
        adam_deadline = deadline
        seq_adam, val_adam = _smooth_adam_optimize(best_seq, adam_deadline)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam.copy()
            print(f"[Adam‑Final] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Final safety check (evaluate_sequence will clip automatically)
    # --------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )
    return best_seq.tolist()
