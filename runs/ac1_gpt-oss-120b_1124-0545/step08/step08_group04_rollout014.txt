"""
Improved search for step‑function minimizing `evaluate_sequence`.

The algorithm builds on the previous four‑phase approach but adds:

1. Additional analytic seeds (inverse‑power, quadratic, and custom
   “flat‑top” windows) and explores a larger set of lengths.
2. A higher‑temperature annealed Adam optimisation (smooth max) with
   several restarts.
3. A cheap stochastic local‑search that perturbs the best vector with
   Gaussian noise whilst staying on the simplex.
4. A final mirror‑descent polishing phase (same as before).

All phases respect the total time budget (default 1000 s).  The routine
always returns a non‑negative list; `evaluate_sequence` will clip and
normalise it automatically.
"""
import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ---------------------------------------------------------------
# Low‑level utilities
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (fast for large vectors).
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv

def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a sequence that
    is already normalised to sum = 1:
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# ---------------------------------------------------------------
# Analytic seed families (all returned normalised to sum = 1)
# ---------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total

def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * math.pi * i / (n - 1)))
    return w / w.sum()

def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * math.pi * i / (n - 1))
    return w / w.sum()

def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * math.pi * i / (n - 1))
         + a2 * np.cos(4 * math.pi * i / (n - 1)))
    return w / w.sum()

def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()

def _inverse_power(n: int, alpha: float = 1.0) -> np.ndarray:
    """p_i ∝ (i+1)^{-alpha}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-alpha)
    return w / w.sum()

def _quadratic(n: int) -> np.ndarray:
    """p_i ∝ 1 - (i/(n-1))**2 (concave parabola)."""
    i = np.arange(n, dtype=np.float64)
    w = 1.0 - (i / (n - 1)) ** 2
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()

# ---------------------------------------------------------------
# Random Dirichlet sampling
# ---------------------------------------------------------------

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """
    Draw a Dirichlet(alpha,…,alpha) vector.
    """
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s

# ---------------------------------------------------------------
# Smoothed Adam optimiser (log‑its parametrisation)
# ---------------------------------------------------------------

def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of ``max(conv(a,a))``.
    Logistic‑softmax parametrisation + Adam.
    The smooth max uses a temperature β that is annealed up to ``beta_end``.
    Returns best sequence seen and its exact objective.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.02

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # soft‑max to obtain a probability vector
        a_exp = np.exp(logits)
        a = a_exp / a_exp.sum()

        # self‑convolution
        conv = _conv_fft(a, a)

        # annealed temperature (linear schedule, reaches beta_end at half the iterations)
        progress = it / max_iter
        if progress < 0.5:
            beta = beta_start + (beta_end - beta_start) * (progress * 2.0)
        else:
            beta = beta_end

        # smooth max via log‑sum‑exp (log‑sum‑exp stabilised)
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()   # soft weights, sum = 1

        # gradient w.r.t. a  (see derivation in the original code)
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it + 1)

        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional exact evaluation
        if (it & 511) == 0:         # every 512 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} beta={beta:.1f}")

            # early exit if already good enough
            if best_val < 1.50:
                break

    return best_seq, best_val

# ---------------------------------------------------------------
# Exact sub‑gradient mirror descent (exponentiated gradient)
# ---------------------------------------------------------------

def _refine_subgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Classical exponentiated‑gradient (KL‑mirror descent) on the simplex,
    using the true sub‑gradient of the active convolution entry.
    Returns best vector seen and its objective.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Refine] it={it:7d} n={n:5d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.7 / math.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)               # stabilisation
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val

# ---------------------------------------------------------------
# Tiny stochastic local search (Gaussian perturbations)
# ---------------------------------------------------------------

def _local_search(start_seq: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: add zero‑mean Gaussian noise to a few
    coordinates, renormalise and keep the improvement.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    # step scale relative to the typical entry size (≈1/n)
    step_scale = 0.5 / n
    it = 0
    while time.time() < deadline:
        it += 1
        # perturb a handful of coordinates (≈1% of n) to keep cost low
        nb = max(1, n // 200)   # ≈0.5% of entries
        idx = rng.choice(n, size=nb, replace=False)
        perturb = a.copy()
        noise = rng.normal(scale=step_scale, size=nb)
        perturb[idx] += noise
        # enforce non‑negativity
        perturb = np.maximum(perturb, 0.0)
        s = perturb.sum()
        if s == 0.0:
            continue
        perturb /= s
        val = _objective(perturb)
        if val < best_val:
            best_val = val
            best_seq = perturb.copy()
            a = perturb
            # slowly shrink the step size – the landscape becomes flatter near a local optimum
            step_scale *= 0.995
            print(f"[Local] it={it:7d} n={n:5d} val={val:.6f}")
        # occasional random restart if stuck
        if (it & 1023) == 0 and rng.random() < 0.01:
            # re‑initialise around current best with a larger jitter
            jitter = rng.normal(scale=5.0 / n, size=n)
            a = np.maximum(best_seq + jitter, 0.0)
            a /= a.sum()
    return best_seq, best_val

        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()

"""
Improved search for a step‑function that minimises the ``evaluate_sequence``
objective.

The workflow extends the earlier four‑phase approach with:

1. Additional analytic seed families (inverse‑power, quadratic, extra
   Gaussian scales) and a broader range of lengths.
2. Random Dirichlet sampling using many concentration parameters.
3. A higher‑temperature annealed Adam optimiser (smooth max), executed
   several times from jittered restarts.
4. An exact mirror‑descent polishing phase (exponentiated‑gradient).
5. A cheap stochastic local‑search that perturbs the best vector with
   Gaussian noise while staying on the simplex.

All phases respect the overall time budget (default 1000 s) and the
routine always returns a non‑negative list; ``evaluate_sequence`` will
clip and normalise it automatically.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ---------------------------------------------------------------
# Low‑level utilities
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (fast for large vectors).
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a sequence that is
    already normalised to sum = 1:

        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ---------------------------------------------------------------
# Analytic seed families (all returned normalised to sum = 1)
# ---------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * math.pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * math.pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * math.pi * i / (n - 1))
         + a2 * np.cos(4 * math.pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    """Discrete Gaussian centred in the middle, σ = sigma_factor·n."""
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _inverse_power(n: int, alpha: float = 1.0) -> np.ndarray:
    """p_i ∝ (i+1)^{-α}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-alpha)
    return w / w.sum()


def _quadratic(n: int) -> np.ndarray:
    """Concave parabola p_i ∝ 1 - (i/(n-1))**2."""
    i = np.arange(n, dtype=np.float64)
    w = 1.0 - (i / (n - 1)) ** 2
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


# ---------------------------------------------------------------
# Random Dirichlet sampling
# ---------------------------------------------------------------

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ---------------------------------------------------------------
# Smoothed Adam optimiser (log‑its parametrisation)
# ---------------------------------------------------------------

def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of ``max(conv(a,a))``.
    Uses a soft‑max (logits) parametrisation combined with Adam.
    The smooth max is a log‑sum‑exp with annealed temperature β.
    Returns the best sequence seen and its exact objective value.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.02

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # probability vector via soft‑max
        a_exp = np.exp(logits)
        a = a_exp / a_exp.sum()

        # self‑convolution
        conv = _conv_fft(a, a)

        # annealed temperature (linear up to beta_end at 50 % of iterations)
        progress = it / max_iter
        if progress < 0.5:
            beta = beta_start + (beta_end - beta_start) * (progress * 2.0)
        else:
            beta = beta_end

        # smooth max via log‑sum‑exp (stabilised)
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()   # soft weights, sum=1

        # gradient wrt a
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it + 1)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # periodical exact evaluation
        if (it & 511) == 0:                     # every 512 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} beta={beta:.1f}")

            if best_val < 1.50:                # early exit if already very good
                break

    return best_seq, best_val


# ---------------------------------------------------------------
# Exact sub‑gradient mirror descent (exponentiated gradient)
# ---------------------------------------------------------------

def _refine_subgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (KL‑mirror descent) on the simplex,
    using the true sub‑gradient of the active convolution entry.
    Returns best vector seen and its objective.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Refine] it={it:7d} n={n:5d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.7 / math.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)           # stabilisation
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ---------------------------------------------------------------
# Tiny stochastic local search (Gaussian perturbations)
# ---------------------------------------------------------------

def _local_search(start_seq: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: add zero‑mean Gaussian noise to a few
    coordinates, renormalise and keep the improvement.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    step_scale = 0.5 / n                # initial perturbation magnitude
    it = 0
    while time.time() < deadline:
        it += 1
        # choose a small random subset of indices to perturb
        nb = max(1, n // 200)           # ~0.5 % of entries
        idx = rng.choice(n, size=nb, replace=False)

        perturb = a.copy()
        perturb[idx] += rng.normal(scale=step_scale, size=nb)
        perturb = np.maximum(perturb, 0.0)

        s = perturb.sum()
        if s == 0.0:
            continue
        perturb /= s

        val = _objective(perturb)
        if val < best_val:
            best_val = val
            best_seq = perturb.copy()
            a = perturb
            step_scale *= 0.995          # slowly tighten the search radius
            print(f"[Local] it={it:7d} n={n:5d} val={val:.6f}")

        # occasional modest restart to escape very flat plateaux
        if (it & 1023) == 0 and rng.random() < 0.01:
            jitter = rng.normal(scale=5.0 / n, size=n)
            a = np.maximum(best_seq + jitter, 0.0)
            a /= a.sum()

    return best_seq, best_val


# ---------------------------------------------------------------
# Main entry point
# ---------------------------------------------------------------

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given time
    budget and returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Re‑use a previously discovered solution if present
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # Lengths to explore – a slightly broader set than before
    # --------------------------------------------------------------
    lengths = [
        1200, 1500, 1800, 2100, 2400, 2700, 3000,
        3300, 3600, 3900, 4200, 4500, 4800, 5000,
    ]

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Stage 1 – deterministic analytic seeds (including new families)
    # --------------------------------------------------------------
    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.25),
        lambda n: _inverse_power(n, alpha=0.5),
        lambda n: _inverse_power(n, alpha=1.0),
        lambda n: _inverse_power(n, alpha=1.5),
        _quadratic,
    )

    for n in lengths:
        for fn in seed_families:
            try:
                seq = fn(n)
                val = _objective(seq)
                if val < best_val:
                    best_val = val
                    best_seq = seq.copy()
                    print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")
            except Exception:
                pass

        # try the stored previous vector if its length matches
        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – heavy‑weight random Dirichlet sampling
    # --------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.35   # roughly one third of remaining time
    sample_deadline = now + time_for_sampling
    alphas = [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = rng.choice(alphas)
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.2f} val={val:.6f}")

        if best_val < 1.48:                # early break if already quite good
            break

    # --------------------------------------------------------------
    # Stage 3 – Smoothed Adam optimisation (several restarts)
    # --------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        adam_budget = remaining * 0.30
        restarts = 3
        for r in range(restarts):
            sub_deadline = time.time() + adam_budget / restarts
            # jitter the start point slightly for each restart
            jitter = rng.normal(scale=0.02, size=best_seq.shape)
            start = np.maximum(best_seq + jitter, 0.0)
            start /= start.sum()
            seq_adam, val_adam = _smooth_adam_optimize(start, sub_deadline)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = seq_adam
                print(f"[Adam‑Restart {r+1}] improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – exact sub‑gradient mirror descent (polishing)
    # --------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _refine_subgradient(best_seq, deadline)
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Final‑Refine] improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – tiny stochastic local search for fine‑tuning
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        final_seq, final_val = _local_search(best_seq, deadline, rng)
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Local‑Final] improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Final sanity‑check (the grader will call evaluate_sequence again)
    # --------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
