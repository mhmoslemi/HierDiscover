Key ideas
---------

   peak to shrink roughly as 1/n while the objective 2·n·max(conv) stays
   comparable; empirical evidence shows the constant factor moves

2. Rich family of parametrised seeds:

   * sparse random supports and deterministic Sidon‑like sets.
   * an up‑scaled version of the previously best construction.

3. Two main continuous‑optimisation phases

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective (ignoring the trivial clipping of evaluate_sequence).
    seq is assumed non‑negative and summed to 1.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window_general(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1).  α>0.
    Edge‑heavy for α<1, bell‑shaped for α>1.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _laplace_window(n: int, scale: float) -> np.ndarray:
    """Symmetric Laplace (double‑exponential) shape."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-np.abs(i - centre) / scale)
    return a / a.sum()


def _generalized_normal(n: int, p: float, sigma_factor: float) -> np.ndarray:
    """
    a_i ∝ exp( - ( |i-centre| / (sigma) ) ** p ),  p>0.
    sigma = sigma_factor * n.
    """
    centre = (n - 1) / 2.0
    sigma = max(sigma_factor * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-np.abs(i - centre) ** p / (sigma ** p))
    return a / a.sum()


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _smooth_random(n: int, rng: np.random.Generator, sigma_factor: float) -> np.ndarray:
    """Generate a random vector and smooth it with a Gaussian kernel."""
    raw = rng.standard_normal(n)
    # Gaussian kernel (same length, centred)
    kern = _gaussian(n, sigma_factor)
    # use FFT for the smoothing
    smoothed = _conv_fft(raw, kern)[:n]  # take first n entries (valid part)
    smoothed = np.clip(smoothed, 0.0, None)
    total = smoothed.sum()
    if total <= 0:
        return _uniform(n)
    return smoothed / total


def _sparse_random_seed(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random support of size `m`."""
    if m >= n:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """Deterministic Sidon‑like support (triangular numbers)."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _quadratic_residue_seed(n: int, m: int) -> np.ndarray:
    """Positions are i² (mod n)."""
    idx = np.mod(np.arange(m) ** 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _upscale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from old length to new_n.
    Afterwards re‑normalise to sum = 1.
    """
    old_n = seq.shape[0]
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    up = np.interp(x_new, x_old, seq)
    up = np.clip(up, 0.0, None)
    total = up.sum()
    return up / total if total > 0 else _uniform(new_n)

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate (log‑sum‑exp) of the convolution peak.
    Returns the best vector seen and its true objective.
    """
    # Use log‑probability parametrisation (soft‑max keeps us on the simplex)
    v = np.log(seed + 1e-30)

    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # smoothing parameter – start modest and grow geometrically
    beta = 1.5
    beta_mul = 1.007

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # smooth max via log‑sum‑exp
        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                # for numerical stability
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k (a*a)_k with respect to a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam moment updates
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 1000000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact non‑smooth objective (exponentiated‑gradient).
    Returns the best vector encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    w = np.zeros(2 * n - 1, dtype=np.float64)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # build sub‑gradient (average over active constraints)
        if active.size:
            w.fill(0.0)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]
        else:
            g = np.zeros_like(a)

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)            # stabilise numerics
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis–Hastings walk that swaps tiny amounts of mass between
    random positions and occasionally applies a global Gaussian jitter.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)   # initial temperature
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # up to 2 % of source mass
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # occasional larger “jump’’
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            s = new_seq.sum()
            if s == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= s
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Seed generation – a diverse pool of starting vectors
# ----------------------------------------------------------------------


def _make_seeds(n: int,
                previous: np.ndarray | None,
                rng: np.random.Generator) -> list[np.ndarray]:
    """
    Assemble a diverse set of start vectors, all normalised to sum = 1.
    """
    seeds: list[np.ndarray] = []

    # Classic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))
    for sigma_factor in np.linspace(0.08, 0.30, 7):
        seeds.append(_gaussian(n, sigma_factor))
    for alpha in np.linspace(0.55, 0.95, 9):
        seeds.append(_beta_window_general(n, alpha))

    # Extended Beta (α up to 5)
    for alpha in np.linspace(1.0, 5.0, 9):
        seeds.append(_beta_window_general(n, alpha))

    # Laplace family
    for scale_frac in np.linspace(0.05, 0.5, 8):
        seeds.append(_laplace_window(n, scale_frac * n))

    # Generalised normal (vary p and width)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        for sigma_frac in np.linspace(0.05, 0.35, 5):
            seeds.append(_generalized_normal(n, p, sigma_frac))

    # Smoothed random vectors
    for sigma_frac in np.linspace(0.03, 0.18, 4):
        for _ in range(2):
            seeds.append(_smooth_random(n, rng, sigma_frac))

    # Sparse random supports
    for m in range(30, 210, 20):          # 30,50,…,190
        for _ in range(2):
            seeds.append(_sparse_random_seed(n, m, rng))

    # Deterministic Sidon‑like constructions
    for m in (50, 70, 90, 110):
        seeds.append(_sidon_like_seed(n, min(m, n)))
        seeds.append(_quadratic_residue_seed(n, min(m, n)))

    # Upscaled previous best (if available)
    if previous is not None and previous.shape[0] > 0:
        if previous.shape[0] == n:
            seeds.append(previous / previous.sum())
        else:
            seeds.append(_upscale_sequence(previous, n))

    # Clean‑up – enforce non‑negativity and renormalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget and returns the best sequence
    discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # Retrieve the previously best construction (if any)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    rng = np.random.default_rng(seed + 12345)

    # Problem size – larger n gives more expressive power
    n = 8000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # 1) Build a rich pool of seeds
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous, rng)

    # ------------------------------------------------------------------
    # 2) Quick evaluation of all seeds – keep the top few
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = seed_vals[0][0].copy()
    global_best_val = seed_vals[0][1]
    print(f"[Seed] best initial objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam on smoothed‑max for each top seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=20000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Adam] seed={idx} improved best to {val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – long sub‑gradient refinement on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                       max_iters=800000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[SubGrad] after long run best = {val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – occasional perturbations + short sub‑gradient bursts
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # small random jitter around the current best
        perturb = global_best_seq.copy()
        noise = rng.normal(scale=0.0015, size=n)
        perturb += noise
        perturb = np.clip(perturb, 0.0, None)
        s = perturb.sum()
        if s == 0.0:
            perturb = _uniform(n)
        else:
            perturb /= s

        seq, val = _subgradient_refine(perturb, n, deadline,
                                       max_iters=250000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Perturbed] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # Phase D – final simulated‑annealing fine‑tuning (uses any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Anneal final] best = {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen but keep it for safety
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")

    # Store for the next round and return as plain list
    globals()["height_sequence_1"] = np.copy(global_best_seq)
    return global_best_seq.tolist()
