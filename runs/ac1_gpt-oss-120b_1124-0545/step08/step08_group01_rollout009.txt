"""
Non‑negative step‑function optimiser (new version).

Key ideas
---------
* Work initially with a modest length (n0≈2000) – cheap to explore many seeds.
* Refine with a pure exponentiated‑gradient (EG) optimiser on the

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (O(n log n)).
    Returns a real array of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate_objective(seq: np.ndarray) -> float:
    """
    Fast surrogate used during the search.
    ``seq`` is assumed to be non‑negative and already normalised to sum = 1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangle (peak in the centre)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two exponentials: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


def _mix(seq1: np.ndarray, seq2: np.ndarray, w: float) -> np.ndarray:
    """Linear interpolation and renormalisation."""
    mixed = w * seq1 + (1.0 - w) * seq2
    mixed = mixed / mixed.sum()
    return mixed

def _generate_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # Gaussian sweep
    for sigma in np.linspace(0.1, 0.5, 5) * n:
        seeds.append(_gaussian(n, sigma))

    # exponential decay
    for r in np.linspace(0.2, 0.9, 9):
        seeds.append(_exp_decay(n, r))

    # beta family
    for a in np.linspace(0.2, 2.0, 4):
        for b in np.linspace(0.4, 5.0, 4):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws
    for conc in (0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (3, 5, 10, 20):
        for conc in (0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # mixture of uniform & triangular
    uniform = _uniform(n)
    tri = _triangular(n)
    for w in np.linspace(0.0, 1.0, 5):
        seeds.append(_mix(uniform, tri, w))

    # previous solution if shape matches
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def _eg_optimize(start_seq: np.ndarray,
                 n: int,
                 deadline: float,
                 max_iters: int = 200000,
                 eta0: float = 0.7,
                 beta_start: float = 0.5,
                 beta_mul: float = 1.002,
                 beta_max: float = 300.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimiser on the log‑sum‑exp surrogate.
    The vector is kept on the simplex (sum = 1, non‑negative).
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_objective(a)

    beta = beta_start
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        cur_val = 2.0 * n * maxc
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[EG] it={it:6d} best surrogate={best_val:.6f}")

        # --- smoothed‑max weights -------------------------------------------------
        shift = maxc
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # --- gradient of weighted max (w.r.t. a) ----------------------------------
        rev_a = a[::-1]
        grad_full = _conv_fft(rev_a, w)      # length 2n‑1
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]  # central n entries

        # --- EG update ------------------------------------------------------------
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # symmetrise occasionally (helps stability)
        if it % 1000 == 0:
            a = (a + a[::-1]) * 0.5

        # increase smoothing temperature (capped)
        beta = min(beta * beta_mul, beta_max)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 600000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror‑descent (exponentiated‑gradient) on the true max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] it={it:7d} best surrogate={best_val:.6f}")

        # sub‑gradient of the *max* at position max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient style update
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # normalise for stability
        a = np.exp(log_a)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300000,
                         init_temp_factor: float = 0.02,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple SA that reshuffles the mass of a random subset (size 2‑10).
    Returns the best sequence and its surrogate value.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate_objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 11)              # subset size 2…10
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _surrogate_objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best surrogate={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.
    Returns the best sequence found as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # Reproducibility
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a small safety margin before the hard 1000 s kill‑switch
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Retrieve any previously found solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a rich pool of seeds (initial length n0)
    # ------------------------------------------------------------------
    n0 = 2000                     # modest length for fast exploration
    print(f"\n--- Seed generation (n={n0}) ---")
    seeds = _generate_seeds(n0, rng, previous)
    print(f"Created {len(seeds)} raw seeds.")

    # evaluate quickly with the surrogate objective (already normalised)
    evaluated = [(vec, _surrogate_objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))          # keep a few elite seeds
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # global best tracker
    global_best_seq = best_seeds[0].copy()
    global_best_val = _surrogate_objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient refinement of the elite seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- EG refinement run {idx+1}/{len(best_seeds)} ---")
        seq_eg, val_eg = _eg_optimize(
            seed_vec,
            n0,
            deadline,
            max_iters=180000,       # generous budget per seed
            eta0=0.8,
            beta_start=0.3,
            beta_mul=1.0015,
            beta_max=250.0
        )
        if seq_eg is not None and val_eg < global_best_val:
            global_best_seq = seq_eg.copy()
            global_best_val = val_eg
            print(f"[overall‑EG] new global best = {val_eg:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long exact sub‑gradient pass on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient fine‑tuning (n0) ---")
        seq_sub, val_sub = _subgradient_refine(
            global_best_seq,
            n0,
            deadline,
            max_iters=600000
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[overall‑SubGrad] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing polishing (n0)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing (n0) ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq,
            n0,
            deadline,
            max_iters=300000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – optional up‑sampling to a longer resolution
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # decide on the next upscale factor (first 2×, then 4× if time permits)
        factor = 2
        n_big = n0 * factor
        if n_big > 16000:          # do not explode the size
            break

        print(f"\n--- Upsampling to n={n_big} ---")
        # linear interpolation onto a finer grid
        x_old = np.linspace(0.0, 1.0, n0, dtype=np.float64)
        x_big = np.linspace(0.0, 1.0, n_big, dtype=np.float64)
        upsampled = np.interp(x_big, x_old, global_best_seq)
        upsampled = upsampled / upsampled.sum()

        # run a short EG phase on the larger vector
        seq_eg_big, val_eg_big = _eg_optimize(
            upsampled,
            n_big,
            deadline,
            max_iters=150000,
            eta0=0.75,
            beta_start=0.4,
            beta_mul=1.001,
            beta_max=300.0
        )
        # follow with sub‑gradient
        seq_sub_big, val_sub_big = _subgradient_refine(
            seq_eg_big,
            n_big,
            deadline,
            max_iters=300000
        )
        # final SA polish
        seq_sa_big, val_sa_big = _simulated_annealing(
            seq_sub_big,
            n_big,
            deadline,
            max_iters=200000
        )

        # pick the best of the three refined versions
        candidates = [(seq_eg_big, val_eg_big),
                      (seq_sub_big, val_sub_big),
                      (seq_sa_big, val_sa_big)]
        candidates.sort(key=lambda x: x[1])
        best_big_seq, best_big_val = candidates[0]

        if best_big_val < global_best_val:
            global_best_seq = best_big_seq.copy()
            global_best_val = best_big_val
            n0 = n_big                     # remember the new resolution
            print(f"[Upsample] new global best surrogate = {best_big_val:.6f}")

        # attempt a second upscale if time still permits
        if time.time() > deadline - 30:   # leave a bit of room for final steps
            break

    # ------------------------------------------------------------------
    # Final sanity check (call evaluate_sequence once)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )

    # expose the best vector for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
