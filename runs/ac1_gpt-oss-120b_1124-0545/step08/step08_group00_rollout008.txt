    best = f.copy()
    best_val = objective(f)
    current = f.copy()
    for iter in range(max_iters):
        if time.time()>deadline: break

        c = np.concatenate([current, np.zeros(n-1)])
        r = np.concatenate([current[:1], np.zeros(n-1)])

def lp_iterate(f, deadline, max_iters=20):
    n = len(f)
    best_f = f.copy()
    best_obj = _objective(f)
    for it in range(max_iters):
        if time.time()>deadline:
            break
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)
        # Build matrix
        A = build_conv_matrix(f)
        # Set up linprog
        c = -np.ones(n)  # minimize -sum => maximize sum
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        # Solve
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print("[LP] iteration", it, "failed", res.message)
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 * (1 + 1e-12):
            # no improvement
            break
        # Normalize
        f = g / sum_g
        cur_obj = _objective(f)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_f = f.copy()
            print(f"[LP] iter {it} improved obj={cur_obj:.6f}")

"""
Propose a step‑function that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The search consists of three stages:

1. **LP tightening (coarse length)** –  
   Starting from a uniform (or down‑sampled) distribution we solve a
   linear program  

        maximise   Σ g_i
        subject to  f * g ≤ M ,   g ≥ 0

   where `M = max(f*f)` and `f * g` is the full linear convolution.
   After solving, `g` is normalised to ℓ₁‑norm = 1, producing a new
   `f`.  Repeating this step (typically 10–12 times) provably reduces
   `M`, i.e. the max‑convolution value.

2. **Upsampling & continuous refinement** –  
   The improved coarse vector (length ≈ 2000) is up‑sampled to the target
   length 4000 by simple repetition and re‑normalised.  This candidate
   is then polished with the Adam optimiser (smooth max surrogate),
   an exact projected sub‑gradient descent, and a short simulated‑annealing
   phase – the same pipeline used in the reference solution.

3. **Result** – the best sequence found is stored in the global name
   `height_sequence_1` and returned as a plain `list[float]`.

The LP stage explores a different part of the search space (convex
tightening) and together with the later local optimisation typically
gets below the previous best bound 1.509 … .
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Surrogate objective: 2·n·max(conv(seq,seq)) assuming Σseq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (Σ=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families (used as seeds)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Place mass on a random small set of coordinates."""
    m = rng.integers(10, 200)          # number of non‑zero entries
    idx = rng.choice(n, size=m, replace=False)
    weights = rng.random(m)
    seq = np.zeros(n, dtype=np.float64)
    seq[idx] = weights
    s = seq.sum()
    return seq / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# LP refinement (coarse length)
# ----------------------------------------------------------------------


def _lp_improve(f: np.ndarray, deadline: float,
                max_iters: int = 15) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve   maximise Σ g_i   s.t.   f * g ≤ M, g ≥ 0,
    where M = max(f*f).  After each solve the vector is normalised
    to ℓ₁‑norm = 1.  Returns the best vector seen and its surrogate value.
    """
    n = f.shape[0]
    best_seq = f.copy()
    best_val = _objective(f)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # current maximal convolution value
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # Build the full convolution matrix A where (A @ g) == f * g
        first_col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
        first_row = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
        A = toeplitz(first_col, first_row)          # shape (2n‑1, n)

        # Linear program: maximise sum(g)  →  minimise -sum(g)
        c = -np.ones(n, dtype=f.dtype)
        b_ub = np.full(2 * n - 1, M, dtype=f.dtype)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            # no improvement possible
            print(f"[LP] iteration {it} gave sum(g)≈1 – stop.")
            break

        f = g / sum_g                         # re‑normalise to ℓ₁ = 1
        cur_val = _objective(f)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = f.copy()
            print(f"[LP] iter {it}: improved surrogate = {cur_val:.6f}")
        else:
            print(f"[LP] iter {it}: surrogate = {cur_val:.6f}")

    return best_seq, best_val

# Adam optimisation (smooth‑max surrogate)
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray, n: int,
                   deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray | None, float]:
    """Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits (stay on simplex via softmax)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.1            # start temperature (smoothness)
    beta_mul = 1.005
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # probabilities, Σa = 1
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # increase smoothing temperature
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp on the convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # simple clipping for stability
        max_norm = 8.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.7
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent
# ----------------------------------------------------------------------


def _subgradient_proj(start_seq: np.ndarray, n: int,
                      deadline: float,
                      max_iters: int = 2_500_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Uses an additive update with Euclidean projection onto the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the max at index max_idx:
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.2 / np.sqrt(it)      # modest diminishing step
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing polishing

def _simulated_annealing(start_seq: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 1_200_000,
                         init_temp_factor: float = 0.025,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles the mass of a random subset of coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9997, min_temp)

        k = rng.integers(2, 12)          # 2 … 11 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

        if seq_adam is not None and val_adam < val_small:
            f_large = seq_adam
            val_small = val_adam
            print(f"[Stage 3A] Adam improved to {val_adam:.6f}")

    time_left = deadline - time.time()
    if time_left > 0:
        print("\n--- Stage 3B: Projected sub‑gradient descent ---")
        t_sub_dead = time.time() + time_left * 0.45
        seq_sub, val_sub = _subgradient_proj(f_large, n_large,
                                             t_sub_dead,
                                             max_iters=3_000_000)
        if val_sub < val_small:
            f_large = seq_sub
            val_small = val_sub
            print(f"[Stage 3B] Sub‑gradient improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final stochastic polishing (simulated annealing)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        print("\n--- Stage 4: Simulated annealing polishing ---")
        t_sa_dead = time.time() + time_left * 0.90
        seq_sa, val_sa = _simulated_annealing(f_large, n_large,
                                              t_sa_dead,
                                              max_iters=1_500_000)
        if val_sa < val_small:
            f_large = seq_sa
            val_small = val_sa
            print(f"[Stage 4] SA improved to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Sanity check with the provided evaluate_sequence
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(f_large.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {val_small:.6f} "
        f"(length {len(f_large)})"
    )

    # store for potential reuse
    globals()["height_sequence_1"] = f_large.tolist()
    return f_large.tolist()
