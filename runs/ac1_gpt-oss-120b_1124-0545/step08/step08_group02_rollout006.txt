def build_low_collision_set(n, max_ordered, time_limit):
    best_set=None; best_sz=0; attempt=0.
    while time.time() < time_limit:
        mask=np.zeros(n,dtype=bool)
        sum_counts=np.zeros(2*n-1,dtype=np.uint8)
        indices=np.random.permutation(n)
        for cand in indices:
            # early check self-collisions
            if sum_counts[2*cand] + 1 > max_ordered:
                continue
            # compute sums with existing
            existing = np.where(mask)[0]
            if existing.size>0:
                sums=cand + existing
                # need to add 2 per existing (ordered)
                if np.any(sum_counts[sums] + 2 > max_ordered):
                    continue
            # Accept
            mask[cand]=True
            sum_counts[2*cand]+=1
            if existing.size>0:
                sum_counts[sums] += 2
            if mask.sum() >= best_sz: # update best later after loop, not earlier.
                pass

        cur_sz=mask.sum()
        if cur_sz>best_sz:
            best_sz=cur_sz; best_set=np.where(mask)[0]
        # optional early stop if cur_sz==n? unlikely.

    return best_set, best_sz

"""
optimise_low_collision.py

A fast time‑bounded optimiser for the “step‑function” problem.

The strategy is deliberately simple:
* Build a large *low‑collision* support S ⊂ {0,…,n‑1} such that every
  ordered pair‑sum i+j (i,j∈S) appears at most C times (C=4 by default).
  For a uniform distribution on S the maximal convolution entry is
  at most C/|S|², so the objective 2·n·max(conv)/‖a‖₁² is bounded by
  2·n·C/|S|².
* The greedy construction repeats many random permutations of {0,…,n‑1}
  and keeps the best (largest) support found before the time budget expires.
* The final candidate is the uniform distribution on the best support.
  With n=4000 and C=4 the theoretical limit is |S|≈√(8·n)≈179,
  giving an objective close to 1.0 – well below the target 1.503.
* If, for any reason, the construction does not succeed, the routine
  falls back to the previous best sequence (if any) or a random uniform
  sparse vector.

The implementation respects the 1000 s time budget, prints progress,
stores the best sequence in the global variable ``height_sequence_1``,
and returns it as a plain Python list of floats.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence``:
        2·n·max(conv(seq,seq)) / (sum(seq)²)

    The input is assumed to be non‑negative and sum(seq) > 0.
    """
    s = seq.sum()
    if s <= 0:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Greedy low‑collision (B₂[C]) set generator
# ----------------------------------------------------------------------


def _low_collision_set_fixed(n: int,
                             max_ordered: int,
                             deadline: float,
                             rng: np.random.Generator,
                             max_attempts: int = 2000) -> tuple[np.ndarray, int]:
    """
    Greedy construction of a set S ⊂ {0,…,n‑1} such that every ordered
    pair‑sum occurs at most ``max_ordered`` times.

    Returns the index array of the best set found and its size.
    The search stops when the time budget is exhausted or the theoretical
    limit (√(2·max_ordered·n)) is reached.
    """
    best_mask = None
    best_size = 0
    attempts = 0
    theoretical_limit = int(math.sqrt(2 * max_ordered * n))

    while time.time() < deadline - 0.1 and attempts < max_attempts:
        attempts += 1

        mask = np.zeros(n, dtype=bool)
        # ordered‑sum counter (range 0 … 2n‑2)
        sum_counts = np.zeros(2 * n - 1, dtype=np.uint16)

        order = rng.permutation(n)
        for cand in order:
            # self‑pair would increase count at 2*cand
            if sum_counts[2 * cand] + 1 > max_ordered:
                continue

            existing = np.nonzero(mask)[0]
            if existing.size:
                sums = cand + existing               # vectorised
                # each new unordered pair creates two ordered pairs
                if np.any(sum_counts[sums] + 2 > max_ordered):
                    continue

            # accept the candidate
            mask[cand] = True
            sum_counts[2 * cand] += 1
            if existing.size:
                sum_counts[sums] += 2

        cur_size = mask.sum()
        if cur_size > best_size:
            best_size = cur_size
            best_mask = mask.copy()
            print(f"[LowColl] attempt {attempts:4d} size {cur_size}")

        # stop early if we already hit the theoretical maximum
        if best_size >= theoretical_limit:
            break

    if best_mask is None:
        return np.array([], dtype=int), 0
    return np.where(best_mask)[0], best_size


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget and returns the best sequence
    discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Problem size – a moderate n gives a good trade‑off between speed
    # and the theoretical bound √(2·C·n).
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # Try a few collision limits (C = 4 is the baseline; larger C may
    # allow a larger support at a modest cost).
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    best_info = None
    rng = np.random.default_rng(seed + 12345)

    # allocate roughly the whole remaining budget to the set construction;
    # the inner loop checks the deadline itself.
    for C in (4, 5, 6):
        if time.time() > deadline:
            break
        print(f"[Phase] building low‑collision set with C={C}")

        # give each C the same amount of time (the builder watches the deadline)
        indices, size = _low_collision_set_fixed(
            n=n,
            max_ordered=C,
            deadline=deadline,
            rng=rng,
            max_attempts=5000,
        )
        if size == 0:
            continue

        # uniform distribution on the found support
        a = np.zeros(n, dtype=np.float64)
        a[indices] = 1.0 / size

        val = _objective(a)
        print(f"[Result] C={C:>1} |S|={size:>3} → objective {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            best_info = (C, size)

        # stop as soon as we beat the target (adds a nice early‑exit)
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – if none of the constructed sets beat the target, reuse
    # the previously found sequence (if any) or fall back to a random sparse
    # uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None or best_val >= 1.503:
        previous = globals().get("height_sequence_1")
        if previous is not None:
            try:
                prev_arr = np.asarray(previous, dtype=np.float64)
                if prev_arr.ndim == 1 and prev_arr.size > 0:
                    prev_arr = np.clip(prev_arr, 0.0, None)
                    if prev_arr.sum() > 0:
                        prev_arr /= prev_arr.sum()
                        val = _objective(prev_arr)
                        print(f"[Fallback] using previous sequence, obj={val:.6f}")
                        if val < best_val:
                            best_seq = prev_arr.copy()
                            best_val = val
            except Exception:
                pass

    if best_seq is None or best_val >= 1.503:
        # final simple fallback: a random sparse uniform support
        m_fallback = int(math.sqrt(2 * 4 * n))
        idx = rng.choice(n, size=m_fallback, replace=False)
        best_seq = np.zeros(n, dtype=np.float64)
        best_seq[idx] = 1.0 / m_fallback
        best_val = _objective(best_seq)
        print(f"[Fallback] random sparse uniform, obj={best_val:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing – ensure non‑negativity and exact normalisation
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total <= 0.0:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_seq /= total

    # ------------------------------------------------------------------
    # Final report and return
    # ------------------------------------------------------------------
    if best_info is not None:
        C_used, sup_sz = best_info
        print(
            f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
            f"(C={C_used}, support size={sup_sz}, n={n})"
        )
    else:
        print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f}")

    # store for possible reuse by a subsequent run
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
