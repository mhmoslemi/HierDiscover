def subgradient_refine2(a, n, deadline, max_iters=400000, lr_factor=1.5):

   a = a.copy()
   best = a.copy()
   best_val = _objective(a)

   eps=1e-30

   for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]  # actual objective
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()

        # compute subgradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated gradient update
        lr = lr_factor / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        # stabilization
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

   return best, best_val

"""
height_sequence optimiser – a multi‑stage search with
  • diversified analytic seeds (including refined geometric / power‑law shapes),
  • scalar‑parameter optimisation (geometric ratio, power‑law exponent),
  • Adam optimiser on a smoothed max‑convolution surrogate (fast, adaptive temperature),
  • exponentiated‑gradient mirror descent on the true max‑convolution objective
    (larger step‑size than the previous baseline),
  • random convex combinations of the best seeds,
  • optional up‑sampling + long refinement.

The routine respects a time‑budget (default 1000 s) and stores the best
vector globally as ``height_sequence_1``.  It returns a plain ``list[float]``
ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used while searching.
    ``seq`` must be non‑negative and sum to 1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic shape families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_powerlaw(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law: (i+1)^‑p + (n‑i)^‑p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p) + np.power(n - i + 1, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Parameter optimisation for one‑dimensional families
# ----------------------------------------------------------------------


def _best_sym_geom(n: int, rng: np.random.Generator) -> np.ndarray:
    """Find the geometric ratio r∈(0,1) that gives the smallest surrogate objective."""
    try:
        from scipy.optimize import minimize_scalar

        def obj(r):
            return _objective(_sym_geom(n, r))

        res = minimize_scalar(
            obj, bounds=(0.01, 0.99), method="bounded",
            options={"xatol": 1e-3, "maxiter": 30}
        )
        r_opt = res.x
        return _sym_geom(n, r_opt)
    except Exception:
        # fallback to a coarse grid if SciPy unavailable
        best = None
        best_val = float("inf")
        for r in np.linspace(0.10, 0.90, 17):
            v = _sym_geom(n, r)
            val = _objective(v)
            if val < best_val:
                best_val, best = val, v
        return best if best is not None else _uniform(n)


def _best_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Find power‑law exponent p≥0 that minimises the surrogate objective."""
    try:
        from scipy.optimize import minimize_scalar

        def obj(p):
            return _objective(_sym_powerlaw(n, p))

        res = minimize_scalar(
            obj, bounds=(0.0, 10.0), method="bounded",
            options={"xatol": 1e-2, "maxiter": 30}
        )
        p_opt = res.x
        return _sym_powerlaw(n, p_opt)
    except Exception:
        # fallback coarse grid
        best = None
        best_val = float("inf")
        for p in np.arange(0.0, 5.01, 0.25):
            v = _sym_powerlaw(n, p)
            val = _objective(v)
            if val < best_val:
                best_val, best = val, v
        return best if best is not None else _uniform(n)

def _generate_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # refined parametric families
    seeds.append(_best_sym_geom(n, rng))
    seeds.append(_best_powerlaw(n, rng))

    # beta grid – a bit finer than before
    for a in np.linspace(0.05, 2.0, 10):
        for b in np.linspace(0.05, 5.0, 10):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (different concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 5.0):
        for _ in range(4):

    # random sparse patterns
    for k in (3, 5, 10, 20):
        for conc in (0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if any and compatible length)
    if previous is not None and previous.shape == (n,):
        v = np.asarray(previous, dtype=np.float64)
        s = v.sum()
        if s > 0:
            seeds.append(v / s)

    return seeds

def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 60000) -> tuple[np.ndarray | None, float]:
    """Adam optimiser on a log‑sum‑exp surrogate.  Uses a faster temperature schedule."""
    eps = 1e-30
    v = np.log(start_vec + eps)            # logits for the softmax

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # temperature (beta) – start low, increase quickly, cap at 300
    beta = 0.2
    beta_mul = 1.02
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # increase smoothing temperature (fast)
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp on the convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k wrt a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr = 0.8 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 300000,
                        lr_factor: float = 1.5) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient mirror descent on the true max‑convolution objective.
    Larger ``lr_factor`` (default 1.5) yields more aggressive early steps.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = lr_factor / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 150000,
                         init_temp_factor: float = 0.02,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles the mass of a random small subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)           # size of the random subset (2…6)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Local optimisation pipeline for a single seed
# ----------------------------------------------------------------------


def _local_optimize(seed_vec: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Apply Adam → mirror‑descent → SA (as time permits) and return the
    best vector discovered.
    """
    # 1. Adam on smooth surrogate
    seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=50000)
    if seq_adam is None:
        seq_adam = seed_vec.copy()
        val_adam = _objective(seq_adam)

    # 2. Mirror‑descent on the true max‑convolution
    seq_sub, val_sub = _subgradient_refine(seq_adam, n, deadline,
                                           max_iters=250000,
                                           lr_factor=1.6)

    # 3. Optional simulated annealing (only if time left)
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(seq_sub, n, deadline,
                                              max_iters=120000)
        candidates = [(seq_adam, val_adam), (seq_sub, val_sub), (seq_sa, val_sa)]
    else:
        candidates = [(seq_adam, val_adam), (seq_sub, val_sub)]

    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val


# ----------------------------------------------------------------------
# Upsampling + long refinement
# ----------------------------------------------------------------------


def _upsample_and_refine(seq: np.ndarray,
                         n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Linear interpolation to double the resolution and a long mirror‑descent."""
    n_big = n * 2
    x_old = np.linspace(0.0, 1.0, n, dtype=np.float64)
    x_big = np.linspace(0.0, 1.0, n_big, dtype=np.float64)
    up = np.interp(x_big, x_old, seq)
    up = up / up.sum()

    # Allocate most of the remaining budget to this refinement
    seq_ref, val_ref = _subgradient_refine(
        up, n_big, deadline,
        max_iters=300000,
        lr_factor=1.4
    )
    return seq_ref, val_ref


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal
    ``evaluate_sequence`` value.  Returns the best sequence found as a
    plain list of floats.
    """
    # --------------------------------------------------------------
    # reproducibility
    # --------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------
    # Try to reuse a previous solution (if any)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev = np.asarray(prev, dtype=np.float64)
        except Exception:
            prev = None
    else:
        prev = None

    # --------------------------------------------------------------
    # Choose a manageable length – 4000 works well for the surrogate
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting optimisation, length = {n} ===")

    # --------------------------------------------------------------
    # Stage 1 – generate a broad pool of seeds and keep the best few
    # --------------------------------------------------------------
    seeds = _generate_seeds(n, rng, prev)
    print(f"Generated {len(seeds)} candidate seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("Top seed objectives (surrogate):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – refine each top seed locally
    # --------------------------------------------------------------
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best seed surrogate value = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Local optimisation run {idx+1}/{len(best_seeds)} ---")
        seq, val = _local_optimize(seed_vec, n, deadline)
        if val < global_best_val:
            global_best_seq, global_best_val = seq, val
            print(f"[Global] new best surrogate = {val:.6f}")

    # --------------------------------------------------------------
    # Stage 2b – explore random convex mixtures of the current top 3
    # --------------------------------------------------------------
    if time.time() < deadline:
        mix_candidates = best_seeds[:3]
        mix_trials = 12
        print(f"\n--- Exploring {mix_trials} random mixtures of top 3 seeds ---")
        for t in range(mix_trials):
            if time.time() > deadline:
                break
            weights = rng.dirichlet(np.ones(len(mix_candidates)))
            mixture = sum(w * v for w, v in zip(weights, mix_candidates))
            mixture = mixture / mixture.sum()
            seq, val = _local_optimize(mixture, n, deadline)
            if val < global_best_val:
                global_best_seq, global_best_val = seq, val
                print(f"[Global] new best (mixture) = {val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – optional up‑sampling to a finer resolution
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Upsampling to double resolution and final refinement ---")
        seq_big, val_big = _upsample_and_refine(global_best_seq, n, deadline)
        if val_big < global_best_val:
            global_best_seq, global_best_val = seq_big, val_big
            n = len(seq_big)
            print(f"[Upsample] new best surrogate = {val_big:.6f}")

    # --------------------------------------------------------------
    # Final sanity check with the official evaluator
    # --------------------------------------------------------------
    try:
        # The evaluator will rescale internally, we just ensure sum > 0
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
