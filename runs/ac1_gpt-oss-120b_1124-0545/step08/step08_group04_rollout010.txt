def _lbfgs_optimize(start_seq, beta, deadline, maxfun_per_stage=200):
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)
    # define objective with gradient
    def fun(logits):
        # compute a via softmax
        max_log = np.max(logits)
        a = np.exp(logits - max_log)
        a_sum = a.sum()
        a = a / a_sum
        conv = _conv_fft(a, a)
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        Z = exp_beta.sum()
        p = exp_beta / Z
        smooth_max = c + np.log(Z) / beta
        # gradient wrt a
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n-1:n-1+n]
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot) * (2.0 * n)  # final objective scaling
        fval = smooth_max * (2.0 * n)
        return fval, grad_logits
    # optimize using L-BFGS-B
    try:
        from scipy.optimize import minimize
    except Exception as e:
        print("scipy not available, falling back to Adam")
        return start_seq, _objective(start_seq)
    max_iter = maxfun_per_stage
    # To respect time, compute remaining time
    maxfun = max_iter
    res = minimize(fun, logits, method='L-BFGS-B', jac=True,
                   options={'maxfun': maxfun, 'maxiter': maxfun, 'ftol': 1e-12, 'disp': False})
    # retrieve final a
    logits_opt = res.x
    max_log = np.max(logits_opt)
    a_opt = np.exp(logits_opt - max_log)
    a_opt = a_opt / a_opt.sum()
    val_opt = _objective(a_opt)
    return a_opt, val_opt

"""
Propose a step‑function that minimizes ``evaluate_sequence``.

The search proceeds in four timed phases:

1. **Broad deterministic seeding** – a large collection of analytic families
   (uniform, windows, Gaussians, geometric, power‑law) is generated for several
   lengths.  All parameter variants are evaluated and the best seed is retained.

2. **Heavy‑weight random sampling** – Dirichlet vectors with a wide range of
   concentration parameters (including very sparse and very smooth cases) are
   drawn for the currently best length.  The best random sample is kept.

3. **Smooth‑maximum L‑BFGS optimisation** – the (soft) max of the
   self‑convolution is approximated by a log‑sum‑exp with an annealed temperature
   β.  For a fixed β the objective and its gradient w.r.t. the soft‑max
   (log‑its) parametrisation are given in closed form; a limited‑memory BFGS
   optimizer (SciPy) is run for a few hundred function evaluations.
   β is increased step‑wise (5 → 15 → 40 → 80 → 200) so that the optimisation
   gradually approaches the true max‑convolution.

4. **Exact sub‑gradient mirror‑descent polishing** – a final exponentiated‑gradient
   (KL‑mirror descent) run refines the solution with the true sub‑gradient of the
   active convolution entry.

The routine respects the total time budget (default 1000 s), always returns a
non‑negative list and stores the best vector in the global variable
``height_sequence_1`` for possible reuse by later runs.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ---------------------------------------------------------------
# Low‑level utilities
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (fast for large vectors).
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a sequence already
    normalised to sum = 1:
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ---------------------------------------------------------------
# Analytic seed families (returned normalised to sum = 1)
# ---------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * math.pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * math.pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * math.pi * i / (n - 1))
         + a2 * np.cos(4 * math.pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _geometric(n: int, r: float) -> np.ndarray:
    """Weights ∝ r**i for i=0..n‑1 (r∈(0,1])."""
    if not (0.0 < r <= 1.0):
        return _uniform(n)
    w = r ** np.arange(n, dtype=np.float64)
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total


def _powerlaw(n: int, p: float) -> np.ndarray:
    """Weights ∝ (i+1)^(-p) with p>0."""
    if p <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total


# ---------------------------------------------------------------
# Random Dirichlet sampling
# ---------------------------------------------------------------

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """
    Draw a Dirichlet(alpha,…,alpha) vector.
    """
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ---------------------------------------------------------------
# Smooth‑max L‑BFGS optimisation
# ---------------------------------------------------------------

def _smooth_obj_and_grad(logits: np.ndarray, n: int, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad) where
        f   = 2·n·smooth_max(conv(a,a))
        grad = ∂f/∂logits     (logits param ⇒ a = softmax(logits))
    The smooth max is the log‑sum‑exp with temperature β.
    """
    # softmax → a
    max_log = np.max(logits)
    a = np.exp(logits - max_log)
    a_sum = a.sum()
    a = a / a_sum

    # self‑convolution
    conv = _conv_fft(a, a)
    c = np.max(conv)                     # max value (for numerical stability)
    exp_beta = np.exp(beta * (conv - c))
    Z = exp_beta.sum()
    p = exp_beta / Z                     # soft‑weights, sum(p)=1

    # smooth max
    smooth_max = c + np.log(Z) / beta

    # gradient w.r.t. a: g_a = 2·conv(p, a_rev)
    conv_pa = _conv_fft(p, a[::-1])      # length 3n‑2
    g_a = 2.0 * conv_pa[n - 1:n - 1 + n]  # slice of length n

    # chain rule through softmax
    dot = np.dot(g_a, a)
    grad_logits = a * (g_a - dot) * (2.0 * n)

    return smooth_max * (2.0 * n), grad_logits


def _lbfgs_optimize(start_seq: np.ndarray,
                    beta: float,
                    deadline: float,
                    maxfun: int = 200) -> tuple[np.ndarray, float]:
    """
    Perform a limited‑memory BFGS optimisation of the smooth objective
    for a fixed temperature β.  The optimisation stops either after ``maxfun``
    function evaluations or when the deadline is reached.
    Returns the best sequence seen (as a probability vector) and its exact
    objective value.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)          # initialise from current probabilities

    # Wrapper returning (value, gradient) for SciPy
    def fun(logits_vec):
        f, g = _smooth_obj_and_grad(logits_vec, n, beta)
        return f, g

    # import lazily – SciPy may be unavailable
    try:
        from scipy.optimize import minimize
    except Exception:
        # fall back to a very short Adam run (implemented later) – here we simply
        # return the start vector.
        return start_seq, _objective(start_seq)

    # Respect the remaining time – each L‑BFGS iteration costs a few ms.
    # Use a very small tolerance; the optimizer will stop after ``maxfun`` calls.
    options = {
        'maxfun': maxfun,
        'maxiter': maxfun,
        'ftol': 1e-12,
        'disp': False,
    }
    # Run the optimiser
    res = minimize(fun, logits,
                   method='L-BFGS-B',
                   jac=True,
                   bounds=None,
                   options=options)

    # Recover the probability vector
    logits_opt = res.x
    max_log = np.max(logits_opt)
    a_opt = np.exp(logits_opt - max_log)
    a_opt = a_opt / a_opt.sum()

    val_opt = _objective(a_opt)
    return a_opt, val_opt


# ---------------------------------------------------------------
# Exact sub‑gradient mirror descent (exponentiated‑gradient)
# ---------------------------------------------------------------

def _refine_subgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Classical exponentiated‑gradient (KL‑mirror descent) on the simplex,
    using the true sub‑gradient of the active convolution entry.
    Returns the best sequence seen and its exact objective.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if (it & 1023) == 0:
                print(f"[Refine] it={it:7d} n={n:5d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = 0.7 / math.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)               # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ---------------------------------------------------------------
# Misc helpers
# ---------------------------------------------------------------

def _resample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation resampling of a 1‑D array.
    """
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0, old_n - 1, new_n, dtype=np.float64)
    new_seq = np.interp(new_x, old_x, seq)
    # clip negatives (numerical noise) and renormalise
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0.0:
        return _uniform(new_n)
    return new_seq / s

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the given time budget and returns the best sequence
    as a plain list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Retrieve previously stored solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – moderate values keep FFT cheap but allow
    # improvement via larger n.
    # ------------------------------------------------------------------
    lengths = [1500, 2000, 2500, 3000, 3500, 4000, 5000, 6000, 8000]

    best_seq = None
    best_val = float("inf")
    best_n   = None

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic seeds (including parameter sweeps)
    # ------------------------------------------------------------------
    seed_families = [
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.25),
    ]

    # geometric families (different decay rates)
    geometric_rs = [0.5, 0.6, 0.7, 0.8, 0.9, 0.95]
    for r in geometric_rs:
        seed_families.append(lambda n, rr=r: _geometric(n, rr))

    # power‑law families
    power_ps = [0.5, 1.0, 1.5, 2.0, 3.0]
    for p in power_ps:
        seed_families.append(lambda n, pp=p: _powerlaw(n, pp))

    # evaluate all seeds
    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                best_n   = n
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

        # also try the globally stored previous vector (resampled to n)
        if previous is not None:
            seq = _resample(previous, n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                best_n   = n
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – massive random Dirichlet sampling on the current best length
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – should never happen
        best_seq = _uniform(2000)
        best_n   = 2000
        best_val = _objective(best_seq)

    alphas = [0.1, 0.5, 1.0, 2.0, 5.0]   # wide spread – very sparse to very smooth
    while time.time() < deadline - 2.0:   # keep a couple seconds for later phases
        n = best_n
        alpha = rng.choice(alphas)
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.2f} val={val:.6f}")

        # early stop if we happen to beat a very low target
        if best_val < 1.48:
            break

    # ------------------------------------------------------------------
    # Phase 3 – smooth‑max L‑BFGS optimisation with annealed β
    # ------------------------------------------------------------------
    # allocate roughly 35 % of the remaining time to this phase
    remaining = deadline - time.time()
    lbfgs_deadline = time.time() + 0.35 * remaining
    beta_schedule = [5.0, 15.0, 40.0, 80.0, 200.0]

    seq_current = best_seq.copy()
    for beta in beta_schedule:
        if time.time() >= lbfgs_deadline:
            break
        # give a couple of hundred function evaluations; if time short, shrink.
        maxfun = 200
        time_per_eval_est = 0.006   # rough estimate for n≈4000
        time_needed = maxfun * time_per_eval_est
        if time.time() + time_needed > lbfgs_deadline:
            maxfun = max(10, int((lbfgs_deadline - time.time()) / time_per_eval_est))
        seq_opt, val_opt = _lbfgs_optimize(seq_current, beta, lbfgs_deadline, maxfun=maxfun)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt.copy()
            best_n   = len(seq_opt)
            print(f"[L‑BFGS β={beta:.1f}] improved best to {best_val:.6f}")

        seq_current = seq_opt   # warm‑start next β

    # ------------------------------------------------------------------
    # Phase 4 – exact sub‑gradient mirror descent (final polishing)
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _refine_subgradient(best_seq, deadline)
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Final‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check and global store
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose the result globally for possible reuse in later runs
    globals()["height_sequence_1"] = best_seq.copy()

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
