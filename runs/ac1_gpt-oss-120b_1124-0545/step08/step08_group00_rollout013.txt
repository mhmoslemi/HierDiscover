"""
Propose a step‑function that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²),

i.e. minimise the L∞‑norm of the self‑convolution of a probability
distribution.  The search consists of three phases:

1. **Broad seed generation** – a large pool of analytically motivated
   shapes (uniform, beta, sine‑window, Tukey, Gaussian, geometric,
   random Dirichlet) is created and normalised to ℓ₁‑norm = 1.

2. **Deterministic refinement** – the best few seeds are refined by
   a projected sub‑gradient descent on the exact objective
   (max‑convolution).  The iterations are limited by a global time
   budget; the algorithm projects onto the probability simplex after
   each additive step.

3. **Stochastic polishing** – a simple simulated‑annealing that reshuffles
   the mass of a handful of coordinates is applied to the best vector
   found, again respecting the remaining time budget.

All steps respect the total 1000 s budget.  The final sequence is stored
in the global name ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Surrogate objective: 2·n·max(conv(seq,seq))  (assumes Σseq = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (Σ=1, v≥0). """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero vector – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) on {1,…,n} (discrete approximation)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

# Optimisation primitives
# ----------------------------------------------------------------------


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on 2·n·max(conv) with Euclidean projection
    onto the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the max at index max_idx (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing additive step
        lr = 0.3 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 600_000,
    init_temp_factor: float = 0.025,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Simple SA polishing that reshuffles a random subset of coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9997, min_temp)

        k = rng.integers(2, 9)          # size of perturbed subset: 2 … 8
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

            print(f"[overall‑SubProj] new global best = {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – stochastic polishing (simulated annealing) on the best
    # vector found so far.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline, max_iters=600_000)
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional – will raise a warning if something
    # went wrong inside evaluate_sequence).
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    # expose for possible reuse by future calls
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
