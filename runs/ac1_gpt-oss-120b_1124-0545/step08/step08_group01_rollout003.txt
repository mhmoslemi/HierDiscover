"""
Non‑negative step‑function optimiser (1000 s budget).

The optimiser works as follows:

1. **Initialisation** – if a previous solution (`height_sequence_1`)
   exists it is taken as the start point; otherwise a cheap pool of
   analytically motivated families (uniform, triangular, symmetric
   geometric, beta, random Dirichlet, sparse …) is generated for a
   moderate length (`n = 4000`).  The best seed (lowest surrogate
   objective 2·n·max(conv)) is kept.

2. **Heavy mirror‑descent** – the current vector lives on the probability
   simplex via an exponentiated‑gradient (mirror‑descent) update.
   At each step the exact sub‑gradient of the true max‑convolution
   (the sum of the contributions of all *active* (tight) convolution
   entries) is used.  A simple back‑tracking line‑search adapts the step
   size; occasional random “local reshuffles’’ provide escape moves.
   The loop runs until the time budget is exhausted, keeping the best
   vector seen.

3. **Optional up‑sampling** – if time permits the best vector is
   linearly interpolated to double the resolution (e.g. 8000 points) and
   refined again with the same mirror‑descent routine.  Extra degrees of
   freedom often lower the constant `C = ∫ f²` and therefore the final
   evaluation value.

4. **Result** – the best sequence is stored globally as
   ``height_sequence_1`` and returned as a plain list of floats ready for
   ``evaluate_sequence``.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    Assumes ``seq`` is non‑negative and already normalised to sum = 1.
    Returns ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * float(np.max(conv))


# --------------------------------------------------------------
# Analytic shape families (all normalised to sum = 1)
# --------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    if a <= 0.0 or b <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = np.power(i, a - 1.0)
    right = np.power(n - i + 1.0, b - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


# --------------------------------------------------------------
# Seed generation (cheap, diverse)
# --------------------------------------------------------------

def _generate_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # symmetric geometric – medium grid
    for r in np.linspace(0.10, 0.90, 17):
        seeds.append(_sym_geom(n, r))

    # symmetric power‑law (i+1)^‑p + (n‑i)^‑p
    for p in np.arange(0.0, 5.01, 0.25):
        i = np.arange(1, n + 1, dtype=np.float64)
        a = np.power(i, -p) + np.power(n - i + 1, -p)
        s = a.sum()
        if s > 0:
            seeds.append(a / s)

    # beta family – grid
    for a in np.linspace(0.05, 2.0, 8):
        for b in np.linspace(0.05, 5.0, 8):
            seeds.append(_beta_shape(n, a, b))

    # mixtures of uniform and symmetric geometric
    uniform = _uniform(n)
    for r in np.linspace(0.20, 0.90, 9):
        base = _sym_geom(n, r)
        for w in np.linspace(0.0, 1.0, 5):
            mixed = w * uniform + (1.0 - w) * base
            mixed = mixed / mixed.sum()
            seeds.append(mixed)

    # random Dirichlet draws
    for conc in (0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (3, 5, 10, 20):
        for conc in (0.20, 0.50):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if compatible length)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds


# --------------------------------------------------------------
# Heavy mirror‑descent refinement (exact max‑convolution sub‑gradient)
# --------------------------------------------------------------

def _refine_max_convolution(start_seq: np.ndarray,
                            deadline: float,
                            max_no_imp: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the true max‑convolution
    objective.  The step size is `lr / sqrt(iter)`.  A tiny back‑tracking
    line‑search and occasional random local reshuffles (simulated‑annealing
    style) are used to escape shallow plateaus.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0
    no_imp = 0                     # how many iterations without improvement

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        # active set – all entries within a tiny relative tolerance
        tol = max_val * 1e-12
        active = (conv >= max_val - tol).astype(np.float64)

        # sub‑gradient of the max: g = 2 * (rev_a * active)[center slice]
        rev_a = a[::-1]
        full = _conv_fft(rev_a, active)          # length 3n‑2
        g = 2.0 * full[n - 1:n - 1 + n]          # central n entries

        # step size schedule
        lr = 0.7 / np.sqrt(it)

        # line‑search: try the current lr, halve it a few times if no improvement
        improved = False
        for _ in range(5):
            log_a = np.log(a + eps) - lr * g
            # keep numbers stable
            log_a -= np.max(log_a)
            cand = np.exp(log_a)
            cand /= cand.sum()
            cand_val = _objective(cand)

            if cand_val < best_val:
                best_val = cand_val
                best_seq = cand.copy()
                a = cand
                improved = True
                no_imp = 0
                if it % 50000 == 0:
                    print(f"[Refine] it={it:,} best={best_val:.6f}")
                break
            else:
                lr *= 0.5     # shrink step and retry

        if not improved:
            no_imp += 1

        # Occasional random “local” reshuffle to escape plateaus
        if it % 10000 == 0 or (no_imp > 50000 and it % 5000 == 0):
            rng = np.random.default_rng()
            k = rng.integers(2, 7)            # move mass among 2‑6 coordinates
            idx = rng.choice(n, size=k, replace=False)
            total = a[idx].sum()
            if total > 0.0:
                new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
                a2 = a.copy()
                a2[idx] = new_alloc
                a2 /= a2.sum()
                val2 = _objective(a2)
                if val2 < best_val:
                    best_val = val2
                    best_seq = a2.copy()
                    a = a2
                    no_imp = 0
                    print(f"[Reshuffle] it={it:,} best={best_val:.6f}")

        # safety break if we have spent too many iterations without any improvement
        if no_imp > max_no_imp:
            break

    return best_seq, best_val


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence as a list.
    The function respects the overall time budget (default 1000 s).
    """
    # ----------------------------------------------------------
    # Setup
    # ----------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # a small safety margin before the hard timeout
    deadline = start_time + budget_s - 5.0

    # ----------------------------------------------------------
    # Load previous solution if available
    # ----------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ----------------------------------------------------------
    # Choose length. 4000 is a good trade‑off; we may double later.
    # ----------------------------------------------------------
    n = 4000

    # ----------------------------------------------------------
    # Stage 1 – obtain a good starting vector
    # ----------------------------------------------------------
    if previous is not None and previous.shape == (n,):
        start_vec = previous.copy()
        start_vec = start_vec / start_vec.sum()
        print("[Init] Using previous solution as start point.")
    else:
        print(f"\n--- Seed generation (n={n}) ---")
        seeds = _generate_seeds(n, rng, previous)
        print(f"Generated {len(seeds)} raw seeds.")
        # evaluate cheaply (surrogate only)
        scored = [(vec, _objective(vec)) for vec in seeds]
        scored.sort(key=lambda x: x[1])
        best_seed = scored[0][0]
        start_vec = best_seed.copy()
        print(f"[Init] Best seed surrogate value = {scored[0][1]:.6f}")

    # ----------------------------------------------------------
    # Stage 2 – heavy mirror‑descent refinement
    # ----------------------------------------------------------
    print("\n--- Heavy refinement (mirror descent) ---")
    refined_seq, refined_val = _refine_max_convolution(start_vec, deadline)
    print(f"[Refine] Best surrogate after refinement = {refined_val:.6f}")

    # ----------------------------------------------------------
    # Stage 3 – optional up‑sampling & further refinement
    # ----------------------------------------------------------
    if time.time() < deadline:
        n_big = n * 2
        print(f"\n--- Upsampling to n={n_big} and further refinement ---")
        x_old = np.linspace(0.0, 1.0, n, dtype=np.float64)
        x_big = np.linspace(0.0, 1.0, n_big, dtype=np.float64)
        upsampled = np.interp(x_big, x_old, refined_seq)
        upsampled = upsampled / upsampled.sum()
        refined_seq2, refined_val2 = _refine_max_convolution(upsampled, deadline)
        if refined_val2 < refined_val:
            refined_seq = refined_seq2
            refined_val = refined_val2
            n = n_big
            print(f"[Upsample‑Refine] Improved surrogate = {refined_val:.6f}")
        else:
            print("[Upsample‑Refine] No improvement over previous best.")

    # ----------------------------------------------------------
    # Final sanity check (exact evaluation)
    # ----------------------------------------------------------
    try:
        exact_val = evaluate_sequence(refined_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        exact_val = None

    print("\n=== Finished ===")
    print(f"Surrogate objective ≈ {refined_val:.6f}")
    if exact_val is not None:
        print(f"Exact evaluate_sequence value = {exact_val:.6f}")

    # expose for possible reuse
    globals()["height_sequence_1"] = refined_seq.tolist()
    return refined_seq.tolist()
