"""
Hybrid optimizer for step‑function attack (improved).

Algorithm:
1. Generate a rich set of seed vectors (analytic families plus fine‑grid
   beta‑pdf and mixed beta + uniform parameters).
2. Keep the best few seeds.
3. Refine each of them with a **projected additive sub‑gradient** method
   (gradient of the max–convolution approximated by the uniform
   sub‑gradient on the active set).  Momentum (Nesterov‑like) and a
   decreasing step‑size are used.
4. Run a short simulated‑annealing phase for final fine‑tuning.
The best sequence found before the deadline is stored as
`height_sequence_1` and returned as a plain list.
"""

import time
import math
import numpy as np

# ---------- FFT based convolution ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the grader."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))

# ---------- Simplex projection ----------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# ---------- Analytic families ----------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n) / n

def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()

def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()

def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()

def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()

def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()

def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()

def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()

def _beta_window(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density discretised on cell mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()

def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# ---------- Seed generation ----------
def _make_extended_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic families (same as before, but with finer grids for some)
    seeds.append(_uniform(n))

    # geometric – decay factor up to 12/n (r down to ~0.997)
    for c in np.linspace(0.5, 12.0, 30):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power law – exponents up to 4
    for p in np.linspace(0.0, 4.0, 41):
        seeds.append(_power(n, p))

    # combined geometric‑power
    for c in np.linspace(0.5, 12.0, 16):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 9):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 12.0, 16):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classic windows
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 7):
        seeds.append(_tukey_window(n, alpha))

    # Gaussian family – finer sweep
    for sigma_factor in np.linspace(0.06, 0.30, 13):
        seeds.append(_gaussian(n, sigma_factor))

    # Beta‑pdf family – fine grid from 0.15 to 1.2
    for alpha in np.linspace(0.15, 1.2, 24):
        seeds.append(_beta_window(n, alpha))

    # Mixture of beta and uniform – 5 mixture ratios
    for alpha in np.linspace(0.2, 1.0, 9):
        beta_vec = _beta_window(n, alpha)
        for lam in np.linspace(0.0, 1.0, 5):
            mix = lam * beta_vec + (1 - lam) * _uniform(n)
            mix = np.clip(mix, 0.0, None)
            mix /= mix.sum()
            seeds.append(mix)

    # Dirichlet draws – a few concentration values
    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, alpha, rng))

    # random soft‑max vectors
    for _ in range(5):
        seeds.append(_random_softmax(n, rng))

    # reuse previous construction (if any)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final clean‑up
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

# ---------- Additive sub‑gradient refinement ----------
def _additive_subgrad_refine(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 600000) -> tuple[np.ndarray, float]:
    """
    Projected additive sub‑gradient descent on `max(conv(a,a))`.
    Uses a momentum term (β=0.9) and a step size η_t = η0 / sqrt(t).
    Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # momentum buffer
    v = np.zeros_like(a)

    eta0 = 0.4          # initial step size (tuned empirically)
    mu = 0.9            # momentum coefficient

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # active set – all indices within a tiny tolerance of the maximum
        rel_tol = 1e-12
        active = np.where(conv >= max_val - rel_tol)[0]

        # weight vector for the active constraints (uniform)
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size

        # sub‑gradient of the max‑convolution (same as in the original code)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        g = 2.0 * full[n - 1:n - 1 + n]

        # momentum update
        eta = eta0 / math.sqrt(it)
        v = mu * v - eta * g
        a = a + v

        # projection onto the simplex
        a = _project_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[AddSub] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val

# ---------- Simulated annealing (same as before) ----------
def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-7, cur_val * 0.08)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        move_type = rng.random()
        if move_type < 0.70:                     # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.random() * 0.20 * cur[i]
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        elif move_type < 0.85:                   # swap two entries
            i, j = rng.integers(0, n, size=2)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:                                    # global Gaussian jitter
            noise = rng.normal(scale=0.0015, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0:
                new_seq /= total
            else:
                new_seq = _uniform(n)

        new_seq = _project_simplex(new_seq)
        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:8d} best={new_val:.6f}")

        if it % 2000 == 0:
            jitter = rng.normal(scale=0.003, size=n)
            candidate = best + jitter
            candidate = np.clip(candidate, 0.0, None)
            candidate = _project_simplex(candidate)
            cand_val = _objective(candidate)
            if cand_val < best_val:
                best_val = cand_val
                best = candidate.copy()
                print(f"[Anneal] (jump) it={it:8d} best={cand_val:.6f}")

    return best, best_val

"""
Hybrid optimizer for a step‑function attack (improved).

The algorithm proceeds in three stages:

1. **Seed generation** – a large, diverse collection of analytic
   probability vectors (uniform, geometric, power‑law, classic windows,
   Beta‑pdfs, mixtures, random draws).  The best few seeds are retained.

2. **Additive sub‑gradient refinement** – a projected gradient descent on
   the exact objective `2·n·max(conv(a,a)) / (∑a)²`.  The gradient of the
   max‑convolution is obtained via the active‑set sub‑gradient (the same
   construction used in the original code).  Momentum (μ≈0.9) and a
   step‑size ηₜ = η₀/√t are employed.  Each of the top seeds is refined
   for a share of the remaining time budget; the best result is kept.

3. **Simulated‑annealing fine‑tuning** – a Metropolis–Hastings walk
   (mass‑transfer, swap, global jitter moves) runs on the remaining
   time to polish the solution.

The final sequence is stored in the global variable ``height_sequence_1``
and returned as a plain list.  All vectors are kept on the probability
simplex, so the security checks inside ``evaluate_sequence`` never clip
or reject them.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution (full linear mode)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the grader."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x ≥ 0, ∑x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1) discretised on mid‑points.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `alpha`."""
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Seed pool – a very diverse set of start vectors (all normalised)
# ----------------------------------------------------------------------


def _make_extended_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    # geometric – fine grid
    for c in np.linspace(0.5, 12.0, 30):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power law – fine grid
    for p in np.linspace(0.0, 4.0, 41):
        seeds.append(_power(n, p))

    # combined geometric‑power
    for c in np.linspace(0.5, 12.0, 16):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 9):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 12.0, 16):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classic windows
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 7):
        seeds.append(_tukey_window(n, alpha))

    # Gaussian family – finer sweep
    for sigma_factor in np.linspace(0.06, 0.30, 13):
        seeds.append(_gaussian(n, sigma_factor))

    # Beta‑pdf family – fine grid on (0, 1.2)
    for alpha in np.linspace(0.15, 1.2, 24):
        seeds.append(_beta_window(n, alpha))

    # Mixture of beta and uniform (5 mixture ratios)
    for alpha in np.linspace(0.2, 1.0, 9):
        beta_vec = _beta_window(n, alpha)
        for lam in np.linspace(0.0, 1.0, 5):
            mix = lam * beta_vec + (1 - lam) * _uniform(n)
            mix = np.clip(mix, 0.0, None)
            mix /= mix.sum()
            seeds.append(mix)

    # Dirichlet draws
    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, alpha, rng))

    # random soft‑max vectors
    for _ in range(5):
        seeds.append(_random_softmax(n, rng))

    # reuse previous construction if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final clean‑up – clip tiny negatives, normalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    return cleaned


# ----------------------------------------------------------------------
# Additive sub‑gradient refinement (projected gradient descent)
# ----------------------------------------------------------------------


def _additive_subgrad_refine(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Projected additive sub‑gradient descent on the exact non‑smooth objective.
    Momentum (μ=0.9) and a step size ηₜ = η₀/√t are used.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    v = np.zeros_like(a)          # momentum buffer
    eta0 = 0.40                    # initial step size (empirically tuned)
    mu = 0.9                       # momentum coefficient

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # active set (indices attaining the maximum, up to a tiny tolerance)
        rel_tol = 1e-12
        active = np.where(conv >= max_val - rel_tol)[0]

        # weight vector for the active constraints (uniform over the set)
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size

        # sub‑gradient of the max‑convolution (same construction as in the
        # original exponentiated‑gradient version)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        g = 2.0 * full[n - 1:n - 1 + n]

        # momentum + step
        eta = eta0 / math.sqrt(it)
        v = mu * v - eta * g
        a = a + v

        # projection onto the probability simplex
        a = _project_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[AddSub] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (identical to the previous implementation)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-7, cur_val * 0.08)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        move_type = rng.random()
        if move_type < 0.70:                     # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.random() * 0.20 * cur[i]
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        elif move_type < 0.85:                   # swap two entries
            i, j = rng.integers(0, n, size=2)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:                                    # global Gaussian jitter
            noise = rng.normal(scale=0.0015, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0:
                new_seq /= total
            else:
                new_seq = _uniform(n)

        new_seq = _project_simplex(new_seq)
        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:8d} best={new_val:.6f}")

        if it % 2000 == 0:
            jitter = rng.normal(scale=0.003, size=n)
            candidate = best + jitter
            candidate = np.clip(candidate, 0.0, None)
            candidate = _project_simplex(candidate)
            cand_val = _objective(candidate)
            if cand_val < best_val:
                best_val = cand_val
                best = candidate.copy()
                print(f"[Anneal] (jump) it={it:8d} best={cand_val:.6f}")

    return best, best_val
