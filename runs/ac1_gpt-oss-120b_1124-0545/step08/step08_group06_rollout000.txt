   for several lengths and – if present – the previous best sequence.
2. **Random Dirichlet sampling** – quick Monte‑Carlo search over many
   lengths and concentration parameters.
3. **Annealed‑soft‑max Adam** – a smooth approximation of the
   ``max(conv)`` term.  Starting from each promising seed we run Adam
   on the smoothed objective while gradually increasing the temperature
   (λ) so that the optimisation first sees a very smooth landscape and
   later focuses on the true maximum.  Gradients are obtained analytically
   via FFT‑based convolutions and the iterate is projected onto the
   probability simplex after every step.
4. **Post‑processing** – linear mixing with the uniform distribution,
   a tiny Gaussian blur, and a final sanity check.

All heavy work is done with NumPy and FFT; the routine respects the
supplied time budget (default 1000 s) and returns a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for the sizes we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Objective that ``evaluate_sequence`` would return when ``seq`` is already
    normalised to sum = 1:  2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients proportional to binom(2i,i)/4**i, normalised."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _geometric_shape(n: int, r: float = 0.999) -> np.ndarray:
    """Truncated geometric distribution a_i ∝ r**i, normalised."""
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


# ----------------------------------------------------------------------
# Random Dirichlet sampling
# ----------------------------------------------------------------------


def _dirichlet(alpha: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector of length n, normalised to 1."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) utilities
# ----------------------------------------------------------------------


def _smooth_max_weights(conv: np.ndarray, lam: float) -> tuple[float, np.ndarray]:
    """
    Compute a smooth approximation of max(conv) using
    log‑sum‑exp with temperature λ (lam) and return both the
    smoothed value and the softmax weights (sum to 1).
    """
    # shift by the maximum for numerical stability
    max_c = conv.max()
    shifted = lam * (conv - max_c)
    # prevent overflow/underflow
    exp_shifted = np.exp(shifted - np.max(shifted))
    sum_exp = exp_shifted.sum()
    smooth = max_c + np.log(sum_exp) / lam
    weights = exp_shifted / sum_exp
    return smooth, weights


# ----------------------------------------------------------------------
# Adam optimiser with annealed smoothing and simplex projection
# ----------------------------------------------------------------------


def _refine_smooth_adam(
    start_seq: np.ndarray,
    deadline: float,
    lam_start: float = 10.0,
    lam_end: float = 200.0,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed objective 2·n·smooth_max(conv).
    ``lam`` (temperature) grows from ``lam_start`` to ``lam_end`` during
    the run (geometric schedule).  After each update the iterate is
    projected onto the simplex.
    Returns the best sequence (and its true objective) seen before the
    deadline.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    # Adam state
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # pre‑compute a reversed view for gradient calculation
    a_rev = a[::-1]

    # geometric schedule for lambda
    log_lambda_ratio = np.log(lam_end / lam_start)

    while time.time() < deadline and t < max_iters:
        t += 1

        # ----------- compute smoothed max and its gradient -------------
        conv = _conv_fft(a, a)                     # full convolution, length 2n‑1
        # temperature for the current iteration
        lam = lam_start * np.exp(log_lambda_ratio * (t / max_iters))
        smooth, w = _smooth_max_weights(conv, lam)

        # gradient of smooth_max w.r.t. a (derived analytically)
        # g_j = 2 * Σ_i a_i * w_{i+j}
        # this is a cross‑correlation; compute via convolution with reversed a
        conv_wa = _conv_fft(a_rev, w)              # length = 3n‑2
        grad = 2.0 * conv_wa[n - 1 : n - 1 + n]    # keep n entries

        # --------------------- Adam update ----------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # simple diminishing base learning rate
        lr = 0.05 / np.sqrt(t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto simplex
        a = _proj_simplex(a_new)

        # occasional tiny perturbation to escape plateaus
        if t % 5000 == 0:
            rng = np.random.default_rng(t)
            a = _proj_simplex(a + rng.normal(scale=1e-5, size=n))

        # Update best if the true (non‑smoothed) objective improves
        if t % 200 == 0 or t == 1:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SmoothAdam] it={t:6d} λ={lam:.1f} val={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Post‑processing helpers
# ----------------------------------------------------------------------


def _mix_with_uniform(seq: np.ndarray, steps: int = 101) -> tuple[np.ndarray, float]:
    """Linear interpolation between ``seq`` and the uniform vector."""
    n = len(seq)
    uniform = _uniform(n)
    best_seq = seq.copy()
    best_val = _objective(seq)
    for k in range(steps):
        t = k / (steps - 1)
        combo = (1.0 - t) * seq + t * uniform
        combo = _proj_simplex(combo)  # just to be safe
        val = _objective(combo)
        if val < best_val:
            best_val = val
            best_seq = combo.copy()
    return best_seq, best_val


def _apply_blur(seq: np.ndarray, radius: int = 1) -> tuple[np.ndarray, float]:
    """Replace each entry by a small moving‑average (Gaussian‑like) blur."""
    if radius <= 0:
        return seq, _objective(seq)
    kernel = np.ones(2 * radius + 1, dtype=np.float64)
    kernel /= kernel.sum()
    blurred = np.convolve(seq, kernel, mode="same")
    blurred = _proj_simplex(blurred)
    return blurred, _objective(blurred)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given
    time budget and returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives a finer bound but costs more.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 16000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic seeds
    # ------------------------------------------------------------------
    for n in lengths:
        for maker in (
            _catalan_shape,
            _uniform,
            lambda n: _geometric_shape(n, r=0.999),
            lambda n: _geometric_shape(n, r=0.995),
        ):
            try:
                seq = maker(n)
            except Exception:
                continue
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={maker.__name__} val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1b – incorporate any previous solution if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                prev_arr = prev_arr / prev_arr.sum()
                val = _objective(prev_arr)
                if val < best_val:
                    best_val = val
                    best_seq = prev_arr.copy()
                    print(f"[Seed] previous solution val={val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stage 2 – massive random Dirichlet sampling (different α values)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    sample_deadline = now + remaining * 0.35   # allocate ~35 % to sampling

    alphas = [0.5, 1.0, 2.0, 4.0]               # explore sparse & smooth regimes
    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = float(rng.choice(alphas))
        seq = _dirichlet(alpha, n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.1f} val={val:.6f}")

        # early exit if we already entered the target region
        if best_val < 1.505:
            break

    # ------------------------------------------------------------------
    # Stage 3 – annealed‑soft‑max Adam refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # roughly half of the remaining budget for refinement
        refine_deadline = time.time() + (deadline - time.time()) * 0.55
        seq_ref, val_ref = _refine_smooth_adam(best_seq, refine_deadline)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = seq_ref
            print(f"[Refine‑Adam] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – post‑processing tricks
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # a) linear mixing with uniform
        seq_mix, val_mix = _mix_with_uniform(best_seq)
        if val_mix < best_val:
            best_val = val_mix
            best_seq = seq_mix
            print(f"[Mix‑Uniform] improved best to {best_val:.6f}")

    if time.time() < deadline:
        # b) tiny blur (radius = 1)
        seq_blur, val_blur = _apply_blur(best_seq, radius=1)
        if val_blur < best_val:
            best_val = val_blur
            best_seq = seq_blur
            print(f"[Blur] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – the grader will also call evaluate_sequence
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
