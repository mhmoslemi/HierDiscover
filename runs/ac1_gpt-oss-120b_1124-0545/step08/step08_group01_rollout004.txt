    # start_vec normalized to sum 1.
    v = np.log(start_vec + eps)  # initial logits
    best_seq = None; best_val = np.inf

    for beta in betas:
        # define objective function on v
        def fun(v):
            a = _softmax(v)  # returns array sum to 1
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            # smoothed log-sum-exp
            w = np.exp(beta * (conv - shift))
            val = (shift + np.log(w.sum()) / beta)  # log-sum-exp divided by beta gives smoothed max
            # We want the actual objective 2 * n * max(conv). But for optimization we approximate using smoothed max.
            # Better to return smoothed max directly; for L-BFGS, any monotonic function works.
            # We'll return smoothed max
            return shift + np.log(w.sum()) / beta

        def grad(v):
            a = _softmax(v)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0.0:
                grad_a = np.zeros_like(a)
            else:
                w_norm = w / w_sum
                rev_a = a[::-1]
                full = _conv_fft(rev_a, w_norm)
                grad_a = 2.0 * full[n - 1:n - 1 + n]  # central entries
            # gradient w.r.t. v
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return grad_v

        # set up minimize
        options = {"maxiter": max_iters_per_beta, "gtol": 1e-12}
        res = opt.minimize(fun, v, method='L-BFGS-B', jac=grad, options=options)

        if cur_val < best_val:

"""
Non‑negative step‑function optimiser (enhanced).

The search proceeds in three stages:

1. **Broad seed generation** – a very diverse set of analytic families
   (uniform, triangular, sine, symmetric‑geometric, power‑law,
   exponential, arcsine‑type, Gaussian) together with random Dirichlet
   and sparse draws is built for a modest length (default 2000).
   All candidates are normalised to sum = 1 and ranked by the fast
   surrogate objective ``2·n·max(conv(seq,seq))``.

2. **Heavy refinement** – the top few seeds are refined by
   *Adam* on a smoothed‑max surrogate (log‑sum‑exp) while staying on
   the probability simplex, followed by a long *exact sub‑gradient*
   mirror‑descent (exponentiated‑gradient) on the true max‑convolution
   objective.  If SciPy is available a short L‑BFGS‑B pass on the same
   smoothed objective is performed as well.

3. **Polishing & up‑sampling** – a simulated‑annealing hill‑climber
   further improves the best vector.  Finally the solution is
   linearly up‑sampled to double the resolution (≈4000 points) and
   refined again with sub‑gradient descent.

The routine respects the supplied time budget (default 1000 s),
stores the best vector globally under the name ``height_sequence_1``,
and returns it as a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# optional SciPy for a short L‑BFGS‑B pass
try:
    import scipy.optimize as opt
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False

# Low‑level utilities
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (fast, O(n log n)).
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps ℝⁿ → simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    Assumes ``seq`` is non‑negative and already normalised to sum = 1.
    Returns ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# --------------------------------------------------------------
# Analytic shape families (all normalised to sum = 1)
# --------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_power_law(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law: (i+1)^‑p + (n‑i)^‑p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p) + np.power(n - i + 1, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-lam * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_arc_power(n: int, p: float) -> np.ndarray:
    """U‑shaped (arcsine‑like) family: ((i+1)*(n-i))⁻ᵖ."""
    i = np.arange(n, dtype=np.float64)
    product = (i + 1.0) * (n - i)
    a = np.power(product, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shape(n: int, sigma_frac: float) -> np.ndarray:
    """Symmetric discrete Gaussian (σ = sigma_frac·n)."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    sigma = sigma_frac * n
    if sigma <= 0.0:
        return _uniform(n)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Full‑support Dirichlet draw."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()

def _generate_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a diverse list of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # ----- deterministic families -----
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # symmetric geometric – medium grid
    for r in np.linspace(0.05, 0.95, 19):
        seeds.append(_sym_geom(n, r))

    # symmetric power‑law – coarse and fine grids
    for p in np.arange(0.0, 5.01, 0.25):
        seeds.append(_sym_power_law(n, p))
    for p in np.arange(0.0, 5.001, 0.10):
        seeds.append(_sym_power_law(n, p))

    # exponential decay – medium grid
    for lam in np.linspace(0.001, 0.020, 12):
        seeds.append(_sym_exponential(n, lam))

    # arcsine‑like power – medium grid
    for p in np.arange(0.2, 1.51, 0.10):
        seeds.append(_sym_arc_power(n, p))

    # Gaussian – sigma as a fraction of n
    for sf in np.linspace(0.02, 0.20, 10):
        seeds.append(_gaussian_shape(n, sf))

    # mixtures of uniform & geometric (to explore flat + spiky)
    uniform = _uniform(n)
    for r in np.linspace(0.10, 0.90, 9):
        base = _sym_geom(n, r)
        for w in np.linspace(0.0, 1.0, 5):
            mixed = w * uniform + (1.0 - w) * base
            mixed = mixed / mixed.sum()
            seeds.append(mixed)

    # random Dirichlet draws (different concentrations)
    for conc in (0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns (few spikes)
    for k in (3, 5, 10, 20):
        for conc in (0.20, 0.50):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if compatible length)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 50000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a log‑sum‑exp (smoothed max) surrogate.
    Uses a log‑softmax parametrisation to stay on the probability simplex.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)           # logits (unconstrained)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5               # initial temperature (smoothing factor)
    beta_mul = 1.001
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                     # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # increase smoothing temperature (capped)
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp on the convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror‑descent (exponentiated‑gradient) on the
    true max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the (conv) at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _lbfgs_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    betas: list[float] | None = None,
    max_iters_per_beta: int = 30,
) -> tuple[np.ndarray | None, float]:
    """
    L‑BFGS‑B optimisation on the log‑sum‑exp surrogate.
    The variable is the log‑softmax logits; after each beta we keep the
    best concrete vector (according to the true surrogate objective).
    """
    if not _HAS_SCIPY:
        return None, float("inf")

    if betas is None:
        betas = [5, 15, 50, 150, 300]

    eps = 1e-30
    v = np.log(start_vec + eps)               # initialise logits

    best_seq = None
    best_val = float("inf")

    for beta in betas:
        if time.time() > deadline:
            break

        def fun(logits):
            a = _softmax(logits)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            w = np.exp(beta * (conv - shift))
            # smoothed max = shift + (1/β)·log Σ w
            return shift + (np.log(w.sum()) / beta)

        def grad(logits):
            a = _softmax(logits)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0.0:
                w_norm = np.zeros_like(w)
            else:
                w_norm = w / w_sum
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w_norm)
            grad_a = 2.0 * full[n - 1:n - 1 + n]
            dot = np.dot(grad_a, a)
            return a * (grad_a - dot)

        options = {"maxiter": max_iters_per_beta, "gtol": 1e-12}
        res = opt.minimize(fun, v, method="L-BFGS-B", jac=grad, options=options)
        v = res.x
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L‑BFGS] beta={beta:.1f} best={cur_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles the mass of a random small subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 8)           # size of the random subset (2…7)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal
    ``evaluate_sequence`` value.
    Returns the best sequence found as a list of floats.
    """
    # --- Reproducibility -------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # ----------------------------------------------------------------------
    # Possibly reuse a previous solution
    # ----------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ----------------------------------------------------------------------
    # Choose an initial length (fast enough for many iterations)
    # ----------------------------------------------------------------------
    n = 2000

    # ----------------------------------------------------------------------
    # Stage 1 – generate a large pool of seeds and keep the top few
    # ----------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _generate_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} candidate seeds.")

    # Evaluate all seeds with the fast surrogate objective
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(15, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed objectives:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best seed objective = {global_best_val:.6f}")

    # ----------------------------------------------------------------------
    # Stage 2 – heavy optimisation of the top seeds
    # ----------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Optimisation run {idx+1}/{len(best_seeds)} ---")

        # Adam phase (smooth max)
        seq_adam, val_adam = _adam_optimize(
            seed_vec, n, deadline, max_iters=40000
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # Optional L‑BFGS pass (if SciPy is available)
        if _HAS_SCIPY:
            seq_lbfgs, val_lbfgs = _lbfgs_optimize(
                global_best_seq, n, deadline, max_iters_per_beta=20
            )
            if seq_lbfgs is not None and val_lbfgs < global_best_val:
                global_best_seq = seq_lbfgs.copy()
                global_best_val = val_lbfgs
                print(f"[overall‑L‑BFGS] new global best = {val_lbfgs:.6f}")

        # Exact sub‑gradient refinement (more iterations on the current best)
        seq_sub, val_sub = _subgradient_refine(
            global_best_seq, n, deadline, max_iters=250000
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[overall‑SubGrad] new global best = {val_sub:.6f}")

    # ----------------------------------------------------------------------
    # Stage 3 – simulated‑annealing polishing
    # ----------------------------------------------------------------------
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=200000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] new global best = {val_sa:.6f}")

    # ----------------------------------------------------------------------
    # Optional up‑sampling to a longer length (if time permits)
    # ----------------------------------------------------------------------
    if time.time() < deadline:
        n_big = n * 2                     # double the resolution
        x_old = np.linspace(0.0, 1.0, n, dtype=np.float64)
        x_big = np.linspace(0.0, 1.0, n_big, dtype=np.float64)
        upsampled = np.interp(x_big, x_old, global_best_seq)
        upsampled = upsampled / upsampled.sum()
        print(f"\n--- Upsampling to n={n_big} and refining ---")
        # Sub‑gradient refinement on the larger problem
        seq_big, val_big = _subgradient_refine(
            upsampled, n_big, deadline, max_iters=150000
        )
        if val_big < global_best_val:
            global_best_seq = seq_big.copy()
            global_best_val = val_big
            n = n_big
            print(f"[Upsample‑Refine] new global best = {val_big:.6f}")

    # ----------------------------------------------------------------------
    # Final sanity check (optional)
    # ----------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose the result for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
