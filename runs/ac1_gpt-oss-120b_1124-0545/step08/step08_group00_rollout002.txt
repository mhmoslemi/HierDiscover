"""
Propose a non‑negative step function (list of floats) that minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm follows three stages:

1️⃣  **Rich seed generation** – deterministic families (uniform, triangular,
    trapezoidal, sine, Tukey, Gaussian, geometric, power‑law) plus a *dense*
    grid of beta‑shaped vectors (α,β∈[0.05,5]) and a variety of power‑law
    exponents.  All seeds are ℓ₁‑normalised (∑=1) and evaluated with the cheap
    surrogate `2·n·max(conv)`.  The best few seeds are kept.

2️⃣  **Heavy continuous optimisation** – starting from the best seed we run
    a long Adam optimisation on a log‑sum‑exp smoothed version of the
    “max‑convolution”.  Temperature (β) is increased until a very sharp
    approximation (β≈800) is reached.  After Adam we perform an exact
    projected sub‑gradient descent that works directly on the true maximum.
    Both phases respect the global time budget.

3️⃣  **Final stochastic polishing** – a short simulated‑annealing phase
    reshuffles a random small subset of coordinates while keeping the
    ℓ₁‑norm constant.  This helps escape shallow minima.

All calculations use the real FFT for fast O(n log n) convolutions.
The routine respects the 1000 s budget, stores the best vector in the
global name ``height_sequence_1`` and returns it as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------- low‑level utilities -----------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Surrogate objective: 2·n·max(conv(seq,seq))  (assumes Σseq = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------- analytic shapes -------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta density on the discrete grid {1,…,n} (α,β>0)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _laplace_shape(n: int, lam: float) -> np.ndarray:
    """Symmetric discrete Laplace (double‑exponential) shape."""
    if lam <= 0.0:
        return _uniform(n)
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# -------------------------- simplex projection -------------------------------

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (Σ=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------- optimisation primitives -------------------------

def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 2_000_000,
                   beta_max: float = 800.0,
                   lr0: float = 1.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    Returns the best vector seen and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)               # logits (softmax yields probabilities)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.1
    beta_mul = 1.003                         # a bit slower temperature increase

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                       # current probabilities (sum=1)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # increase smoothing temperature
        beta = min(beta * beta_mul, beta_max)

        # smooth max via log‑sum‑exp (weights w_k)
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k wrt a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)            # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping (helps stability)
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-12
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 1_500_000,
                      top_k: int = 3,
                      lr0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Exact projected sub‑gradient descent on the true max‑convolution.
    Gradient is summed over the `top_k` largest convolution entries.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # indices of the `top_k` largest convolution entries
        if top_k >= len(conv):
            idxs = np.arange(len(conv))
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]

        # accumulate gradient contributions
        g = np.zeros_like(a)
        for idx in idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = idx - i_range
            # each pair contributes twice (i!=j) and once when i==j; the formula
            # 2 * a[rev_idx] works for all cases because when i == rev_idx
            # we still count the diagonal term correctly.
            g[i_range] += 2.0 * a[rev_idx]

        lr = lr0 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.03,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Simple SA that reallocates mass over a random small subset of coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 12)          # choose 2 … 11 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# --------------------------- seed collection --------------------------------

def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse set of ℓ₁‑normalised candidate vectors.
    In addition to the families from the reference code we also add:
      • a dense grid of Beta(α,β) shapes (α,β∈[0.05,5])
      • a finer power‑law grid (p∈[0,8])
      • Laplace shapes with various decay rates
    """
    seeds: list[np.ndarray] = []

    # deterministic families (same as reference)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.15))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))

    for r in np.linspace(0.2, 0.98, 10):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 4.0, 15):
        seeds.append(_power(n, p))

    # Beta grid (dense)
    alphas = np.linspace(0.05, 5.0, 30)
    betas = np.linspace(0.05, 5.0, 30)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # finer power‑law
    for p in np.linspace(0.0, 8.0, 41):
        seeds.append(_power(n, p))

    # Laplace family
    for lam in np.logspace(-3.5, -0.5, 12):
        seeds.append(_laplace_shape(n, lam))

    # Gaussian windows
    for sigma_frac in (0.15, 0.25, 0.35, 0.45):
        seeds.append(_gaussian(n, sigma_frac))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws (heavy‑tailed)
    for _ in range(3):
        seeds.append(_random_exponential(n, rng))

    # previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ---------------------------- main entry point ------------------------------

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function respects the supplied time budget
    and returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # tiny safety margin

    # ------------------------------------------------------------------
    # length (4000 works well, and the objective is scale‑invariant)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – diverse seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Generating seeds for length {n} ---")
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    # evaluate surrogate for all seeds
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # optionally run a tiny differential‑evolution on the beta family
    # (this is cheap compared to the heavy stages)
    def beta_surrogate(params):
        a, b = params
        vec = _beta_shape(n, a, b)
        return _objective(vec)

    # bound the search to the same region we already sampled
    bounds = [(0.05, 5.0), (0.05, 5.0)]
    print("\n[DE] Starting differential evolution on (α,β) beta family...")
    de_result = differential_evolution(beta_surrogate,
                                       bounds,
                                       maxiter=50,
                                       popsize=12,
                                       tol=1e-7,
                                       seed=seed,
                                       polish=True,
                                       updating='deferred',
                                       workers=1)  # keep single‑process

    de_vec = _beta_shape(n, de_result.x[0], de_result.x[1])
    de_val = _objective(de_vec)
    print(f"[DE] best surrogate = {de_val:.6f} (α={de_result.x[0]:.4f}, β={de_result.x[1]:.4f})")

    # insert DE result into the pool of seeds
    best_seeds.append(de_vec)
    best_seeds = sorted(best_seeds, key=lambda v: _objective(v))[:top_k]

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Allocate time: leave ~100 s for final SA, the rest for heavy opt.
    # ------------------------------------------------------------------
    final_sa_time = 100.0
    opt_deadline = deadline - final_sa_time
    if opt_deadline <= time.time():
        opt_deadline = time.time() + 1.0  # ensure at least tiny optimisation window

    # ------------------------------------------------------------------
    # Stage 2 – heavy optimisation (Adam + exact sub‑gradient)
    # ------------------------------------------------------------------
    print("\n--- Heavy optimisation (Adam) ---")
    seq_adam, val_adam = _adam_optimize(global_best_seq, n, opt_deadline,
                                        max_iters=2_000_000,
                                        beta_max=800.0,
                                        lr0=1.0)
    if seq_adam is not None and val_adam < global_best_val:
        global_best_seq = seq_adam.copy()
        global_best_val = val_adam
        print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    print("\n--- Exact projected sub‑gradient descent ---")
    seq_proj, val_proj = _subgradient_proj(global_best_seq, n, opt_deadline,
                                           max_iters=1_500_000,
                                           top_k=3,
                                           lr0=0.9)
    if val_proj < global_best_val:
        global_best_seq = seq_proj.copy()
        global_best_val = val_proj
        print(f"[overall‑SubProj] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final stochastic polishing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(global_best_seq, n, deadline,
                                              max_iters=800_000,
                                              init_temp_factor=0.03)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Sanity check (optional)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
