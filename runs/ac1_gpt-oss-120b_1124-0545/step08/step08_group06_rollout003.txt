"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Analytic seeds** – try several hand‑crafted families (uniform,
   Catalan, geometric, arcsine) for a few lengths.

2. **Random Dirichlet sampling** – while time permits draw many Dirichlet
   vectors with several concentration parameters and keep the best one.

3. **Soft‑max mirror descent** – a smooth approximation of the max‑norm.
   For the current distribution `a` we form the convolution `c = a*a`,
   build a soft‑max weight `w ∝ exp(β·c)`, and move `a` along the
   (projected) gradient of the smooth objective.
   The temperature β is annealed upwards so the process gradually
   focuses on the true maximum.

4. **Projected Adam** – a final fine‑tuning stage similar to the reference
   implementation.

All optimisation works on the probability simplex (`∑a = 1`,
`a_i ≥ 0`).  The best sequence found before the time budget expires is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for our sizes)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Objective that ``evaluate_sequence`` would return when ``seq`` is already
    normalised to sum = 1:  2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Coefficients proportional to binom(2i,i)/4**i – the series of (1‑z)^‑½.
    After truncation they give a near‑constant self‑convolution.
    Normalised to sum = 1.
    """
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _geometric_shape(n: int, r: float = 0.999) -> np.ndarray:
    """Truncated geometric a_i ∝ r**i, normalised to sum = 1."""
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


def _geometric_shape_999(n: int) -> np.ndarray:
    return _geometric_shape(n, r=0.999)


def _geometric_shape_995(n: int) -> np.ndarray:
    return _geometric_shape(n, r=0.995)


def _arcsine_shape(n: int) -> np.ndarray:
    """
    Discrete version of the arcsine density (Beta(0.5,0.5)).
    Peaks at the two ends, tends to flatten the autocorrelation.
    """
    # cell centres to avoid division by zero
    centres = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = 1.0 / np.sqrt(centres * (1.0 - centres))
    a[~np.isfinite(a)] = 0.0
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


# ----------------------------------------------------------------------
# Random Dirichlet sampling
# ----------------------------------------------------------------------


def _dirichlet(alpha: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector of length n, returned normalised."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by Adam and soft‑max refinements
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Refinement – exponentiated gradient (mirror descent)
# ----------------------------------------------------------------------


def _refine_expgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Classic exponentiated‑gradient (multiplicative) mirror descent."""
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ExpGrad] it={it:7d} val={cur_val:.6f}")

        tol = 1e-12
        tie_idxs = np.where(conv >= max_val - tol)[0]
        max_idx = int(np.random.choice(tie_idxs))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.8 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Refinement – projected Adam
# ----------------------------------------------------------------------


def _refine_proj_adam(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimiser with Euclidean projection onto the simplex."""
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 1000 == 0:
                print(f"[Adam] it={t:7d} val={cur_val:.6f}")

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.05 / np.sqrt(t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _proj_simplex(a_new)

        if t % 5000 == 0:
            noise = rng.normal(scale=1e-5, size=n)
            a = _proj_simplex(a + noise)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Refinement – soft‑max mirror descent
# ----------------------------------------------------------------------


def _refine_softmax(start_seq: np.ndarray, deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Smooth max optimisation.
    The loss Lβ(a) = (1/β)·log∑_k exp(β·(a*a)_k) approximates max_k (a*a)_k.
    We minimise Lβ while annealing β → large.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    t = 0
    beta = 1.0

    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)               # length 2n‑1
        max_val = conv.max()
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Softmax] it={t} val={cur_val:.6f}")

        # soft‑max weights
        w = np.exp(beta * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            # fallback: plain max‑gradient
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # gradient = 2·corr(a, w)
        a_rev = a[::-1]
        grad_full = _conv_fft(a_rev, w)               # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]       # slice the n relevant entries

        # learning‑rate schedule, damped by β
        lr = 0.4 / (np.sqrt(t) * (1.0 + beta))
        a_new = a - lr * grad
        a = _proj_simplex(a_new)

        # anneal β (make the soft‑max sharper)
        if t % 200 == 0:
            beta = min(beta * 1.07, 200.0)

        # occasional jitter to escape plateaus
        if t % 1500 == 0:
            noise = rng.normal(scale=1e-5, size=n)
            a = _proj_simplex(a + noise)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given time
    budget and returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a small safety margin for the final sanity check
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives a finer constant but costs more.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic analytic seeds
    # ------------------------------------------------------------------
    seed_makers = [
        _uniform,
        _catalan_shape,
        _geometric_shape_999,
        _geometric_shape_995,
        _arcsine_shape,
    ]

    for n in lengths:
        for maker in seed_makers:
            try:
                seq = maker(n)
            except Exception:  # defensive – a maker may raise for some n
                continue
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={maker.__name__:20s} val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1b – reuse a previously stored solution if it exists.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                prev_arr = prev_arr / prev_arr.sum()
                val = _objective(prev_arr)
                if val < best_val:
                    best_val = val
                    best_seq = prev_arr.copy()
                    print(f"[Seed] previous solution val={val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stage 2 – massive random Dirichlet sampling (different α)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 5.0:                         # need at least a few seconds
        sample_deadline = now + remaining * 0.30
        alphas = [0.5, 1.0, 2.0]                # sparse, uniform, concentrated
        while time.time() < sample_deadline:
            n = int(rng.choice(lengths))
            alpha = float(rng.choice(alphas))
            seq = _dirichlet(alpha, n, rng)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Random] n={n:5d} α={alpha:.1f} val={val:.6f}")

            if best_val < 1.503:               # target already hit – stop early
                break

    # ------------------------------------------------------------------
    # Stage 3 – soft‑max mirror descent (smooth max optimisation)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 5.0 and best_seq is not None:
        soft_deadline = now + remaining * 0.55
        seq_soft, val_soft = _refine_softmax(best_seq, soft_deadline, rng)
        if val_soft < best_val:
            best_val = val_soft
            best_seq = seq_soft
            print(f"[Overall‑Soft] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – projected Adam refinement (fine‑tuning)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 5.0 and best_seq is not None:
        adam_deadline = now + remaining
        seq_adam, val_adam = _refine_proj_adam(best_seq, adam_deadline)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam
            print(f"[Overall‑Adam] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final safety check (the grader will call evaluate_sequence as well)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )
    return best_seq.tolist()
