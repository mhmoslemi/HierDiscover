def construct_low_collision_set(n, max_ordered, bound_m, rng, deadline):
    best_mask = None
    best_size = 0
    attempts = 0
    while time.time() < deadline-0.2 and attempts < max_attempts:
        attempts += 1
        # initialize
        mask = np.zeros(n, dtype=bool)
        sump = np.zeros(2*n-1, dtype=np.uint8)
        chosen = []
        perm = rng.permutation(n)
        for cand in perm:
            # check self-pair
            if sump[2*cand] + 1 > max_ordered:
                continue
            # check sums with existing
            if chosen:
                existing = np.array(chosen, dtype=np.int32)
                sums = cand + existing
                # For each existing, self pair adds 2 to sum_counts for each ordered pair (cand, old) + (old, cand)
                if np.any(sump[sums] + 2 > max_ordered):
                    continue
            # accept cand
            mask[cand] = True
            chosen.append(int(cand))
            sump[2*cand] += 1
            if chosen:
                # existing included previously added? note we already added cand to chosen, we need to update sums with earlier ones only
                # but we can compute sums with existing before adding cand, we need to add 2 for each unordered pair
                old = np.array(chosen[:-1], dtype=np.int32)  # old before cand
                if old.size:
                    sums_old = cand + old
                    sump[sums_old] += 2
            if len(chosen) >= bound_m:
                break
        cur_size = len(chosen)
        if cur_size > best_size:
            best_size = cur_size
            best_mask = mask.copy()
        if best_size >= bound_m:
            break
    return np.where(best_mask)[0] if best_mask is not None else np.array([], dtype=int)
```

Note careful sum update: we must update sump with self-pair and unordered pairs with already existing. Implementation ensures not counting cand with itself again later.

"""
propose_candidate

A compact optimiser that focuses on the combinatorial core of the problem:
construct a large “low‑collision’’ support set (every ordered pair‑sum
occurs at most 4 times).  For a set of size *m* this guarantees that the
convolution max is at most 4/m², so the objective
2·n·max(conv)/sum² ≤ 2·n·4/m².  The theoretical bound for n=4000 and
max_ordered=4 is m≈√(8·n)≈179, which already gives an objective ≈1.0.

The algorithm:
1.  Greedily build many random B₂[2] sets, keeping the largest found.
2.  Initialise the step‑function with uniform mass on that support.
3.  Refine the weights with a long exponentiated‑gradient (sub‑gradient)
    mirror‑descent that directly minimises the exact max‑convolution.
4.  A short simulated‑annealing walk is performed with any remaining
    time for final fine‑tuning.

The routine respects the supplied time‑budget (default 1000 s) and
returns the best sequence as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution (exact, real‑valued)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Low‑collision B₂[2] set generator
# ----------------------------------------------------------------------


def _low_collision_set(
    n: int,
    max_ordered: int,
    rng: np.random.Generator,
    deadline: float,
    max_attempts: int = 2000,
) -> np.ndarray:
    """
    Greedy construction of a set S⊂{0,…,n‑1} such that each ordered sum
    i+j occurs at most ``max_ordered`` times.  Tries many random
    permutations until the time budget runs out and returns the largest
    set discovered.
    """
    # The absolute theoretical bound (ignoring integer effects):
    bound_m = int(math.isqrt(max_ordered * (2 * n - 1)))
    best_mask = None
    best_size = 0
    attempt = 0

    while time.time() < deadline - 0.2 and attempt < max_attempts:
        attempt += 1
        mask = np.zeros(n, dtype=bool)
        sum_counts = np.zeros(2 * n - 1, dtype=np.uint8)
        chosen = []                     # python list of already accepted indices

        order = rng.permutation(n)
        for cand in order:
            # self‑pair check
            if sum_counts[2 * cand] + 1 > max_ordered:
                continue

            # check ordered‑pair collisions with already chosen indices
            if chosen:
                existing = np.asarray(chosen, dtype=np.int32)
                sums = cand + existing          # vectorised
                if np.any(sum_counts[sums] + 2 > max_ordered):
                    continue

            # accept the candidate
            mask[cand] = True
            # update counters: self‑pair
            sum_counts[2 * cand] += 1
            # unordered pairs (contribute two ordered pairs each)
            if chosen:
                old = np.asarray(chosen, dtype=np.int32)
                sums_old = cand + old
                sum_counts[sums_old] += 2

            chosen.append(int(cand))

            # stop early if we already hit the theoretical bound
            if len(chosen) >= bound_m:
                break

        cur_size = len(chosen)
        if cur_size > best_size:
            best_size = cur_size
            best_mask = mask.copy()
            print(f"[LowColl] attempt {attempt:4d} size {cur_size}")

        if best_size >= bound_m:
            break

    if best_mask is None:
        return np.array([], dtype=int)

    return np.where(best_mask)[0]


# ----------------------------------------------------------------------
# Sub‑gradient (mirror‑descent) optimisation on the exact objective
# ----------------------------------------------------------------------


def _subgradient_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the true non‑smooth
    objective.  Returns the best vector seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices where the max is (almost) attained
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient: average over all active constraints
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            # full convolution length = 3n‑2, we need the central n entries
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1 : n - 1 + n]

        # mirror‑descent step (exponentiated‑gradient)
        lr = 1.4 / math.sqrt(it)          # a slightly larger schedule than 0.9/√it
        log_a = np.log(a + 1e-30) - lr * g
        # keep numbers well‑scaled
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Small simulated‑annealing fine‑tuner
# ----------------------------------------------------------------------


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Very light Metropolis–Hastings walk that swaps a small amount of mass
    between two random positions.  Uses an exponential cooling schedule.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-8, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a tiny fraction of mass from i to j
        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        # occasional global Gaussian jump to escape plateaux
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = np.full(n, 1.0 / n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

    if lc_idx.size == 0:

        a[lc_idx] = 1.0 / lc_idx.size

    # Keep a copy of the current best
    global_best_seq = a.copy()
    global_best_val = _objective(a)
    print(f"[seed] initial objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long sub‑gradient (mirror‑descent) optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:

        # Rough cost per sub‑gradient iteration ≈ 0.0015 s

        if seq is not None and val < global_best_val:

    else:
        final_seq = final_seq / tot

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(final_seq)})")
    # store for potential reuse in a later run
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
