def _project_simplex(v):
    v = v - np.max(v)  # maybe sub one? Not needed
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

Implementation of _project_simplex:

```
def _project_simplex(v):
    """Project onto the probability simplex (nonnegative, sum=1)."""
    # based on algorithm by Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all components zero or negative, uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm (time‑aware, < 1000 s)

1. **Analytic seeds** – generate a handful of classic windows (uniform,
   Catalan, Hann, Hamming, Blackman, Gaussians) for several lengths.
   Keep the best.

2. **Random Dirichlet sampling** – while ≈ 30 % of the budget remains,
   draw Dirichlet vectors with α∈{0.5,1,2} for the same lengths and keep
   any improvement.

3. **Greedy local refinement** – use a simple stochastic hill‑climbing
   (move a tiny amount of mass from a random index *i* to another *j*,
   keep the sum = 1 and non‑negativity).  The objective is recomputed
   via fast FFT convolution; any improvement is accepted.  This exploits
   the fact that the optimum is a point on the probability simplex and
   small “mass‑shuffling’’ steps often lower the maximal self‑convolution.
   The loop runs until the time budget is exhausted or the target
   1.503 is reached.

All numbers are kept non‑negative and normalised to sum = 1, therefore
the internal objective ``_objective`` equals the value returned by
``evaluate_sequence`` after the grader’s normalisation.

The routine returns a plain ``list[float]``; the grader will re‑evaluate
it with the security checks supplied in the task description.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from math import sqrt, pi

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a sequence already normalised
    to sum = 1:
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    tot = coeffs.sum()
    if tot == 0.0:
        return _uniform(n)
    return coeffs / tot


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0 - a1 * np.cos(2 * pi * i / (n - 1)) + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    """Discrete Gaussian centred in the middle, σ = sigma_factor·n."""
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _gaussian_015(n: int) -> np.ndarray:
    return _gaussian(n, sigma_factor=0.15)


def _gaussian_025(n: int) -> np.ndarray:
    return _gaussian(n, sigma_factor=0.25)


# ----------------------------------------------------------------------
# Random Dirichlet sampling
# ----------------------------------------------------------------------


def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet(alpha,…,alpha) vector."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, sum(x)=1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Greedy local search (mass‑shuffling hill climbing)
# ----------------------------------------------------------------------


def _local_search(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex.
    At each step a tiny amount of mass is transferred from one index to another.
    Any improvement in the objective is kept.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        # pick two distinct indices, ensure source has some mass
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # transfer a small random fraction (≤ 1 % of the source weight)
        delta = rng.random() * 0.01 * a[i]
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # numerical safety – keep non‑negativity
        if a_new[i] < 0.0:
            continue

        val = _objective(a_new)
        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            a = a_new
            print(f"[Local] it={it:7d} val={val:.6f}")
            if best_val < 1.503:
                break
        elif val < _objective(a):
            # accept any strict improvement over current point
            a = a_new

    return best_seq, best_val

    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------

    previous = globals().get("height_sequence_1")

        except Exception:

    else:
        previous = None

    # ------------------------------------------------------------------
