"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.
-----------------------------------------------------------------------
The algorithm is a multi‑stage search:

1. **Analytic seeds** – a collection of classic windows (uniform,
   Catalan, geometric, Hann, Hamming, Blackman, Gaussian, Beta).  Each
   candidate is normalised to unit L¹‑norm and evaluated.

2. **Random Dirichlet sampling** – draws from Dirichlet(α,…,α) with a
   variety of concentration parameters.  The best sample is kept.

3. **Smooth‑Adam refinement** – a gradient method that minimises a smooth
   surrogate of the ``max`` in the objective:
        Φ(f)= (1/β)·log Σₖ exp(β·(f∗f)ₖ) .
   The gradient of Φ can be written as a convolution, allowing a fast
   O(n log n) implementation via FFTs.  An Adam optimiser with
   Euclidean projection onto the probability simplex is used.
   The “temperature’’ β is increased slowly so that the surrogate
   converges to the true max.

4. **Optional fine‑tuning** – the original exponentiated‑gradient
   (mirror‑descent) and projected‑Adam refinements are run on any
   remaining time budget.

All stages respect a global time budget (default 1000 s) and keep the
best sequence encountered.  The returned list is guaranteed to be
non‑negative and sum‑normalised, so ``evaluate_sequence`` never clips it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for the sizes we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective of ``evaluate_sequence`` for a sequence already
    normalised to sum = 1 (i.e. 2·n·max(conv(seq,seq))).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by Adam refinements
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """a_i ∝ binom(2i,i)/4**i (coefficients of (1‑z)^‑½)."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _geometric_shape(n: int, r: float = 0.999) -> np.ndarray:
    """Truncated geometric distribution a_i ∝ r**i."""
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


def _hann_window(n: int) -> np.ndarray:
    """Hann (raised‑cosine) window."""
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    total = w.sum()
    return w / total


def _hamming_window(n: int) -> np.ndarray:
    """Hamming window."""
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    total = w.sum()
    return w / total


def _blackman_window(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    w = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    total = w.sum()
    return w / total


def _gaussian_window(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred at the middle. sigma = sigma_factor·n."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = a.sum()
    return a / total


def _beta_shape(n: int, a: float = 2.0, b: float = 2.0) -> np.ndarray:
    """Discrete Beta‑like shape ∝ x^{a‑1} (1‑x)^{b‑1} on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 1.0  # shift to avoid 0⁰
    x = i / (n + 1.0)
    a_vals = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    total = a_vals.sum()
    if total == 0.0:
        return _uniform(n)
    return a_vals / total


# ----------------------------------------------------------------------
# Random Dirichlet sampling (α may vary)
# ----------------------------------------------------------------------


def _dirichlet(alpha: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector of length n (already L¹‑normalised)."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s

# Refinement – exponentiated gradient (mirror descent)
# ----------------------------------------------------------------------


def _refine_expgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent with the negative‑entropy geometry (exponentiated gradient)."""
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ExpGrad] it={it:7d} val={cur_val:.6f}")

        # pick a random active index among those within a tiny tolerance
        tol = 1e-12
        tie_idxs = np.where(conv >= max_val - tol)[0]
        max_idx = int(np.random.choice(tie_idxs))

        # sub‑gradient for the selected max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.8 / np.sqrt(it)          # diminishing learning rate
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # improve numerical stability
        a = np.exp(log_a)
        a /= a.sum()
    return best_seq, best_val

# Refinement – projected Adam (plain)
# ----------------------------------------------------------------------


def _refine_proj_adam(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Projected Adam optimiser on the exact max‑convolution objective."""
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 1000 == 0:
                print(f"[Adam] it={t:7d} val={cur_val:.6f}")

        # sub‑gradient of the maximal entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.05 / np.sqrt(t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _proj_simplex(a_new)

        # occasional tiny perturbation
        if t % 5000 == 0:
            noise = rng.normal(scale=1e-5, size=n)
            a = _proj_simplex(a + noise)

    return best_seq, best_val

def _refine_smooth_adam(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        Φ(f) = (1/β)·log Σₖ exp(β·(f∗f)ₖ)
    The surrogate approaches the true max as β grows.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    rng = np.random.default_rng()

    # temperature schedule – start soft, end sharp
    temp = 0.5          # β in the definition above
    max_temp = 50.0
    inc_factor = 1.001

    while time.time() < deadline:
        t += 1

        conv = _conv_fft(a, a)                     # length 2n‑1
        max_c = conv.max()

        # soft‑max weights (stable computation)
        exp_vals = np.exp(temp * (conv - max_c))
        w = exp_vals / exp_vals.sum()             # Σₖ wₖ = 1

        # gradient = 2·(a_rev ∗ w) restricted to n entries
        a_rev = a[::-1]
        grad_full = _conv_fft(a_rev, w)            # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice of length n

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.05 / np.sqrt(t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _proj_simplex(a_new)

        # temperature increase
        if temp < max_temp:
            temp *= inc_factor

        # occasional exact evaluation
        if t % 200 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SmoothAdam] it={t:7d} β={temp:.3f} val={cur_val:.6f}")

        # tiny random jitter every few thousand steps (helps escape plateaus)
        if t % 5000 == 0:
            jitter = rng.normal(scale=1e-5, size=n)
            a = _proj_simplex(a + jitter)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given
    time budget and returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # tiny safety margin for the final sanity check
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives more flexibility but costs more.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic analytic seeds
    # ------------------------------------------------------------------
    seed_makers = (
        _catalan_shape,
        _uniform,
        lambda n: _geometric_shape(n, r=0.999),
        lambda n: _geometric_shape(n, r=0.995),
        _hann_window,
        _hamming_window,
        _blackman_window,
        lambda n: _gaussian_window(n, sigma_factor=0.25),
        lambda n: _gaussian_window(n, sigma_factor=0.35),
        lambda n: _beta_shape(n, a=2.0, b=2.0),
        lambda n: _beta_shape(n, a=0.8, b=0.8),
    )
    for n in lengths:
        for maker in seed_makers:
            try:
                seq = maker(n)
                val = _objective(seq)
                if val < best_val:
                    best_val = val
                    best_seq = seq.copy()
                    print(f"[Seed] n={n:5d} type={maker.__name__} val={val:.6f}")
            except Exception:
                # any unexpected numerical issue – skip this seed
                continue

    # ------------------------------------------------------------------
    # Include a previously submitted solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                prev_arr = prev_arr / prev_arr.sum()
                val = _objective(prev_arr)
                if val < best_val:
                    best_val = val
                    best_seq = prev_arr.copy()
                    print(f"[Seed] previous solution val={val:.6f}")
        except Exception:
            pass

    # Ensure we have **some** starting point
    if best_seq is None:
        # fallback to uniform of the smallest length
        n0 = lengths[0]
        best_seq = _uniform(n0)
        best_val = _objective(best_seq)

    # ------------------------------------------------------------------
    # Stage 2 – massive random Dirichlet sampling
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    # allocate roughly half of the remaining budget to random sampling
    sample_deadline = now + remaining * 0.45

    alphas = [0.2, 0.5, 1.0, 2.0, 5.0]
    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = float(rng.choice(alphas))
        seq = _dirichlet(alpha, n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.1f} val={val:.6f}")

        # early exit if we already beat the target
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # Stage 3 – smooth‑Adam refinement (core improvement step)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # give about 30 % of the remaining budget to the smooth optimiser
        smooth_deadline = time.time() + (deadline - time.time()) * 0.30
        seq_smooth, val_smooth = _refine_smooth_adam(best_seq, smooth_deadline)
        if val_smooth < best_val:
            best_val = val_smooth
            best_seq = seq_smooth
            print(f"[Overall‑SmoothAdam] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – fallback refinements (exp‑gradient & projected Adam)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # remaining time allocated to the classic refinements
        final_deadline = deadline
        seq_exp, val_exp = _refine_expgradient(best_seq, final_deadline * 0.50)
        if val_exp < best_val:
            best_val = val_exp
            best_seq = seq_exp
            print(f"[Overall‑Exp] improved best to {best_val:.6f}")

    if time.time() < deadline:
        seq_proj, val_proj = _refine_proj_adam(best_seq, deadline)
        if val_proj < best_val:
            best_val = val_proj
            best_seq = seq_proj
            print(f"[Overall‑Adam] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – the grader will also call evaluate_sequence
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    # Return a plain Python list as required by the grader
    return best_seq.tolist()
