"""
Simulated‑annealing optimizer for the step‑function lower bound.

The algorithm:
* builds a small but diverse pool of seed vectors (uniform, geometric,
  power‑law, random softmax, sparse random supports);
* picks the best few seeds using the exact objective;
* for each selected seed runs a fast simulated‑annealing / hill‑climbing
  routine that repeatedly transfers a small amount of mass between two
  positions (or moves an entire mass to a fresh index).  The move is
  accepted if it improves the objective or with a probability based on a
  decreasing temperature.
* occasional large Gaussian “jumps’’ are injected to escape plateaus.
* the best sequence found before the time budget expires is stored in the
  global variable ``height_sequence_1`` and returned as a plain list of
  floats.

The implementation works with the full‑length vector (default n=4000) but
keeps the per‑iteration cost low by using an FFT‑based convolution
(`_conv_fft`) which is ~O(n log n).  All numbers are kept non‑negative and
normalised to sum = 1, so the clipping performed inside
``evaluate_sequence`` never changes the candidate.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (exact, real‑valued)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full mode).
    Returns an array of length len(a) + len(b) - 1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the grader (2·n·max(conv(seq,seq))).
    The sequence is assumed to be non‑negative and normalised to sum = 1.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * seq.size * conv.max())


# ----------------------------------------------------------------------
# Simple families for seed generation (all return a normalised np.ndarray)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _sparse_random(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random support of size m (1 ≤ m ≤ n)."""
    if m >= n:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser (dense representation)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float,
                     max_no_imp: int = 20000) -> tuple[np.ndarray, float]:
    """
    Fast SA / hill‑climbing on the exact non‑smooth objective.
    Returns the best sequence seen and its objective.
    """
    rng = np.random.default_rng()
    n = start_seq.size

    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    # initial temperature – a small fraction of the current objective
    T = max(1e-9, cur_val * 0.03)
    decay = 0.99995

    it = 0
    no_imp = 0

    while time.time() < deadline:
        it += 1
        T *= decay
        # --- propose a move -------------------------------------------------
        move_type = rng.integers(3)  # 0: tiny shift, 1: move to new index, 2: full transfer
        new = cur.copy()

        # pick a source index (must have some mass)
        src = rng.integers(n)
        while new[src] < 1e-12:
            src = rng.integers(n)

        if move_type == 0:
            # tiny transfer of a fraction of src to a random destination
            dst = rng.integers(n)
            while dst == src:
                dst = rng.integers(n)
            max_eps = new[src]
            eps = rng.random() * 0.04 * max_eps  # ≤4 % of source mass
            new[src] -= eps
            new[dst] += eps

        elif move_type == 1:
            # transfer a fraction to a brand‑new index (not currently in support)
            dst = rng.integers(n)
            while new[dst] > 1e-12:  # enforce ‘new’ position
                dst = rng.integers(n)
            max_eps = new[src]
            eps = rng.random() * 0.2 * max_eps   # up to 20 % of source mass
            new[src] -= eps
            new[dst] += eps

        else:  # move_type == 2
            # move the *entire* mass of src to a fresh index
            dst = rng.integers(n)
            while new[dst] > 1e-12:
                dst = rng.integers(n)
            eps = new[src]
            new[src] = 0.0
            new[dst] = eps

        # renormalise (tiny numerical drift may appear)
        total = new.sum()
        if total == 0.0:
            continue
        new /= total

        new_val = _objective(new)

        # --- acceptance test -----------------------------------------------
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_val = new_val
            no_imp = 0
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[Anneal] it={it:7d} best={best_val:.6f}")

        else:
            no_imp += 1

        # occasional large Gaussian jump to shake out of plateaus
        if it % 2000 == 0:
            jump = cur + rng.normal(scale=0.005, size=n)
            jump = np.clip(jump, 0.0, None)
            if jump.sum() == 0.0:
                continue
            jump /= jump.sum()
            jump_val = _objective(jump)
            if jump_val < cur_val or rng.random() < math.exp(-(jump_val - cur_val) / (T + 1e-12)):
                cur = jump
                cur_val = jump_val
                if jump_val < best_val:
                    best_val = jump_val
                    best_seq = jump.copy()
                    print(f"[Jump]   it={it:7d} best={best_val:.6f}")

        # restart if stuck for a while
        if no_imp >= max_no_imp:
            # re‑initialise from the current best (adds a small perturbation)
            cur = best_seq.copy()
            cur += rng.normal(scale=0.002, size=n)
            cur = np.clip(cur, 0.0, None)
            if cur.sum() == 0.0:
                cur = best_seq.copy()
            else:
                cur /= cur.sum()
            cur_val = _objective(cur)
            no_imp = 0
            T = max(1e-9, cur_val * 0.03)  # warm‑up temperature again

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation (diverse, cheap)
# ----------------------------------------------------------------------


def _make_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    seeds = []
    seeds.append(_uniform(n))
    # a few geometric families
    for r in np.linspace(0.2, 0.9, 5):
        seeds.append(_geom(n, r))
    # power‑law families
    for p in np.linspace(0.0, 2.5, 6):
        seeds.append(_power(n, p))
    # random softmax
    for _ in range(5):
        seeds.append(_random_softmax(n, rng))
    # sparse random supports with varied sizes
    for m in (30, 60, 90, 120, 150):
        for _ in range(2):
            seeds.append(_sparse_random(n, m, rng))
    # include a previously discovered construction if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev = np.asarray(previous, dtype=np.float64)
            if prev.shape == (n,):
                seeds.append(prev / prev.sum())
        except Exception:
            pass
    # final clean‑up: ensure non‑negativity and normalisation
    clean = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            clean.append(s / tot)
    return clean


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function vector that minimises
    ``evaluate_sequence``.  The routine respects the allotted time budget
    and returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 1.5

    rng = np.random.default_rng(seed + 12345)
    n = 4000
    print(f"\n=== Optimising length {n} (budget {budget_s}s) ===")

    # ------------------------------------------------------------------
    # 1) Generate a modest pool of diverse seeds
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, rng)

    # ------------------------------------------------------------------
    # 2) Quick evaluation of seeds – keep the best few
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [s for s, _ in seed_vals[:top_k]]
    global_best_seq = seed_vals[0][0].copy()
    global_best_val = seed_vals[0][1]
    print(f"[seed] best initial objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Short annealing runs from the top seeds (use a fraction of budget)
    # ------------------------------------------------------------------
    allocated = (budget_s * 0.2)  # 20 % of the time for quick diversifications
    quick_deadline = start_time + allocated
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > quick_deadline:
            break
        print(f"[quick] start SA from seed {idx}")
        seq, val = _anneal_optimize(seed_vec, quick_deadline)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[quick] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # 4) Long annealing from the current best (use the rest of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[main] long annealing phase")
        seq, val = _anneal_optimize(global_best_seq, deadline)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[main] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # 5) Final sanity check – fallback to uniform if something went wrong
    # ------------------------------------------------------------------
    if global_best_seq is None or not np.isfinite(global_best_val):
        print("[fallback] using uniform distribution")
        global_best_seq = _uniform(n)

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})")
    globals()["height_sequence_1"] = np.copy(global_best_seq)
    return global_best_seq.tolist()
