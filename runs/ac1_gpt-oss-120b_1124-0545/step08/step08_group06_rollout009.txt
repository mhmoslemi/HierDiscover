"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Deterministic seeds** – a few analytical families (uniform,
   Catalan, geometric) for a few modest lengths (2000, 4000, 8000).

2. **Random Dirichlet sampling** – while time permits we draw many
   Dirichlet vectors with several α‑parameters and keep the best one.

3. **Two complementary refinements**  
   * **Full‑active exponentiated‑gradient** – a mirror‑descent update
     that uses the *average* sub‑gradient of *all* active convolution
     entries (rather than a single random one).  The learning‑rate
     decays as 1/√it and the iterate is projected onto the probability
     simplex after every step.
   * **Projected sub‑gradient descent** – an additive update with a
     similar schedule, followed by an Euclidean projection onto the
     simplex.  Occasional tiny Gaussian noise is added to escape shallow
     plateaus.

4. **Local hill‑climbing** – small random transfers of mass between two
   coordinates; an improvement is kept immediately.  This final stage
   polishes the best vector found so far.

All stages are time‑aware; each consumes a fraction of the remaining
budget and the best vector over the whole run is returned as a plain
Python list.  Only the standard library, NumPy and SciPy (for the
optional linear‑programming baseline) are used.
"""

import time
import numpy as np
from scipy.optimize import linprog  # optional; used only if available

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for our sizes)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Objective equivalent to ``evaluate_sequence`` when the vector is
    already normalised to sum = 1:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by all refinement stages
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Catalan‑type coefficients: c_i = binom(2i,i) / 4**i.
    After truncation the (full) self‑convolution is constant for the
    first n entries.  Normalised to sum = 1.
    """
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2.0 * i) * (2.0 * i - 1.0)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _geometric_shape(n: int, r: float = 0.999) -> np.ndarray:
    """Truncated geometric distribution a_i ∝ r**i, normalised to sum = 1."""
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


# ----------------------------------------------------------------------
# Random Dirichlet sampling
# ----------------------------------------------------------------------


def _dirichlet(alpha: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector and normalise to sum = 1."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Sub‑gradient helpers
# ----------------------------------------------------------------------


def _subgradient_maxconv(a: np.ndarray, tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Return a (sub‑)gradient of the max‑convolution functional and the
    current maximal convolution value.
    The gradient is the *average* of the gradients of all active
    constraints (i.e. all k where conv[k] is within ``tol`` of the max).
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    max_val = conv.max()
    active = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(a)
    # Aggregate contributions from every active shift
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i               # partner index inside the vector
        g[idx_i] += 2.0 * a[rev_idx]      # derivative of a_i*a_{k-i}
    if len(active) > 0:
        g /= float(len(active))           # average – improves stability
    return g, max_val


# ----------------------------------------------------------------------
# Refinement – full‑active exponentiated gradient (mirror descent)
# ----------------------------------------------------------------------


def _refine_expgradient_full(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using the negative entropy (exponentiated gradient).
    The sub‑gradient is the *average* over all active convolution entries.
    Returns the best vector observed before ``deadline``.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        g, max_val = _subgradient_maxconv(a)
        # learning rate decays like 1/√it; a slightly smaller prefactor works well
        lr = 0.3 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)               # improve numerical stability
        a = np.exp(log_a)
        a /= a.sum()
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ExpFull] it={it:7d} val={cur_val:.6f}")
        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            rng = np.random.default_rng()
            a = _proj_simplex(a + rng.normal(scale=1e-6, size=n))
    return best_seq, best_val


# ----------------------------------------------------------------------
# Refinement – projected additive sub‑gradient descent
# ----------------------------------------------------------------------


def _refine_proj_subgrad(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple additive sub‑gradient descent with Euclidean projection onto
    the simplex.  The gradient is the average over all active constraints.
    Returns the best vector seen before ``deadline``.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    while time.time() < deadline:
        it += 1
        g, max_val = _subgradient_maxconv(a)
        lr = 0.5 / np.sqrt(it)               # diminishing stepsize
        a_new = a - lr * g
        a = _proj_simplex(a_new)
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Subgrad] it={it:7d} val={cur_val:.6f}")
        # tiny random perturbation every few thousand steps
        if it % 5000 == 0:
            rng = np.random.default_rng()
            a = _proj_simplex(a + rng.normal(scale=1e-7, size=n))
    return best_seq, best_val

# Local hill‑climbing refinement
# ----------------------------------------------------------------------


def _refine_local_search(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Random local mass transfers (i→j) with immediate acceptance if the
    objective improves.  A simple way to polish the current best point.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        # move at most a small fraction of the smaller entry
        delta_max = min(best_seq[i], 0.001)
        if delta_max <= 0:
            continue
        delta = rng.uniform(0.0, delta_max)
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _proj_simplex(cand)   # keep on simplex and remove numerical negatives
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand
            print(f"[Local] i={i}→j={j} δ={delta:.6e} val={val:.6f}")
    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional LP helper (original “linear‑programming” idea)
# ----------------------------------------------------------------------


def _lp_improvement(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the linear program that maximises Σ g_j subject to
        f * g ≤ M   (full convolution)
        g_j ≥ 0.
    Returns the optimised g (not rescaled).  Uses SciPy's ``linprog`` with a
    sparse constraint matrix.  If SciPy is unavailable or the solve fails,
    the function returns the input ``f`` unchanged.
    """
    try:
        n = f.shape[0]
        L = 2 * n - 1
        # Build sparse matrix A where A[k,i] = f_i if 0 ≤ k-i < n else 0
        from scipy import sparse

        data = []
        rows = []
        cols = []
        for i in range(n):
            # The i‑th coefficient of f contributes to rows k = i … i+n-1
            row_start = i
            row_end = i + n   # exclusive
            rows.extend(range(row_start, row_end))
            cols.extend([i] * n)
            data.extend([f[i]] * n)

        A = sparse.csr_matrix((data, (rows, cols)), shape=(L, n))
        b = np.full(L, M, dtype=np.float64)

        # maximise Σ g_j  → minimise –Σ g_j
        c = -np.ones(n, dtype=np.float64)

        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": 30},
        )
        if res.success:
            return res.x
        else:
            return f
    except Exception:
        return f

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget (default 1000 s) and returns the best sequence found as a
    plain Python list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin so that the final sanity check does not run out of time
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Lengths to explore (larger n gives a finer theoretical lower bound)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic analytical seeds
    # ------------------------------------------------------------------
    for n in lengths:
        for maker in (
            _catalan_shape,
            _uniform,
            lambda n: _geometric_shape(n, r=0.999),
            lambda n: _geometric_shape(n, r=0.995),
        ):
            try:
                seq = maker(n)
                val = _objective(seq)
                if val < best_val:
                    best_val = val
                    best_seq = seq.copy()
                    print(f"[Seed] n={n:5d} type={maker.__name__} val={val:.6f}")
            except Exception:
                continue

    # Stage 2 – re‑use a previously found solution if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size > 0 and np.all(arr >= 0):
                arr = arr / arr.sum()
                val = _objective(arr)
                if val < best_val:
                    best_val = val
                    best_seq = arr.copy()
                    print(f"[Seed] previous solution val={val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stage 3 – massive Dirichlet sampling while time permits
    # ------------------------------------------------------------------
    now = time.time()

    alphas = [0.5, 1.0, 2.0]
    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = float(rng.choice(alphas))
        cand = _dirichlet(alpha, n, rng)
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Random] n={n:5d} α={alpha:.1f} val={val:.6f}")
        # early exit when we already beat the target comfortably
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # Stage 4 – optional LP “boost” (one shot) if we still have time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # compute current M = max(conv(f,f))
        M = np.max(_conv_fft(best_seq, best_seq))
        # attempt a single LP improvement (may be costly, use a tiny time slice)
        lp_deadline = time.time() + (deadline - time.time()) * 0.10
        # limit runtime of the LP solve by a timeout in ``linprog`` options
        try:
            g0 = _lp_improvement(best_seq, M)
            # rescale g0 to the canonical ℓ1‑norm √(2n) as described in the paper
            scale = np.sqrt(2.0 * len(best_seq)) / g0.sum()
            g = g0 * scale
            # simple convex combination (small step)
            t = 0.2
            cand = (1 - t) * best_seq + t * g
            cand = cand / cand.sum()
            val = _objective(cand)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[LPBoost] improved value to {val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stage 5 – full‑active exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + (deadline - time.time()) * 0.30
        seq_exp, val_exp = _refine_expgradient_full(best_seq, exp_deadline)
        if val_exp < best_val:
            best_val = val_exp
            best_seq = seq_exp
            print(f"[Overall‑Exp] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – projected additive sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.20
        seq_sub, val_sub = _refine_proj_subgrad(best_seq, sub_deadline)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = seq_sub
            print(f"[Overall‑Sub] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 7 – final local greedy polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        local_deadline = deadline
        seq_loc, val_loc = _refine_local_search(best_seq, local_deadline)
        if val_loc < best_val:
            best_val = val_loc
            best_seq = seq_loc
            print(f"[Overall‑Local] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will also invoke evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    # Return a plain Python list as required by the grader
    return best_seq.tolist()
