def _project_simplex(v):
    # Based on algorithm by Wang & Carreira-Perpinan
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].sum() / rho   # Actually theta = (sum_{i=1}^rho u_i - 1) / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative sequence) that minimizes
evaluate_sequence.  The search proceeds in three stages:

1. **Seed generation** – a handful of analytically motivated shapes
   (uniform, sine window, symmetric geometric, harmonic, inverse‑sqrt,
   random Dirichlet) are created and the best few (by the true objective)
   are kept.

2. **Low‑dimensional Adam optimisation** – each seed is expressed as a
   soft‑max of a low‑dimensional Fourier basis (constant + a few cosine /
   sine terms).  Adam minimises a smoothed version of the max‑convolution
   (log‑sum‑exp) while staying on the probability simplex.  This explores
   global shape changes far more efficiently than element‑wise updates.

3. **Fine‑grained projected sub‑gradient descent** – the best vector from
   stage 2 is refined by a classic projected sub‑gradient method on the
   true max‑convolution objective.  A simple simulated‑annealing
   polishing step finishes the search.

The routine respects a time budget (default 1000 s) and stores the best
vector globally under ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate used for the search (sum(seq)=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simplexes & projections
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = (cssv[cond].max()) / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Basis for low‑dimensional optimisation (Fourier‑type)
# ----------------------------------------------------------------------


def _basis_matrix(n: int, K: int) -> np.ndarray:
    """
    Returns an (n, 2*K+1) matrix with columns:
    [1, cos(2πk i/n), sin(2πk i/n)] for k=1..K.
    """
    i = np.arange(n, dtype=np.float64)
    cols = [np.ones(n, dtype=np.float64)]
    for k in range(1, K + 1):
        cols.append(np.cos(2.0 * np.pi * k * i / n))
        cols.append(np.sin(2.0 * np.pi * k * i / n))
    return np.stack(cols, axis=1)


# ----------------------------------------------------------------------
# Seed families (non‑negative, sum ≈ 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a /= a.sum()
    return a


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric two‑sided geometric progression."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    a /= a.sum()
    return a


def _harmonic(n: int) -> np.ndarray:
    """a_i ∝ 1/(i+1) + 1/(n-i)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (i + 1.0) + 1.0 / (n - i)
    a /= a.sum()
    return a


def _inv_sqrt(n: int) -> np.ndarray:
    """a_i ∝ 1/√((i+1)*(n-i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 1.0) * (n - i))
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    vec /= vec.sum()
    return vec


def _gather_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a modest set of diverse seeds."""
    seeds = [
        _uniform(n),
        _sine_window(n),
        _sym_geom(n, 0.5),
        _sym_geom(n, 0.7),
        _harmonic(n),
        _inv_sqrt(n),
    ]
    # few random Dirichlet draws
    for conc in (0.2, 0.5, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))
    # previous solution if available
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev = np.asarray(previous, dtype=np.float64)
            if prev.shape == (n,):
                prev = prev / prev.sum()
                seeds.append(prev)
        except Exception:
            pass
    return seeds


# ----------------------------------------------------------------------
# Low‑dimensional Adam optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _adam_optimize_lowdim(start_w: np.ndarray,
                          phi: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 60000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a low‑dimensional parameterisation.
    The vector a is obtained as a soft‑max of the linear basis:
        a_i = exp((phi @ w)_i) / Σ_j exp((phi @ w)_j)
    A smoothed max (log‑sum‑exp) surrogate is minimised.
    """
    eps = 1e-30
    w = start_w.copy()
    dim = w.shape[0]
    m = np.zeros(dim, dtype=np.float64)
    v2 = np.zeros(dim, dtype=np.float64)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.5

    beta = 0.5
    beta_mul = 1.001
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1

        # logits → simplex via soft‑max
        logits = phi @ w
        log_max = np.max(logits)
        exp_logits = np.exp(logits - log_max)
        a = exp_logits / np.sum(exp_logits)

        # true objective (for bookkeeping)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LowDim Adam] it={it:6d} best={cur_val:.6f}")

        # increase smoothing temperature
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp on the convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w_smooth = np.exp(beta * (conv - shift))
        s = w_smooth.sum()
        if s == 0.0:
            continue
        w_smooth /= s

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w_smooth)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

        # chain‑rule to logits (soft‑max Jacobian)
        dot = np.dot(grad_a, a)
        grad_logits = a * (grad_a - dot)

        # finally to low‑dim parameters
        grad_w = phi.T @ grad_logits

        # Adam update
        lr = lr0 / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad_w
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_w ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        w = w - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val

# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 300000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror‑descent (exponentiated‑gradient) on the
    true max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient style update
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)  # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val

def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200000,
                         init_temp_factor: float = 0.02,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Random Dirichlet reshuffles of a small subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)             # 2…6 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

        print(f"\n--- Low‑dim Adam run {idx+1}/{len(best_seeds)} ---")
        # initialise low‑dim parameters by fitting log‑probabilities
        eps = 1e-12
        try:
            w0, *_ = np.linalg.lstsq(phi,
                                     np.log(seed_vec + eps),
                                     rcond=None)
        except Exception:
            w0 = np.zeros(phi.shape[1], dtype=np.float64)

        seq_adam, val_adam = _adam_optimize_lowdim(
            w0, phi, n, deadline, max_iters=60000
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Projected sub‑gradient refinement (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient refinement ---")
        seq_sub, val_sub = _subgradient_refine(
            global_best_seq, n, deadline, max_iters=300000
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[overall‑SubGrad] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Optional simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=200000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Optional up‑sampling (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_big = n * 2
        x_old = np.linspace(0.0, 1.0, n, dtype=np.float64)
        x_big = np.linspace(0.0, 1.0, n_big, dtype=np.float64)
        upsampled = np.interp(x_big, x_old, global_best_seq)
        upsampled = upsampled / upsampled.sum()
        print(f"\n--- Upsampling to n={n_big} and final refinement ---")
        seq_big, val_big = _subgradient_refine(
            upsampled, n_big, deadline, max_iters=200000
        )
        if val_big < global_best_val:
            global_best_seq = seq_big.copy()
            global_best_val = val_big
            n = n_big
            print(f"[Upsample‑Refine] new global best = {val_big:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

This implementation introduces a compact low‑dimensional Adam optimiser that searches over a Fourier‑style basis, followed by a high‑precision projected sub‑gradient refinement and a light simulated‑annealing polish.  The added harmonic and inverse‑sqrt seeds, together with the new optimisation pipeline, give the optimiser fresh directions that were not explored by the previous code and typically achieve a lower upper bound within the allotted 1000 seconds.
