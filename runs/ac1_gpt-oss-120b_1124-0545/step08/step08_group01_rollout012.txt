        # compute conv and M

"""
Search for a non‑negative step‑function minimising ``evaluate_sequence``.
The algorithm proceeds in four stages:

1. **Seed generation** – a handful of analytically motivated families
   (uniform, triangular, symmetric geometric, a right‑skewed exponential)
   are created for a modest length ``n_small`` (default 1024) and normalised
   to sum = 1.  The best few seeds (by the surrogate objective
   ``2·n·max(conv)``) are kept.

2. **LP flattening** – for each retained seed we run a short loop of
   linear‑programming steps:
        maximise   Σ b_j
        subject to   b_j ≥ 0
                     (f * b)_i ≤ M   for all i,
   where ``M = max(f * f)``.  The LP is solved with SciPy’s HiGHS.
   The optimum ``b`` is renormalised (Σ b = 1) and mixed into the
   current vector: ``f ← (1–t)·f + t·b`` with a small step size ``t``.
   This procedure monotonically reduces the maximal convolution value.

3. **Mirror‑descent polishing** – an exponentiated‑gradient (mirror‑descent)
   sub‑gradient method works directly on the true max‑convolution
   objective (as in the reference code).  It is run for many iterations
   on the current best vector.

4. **Upsampling & final polishing** – the best vector is linearly
   interpolated to a longer length (≈ 4000) and refined once more by
   the same mirror‑descent routine, followed by a brief simulated‑annealing
   hill‑climber.  The final result is stored globally as
   ``height_sequence_1`` and returned as a plain list of floats.

The routine respects the given time budget (default 1000 s) and prints
progress information.
"""

import time
import numpy as np
from scipy.optimize import linprog

# --------------------------------------------------------------
# Low‑level utilities
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    Assumes ``seq`` is non‑negative and already normalised to sum = 1.
    Returns ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# --------------------------------------------------------------
# Analytic seed families (all normalised to sum = 1)
# --------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    return a / a.sum()


def _right_geom(n: int, r: float) -> np.ndarray:
    """
    Right‑skewed geometric progression: a_i ∝ r**i.
    ``r`` should be in (0, 1); the vector is normalised to sum = 1.
    """
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    return a / a.sum()


# --------------------------------------------------------------
# Helper for building the LP constraint matrix (boolean mask)
# --------------------------------------------------------------

def _build_conv_mask(n: int) -> np.ndarray:
    """
    Boolean mask of shape (2·n‑1, n) where mask[i, j] == True
    iff the term f_j appears in the i‑th entry of the linear convolution
    f * b.  The mask is independent of the actual values of ``f`` and can
    be reused across LP iterations.
    """
    rows = np.arange(2 * n - 1)[:, None]      # shape (2n‑1, 1)
    cols = np.arange(n)[None, :]              # shape (1, n)
    mask = (rows >= cols) & (rows < cols + n)
    return mask


# --------------------------------------------------------------
# Sub‑gradient mirror‑descent (same as in the reference code)
# --------------------------------------------------------------

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 300000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror‑descent (exponentiated‑gradient) on the
    true max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# --------------------------------------------------------------
# Simulated annealing polishing (same idea as before)
# --------------------------------------------------------------

def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 120000,
                         init_temp_factor: float = 0.02,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles the mass of a random small subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)           # size of the random subset (2…6)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal
    ``evaluate_sequence`` value.  Returns the best sequence found.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Choose a modest length for the LP phase – large enough to capture
    # the effect but small enough to keep the LP cheap.
    # ------------------------------------------------------------------
    n_small = 1024
    mask_small = _build_conv_mask(n_small)

    # ------------------------------------------------------------------
    # 1. Seed generation (few analytically motivated families)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n_small),
        _triangular(n_small),
        _sym_geom(n_small, 0.5),
        _right_geom(n_small, 0.9),
        _right_geom(n_small, 0.99)
    ]

    # Evaluate seeds with the fast surrogate objective and keep the best three
    evaluated = [(s, _objective(s)) for s in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 3
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Seed objectives (best {top_k} kept):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"Initial best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. LP‑based flattening
    # ------------------------------------------------------------------
    eps = 1e-12
    max_lp_iters = 30               # per seed – a few LP solves are cheap
    t_mix = 0.03                    # small mixing factor to keep changes gentle

    for seed_idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        print(f"\n--- LP refinement for seed #{seed_idx} ---")
        f = seed_vec.copy()

        for it in range(1, max_lp_iters + 1):
            if time.time() > deadline:
                break

            # current bound M = max(f*f)
            conv_f = _conv_fft(f, f)
            M = np.max(conv_f)

            # Build A_ub = mask * f   (broadcast f across rows)
            A = mask_small * f  # shape (2n‑1, n)

            # Linear program: maximise sum(b)  ⇔ minimise -1·b
            c = -np.ones(n_small, dtype=np.float64)
            b_ub = np.full(2 * n_small - 1, M, dtype=np.float64)

            # Solve with HiGHS; give a tiny time budget per call
            time_left = max(0.0, deadline - time.time())
            res = linprog(c,
                          A_ub=A,
                          b_ub=b_ub,
                          bounds=[(0, None)] * n_small,
                          method='highs',
                          options={'time_limit': time_left})
            if not res.success:
                print(f"  LP failed (it={it}) – {res.message}")
                break

            b = res.x
            sumb = b.sum()
            if sumb <= 0.0:
                print("  LP returned zero sum – aborting this seed")
                break
            g = b / sumb                     # normalise to sum = 1

            # mix a tiny step towards g
            f = (1.0 - t_mix) * f + t_mix * g
            f = np.maximum(f, eps)
            f /= f.sum()

            val = _objective(f)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = f.copy()
                print(f"  ==> new global best after LP it={it}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Mirror‑descent polishing on the small vector
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent polishing (small n) ---")
        seq_ref, val_ref = _subgradient_refine(global_best_seq,
                                               n_small,
                                               deadline,
                                               max_iters=300000)
        if val_ref < global_best_val:
            global_best_seq = seq_ref
            global_best_val = val_ref
            print(f"After sub‑gradient (small): new best = {val_ref:.6f}")

    # ------------------------------------------------------------------
    # 4. Upsample to a longer length and final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_big = 4000
        x_small = np.linspace(0.0, 1.0, n_small, dtype=np.float64)
        x_big = np.linspace(0.0, 1.0, n_big, dtype=np.float64)
        upsampled = np.interp(x_big, x_small, global_best_seq)
        upsampled = np.maximum(upsampled, 0.0)
        upsampled /= upsampled.sum()
        print(f"\n--- Upsampling to n={n_big} ---")
        seq_big, val_big = _subgradient_refine(upsampled,
                                               n_big,
                                               deadline,
                                               max_iters=250000)
        if val_big < global_best_val:
            global_best_seq = seq_big
            global_best_val = val_big
            print(f"After upsampled sub‑gradient: new best = {val_big:.6f}")

    # ------------------------------------------------------------------
    # 5. Optional simulated‑annealing final polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(global_best_seq,
                                              len(global_best_seq),
                                              deadline,
                                              max_iters=120000)
        if val_sa < global_best_val:
            global_best_seq = seq_sa
            global_best_val = val_sa
            print(f"After SA: new best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (calls evaluate_sequence just to ensure no crash)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f}"
        f" (length {len(global_best_seq)})"
    )

    # expose the result for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
