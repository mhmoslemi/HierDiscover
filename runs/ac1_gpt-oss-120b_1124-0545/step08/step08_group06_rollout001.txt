def _grad_softmax_max(a, beta):
    conv = _conv_fft(a, a)
    max_c = conv.max()
    # compute w = softmax(beta * (conv - max_c)) for numerical stability.
    # effectively w sum 1

"""
Propose a non‑negative step‑function that minimizes the evaluation
objective.  The algorithm consists of:

1. Deterministic seed families (uniform, Catalan, geometric, sinusoid,
   Beta‑shaped) for several lengths.
2. Massive random sampling (Dirichlet and Beta‑shaped) with many
   concentration parameters.
3. Two gradient‑based refinements:
   * Exponentiated‑gradient using *all* active convolution entries
     (not just a single maximum).
   * Adam on a *soft‑max* smoothed version of the max‑convolution
     (continuation on the smoothing parameter β).
4. A cheap random‑swap hill‑climber for final polishing.

All iterates are projected onto the probability simplex (non‑negative,
unit sum).  The routine respects the given time budget and returns the
best list found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for the sizes we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Objective that ``evaluate_sequence`` would return when ``seq`` is already
    normalised to sum = 1:  2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Coefficients proportional to the Catalan numbers
    binom(2i,i)/4**i (coefficients of (1‑z)^‑½).  Normalised to sum = 1.
    """
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _geometric_shape(n: int, r: float = 0.999) -> np.ndarray:
    """Truncated geometric distribution a_i ∝ r**i."""
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


def _sinusoid_shape(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n), which is positive on (0,n)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑distribution shape:
          a_i ∝ (i+1)^{α‑1} · (n‑i)^{β‑1}
    """
    i = np.arange(n, dtype=np.float64)
    # add a tiny epsilon to avoid zero‑to‑zero powers when α or β < 1
    eps = 1e-12
    a = np.power(i + 1 + eps, alpha - 1.0) * np.power(n - i + eps, beta - 1.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


# ----------------------------------------------------------------------
# Random generation utilities
# ----------------------------------------------------------------------


def _dirichlet(alpha: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector, normalised to sum = 1."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


def _random_beta_shape(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Randomly sample α,β from a wide range and return the corresponding
    Beta‑shaped probability vector.
    """
    # α,β ~ LogUniform(0.2, 5) (more weight on values close to 1)
    log_a = rng.uniform(np.log(0.2), np.log(5.0))
    log_b = rng.uniform(np.log(0.2), np.log(5.0))
    alpha = np.exp(log_a)
    beta = np.exp(log_b)
    return _beta_shape(n, alpha, beta)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by all refinements
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive; return uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Refinement – exponentiated gradient with full active set
# ----------------------------------------------------------------------


def _refine_expgradient_full(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using the negative entropy (exponentiated gradient) but
    using *all* convolution indices that are within a relative tolerance
    of the current maximum.  Returns the best vector seen before the
    deadline.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30
    it = 0
    # relative tolerance for “active set”
    rel_tol = 1e-6
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ExpGradFull] it={it:7d} val={cur_val:.6f}")

        # active indices (within rel_tol of the maximum)
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient accumulated over all active entries
        g = np.zeros_like(a)
        for idx in active:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = idx - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        # average it to keep magnitude reasonable
        if active.size > 1:
            g /= active.size

        # exponentiated‑gradient update (multiplicative)
        lr = 0.6 / np.sqrt(it)  # diminishing step size
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)  # for numerical stability
        a = np.exp(log_a)
        a /= a.sum()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Refinement – soft‑max Adam (continuation on β)
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max weights w_k = exp(β·(conv_k - max(conv))) / Σ.
    """
    max_c = conv.max()
    # subtract max to avoid overflow
    shifted = beta * (conv - max_c)
    # clip to avoid NaNs for extreme β
    shifted = np.clip(shifted, -700, 700)
    exp_term = np.exp(shifted)
    w = exp_term / exp_term.sum()
    return w


def _grad_softmax_max(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed max defined by soft‑max weights.
    Returns ∇_a  (soft‑max‑max) .  The gradient formula is
        g_i = 2 * Σ_k w_k · a_{k-i}
    which can be computed via a convolution with a reversed copy of a.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    w = _softmax_weights(conv, beta)
    a_rev = a[::-1]
    conv_wr = _conv_fft(w, a_rev)                # length = (2n‑1) + n‑1 = 3n‑2
    g = 2.0 * conv_wr[n - 1 : n - 1 + n]         # extract the relevant slice
    return g


def _refine_softmax_adam(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max objective using a continuation on
    the soft‑max temperature β.  Returns the best vector seen before the
    deadline.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # continuation schedule for β
    beta_vals = np.geomspace(1.0, 200.0, num=8)   # 1 → 200 geometrically
    beta_idx = 0
    beta = beta_vals[beta_idx]

    # allocate equal time for each β‑stage
    stage_deadlines = []
    remaining = deadline - time.time()
    per_stage = remaining / len(beta_vals)
    for _ in beta_vals:
        stage_deadlines.append(time.time() + per_stage)

    it = 0
    rng = np.random.default_rng()

    for beta, stage_deadline in zip(beta_vals, stage_deadlines):
        while time.time() < stage_deadline:
            it += 1
            # compute current convolution and gradient w.r.t. smoothed max
            conv = _conv_fft(a, a)
            cur_max = np.max(conv)
            cur_val = 2.0 * n * cur_max
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                if it % 500 == 0:
                    print(f"[SoftmaxAdam] it={it:7d} β={beta:.1f} val={cur_val:.6f}")

            # gradient of smoothed max
            g = _grad_softmax_max(a, beta)

            # Adam moment updates
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # base learning rate – diminishes with iteration count
            lr = 0.08 / np.sqrt(it)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # project back onto the simplex
            a = _proj_simplex(a_new)

            # occasional jitter to escape plateaus
            if it % 4000 == 0:
                noise = rng.normal(scale=1e-5, size=n)
                a = _proj_simplex(a + noise)

        # move to next β automatically (stage_deadlines already set)
    return best_seq, best_val

# ----------------------------------------------------------------------


def _refine_random_swaps(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: repeatedly move a tiny amount of mass from
    one index to another (preserving the simplex).  Accept only
    improving moves (deterministic hill‑climbing).  Returns best found.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    rng = np.random.default_rng()
    # scale of the random transfer – start relatively large and shrink
    delta_scale = 0.02

    while time.time() < deadline:
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # propose moving a fraction of mass from i to j
        delta = delta_scale * rng.random() * a[i]   # never exceed a[i]
        if delta <= 0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        # project is unnecessary because we keep sum and non‑negativity by construction
        val = _objective(a_candidate)
        if val < best_val:
            best_val = val
            best_seq = a_candidate.copy()
            a = a_candidate
            # shrink step size a bit after a success
            delta_scale = max(1e-4, delta_scale * 0.99)
            print(f"[SwapHill] val={val:.6f}  delta_scale={delta_scale:.5f}")
        else:
            # occasionally shrink even when no improvement (to explore finer moves)
            delta_scale *= 0.9999
    return best_seq, best_val
