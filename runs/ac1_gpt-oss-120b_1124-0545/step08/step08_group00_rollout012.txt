"""
Step‑function optimizer for ``evaluate_sequence``

The algorithm builds a diverse pool of ℓ₁‑normalised seeds (uniform,
geometric, sine‑window, Tukey, linear‑decreasing, random Dirichlet …)
and keeps the most promising ones.  Each seed is refined by two
continuous phases:

1. **Smoothed‑max gradient descent** – we maximise the soft‑max of the
   convolution entries (log‑sum‑exp) with a temperature that is slowly
   increased (β → large).  The variables stay on the probability simplex
   via Euclidean projection after every update.

2. **Exact sub‑gradient descent** – the true max‑convolution is tackled
   with an additive step and a line‑search over a handful of step‑size
   candidates, again projecting onto the simplex.

A final short simulated‑annealing polish perturbs a random subset of
coordinates.  The whole search respects the supplied time budget
(≈1000 s) and returns the best sequence found.  The best sequence is also
stored in the global name ``height_sequence_1`` for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _max_conv(seq: np.ndarray) -> float:
    """Maximum entry of the (full) self‑convolution."""
    return np.max(_conv_fft(seq, seq))


def _surrogate(seq: np.ndarray, n: int) -> float:
    """Surrogate objective: 2·n·max(conv(seq,seq)) (assumes Σseq = 1)."""
    return 2.0 * n * _max_conv(seq)


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def _linear_decreasing(n: int) -> np.ndarray:
    """a_i ∝ (n-i)   (monotonically decreasing, zero at the end)."""
    idx = np.arange(n, dtype=np.float64)
    a = n - idx
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------


def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_tukey_window(n, alpha=0.3))
    seeds.append(_tukey_window(n, alpha=0.5))
    seeds.append(_tukey_window(n, alpha=0.7))
    seeds.append(_geom(n, r=0.99))
    seeds.append(_geom(n, r=0.995))
    seeds.append(_linear_decreasing(n))

    # random Dirichlet draws (different concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        prev = np.asarray(previous, dtype=np.float64)
        if prev.sum() > 0:
            seeds.append(prev / prev.sum())

    return seeds


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _smooth_max_descent(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    beta0: float = 0.5,
    beta_mul: float = 1.001,
    beta_max: float = 5_000.0,
    lr0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a log‑sum‑exp (soft‑max) surrogate.
    Variables stay on the simplex via Euclidean projection.
    """
    a = start_vec.copy()
    best = a.copy()
    best_val = _surrogate(a, n)

    beta = beta0
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max weights over the convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k   (see derivation in the description)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length = 3·n‑2
        grad = 2.0 * full[n - 1:n - 1 + n]        # central n entries

        # step size (diminishing) and Euclidean projection onto simplex
        lr = lr0 / np.sqrt(it)
        a = _proj_simplex(a - lr * grad)

        cur_val = _surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[Smooth] it={it:6d} best={best_val:.6f}")

        # raise temperature slowly – the surrogate becomes a tighter
        # approximation of the true max.
        beta = min(beta * beta_mul, beta_max)

    return best, best_val


def _subgradient_descent(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Uses a tiny line‑search over a few step‑size factors and projects onto
    the probability simplex after each update.
    """
    a = start_vec.copy()
    best = a.copy()
    best_val = _surrogate(a, n)

    factors = (1.0, 0.5, 0.2, 0.1, 0.05)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active max‑entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # tiny line‑search – try a few step‑sizes
        improved = False
        for f in factors:
            lr = f * 0.5 / np.sqrt(it)
            cand = _proj_simplex(a - lr * g)
            cand_val = _surrogate(cand, n)
            if cand_val < cur_val:
                a = cand
                improved = True
                break
        if not improved:
            # fall back to a very small step
            lr = 0.01 / np.sqrt(it)
            a = _proj_simplex(a - lr * g)

    return best, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    init_temp: float = 0.02,
) -> tuple[np.ndarray, float]:
    """Simple SA that redistributes mass among a random subset of coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate(cur, n)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9997, 1e-12)

        k = rng.integers(2, 12)                # size of the reshuffle block
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _surrogate(new, n)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and returns the best sequence found.  The best sequence is also stored
    in the global name ``height_sequence_1`` for possible reuse.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 3.0       # safety margin

    # ------------------------------------------------------------------
    # Length selection – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Gather a generous variety of seeds.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    all_seeds = _collect_seeds(n, rng, previous)

    print(f"\n--- Evaluating {len(all_seeds)} seeds (n={n}) ---")
    evaluated = [(vec, _surrogate(vec, n)) for vec in all_seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # initialise global best
    global_best = best_seeds[0].copy()
    global_best_val = _surrogate(global_best, n)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – smoothed‑max descent (Adam‑like updates)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Smoothed‑max optimisation #{idx+1}/{len(best_seeds)} ---")
        seq, val = _smooth_max_descent(
            start_vec=seed_vec,
            n=n,
            deadline=deadline,
            max_iters=200_000,
        )
        if val < global_best_val:
            global_best, global_best_val = seq, val
            print(f"[Overall] new global best (smooth) = {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exact sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exact sub‑gradient descent phase ---")
        seq, val = _subgradient_descent(
            start_vec=global_best,
            n=n,
            deadline=deadline,
            max_iters=300_000,
        )
        if val < global_best_val:
            global_best, global_best_val = seq, val
            print(f"[Overall] new global best (sub‑grad) = {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final stochastic polishing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq, val = _simulated_annealing(
            start_seq=global_best,
            n=n,
            deadline=deadline,
            max_iters=500_000,
        )
        if val < global_best_val:
            global_best, global_best_val = seq, val
            print(f"[Overall] final global best = {val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check using the provided evaluator.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best.tolist())
        print(f"\n>>> evaluate_sequence returned {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best.tolist()
    return global_best.tolist()
