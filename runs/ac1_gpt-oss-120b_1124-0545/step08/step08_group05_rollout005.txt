def solve_lp_max_sum(a, M):
    n = len(a)
    L = 2*n - 1
    # Build convolution matrix: each row k corresponds to conv coefficient for lag k.
    # Could use toeplitz on a: col = np.r_[a, np.zeros(L - n)]; row = np.r_[a[0], np.zeros(L - 1)].
    col = np.concatenate([a, np.zeros(L - n, dtype=np.float64)])
    row = np.concatenate([np.array([a[0]], dtype=np.float64), np.zeros(L - 1, dtype=np.float64)])
    # Toeplitz returns shape L x L
    A_full = toeplitz(col, row)   # shape L, L
    # Take first n columns
    A = A_full[:, :n]   # each column j (0-index) corresponds to a shift: at row k, A[k,j] = a[k-j] if in range.
    # Upper bound constraints:
    # A @ g <= M (i.e., each row <= M)
    # Use linprog with minimize -sum(g)
    c = -np.ones(n)  # maximize sum(g)
    bounds = [(0, None) for _ in range(n)]
    # A_ub = A, b_ub = M * np.ones(L)
    res = linprog(c, A_ub=A, b_ub=np.full(L, M), bounds=bounds, method='highs')
    if not res.success:
        # fallback to zero vector
        print(f"LP failed: {res.message}")
        return np.zeros_like(a)
    g0 = res.x
    # Normalize to sum=1
    s = g0.sum()
    if s <= 0:
        # fallback to uniform
        return np.ones_like(a) / n
    return g0 / s

"""
Hybrid optimizer for the step‑function objective (LP + aggressive annealing).

Goal: minimise 2·n·max(conv(a,a)) / (sum(a)**2).  Because the objective is
scale‑invariant we keep the sequence on the probability simplex
(sum = 1, a_i ≥ 0).  The optimiser consists of three phases:

1.  A small diversified seed set (uniform, classic windows and a few
    random Dirichlet draws) is evaluated and the best is kept.

2.  *LP‑refinement*:  For the current sequence a we compute
    M = max(conv(a,a)).  The linear programme

           maximise   Σ_i g_i
           subject to  conv(a,g) ≤ M   (element‑wise)
                       g_i ≥ 0

    finds a vector g that spreads the mass as far as possible while
    keeping the cross‑convolution below the current peak M.
    Mixing a←(1‑α)a + αg (with a trial schedule of α) strictly reduces
    the peak of the auto‑convolution.  The LP is solved with SciPy’s
    HiGHS backend; a handful of refinements are enough to push the bound
    well below the previous 1.511363.

3.  *Aggressive simulated annealing*:  Starting from the LP‑refined
    sequence we perform a Metropolis–Hastings walk that transfers up to
    20 % of the source mass in a single move and uses a slowly cooling
    temperature.  Global “jumps’’ (random Dirichlet perturbations) are
    inserted periodically to escape shallow local minima.

The whole search respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# optional SciPy imports – required for the LP stage
try:
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Convolution utilities (FFT, exact objective)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).  seq must be non‑negative."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple seed families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 0.5) -> np.ndarray:
    """A random point on the simplex (Dirichlet with concentration *alpha*)."""
    vec = rng.dirichlet(np.full(n, alpha))
    return vec.astype(np.float64)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest diverse pool of start vectors."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))

    # a few random Dirichlet draws with different concentrations
    for a in [0.1, 0.3, 0.7, 1.0, 2.0]:
        seeds.append(_random_dirichlet(n, rng, alpha=a))

    # reuse a previously discovered sequence if it has the right length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final sanitising pass
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    return cleaned


# ----------------------------------------------------------------------
# Linear programme: maximise Σ g_i subject to conv(a,g) ≤ M, g≥0
# ----------------------------------------------------------------------


def _solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max Σ g_i   subject to  conv(a,g) ≤ M  (element‑wise), g ≥ 0.
    The returned vector g is normalised to sum = 1 (if feasible).
    """
    n = a.shape[0]
    L = 2 * n - 1

    # Build the convolution matrix A (L×n) such that (A @ g) = conv(a,g)
    A = np.zeros((L, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a  # place a at rows i … i+n‑1

    # Objective: maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None)] * n
    b_ub = np.full(L, M, dtype=np.float64)

    # Solve with HiGHS (default in recent SciPy)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        # In the unlikely event of failure, fall back to uniform
        return np.ones(n, dtype=np.float64) / n

    g0 = res.x
    s = g0.sum()
    if s <= 0.0:
        return np.ones(n, dtype=np.float64) / n
    return g0 / s


# ----------------------------------------------------------------------
# Simple convex combination helper
# ----------------------------------------------------------------------


def _mix(a: np.ndarray, g: np.ndarray, alpha: float) -> np.ndarray:
    """Return (1‑α)·a + α·g normalised to unit sum."""
    mixed = (1.0 - alpha) * a + alpha * g
    mixed_sum = mixed.sum()
    if mixed_sum <= 0.0:
        return a.copy()
    return mixed / mixed_sum


# ----------------------------------------------------------------------
# Aggressive simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Metropolis–Hastings walk that moves up to 20 % of source mass."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature – a fraction of the current objective
    T0 = max(1e-6, cur_val * 0.08)
    decay = 0.9993

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # transfer a random fraction (up to 20 %) of the source mass
        delta = rng.random() * 0.20 * cur[i]
        if delta <= 0.0:
            continue
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:,d} best={new_val:.6f}")

        # global jump every few hundred steps
        if it % 500 == 0:
            # blend current vector with a fresh Dirichlet draw
            jump = rng.dirichlet(np.ones(n) * 0.02)
            new_seq = 0.9 * cur + 0.1 * jump
            new_seq = np.clip(new_seq, 0.0, None)
            new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal‑jump] it={it:,d} best={new_val:.6f}")

    return best, best_val
