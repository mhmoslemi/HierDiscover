def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for the sizes we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Objective for a vector already normalised to sum‑=‑1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex Σx=1, x≥0."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a /= a.sum()
    return a


def _geometric(n: int, r: float) -> np.ndarray:
    """a_i ∝ r^i."""
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    a /= a.sum()
    return a


def _beta_shape(n: int, α: float, β: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1}."""
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    # avoid exact 0/1 when α or β < 1
    eps = 1e-12
    t = np.clip(t, eps, 1.0 - eps)
    a = (t ** (α - 1.0)) * ((1.0 - t) ** (β - 1.0))
    a /= a.sum()
    return a


def _dirichlet(alpha: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a symmetric Dirichlet(α,…,α) and normalise to sum = 1."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    total = x.sum()
    if total == 0.0:
        return _uniform(n)
    return x / total


# ----------------------------------------------------------------------
# Gradient of the soft‑max‑weighted convolution
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute ∇ (log‑sum‑exp_{k}(β·conv_k))  where conv = a * a.
    The gradient is obtained analytically using two FFT convolutions.
    """
    n = len(a)
    conv = _conv_fft(a, a)                     # length 2n‑1
    max_val = conv.max()
    # soft‑max weights (max entry gets weight 1)
    w = np.exp(beta * (conv - max_val))        # shape (2n‑1,)

    # g_i = Σ_j w_{i+j} a_j   -> can be written as a slice of conv(w, a_rev)
    a_rev = a[::-1]
    conv_wr = _conv_fft(w, a_rev)              # length 3n‑2
    # slice i+n‑1 gives Σ_j w_{i+j} a_j for i=0…n‑1
    g = conv_wr[n - 1 : n - 1 + n]

    # subtract the diagonal term (k = 2i) from the gradient
    diag = np.zeros_like(a)
    idx = np.arange(n)
    mask = 2 * idx < w.shape[0]
    diag[mask] = w[2 * idx[mask]] * a[mask]

    grad = 2.0 * g - diag
    return grad


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def _adam_softmax(start_seq: np.ndarray,
                  deadline: float,
                  beta_start: float = 1.0,
                  beta_end: float = 60.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser applied to the soft‑max‑approximation of the
    ``max(conv)`` objective.  The soft‑max temperature β is increased
    linearly from *beta_start* to *beta_end*.
    """
    a = start_seq.copy()
    n = len(a)

    best_seq = a.copy()
    best_val = _objective(a)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1
        # linearly schedule β
        beta = beta_start + (beta_end - beta_start) * min(1.0,
                                                          (t / 2000.0))

        grad = _softmax_gradient(a, beta)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # diminishing step‑size
        lr = 0.15 / np.sqrt(t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project back onto simplex (non‑negative, Σ=1)
        a = _proj_simplex(a_new)

        # occasional tiny random jitter to escape plateaus
        if t % 4000 == 0:
            a += rng.normal(scale=1e-5, size=n)
            a = _proj_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam‑soft] it={t:6d} β={beta:.1f} val={cur_val:.6f}")

    return best_seq, best_val


def _exp_gradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent with negative entropy) using
    the exact sub‑gradient of the active maximum entry.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ExpGrad] it={it:6d} val={cur_val:.6f}")

        # select a random index among those within a tiny tolerance
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]
        k = int(rng.choice(active))

        # sub‑gradient of this entry (ordered‑pair version)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx_i
        # products counted twice for i≠rev, once for i=rev – using the
        # same 2·a[rev] rule (the diagonal case is harmlessly slighty
        # over‑estimated, which is fine for a sub‑gradient).
        g[idx_i] = 2.0 * a[rev]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The routine respects the time budget and
    returns the best solution as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final sanity check
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a problem size (larger n gives a finer bound but costs more)
    # ------------------------------------------------------------------
    n = 8000                      # works well on the allotted time

    # ------------------------------------------------------------------
    # Stage 1 – generate a diverse collection of seeds
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # analytic families
    families = [
        (lambda n: _uniform(n), "uniform"),
        (lambda n: _power_law(n, 0.5), "power‑0.5"),
        (lambda n: _power_law(n, 1.0), "power‑1.0"),
        (lambda n: _power_law(n, 1.5), "power‑1.5"),
        (lambda n: _geometric(n, 0.9999), "geom‑0.9999"),
        (lambda n: _geometric(n, 0.9995), "geom‑0.9995"),
        (lambda n: _beta_shape(n, 0.6, 0.6), "beta‑0.6‑0.6"),
        (lambda n: _beta_shape(n, 1.0, 2.0), "beta‑1‑2"),
        (lambda n: _beta_shape(n, 2.0, 1.0), "beta‑2‑1"),
    ]

    for maker, name in families:
        try:
            seq = maker(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] type={name:<12} val={val:.6f}")
        except Exception:
            pass

    # a few random Dirichlet draws (different concentration parameters)
    alphas = [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]
    while time.time() < deadline * 0.30:     # ~30 % of total budget
        alpha = float(rng.choice(alphas))
        seq = _dirichlet(alpha, n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] α={alpha:.2f} val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – soft‑max Adam optimisation (the heavy work)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = time.time() + (deadline - time.time()) * 0.55
        seq_adam, val_adam = _adam_softmax(best_seq, adam_deadline,
                                           beta_start=1.0,
                                           beta_end=80.0)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam
            print(f"[Overall‑Adam] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – mirror‑descent (exponentiated‑gradient) fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = deadline
        seq_exp, val_exp = _exp_gradient(best_seq, exp_deadline)
        if val_exp < best_val:
            best_val = val_exp
            best_seq = seq_exp
            print(f"[Overall‑Exp] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – the grader will also call evaluate_sequence
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print("\n=== Finished ===")
    print(f"Best objective ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
