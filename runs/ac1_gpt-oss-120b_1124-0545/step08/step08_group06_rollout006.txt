"""
Propose a step‑function (list of non‑negative heights) that minimizes
the supplied ``evaluate_sequence`` objective.

Algorithm
---------
1. **Analytic seeds** – a few well‑known shapes (uniform, Catalan,
   geometric) are evaluated for lengths 4000 and 8000.

2. **Smooth‑Adam optimizer** – the vector is parameterised as a soft‑max
   of free variables `x`.  The true objective
   ``2·n·max(conv(a,a))`` (with ``a = softmax(x)``) is approximated by a
   smooth maximum `M_γ = logsumexp(γ·conv) / γ`.  The gradient of
   `M_γ` w.r.t. `x` can be written in closed form using one correlation
   (`w ⋆ a`) where `w` are the soft‑max weights of the convolution
   entries.  Adam with an adaptive learning‑rate and a linearly growing
   smoothing parameter `γ` is run for the remaining budget; the best
   point (according to the exact non‑smooth objective) is retained.

3. **Random Dirichlet restarts** – the smooth optimizer is launched from
   a few random Dirichlet draws to avoid local minima.

The routine respects the global time budget (default 1000 s) and
returns a plain Python list ready for ``evaluate_sequence``.  Only
standard‑library modules and NumPy are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for the sizes we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective that ``evaluate_sequence`` would return when ``a`` is
    already normalised to sum = 1.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Coefficients proportional to (2i choose i)/4**i (the series of (1‑z)^‑½),
    truncated to length n and normalised to sum = 1.
    """
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2.0 * i) * (2.0 * i - 1.0)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _geometric_shape(n: int, r: float = 0.999) -> np.ndarray:
    """
    Truncated geometric distribution a_i ∝ r**i, normalised to sum = 1.
    """
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


# ----------------------------------------------------------------------
# Random Dirichlet sampling (α may vary)
# ----------------------------------------------------------------------


def _dirichlet(alpha: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector of length n and normalise to sum = 1."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Correlation (valid) – needed for the exact gradient of the smooth
# max‑approximation.
# ----------------------------------------------------------------------


def _corr_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid correlation w ⋆ a (length = len(w) - len(a) + 1).
    Implemented via FFT for speed.
    """
    rev_a = a[::-1]                     # reverse a
    conv = _conv_fft(w, rev_a)          # full convolution
    n = len(a)
    start = n - 1
    end = start + n
    return conv[start:end]              # shape (n,)


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (soft‑max parametrisation)
# ----------------------------------------------------------------------


def _smooth_adam(start_x: np.ndarray,
                 sub_deadline: float,
                 rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the exact objective using a smooth max
    (log‑sum‑exp) surrogate.  Returns the best *probability* vector seen
    (i.e. a_i ≥ 0, Σa_i = 1) and its exact objective value.
    """
    x = start_x.copy()
    n = len(x)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # tracking best solution
    best_a = None
    best_val = float("inf")

    # schedule for the smoothing parameter γ
    gamma_start = 2.0
    gamma_end = 30.0
    # we will linearly increase γ from start to end during the run
    total_iters_est = max(1, int(5e5))   # a generous cap for the schedule

    it = 0
    while time.time() < sub_deadline:
        it += 1
        # ----- 1) soft‑max to obtain a probability vector -----
        mx = np.max(x)
        exp_x = np.exp(x - mx)           # stabilised
        a = exp_x / exp_x.sum()

        # ----- 2) convolution a*a (full) -----
        conv = _conv_fft(a, a)          # length 2n‑1

        # ----- 3) smooth max via log‑sum‑exp -----
        gamma = gamma_start + (gamma_end - gamma_start) * min(1.0, it / total_iters_est)
        q = gamma * conv
        q_max = np.max(q)
        w = np.exp(q - q_max)           # unnormalised soft‑max weights
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum                    # now w_i ≥ 0, Σw_i = 1

        # ----- 4) gradient of the *exact* objective w.r.t. a -----
        #   ∂Obj/∂a = 4 n · ( w ⋆ a )
        g_a = 4.0 * n * _corr_valid(w, a)    # shape (n,)

        # ----- 5) chain rule through the soft‑max -----
        #   ∂Obj/∂x = a * (g_a - (a·g_a))
        scalar = np.dot(g_a, a)               # a·g_a
        grad = a * (g_a - scalar)             # shape (n,)

        # ----- 6) Adam update on the free parameters x -----
        lr = 0.1 / np.sqrt(it)                # diminishing base lr
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny perturbation to escape shallow plateaus
        if it % 5000 == 0:
            x += rng.normal(scale=1e-4, size=n)

        # ----- 7) evaluate the *exact* (non‑smooth) objective occasionally -----
        if it % 1000 == 0 or time.time() > sub_deadline - 0.1:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[SmoothAdam] it={it:6d} val={cur_val:.6f} γ={gamma:.2f}")

        # early stop if we have already hit the target bound
        if best_val < 1.503:
            break

    if best_a is None:          # fallback – return last iterate
        best_a = np.exp(x - np.max(x))
        best_a /= best_a.sum()
        best_val = _objective(best_a)

    return best_a, best_val
