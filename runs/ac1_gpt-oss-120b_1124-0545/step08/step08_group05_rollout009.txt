Goal: minimise  2·n·max(conv(a,a)) / (sum(a)**2)  over non‑negative vectors a.

1.  Build a very diverse pool of seed windows (uniform, geometric,
    power‑law, classic spectral windows, Chebyshev and DPSS windows,

2.  Keep the best few seeds (by the exact objective) and, for each of them,

    reduces the exact max‑convolution further.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The input is assumed to be a non‑negative vector.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window, β ≥ 0."""
    if not SCIPY_AVAILABLE or beta <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    nminus1 = n - 1
    x = (2.0 * i / nminus1) - 1.0
    a = special.i0(beta * np.sqrt(1.0 - x * x)) / special.i0(beta)
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1), discretised on mid‑points.
    α ∈ (0,1] gives edge‑heavy shapes; α=1 is uniform.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_general(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    General discrete Beta(α,β) shape on {1,…,n}.
    α,β > 0 (values <1 concentrate mass near the edges).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()

def _chebwin(n: int, attenuation: float = 100.0) -> np.ndarray:
    """Chebyshev window – symmetric, controllable side‑lobe attenuation."""
    if not SCIPY_AVAILABLE:
        return _uniform(n)
    # SciPy's chebwin signature: chebwin(M, at=attenuation)
    w = signal.chebwin(n, at=attenuation)
    w = np.clip(w, 0.0, None)
    return w / w.sum()


def _dpss_window(n: int, NW: float = 3.5) -> np.ndarray:
    """Discrete Prolate Spheroidal Sequence (Slepian) window."""
    if not SCIPY_AVAILABLE:
        return _uniform(n)
    # Returns shape (Kmax, n). We ask for the first DPSS.
    w = signal.windows.dpss(n, NW, Kmax=1)[0]
    w = np.clip(w, 0.0, None)
    return w / w.sum()

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 400000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact non‑smooth objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over active constraints)
        g = np.zeros_like(a)
        if active.size > 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient step with a 1/√t schedule
        lr = 0.8 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # keep numbers bounded
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val

def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Metropolis–Hastings walk with larger mass‑transfer moves.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # start temperature a fraction of the current value
    T0 = max(1e-6, cur_val * 0.06)
    decay = 0.9992

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct positions
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # transfer a random fraction (up to 5 %) of the source mass
        delta = rng.random() * 0.05 * cur[i]
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # occasional global perturbation (every 800 steps)
        if it % 800 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Global variables used by the LBFGS surrogate objective
# ----------------------------------------------------------------------
_current_n: int = 0
_current_alpha: float = 1.0


def _smooth_obj_grad(x: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smooth surrogate (log‑sum‑exp) of max(conv) and its gradient.
    Operates on the log‑softmax parametrisation of the simplex.
    """
    n = _current_n
    alpha = _current_alpha

    # soft‑max mapping (ensures non‑negativity and sum‑1)
    max_x = np.max(x)
    exp_x = np.exp(x - max_x)
    a = exp_x / exp_x.sum()

    # full auto‑convolution
    conv = _conv_fft(a, a)
    conv_max = conv.max()

    # soft‑max weights over convolution entries
    exp_term = np.exp(alpha * (conv - conv_max))
    w = exp_term / exp_term.sum()

    # gradient of smooth max w.r.t. a  (2 * rev_a * w)  --> multiplied by 2n later
    rev_a = a[::-1]
    conv_rev_w = _conv_fft(rev_a, w)
    grad_f_a = 4.0 * n * conv_rev_w[n - 1:n - 1 + n]   # 2*n * 2 = 4*n

    # transform gradient to the log‑softmax coordinates
    dot = np.dot(grad_f_a, a)
    grad_x = a * (grad_f_a - dot)

    # smooth approximation of the max
    smooth_max = conv_max + np.log(exp_term.sum()) / alpha
    f_val = 2.0 * n * smooth_max

    return float(f_val), grad_x.astype(np.float64)

# LBFGS‑based continuation optimisation
# ----------------------------------------------------------------------


def _lbfgs_continuation(seed_seq: np.ndarray,
                        n: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimise the smooth surrogate with a sequence of increasing smoothness
    parameters α using SciPy's L‑BFGS‑B.  Returns the best exact objective
    encountered during the continuation.
    """
    eps = 1e-30
    x = np.log(seed_seq + eps)          # initialise log‑softmax variables
    best_seq = seed_seq.copy()
    best_val = _objective(best_seq)

    alphas = [30.0, 120.0, 300.0, 600.0]

    for alpha in alphas:
        if time.time() > deadline - 5.0:
            break
        global _current_n, _current_alpha
        _current_n = n
        _current_alpha = alpha

        # Use SciPy L‑BFGS‑B; if SciPy is missing we break out.
        if not SCIPY_AVAILABLE:
            break
        try:
            res = optimize.minimize(_smooth_obj_grad,
                                    x,
                                    method='L-BFGS-B',
                                    jac=True,
                                    options={'maxiter': 300, 'disp': False})
        except Exception as e:          # pragma: no cover
            print("[LBFGS] optimisation failed:", e)
            break

        x = res.x
        # recover a from logits
        a = np.exp(x - np.max(x))
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[LBFGS] α={alpha:.0f} improved to {val:.6f}")

    return best_seq, best_val

            previous = None

    # ------------------------------------------------------------------
    # 0) Evaluate the previous vector (if any)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    if previous is not None and previous.shape == (n,):
        val = _objective(previous)
        best_seq = previous.copy()
        best_val = val
        print(f"[prev] n={n} val={val:.6f}")

    # ------------------------------------------------------------------
    # 1) Build a rich pool of seeds and keep the top few
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous)
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_seeds = [s for s, _ in seed_vals[:8]]

    if seed_vals[0][1] < best_val:
        best_val = seed_vals[0][1]
        best_seq = seed_vals[0][0].copy()
        print(f"[seed] new best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Continuation with LBFGS on the smooth surrogate (primary phase)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20:
        for idx, s in enumerate(top_seeds):
            if time.time() > deadline - 5:
                break
            seq, val = _lbfgs_continuation(s, n, deadline)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[overall LBFGS] seed#{idx} best={val:.6f}")

    # ------------------------------------------------------------------
    # 3) Short sub‑gradient mirror‑descent refinement (smooth → exact)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        seq, val = _subgradient_refine(best_seq, n, deadline,
                                       max_iters=300000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[overall SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # 4) Final simulated‑annealing walk (uses any remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(best_seq, n, deadline,
                                    max_iters=None)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[overall Anneal] best={val:.6f}")

    # ------------------------------------------------------------------
    # 5) Fallback (very unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    # store for possible subsequent runs
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
