def _smooth_optimize(start_seq, n, deadline):
    # Convert start_seq to logits
    eps = 1e-30
    logits = np.log(start_seq + eps)
    # Adam parameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    t = 0
    best_seq = start_seq.copy()
    best_val = _objective(best_seq)
    lambda_min = 5.0
    lambda_max = 200.0
    # We'll increase lambda linearly over iterations
    while time.time() < deadline:
        t += 1
        # linearly increase lambda based on progress fraction
        frac = (t * 1.0) / 2000000.0  # assume up to many iterations
        lam = lambda_min + (lambda_max - lambda_min) * min(frac, 1.0)
        # compute probabilities a
        max_log = np.max(logits)
        exp_logits = np.exp(logits - max_log)
        a = exp_logits / exp_logits.sum()
        # conv
        conv = _conv_fft(a, a)
        # compute softmax weights
        # shift for numerical stability
        shift = conv - conv.max()
        exp_shift = np.exp(lam * shift)
        w = exp_shift / exp_shift.sum()
        # gradient w.r.t a
        grad_a = 2.0 * lam * _conv_fft(w, a[::-1])[n-1:n-1+n]
        # gradient w.r.t logits
        dot = np.dot(grad_a, a)
        grad_logits = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_logits
        v = beta2 * v + (1 - beta2) * (grad_logits ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # evaluate objective
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[smooth] iter={t} lam={lam:.2f} best={cur_val:.6f}")

        # stop after many iterations if time limited
        if t % 20000 == 0 and time.time() > deadline:
            break

    return best_seq, best_val

"""
Hybrid optimiser – fine‐grid search + exponentiated‑gradient,
smooth Adam descent (soft‑max approximation of the max) and
final simulated‑annealing.  The routine works entirely on the
probability simplex (∑a=1) so the clipping performed by
``evaluate_sequence`` never alters the result.  It keeps the best
vector found in the global name ``height_sequence_1`` for possible
reuse by later runs.

The main ideas
--------------
* Very fine 1‑parameter scans of the analytic families that are known
  to give good‑looking step‑functions (geometric, symmetric geometric,
  power‑law, Beta‑window, Gaussian, Tukey).  The scans use thousands
  of points – cheap because the objective is evaluated with an FFT.
* A short stochastic phase (random Dirichlet / soft‑max draws) to add
  diversity.
* Exponentiated‑gradient mirror descent (the same sub‑gradient method
  used previously) to polish the current best seed.
* A smooth Adam optimiser where the non‑smooth max is replaced by a
  log‑sum‑exp with an annealing λ‑schedule; this provides a
  well‑behaved gradient that can escape the plateaus of the plain
  sub‑gradient method.
* A final Metropolis‑Hastings annealer with several move types and
  an adaptive temperature to finish the optimisation.

All heavy work is timed – the function stops well before the supplied
budget (default 1000 s) and returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution (full mode)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Probability simplex projection (used by the annealer)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families (all return a *probability* vector)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1), discretised on mid‑points.
    α ∈ (0,1] gives edge‑heavy shapes; α=1 is uniform.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `alpha`."""
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Sub‑gradient mirror‑descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 600000) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the exact objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over active constraints)
        g = np.zeros_like(a)
        if len(active) > 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / len(active)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth Adam optimiser (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------
def _smooth_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed objective
        L(λ) = logsumexp( λ·conv(a,a) )
    λ is increased linearly from λ_min to λ_max.
    """
    eps = 1e-30
    # initialise logits for the softmax parametrisation
    logits = np.log(start_seq + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam_min = 5.0
    lam_max = 200.0

    best_seq = _project_simplex(np.exp(logits - np.max(logits)))
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # annealing of λ
        lam = lam_min + (lam_max - lam_min) * min(it / max_iters, 1.0)

        # current probability vector
        max_log = np.max(logits)
        a = np.exp(logits - max_log)
        a /= a.sum()

        # convolution a*a
        conv = _conv_fft(a, a)

        # soft‑max weights over the convolution entries
        shift = conv - conv.max()          # shift for numerical stability
        w = np.exp(lam * shift)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t a : 2·λ·(w * a_rev)  (slice the central n entries)
        grad_a = 2.0 * lam * _conv_fft(w, a[::-1])[n - 1:n - 1 + n]

        # gradient w.r.t logits (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasionally evaluate the true objective
        if it % 2000 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Smooth] it={it:6d} lam={lam:.2f} best={cur_val:.6f}")

    # final evaluation
    final_a = np.exp(logits - np.max(logits))
    final_a /= final_a.sum()
    final_val = _objective(final_a)
    if final_val < best_val:
        best_val = final_val
        best_seq = final_a.copy()
        print(f"[Smooth] final best={final_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Metropolis‑Hastings annealer with several move types."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-7, cur_val * 0.12)          # a bit hotter than before
    decay = 0.9996

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        move_type = rng.random()
        if move_type < 0.65:                     # mass transfer (large early)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.random() * 0.30 * cur[i]   # up to 30 % transfer
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        elif move_type < 0.85:                   # swap two entries
            i, j = rng.integers(0, n, size=2)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:                                    # global Gaussian jitter
            noise = rng.normal(scale=0.0020, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0:
                new_seq /= total
            else:
                new_seq = _uniform(n)

        # keep on the simplex
        new_seq = _project_simplex(new_seq)

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:8d} best={new_val:.6f}")

        # occasional stronger random perturbation (every 2500 steps)
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.004, size=n)
            cand = best + jitter
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand)
            cand_val = _objective(cand)
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
                print(f"[Anneal] (jump) it={it:8d} best={cand_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  Returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # a safety margin

    # ------------------------------------------------------------------
    # Length – a modest 4000 gives a good speed/quality trade‑off
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Reuse previous construction if it has the right length
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0) evaluate previous construction (if any)
    # ------------------------------------------------------------------
    if previous is not None and previous.shape == (n,):
        val = _objective(previous)
        best_seq = previous.copy()
        best_val = val
        print(f"[prev] val={val:.6f}")

    # ------------------------------------------------------------------
    # 1) Very fine grid search over a handful of 1‑parameter families
    # ------------------------------------------------------------------
    families = [
        ("geom", _geom, np.linspace(0.5, 0.9999, 2000)),
        ("sym_geom", _sym_geom, np.linspace(0.5, 0.9999, 2000)),
        ("power", _power, np.linspace(0.0, 5.0, 201)),
        ("beta", _beta_window, np.linspace(0.3, 2.5, 221)),
        ("gaussian", _gaussian, np.linspace(0.04, 0.40, 181)),
        ("tukey", lambda n, a: _tukey_window(n, a), np.linspace(0.0, 1.0, 101)),
    ]

    for name, generator, grid in families:
        for param in grid:
            seq = generator(n, param)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[grid] {name:9s} param={param:.6f}  best={val:.6f}")
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2) Random Dirichlet / soft‑max seeds for extra diversity
    # ------------------------------------------------------------------
    for _ in range(30):
        seq = _random_dirichlet(n, 0.2, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[rand Dir] new best={val:.6f}")
        if time.time() > deadline:
            break

    for _ in range(15):
        seq = _random_softmax(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[rand Soft] new best={val:.6f}")
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 3) Sub‑gradient mirror‑descent refinement (exp‑gradient) on the best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        seq, val = _subgradient_refine(best_seq, n, deadline, max_iters=300000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[SubGrad] refined best={val:.6f}")

    # ------------------------------------------------------------------
    # 4) Smooth Adam descent with annealing λ (log‑sum‑exp surrogate)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10.0:
        seq, val = _smooth_optimize(best_seq, n, deadline, max_iters=150000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Smooth] after Adam best={val:.6f}")

    # ------------------------------------------------------------------
    # 5) Simulated‑annealing fine‑tuning (uses remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(best_seq, n, deadline, max_iters=None)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Anneal] final best={val:.6f}")

    # ------------------------------------------------------------------
    # 6) Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
