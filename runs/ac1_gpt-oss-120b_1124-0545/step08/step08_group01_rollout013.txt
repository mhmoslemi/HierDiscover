Implement a function _project_monotonic_decreasing(seq): returns monotonic decreasing version equal to isotonic regression. Implementation: use the convex pooling algorithm: iterate over w = list of values and weights (each weight=1). For decreasing, we want b_i >= b_{i+1}. We'll perform PAVA on reversed sequence (or on negative). Simpler: treat -seq and call _isotonic_increasing, then return -result. We'll implement _isotonic_increasing using PAVA: create lists of level values and counts, then while any adjacent level value violates monotonicity (prev > next), merge them.

Pseudo:

def _isotonic_increasing(y):
    n = len(y)
    solution = y.astype(np.float64)
    # pooling
    level = solution.copy()
    weight = np.ones(n)
    i = 0
    while i < n-1:
        if level[i] <= level[i+1] + 1e-12:
            i += 1
        else:
            # merge i and i+1
            total_weight = weight[i] + weight[i+1]
            avg = (level[i]*weight[i] + level[i+1]*weight[i+1]) / total_weight
            level[i] = avg
            weight[i] = total_weight
            # remove i+1
            level = np.delete(level, i+1)
            weight = np.delete(weight, i+1)
            # step back if possible
            if i > 0:
                i -= 1
    # now expand levels to original size
    # each zone has weight nodes with the same avg
    b = np.empty(n, dtype=np.float64)
    idx = 0
    for val, w in zip(level, weight):
        b[idx:idx+int(w)] = val

    sums = y.copy()

"""
Non‑negative step‑function optimiser (power‑law search).

The algorithm is deliberately simple but mathematically motivated.
For a decreasing power‑law a_i ∝ (i+1)^(-p) the peak of the
self‑convolution can be expressed analytically and the optimal
exponent p lies around 0.58 for n≈4000.  We locate the best p by a
golden‑section search on the cheap surrogate objective
    2·n·max(conv(a,a))
and then polish the result with a short Adam run and an exact
sub‑gradient mirror‑descent.  The whole routine respects the
provided time budget (≈1000 s) and returns the best sequence found
as a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Fast convolution utilities (real FFT, O(n log n))
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Surrogate objective used during the search.
    Assumes ``seq`` is non‑negative and already normalised to sum = 1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# --------------------------------------------------------------
# Power‑law family (a_i ∝ (i+1)^-p)
# --------------------------------------------------------------

def _powerlaw_seq(p: float, n: int) -> np.ndarray:
    """Return a normalised decreasing power‑law vector of length n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    # safety – fall back to uniform if something goes wrong
    if s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


# --------------------------------------------------------------
# Adam optimiser (same surrogate as above)
# --------------------------------------------------------------

def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 60000) -> tuple[np.ndarray | None, float]:
    """Adam optimiser on a log‑sum‑exp surrogate of the max‑conv."""
    eps = 1e-30
    logit = np.log(start_vec + eps)            # logits for the simplex

    m = np.zeros_like(logit)
    v2 = np.zeros_like(logit)

    beta = 0.5            # initial temperature for log‑sum‑exp
    beta_mul = 1.001
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(logit)                     # stays on the probability simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # increase smoothing temperature (capped)
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp on the convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]    # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_logit = a * (grad_a - dot)

        # clip for stability
        max_norm = 10.0
        norm = np.linalg.norm(grad_logit)
        if norm > max_norm:
            grad_logit = grad_logit * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad_logit
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_logit ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        logit = logit - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# --------------------------------------------------------------
# Exact sub‑gradient mirror‑descent (exponentiated‑gradient)
# --------------------------------------------------------------

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 300000) -> tuple[np.ndarray, float]:
    """Refine a sequence with the true max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(max_idx, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

    return best_seq, best_val
