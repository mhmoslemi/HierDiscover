    perm = np.empty(n, dtype=int)
    left = 0

    idx = 0

        idx += 1
        if left != right:
            perm[idx] = right
            idx += 1
        left += 1
        right -= 1
    return perm

   optimisation runs as long as the allocated budget lasts.

4. **Permutation hill‑climb** – the multiset of values of the current
   best vector is kept fixed while we search for a better ordering.
   A large number of random permutations (and a simple swap‑based
   simulated‑annealing hill‑climb) are evaluated; the best ordering
   found replaces the current best.

5. **Exact sub‑gradient mirror descent** – a short KL‑mirror‑descent
   polishing stage (exponentiated‑gradient) finishes the search.

All phases respect the overall time budget (default 1000 s).  The
returned list is normalised (sum = 1) and therefore safe for the
grader.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from math import sqrt, log, pi, exp

# ---------------------------------------------------------------
# Low‑level utilities
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a normalised sequence
    (sum == 1):  2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ---------------------------------------------------------------
# Analytic seed families (all returned normalised to sum = 1)
# ---------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * pi * i / (n - 1))
         + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    """Discrete Gaussian centred in the middle, σ = sigma_factor·n."""
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


# ---------------------------------------------------------------
# Permutation utilities (spread the mass)
# ---------------------------------------------------------------

def _alternating_extreme_permutation(n: int) -> np.ndarray:
    """[0, n‑1, 1, n‑2, …] – interleaves large and small positions."""
    perm = np.empty(n, dtype=int)
    left, right, idx = 0, n - 1, 0
    while left <= right:
        perm[idx] = left
        idx += 1
        if left != right:
            perm[idx] = right
            idx += 1
        left += 1
        right -= 1
    return perm


def _apply_desc_permutation(vals: np.ndarray, perm: np.ndarray) -> np.ndarray:
    """Place the values sorted descending according to ``perm``."""
    desc = np.sort(vals)[::-1]          # descending
    out = np.empty_like(desc)
    out[perm] = desc
    # normalise to sum = 1 (preserves ordering)
    out /= out.sum()
    return out


# ---------------------------------------------------------------
# Random Dirichlet sampling
# ---------------------------------------------------------------

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet(alpha,…,alpha) vector."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ---------------------------------------------------------------
# Smoothed Adam optimiser (log‑its parametrisation)
# ---------------------------------------------------------------

def _smooth_adam_optimize(start_seq: np.ndarray,
                          deadline: float,
                          beta_start: float = 5.0,
                          beta_end: float = 200.0,
                          max_iter: int = 7_000_000) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of ``max(conv(a,a))``.
    *Logits* → softmax → the probability vector.
    The smooth‑max is a log‑sum‑exp with annealed temperature β.
    Returns the best sequence seen and its exact objective.
    """
    n = len(start_seq)
    eps = 1e-30

    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.02

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        a = np.exp(logits)
        a /= a.sum()

        conv = _conv_fft(a, a)

        # linear schedule for β (first half of iters the temperature grows)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.5))

        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()                 # soft‑weights

        # gradient w.r.t. a (2 * convolution of p with reversed a)
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if (it & 2047) == 0:      # every 2048 steps
            val = _objective(np.exp(logits) / np.exp(logits).sum())
            if val < best_val:
                best_val = val
                best_seq = np.exp(logits) / np.exp(logits).sum()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

            if best_val < 1.500:      # aggressive early stop
                break

    return best_seq, best_val


# ---------------------------------------------------------------
# Exact sub‑gradient mirror descent (exponentiated‑gradient)
# ---------------------------------------------------------------

def _refine_subgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Classical KL‑mirror descent on the simplex using the true
    sub‑gradient of the active convolution entry.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Refine] it={it:7d} n={n:5d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.7 / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ---------------------------------------------------------------
# Random permutation / hill‑climb search
# ---------------------------------------------------------------

def _permute_hillclimb(base_seq: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Keep the multiset of values fixed and search for a better ordering.
    A huge number of random permutations (and occasional two‑point swaps)
    are tried; the best ordering found is returned.
    """
    n = len(base_seq)
    best_seq = base_seq.copy()
    best_val = _objective(best_seq)

    # a first fast random‑permutation sweep
    attempts = 0
    while time.time() < deadline and attempts < 300_000:
        perm = rng.permutation(n)
        cand = base_seq[perm]
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Perm] attempt={attempts:6d} val={val:.6f}")
            if best_val < 1.503:
                break
        attempts += 1

    # small simulated‑annealing swap‑based local search
    cur_seq = best_seq.copy()
    cur_val = best_val
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cand = cur_seq.copy()
        cand[i], cand[j] = cand[j], cand[i]
        val = _objective(cand)
        # simple simulated annealing acceptance
        temp = max(0.001, 0.1 / sqrt(it))
        accept = (val < cur_val) or (rng.random() < np.exp(-(val - cur_val) / temp))
        if accept:
            cur_seq, cur_val = cand, val
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"[Swap] it={it:7d} new_best={val:.6f}")
                if best_val < 1.503:
                    break
    return best_seq, best_val


# ---------------------------------------------------------------
# Main entry point
# ---------------------------------------------------------------

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes the
    ``evaluate_sequence`` objective.  Returns the best sequence found
    as a plain list (normalised to sum = 1).
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Try to reuse a previously stored solution if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – a compromise between expressive power and speed
    # ------------------------------------------------------------------
    lengths = [2500, 3000, 3500, 4000, 4500, 5000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic seeds (with spread permutation)
    # ------------------------------------------------------------------
    seed_families = (
        ("uniform", _uniform),
        ("catalan", _catalan_shape),
        ("hann", _hann),
        ("hamming", _hamming),
        ("blackman", _blackman),
        ("gauss15", lambda n: _gaussian(n, sigma_factor=0.15)),
        ("gauss25", lambda n: _gaussian(n, sigma_factor=0.25)),
    )

    alt_perm_cache = {}   # cache permutations for each length

    for n in lengths:
        # basic families
        for name, fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[Seed] n={n:5d} type={name:10s} val={val:.6f}")

            # spread version (alternating extremes)
            if n not in alt_perm_cache:
                alt_perm_cache[n] = _alternating_extreme_permutation(n)
            perm = alt_perm_cache[n]
            seq_spread = _apply_desc_permutation(seq, perm)
            val_spread = _objective(seq_spread)
            if val_spread < best_val:
                best_val, best_seq = val_spread, seq_spread.copy()
                print(f"[Seed‑Spread] n={n:5d} type={name:10s} val={val_spread:.6f}")

        # also try the previously stored vector if it matches the length
        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val, best_seq = val, previous.copy()
                print(f"[Seed‑Prev] n={n:5d} val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – massive random Dirichlet sampling (several alphas)
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.35   # ~35 % of remaining budget
    sample_deadline = now + time_for_sampling
    alphas = [0.3, 0.5, 0.8, 1.0, 2.0]

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.2f} val={val:.6f}")
            if best_val < 1.505:
                break

    # ------------------------------------------------------------------
    # Phase 3 – Smoothed Adam optimisation
    # ------------------------------------------------------------------
    if best_seq is not None:
        opt_deadline = time.time() + (deadline - time.time()) * 0.25   # ~25 % of remaining
        seq_adam, val_adam = _smooth_adam_optimize(best_seq, opt_deadline,
                                                   beta_start=5.0, beta_end=200.0,
                                                   max_iter=8_000_000)
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam
            print(f"[Adam‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – permutation hill‑climb (keep multiset of values)
    # ------------------------------------------------------------------
    if best_seq is not None:
        perm_deadline = time.time() + (deadline - time.time()) * 0.30   # ~30 % of remaining
        seq_perm, val_perm = _permute_hillclimb(best_seq, perm_deadline, rng)
        if val_perm < best_val:
            best_val, best_seq = val_perm, seq_perm
            print(f"[Perm‑Climb] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – exact sub‑gradient mirror descent (final polish)
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _refine_subgradient(best_seq, deadline)
        if final_val < best_val:
            best_val, best_seq = final_val, final_seq
            print(f"[Final‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check (the grader will call evaluate_sequence again)
    # ------------------------------------------------------------------
    result = best_seq.tolist()
    try:
        evaluate_sequence(result)
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )
    return result
