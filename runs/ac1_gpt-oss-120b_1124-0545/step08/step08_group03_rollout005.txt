"""
Random‑sparse search for a low‑autocorrelation step‑function.

The algorithm works with a fixed length `n = 4000`.  It repeatedly
draws a random support of size `m` (≈ 800‑2500) and places equal mass
`1/m` on the chosen indices.  The exact objective
`2·n·max(conv(a,a))/(sum(a)^2)` is evaluated with an FFT‑based convolution.
The best vector seen is kept.  When the time budget is almost exhausted a
short exponentiated‑gradient (sub‑gradient) refinement is performed to
polish the current best candidate.  The final best sequence is stored in
the global variable ``height_sequence_1`` and returned as a plain list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT convolution (exact, real‑valued)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence up to clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(seq,seq))/sum(seq)^2.
    The input is assumed to be a non‑negative ndarray.
    """
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple exponentiated‑gradient refinement (sub‑gradient on exact objective)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact non‑smooth objective.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    w = np.zeros(2 * n - 1, dtype=np.float64)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # build the sub‑gradient (average over active constraints)
        g = np.zeros_like(a)
        if active.size:
            w.fill(0.0)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient step
        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)            # keep numerics stable
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the supplied time budget and returns the best
    sequence discovered as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # initialise RNG and timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    rng = np.random.default_rng(seed + 1)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Random‑sparse optimiser, length {n} ===")

    # ------------------------------------------------------------------
    # start from the previous solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.ndim == 1 and prev_arr.shape[0] == n:
                best_seq = prev_arr / prev_arr.sum()   # normalise to sum=1
                best_val = _objective(best_seq)
                print(f"[Init] previous best objective = {best_val:.6f}")
            else:
                raise ValueError
        except Exception:
            best_seq = np.full(n, 1.0 / n, dtype=np.float64)
            best_val = _objective(best_seq)
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)

    # ------------------------------------------------------------------
    # Random sparse sampling loop
    # ------------------------------------------------------------------
    it = 0
    # choose a range for the support size – empirically values around
    # 1000‑2000 give good peaks
    min_m, max_m = 800, 2500

    while True:
        now = time.time()
        # leave a little time for a final refinement pass
        if now >= deadline - 20.0:
            break

        # draw a random support size
        m = int(rng.integers(min_m, max_m + 1))
        # random distinct indices
        idx = rng.choice(n, size=m, replace=False)

        a = np.zeros(n, dtype=np.float64)
        a[idx] = 1.0 / m                       # equal mass on the support
        val = _objective(a)

        if val < best_val:
            best_val = val
            best_seq = a
            print(f"[Random] it={it:5d} m={m:4d} obj={val:.6f}")

        it += 1

    print(f"[Random] finished after {it} samples; best so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional polishing with sub‑gradient (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        seq, val = _subgradient_refine(best_seq, n, deadline,
                                       max_iters=200_000)
        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"[Polish] after sub‑gradient best={val:.6f}")

    # ------------------------------------------------------------------
    # finalise and store the result
    # ------------------------------------------------------------------
    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
          f"(length {len(best_seq)})")
    globals()["height_sequence_1"] = np.copy(best_seq)
    # convert to plain Python list of floats as required by the grader
    return best_seq.tolist()
