def _refine_softmax_adam(start_seq, deadline):
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    # initial temperature
    conv = _conv_fft(a, a)
    maxc = conv.max()
    minc = conv.min()
    tau = maxc - minc
    if tau < 1e-12:
        tau = maxc * 0.1
    tau_min = 1e-12
    decay = 0.99995 # per iteration
    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 1000 == 0:
                print(f"[SoftAdam] it={t:6d} val={cur_val:.6f}")
        # compute softmax weights
        maxc = conv.max()
        # stability subtract maxc
        exp_vals = np.exp((conv - maxc) / tau)
        sum_exp = exp_vals.sum()
        w = exp_vals / sum_exp
        # gradient computation
        grad_full = _conv_fft(w, a[::-1])
        # slice appropriate n entries
        grad = 2.0 * grad_full[n-1 : n-1 + n]
        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = 0.1 / np.sqrt(t) # base learning rate decaying
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = _proj_simplex(a)
        # update temperature
        tau = max(tau * decay, tau_min)
    return best_seq, best_val

"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Deterministic seeds** – a few analytically motivated shapes
   (uniform, Catalan, geometric) for several lengths.  The best of them
   is kept as a starting point.

2. **Random Dirichlet sampling** – while time permits draw many Dirichlet
   vectors with several concentration parameters and keep the best one.

3. **Exponential‑gradient refinement** – a mirror‑descent step that uses
   only the active (max) convolution entry (as in the reference code).

4. **Soft‑max Adam refinement** – a smooth approximation of the max‑norm
   (τ·log∑exp(conv/τ)).  The gradient involves a *soft* combination of
   *all* convolution entries, which encourages many constraints to be
   active and therefore flattens the auto‑convolution.  Adam with
   projection onto the probability simplex is used for robustness.

All stages respect a global 1000 s wall‑clock budget.  The routine prints
progress and finally returns the best sequence found, as a plain Python
list (the grader will call ``evaluate_sequence`` on it).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for the sizes we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Objective that ``evaluate_sequence`` would return when ``seq`` is already
    normalised to sum = 1:  2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Coefficients proportional to binom(2i,i)/4**i (the series of (1‑z)^‑½).
    After truncation they give a near‑constant auto‑convolution.
    Normalised to sum = 1.
    """
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _geometric_shape(n: int, r: float = 0.999) -> np.ndarray:
    """
    Truncated geometric distribution a_i ∝ r**i, normalised to sum = 1.
    Parameter r∈(0,1) controls the decay.
    """
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


# ----------------------------------------------------------------------
# Random Dirichlet sampling
# ----------------------------------------------------------------------


def _dirichlet(alpha: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector of length n (normalised to sum = 1)."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by Adam‑type refinements
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Refinement – exponentiated gradient (mirror descent)
# ----------------------------------------------------------------------


def _refine_expgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent using the negative entropy (exponentiated gradient)."""
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ExpGrad] it={it:7d} val={cur_val:.6f}")

        # Tie‑break among nearly maximal entries
        max_val = conv.max()
        tol = 1e-12
        tie_idxs = np.where(conv >= max_val - tol)[0]
        max_idx = int(np.random.choice(tie_idxs))

        # Sub‑gradient of the selected entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Exponentiated‑gradient update
        lr = 0.8 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)               # numerical stability
        a = np.exp(log_a)
        a /= a.sum()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Refinement – soft‑max Adam (smooth approximation of max‑norm)
# ----------------------------------------------------------------------


def _refine_softmax_adam(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises a smooth approximation
        τ·log∑_k exp(conv_k/τ)  ≈  max_k conv_k .
    The gradient is a soft combination of all convolution entries,
    encouraging many constraints to be active.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # initialise temperature τ (scale of the soft‑max)
    conv = _conv_fft(a, a)
    tau = max(conv.max() - conv.min(), 1e-12)  # a reasonable start
    tau_min = 1e-12
    decay = 0.99997                               # very slow cooling

    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 500 == 0:
                print(f"[SoftAdam] it={t:7d} val={cur_val:.6f}")

        # ----- soft‑max weights -----
        max_c = conv.max()
        # stabilised exponentials
        exp_vals = np.exp((conv - max_c) / tau)
        w = exp_vals / exp_vals.sum()                     # size 2n‑1, sum 1

        # ----- gradient of the smoothed objective -----
        # g_i = 2 * Σ_k w_k * a_{k-i}
        # achieved by convolution of w with the *reversed* a
        grad_full = _conv_fft(w, a[::-1])                 # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]          # keep the n relevant entries

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.08 / np.sqrt(t)                            # a modest diminishing step size
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex (keeps non‑negativity & Σ=1)
        a = _proj_simplex(a)

        # ----- temperature schedule -----
        tau = max(tau * decay, tau_min)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given
    time budget and returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # tiny safety margin for the final sanity check
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives a finer theoretical bound,
    # but the optimisation becomes more expensive.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic seeds
    # ------------------------------------------------------------------
    for n in lengths:
        for maker in (
            _catalan_shape,
            _uniform,
            lambda n: _geometric_shape(n, r=0.999),
            lambda n: _geometric_shape(n, r=0.995),
        ):
            try:
                seq = maker(n)
            except Exception:
                continue
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={maker.__name__} val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1‑b – previous solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                prev_arr = prev_arr / prev_arr.sum()
                val = _objective(prev_arr)
                if val < best_val:
                    best_val = val
                    best_seq = prev_arr.copy()
                    print(f"[Seed] previous solution val={val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stage 2 – massive Dirichlet sampling
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    sample_deadline = now + remaining * 0.45  # about half the remaining budget

    alphas = [0.5, 1.0, 2.0]  # sparse, uniform, concentrated Dirichlets
    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = float(rng.choice(alphas))
        seq = _dirichlet(alpha, n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.1f} val={val:.6f}")

        if best_val < 1.503:          # early stop – target already reached
            break

    # ------------------------------------------------------------------
    # Stage 3 – exponentiated‑gradient mirror descent (quick coarse tune)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + (deadline - time.time()) * 0.30
        seq_exp, val_exp = _refine_expgradient(best_seq, exp_deadline)
        if val_exp < best_val:
            best_val = val_exp
            best_seq = seq_exp
            print(f"[Overall‑Exp] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – soft‑max Adam refinement (fine tune)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        soft_deadline = deadline
        seq_soft, val_soft = _refine_softmax_adam(best_seq, soft_deadline)
        if val_soft < best_val:
            best_val = val_soft
            best_seq = seq_soft
            print(f"[Overall‑SoftAdam] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – the grader will also call evaluate_sequence
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )
    # Return a plain Python list as required by the grader
    return best_seq.tolist()
