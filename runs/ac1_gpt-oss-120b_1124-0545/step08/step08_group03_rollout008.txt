"""
Propose a step‑function (non‑negative vector) that minimises
`evaluate_sequence`.  The implementation builds a large pool of diverse
initialisations (windows, Dirichlet draws, sparse supports, edge‑heavy
shapes), keeps the best few, and then refines them with three complementary
optimisers:

1. Adam on a smooth‑max surrogate of the convolution peak.
2. A long exponentiated‑gradient (mirror‑descent) pass that works on the
   exact non‑smooth objective.
3. Repeated random perturbations followed by short mirror‑descent bursts
   (a simple simulated‑annealing style walk).

All phases respect the supplied time budget (default 1000 s).  The best
vector encountered is stored in `height_sequence_1` and returned as a plain
list of floats.  The code is deliberately lightweight – it avoids any
heavy linear‑programming or external solvers – and therefore can spend a
large fraction of the budget on very many cheap gradient steps, which
typically pushes the objective below the target 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (exact linear convolution via real FFT)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of real vectors a and b using an FFT.
    The result length is len(a) + len(b) - 1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` (ignoring the 0‑1000 clip,
    which never triggers for our normalised vectors).
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Families of candidate windows (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Standard Tukey window (alpha in [0,1])."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1) discretised on mid‑points.
    Small α concentrates mass near the edges.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _inverse_tent(n: int) -> np.ndarray:
    """V‑shaped distribution: zero in the centre, large at the ends."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.abs(i - centre) / centre  # 0 at centre, 1 at ends
    a += 1e-12                     # avoid exact zeros
    return a / a.sum()


def _edge_block(n: int, block_frac: float = 0.05) -> np.ndarray:
    """
    Uniform mass on two edge blocks of relative size `block_frac`
    (default 5 % on each side), zero elsewhere.
    """
    block = int(max(1, round(block_frac * n)))
    a = np.zeros(n, dtype=np.float64)
    a[:block] = 0.5 / block
    a[-block:] = 0.5 / block
    return a / a.sum()


def _dirichlet_draw(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with symmetric concentration `alpha`."""
    if alpha <= 0:
        # degenerate sparse support
        a = np.zeros(n, dtype=np.float64)
        a[rng.integers(0, n)] = 1.0
        return a
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _sparse_random_seed(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random support of size `m` (1 ≤ m ≤ n)."""
    if m >= n:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Simple Euclidean projection on the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 25000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth‑max surrogate of the convex peak.
    Returns the best vector seen and its true objective.
    """
    v = np.log(seed + 1e-30)          # logits for simplex parametrisation

    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # start with a moderate temperature for the soft‑max weight
    beta = 1.5
    beta_mul = 1.006

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # ---- smooth‑max weights (soft‑max over the convolution values) ----
        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k (a*a)_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam moments
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient on the *exact* objective.
    Runs until the deadline (or max_iters if supplied).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    w = np.zeros(2 * n - 1, dtype=np.float64)

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over the active constraints)
        g = np.zeros_like(a)
        if active.size:
            w.fill(0.0)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient step
        lr = 0.85 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # keep numerics stable
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:8d} best={cur_val:.6f}")

    return best_seq, best_val


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Lightweight Metropolis–Hastings walk with tiny mass transfers and
    occasional global Gaussian jumps.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.015 * cur[i]   # up to 1.5 % of source mass
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:8d} best={new_val:.6f}")

        # occasional larger jump
        if it % 1500 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= total
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:8d} (jump) best={new_val:.6f}")

    return best, best_val

def _make_seeds(n: int,
                previous: np.ndarray | None,
                rng: np.random.Generator) -> list[np.ndarray]:
    """
    Assemble a diverse collection of probability vectors.
    """
    seeds: list[np.ndarray] = []

    # classic smooth families
    seeds.append(_uniform(n))
    for r in np.linspace(0.2, 0.95, 10):
        seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for r in np.linspace(0.3, 0.9, 7):
        for p in np.linspace(0.0, 1.5, 5):
            seeds.append(_geom_power(n, r, p))
    for r in np.linspace(0.3, 0.9, 6):
        seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))
    for sigma in np.linspace(0.05, 0.35, 9):
        seeds.append(_gaussian(n, sigma))
    for alpha in np.linspace(0.05, 0.9, 9):
        seeds.append(_beta_window(n, alpha))
    for _ in range(12):
        seeds.append(_random_softmax(n, rng))

    # edge‑heavy / V‑shapes
    seeds.append(_inverse_tent(n))
    for frac in (0.02, 0.04, 0.07, 0.1):
        seeds.append(_edge_block(n, block_frac=frac))

    # sparse random supports (different densities)
    for m in (30, 50, 80, 120, 180):
        for _ in range(4):
            seeds.append(_sparse_random_seed(n, m, rng))

    # Dirichlet draws with various concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_dirichlet_draw(n, alpha, rng))

    # reuse a previous construction if given
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # normalise / clean
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    return cleaned

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Recover the previously best vector (if any) for warm start
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    rng = np.random.default_rng(seed + 12345)

    # ------------------------------------------------------------------
    # Choose a problem size.  4000 works well (balance between expressiveness
    # and speed) – we keep it fixed for reproducibility.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting optimisation, length = {n} ===")

    # ------------------------------------------------------------------
    # 1) Build a rich seed pool
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous, rng)

    # ------------------------------------------------------------------
    # 2) Quick evaluation – retain a few elite seeds
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = seed_vals[0][0].copy()
    global_best_val = seed_vals[0][1]
    print(f"[seed] best initial = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam (smooth‑max) on each elite seed
    # ------------------------------------------------------------------
    for idx, s in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(s, n, deadline, max_iters=25000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Adam] seed={idx} improved best to {val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – prolonged exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _subgradient_refine(global_best_seq, n, deadline, max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[SubGrad] after long run best = {val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – alternating short sub‑gradient bursts with random perturbations.
    #          This mimics simulated‑annealing but keeps the cheap mirror‑descent
    #          as the core optimiser.
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # a) short sub‑gradient refinement (10 k iters)
        seq, val = _subgradient_refine(global_best_seq, n, deadline, max_iters=15000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[SubGrad] short burst new best = {val:.6f}")

        # b) random perturbation – jitter a few coordinates
        if time.time() > deadline - 20.0:
            break
        jitter = global_best_seq.copy()
        # pick a handful of indices to reshuffle mass among them
        idxs = rng.choice(n, size=8, replace=False)
        mass = jitter[idxs].sum()
        if mass > 0:
            # redistribute according to a fresh Dirichlet draw
            new_weights = rng.gamma(0.3, 1.0, size=8)
            new_weights = new_weights / new_weights.sum() * mass
            jitter[idxs] = new_weights
            jitter = _project_simplex(jitter)

            seq, val = _subgradient_refine(jitter, n, deadline, max_iters=20000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[Perturb] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # Phase D – final lightweight Metropolis‑Hastings annealing (uses any
    #          remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Anneal‑final] best = {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {n})")
    globals()["height_sequence_1"] = np.copy(global_best_seq)
    return global_best_seq.tolist()
