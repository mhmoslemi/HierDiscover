def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (fast for large vectors).
    Returns a real array of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv

    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (return normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = a0 - a1 * np.cos(2 * pi * i / (n - 1)) + a2 * np.cos(4 * pi * i / (n - 1))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total


def _power_law(n: int, p: float = 0.5) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) , normalized."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total


def _kaiser_window(n: int, beta: float = 4.0) -> np.ndarray:
    """Kaiser window with shape parameter β."""
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha  # in [-1,1]
    w = i0(beta * np.sqrt(1 - t * t)) / i0(beta)
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total


def _lanczos_window(n: int, a: float = 3.0) -> np.ndarray:
    """
    Lanczos (sinc) window.  We square the sinc to guarantee non‑negativity.
    """
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha  # [-1,1]
    w = np.sinc(a * t) ** 2
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total


# ----------------------------------------------------------------------
# Random Dirichlet sampling
# ----------------------------------------------------------------------


def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """
    Sample a Dirichlet(alpha,…,alpha) vector (alpha > 0).
    """
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑its parametrisation)
# ----------------------------------------------------------------------


def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 0.1,
    beta_end: float = 200.0,
    max_iter: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)) using Adam.
    The smoothing uses a temperature β that is linearly annealed.
    Returns the best sequence found (norm‑1) and its exact objective.
    """
    n = len(start_seq)
    eps = 1e-30

    # initialise logits = log(a)
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999
    lr0 = 0.04  # a bit larger than the previous version

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # soft‑max to obtain probabilities
        a = np.exp(logits)
        a = a / a.sum()

        # self‑convolution
        conv = _conv_fft(a, a)

        # annealed temperature
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.5))

        # smooth max via log‑sum‑exp
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()  # soft‑weights, sum = 1

        # gradient w.r.t. a : g = 2 * (p * a) *convolution
        conv_pa = _conv_fft(p, a[::-1])          # length = 3n‑2
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]     # central slice, length n

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # periodic exact evaluation
        if (it & 1023) == 0:  # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:8d} n={n:5d} val={val:.6f} beta={beta:.1f}")

            if best_val < 1.49:  # generous early‑stop
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient mirror descent (exponentiated gradient)
# ----------------------------------------------------------------------


def _refine_subgradient(
    start_seq: np.ndarray,
    deadline: float,
    max_iter: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Classical KL‑mirror descent (exponentiated gradient) using the true
    sub‑gradient of the active convolution entry.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Refine] it={it:8d} n={n:5d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size decays as 1/√it
        lr = 0.7 / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)        # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given
    time budget and returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # a tiny safety margin

    # ------------------------------------------------------------------
    # Possibly reuse a previously stored solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                prev_arr = prev_arr / prev_arr.sum()
                previous = prev_arr
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore (moderate to large, still FFT‑friendly)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic seeds
    # ------------------------------------------------------------------
    seed_families = (
        _uniform,
        _catalan_shape,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.12),
        lambda n: _gaussian(n, sigma_factor=0.20),
        lambda n: _power_law(n, p=0.45),
        lambda n: _power_law(n, p=0.55),
        lambda n: _kaiser_window(n, beta=2.0),
        lambda n: _kaiser_window(n, beta=4.0),
        lambda n: _kaiser_window(n, beta=6.0),
        lambda n: _lanczos_window(n, a=2.0),
        lambda n: _lanczos_window(n, a=3.0),
    )

    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:14s} val={val:.6f}")

        # try previous solution if compatible length
        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – massive random Dirichlet sampling
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.40  # 40 % of remaining budget
    sample_deadline = now + time_for_sampling
    alphas = [0.3, 0.5, 1.0, 2.0]

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.2f} val={val:.6f}")

        if best_val < 1.49:  # early break if we are already very good
            break

    # ------------------------------------------------------------------
    # Phase 3 – Smoothed Adam optimisation (multiple starts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # allocate roughly 30 % of the remaining time
        phase3_deadline = time.time() + (deadline - time.time()) * 0.30
        # we may try a few restarts with small perturbations
        restarts = 3
        for r in range(restarts):
            # add a tiny random perturbation to avoid identical starting point
            pert = best_seq * (1.0 + 0.01 * rng.normal(size=best_seq.shape))
            pert = np.clip(pert, 0, None)
            pert = pert / pert.sum()
            seq_opt, val_opt = _smooth_adam_optimize(pert, phase3_deadline)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = seq_opt
                print(f"[Adam‑Refine] restart={r} improved best to {best_val:.6f}")

            if time.time() >= phase3_deadline:
                break

    # ------------------------------------------------------------------
    # Phase 4 – Exact sub‑gradient mirror descent (final polishing)
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _refine_subgradient(best_seq, deadline)
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Final‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Sanity check – the grader will invoke evaluate_sequence again,
    # but we check it once here to catch accidental bugs.
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
