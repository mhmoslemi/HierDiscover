def _lp_mixing_step(f_seq, n, deadline):
    # f_seq normalized sum=1.
    M = np.max(_conv_fft(f_seq, f_seq))
    # Build sparse Toeplitz convolution matrix constraints
    # Build data arrays
    # row indices = (offsets[:,None] + np.arange(n)[None,:]).ravel()
    # col indices = np.tile(np.arange(n), n)
    # data = np.repeat(f_seq, n)
    row = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    col = np.tile(np.arange(n), n)
    data = np.repeat(f_seq, n)
    A = csr_matrix((data, (row, col)), shape=(2*n-1, n))
    b = np.full(2*n-1, M, dtype=np.float64)

    # linear program: maximize sum(g)  <=> minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)

    # use HiGHS, set time limit as remaining
    remaining = deadline - time.time()
    if remaining <= 0:
        return None, float('inf')
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'time_limit': remaining})
    if not res.success:
        return None, float('inf')
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return None, float('inf')
    g_norm = g / sum_g
    # mixing: average with f
    new = 0.5 * (f_seq + g_norm)
    val = _objective(new)
    return new, val

    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate (and exact, for sum(seq)=1) objective:
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (row‑vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (Σ = 1, v ≥ 0).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    # geometric progression a_i = r**i
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed collection (deterministic + random families)
# ----------------------------------------------------------------------


def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.15))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))

    # Geometric families
    for r in np.linspace(0.2, 0.98, 10):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # Power‑law family
    for p in np.linspace(0.0, 4.0, 15):
        seeds.append(_power(n, p))

    # Beta family
    alphas = np.linspace(0.3, 2.0, 6)
    betas = np.linspace(0.3, 4.0, 6)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows
    for sigma_frac in (0.15, 0.25, 0.35, 0.45):
        seeds.append(_gaussian(n, sigma_frac))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws (heavy‑tailed)
    for _ in range(3):
        seeds.append(_random_exponential(n, rng))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best sequence found and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)  # work in log‑softmax space

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 5.0            # start temperature (moderately smooth)
    beta_mul = 1.005
    beta_max = 2000.0

    best_seq = None
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(v)                     # probabilities, Σa = 1
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # increase smoothing temperature
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp on the convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k · conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 8.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.7
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Uses additive updates with Euclidean projection onto the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the max at index max_idx:
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # additive step (decrease direction) and simplex projection
        lr = 0.25 / np.sqrt(it)   # slightly larger initial step
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _lp_mixing_step(
    f_seq: np.ndarray,
    n: int,
    deadline: float,
) -> tuple[np.ndarray | None, float]:
    """
    Linear‑programming mixing step.
    Solve:
        maximise Σ g_j
        s.t.   (f * g)_k ≤ M   for all k,
               g_j ≥ 0.
    The solution g is rescaled to have unit ℓ₁‑norm and then averaged with f.
    """
    M = np.max(_conv_fft(f_seq, f_seq))

    # Build the Toeplitz convolution matrix T_f (shape (2n‑1, n)).
    # Row i, column j gets f_{i-j} if i-j in [0,n‑1], else 0.
    # This can be constructed efficiently with broadcasting.
    row = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.repeat(f_seq, n)

    T_f = csr_matrix((data, (row, col)), shape=(2 * n - 1, n))

    # Linear program data
    c = -np.ones(n, dtype=np.float64)          # minimise –Σ g_j  ⇔ maximise Σ g_j
    b = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Respect remaining time
    remaining = deadline - time.time()
    if remaining <= 0.0:
        return None, float("inf")

    # HiGHS solver via SciPy
    options = {"time_limit": remaining}
    res = linprog(c, A_ub=T_f, b_ub=b, bounds=bounds, method="highs", options=options)

    if not res.success:
        return None, float("inf")
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0:
        return None, float("inf")

    # Rescale to unit ℓ₁‑norm and average with f
    g_norm = g / sum_g
    mixed = 0.5 * (f_seq + g_norm)

    val = _objective(mixed)
    return mixed, val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_200_000,
    init_temp_factor: float = 0.025,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing that rescales the mass of a random
    subset of coordinates at each step.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9997, min_temp)

        k = rng.integers(2, 12)          # size of the randomly chosen subset
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget (default 1000 s) and returns
    the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Length choice – 4000 offers a good balance.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any) for warm start.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.shape != (n,) or previous.sum() == 0.0:
                previous = None
            else:
                previous = previous / previous.sum()
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a rich pool of seeds.
    # ------------------------------------------------------------------
    print(f"\n--- Generating seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)

    # Add a larger random sweep over the analytic families
    for _ in range(200):
        seeds.append(_beta_shape(n, rng.uniform(0.1, 3.0), rng.uniform(0.1, 3.0)))
        seeds.append(_power(n, rng.uniform(0.0, 5.0)))
        seeds.append(_gaussian(n, rng.uniform(0.1, 0.6)))
        seeds.append(_tukey_window(n, rng.uniform(0.05, 0.95)))
        seeds.append(_geom(n, rng.uniform(0.2, 0.99)))
        seeds.append(_random_dirichlet(n, rng.uniform(0.05, 2.0), rng))

    # Evaluate all seeds (cheap surrogate)
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # Initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – heavy optimisation (Adam → sub‑gradient → LP mixing)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Optimisation run {idx+1}/{len(best_seeds)} ---")

        # Adam phase
        seq_adam, val_adam = _adam_optimize(
            seed_vec, n, deadline, max_iters=200_000
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam] new global best = {val_adam:.6f}")

        # Sub‑gradient phase
        seq_proj, val_proj = _subgradient_proj(
            global_best_seq, n, deadline, max_iters=500_000
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[Proj] new global best = {val_proj:.6f}")

        # LP mixing step (once per seed)
        if time.time() < deadline:
            seq_lp, val_lp = _lp_mixing_step(global_best_seq, n, deadline)
            if seq_lp is not None and val_lp < global_best_val:
                global_best_seq = seq_lp.copy()
                global_best_val = val_lp
                print(f"[LP   ] new global best = {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final stochastic polishing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=1_200_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA   ] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"Final evaluate_sequence value: {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # store for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
