"""
Propose a candidate step‑function (non‑negative heights) whose
objective

    2·n·max(conv(a,a)) / (sum(a)^2)

is as small as possible.

The main idea
----------------
* Build a *low‑collision* support set S ⊂ {0,…,n‑1} such that every
  ordered sum i+j (i,j∈S) occurs at most C=4 times.
  For a set of size m the uniform distribution on S gives
      max(conv) = C / m²  and  objective ≈ 2·C·n / m² .
  The maximum feasible size satisfies  m² ≤ C·(2n‑1)  (average‑collision bound),
  i.e.  m ≤ √[C·(2n‑1)].  For n=4000 and C=4 this theoretical bound is
  ≈ 179.  Getting a set with m≈175 already yields an objective below 1.4,
  well under the target 1.503.

* The greedy construction from the previous code is kept but
  – many more restarts are performed,
  – after each greedy run a cheap *expansion* phase tries to add more
    indices that still respect the collision limit,
  – the best mask found (largest |S|) is kept.

* If the uniform distribution on the best mask already beats the target,
  it is returned immediately.  Otherwise a short Adam / sub‑gradient
  refinement (as in the reference implementation) is executed.

The whole procedure respects the supplied time budget (default 1000 s)
and returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (exact, real‑valued via FFT)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The input is assumed to be a non‑negative vector.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple families (kept for fallback / seed diversity)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Low‑collision set construction (C = max ordered collisions)
# ----------------------------------------------------------------------


def _expand_support(mask: np.ndarray,
                    sum_counts: np.ndarray,
                    max_ordered: int,
                    rng: np.random.Generator,
                    max_trials: int = 2000) -> tuple[np.ndarray, np.ndarray]:
    """
    After a greedy run, try to insert additional indices while keeping the
    collision limit.  This cheap local improvement can boost the final size.
    """
    n = mask.size
    for _ in range(max_trials):
        cand = rng.integers(0, n)
        if mask[cand]:
            continue
        # self‑pair
        if sum_counts[2 * cand] >= max_ordered:
            continue
        existing = np.nonzero(mask)[0]
        if existing.size:
            sums = cand + existing
            if np.any(sum_counts[sums] + 2 > max_ordered):
                continue
        # accept
        mask[cand] = True
        sum_counts[2 * cand] += 1
        if existing.size:
            sum_counts[sums] += 2
    return mask, sum_counts


def _low_collision_set_improved(n: int,
                                max_ordered: int,
                                rng: np.random.Generator,
                                deadline: float,
                                max_total_attempts: int = 5000) -> np.ndarray | None:
    """
    Greedy construction of a set S ⊂ {0,…,n‑1} such that each ordered sum
    i+j occurs at most `max_ordered` times.  The routine performs many
    restarts and a cheap expansion phase, returning the mask of the best
    (largest) set found.
    """
    best_mask = None
    best_size = 0
    # theoretical upper bound from average‑collision argument
    theoretical_max = int(np.sqrt(max_ordered * (2 * n - 1)))
    attempts = 0

    while time.time() < deadline - 0.5 and attempts < max_total_attempts:
        attempts += 1
        mask = np.zeros(n, dtype=bool)
        sum_counts = np.zeros(2 * n - 1, dtype=np.uint8)

        order = rng.permutation(n)
        for cand in order:
            # self‑pair check
            if sum_counts[2 * cand] + 1 > max_ordered:
                continue

            # pair with existing elements
            existing = np.nonzero(mask)[0]
            if existing.size:
                sums = cand + existing
                # each new pair adds two ordered pairs
                if np.any(sum_counts[sums] + 2 > max_ordered):
                    continue

            # accept candidate
            mask[cand] = True
            sum_counts[2 * cand] += 1
            if existing.size:
                sum_counts[sums] += 2

        # cheap expansion
        mask, sum_counts = _expand_support(mask, sum_counts, max_ordered, rng)

        cur_size = mask.sum()
        if cur_size > best_size:
            best_size = cur_size
            best_mask = mask.copy()
            print(f"[LowColl] attempt {attempts:4d} size {cur_size}")

        if best_size >= theoretical_max:
            # cannot be improved substantially
            break

    if best_mask is None or best_size == 0:
        return None
    print(f"[LowColl] best size after {attempts} attempts = {best_size}")
    return np.where(best_mask)[0]


# ----------------------------------------------------------------------
# Simple Adam optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 12000) -> tuple[np.ndarray, float]:
    """
    Adam on a smoothed‑max surrogate of max(conv(a,a)).
    Returns the best vector seen and its true objective.
    """
    v = np.log(seed + 1e-30)          # logits for simplex parametrisation
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 1.3
    beta_mul = 1.007

    best_seq = seed.copy()
    best_val = _objective(seed)
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k · (a*a)_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam moment update
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Sub‑gradient (exponentiated gradient) refinement for the exact objective
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 300000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact non‑smooth objective.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over active constraints)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]
        else:
            g = np.zeros_like(a)

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 42,

    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  A fast low‑collision construction is tried first;
    if it already beats the target (≈1.503) the result is returned.
    Otherwise a short Adam + sub‑gradient refinement is performed.
    The routine respects the supplied time budget and returns the best
    sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 is a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000
    max_ordered = 4                     # allowed ordered‑sum multiplicity
    print(f"\n=== Optimising length {n} (max_ordered={max_ordered}) ===")

    rng = np.random.default_rng(seed + 12345)

    # ------------------------------------------------------------------
    # Phase 0 – aggressive low‑collision support search
    # ------------------------------------------------------------------
    low_coll_deadline = min(deadline - 5.0, start_time + 0.35 * budget_s)
    idx_set = _low_collision_set_improved(
        n=n,
        max_ordered=max_ordered,
        rng=rng,
        deadline=low_coll_deadline,
        max_total_attempts=4000
    )

    if idx_set is not None and idx_set.size > 0:
        a = np.zeros(n, dtype=np.float64)
        a[idx_set] = 1.0 / idx_set.size          # uniform on the support
        val = _objective(a)
        print(f"[Result] low‑collision uniform (|S|={idx_set.size}) obj={val:.6f}")

        if val < 1.503:                           # target already met
            final_seq = a / a.sum()
            globals()["height_sequence_1"] = final_seq.copy()
            print("\n=== Finished early (target reached) ===")
            return final_seq.tolist()

        # keep it as the current best for further refinement
        best_seq = a.copy()
        best_val = val
    else:
        # fallback to a simple uniform start
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print(f"[Fallback] uniform start obj={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam on a smooth‑max surrogate (shortened)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        seq, val = _adam_optimize(best_seq, n, deadline, max_iters=12000)
        if seq is not None and val < best_val:
            best_seq, best_val = seq.copy(), val
            print(f"[Overall Adam] improved best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact sub‑gradient mirror‑descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        seq, val = _subgradient_refine(best_seq, n, deadline,
                                       max_iters=200000)
        if seq is not None and val < best_val:
            best_seq, best_val = seq.copy(), val
            print(f"[Overall SubGrad] improved best={val:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation & return
    # ------------------------------------------------------------------
    final_seq = np.maximum(best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
        total = final_seq.sum()
    final_seq = final_seq / total

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(final_seq)})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
