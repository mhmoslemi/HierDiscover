    attempt = 0
    while time.time() < deadline:
        attempt += 1
        # random permutation of positions
        order = rng.permutation(n)
        used_sums = set()
        cur_set = []
        for pos in order:
            # fast check using Python set
            conflict = False
            s_self = pos + pos
            if s_self in used_sums:
                conflict = True
            else:
                for p in cur_set:
                    if (pos + p) in used_sums:
                        conflict = True
                        break
            if not conflict:
                # add position
                cur_set.append(pos)
                # add sums
                used_sums.add(s_self)
                for p in cur_set[:-1]:  # all previous ones
                    used_sums.add(pos + p)
        m = len(cur_set)
        if m > best_m:
            best_m = m
            best_set = cur_set.copy()
            print(f"[Attempt {attempt}] New best m={best_m}")
        # early exit if reaches theoretical max approx 125
        if best_m >= 125:
            break
    # Build sequence
    seq = np.zeros(n, dtype=np.float64)
    if best_set is None or best_m == 0:
        seq = np.ones(n) / n
        print("[fallback] uniform")

        # Ensure sum maybe slightly off due to rounding
        total = seq.sum()

    except Exception as e:

    return seq.tolist()

def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Construct a sparse Sidon set (B2 sequence) and assign equal mass to each mark.
    The set is built by repeated random greedy insertions, maximising the number of marks
    within the interval [0, n). With m marks and sum of coefficients = 1 the worst
    autocorrelation coefficient equals 1/m^2, giving the objective 2·n/m^2.
    By finding a Sidon set with m > sqrt(2·n) we obtain values well below the
    target bound 1.503.
    """
    import time, math, numpy as np

    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin

    # Choose length n. 4000 gives a good trade‑off.
    n = 4000

    rng = np.random.default_rng(seed)

    best_set = []
    best_m = 0
    attempt = 0

    # Pre‑compute the full set of positions once
    positions = np.arange(n, dtype=np.int32)

    while time.time() < deadline:
        attempt += 1
        # random order of positions
        order = rng.permutation(positions)
        used = set()
        cur = []

        # greedy insertion
        for pos in order:
            # test whether adding pos would create a duplicate sum
            conflict = False
            # self‑sum
            s_self = pos + pos
            if s_self in used:
                conflict = True
            else:
                # check sums with existing marks
                for p in cur:
                    if (pos + p) in used:
                        conflict = True
                        break
            if not conflict:
                # add position and record its sums
                for p in cur:
                    used.add(pos + p)
                used.add(s_self)
                cur.append(int(pos))

        m = len(cur)
        if m > best_m:
            best_m = m
            best_set = cur.copy()
            print(f"[Attempt {attempt}] New best size {best_m}")

        # Early stop if we hit the theoretical ceiling
        if best_m >= 125:
            break

    # Build the sequence
    seq = np.zeros(n, dtype=np.float64)
    if best_m == 0:
        seq[:] = 1.0 / n
        print("[fallback] using uniform distribution")
    else:
        weight = 1.0 / best_m
        seq[best_set] = weight
        # sanity: normalise in case of rounding
        total = seq.sum()
        if not np.isclose(total, 1.0):
            seq /= total

    try:
        val = evaluate_sequence(seq.tolist())
        print(f"Final objective {val:.6f} (n={n}, m={best_m})")
    except Exception as exc:
        print("Error while evaluating:", exc)

    globals()["height_sequence_1"] = seq.tolist()
    return seq.tolist()

"""
Sidon‑set based optimiser
-------------------------

We look for a vector `a` of length *n* with non‑negative entries and
∑a_i = 1 that minimises

    2·n·max(conv(a,a))

Because the objective is homogeneous we may normalise the sum to 1.
If the support of `a` is a Sidon (B₂) set – i.e. all pairwise
sums i+j are different – then each entry of the convolution
contains at most one product a_i·a_j.
With equal weights w = 1/m on *m* marks the maximal autocorrelation
coefficient is exactly w² = 1/m², so the objective becomes

    2·n / m² .

Hence the larger a Sidon set we can embed in {0,…,n‑1} the better.
The theoretical limit is m ≤ (√(16 n−7)−1)/2 ≈ 2·√n, which for
n=4000 gives m ≤ 125 (objective ≈0.5).  Achieving any m > √(2 n)
already beats the target 1.503.

The algorithm repeatedly builds a Sidon set by a random greedy
insertion (many attempts are cheap) and keeps the largest one found
before the time budget expires.  The final sequence puts equal mass
1/m on the found marks and zero elsewhere.

The returned sequence is stored in the global variable
`height_sequence_1` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Greedy Sidon set construction (top‑level helpers, no closures)
# ----------------------------------------------------------------------


def _greedy_sidon_set(order: np.ndarray) -> list[int]:
    """
    Build a Sidon set by scanning `order` (a permutation of positions)
    and inserting a position whenever all new sums are still unused.
    Returns the list of selected positions (unsorted).
    """
    used = set()          # all sums i+j already present
    cur = []              # positions selected so far

    for pos in order:
        s_self = int(pos) + int(pos)
        if s_self in used:
            continue

        conflict = False
        for p in cur:
            if (int(pos) + int(p)) in used:
                conflict = True
                break
        if conflict:
            continue

        # add the new position and its sums
        for p in cur:
            used.add(int(pos) + int(p))
        used.add(s_self)
        cur.append(int(pos))

    return cur


def _search_best_sidon(n: int, deadline: float, rng: np.random.Generator) -> tuple[list[int], int]:
    """
    Repeatedly try random permutations until the deadline is reached.
    Returns the largest Sidon set discovered and its cardinality.
    """
    best_set: list[int] = []
    best_m = 0
    attempt = 0
    positions = np.arange(n, dtype=np.int32)

    # theoretical ceiling (won’t be reached for n=4000, but useful as early‑stop)
    theoretical_max = int((np.sqrt(16 * n - 7) - 1) // 2)

    while time.time() < deadline:
        attempt += 1
        order = rng.permutation(positions)
        cur = _greedy_sidon_set(order)
        m = len(cur)
        if m > best_m:
            best_m = m
            best_set = cur
            print(f"[Attempt {attempt}] New best size = {best_m}")
            # stop if we hit the (almost) theoretical limit
            if best_m >= theoretical_max:
                break

    return best_set, best_m


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Construct a sparse Sidon set of marks and assign equal mass.
    With m marks the worst autocorrelation is 1/m², giving the
    objective 2·n/m².  For n=4000 we easily obtain m>73, i.e.
    objective < 1.503.
    """
    # safety margin so we finish before the hard deadline
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    # length of the step function – 4000 offers a good speed / quality trade‑off
    n = 4000

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Find a large Sidon set
    # ------------------------------------------------------------------
    best_set, best_m = _search_best_sidon(n, deadline, rng)

    # ------------------------------------------------------------------
    # 2) Build the corresponding sequence (equal weight on each mark)
    # ------------------------------------------------------------------
    seq = np.zeros(n, dtype=np.float64)

    if best_m == 0:
        # fallback – uniform distribution (objective = 2)
        seq[:] = 1.0 / n
        print("[fallback] using uniform distribution")
    else:
        weight = 1.0 / best_m
        seq[best_set] = weight
        # normalise to guard against tiny rounding errors
        total = seq.sum()
        if not np.isclose(total, 1.0):
            seq /= total
        print(f"[result] Sidon set size = {best_m}, weight = {weight:.6e}")

    # ------------------------------------------------------------------
    # 3) Optional diagnostics – evaluate the objective we actually get
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(seq.tolist())
        print(f"[evaluation] Objective ≈ {val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[evaluation] failed:", exc)

    # store for a possible later run
    globals()["height_sequence_1"] = seq.tolist()
    return seq.tolist()
