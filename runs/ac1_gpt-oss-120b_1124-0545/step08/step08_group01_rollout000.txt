def _project_simplex(v):
    # similar to algorithm in Duchi et al. 2008.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Non‑negative step‑function optimiser – version 2.

Goal
----
Find a non‑negative vector `a` of length `n` (default 4000) that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

Because the objective is scale‑invariant we normalise every candidate to
`sum(a)=1`.  The problem is then to minimise `max(conv(a,a))`.

Approach
--------
1. **Rich seed pool** – deterministic families (uniform, triangular,
   sine‑window, symmetric exponential, symmetric power‑law, beta,
   two‑block) plus random Dirichlet / sparse draws.  All seeds are
   normalised to a probability simplex and the best few are kept.

2. **Smooth‑max Adam** – minimise a log‑sum‑exp surrogate of the
   convolution using Adam on the log‑softmax parametrisation.
   The smoothing temperature β is slowly increased up to a large value
   (≈ 5000) so that the surrogate approaches the true maximum.

3. **Projected Gradient Descent (PGD)** – starting from the best Adam
   result, perform many iterations of classic projected gradient descent
   on the exact max‑convolution objective.  The gradient is the
   sub‑gradient at the current maximal index; after each step we project
   back onto the probability simplex (Duchi et al. 2008).  A simple
   momentum term and gradient clipping accelerate convergence.

4. **Simulated Annealing (SA)** – a cheap hill‑climber that repeatedly
   redistributes the mass of a random small subset of coordinates.
   The temperature is annealed geometrically; always‐accept moves that
   improve the surrogate and occasionally accept worse moves.

5. **Upsampling** – if time permits, linearly upsample the best vector
   to twice the resolution and repeat PGD + SA.  The upsampled vector
   inherits the smooth shape and can achieve a lower maximum convolution
   value.

All stages respect a global deadline (`budget_s` seconds).  The best
vector found is stored in the global name ``height_sequence_1`` and
returned as a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Surrogate objective used during the search: 2·n·max(conv(seq,seq)).
    Assumes ``seq`` is non‑negative and already normalised to sum = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (Duchi et al.)."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families – all return a normalised probability vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    if lam <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-lam * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_power(n: int, p: float) -> np.ndarray:
    """sin(π·(i+1)/(n+1)) raised to power p."""
    if p <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1)) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _two_block(n: int, edge_frac: float, ratio: float) -> np.ndarray:
    """
    Piecewise constant: edges of length edge_frac·n on each side have
    value = ratio·inner, inner is the value on the middle block.
    """
    if edge_frac <= 0.0 or edge_frac >= 0.5:
        return _uniform(n)
    edge_len = max(1, int(round(edge_frac * n)))
    interior_len = n - 2 * edge_len
    if interior_len <= 0:
        return _uniform(n)

    # let inner = x, edge = ratio * x
    denom = 2 * edge_len * ratio + interior_len
    inner = 1.0 / denom
    edge = ratio * inner

    a = np.empty(n, dtype=np.float64)
    a[:edge_len] = edge
    a[-edge_len:] = edge
    a[edge_len:-edge_len] = inner
    return a / a.sum()  # just in case of rounding


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()

def _generate_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # symmetric geometric – medium grid
    for r in np.linspace(0.10, 0.90, 17):
        seeds.append(_sym_geom(n, r))

    # symmetric power‑law
    for p in np.arange(0.0, 5.01, 0.25):
        i = np.arange(1, n + 1, dtype=np.float64)
        a = np.power(i, -p) + np.power(n - i + 1, -p)
        s = a.sum()
        if s > 0:
            seeds.append(a / s)

    # symmetric exponential – several decay rates
    for lam in (0.001, 0.005, 0.010, 0.020, 0.050, 0.100):
        seeds.append(_sym_exp(n, lam))

    # sin‑power family
    for p in (0.5, 1.0, 2.0, 3.0, 5.0, 8.0, 12.0):
        seeds.append(_sin_power(n, p))

    # two‑block piecewise constant
    for ef in (0.01, 0.02, 0.05, 0.10):
        for r in (0.5, 1.0, 2.0, 5.0, 10.0):
            seeds.append(_two_block(n, ef, r))

    # beta family – finer grid
    for a in np.linspace(0.05, 2.0, 9):
        for b in np.linspace(0.05, 5.0, 9):
            seeds.append(_beta_shape(n, a, b))

    # mixtures of uniform and symmetric geometric
    uniform = _uniform(n)
    for r in np.linspace(0.20, 0.90, 9):
        base = _sym_geom(n, r)
        for w in np.linspace(0.0, 1.0, 5):
            mixed = w * uniform + (1.0 - w) * base
            mixed = mixed / mixed.sum()
            seeds.append(mixed)

    # random Dirichlet draws
    for conc in (0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (3, 5, 10, 20):
        for conc in (0.20, 0.50):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution if compatible
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 60000,
                   beta_max: float = 5000.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a log‑sum‑exp (smoothed max) surrogate.
    The vector is parametrised by logits and kept on the simplex via
    the softmax.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5
    beta_mul = 1.001

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                 # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # increase smoothing temperature (capped)
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp on the convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _pgd_optimize(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int = 300000,
                  momentum: float = 0.9,
                  clip_norm: float = 20.0) -> tuple[np.ndarray, float]:
    """
    Projected (with optional momentum) gradient descent on the exact
    max‑convolution objective.
    """
    a = start_seq.copy()
    best = a.copy()
    best_val = _objective(a)

    vel = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if it % 5000 == 0:
                print(f"[PGD] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the maximal index
        grad = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        grad[idx] = 2.0 * a[rev_idx]

        # optional momentum
        vel = momentum * vel - grad

        step = 0.5 / np.sqrt(it)          # diminishing step size
        a_new = a + step * vel

        # clip gradient norm for stability (applied after momentum)
        norm = np.linalg.norm(a_new - a)
        if norm > clip_norm:
            a_new = a + (clip_norm / norm) * (a_new - a)

        # projection onto simplex
        a = _project_simplex(a_new)

    return best, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 250000,
                         init_temp_factor: float = 0.02,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles the mass of a random small subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)           # subset size 2…6
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new.copy()
                print(f"[SA]  it={it:7d} best={new_val:.6f}")

    return best, best_val

                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal ``evaluate_sequence``.
    Returns the best sequence found (list of floats) and stores it in the
    global variable ``height_sequence_1`` for possible reuse.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Reuse previous solution if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose length – 4000 is a good trade‑off
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – generate a large pool of seeds and keep the top few
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _generate_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} candidate seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed objectives:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – heavy optimisation (Adam → PGD) on the top seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Optimisation run {idx+1}/{len(best_seeds)} ---")

        # 1) Adam on smoothed max
        seq_adam, val_adam = _adam_optimize(
            seed_vec, n, deadline, max_iters=50000, beta_max=5000.0
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # 2) Projected Gradient Descent (exact max)
        seq_pgd, val_pgd = _pgd_optimize(
            global_best_seq, n, deadline, max_iters=200000
        )
        if val_pgd < global_best_val:
            global_best_seq = seq_pgd.copy()
            global_best_val = val_pgd
            print(f"[overall‑PGD] new global best = {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=250000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – optional up‑sampling and refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_big = n * 2
        x_old = np.linspace(0.0, 1.0, n, dtype=np.float64)
        x_big = np.linspace(0.0, 1.0, n_big, dtype=np.float64)
        upsampled = np.interp(x_big, x_old, global_best_seq)
        upsampled = upsampled / upsampled.sum()
        print(f"\n--- Upsampling to n={n_big} and refining ---")

        # PGD on the larger vector
        seq_big, val_big = _pgd_optimize(
            upsampled, n_big, deadline, max_iters=150000
        )
        if val_big < global_best_val:
            global_best_seq = seq_big.copy()
            global_best_val = val_big
            n = n_big
            print(f"[Upsample‑PGD] new global best = {val_big:.6f}")

        # final SA on upsampled version (if time permits)
        if time.time() < deadline:
            seq_sa2, val_sa2 = _simulated_annealing(
                global_best_seq, n, deadline, max_iters=150000
            )
            if val_sa2 < global_best_val:
                global_best_seq = seq_sa2.copy()
                global_best_val = val_sa2
                print(f"[Upsample‑SA] new global best = {val_sa2:.6f}")

    # ------------------------------------------------------------------
    # Sanity check with the provided evaluate_sequence (optional)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose the result for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

This implementation expands the seed pool with additional analytic families, uses a more aggressive Adam schedule (higher smoothing temperature), adds a dedicated projected‑gradient‑descent phase with momentum, and applies simulated‑annealing polishing both before and after an optional up‑sampling step. All stages respect the 1000‑second budget and store the best-found sequence in `height_sequence_1`.
