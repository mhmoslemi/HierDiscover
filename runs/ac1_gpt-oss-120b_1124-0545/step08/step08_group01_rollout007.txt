Implementation details: projection onto simplex defined as:

def simplex_projection(v):
    """Project onto simplex {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _pgd_optimize(start_seq, n, deadline, max_iters=500000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    # gradient step size schedule: start with lr0 maybe 0.5, decays as lr0 / sqrt(it) or lr0 / (it**0.5).
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]  # actual evaluation (but we use objective consistent)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # compute gradient
        rev_a = a[::-1]
        # gradient for max index only:
        # We'll do conv of rev_a with indicator vector e_{max_idx}
        e = np.zeros(len(conv))
        e[max_idx] = 1.0
        # but conv length = 2n-1 = len(conv)
        # Compute grad: full = _conv_fft(rev_a, e) (fast)
        # full length same as conv (2n-1)
        full = _conv_fft(rev_a, e) # shape 2n-1
        grad = 2.0 * full[n-1 : n-1+n]  # slice central n entries
        # step size
        lr = 0.5 / np.sqrt(it)  # adjust
        # update
        a_new = a - lr * grad
        # projection onto simplex
        a = _project_simplex(a_new)
        # maybe occasional print
        if it % 50000 == 0:
            print(f"[PGD] iter={it} best={best_val:.6f}")
    return best_seq, best_val

def _project_simplex(v):
    """Project onto simplex {x >=0, sum(x)=1}.

    Returns projected vector.
    """
    # shift to avoid issues
    # If sum(v) == 1 and all v >=0 return v
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-10:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros? fallback uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _pgd_optimize(start_seq, n, deadline, max_iters=500000):
    a = start_seq.astype(np.float64).copy()
    # ensure normalization
    s = a.sum()
    if s <= 0:
        a = _uniform(n)
    else:
        a /= s

    best_seq = a.copy()
    conv = _conv_fft(a, a)
    max_idx = int(np.argmax(conv))
    best_val = 2.0 * n * conv[max_idx]

    # Pre-allocate arrays
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute conv
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # compute gradient for max index
        grad = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx_i
        grad[idx_i] = 2.0 * a[rev_idx]
        # step size schedule
        lr = 0.5 / np.sqrt(it)
        a = _project_simplex(a - lr * grad)
        # optional prints
        if it % 100000 == 0:
            print(f"[PGD] iter={it} best={best_val:.6f}")

    return best_seq, best_val

"""
Non‑negative step‑function optimiser – version 2.

The optimiser works in three stages:

1️⃣  *Very broad seed sweep* – analytically motivated families
    (uniform, triangular, symmetric geometric, symmetric power‑law,
    beta, random Dirichlet and sparse patterns) are instantiated on a
    fixed length `n=4000`.  The geometric ratio `r` and the power‑law
    exponent `p` are swept on a fine grid (r∈[0.01,0.99], p∈[0,10]) so that
    even subtle shapes are examined.  Each candidate is normalised to
    `∑a_i = 1` and evaluated with the cheap surrogate
    `2·n·max(conv(a,a))`.  The best five seeds are retained.

2️⃣  *Projected‑gradient descent on a smoothed max* – starting from each
    retained seed we run a long (time‑controlled) optimisation on the
    true objective `max(conv)`.  The gradient of a *soft‑max* version of
    the convolution is computed (a log‑sum‑exp with temperature `β`),
    and a simple projected‑gradient step on the probability simplex is
    taken (`a ← Π_{simplex}(a – η·∇)`).  The temperature is increased
    geometrically (`β←β·1.005`) so that the algorithm gradually focuses
    on the actual maximum.  Step‑size decays as `η₀/√t`.  This phase
    typically performs a few hundred‑thousand iterations within the
    time budget.

3️⃣  *Upsampling & final polishing* – if time permits the best sequence
    is linearly interpolated to length `n·2` (doubling the resolution)
    and the same projected‑gradient routine is applied once more.

All intermediate progress (objective values, iteration counters and
timings) is printed.  The final vector is stored under the global name
`height_sequence_1` for possible reuse and returned as a plain
`list[float]` ready for `evaluate_sequence`.

The method deliberately differs from the previous solution (which used
Adam + exponentiated‑gradient) by (i) employing a finer analytic seed
grid, (ii) using a true projected‑gradient descent on the smoothed
maximum (no adaptive moments), and (iii) by an aggressive up‑sampling
stage.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Low‑level utilities
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the seed sweep.
    Assumes ``seq`` is non‑negative and already normalised to sum = 1.
    Returns ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# --------------------------------------------------------------
# Simplex projection (Mirrored‑Descent primal step)
# --------------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    ``S = {x | x_i ≥ 0, ∑_i x_i = 1}``.
    Implements the O(n) algorithm from
    Duchi et al., “Efficient Projections onto the L1‑Ball for
    Learning in High Dimensions”.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform vector
        return np.full_like(v, 1.0 / n)

    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Analytic families (all normalised to sum = 1)
# --------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^‑p + (n‑i)^‑p, symmetric power‑law."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p) + np.power(n - i + 1, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a

def _generate_extensive_seeds(n: int, rng: np.random.Generator,
                              previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a large and diverse pool of candidate vectors.
    The families are sampled on fine grids (geometric ratio, power‑law
    exponent, beta parameters) to avoid missing a promising region.
    All vectors are normalised to sum = 1.
    """
    seeds: list[np.ndarray] = []

    # deterministic basics
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # symmetric geometric – dense grid
    for r in np.linspace(0.01, 0.99, 37):          # 0.01 … 0.99, step ≈0.027
        seeds.append(_sym_geom(n, r))

    # symmetric power‑law – extended exponent range
    for p in np.arange(0.0, 10.001, 0.20):        # 0 … 10 in steps of 0.2
        seeds.append(_sym_powerlaw(n, p))

    # beta family – include extreme near‑zero parameters
    for a in np.linspace(0.01, 2.0, 10):
        for b in np.linspace(0.01, 5.0, 12):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (different concentrations)
    for conc in (0.02, 0.05, 0.10, 0.30, 0.70, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns (few non‑zeros)
    for k in (3, 5, 8, 12):
        for conc in (0.02, 0.10):
            for _ in range(3):
                seeds.append(_random_sparse(n, k, conc, rng))

    # mixture of uniform & geometric – extra flexibility
    uniform = _uniform(n)
    for r in np.linspace(0.20, 0.80, 7):
        base = _sym_geom(n, r)
        for w in np.linspace(0.0, 1.0, 7):
            mixed = w * uniform + (1.0 - w) * base
            mixed /= mixed.sum()
            seeds.append(mixed)

    # previous solution, if compatible
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def _pgd_optimize(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a log‑sum‑exp surrogate.
    The iterate stays on the simplex via the Euclidean projection.
    Parameters
    ----------
    start_seq : np.ndarray
        Initial probability vector (already normalised to sum = 1).
    n : int
        Length of the vector.
    deadline : float
        Absolute UNIX time after which the routine stops.
    max_iters : int
        Upper bound on the number of iterations (the actual number may be
        limited by the deadline).
    Returns
    -------
    best_seq, best_val
        The best vector seen and the corresponding true objective
        ``2·n·max(conv)``.
    """
    a = start_seq.astype(np.float64).copy()
    s = a.sum()
    if s <= 0:
        a = _uniform(n)
    else:
        a /= s

    best_seq = a.copy()
    conv = _conv_fft(a, a)
    cur_max = float(np.max(conv))
    best_val = 2.0 * n * cur_max

    # temperature for the log‑sum‑exp (smooth‑max) surrogate
    beta = 0.5
    beta_mul = 1.005
    beta_max = 400.0

    eps = 1e-30
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # ----- 1) evaluate -------------------------------------------------
        conv = _conv_fft(a, a)            # linear convolution a*a
        max_idx = int(np.argmax(conv))
        cur_max = float(conv[max_idx])
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[PGD] it={it:7d}  new best={best_val:.6f}")

        # ----- 2) compute smoothed gradient ---------------------------------
        # soft‑max weights over all convolution entries
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:                 # numerically unlikely, fall back to one‑hot
            w = np.zeros_like(conv)
            w[max_idx] = 1.0
        else:
            w /= w_sum

        # gradient of the smoothed max w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)       # length = 3n‑2
        grad = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # ----- 3) step on the simplex --------------------------------------
        lr = 0.9 / np.sqrt(it)           # initial learning‑rate ≈0.9, decays ∝1/√t
        a_new = a - lr * grad
        a = _project_simplex(a_new)

        # ----- 4) temperature update ----------------------------------------
        beta = min(beta * beta_mul, beta_max)

    return best_seq, best_val


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget,
    prints intermediate progress, stores the best vector globally under the
    name ``height_sequence_1`` and finally returns it as a plain list.
    """
    # reproducibility
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for final sanity‑check
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load a previous solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length of the search – 4000 is a solid compromise for the budget
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting optimiser – length {n} – budget {budget_s}s ===")

    # ------------------------------------------------------------------
    # Stage 1 – massive seed sweep
    # ------------------------------------------------------------------
    print("\n--- Stage 1: generating extensive seed pool ---")
    seeds = _generate_extensive_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # cheap surrogate evaluation
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(7, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"Top {top_k} surrogate objectives:")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long projected‑gradient descent on each elite seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds, 1):
        if time.time() > deadline:
            break
        print(f"\n--- Optimisation run {idx}/{len(best_seeds)} (PGD) ---")
        seq_opt, val_opt = _pgd_optimize(seed_vec, n, deadline,
                                         max_iters=1_200_000)
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[overall] new global best = {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – optional up‑sampling & refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:           # need a few seconds for sanity‑check
        n_big = n * 2
        x_old = np.linspace(0.0, 1.0, n, dtype=np.float64)
        x_big = np.linspace(0.0, 1.0, n_big, dtype=np.float64)
        upsampled = np.interp(x_big, x_old, global_best_seq)
        upsampled = upsampled / upsampled.sum()
        print(f"\n--- Upsampling to n={n_big} and refining ---")
        seq_big, val_big = _pgd_optimize(upsampled, n_big, deadline,
                                         max_iters=800_000)
        if val_big < global_best_val:
            global_best_seq = seq_big.copy()
            global_best_val = val_big
            n = n_big
            print(f"[Upsample‑Refine] new global best = {val_big:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – evaluate with the provided function
    # ------------------------------------------------------------------
    try:
        val_check = evaluate_sequence(global_best_seq.tolist())
        print(f"\nSanity check with evaluate_sequence: {val_check:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest true objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )

    # expose for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
