def _conv_fft(x,y):
    L = len(x)+len(y)-1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    prod = fx*fy
    conv = np.fft.irfft(prod, n=N)[:L]
    return conv
```

def _refine_softmax_adam(start_seq, deadline):
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    # Adam params
    beta1=0.9; beta2=0.999; eps=1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    # parameters for softmax
    tau = 0.5
    tau_inc = (10.0 - 0.5) / (deadline - time.time())  # adapt gradually up to 10?
    # Use learning rate schedule
    base_lr = 0.05
    while time.time() < deadline:
        t += 1
        # compute convolution
        conv = _conv_fft(a, a)
        maxc = conv.max()
        cur_val = 2.0 * n * maxc
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SoftAdam] it={t} val={cur_val:.6f} tau={tau:.3f}")
        # compute softmax weights
        # subtract max for stability
        w = np.exp(tau * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # gradient via correlation
        w_rev = w[::-1]
        grad_full = _conv_fft(w_rev, a)  # length 3n-2
        start = len(w) - 1  # = 2n-2
        g = 2.0 * grad_full[start : start + n]  # slice length n
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = base_lr / np.sqrt(t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex
        a = _proj_simplex(a_new)
        # increase tau
        if tau < 20.0:
            tau = min(20.0, tau + tau_inc)
        # occasional noise
        if t % 5000 == 0:
            a = _proj_simplex(a + np.random.default_rng().normal(scale=1e-5, size=n))
    return best_seq, best_val
```

"""
Improved search for a non‑negative step function that minimizes
``evaluate_sequence``.

Main ideas
----------
1. Deterministic analytic seeds (uniform, Catalan, geometric, sine,
   inverse‑sqrt, parabolic) for a few lengths.

2. Random Dirichlet sampling with a wider set of concentration
   parameters (including very small and large α).

3. Three refinement stages, all time‑aware:
   • Exponentiated‑gradient (mirror descent) – as before.
   • Projected Adam – as before.
   • **Soft‑max‑Adam** – a smooth approximation of the max
     (soft‑max) is used to obtain a dense gradient that pushes all
     large convolution entries down simultaneously.  The temperature
     τ is increased during the run to approach the true max.

All helper functions are top‑level; no closures or lambdas are used.
The routine respects the supplied time budget (default 1000 s) and
returns a plain Python list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for the sizes we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Objective that ``evaluate_sequence`` would return when ``seq`` is
    already normalised to sum = 1:  2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Coefficients proportional to binom(2i,i)/4**i (the series of (1‑z)^‑½).
    After truncation they give a near‑constant auto‑convolution.
    Normalised to sum = 1.
    """
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _geometric_shape(n: int, r: float = 0.999) -> np.ndarray:
    """
    Truncated geometric distribution a_i ∝ r**i, normalised to sum = 1.
    The parameter r∈(0,1) controls the decay; r close to 1 is almost flat.
    """
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


def _sine_shape(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n), positive on (0,π)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n)
    total = a.sum()
    return a / total


def _inverse_sqrt_shape(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _parabolic_shape(n: int) -> np.ndarray:
    """Concave quadratic peak: a_i = 1 - ((i-(n-1)/2) / ((n-1)/2))²."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - ((i - centre) / centre) ** 2
    a[a < 0] = 0.0
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


# ----------------------------------------------------------------------
# Random Dirichlet sampling (α may vary)
# ----------------------------------------------------------------------


def _dirichlet(alpha: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Draw a Dirichlet(α,…,α) vector of length n using Gamma variables.
    Returns a vector normalised to sum = 1.
    """
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by Adam refinements
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Refinement – exponentiated gradient (mirror descent)
# ----------------------------------------------------------------------


def _refine_expgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using the negative entropy (exponentiated gradient).
    Returns the best vector seen before the deadline.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ExpGrad] it={it:7d} val={cur_val:.6f}")

        # Tie‑break randomly among entries close to the maximum
        tol = 1e-12
        tie_idxs = np.where(conv >= max_val - tol)[0]
        max_idx = int(np.random.choice(tie_idxs))

        # Sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Exponentiated‑gradient update
        lr = 0.8 / np.sqrt(it)           # diminishing step size
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)           # improve numerical stability
        a = np.exp(log_a)
        a /= a.sum()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Refinement – projected Adam (adaptive moments)
# ----------------------------------------------------------------------


def _refine_proj_adam(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with Euclidean projection onto the simplex after each step.
    Returns the best vector seen before the deadline.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 1000 == 0:
                print(f"[Adam] it={t:7d} val={cur_val:.6f}")

        # Sub‑gradient of the maximal entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.05 / np.sqrt(t)           # simple diminishing base learning rate
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _proj_simplex(a_new)

        # occasional tiny random perturbation to escape shallow plateaus
        if t % 5000 == 0:
            noise = rng.normal(scale=1e-5, size=n)
            a = _proj_simplex(a + noise)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Refinement – soft‑max Adam (smooth approximation of max)
# ----------------------------------------------------------------------


def _refine_softmax_adam(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation using a soft‑max approximation of the max‑convolution.
    Gradient is the correlation of the soft‑max weights with the current
    vector (dense sub‑gradient).  Temperature τ is increased during the run.
    Returns the best vector seen before the deadline.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    rng = np.random.default_rng()

    # start and end temperature for the soft‑max (τ controls sharpness)
    tau = 0.5
    tau_target = 20.0
    # linear schedule
    tau_inc = (tau_target - tau) / max(1.0, deadline - time.time())

    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        maxc = conv.max()
        cur_val = 2.0 * n * maxc

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SoftAdam] it={t:7d} val={cur_val:.6f} τ={tau:.2f}")

        # ----- soft‑max weights -----
        # w_k ∝ exp(τ·(conv_k – maxc))  (stable, all weights sum to 1)
        w = np.exp(tau * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # ----- gradient = 2 * correlation(w, a) -----
        # correlation can be obtained via a convolution with the reversed w
        w_rev = w[::-1]
        grad_full = _conv_fft(w_rev, a)               # length = 3n‑2
        start = len(w) - 1                            # = 2n‑2
        g = 2.0 * grad_full[start : start + n]        # dense sub‑gradient

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.05 / np.sqrt(t)                        # diminishing step
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _proj_simplex(a_new)

        # ---- temperature schedule ----
        if tau < tau_target:
            tau = min(tau_target, tau + tau_inc)

        # occasional random jitter to avoid stagnation
        if t % 5000 == 0:
            a = _proj_simplex(a + rng.normal(scale=1e-5, size=n))

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given
    time budget (default 1000 s) and returns the best sequence found as
    a plain list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives a finer bound but costs more
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic analytic seeds
    # ------------------------------------------------------------------
    seed_makers = (
        _catalan_shape,
        _uniform,
        lambda n: _geometric_shape(n, r=0.999),
        lambda n: _geometric_shape(n, r=0.995),
        _sine_shape,
        _inverse_sqrt_shape,
        _parabolic_shape,
    )
    for n in lengths:
        for maker in seed_makers:
            try:
                seq = maker(n)
                val = _objective(seq)
                if val < best_val:
                    best_val = val
                    best_seq = seq.copy()
                    print(f"[Seed] n={n:5d} type={maker.__name__} val={val:.6f}")
            except Exception:
                continue

    # ------------------------------------------------------------------
    # Stage 2 – incorporate any previously stored solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                prev_arr = prev_arr / prev_arr.sum()
                val = _objective(prev_arr)
                if val < best_val:
                    best_val = val
                    best_seq = prev_arr.copy()
                    print(f"[Seed] previous solution val={val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stage 3 – massive random Dirichlet sampling (various α)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    sample_deadline = now + remaining * 0.45   # ~45 % of the remaining budget

    alphas = [0.2, 0.5, 1.0, 2.0, 5.0]          # explore sparse & smooth regimes
    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = float(rng.choice(alphas))
        seq = _dirichlet(alpha, n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.1f} val={val:.6f}")
        # early stop if we already beat a reasonable target
        if best_val < 1.48:
            break

    # ------------------------------------------------------------------
    # Stage 4 – exponentiated‑gradient mirror descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + (deadline - time.time()) * 0.30
        seq_exp, val_exp = _refine_expgradient(best_seq, exp_deadline)
        if val_exp < best_val:
            best_val = val_exp
            best_seq = seq_exp
            print(f"[Overall‑Exp] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – projected Adam refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = time.time() + (deadline - time.time()) * 0.25

    # ------------------------------------------------------------------
    # Stage 6 – soft‑max Adam refinement (dense gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        soft_deadline = deadline
        seq_soft, val_soft = _refine_softmax_adam(best_seq, soft_deadline)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
