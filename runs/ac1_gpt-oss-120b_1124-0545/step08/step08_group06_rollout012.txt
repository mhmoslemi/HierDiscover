"""
Propose a non‑negative step‑function (list of heights) that minimises
the ``evaluate_sequence`` objective.

Algorithm
---------
1. **Deterministic families** – for a few lengths (4000, 8000) generate
   analytically motivated distributions (uniform, Catalan, geometric,
   sine‑based, Beta‑family, power‑sine).  Keep the best one.

2. **Random sampling** – while time permits draw many random vectors:
   * Dirichlet(α,…,α) with α drawn log‑uniform in [0.3,5].
   * Discrete Beta(α,β) obtained from the regularised incomplete beta
     function (α,β drawn log‑uniform).

3. **Mirror descent (exponentiated gradient)** – starting from the best
   vector, perform a multiplicative update using the sum of gradients
   of the *top‑k* convolution entries (k≈5).  The learning rate
   decays as 1/√t.  The best iterate seen before the sub‑deadline is
   retained.

4. **Projected Adam** – an adaptive‑gradient step with Euclidean
   projection onto the probability simplex.  A tiny Gaussian perturbation
   is added every few thousand steps to escape shallow plateaus.

5. **Final check** – the resulting list is fed to ``evaluate_sequence``
   (the grader will do the same) and returned.

The routine respects the supplied time budget (default 1000 s) and
prints progress information.  All helpers are top‑level, no closures
or lambdas are used.
"""

import time
import numpy as np
from scipy.special import betainc
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for the sizes we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Objective that ``evaluate_sequence`` would return when ``seq`` is already
    normalised to sum = 1:  2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Coefficients proportional to binom(2 i,i)/4**i (the series of (1‑z)^‑½).
    Normalised to sum = 1.
    """
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _geometric_shape(n: int, r: float = 0.9995) -> np.ndarray:
    """Truncated geometric distribution a_i ∝ r**i, normalised to sum = 1."""
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _sin_shape(n: int) -> np.ndarray:
    """p_i ∝ sin(π·i/(n‑1)).  Zero at the ends, symmetric."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _sin_power_shape(n: int, gamma: float = 2.0) -> np.ndarray:
    """p_i ∝ sin(π·i/(n‑1))**γ , γ>0."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1)) ** gamma
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discrete Beta(α,β) on {0,…,n‑1}.  The probability of index i is
    I_{(i+1)/n}(α,β) − I_{i/n}(α,β) where I is the regularised incomplete
    beta function.  Returns a vector that sums to 1.
    """
    # CDF values at the bin edges
    x = np.linspace(0.0, 1.0, n + 1)
    cdf = betainc(a, b, x)
    probs = np.diff(cdf)               # length n
    total = probs.sum()
    if total <= 0.0:
        return _uniform(n)
    return probs / total


# ----------------------------------------------------------------------
# Random generators
# ----------------------------------------------------------------------


def _dirichlet(alpha: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Draw a Dirichlet(α,…,α) vector (α>0) of length n and normalise to sum = 1.
    """
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Simplex projection – Euclidean (used by Adam)
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Refinement – exponentiated gradient (multi‑constraint)
# ----------------------------------------------------------------------


def _refine_expgradient_multi(start_seq: np.ndarray,
                              deadline: float,
                              top_k: int = 5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the probability simplex using the negative entropy.
    The sub‑gradient is the sum of the gradients of the *top_k* largest
    convolution entries (ties are automatically handled).
    Returns the best sequence seen before ``deadline``.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ExpGrad] it={it:7d} val={cur_val:.6f}")

        # indices of the k largest entries (k ≥ 1)
        k = min(top_k, conv.size)
        top_idxs = np.argpartition(-conv, k - 1)[:k]

        # aggregated gradient of the selected entries
        g = np.zeros_like(a)
        for idx in top_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = idx - idx_i
            g[idx_i] += 2.0 * a[rev_idx]

        # multiplicative (exponentiated) update
        lr = 0.9 / np.sqrt(it)          # learning‑rate schedule
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # numerical stability
        a = np.exp(log_a)
        a /= a.sum()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Refinement – projected Adam (adaptive moments)
# ----------------------------------------------------------------------


def _refine_proj_adam(start_seq: np.ndarray,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with Euclidean projection onto the simplex after each step.
    Returns the best sequence seen before ``deadline``.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 1000 == 0:
                print(f"[Adam] it={t:7d} val={cur_val:.6f}")

        # gradient of the active convolution entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.15 / np.sqrt(t)          # a slightly larger base learning rate
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _proj_simplex(a_new)

        # occasional tiny random perturbation to avoid plateaus
        if t % 5000 == 0:
            noise = rng.normal(scale=1e-5, size=n)
            a = _proj_simplex(a + noise)

    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the

    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin for the final sanity check
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Lengths to explore – a couple of medium‑size vectors give a good

    # ------------------------------------------------------------------
    lengths = [4000, 8000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    for n in lengths:
        for maker in (
            _catalan_shape,

            except Exception:

            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={maker.__name__} val={val:.6f}")

        # power‑sine family
