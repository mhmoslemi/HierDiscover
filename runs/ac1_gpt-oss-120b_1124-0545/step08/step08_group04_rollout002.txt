"""
Propose a step‑function that minimizes ``evaluate_sequence``.

The search is performed in five time‑aware stages:

1. **Analytic seeds** – a collection of closed‑form families
   (uniform, Catalan, Hann, Hamming, Blackman, Gaussian, Beta‑shaped)
   is generated for several lengths; the best is kept.

2. **Random Dirichlet sampling** – many Dirichlet vectors with a variety
   of concentration parameters are drawn while time permits.

3. **Global optimisation with Differential Evolution** – the current
   best length is used to optimise a *mixture of Gaussians* on the simplex.
   The mixture is parametrised by non‑negative weights, means and
   standard deviations; the optimiser searches the 16‑dimensional space
   (5 Gaussians + uniform component).  This provides a powerful global
   search that can escape the local minima reached by the previous phases.

4. **Smooth Adam refinement** – starting from the best vector obtained so
   far we minimise a smooth approximation of ``max(conv(a,a))`` using a
   log‑sum‑exp surrogate and Adam on the soft‑max (log‑its) parametrisation.

5. **Exact mirror‑descent polishing** – a few thousand iterations of
   exponentiated‑gradient (KL‑mirror descent) on the true sub‑gradient of
   the active convolution entry are performed for a final squeeze.

All phases respect the overall time budget (default 1000 s) and the
function returns a plain Python list of non‑negative heights.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ---------------------------------------------------------------
# Low‑level convolution utilities
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a normalised (sum = 1) vector:
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ---------------------------------------------------------------
# Analytic seed families (all returned normalised to sum = 1)
# ---------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * math.pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * math.pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * math.pi * i / (n - 1))
         + a2 * np.cos(4 * math.pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred in the middle, σ = sigma_factor·n."""
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total


def _beta_shape(n: int, p: float = 0.30) -> np.ndarray:
    """
    Discrete Beta‑type shape proportional to ((i+1)*(n-i))^{-p}.
    For p>0 the ends are singular, which often reduces the peak of the
    self‑convolution.
    """
    i = np.arange(n, dtype=np.float64)
    # avoid exact zero when i=0 or i=n-1 by adding 1 to each factor
    w = ((i + 1) * (n - i)) ** (-p)
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total


# ---------------------------------------------------------------
# Random Dirichlet sampling
# ---------------------------------------------------------------

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """
    Draw a Dirichlet(alpha,…,alpha) vector.
    For alpha=1 we use exponentials; for other alphas we use Gamma.
    """
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ---------------------------------------------------------------
# Smoothed Adam optimiser (log‑its parametrisation)
# ---------------------------------------------------------------

def _smooth_adam_optimize(start_seq: np.ndarray,
                          deadline: float,
                          beta_start: float = 5.0,
                          beta_end: float = 50.0,
                          max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of ``max(conv(a,a))``.
    * a is represented by logits -> softmax(a).
    * The smooth max uses a log‑sum‑exp with an annealed temperature β.
    * Adam with decaying step size is employed.
    Returns the best sequence seen and its exact objective value.
    """
    n = len(start_seq)
    eps = 1e-30

    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999
    lr0 = 0.02

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        a = np.exp(logits)
        a /= a.sum()

        conv = _conv_fft(a, a)

        # linear annealing of β
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.5))

        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()

        # gradient of the smoothed max w.r.t. a
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} beta={beta:.1f}")
            if best_val < 1.50:
                break

    return best_seq, best_val


# ---------------------------------------------------------------
# Exact sub‑gradient mirror descent (exponentiated gradient)
# ---------------------------------------------------------------

def _refine_subgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Classical KL‑mirror descent on the simplex using the true sub‑gradient
    of the active convolution entry.
    Returns the best vector seen and its objective.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Refine] it={it:7d} n={n:5d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.7 / math.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ---------------------------------------------------------------
# Mixture of Gaussians (global DE search)
# ---------------------------------------------------------------

def _build_mixture(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a probability vector on the simplex from a parameter vector.
    The model consists of K Gaussian components plus an optional uniform
    component.  ``params`` layout (per component): weight, mean, sigma,
    followed by a final uniform_weight ∈ [0,1].
    """
    K = (len(params) - 1) // 3
    uniform_weight = float(params[-1])
    raw_weights = np.maximum(params[0:3 * K:3], 1e-12)
    mus = np.clip(params[1:3 * K:3], 0.0, 1.0)
    sigmas = np.clip(params[2:3 * K:3], 0.01, 0.5)

    weights = raw_weights / raw_weights.sum()

    x = np.arange(n, dtype=np.float64) / (n - 1)  # domain [0,1]
    seq = np.zeros(n, dtype=np.float64)
    for w, mu, sigma in zip(weights, mus, sigmas):
        seq += w * np.exp(-0.5 * ((x - mu) / sigma) ** 2)

    # add the uniform component
    seq = uniform_weight * (1.0 / n) + (1.0 - uniform_weight) * seq
    seq = np.maximum(seq, 0.0)

    total = seq.sum()
    if total == 0.0:
        return _uniform(n)
    return seq / total


class DeObjective:
    """Callable wrapper used by scipy.optimize.differential_evolution."""
    def __init__(self, n: int):
        self.n = n

    def __call__(self, params: np.ndarray) -> float:
        seq = _build_mixture(params, self.n)
        return _objective(seq)


def _de_optimize(n: int, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Global optimisation of a Gaussian mixture by Differential Evolution.
    The time budget is respected by limiting the number of iterations.
    """
    from scipy.optimize import differential_evolution

    K = 5                       # number of Gaussian components
    bounds = []
    for _ in range(K):
        bounds.append((0.0, 1.0))      # weight
        bounds.append((0.0, 1.0))      # mean
        bounds.append((0.01, 0.5))     # sigma
    bounds.append((0.0, 1.0))          # uniform weight

    # a modest number of iterations – we adapt it to the remaining time
    remaining = deadline - time.time()
    # Rough estimate: a single DE iteration (population sweep) takes
    # O(popsize * dim) cheap evaluations; we keep it to ≤ 80 iters.
    max_iter = 80
    popsize = 12

    obj = DeObjective(n)
    result = differential_evolution(
        obj,
        bounds,
        maxiter=max_iter,
        popsize=popsize,
        polish=False,
        init='latinhypercube',
        seed=int(rng.integers(2**32 - 1)),
        updating='deferred',
        # ``workers`` is left at the default (single‑process) to avoid
        # pickling issues in the execution sandbox.
    )
    best_seq = _build_mixture(result.x, n)
    best_val = _objective(best_seq)
    print(f"[DE] n={n:5d} best_val={best_val:.6f}")
    return best_seq, best_val

                print(f"[Seed] n={n:5d} type=Previous  val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – massive random Dirichlet sampling (time‑aware)
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.35   # ~35 % of remaining budget
    sample_deadline = now + time_for_sampling
    alphas = [0.2, 0.5, 1.0, 2.0]

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            best_n = n
            print(f"[Random] n={n:5d} α={alpha:.2f} val={val:.6f}")
        if best_val < 1.50:
            break

    # ------------------------------------------------------------------
    # Stage 2 – Global optimisation with Differential Evolution
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # allocate up to ~30 % of the remaining time to DE
        de_deadline = time.time() + (deadline - time.time()) * 0.30
        seq_de, val_de = _de_optimize(best_n, de_deadline, rng)
        if val_de < best_val:
            best_val = val_de
            best_seq = seq_de
            print(f"[Update] after DE best_val={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Smoothed Adam optimisation (fine‑grained local search)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        opt_deadline = time.time() + (deadline - time.time()) * 0.20
        seq_adam, val_adam = _smooth_adam_optimize(best_seq, opt_deadline)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam
            print(f"[Adam‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Exact sub‑gradient mirror descent (final polish)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        final_seq, final_val = _refine_subgradient(best_seq, deadline)
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Final‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check (the grader will call evaluate_sequence again)
    # ------------------------------------------------------------------
    try:
        # evaluate_sequence is supplied by the grader – this call merely
        # guards against accidental NaNs / infinities.
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print("\n=== Finished ===")
    print(f"Best objective ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
