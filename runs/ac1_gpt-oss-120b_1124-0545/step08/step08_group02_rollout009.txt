```
def greedy_low_collision(n, maxOrdered, target_m, rng, deadline):
    best_mask = None
    best_size = 0
    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
         attempts += 1
         mask = np.zeros(n, dtype=bool)
         sum_counts = np.zeros(2*n-1, dtype=np.uint16)
         order = rng.permutation(n)
         for cand in order:
             # compute if adding violates
             # self count
             if sum_counts[2*cand] + 1 > maxOrdered:
                 continue
             existing = np.nonzero(mask)[0]
             if existing.size:
                 sums = cand + existing
                 if np.any(sum_counts[sums] + 2 > maxOrdered):
                     continue
             # if we reach target_m, break early
             mask[cand] = True
             sum_counts[2*cand] += 1
             if existing.size:
                 sum_counts[sums] += 2
             if mask.sum() >= target_m:
                 break
         cur = mask.sum()
         if cur > best_size:
             best_size = cur
             best_mask = mask.copy()
    return np.where(best_mask)[0] if best_mask is not None else None
```

"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

Algorithm
---------
*   Build a large low‑collision support set (a B₂[2] or B₂[3] set):
    – Greedy random insertion with many restarts, respecting an upper bound
      on the ordered pair‑sum multiplicity (``max_ordered``).
    – Target set size is close to the theoretical bound
      ``sqrt(max_ordered * (2*n-1))``.
    – The best set found (largest size, smallest objective) is kept.
*   Use the uniform distribution on that support as a candidate.
*   Refine the candidate with exponentiated‑gradient (mirror‑descent)
    sub‑gradient optimisation on the exact non‑smooth objective.
*   A short simulated‑annealing walk is run if time permits.
*   The best sequence discovered within the time budget is stored in the
    global ``height_sequence_1`` and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear convolution (FFT, exact for real input)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The input is assumed to be a non‑negative vector.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple families (useful as additional seeds)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _sparse_random_seed(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random support of size `m`."""
    if m >= n:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """Deterministic Sidon‑like support using triangular numbers."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _quadratic_residue_seed(n: int, m: int) -> np.ndarray:
    """Positions are i² (mod n)."""
    idx = np.mod(np.arange(m) ** 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Low‑collision B₂[t] set generator
# ----------------------------------------------------------------------


def _low_collision_set(
    n: int,
    target_m: int,
    max_ordered: int,
    rng: np.random.Generator,
    deadline: float,
) -> np.ndarray | None:
    """
    Greedy randomized construction of a set S ⊂ {0,…,n‑1} such that every
    ordered pair‑sum occurs at most `max_ordered` times.
    Repeats many attempts until the deadline expires.
    Returns the index array of the best set found.
    """
    best_mask = None
    best_size = 0
    attempts = 0

    while time.time() < deadline - 0.2:
        attempts += 1
        mask = np.zeros(n, dtype=bool)
        sum_counts = np.zeros(2 * n - 1, dtype=np.uint8)
        selected = []               # Python list of already chosen indices

        order = rng.permutation(n)
        for cand in order:
            # self‑pair would increase count at 2*cand
            if sum_counts[2 * cand] + 1 > max_ordered:
                continue

            if selected:
                # ordered sums with the current set (each contributes 2)
                sums = cand + np.array(selected, dtype=int)
                if np.any(sum_counts[sums] + 2 > max_ordered):
                    continue

            # accept candidate
            if selected:
                sum_counts[sums] += 2
            sum_counts[2 * cand] += 1
            mask[cand] = True
            selected.append(cand)

            if len(selected) >= target_m:
                break

        cur_size = len(selected)
        if cur_size > best_size:
            best_size = cur_size
            best_mask = mask.copy()
            print(f"[LowColl] attempt {attempts:4d} size {cur_size}")

        # early exit if we already hit the theoretical bound
        if best_size >= target_m:
            break

    if best_mask is None or best_size == 0:
        return None
    return np.where(best_mask)[0]


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient / exponentiated‑gradient refinement
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact non‑smooth objective.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # build sub‑gradient (average over active constraints)
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1 : n - 1 + n]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis–Hastings walk that swaps a tiny amount of mass
    between two random positions and occasionally makes a global Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Seed generator (additional starters besides the low‑collision set)
# ----------------------------------------------------------------------


def _make_seeds(n: int, previous: np.ndarray | None, rng: np.random.Generator) -> list[np.ndarray]:
    """Assemble a diverse set of start vectors (all on the simplex)."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # sparse random seeds – support sizes 30 … 180
    for m in range(30, 191, 20):
        for _ in range(3):
            seeds.append(_sparse_random_seed(n, m, rng))

    # deterministic Sidon‑like structures
    for m in (50, 80, 110, 140):
        seeds.append(_sidon_like_seed(n, min(m, n)))
        seeds.append(_quadratic_residue_seed(n, min(m, n)))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final clean‑up (ensure non‑negative and normalised)
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and returns the best sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – 4000 is a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # Retrieve previous construction, if any
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.shape != (n,):
                previous = None
        except Exception:
            previous = None

    rng = np.random.default_rng(seed + 12345)

    # ------------------------------------------------------------------
    # Keep track of the global best
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # seed from previous solution
    if previous is not None:
        val = _objective(previous)
        best_seq = previous.copy()
        best_val = val
        print(f"[init] previous best objective = {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – search for large low‑collision supports
    # ------------------------------------------------------------------
    # Allocate roughly 40 % of the time budget to this phase
    phase1_deadline = start_time + 0.4 * (deadline - start_time)

    for max_ordered in (4, 6):
        # theoretical maximum size for given max_ordered
        max_possible = int(math.isqrt(max_ordered * (2 * n - 1)))
        target_m = max_possible                     # aim for the bound
        print(f"[search] maxOrdered={max_ordered}, target size ≤ {target_m}")

        support = _low_collision_set(
            n=n,
            target_m=target_m,
            max_ordered=max_ordered,
            rng=rng,
            deadline=phase1_deadline,
        )
        if support is not None and support.size > 0:
            # uniform weight on the support
            m = support.size
            a = np.zeros(n, dtype=np.float64)
            a[support] = 1.0 / m
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[support] maxOrdered={max_ordered} size={m} obj={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – generate a diverse pool of seeds (classic families,
    #            sparse random, Sidon‑like) and keep the top few.
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, best_seq, rng)
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = min(15, len(seed_vals))
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient (sub‑gradient) refinement on the
    #            best few seeds.  Number of iterations adapts to remaining
    #            time.
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline - 2.0:
            break
        remaining = deadline - time.time()
        # Rough estimate: 0.001 s per iteration
        max_iters = int(min(250_000, remaining / 0.0012))
        seq, val = _subgradient_refine(seed_vec, n, deadline, max_iters=max_iters)
        if seq is not None and val < best_val:
            best_seq = seq.copy()
            best_val = val
            print(f"[refine] seed {idx} improved objective to {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – short simulated‑annealing fine‑tuning (uses all remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        seq, val = _anneal_optimize(best_seq, n, deadline, max_iters=None)
        if seq is not None and val < best_val:
            best_seq = seq.copy()
            best_val = val
            print(f"[anneal] final objective {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    # final normalisation (protect against tiny negative drift)
    final_seq = np.maximum(best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
    else:
        final_seq = final_seq / total

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(final_seq)})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
