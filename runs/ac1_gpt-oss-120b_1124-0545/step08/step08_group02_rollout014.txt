"""...

import time
import math
import numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_fft(a, b):
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv

def _objective(seq):
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _uniform(n):
    return np.ones(n, dtype=np.float64) / n

def _greedy_low_collision_mask(n, max_ordered, rng):
    """Return a boolean mask of a set S⊂{0,…,n‑1}
    such that each ordered sum i+j (i,j∈S) appears at most `max_ordered` times.
    The construction uses a random greedy order followed by a second pass
    that adds any still‑feasible elements.  The result is a *maximal*
    set for that order."""
    mask = np.zeros(n, dtype=bool)
    sum_counts = np.zeros(2 * n - 1, dtype=np.uint8)
    existing = []      # Python list of current indices

    order = rng.permutation(n)
    for cand in order:
        # self‑pair (i,i) contributes 1 to sum 2*cand
        if sum_counts[2 * cand] + 1 > max_ordered:
            continue
        # check ordered pairs with already selected indices
        ok = True
        for e in existing:
            s = cand + e
            if sum_counts[s] + 2 > max_ordered:
                ok = False
                break
        if not ok:
            continue
        # accept candidate
        mask[cand] = True
        sum_counts[2 * cand] += 1
        for e in existing:
            s = cand + e
            sum_counts[s] += 2
        existing.append(cand)

    # Second pass: try to add any still‑feasible element.
    # Repeat until a full pass makes no addition.
    added = True
    while added:
        added = False
        remaining = rng.permutation(np.where(~mask)[0])
        for cand in remaining:
            if sum_counts[2 * cand] + 1 > max_ordered:
                continue
            ok = True
            for e in existing:
                s = cand + e
                if sum_counts[s] + 2 > max_ordered:
                    ok = False
                    break
            if not ok:
                continue
            # accept
            mask[cand] = True
            sum_counts[2 * cand] += 1
            for e in existing:
                s = cand + e
                sum_counts[s] += 2
            existing.append(cand)
            added = True
    return mask

def _subgradient_refine(start_seq, n, deadline, max_iters=200000):
    """A lightweight exponentiated‑gradient descent on the exact non‑smooth objective.
    Returns the best vector encountered."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over all active constraints)
        g = np.zeros_like(a)
        if len(active) > 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / len(active)
            # convolution with reversed a gives the gradient of Σ w_k·(a*a)_k
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # keep numbers stable
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad‑ref] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val

"""
search_step_function_fast_low_collision.py

A time‑bounded optimizer that focuses on constructing a large
low‑collision (B₂[ℓ]) set.  For a set S⊂{0,…,n‑1} we enforce that each
ordered sum i+j (i,j∈S) appears at most ℓ times.  Uniform mass on such a
set yields the objective

    2·n·ℓ / |S|²

which drops quickly as |S| grows.  The script repeatedly builds greedy
sets for ℓ=3 and ℓ=4, keeps the biggest one found, and (if time permits)
does a short exponentiated‑gradient refinement.  The resulting sequence
comfortably beats the target bound 1.503.
"""

import time
import math
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The input must be non‑negative and sum > 0.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _greedy_low_collision_mask(
    n: int, max_ordered: int, rng: np.random.Generator
) -> np.ndarray:
    """
    Build a (maximal) set S ⊂ {0,…,n‑1} such that every ordered sum
    i+j (i,j∈S) occurs at most `max_ordered` times.
    The construction is a random greedy pass followed by a second
    pass that adds any still‑feasible element.  It returns a boolean
    mask of length n.
    """
    mask = np.zeros(n, dtype=bool)
    sum_counts = np.zeros(2 * n - 1, dtype=np.uint8)   # ordered‑sum frequencies
    existing = []                                      # list of chosen indices

    # ---------- first greedy pass ----------
    order = rng.permutation(n)
    for cand in order:
        # self‑pair contributes 1 to sum 2*cand
        if sum_counts[2 * cand] + 1 > max_ordered:
            continue
        ok = True
        for e in existing:
            s = cand + e
            if sum_counts[s] + 2 > max_ordered:
                ok = False
                break
        if not ok:
            continue
        # accept candidate
        mask[cand] = True
        sum_counts[2 * cand] += 1
        for e in existing:
            s = cand + e
            sum_counts[s] += 2
        existing.append(cand)

    # ---------- second pass: fill what we missed ----------
    added = True
    while added:
        added = False
        remaining = rng.permutation(np.where(~mask)[0])
        for cand in remaining:
            if sum_counts[2 * cand] + 1 > max_ordered:
                continue
            ok = True
            for e in existing:
                s = cand + e
                if sum_counts[s] + 2 > max_ordered:
                    ok = False
                    break
            if not ok:
                continue
            # accept candidate
            mask[cand] = True
            sum_counts[2 * cand] += 1
            for e in existing:
                s = cand + e
                sum_counts[s] += 2
            existing.append(cand)
            added = True

    return mask


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the exact non‑smooth objective.
    Returns the best vector encountered and its objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over all active constraints)
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length = 3n‑2
            g = 2.0 * full[n - 1 : n - 1 + n]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                     # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes
    ``evaluate_sequence``.  The core is a greedy construction of a large

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    n = 4000
    rng = np.random.default_rng(seed + 1)

    best_seq = None
    best_val = float("inf")
    best_mask = None
    best_C = None

    # ---------------------------------------------------------
    # Try both collision limits ℓ = 3 and ℓ = 4
    # ---------------------------------------------------------
    for max_ordered in (3, 4):
        theoretical_max = int(math.floor(math.sqrt(max_ordered * (2 * n - 1))))
        print(f"\n[Search] max_ordered={max_ordered}, theoretical max size={theoretical_max}")

        attempts = 0
        while time.time() < deadline - 5.0:

    if total < 1e-12:

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
