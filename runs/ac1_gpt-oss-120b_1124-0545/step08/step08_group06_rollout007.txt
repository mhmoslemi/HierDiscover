"""
step_sequence_optimizer.py

A simple yet powerful optimizer for the ``evaluate_sequence`` objective.
The main idea is an alternating linear‑program / mirror‑descent scheme:

1. **Linear‑program step** – given a current distribution *f*,
   solve a linear program that maximises the total mass of a new
   distribution *g* while keeping the convolution ``f * g`` below the
   current worst value ``M = max(f*f)``.  Because the constraints are
   linear in *g* this sub‑problem is cheap (a single LP).

2. **Replacement** – if the LP produces a *g* that yields a smaller
   worst convolution value then we replace *f* by the normalised *g*.
   Repeating this few times drives the sequence towards a shape with a
   flatter self‑convolution.

3. **Fine‑grained refinement** – after the LP loop we apply a few
   iterations of exponentiated‑gradient (mirror descent) to polish the
   solution.

The whole routine respects the supplied time budget (default 1000 s) and
always returns a non‑negative list whose sum is positive, as required by
the grader.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for the sizes we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Objective that ``evaluate_sequence`` would return when ``seq`` is already
    normalised to sum = 1:  2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (used as a deterministic starting point)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Coefficients proportional to binom(2i,i)/4**i (the series of (1‑z)^‑½).
    After truncation they give a near‑constant auto‑convolution.
    Normalised to sum = 1.
    """
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


# ----------------------------------------------------------------------
# Exponentiated‑gradient refinement (mirrors the previous solution)
# ----------------------------------------------------------------------


def _refine_expgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using the negative entropy (exponentiated gradient).
    Returns the best vector seen before the deadline.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ExpGrad] it={it:7d} val={cur_val:.6f}")

        # pick a random index among the (almost) maximal entries
        tol = 1e-12
        max_val = conv.max()
        tie_idxs = np.where(conv >= max_val - tol)[0]
        max_idx = int(np.random.choice(tie_idxs))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (multiplicative) update
        lr = 0.8 / np.sqrt(it)  # diminishing step size
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)  # improve numerical stability
        a = np.exp(log_a)
        a /= a.sum()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑program sub‑step
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that (A @ g) = np.convolve(f, g).
    The matrix is very sparse – each row contains at most n non‑zeros.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        j_vals = k - i_vals                     # the index of g
        rows.extend([k] * len(i_vals))
        cols.extend(j_vals.tolist())
        data.extend(f[i_vals].tolist())
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n), dtype=np.float64)


def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        max 1ᵀ g
        s.t. A @ g ≤ M·1 , g ≥ 0
    where A is the convolution matrix for ``f``.
    Returns a normalised solution (sum(g)=1) or ``None`` on failure.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # linprog minimises → maximise sum(g)
    bounds = [(0, None)] * n

    # ``highs`` is fast and handles sparse matrices natively.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        return None

    g = res.x
    # Numerical noise may leave tiny negatives – clip them.
    g = np.maximum(g, 0.0)
    s = g.sum()
    if s <= 0.0:
        return None
    return g / s
