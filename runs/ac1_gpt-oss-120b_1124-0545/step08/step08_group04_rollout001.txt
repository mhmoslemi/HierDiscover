def simulated_annealing(seq, deadline):
    a = seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = objective(a)
    # Represent as logits = log(a)
    logits = np.log(a + 1e-16)
    temp0 = 0.5  # initial temperature for logit perturbations scale
    it = 0
    while time.time() < deadline:
        it += 1
        # propose new logits with Gaussian noise
        step = temp0 * np.exp(-it / (deadline - start_time) * 5)  # decays
        proposal_logits = logits + np.random.normal(scale=step, size=n)
        # softmax
        proposal = np.exp(proposal_logits)
        proposal = proposal / proposal.sum()
        val = objective(proposal)
        if val < best_val or np.random.rand() < np.exp(-(val - best_val)/ (temp0 * max(1e-9, (deadline - time.time())))):
            logits = proposal_logits
            a = proposal
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[SA] it={it} val={val:.6f}")
    return best_seq, best_val

def coordinate_descent(seq, deadline):
    a = seq.copy()
    n = len(a)
    best_val = objective(a)
    best_seq = a.copy()
    rng = np.random.default_rng()
    step_size_initial = 0.01
    it=0
    while time.time()<deadline:
        it+=1
        i,j = rng.integers(0,n,2)
        if i==j: continue
        # compute permissible delta
        delta = min(step_size_initial, a[i])
        # propose moving delta from i to j
        a_pro = a.copy()
        a_pro[i] -= delta
        a_pro[j] += delta
        val = objective(a_pro)
        if val < best_val:
            best_val = val
            best_seq = a_pro.copy()
            a = a_pro
            print(f"[CD] it={it} val={val:.6f}")
        # adapt step size maybe decay
        if it % 1000 == 0:
            step_size_initial *= 0.9
    return best_seq, best_val

def _upsample(seq, new_n):
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0,1,old_n)
    x_new = np.linspace(0,1,new_n)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0)
    sm = up.sum()
    if sm == 0:
        return np.full(new_n, 1.0/new_n)
    return up/sm

def _smooth_adam_optimize(start_seq, deadline, rng):
    n = len(start_seq)
    best_seq = start_seq.copy()
    best_val = _objective(best_seq)
    # parameters
    max_total_iter = 5_000_000  # may be limited by time
    # We will run multiple restarts if time permits
    restart = 0
    while time.time() < deadline:
        # for each restart, random perturbation
        logits = np.log(best_seq + 1e-30) + rng.normal(scale=0.1, size=n)
        m = np.zeros_like(logits)
        v = np.zeros_like(logits)
        beta1, beta2 = 0.9, 0.999
        lr0 = 0.02
        it = 0
        # allocate a sub-deadline for this restart: maybe the remaining time divided by remaining restarts
        # We'll stop if total iterations exceed some limit or time limit near
        while time.time() < deadline:
            it += 1
            # compute a
            a = np.exp(logits)
            a_sum = a.sum()
            a = a / a_sum
            # compute conv
            conv = _conv_fft(a, a)
            # smooth max temperature schedule
            beta = 5.0 + (200.0 - 5.0) * min(1.0, it / (max_total_iter * 0.3))
            # log-sum-exp
            c = np.max(conv)
            exp_beta = np.exp(beta * (conv - c))
            p = exp_beta / exp_beta.sum()
            # gradient w.r.t a
            conv_pa = _conv_fft(p, a[::-1])
            g_a = 2.0 * conv_pa[n-1:n-1+n]
            dot = np.dot(g_a, a)
            grad_logits = a * (g_a - dot)
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_logits
            v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            lr = lr0 / math.sqrt(it)
            logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)
            # periodic evaluation
            if (it & 1023) == 0:
                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[Adam restart{restart}] it={it} val={val:.6f}")
                # early stop if time short
                if time.time() > deadline - 0.1:
                    break
            if it >= max_total_iter:
                break
        restart += 1
        # check if enough time remains for another restart
        if time.time() > deadline - 10:
            break
    return best_seq, best_val

"""
Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Analytic seeds** – a handful of classic windows (uniform, Catalan,
   Hann, Hamming, Blackman, Gaussian) are generated for several lengths.
2. **Random sampling** – Dirichlet vectors with many concentration
   parameters and sparse binary patterns are drawn while time permits.
3. **Smooth Adam optimisation** – the sequence is represented by soft‑max
   logits.  A smooth approximation of the max‑convolution (log‑sum‑exp
   with linearly annealed temperature) is minimised by Adam; several
   short restarts are performed.
4. **Simulated‑annealing refinement** – starting from the best vector,
   logit perturbations are accepted with a temperature schedule,
   allowing the optimizer to escape shallow local minima.

All phases respect a global time budget (default 1000 s).  The routine
returns the best sequence (as a plain list) it has found.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` when ``seq`` is already normalised
    to sum = 1:
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients proportional to binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * math.pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * math.pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * math.pi * i / (n - 1))
         + a2 * np.cos(4 * math.pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _gaussian_sigma015(n: int) -> np.ndarray:
    return _gaussian(n, sigma_factor=0.15)


def _gaussian_sigma025(n: int) -> np.ndarray:
    return _gaussian(n, sigma_factor=0.25)


# ----------------------------------------------------------------------
# Random generation utilities
# ----------------------------------------------------------------------


def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet(alpha,…,alpha) vector."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


def _random_sparse_binary(n: int, rng: np.random.Generator, sparsity: float) -> np.ndarray:
    """
    Generate a binary (0/1) vector with a given sparsity fraction,
    then normalise to sum = 1.
    """
    k = max(1, int(sparsity * n))
    idx = rng.choice(n, size=k, replace=False)
    seq = np.zeros(n, dtype=np.float64)
    seq[idx] = 1.0 / k
    return seq


# ----------------------------------------------------------------------
# Helper to up‑sample a vector to a new length (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    sm = up.sum()
    if sm == 0.0:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    return up / sm


# ----------------------------------------------------------------------
# Phase 3 – Smooth Adam optimisation (soft‑max parametrisation)
# ----------------------------------------------------------------------


def _smooth_adam_optimize(start_seq: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution with Adam.
    Multiple short restarts (each with a small random perturbation of the
    initial logits) are performed while time permits.
    """
    n = len(start_seq)
    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    # Hyper‑parameters (chosen to be reasonably aggressive)
    beta_start = 5.0
    beta_end = 150.0
    max_iter_per_restart = 150_000

    restart = 0
    while time.time() < deadline:
        # Initialise logits with a small random perturbation
        logits = np.log(best_seq + 1e-30) + rng.normal(scale=0.1, size=n)

        # Adam accumulator variables
        m = np.zeros_like(logits)
        v = np.zeros_like(logits)
        beta1, beta2 = 0.9, 0.999
        lr0 = 0.018

        it = 0
        while time.time() < deadline and it < max_iter_per_restart:
            it += 1

            # Soft‑max to obtain a probability vector a (sum = 1)
            a = np.exp(logits)
            a = a / a.sum()

            # Full self‑convolution
            conv = _conv_fft(a, a)

            # Linear annealing of the temperature β
            beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter_per_restart * 0.3))

            # Smooth max via log‑sum‑exp (stabilised)
            c = np.max(conv)
            exp_beta = np.exp(beta * (conv - c))
            p = exp_beta / exp_beta.sum()                     # soft‑weights

            # Gradient w.r.t. a : g_a = 2·(p * a)  (convolution form)
            conv_pa = _conv_fft(p, a[::-1])                   # length = 3n‑2
            g_a = 2.0 * conv_pa[n - 1:n - 1 + n]               # slice of length n

            # Chain rule through the soft‑max
            dot = np.dot(g_a, a)
            grad_logits = a * (g_a - dot)

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_logits
            v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = lr0 / math.sqrt(it)                         # 1/√t decay
            logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

            # Periodic exact evaluation
            if (it & 1023) == 0:  # every 1024 steps
                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[Adam restart{restart}] it={it:7d} n={n:5d} val={val:.6f}")

                # Early exit when we have a very good solution
                if best_val < 1.45:
                    return best_seq, best_val

        restart += 1
        # Leave a small safety margin for the next phases
        if time.time() > deadline - 5.0:
            break

    return best_seq, best_val

# Phase 4 – Simulated‑annealing refinement on the simplex
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple logit‑space simulated annealing.
    Accepts uphill moves with a temperature that decays linearly
    from 0.5 to 0.01.
    """
    n = len(start_seq)
    cur_a = start_seq.copy()
    cur_logits = np.log(cur_a + 1e-30)
    cur_val = _objective(cur_a)

    best_a = cur_a.copy()
    best_val = cur_val

    start_time = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        # linear temperature schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        temp = 0.5 * (1.0 - progress) + 0.01   # from 0.5 down to 0.01

        # propose a new point in logit space
        prop_logits = cur_logits + rng.normal(scale=temp, size=n)
        exp_logits = np.exp(prop_logits - np.max(prop_logits))
        prop_a = exp_logits / exp_logits.sum()
        prop_val = _objective(prop_a)

        if prop_val < cur_val:
            # deterministic improvement
            cur_logits, cur_a, cur_val = prop_logits, prop_a, prop_val
            if prop_val < best_val:
                best_val, best_a = prop_val, prop_a.copy()
                print(f"[SA] it={it:7d} n={n:5d} val={best_val:.6f}")
        else:
            # probabilistic uphill step
            delta = prop_val - cur_val
            accept_prob = math.exp(-delta / (temp + 1e-12))
            if rng.random() < accept_prob:
                cur_logits, cur_a, cur_val = prop_logits, prop_a, prop_val

    return best_a, best_val

            print(f"[Adam] improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – Simulated‑annealing polishing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(best_seq, deadline, rng)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa
            print(f"[SA] improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Final sanity check (the grader will invoke evaluate_sequence again)
    # --------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
