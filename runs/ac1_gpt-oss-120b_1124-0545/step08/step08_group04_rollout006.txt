def _refine_subgradient_multi(start_seq, deadline, tolerance=1e-12, lr0=0.7):
   a = start_seq.copy()
   best_seq = a.copy()
   best_val = _objective(a)
   n = len(a)
   eps = 1e-30
   it = 0
   while time.time() < deadline:
       conv = _conv_fft(a, a)
       max_conv = np.max(conv)
       cur_val = 2.0 * n * max_conv
       if cur_val < best_val:
           best_val = cur_val
           best_seq = a.copy()
       # Determine active indices within tolerance
       # absolute tolerance maybe (max_conv - conv) < 1e-12, but due to floating errors use relative
       # use mask
       mask = conv >= max_conv - 1e-12
       active_idx = np.where(mask)[0]
       # Compute subgradient as average of gradients for all active indices.
       g = np.zeros_like(a)
       for k in active_idx:
           i_start = max(0, k - (n - 1))
           i_end = min(n - 1, k)

           rev_idx = k - idx_i

"""
Optimising a non‑negative step function (a probability vector) for the
objective

    evaluate_sequence(a) = 2·n·max(conv(a,a))

where n = len(a) and conv is the ordinary linear convolution.
The objective is invariant to scaling, therefore we keep the vector
normalised to sum = 1 and minimise  max(conv).

The search consists of four time‑aware stages:

1. Deterministic seeds  – a few analytically defined families
   (uniform, Catalan, Hann, Hamming, Blackman, Gaussian, power‑law,
   exponential, arcsine‑like) are generated for several lengths.  The
   best of these seeds is stored.

2. Random exploration – Dirichlet samples with a wide range of
   concentration parameters as well as random power‑law / exponential
   shapes are drawn while time permits.

3. Smoothed Adam optimisation – starting from the current best vector
   the smooth approximation  log‑sum‑exp(β·conv)  of the max‑convolution
   is minimised.  The temperature β is annealed from 1 to 200,
   the learning‑rate follows a 1/√t schedule and Adam’s moments are
   used.  No premature early‑stop is applied; the optimisation runs
   until the allocated time budget is exhausted.

4. Multi‑peak mirror‑descent – a final polishing based on an exact
   sub‑gradient of the true max‑convolution.  All indices that are
   within 1e‑12 of the current maximum are treated as active constraints
   (instead of a single one) and an exponentiated‑gradient (KL‑mirror
   descent) step is taken.  This often squeezes the final objective a
   little further.

All phases respect the overall 1000 s budget.  The routine returns a
plain Python list; the grader will call ``evaluate_sequence`` again for
the final verification.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from math import pi, sqrt, log, exp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for the sizes we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a normalised probability
    vector (sum == 1).  Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (all returned normalised to sum == 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i  – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * pi * i / (n - 1))
         + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _powerlaw(n: int, p: float = 0.5) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (monotonically decreasing)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    return w / w.sum()


def _exponential(n: int, lam: float = 0.001) -> np.ndarray:
    """a_i ∝ exp(-lam·i)  (decreasing exponential)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * i)
    return w / w.sum()


def _arcsine_like(n: int) -> np.ndarray:
    """
    Discrete analogue of the arcsine density:
        w_i ∝ 1 / sqrt( (i+1) * (n-i) )
    symmetric, vanishing in the middle, peaking at the two ends.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    w = 1.0 / np.sqrt(i * (n - i + 1))
    return w / w.sum()


# ----------------------------------------------------------------------
# Random helpers
# ----------------------------------------------------------------------


def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Draw a Dirichlet(alpha,…,alpha) vector."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


def _random_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random exponent p ∈ (0.1,1.5) → power‑law shape."""
    p = rng.uniform(0.1, 1.5)
    return _powerlaw(n, p)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random decay λ ∈ (1e‑4,5e‑3) → exponential shape."""
    lam = rng.uniform(1e-4, 5e-3)
    return _exponential(n, lam)


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑its parametrisation)
# ----------------------------------------------------------------------


def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 1.0,
    beta_end: float = 200.0,
    lr0: float = 0.03,
    max_iter: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)) using Adam on the
    soft‑max (log‑its) parametrisation.

    * `beta_start` / `beta_end` – temperature of the log‑sum‑exp
      smoothing; it is annealed linearly over the whole iteration budget.
    * `lr0` – base learning rate (decays as 1/√t).
    """
    n = len(start_seq)
    eps = 1e-30

    # initialise logits from the starting probability vector
    logits = np.log(start_seq + eps)

    # Adam moments
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # probability vector a = softmax(logits)
        a = np.exp(logits)
        a_sum = a.sum()
        a = a / a_sum

        # full self‑convolution
        conv = _conv_fft(a, a)                      # length = 2n‑1

        # anneal temperature
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / max_iter)

        # smooth max via stable log‑sum‑exp
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()               # soft‑weights, sum = 1

        # gradient wrt a :  g_a = 2 * (p ⋆ a_rev)
        conv_pa = _conv_fft(p, a[::-1])             # length = 3n‑2
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]         # slice of length n

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule
        lr = lr0 / sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # periodic exact evaluation
        if (it & 1023) == 0:          # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} beta={beta:.1f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient mirror descent (multi‑peak version)
# ----------------------------------------------------------------------


def _refine_subgradient_multi(
    start_seq: np.ndarray,
    deadline: float,
    lr0: float = 0.7,
    eps: float = 1e-30,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent on the simplex using the true sub‑gradient of the
    max‑convolution.  All indices whose convolution value is within
    1e‑12 of the current maximum are treated as active.
    """
    a = start_seq.copy()
    n = len(a)

    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_conv = np.max(conv)
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Refine] it={it:7d} n={n:5d} best={cur_val:.6f}")

        # active set (all indices essentially tied for the max)
        active = np.where(conv >= max_conv - 1e-12)[0]

        # accumulate sub‑gradient over the active set
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += a[rev_idx]

        # average over the active constraints
        if active.size > 0:
            g = 2.0 * g / active.size
        else:
            g = 2.0 * g

        # exponentiated‑gradient step
        lr = lr0 / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise before exponentiation
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising
    ``evaluate_sequence`` under the given time budget.
    The function returns the best sequence found (as a plain list).
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # ------------------------------------------------------------------
    # Recover a previously computed solution (if any) for possible reuse
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to investigate – a mix of moderate and a slightly larger one
    # ------------------------------------------------------------------
    lengths = [1500, 2000, 2500, 3000, 3500, 4000, 5000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic analytical seeds
    # ------------------------------------------------------------------
    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.25),
        lambda n: _powerlaw(n, p=0.5),
        lambda n: _powerlaw(n, p=0.8),
        lambda n: _exponential(n, lam=0.001),
        _arcsine_like,
    )

    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – massive random Dirichlet / power‑law sampling
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.30   # 30 % of remaining budget
    sample_deadline = now + time_for_sampling
    alphas = [0.2, 0.5, 1.0, 2.0, 5.0]

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        # randomly decide which generator to use
        choice = rng.integers(0, 3)
        if choice == 0:
            alpha = rng.choice(alphas)
            seq = _random_dirichlet(n, rng, alpha=alpha)
        elif choice == 1:
            seq = _random_powerlaw(n, rng)
        else:
            seq = _random_exponential(n, rng)

        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] n={n:5d} val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Smoothed Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # allocate roughly 30 % of the *remaining* budget to Adam
        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.30

        # first run from the current best
        seq_adam, val_adam = _smooth_adam_optimize(
            best_seq, adam_deadline, beta_start=1.0, beta_end=200.0, lr0=0.03
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam
            print(f"[Adam‑Refine‑1] improved best to {best_val:.6f}")

        # a couple of short restarts with jittered starts (if time permits)
        restarts = 2
        for r in range(restarts):
            if time.time() >= adam_deadline:
                break
            jitter = rng.normal(scale=0.05, size=best_seq.shape)
            start = best_seq * np.exp(jitter)   # multiplicative perturbation
            start = np.clip(start, 0, None)
            start /= start.sum()
            seq_adam2, val_adam2 = _smooth_adam_optimize(
                start, adam_deadline, beta_start=1.0, beta_end=200.0, lr0=0.03
            )
            if val_adam2 < best_val:
                best_val = val_adam2
                best_seq = seq_adam2
                print(f"[Adam‑Refine‑{r+2}] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Exact multi‑peak sub‑gradient mirror descent (final polish)
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _refine_subgradient_multi(best_seq, deadline)
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Final‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will re‑evaluate this)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    # expose the solution for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
