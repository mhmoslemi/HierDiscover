def loss_and_grad(v):
    a = softmax(v)
    conv = _conv_fft(a, a)
    max_conv_val = conv.max()
    # smooth max:
    shift = max_conv_val
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0:
        smooth_max = shift
        w_norm = np.full_like(conv, 1.0/conv.size)  # arbitrary
    else:
        smooth_max = shift + np.log(w_sum) / beta
        w_norm = exp_term / w_sum
    # compute gradient wrt a
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w_norm)
    grad_a = 2.0 * full[n-1:n-1+n]  # shape n
    # chain rule through softmax: grad_v = a * (grad_a - np.dot(grad_a, a))
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    # compute loss: 2*n * smooth_max / (a.sum()**2)
    # a.sum() = 1

search_robust_lbfgs.py
======================

A time‑bounded optimizer for the step‑function lower‑bound problem.

Main ideas
----------
1.  Build a large, diverse pool of seed vectors (classical windows,
    random soft‑max, sparse supports, Sidon‑like constructions, and the
    previous best if any).
2.  Evaluate the seeds quickly and keep the few best.
3.  For each promising seed run a **multistage L‑BFGS** optimisation on a
    *smooth‑max* surrogate of the convolution peak.  The surrogate is a
    log‑sum‑exp with a temperature schedule β∈{0.5,1,2,5,10,20}.
    The variables are logits of a soft‑max, guaranteeing non‑negativity
    and a unit ℓ₁‑norm.
4.  Refine the best solution with a long exponentiated‑gradient
    (mirror‑descent) pass on the exact non‑smooth objective.
5.  Spend any remaining time in a lightweight simulated‑annealing
    walk to escape shallow minima.

final sequence is stored in the global variable ``height_sequence_1``

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` (ignoring the clipping
    of values to [0,1000] – our constructions stay well inside that range).
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:

    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()

    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()

    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()

    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))

def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:

    a = np.clip(a, 0.0, None)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 400000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact non‑smooth objective.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    w = np.zeros(2 * n - 1, dtype=np.float64)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # build the sub‑gradient (average over active constraints)
        g = np.zeros_like(a)
        if active.size:
            w.fill(0.0)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # keep numerics stable
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val

def _lbfgs_loss_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (loss, grad) for the smooth‑max objective.
    Variables `v` are logits; the actual probabilities are `a = softmax(v)`.
    The smooth max of `conv(a,a)` is
        max_smooth = max(conv) + (1/β)·log Σ exp(β·(conv−max(conv))).
    """
    a = _softmax(v)                     # lies on the simplex, sum(a)=1
    conv = _conv_fft(a, a)
    max_conv = conv.max()
    shift = max_conv                     # for numerical stability
    exp_term = np.exp(beta * (conv - shift))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        smooth_max = shift
        w_norm = np.full_like(conv, 1.0 / conv.size)
    else:
        smooth_max = shift + np.log(w_sum) / beta
        w_norm = exp_term / w_sum

    # gradient wrt a : 2 * (rev_a * w_norm)  (same as in the original code)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w_norm)
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain‑rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    loss = 2.0 * n * smooth_max       # denominator = sum(a)^2 = 1
    return loss, grad_v


def _lbfgs_optimize(seed: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule=None,
                    maxiter_per_stage: int = 200) -> tuple[np.ndarray, float]:
    """
    Multistage L‑BFGS on increasingly sharp smooth‑max surrogates.
    Returns the best vector found and its true objective.
    """
    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0]

    # Initialise logits from the seed (add a tiny epsilon to avoid log(0))
    v = np.log(seed + 1e-30)

    best_seq = seed.copy()
    best_val = _objective(seed)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # SciPy minimise – expects a function returning (fun, grad) if jac=True
        res = minimize(_lbfgs_loss_and_grad,
                       v,
                       args=(n, beta),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_stage,
                                'disp': False})
        v = res.x
        a = _softmax(v)
        val = _objective(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[L‑BFGS] β={beta:.2f} val={val:.6f}")

    return best_seq, best_val

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))
    for sigma_factor in np.linspace(0.08, 0.30, 7):
        seeds.append(_gaussian(n, sigma_factor))
    for alpha in np.linspace(0.55, 0.95, 9):
        seeds.append(_beta_window(n, alpha))
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # sparse random seeds – a family that often yields a smaller max‑conv
    for m in range(30, 210, 20):          # 30,50,…,190
        for _ in range(4):
            seeds.append(_sparse_random_seed(n, m, rng))

    # deterministic Sidon‑like structures
    for m in (50, 70, 90, 110):
        seeds.append(_sidon_like_seed(n, min(m, n)))
        seeds.append(_quadratic_residue_seed(n, min(m, n)))

    # reuse previous construction if present
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # clean‑up – enforce non‑negativity and renormalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget and returns the best
    sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Load a previous construction if the grader provided one
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    rng = np.random.default_rng(seed + 1)

    # ------------------------------------------------------------------
    # Problem size – 4000 is a good trade‑off between expressiveness and speed
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # 1) Build a rich seed pool
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous, rng)

    # ------------------------------------------------------------------
    # 2) Quick evaluation of seeds – keep the top few
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = seed_vals[0][0].copy()
    global_best_val = seed_vals[0][1]
    print(f"[seed] best initial={global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – L‑BFGS multistage smooth‑max optimisation
    # ------------------------------------------------------------------
    if SCIPY_AVAILABLE:
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _lbfgs_optimize(seed_vec, n, deadline,
                                       beta_schedule=[0.5, 1.0, 2.0, 5.0, 10.0, 20.0],
                                       maxiter_per_stage=150)
            if seq is not None and val < global_best_val:
                global_best_val, global_best_seq = val, seq.copy()
                print(f"[L‑BFGS] seed={idx} best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – long exponentiated‑gradient refinement (exact objective)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                       max_iters=400000)
        if seq is not None and val < global_best_val:
            global_best_val, global_best_seq = val, seq.copy()
            print(f"[SubGrad] after long run best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – final simulated‑annealing fine‑tuning (if time permits)
    # ------------------------------------------------------------------
    def _anneal_optimize(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int | None = None) -> tuple[np.ndarray, float]:
        """Light‑weight Metropolis‑Hastings walk."""
        rng_local = np.random.default_rng(seed + 1234)
        cur = start_seq.copy()
        cur_val = _objective(cur)
        best = cur.copy()
        best_val = cur_val

        T0 = max(1e-6, cur_val * 0.04)
        decay = 0.9995
        it = 0
        while (max_iters is None or it < max_iters) and time.time() < deadline:
            it += 1
            T = T0 * (decay ** it)

            i, j = rng_local.integers(0, n, size=2)
            while i == j:
                j = rng_local.integers(0, n)

            delta = rng_local.random() * 0.02 * cur[i]
            if delta > cur[i]:
                delta = cur[i]

            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            new_val = _objective(new_seq)

            if new_val < cur_val or rng_local.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur, cur_val = new_seq, new_val
                if new_val < best_val:
                    best_val, best = new_val, new_seq.copy()
                    print(f"[Anneal] it={it:6d} best={new_val:.6f}")

            # occasional global jump
            if it % 1000 == 0:
                noise = rng_local.normal(scale=0.001, size=n)
                jump = cur + noise
                jump = np.clip(jump, 0.0, None)
                if jump.sum() == 0.0:
                    jump = _uniform(n)
                else:
                    jump /= jump.sum()
                jump_val = _objective(jump)
                accept = jump_val < cur_val or rng_local.random() < math.exp(-(jump_val - cur_val) / (T + 1e-12))
                if accept:
                    cur, cur_val = jump, jump_val
                    if jump_val < best_val:
                        best_val, best = jump_val, jump.copy()
                        print(f"[Anneal] it={it:6d} (jump) best={jump_val:.6f}")

        return best, best_val

    if time.time() < deadline - 5.0:
        seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_val, global_best_seq = val, seq.copy()
            print(f"[Anneal final] best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    globals()["height_sequence_1"] = np.copy(global_best_seq)
    return global_best_seq.tolist()
