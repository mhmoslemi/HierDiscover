def _adam_optimize_mask(seed_vec, mask, n, deadline, max_iters=20000):
    v = np.log(seed_vec + 1e-30)  # logits
    # set masked-out logits to very negative
    v[~mask] = -1e9
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr0 = 0.7
    beta = 1.5
    beta_mul = 1.007
    best_seq = None
    best_val = float('inf')
    t = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        # enforce mask
        a[~mask] = 0.0
        sum_a = a.sum()
        if sum_a == 0:
            # reinitialize uniform over mask
            a = np.zeros_like(a)
            a[mask] = 1.0 / mask.sum()
        else:
            a /= sum_a
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[AdamMask] it={it} best={cur_val:.6f}")
        # smoothed max
        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0:
            continue
        w /= w_sum
        # gradient
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)  # length = 3n-2
        grad_a = 2.0 * full[n-1:n-1+n]
        # chain rule
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # zero out gradient for masked-out positions (keep them -inf)
        grad_v[~mask] = 0.0
        # Adam updates
        m1 = beta1 * m1 + (1 - beta1) * grad_v
        m2 = beta2 * m2 + (1 - beta2) * (grad_v**2)
        m1_hat = m1 / (1 - beta1**t)
        m2_hat = m2 / (1 - beta2**t)
        lr = lr0 / np.sqrt(t)
        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        # ensure masked-out logits remain low
        v[~mask] = -1e9
    return best_seq, best_val

"""
Propose a step‑function with a very low maximal self‑convolution.
The core idea is to build a large B₂[2] (low‑collision) support set,
i.e. a set S⊂{0,…,n‑1} such that every ordered pair‑sum i+j (i,j∈S)
appears at most four times.  For n=4000 such a set can easily have
|S|≈150–180, giving a theoretical bound

    2·n·max(conv(a,a)) / (∑a)² ≤ 2·n·4/|S|² = 32000/|S|²  < 1.503

if |S|≥147.  The algorithm works in three stages:

1. **Low‑collision set generation** – a fast greedy construction
   with many random restarts, followed by a few random swaps.
2. **Weight optimisation on the fixed support** – a short Adam
   run (log‑softmax parametrisation) that respects the support mask,
   plus an optional sub‑gradient mirror‑descent.
3. **Final normalisation** – the best vector found is projected back
   onto the probability simplex and returned.

All heavy computation (convolutions) is performed with an FFT‑based
routine for speed.  The whole search respects the supplied time budget
(≈1000 s) and never returns an invalid sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution (exact, real‑valued)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (∑seq)².
    The input is assumed to be a non‑negative vector.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _uniform(n: int) -> np.ndarray:
    """Uniform probability vector of length n."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# B₂[2] low‑collision set construction
# ----------------------------------------------------------------------


def _b2_2_set_generator(n: int,
                         target_m: int,
                         max_ordered: int,
                         rng: np.random.Generator,
                         deadline: float) -> np.ndarray | None:
    """
    Greedy + random‑swap construction of a B₂[2] set S⊂{0,…,n‑1}
    with |S| as large as possible (up to target_m).  The ordered
    pair‑sum count is kept ≤ max_ordered (normally 4).
    Returns the index array of the best set found.
    """
    best_mask = np.zeros(n, dtype=bool)
    best_sz = 0
    attempt = 0

    while time.time() < deadline - 0.5 and attempt < 2000:
        attempt += 1
        mask = np.zeros(n, dtype=bool)
        sum_counts = np.zeros(2 * n - 1, dtype=np.int16)

        # -------- greedy addition passes ----------
        added = True
        while added and time.time() < deadline - 0.5:
            added = False
            order = rng.permutation(n)
            for idx in order:
                if mask[idx]:
                    continue
                # self‑pair
                if sum_counts[2 * idx] + 1 > max_ordered:
                    continue
                # pairs with already chosen indices
                existing = np.where(mask)[0]
                if existing.size:
                    sums = idx + existing
                    if np.any(sum_counts[sums] + 2 > max_ordered):
                        continue
                # accept
                mask[idx] = True
                sum_counts[2 * idx] += 1
                if existing.size:
                    sum_counts[sums] += 2
                added = True
                if mask.sum() >= target_m:
                    break
            if mask.sum() >= target_m:
                break

        cur_sz = mask.sum()
        if cur_sz > best_sz:
            best_sz = cur_sz
            best_mask = mask.copy()
            print(f"[B2_2] attempt {attempt:4d} size {cur_sz}")

        # -------- random swaps to improve the set ----------
        # try up to 400 swaps per attempt
        for _ in range(400):
            if time.time() > deadline - 0.5:
                break
            absent = np.where(~mask)[0]
            present = np.where(mask)[0]
            if absent.size == 0:
                break
            i = rng.choice(absent)
            j = rng.choice(present)
            # tentative swap
            new_mask = mask.copy()
            new_mask[i] = True
            new_mask[j] = False
            # fast check via convolution of the indicator vector
            conv = np.convolve(new_mask.astype(np.int8), new_mask.astype(np.int8))
            if conv.max() <= max_ordered:
                mask = new_mask
                # after a successful swap try to add more elements
                # (continue the outer greedy loop)
                added = True
                break  # break swap loop and re‑enter greedy addition

        # after swaps, try one more greedy pass to squeeze more elements
        if added and time.time() < deadline - 0.5:
            continue   # restart greedy pass with the improved mask

    if best_sz == 0:
        return None
    return np.where(best_mask)[0]


# ----------------------------------------------------------------------
# Adam optimisation respecting a support mask
# ----------------------------------------------------------------------


def _adam_optimize_mask(seed: np.ndarray,
                        mask: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 25000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser of the smooth‑max surrogate of the max‑convolution.
    Works on the full vector but keeps entries outside *mask* at zero.
    """
    v = np.log(seed + 1e-30)            # logits
    v[~mask] = -1e9                     # lock out forbidden positions
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr0 = 0.7
    beta = 1.5
    beta_mul = 1.007

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        a[~mask] = 0.0
        s = a.sum()
        if s == 0.0:
            a = np.zeros_like(a)
            a[mask] = 1.0 / mask.sum()
        else:
            a /= s

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[AdamMask] it={it:5d} best={cur_val:.6f}")

        # ---- smooth‑max of the convolution ----
        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w_k·(a*a)_k ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ---- chain rule through soft‑max ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        grad_v[~mask] = 0.0                 # no updates off‑support

        # ---- Adam moments ----
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        v[~mask] = -1e9                     # re‑enforce the mask

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient (max‑only) optimisation on a fixed support
# ----------------------------------------------------------------------


def _subgradient_optimize_mask(start_seq: np.ndarray,
                               mask: np.ndarray,
                               deadline: float,
                               max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true non‑smooth objective,
    using the gradient of the *single* maximal convolution entry.
    The support is kept fixed.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # gradient of the active constraint (ordered sums)
        g = np.zeros_like(a)
        idx = np.arange(a.size)
        j = k_max - idx
        valid = (j >= 0) & (j < a.size)
        g[valid] = a[j[valid]]
        g = 2.0 * g

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a[~mask] = -1e9
        a = _softmax(log_a)
        a[~mask] = 0.0
        s = a.sum()
        if s == 0.0:
            a = np.zeros_like(a)
            a[mask] = 1.0 / mask.sum()
        else:
            a /= s

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGradMask] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple simulated annealing (no mask) – final polish
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Metropolis‑Hastings walk that moves a tiny amount of mass between two
    random positions and occasionally performs a global Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        if it % 1000 == 0:
            # occasional global jump
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and returns the best sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # small safety margin

    # problem size – 4000 balances quality and speed well
    n = 4000
    print(f"\n=== Optimisation start (n={n}) ===")

    rng = np.random.default_rng(seed + 1)

    # ------------------------------------------------------------------
    # 1️⃣ Build a large B₂[2] set (max ordered collisions ≤ 4)
    # ------------------------------------------------------------------
    target_m = 150
    max_ordered = 4
    support = _b2_2_set_generator(
        n=n,
        target_m=target_m,
        max_ordered=max_ordered,
        rng=rng,
        deadline=deadline
    )

    if support is None or support.size < 120:
        # fallback: use a uniform vector (guaranteed to be valid)
        print("[Fallback] low‑collision set too small – using uniform.")
        a = _uniform(n)
        mask = np.ones(n, dtype=bool)
    else:
        mask = np.zeros(n, dtype=bool)
        mask[support] = True
        a = np.zeros(n, dtype=np.float64)
        a[support] = 1.0 / support.size
        a /= a.sum()
        print(f"[Support] size={support.size} initial_obj={_objective(a):.6f}")

    # keep the best vector found so far
    best_seq = a.copy()
    best_val = _objective(a)

    # ------------------------------------------------------------------
    # 2️⃣ Adam optimisation on the fixed support
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        seq_adam, val_adam = _adam_optimize_mask(best_seq, mask, n, deadline,
                                                max_iters=25000)
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam
            best_val = val_adam
            print(f"[Overall] after Adam best={val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Exact sub‑gradient descent (max‑only) on the same support
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        seq_sub, val_sub = _subgradient_optimize_mask(best_seq, mask,
                                                      deadline,
                                                      max_iters=50000)
        if seq_sub is not None and val_sub < best_val:
            best_seq = seq_sub
            best_val = val_sub
            print(f"[Overall] after SubGrad best={val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final simulated‑annealing fine‑tuning (unconstrained)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        seq_ann, val_ann = _anneal_optimize(best_seq, n, deadline)
        if seq_ann is not None and val_ann < best_val:
            best_seq = seq_ann
            best_val = val_ann
            print(f"[Overall] after Anneal best={val_ann:.6f}")

    # ------------------------------------------------------------------
    # Normalise and store the result
    # ------------------------------------------------------------------
    final_seq = np.maximum(best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
    else:
        final_seq = final_seq / total

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(final_seq)})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
