"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.  The implementation follows several complementary
ideas:

1. **Analytic seeds** – a few deterministic families (uniform, Catalan,
   geometric, triangular) for several lengths.
2. **Dirichlet sampling** – massive random draws with various concentration
   parameters.
3. **Exponentiated‑gradient (mirror descent)** – the classic sub‑gradient
   method used in the reference solution.
4. **Smooth‑max gradient descent** – minimise a soft‑max approximation of the
   max‑convolution using projected gradient steps (β is increased
   gradually).
5. **Hill‑climbing perturbations** – tiny random mass transfers between
   coordinates to escape shallow plateaus.

All stages respect a global time budget (default 1000 s).  The best
sequence seen is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for the sizes we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _corr_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation (x ⋆ y)[i] = Σ_k x_k y_{k+i}.
    Returned length is len(x)+len(y)-1, with index (len(y)-1) corresponding
    to shift 0.  Implemented via FFT: ifft(FFT(x) * conj(FFT(y))).
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    corr = np.fft.irfft(fx * np.conj(fy), n=N)[:L]
    return corr


def _objective(seq: np.ndarray) -> float:
    """
    Value of ``evaluate_sequence`` when ``seq`` already satisfies
    ``sum(seq) == 1`` (the scaling factor cancels out).  It is
    ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i (truncated series of (1‑z)^‑½)."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2.0 * i) * (2.0 * i - 1.0)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _geometric_shape(n: int, r: float = 0.999) -> np.ndarray:
    """Truncated geometric distribution a_i ∝ r**i."""
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular hill (max at the centre)."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(idx - mid) / (mid + 1.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total

# Random Dirichlet sampling
# ----------------------------------------------------------------------


def _dirichlet(alpha: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector and normalise to sum = 1."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x ≥ 0, Σ x = 1} (O(n log n) algorithm)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Refinement – exponentiated gradient (mirror descent)
# ----------------------------------------------------------------------


def _refine_expgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with the negative‑entropy potential.
    Returns the best vector seen before ``deadline``.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ExpGrad] it={it:7d} val={cur_val:.6f}")

        # tie‑breaking: random choice among near‑maximal indices
        tol = 1e-12
        ties = np.where(conv >= max_val - tol)[0]
        k = int(np.random.choice(ties))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        # exponentiated‑gradient update
        lr = 0.8 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)            # numerical stabilisation
        a = np.exp(log_a)
        a /= a.sum()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Refinement – smooth‑max gradient descent
# ----------------------------------------------------------------------


def _refine_smoothmax(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of the max‑convolution:
        J_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k).

    The gradient of J_β is a weighted sum of the ordinary sub‑gradients.
    We increase β gradually (β = 0.5, 2, 8, 32, 128) to approach the true
    max.  Projected gradient descent on the simplex is used.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    # schedule of β values (softmax temperature)
    betas = [0.5, 2.0, 8.0, 32.0, 128.0]
    beta_idx = 0
    step = 0
    while time.time() < deadline and beta_idx < len(betas):
        beta = betas[beta_idx]
        # a modest number of inner steps per β; we stop earlier if time runs out
        inner_steps = 2000
        for _ in range(inner_steps):
            if time.time() >= deadline:
                break
            step += 1
            conv = _conv_fft(a, a)                     # length 2n‑1
            max_c = conv.max()
            # soft‑max weights (stable)
            w = np.exp(beta * (conv - max_c))
            sum_w = w.sum()
            # correlation w ⋆ a  (g_i = 2·Σ_k w_k·a_{k+i})
            corr_full = _corr_fft(w, a)                # length 3n‑2
            start = n - 1
            grad = 2.0 * corr_full[start:start + n] / sum_w

            # simple diminishing learning rate
            lr = 0.5 / np.sqrt(step)
            a_new = a - lr * grad
            a = _proj_simplex(a_new)

            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Smooth] β={beta:.1f} step={step} val={cur_val:.6f}")

        beta_idx += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# Refinement – simple hill‑climbing perturbations
# ----------------------------------------------------------------------


def _refine_hillclimb(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Tiny random mass transfers between two coordinates.
    Accepts only improving moves (deterministic hill‑climb).
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        # Gaussian perturbation with zero mean, small std
        delta = rng.normal(scale=1e-4)
        if a[i] - delta < 0.0 or a[j] + delta < 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        cur_val = _objective(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()
            a = a_new
            print(f"[Hill] it={it:7d} val={cur_val:.6f}")
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final sanity check
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Lengths to explore – a modest spread (larger lengths give finer asymptotics)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic seeds
    # ------------------------------------------------------------------
    seed_fns = (
        _catalan_shape,
        _uniform,
        lambda n: _geometric_shape(n, r=0.999),
        lambda n: _geometric_shape(n, r=0.995),
        _triangular_shape,
    )
    for n in lengths:
        for maker in seed_fns:
            try:
                seq = maker(n)
            except Exception:   # pragma: no cover
                continue
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={maker.__name__} val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – include a possible previous solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                prev_arr = prev_arr / prev_arr.sum()
                val = _objective(prev_arr)
                if val < best_val:
                    best_val = val
                    best_seq = prev_arr.copy()
                    print(f"[Seed] previous solution val={val:.6f}")
        except Exception:   # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # Stage 3 – massive Dirichlet sampling
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    # approx. 30 % of the remaining time for sampling
    sample_deadline = now + remaining * 0.30

    alphas = [0.1, 0.5, 1.0, 2.0, 5.0]
    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = float(rng.choice(alphas))
        seq = _dirichlet(alpha, n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.1f} val={val:.6f}")

        if best_val < 1.503:   # early stop if we have already reached target
            break

    # ------------------------------------------------------------------
    # Stage 4 – exponentiated‑gradient mirror descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + (deadline - time.time()) * 0.25
        seq_exp, val_exp = _refine_expgradient(best_seq, exp_deadline)
        if val_exp < best_val:
            best_val = val_exp
            best_seq = seq_exp
            print(f"[Overall‑Exp] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – smooth‑max gradient descent (β annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        smooth_deadline = time.time() + (deadline - time.time()) * 0.25
        seq_smooth, val_smooth = _refine_smoothmax(best_seq, smooth_deadline)
        if val_smooth < best_val:
            best_val = val_smooth
            best_seq = seq_smooth
            print(f"[Overall‑Smooth] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – hill‑climbing local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        seq_hill, val_hill = _refine_hillclimb(best_seq, hill_deadline)
        if val_hill < best_val:
            best_val = val_hill
            best_seq = seq_hill
            print(f"[Overall‑Hill] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    # Return as plain Python list as required by the grader
    return best_seq.tolist()
