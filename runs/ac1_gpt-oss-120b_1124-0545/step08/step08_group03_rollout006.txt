"""
Hybrid heuristic optimizer for the step‑function bound.

The routine builds a very diverse pool of initial vectors (classical windows,
heavy‑edge beta densities, high‑order power laws, exponential‑decay and
deterministic Sidon‑like patterns).  The best few seeds are refined with
three complementary phases:

1. **Adam on a smooth‑max surrogate** – the max‑convolution is approximated by
   a log‑sum‑exp (temperature β).  β is slowly increased so that the smooth
   surrogate tightens while Adam adaptively adjusts the logits of a simplex
   parametrisation.

2. **Exponentiated‑gradient (mirror‑descent) on the exact non‑smooth objective**.
   The sub‑gradient is the average of the gradients of all *active* convolution
   entries (those within a tiny tolerance of the current maximum).  A
   decreasing learning‑rate 0.9/√t together with the exponential map keeps the
   iterates on the probability simplex.

3. **Final annealing** – a Metropolis‑Hastings walk that repeatedly moves a
   small random amount of mass between two positions and occasionally makes a
   global Gaussian jitter.  This helps escaping shallow local minima.

If `cvxpy` is available a cheap “log‑sum‑exp” LP‑style refinement is inserted
between the two gradient phases – it solves a convex approximation of the
original problem and often yields an immediate objective drop.

All phases respect the supplied time budget (default 1000 s).  The best vector
found is stored in the global variable `height_sequence_1` and returned as a
plain list of Python floats.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact version of the public evaluate_sequence (without the 0‑1000 clipping,
    which never activates for our normalised vectors).
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1), discretised on mid‑points.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_heavy(n: int, alpha: float) -> np.ndarray:
    """
    Very heavy edge version – α ∈ (0,0.5] concentrates mass near the ends.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(‑lam·|i‑centre|)."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_power(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ i^(‑p) + (n‑i)^(‑p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p) + 1.0 / ((n - idx + 1) ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()

def _sparse_random_seed(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Uniform mass on a random support of size `m` (1 ≤ m ≤ n).
    """
    if m >= n:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """
    Deterministic “Sidon‑like” support using triangular numbers.
    """
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _quadratic_residue_seed(n: int, m: int) -> np.ndarray:
    """
    Positions i² (mod n).  For a prime n this set is a Sidon set;
    for composite n it still yields a fairly flat sum‑set.
    """
    idx = np.mod(np.arange(m) ** 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _log_sum_exp(x: np.ndarray, beta: float) -> float:
    """Smooth max: (1/β)·log Σ exp(β·x_i)."""
    mx = np.max(x)
    return mx + (1.0 / beta) * np.log(np.sum(np.exp(beta * (x - mx))))

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate (log‑sum‑exp) of max(conv(a,a)).
    Returns the best sequence seen and its true objective.
    """
    # Initialise logits (log‑softmax variables)
    v = np.log(seed + 1e-30)

    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # temperature schedule for the smooth‑max
    beta = 1.2          # start modest
    beta_mul = 1.008    # geometric increase → tighter approximation

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                     # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k (a*a)_k  (see derivation in the original script)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam moments
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 800000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the exact non‑smooth objective.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    w = np.zeros(2 * n - 1, dtype=np.float64)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over the active constraints)
        g = np.zeros_like(a)
        if active.size:
            w.fill(0.0)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # keep numerics stable
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _convex_refine(seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int = 200000) -> np.ndarray:
    """
    A lightweight convex surrogate using a log‑sum‑exp relaxation.
    For each lag k we enforce  log Σ_i exp(b_i + b_{k‑i}) ≤ t.
    The variables are b_i = log a_i (real) and the scalar t = log M.
    If cvxpy is present, we solve the problem and return the (renormalised)
    probabilities; otherwise we simply return the original sequence.
    """
    if not CVXPY_AVAILABLE:
        return seq

    # limit the problem size to keep it fast – we pick a subset of
    # the most relevant lags (the centre ± 0.4·n).  The extremes are already
    # small for any reasonable distribution.
    centre = n - 1
    radius = int(0.45 * n)
    ks = np.arange(centre - radius, centre + radius + 1)

    b = cp.Variable(n)
    t = cp.Variable()
    constraints = [cp.sum(cp.exp(b)) == 1.0]

    for k in ks:
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i = np.arange(i_low, i_high + 1)
        expr = b[i] + b[k - i]
        constraints.append(cp.log_sum_exp(expr) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS,
                   max_iters=8000,
                   abstol=1e-8,
                   reltol=1e-8,
                   feastol=1e-8)
    except Exception:  # pragma: no cover
        return seq

    if b.value is None:
        return seq

    a = np.exp(np.asarray(b.value).flatten())
    a_sum = a.sum()
    if a_sum == 0.0:
        return seq
    a /= a_sum
    return a


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk that repeatedly transfers a tiny
    amount of mass between two random positions and occasionally adds
    a global Gaussian perturbation.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)   # initial temperature
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move a modest fraction (up to 2 %) of the source mass
        delta = rng.random() * 0.02 * cur[i]
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        # every 1000 steps perform a global jitter
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= total
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

def _make_seeds(n: int,
                previous: np.ndarray | None,
                rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse pool of start vectors, all normalised to sum = 1."""
    seeds: list[np.ndarray] = []

    # Classical windows
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 5):
        seeds.append(_tukey_window(n, alpha))

    # Heavier edge‑biased Betas (α down to 0.1)
    for alpha in np.linspace(0.10, 0.50, 9):
        seeds.append(_beta_heavy(n, alpha))

    # Symmetric power laws and exponential decays
    for p in np.linspace(0.5, 3.0, 6):
        seeds.append(_sym_power(n, p))
    for lam in np.linspace(0.001, 0.02, 5):
        seeds.append(_exp_decay(n, lam))

    # Geometric / power mixtures
    for r in np.linspace(0.95, 0.999, 4):
        seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.5, 6):
        seeds.append(_power(n, p))
    for r in np.linspace(0.95, 0.999, 3):
        for p in np.linspace(0.0, 2.0, 4):
            seeds.append(_geom_power(n, r, p))

    # Random softmaxes
    for _ in range(8):
        seeds.append(_random_softmax(n, rng))

    # Sparse random supports – a family that often yields a lower max‑conv
    for m in range(30, 210, 30):          # 30,60,…,180
        for _ in range(3):
            seeds.append(_sparse_random_seed(n, m, rng))

    # Deterministic Sidon‑like structures
    for m in (50, 80, 110, 140):
        seeds.append(_sidon_like_seed(n, min(m, n)))
        seeds.append(_quadratic_residue_seed(n, min(m, n)))

    # Re‑use the previous construction if available
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # Guard against invalid entries
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget and returns the best sequence
    discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Load a previous construction if the grader supplied one
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    rng = np.random.default_rng(seed + 1)

    # ------------------------------------------------------------------
    # Problem size – 4000 is a good compromise
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # 1) Assemble a rich seed pool
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous, rng)

    # ------------------------------------------------------------------
    # 2) Quick evaluation – keep the top few the most promising seeds
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = seed_vals[0][0].copy()
    global_best_val = seed_vals[0][1]
    print(f"[seed] best initial={global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam on the smoothed‑max for each top seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=25000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Adam] seed={idx} best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – optional convex surrogate refinement (fast, cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline and CVXPY_AVAILABLE:
        seq = _convex_refine(global_best_seq, n, deadline)
        val = _objective(seq)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Convex] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – long exponentiated‑gradient refinement on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                       max_iters=900000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[SubGrad] after long run best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase D – perturbation + short sub‑gradient passes (helps escape)
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # a small random jitter around the current best
        perturb = global_best_seq.copy()
        noise = rng.normal(scale=0.0015, size=n)
        perturb += noise
        perturb = np.clip(perturb, 0.0, None)
        total = perturb.sum()
        if total == 0.0:
            perturb = _uniform(n)
        else:
            perturb /= total

        seq, val = _subgradient_refine(perturb, n, deadline,
                                       max_iters=250000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Perturb] new best={val:.6f}")

        if time.time() > deadline - 20.0:
            break

    # ------------------------------------------------------------------
    # Phase E – final simulated‑annealing fine‑tuning (uses any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Anneal final] best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event we have no candidate
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    globals()["height_sequence_1"] = np.copy(global_best_seq)
    return global_best_seq.tolist()

# Additional window helpers (used inside seed generation)
# ----------------------------------------------------------------------


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()
