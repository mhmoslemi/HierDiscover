"""
Hybrid optimiser for the step‑function attack (≈1000 s budget).

Goal
----
Find a non‑negative vector `a` (length *n*) minimising

    2·n·max(conv(a,a)) / (sum(a)**2)

The objective is 0‑homogeneous, therefore we work on the probability
simplex (∑a = 1).  The optimiser consists of three stages:

1. **Rich seed generation** – a collection of analytic families (Beta,
   geometric, windows, random Dirichlet, …) and a reuse of the previous
   construction.  The best few seeds are kept.

2. **Smooth‑gradient optimisation** – we minimise a smooth surrogate
   of ``max(conv)`` using the log‑sum‑exp approximation.
   The surrogate is
        Fλ(a) = (1/λ)·log(∑k exp(λ·(a*a)k))
   Its gradient can be written in closed form via a single
   FFT‑based correlation, so each iteration costs O(n log n).
   The parameters are the log‑softmax of the probabilities and we use an
   Adam optimiser with a schedule that slowly increases λ (smooth → exact).

3. **Simulated‑annealing fine‑tuning** – a Metropolis–Hastings walk that
   explores the neighbourhood of the best vector obtained above.  The
   walk uses several move types (mass transfer, swap, Gaussian jitter)
   and a simple exponential cooling schedule.

All stages respect the time budget; intermediate best vectors are printed
and the final one is stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution / correlation (real, power‑of‑two length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a ⊛ b (real FFT)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _corr_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation (w ⋆ a) of lengths (2·n‑1) and n.
    Returns a vector of length n:
        (w ⋆ a)[i] = Σ_k w[i+k]·a[k]
    Computed via FFT: (w ⋆ a) = ifft(conj(fft(w))·fft(a)).
    """
    L = w.size + a.size - 1
    N = _next_pow2(L)
    fw = np.fft.rfft(w, n=N)
    fa = np.fft.rfft(a, n=N)
    corr = np.fft.irfft(np.conj(fw) * fa, n=N)[:L]
    start = w.size - 1
    return corr[start:start + a.size]


# ----------------------------------------------------------------------
# Exact objective (for monitoring / final evaluation)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Analytic families (all return a *probability* vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) pdf on (0,1) discretised on mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _make_seed_pool(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # uniform
    seeds.append(_uniform(n))

    # beta‑pdfs (edge‑heavy to centre‑heavy)
    for a in np.linspace(0.25, 2.0, 16):
        seeds.append(_beta_window(n, a))

    # few Dirichlet draws (very sparse / very dense)
    for a in [0.05, 0.2, 0.8, 1.5, 3.0]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, a, rng))

    # random soft‑max vectors
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # reuse previous construction if it has the right length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final clean‑up – clip tiny negatives, renormalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    return cleaned


# ----------------------------------------------------------------------
# Smoothed‑gradient optimiser (Adam on the log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _smooth_gradient_optimize(start_seq: np.ndarray,
                              n: int,
                              deadline: float,
                              max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Minimises a smooth approximation of ``max(conv)``.
    Uses Adam on the log‑softmax parameters.
    Returns the best vector seen and its exact objective value.
    """
    # logarithmic parametrisation (ensures positivity and sum‑1 after exp‑norm)
    x = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # smoothing parameter λ – starts modest, grows slowly
    lam = 5.0
    lam_growth = 1.03          # multiplicative growth every ``lam_step`` iters
    lam_step = 4000

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for it in range(1, max_iters + 1):
        # time check
        if time.time() > deadline:
            break

        # probability vector from logits
        a = np.exp(x)
        a /= a.sum()

        # convolution and smooth weights
        conv = _conv_fft(a, a)
        max_c = conv.max()
        # shift by the maximum for numeric stability
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w_norm = w / w_sum

        # gradient w.r.t. a :  ∂F/∂a = Σ_k w_norm_k ∂conv_k/∂a
        # ∂conv_k/∂a_i = a_{k-i} + a_{i-k} → cross‑correlation
        grad_a = 2.0 * _corr_fft(w_norm, a)

        # chain rule: ∂F/∂x = a * (grad_a - <a,grad_a>)
        dot_ag = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot_ag)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (decays like 1/√t)
        lr = 0.12 / (math.sqrt(it) + 1e-12)

        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional exact evaluation
        if it % 2000 == 0:
            a = np.exp(x)
            a /= a.sum()
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SmoothGrad] it={it:7d} best={cur_val:.6f}")

        # increase λ slowly – makes the surrogate sharper over time
        if it % lam_step == 0:
            lam = min(lam * lam_growth, 300.0)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (identical to the previous implementation)
# ----------------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # temperature = 8 % of current objective, geometric cooling
    T0 = max(1e-7, cur_val * 0.08)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        mv = rng.random()
        if mv < 0.70:          # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = rng.random() * 0.20 * cur[i]
                new_seq = cur.copy()
                new_seq[i] -= delta
                new_seq[j] += delta
            else:
                continue
        elif mv < 0.85:        # swap two entries
            i, j = rng.integers(0, n, size=2)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:                 # global Gaussian jitter
            noise = rng.normal(scale=0.0015, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            s = new_seq.sum()
            new_seq = new_seq / s if s > 0 else _uniform(n)

        # keep on the simplex
        new_seq = np.maximum(new_seq, 0.0)
        new_seq /= new_seq.sum()

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:8d} best={new_val:.6f}")

        # periodic strong jump
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.003, size=n)
            cand = best + jitter
            cand = np.clip(cand, 0.0, None)
            cand /= cand.sum()
            cand_val = _objective(cand)
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
                print(f"[Anneal] (jump) it={it:8d} best={cand_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  Returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Length – 4000 balances speed and resolution
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Optional reuse of a previous construction
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev = np.asarray(prev, dtype=np.float64)
        except Exception:
            prev = None

    # ------------------------------------------------------------------
    # 0) evaluate previous construction (if any)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    if prev is not None and prev.shape == (n,):
        val = _objective(prev)
        best_seq = prev.copy()
        best_val = val
        print(f"[prev] val={val:.6f}")

    # ------------------------------------------------------------------
    # 1) Seed pool
    # ------------------------------------------------------------------
    seeds = _make_seed_pool(n, prev)
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    # keep top 8 seeds for the costly optimisation stages
    top_seeds = [s for s, _ in seed_vals[:8]]

    if seed_vals[0][1] < best_val:
        best_val = seed_vals[0][1]
        best_seq = seed_vals[0][0].copy()
        print(f"[seed] new best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Smooth‑gradient optimiser (Adam on log‑softmax)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        for s in top_seeds:
            seq, val = _smooth_gradient_optimize(s, n, deadline,
                                                 max_iters=250_000)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[overall SmoothGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # 3) Simulated‑annealing fine‑tuning (uses remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(best_seq, n, deadline,
                                    max_iters=None)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[overall Anneal] best={val:.6f}")

    # ------------------------------------------------------------------
    # 4) Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
