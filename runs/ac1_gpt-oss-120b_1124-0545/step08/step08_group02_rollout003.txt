"""
Propose a step‑function that beats the 1.503 target.

Algorithm
---------
1.  Fix a length `n = 4000`.  The objective for a probability vector `a`
    (sum = 1) is 
        obj(a) = 2·n·max_i (a * a)_i .
    For a uniform distribution on a support `S` of size `m`
    the convolution counts are integer:
        cnt_s = #{(i,j)∈S² : i+j=s}   (ordered pairs)
    and  (a*a)_s = cnt_s / m².
    Hence   obj = 2·n·max(cnt) / m².
    If we can find a set `S` with many elements and a small
    bound `L = max(cnt)`, the objective becomes
        obj ≤ 2·n·L / m².

2.  We search for a large *low‑collision* set `S ⊂ {0,…,n‑1}` such that
    every ordered sum occurs at most `L = 5` times (the B₂[5] property).
    For a random greedy construction the expected size is close to
    √(L·(2n‑1)) ≈ 200 for `n = 4000`.  A uniform distribution on such a
    set would give `obj ≤ 2·n·5 / 200² = 1.0`, comfortably below 1.503.

3.  The search consists of many independent greedy attempts (as many as
    time permits).  Each attempt processes a random permutation of the
    indices, accepting a candidate only if the ordered‑pair‑sum counter
    would stay ≤ `L`.  The best set (largest `m`, breaking ties by the
    resulting objective) is kept.

4.  After the greedy phase we try a cheap *completion* step: iterate over
    the still‑missing indices in random order and add any that keep the
    bound ≤ `L`.  This often enlarges the set a little.

5.  The final sequence is the uniform probability vector on the best set
    (`a_i = 1/m` for i∈S, else 0).  If any time remains we perform a very
    short Adam optimisation on a smooth‑max surrogate to polish the
    weights.

The whole routine respects the supplied time budget (default 1000 s) and
stores the best sequence in the global variable `height_sequence_1`.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Fast objective for a uniform distribution on a support set.
# ------------------------------------------------------------
def _uniform_obj(n: int, support_size: int, max_cnt: int) -> float:
    """Return the exact objective value for a uniform distribution
    on a support of size `support_size` whose worst ordered‑sum count is
    `max_cnt`."""
    if support_size == 0:
        return float("inf")
    return 2.0 * n * max_cnt / (support_size * support_size)


# ------------------------------------------------------------
# Greedy construction of a B₂[L] set (ordered sums ≤ L).
# ------------------------------------------------------------
def _greedy_b2_set(
    n: int,
    max_ordered: int,
    rng: np.random.Generator,
    deadline: float,
) -> np.ndarray | None:
    """Return a Boolean mask of a set built by a single greedy pass.
    If the deadline is exceeded during the construction, None is returned."""
    mask = np.zeros(n, dtype=bool)
    # ordered‑pair sum counter: length 2n‑1, values are small (≤max_ordered)
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)

    order = rng.permutation(n)
    for cand in order:
        if time.time() > deadline:
            return None

        # self‑pair (i,i) contributes 1 to sum 2*cand
        if sum_counts[2 * cand] + 1 > max_ordered:
            continue

        # existing indices already chosen
        existing = np.nonzero(mask)[0]
        if existing.size:
            sums = cand + existing          # vectorised
            # each new unordered pair adds 2 ordered contributions
            if np.any(sum_counts[sums] + 2 > max_ordered):
                continue

        # accept the candidate
        mask[cand] = True
        sum_counts[2 * cand] += 1
        if existing.size:
            sum_counts[sums] += 2

    return mask


# ------------------------------------------------------------
# Full search for the largest B₂[L] set under a time budget.
# ------------------------------------------------------------
def _search_best_low_collision_set(
    n: int,
    max_ordered: int,
    deadline: float,
) -> np.ndarray | None:

    Returns the index array of the best set found (largest size,
    breaking ties by the resulting uniform objective)."""
    rng = np.random.default_rng()
    best_mask = np.zeros(n, dtype=bool)
    best_size = 0
    best_obj = float("inf")
    attempt = 0

    # The theoretical maximum size (ignoring combinatorial subtleties)
    max_theoretical = int(math.floor(math.sqrt(max_ordered * (2 * n - 1))))

    while time.time() < deadline - 0.5:
        attempt += 1
        mask = _greedy_b2_set(n, max_ordered, rng, deadline)
        if mask is None:
            break
        cur_size = mask.sum()
        if cur_size == 0:
            continue

        # compute ordered‑pair‑sum counts via convolution (fast, integer)
        cnt = np.convolve(mask.astype(np.int16), mask.astype(np.int16))
        cur_max = int(cnt.max())
        cur_obj = _uniform_obj(n, cur_size, cur_max)

        if cur_size > best_size or (cur_size == best_size and cur_obj < best_obj):

            best_size = cur_size

    # --------------------------------------------------------
    cnt = np.convolve(best_mask.astype(np.int16), best_mask.astype(np.int16))
    missing = np.where(~best_mask)[0]
    rng.shuffle(missing)

        cnt[2 * cand] += 1
        best_mask[cand] = True
        best_size += 1

    print(f"[Final] best set size {best_size}, max_cnt {int(cnt.max())}")
    return np.where(best_mask)[0]


# ------------------------------------------------------------
# Tiny Adam optimizer on the smooth‑max surrogate.
# ------------------------------------------------------------
def _adam_optimize(
    x0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 8000,
) -> np.ndarray:
    """Runs a very short Adam optimisation, starting from `x0`
    (already on the simplex).  Returns the best vector seen."""
    # soft‑max parametrisation
    v = np.log(x0 + 1e-30)

    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr0 = 0.5

    # smooth‑max temperature schedule
    beta = 1.2
    beta_mul = 1.006

    best_seq = x0.copy()
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = np.exp(v - np.max(v))
        a /= a.sum()                     # simplex
        # concrete objective (exact)
        conv = np.convolve(a, a)
        cur_val = 2.0 * n * conv.max()   # sum(a)=1 ⇒ denominator = 1
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it {it:5d} best {cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of weighted sum Σ w_k (a*a)_k
        rev_a = a[::-1]
        full = np.convolve(rev_a, w)               # length 3n‑2
        grad = 2.0 * full[n - 1 : n - 1 + n]       # size n

        # ADAM
        m1 = beta1 * m1 + (1 - beta1) * grad
        m2 = beta2 * m2 + (1 - beta2) * (grad ** 2)

        m1_hat = m1 / (1 - beta1 ** it)
        m2_hat = m2 / (1 - beta2 ** it)

        lr = lr0 / math.sqrt(it)
        v -= lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq

        print(f"[Result] uniform on support: m={m}, max_cnt={max_cnt}, obj≈{approx_obj:.6f}")

    # --------------------------------------------------------
    # Phase 2 – optional short Adam polishing (if time permits)
    # --------------------------------------------------------
    if time.time() < deadline - 0.5:
        seq = _adam_optimize(seq, n, deadline, max_iters=15000)

    # final sanitisation
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total < 1e-12:
        seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        seq /= total

    # store globally for possible reuse by subsequent runs
    globals()["height_sequence_1"] = seq.copy()

    # final objective (using the already‑imported evaluate_sequence)
    try:
        final_val = evaluate_sequence(seq.tolist())
        print(f"\n=== Finished ===\nFinal objective ≈ {final_val:.6f}")
    except Exception:   # pragma: no cover
        print("\n=== Finished ===\n(Unable to evaluate final objective)")

    return seq.tolist()
