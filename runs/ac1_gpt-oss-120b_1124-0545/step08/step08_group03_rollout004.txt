"""
Improved step‑function optimizer.

The search proceeds in four stages:

1. **Broad seeding** – many deterministic windows (uniform, Hann, Blackman,
   Tukey, Gaussian …) together with a large batch of random Dirichlet
   vectors (different concentration parameters).  All seeds are projected
   onto the probability simplex and the best few are kept.

2. **Smooth‑max Adam refinement** – each top seed is optimised with Adam on
   a soft‑max approximation of the convolution supremum.  The temperature
   (β) is increased geometrically, giving a progressively sharper
   approximation.

3. **Exact sub‑gradient (mirror‑descent) refinement** – the non‑smooth
   objective `2·n·max(conv(a,a))/sum(a)^2` is minimised directly by
   exponentiated‑gradient steps using the true active set of constraints.
   This phase runs for as many iterations as time permits.

4. **Final annealing** – a Metropolis‑Hastings walk that combines the two
   elementary moves used in the original code (mass transfer) with a
   *swap* move (exchanging two entries).  A very slow exponential cooling
   schedule is used; any improvement updates the incumbent solution.

All phases respect the global time budget (default 1000 s).  The best
vector found is stored in the global variable ``height_sequence_2`` and
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (exact, real‑valued)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by `evaluate_sequence` (sum is forced to 1)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple families (all return a probability vector on the simplex)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _random_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc`."""
    if conc <= 0:
        # fallback to sparse random
        a = np.zeros(n, dtype=np.float64)
        a[rng.integers(0, n)] = 1.0
        return a
    g = rng.gamma(shape=conc, scale=1.0, size=n)
    s = g.sum()
    return g / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,

    Adam on a smooth‑max surrogate of max(conv(a,a)).
    Returns the best vector seen and its true objective.
    """
    # initialise logits (log‑space simplex)
    v = np.log(seed + 1e-30)

    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 1.5          # starting inverse‑temperature
    beta_mul = 1.007    # geometric increase

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                     # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        #  ---- smooth‑max weights ----
        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k (a*a)_k  (see derivation in original code)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam moments
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 600000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the exact non‑smooth objective.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    w = np.zeros(2 * n - 1, dtype=np.float64)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # build sub‑gradient (average over active constraints)
        g = np.zeros_like(a)
        if active.size:
            w.fill(0.0)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient step on the simplex
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # keep numerics stable
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with both mass‑transfer and value‑swap moves.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)   # initial temperature
    decay = 0.9998                  # slower cooling than the original code

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # ---- propose a move -------------------------------------------------
        if rng.random() < 0.7:  # mass‑transfer move (as before)
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            delta = rng.random() * 0.02 * cur[i]   # up to 2 % of source mass
            if delta > cur[i]:
                delta = cur[i]
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        else:  # swap move – exchanges two entries
            i, j = rng.integers(0, n, size=2)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = cur[j], cur[i]

        new_val = _objective(new_seq)

        # ---- Metropolis acceptance -----------------------------------------
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        # occasional global Gaussian jump (helps escape shallow minima)
        if it % 2000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            tmp = cur + noise
            tmp = np.clip(tmp, 0.0, None)
            s = tmp.sum()
            if s == 0.0:
                tmp = _uniform(n)
            else:
                tmp /= s
            new_val = _objective(tmp)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = tmp
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = tmp.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

def _make_seeds(n: int,
                previous: np.ndarray | None,
                rng: np.random.Generator) -> list[np.ndarray]:
    """
    Assemble a diverse set of start vectors, all normalised to sum = 1.
    """
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    for r in np.linspace(0.9, 0.1, 9):
        if r > 0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for alpha in np.linspace(0.2, 0.8, 5):
        seeds.append(_tukey_window(n, alpha))
    seeds.append(_hann_window(n))
    seeds.append(_blackman_window(n))
    for sigma in np.linspace(0.07, 0.30, 6):
        seeds.append(_gaussian(n, sigma))
    for _ in range(12):
        seeds.append(_random_softmax(n, rng))

    # random Dirichlet vectors – wide range of concentration parameters
    for conc in [0.05, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(30):
            seeds.append(_random_dirichlet(n, rng, conc))

    # reuse previous construction if present (permute it a little)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())
        for _ in range(15):
            perm = rng.permutation(n)
            seeds.append(previous[perm] / previous[perm].sum())

    # clean‑up – enforce non‑negativity and renormalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the supplied time budget and returns the
    best sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 3.0   # safety margin

    # ------------------------------------------------------------------
    # Load a previous construction if the grader provided one
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    rng = np.random.default_rng(seed + 1)

    # ------------------------------------------------------------------
    # Problem size – 4000 is a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # 1) Build a rich seed pool
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous, rng)

    # ------------------------------------------------------------------
    # 2) Quick evaluation of seeds – keep the most promising few
    # ------------------------------------------------------------------
    seed_vals = []
    for s in seeds:
        if time.time() > deadline:
            break
        val = _objective(s)
        seed_vals.append((s, val))
    seed_vals.sort(key=lambda x: x[1])
    top_k = min(20, len(seed_vals))
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = seed_vals[0][0].copy()
    global_best_val = seed_vals[0][1]
    print(f"[seed] best initial = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam on smooth‑max for each top seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=20000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Adam] seed={idx} best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – long sub‑gradient refinement on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                       max_iters=800000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[SubGrad] after long run best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – convex mixing of the best with other promising seeds
    # ------------------------------------------------------------------
    if time.time() < deadline:
        for _ in range(15):
            if time.time() > deadline:
                break
            other = top_seeds[rng.integers(len(top_seeds))]
            alpha = rng.random()
            mixed = (1 - alpha) * global_best_seq + alpha * other
            mixed = _project_simplex(mixed)
            val = _objective(mixed)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = mixed.copy()
                print(f"[mix] new best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase D – final annealing with swap moves
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                    max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Anneal final] best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    # Store for possible future runs
    globals()["height_sequence_2"] = np.copy(global_best_seq)
    return global_best_seq.tolist()
