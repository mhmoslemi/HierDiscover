        a = a_exp / s

"""
Propose a step‑function minimizing the evaluate_sequence objective.
The algorithm builds a diverse pool of analytic and random seed vectors,
selects the most promising few and refines them with three optimisation
stages:

1.  Exponential‑gradient (mirror‑descent) on a smoothed max‑convolution
    surrogate – fast multiplicative updates that stay on the probability
    simplex.

2.  Exact projected sub‑gradient descent on the true max‑convolution
    (the same update used in the reference solution, but with an
    adaptive step‐size).

3.  A short simulated‑annealing polishing that reshuffles the mass of a
    random subset of coordinates.

All stages respect the overall time budget and the best sequence found
is stored in the global name ``height_sequence_1`` and returned as a list
of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast, O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray, n: int) -> float:
    """Surrogate objective 2·n·max(conv(seq,seq)) for a sequence with Σseq = 1."""
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (Σ=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.15))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))

    # Geometric families
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # Power‑law family
    for p in np.linspace(0.0, 6.0, 20):
        seeds.append(_power(n, p))

    # Beta family – broaden search
    alphas = np.concatenate((np.logspace(-2, -0.3, 5), np.linspace(0.3, 2.0, 6)))
    betas = np.concatenate((np.logspace(-2, -0.3, 5), np.linspace(0.3, 4.0, 6)))
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, float(a), float(b)))

    # Gaussian windows
    for sigma_frac in (0.10, 0.15, 0.20, 0.30, 0.40):
        seeds.append(_gaussian(n, sigma_frac))

    # Tukey windows
    for alpha in (0.05, 0.1, 0.2, 0.4, 0.6, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws – a few concentrations
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws (heavy‑tailed)
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Perturb previous best (if any) – tiny Gaussian jitter
    if previous is not None and previous.shape == (n,):
        base = previous / previous.sum()
        for _ in range(3):
            noise = rng.normal(scale=0.02, size=n)
            cand = base + noise
            cand = np.maximum(cand, 0.0)
            if cand.sum() > 0:
                cand = cand / cand.sum()
                seeds.append(cand)

    return seeds

def _exponential_gradient_descent(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Multiplicative update: a_i ← a_i * exp(-η·g_i) with projection by
    normalisation (i.e. KL‑projection onto the simplex).
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    it = 1
    eta0 = 0.8  # initial step size

    while time.time() < deadline and (max_iters is None or it <= max_iters):
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[EG] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # learning rate – 1/√t schedule
        eta = eta0 / np.sqrt(it)

        # multiplicative update (KL‑mirror‑descent)
        a = a * np.exp(-eta * g)

        # re‑normalise onto the simplex (KL‑projection)
        s = a.sum()
        if s <= 0.0:
            # fallback to Euclidean projection if numerical issues arise
            a = _proj_simplex(a - eta * g)
        else:
            a = a / s

        it += 1

    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exact additive sub‑gradient descent with Euclidean projection onto the simplex.
    Step size decays as 1/√t; the routine aborts when the deadline is reached.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the max (same as in EG)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # additive step and simplex projection
        lr = 0.3 / np.sqrt(it)  # a slightly larger base learning rate than the reference
        a = a - lr * g
        a = _proj_simplex(a)

        it += 1

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.03,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Simple SA that redistributes the mass of a random subset of coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 1
    while time.time() < deadline and it <= max_iters:
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # 2 … 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            it += 1
            continue

        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new, n)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

        it += 1

    return best_seq, best_val

    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed collection (extended parameter search)
# ----------------------------------------------------------------------


def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.15))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))

    # Geometric families
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # Power‑law family (extended range)
    for p in np.linspace(0.0, 6.0, 20):
        seeds.append(_power(n, p))

    # Beta family – broad sweep on a log‑scale and linear scale
    alphas = np.concatenate((np.logspace(-2, -0.3, 5), np.linspace(0.3, 2.0, 6)))
    betas = np.concatenate((np.logspace(-2, -0.3, 5), np.linspace(0.3, 4.0, 6)))
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, float(a), float(b)))

    # Gaussian windows
    for sigma_frac in (0.10, 0.15, 0.20, 0.30, 0.40):
        seeds.append(_gaussian(n, sigma_frac))

    # Tukey windows
    for alpha in (0.05, 0.1, 0.2, 0.4, 0.6, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws (heavy‑tailed)
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Perturb previous solution (if any)
    if previous is not None and previous.shape == (n,):
        base = previous / previous.sum()
        for _ in range(3):
            noise = rng.normal(scale=0.02, size=n)
            cand = base + noise
            cand = np.maximum(cand, 0.0)
            if cand.sum() > 0.0:
                cand = cand / cand.sum()
                seeds.append(cand)

    return seeds

# Optimisation primitives
# ----------------------------------------------------------------------


def _exponential_gradient_descent(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Multiplicative update: a_i ← a_i·exp(-η·g_i) with KL‑projection
    (simple normalisation) onto the probability simplex.
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    it = 1
    eta0 = 0.8  # initial step size

    while time.time() < deadline and (max_iters is None or it <= max_iters):
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[EG] it={it:7d} best={cur_val:.6f}")

        # Sub‑gradient of the current maximal entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Learning rate – 1/√t schedule
        eta = eta0 / np.sqrt(it)

        # Multiplicative (KL) update
        a = a * np.exp(-eta * g)

        # Normalise (KL‑projection onto the simplex)
        s = a.sum()
        if s <= 0.0:
            a = _proj_simplex(a - eta * g)  # fallback
        else:
            a = a / s

        it += 1

    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exact additive sub‑gradient descent with Euclidean projection onto the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # Sub‑gradient of the maximal convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Additive step with a slightly larger base step size
        lr = 0.3 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        it += 1

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.03,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles mass among a randomly chosen subset of coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 1
    while time.time() < deadline and it <= max_iters:
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # 2 … 12 random indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            it += 1
            continue

        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new, n)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

        it += 1

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget and returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 3.0  # safety margin

    # Retrieve previous solution if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length – 4000 offers a good speed / quality trade‑off.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – generate diverse seed pool and keep the most promising ones.
    # ------------------------------------------------------------------
    print(f"\n--- Generating seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _objective(vec, n)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 4
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq, n)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – fast exponential‑gradient (mirror‑descent) refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_eg, val_eg = _exponential_gradient_descent(
            global_best_seq, n, deadline, max_iters=None
        )
        if val_eg < global_best_val:
            global_best_seq = seq_eg.copy()
            global_best_val = val_eg
            print(f"[overall‑EG] new global best = {val_eg:.6f}")

    # ------------------------------------------------------------------
    # Stage 2b – exact projected sub‑gradient refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_proj, val_proj = _subgradient_proj(
            global_best_seq, n, deadline, max_iters=2_000_000
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[overall‑SubProj] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final stochastic polishing (simulated annealing).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=800_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}")
        print(f"Official evaluate_sequence value = {final_score:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_score = None

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
