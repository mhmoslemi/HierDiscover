def adam_optimize(start_seq, n, deadline, max_iters=200000):
    p = start_seq.copy()
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    best_seq=p.copy()
    best_val=_objective(p)
    alpha=10.0
    t=0
    while time.time()<deadline and t<max_iters:
        t+=1
        # adapt alpha schedule
        if t<20000:
            alpha=30.0
        elif t<100000:
            alpha=120.0
        else:
            alpha=300.0

        conv = _conv_fft(p,p)
        maxc=conv.max()
        # compute softmax
        exps = np.exp(alpha*(conv - maxc))
        w = exps / exps.sum()
        # gradient
        rev = p[::-1]
        grad = 2.0 * _conv_fft(rev, w)
        # Adam update for p (on log domain?). We'll use direct p.
        m = beta1 * m + (1-beta1)*grad
        v = beta2 * v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = 0.01 * m_hat / (np.sqrt(v_hat) + eps)   # lr may be 0.01
        p = p - step
        # ensure positivity projection
        p = project_simplex(p)   # projects onto simplex
        # evaluate occasionally:
        if t%2000==0:
            val = _objective(p)
            if val < best_val:
                best_val=val
                best_seq=p.copy()
                print(f"[Adam] iter={t} best={val:.6f}")
    return best_seq, best_val

"""
Improved optimiser for the step‑function objective.

Main ideas
----------

* A very rich seed pool (many classic and parametric windows, plus
  random Dirichlet samples) is built and the best few are kept.
* Each seed is refined with an Adam‑based projected‑gradient descent on a
  smooth approximation of the max‑convolution.  The smoothness parameter
  (log‑sum‑exp temperature) is increased gradually so that the optimiser
  first sees a well‑behaved landscape and finally targets the true
  non‑smooth objective.
* After the gradient phase a short simulated‑annealing walk is performed
  to escape any remaining shallow local minima.
* All vectors are kept on the probability simplex (non‑negative,
  sum = 1).  The objective is scale‑invariant, therefore returning a
  probability vector is sufficient for ``evaluate_sequence``.
* The routine respects the supplied time budget (default 1000 s) and
  returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, real‑valued)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The input is assumed to be a non‑negative vector on the simplex.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple families (all return a probability vector on the simplex)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.special import i0
    except Exception:  # pragma: no cover
        return _uniform(n)
    if beta <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    nminus1 = n - 1
    x = (2.0 * i / nminus1) - 1.0
    a = i0(beta * np.sqrt(1.0 - x * x)) / i0(beta)
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _beta_general(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (α,β)‑Beta shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    """Dirichlet with equal concentration `conc` (>0)."""
    x = rng.gamma(conc, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else _uniform(n)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse set of probability vectors.
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic families (coarse sweeps)
    seeds.append(_uniform(n))

    for r in np.linspace(0.1, 0.95, 17):
        seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    for r in np.linspace(0.1, 0.9, 13):
        for p in np.linspace(0.0, 3.0, 13):
            seeds.append(_geom_power(n, r, p))

    for r in np.linspace(0.1, 0.9, 13):
        seeds.append(_sym_geom(n, r))

    # classic windows
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    for alpha in np.linspace(0.0, 1.0, 21):
        seeds.append(_tukey_window(n, alpha))

    for beta in np.linspace(0.0, 30.0, 16):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.05, 0.5, 19):
        seeds.append(_gaussian(n, sigma))

    for a in np.linspace(0.1, 3.0, 13):
        for b in np.linspace(0.1, 3.0, 13):
            if abs(a - b) < 1e-12:
                continue
            seeds.append(_beta_general(n, a, b))

    # random Dirichlet draws (different concentrations)
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(4):
            seeds.append(_random_dirichlet(n, rng, conc))

    # reuse previous once if shape matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # clean‑up – enforce non‑negativity and normalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex (used by gradient methods)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam‑based projected gradient minimisation on a smooth max‑approx.
# ----------------------------------------------------------------------


def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 400000) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the log‑sum‑exp surrogate of max(conv).
    Returns the best vector seen and its exact objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # smoothing schedule – start gentle, finish sharp
        if t < 100000:
            alpha = 30.0
        elif t < 250000:
            alpha = 120.0
        else:
            alpha = 300.0

        conv = _conv_fft(a, a)
        maxc = conv.max()

        # soft‑max weights (log‑sum‑exp)
        exps = np.exp(alpha * (conv - maxc))
        w = exps / exps.sum()

        # gradient of the smooth surrogate
        rev_a = a[::-1]
        grad = 2.0 * _conv_fft(rev_a, w)   # shape (n,)

        # Adam update on the *plain* probabilities (projected later)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # base learning rate – slowly decaying
        base_lr = 0.02 / math.sqrt(t)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

        # occasional exact evaluation
        if t % 2000 == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={t:7d} best={val:.6f}")

    return best_seq, best_val

# Simulated‑annealing fine‑tuning (as in the previous version)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.06)
    decay = 0.9992
    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        delta = rng.random() * 0.05 * cur[i]
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        if it % 800 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq = new_seq / new_seq.sum()
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")
    return best, best_val
