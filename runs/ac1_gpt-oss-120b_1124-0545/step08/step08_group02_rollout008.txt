def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The vector must be non‑negative; the function does not enforce the
    constraints.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)

    s = a.sum()

def _sine_window(n: int) -> np.ndarray:

def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:

    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()

    i = np.arange(n, dtype=np.float64)
    a = (
        0.42

def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density (α∈(0,1]).
       Edge‑heavy when α is small, uniform when α=1.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Low‑collision (B₂[t]) set generator
# ----------------------------------------------------------------------
def _low_collision_set(
    n: int,
    target_m: int,
    max_ordered: int,
    rng: np.random.Generator,
    deadline: float,
) -> np.ndarray | None:
    """
    Greedy construction of a set S ⊂ {0,…,n‑1} such that every ordered
    pair‑sum i+j (i,j∈S) occurs at most `max_ordered` times.
    The routine repeatedly tries random permutations until it either
    reaches `target_m` elements or the time budget expires.
    Returns the index array of the best set encountered.
    """
    best_mask = None
    best_size = 0
    attempt = 0
    # The number of attempts is not hard‑capped; we stop when time is low.
    while time.time() < deadline - 0.2:
        attempt += 1
        mask = np.zeros(n, dtype=bool)
        sum_counts = np.zeros(2 * n - 1, dtype=np.uint8)  # ordered‑sum counter

        order = rng.permutation(n)
        for cand in order:
            # check diagonal contribution
            if sum_counts[cand * 2] + 1 > max_ordered:
                continue

            # compute sums with already chosen indices
            existing = np.nonzero(mask)[0]
            if existing.size:
                sums = cand + existing
                if np.any(sum_counts[sums] + 2 > max_ordered):
                    continue

            # accept candidate
            mask[cand] = True
            sum_counts[cand * 2] += 1
            if existing.size:
                sum_counts[sums] += 2

            if mask.sum() >= target_m:
                break

        cur_size = mask.sum()
        if cur_size > best_size:
            best_size = cur_size
            best_mask = mask.copy()
            print(f"[LowColl] attempt {attempt:4d} size {cur_size}")

        # early exit if theoretical maximum for this K is reached
        # theoretical max ≈ sqrt(2 * n * max_ordered)
        theo_max = int(math.sqrt(2.0 * n * max_ordered))
        if best_size >= min(target_m, theo_max):
            break

    if best_mask is None or best_size == 0:
        return None
    return np.where(best_mask)[0]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 50000,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate of max(conv(a,a)).
    Returns the best vector seen and its true objective.
    """
    v = np.log(seed + 1e-30)  # logits to keep simplex parametrisation

    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # start with a moderate smoothing factor
    beta = 1.5
    beta_mul = 1.01  # a bit more aggressive than before

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # --- smooth‑max weights ---
        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k · (a*a)_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the exact non‑smooth objective.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient: average over all active constraints
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)            # keep numbers stable
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _lp_refine(
    f: np.ndarray,
    n: int,
    deadline: float,
    t_mix: float = 0.35,
    max_lp_iters: int = 7,
) -> np.ndarray:
    """
    Perform several cheap LP refinements.
    For each refinement we:
        * compute current max M = max(conv(f,f))
        * collect active indices where M is attained
        * solve maximise Σ g   subject to   f * g ≤ M   and g ≥ 0
    The solution g is mixed back into f.
    """
    if not CVXPY_AVAILABLE:
        return f

    a = f.copy()
    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - 1e-9)[0]

        g = cp.Variable(n)
        constraints = [g >= 0]

        # linear constraints for each active position
        for k in active:
            coeff = np.zeros(n, dtype=np.float64)
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i = np.arange(i_low, i_high + 1)
            j = k - i
            coeff[j] = a[i]
            constraints.append(coeff @ g <= M + 1e-12)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2500,
                       abstol=1e-9, reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if g.value is None:
            break

        g_opt = np.asarray(g.value).flatten()
        tot = g_opt.sum()
        if tot <= 0:
            break
        g_opt = g_opt / tot

        a = (1.0 - t_mix) * a + t_mix * g_opt
        a = _project_simplex(a)

    return a


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk.
    Moves a tiny amount of mass between two random positions and occasionally
    adds a global Gaussian perturbation.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.05)
    decay = 0.9994

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # transfer a small fraction of mass from i to j
        delta = rng.random() * 0.03 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        # occasional global jump
        if it % 1200 == 0:
            noise = rng.normal(scale=0.0015, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            s = new_seq.sum()
            if s == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= s
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Seed generation (including low‑collision constructions)
# ----------------------------------------------------------------------
def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """
    Assemble a diverse set of start vectors, all normalised to sum = 1.
    Classic families, random sparse vectors and deterministic Sidon‑like
    structures are added.
    """
    seeds: list[np.ndarray] = []

    # classic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))
    for sigma_factor in np.linspace(0.08, 0.30, 7):
        seeds.append(_gaussian(n, sigma_factor))
    for alpha in np.linspace(0.55, 0.95, 9):
        seeds.append(_beta_window(n, alpha))
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # sparse random seeds – support sizes 30 … 200
    for m in range(30, 210, 20):
        for _ in range(4):
            seeds.append(_sparse_random_seed(n, m, rng))

    # deterministic Sidon‑like structures
    for m in (50, 80, 110, 140):
        seeds.append(_sidon_like_seed(n, min(m, n)))
        seeds.append(_quadratic_residue_seed(n, min(m, n)))

    # reuse previous if appropriate
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final clean‑up
    cleaned: list[np.ndarray] = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence.
    The routine respects the supplied time budget (seconds) and returns the
    best sequence found as a plain list of Python floats.
    """
    # --------------------------------------------------------------
    # Basic initialisation
    # --------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    rng = np.random.default_rng(seed + 1)

    # try to reuse a previously stored result
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # --------------------------------------------------------------
    # Problem size – 4000 offers a good trade‑off
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # --------------------------------------------------------------
    # Build an initial pool of seeds
    # --------------------------------------------------------------
    seeds = _make_seeds(n, previous, rng)

    # --------------------------------------------------------------
    # Add a few low‑collision support sets (ordered collisions ≤ 6)
    # --------------------------------------------------------------
    low_coll_deadline = deadline - 30.0          # keep ~30 s for the rest
    # try a few different collision caps, starting with the most aggressive (6)
    for K, target in [(6, 200), (5, 190), (4, 180)]:
        if time.time() > low_coll_deadline:
            break
        lc_idx = _low_collision_set(
            n=n,
            target_m=target,
            max_ordered=K,
            rng=rng,
            deadline=low_coll_deadline,
        )
        if lc_idx is not None and lc_idx.size > 0:
            a = np.zeros(n, dtype=np.float64)
            a[lc_idx] = 1.0 / lc_idx.size
            a = _project_simplex(a)   # just to be safe
            seeds.append(a)
            print(f"[seed] added low‑collision set K={K} size {lc_idx.size}")

    # --------------------------------------------------------------
    # Quick evaluation of seeds, keep the top few
    # --------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = min(15, len(seed_vals))
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = seed_vals[0][0].copy()
    global_best_val = seed_vals[0][1]
    print(f"[seed] initial best obj = {global_best_val:.6f}")

    # Early exit if a seed already beats the target (highly unlikely but cheap)
    if global_best_val < 1.503:
        print("[early] found a seed already below target")
        final_seq = np.maximum(global_best_seq, 0.0)
        final_seq /= final_seq.sum()
        globals()["height_sequence_1"] = final_seq.copy()
        return final_seq.tolist()

    # --------------------------------------------------------------
    # Phase A – Adam optimisation on the promising seeds
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline - 10.0:   # keep a margin for later phases
            break
        seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=40000)
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall Adam] seed={idx} best={val:.6f}")

    # --------------------------------------------------------------
    # Phase B – Exact sub‑gradient (mirror‑descent)
    # --------------------------------------------------------------
    if time.time() < deadline - 8.0:
        remaining = deadline - time.time()
        # estimate cost of one sub‑gradient iteration ≈ 0.0016 s
        max_sub_iters = int(min(800000, remaining / 0.0016))
        seq, val = _subgradient_refine(
            global_best_seq, n, deadline, max_iters=max_sub_iters
        )
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall SubGrad] best={val:.6f}")

    # --------------------------------------------------------------
    # Phase C – Repeated LP refinement (interleaved with short sub‑grads)
    # --------------------------------------------------------------
    if CVXPY_AVAILABLE and time.time() < deadline - 5.0:
        lp_seq = global_best_seq.copy()
        lp_pass = 0
        while time.time() + 5.0 < deadline and lp_pass < 12:
            lp_pass += 1
            lp_seq = _lp_refine(lp_seq, n, deadline, t_mix=0.35, max_lp_iters=7)
            # short sub‑gradient clean‑up after each LP pass
            sub_seq, sub_val = _subgradient_refine(
                lp_seq, n, deadline, max_iters=40000
            )
            if sub_seq is not None and sub_val < global_best_val:
                global_best_seq = sub_seq.copy()
                global_best_val = sub_val
                print(f"[LP+Sub] new best {sub_val:.6f}")
                lp_seq = sub_seq.copy()
            else:
                # even if not better, keep the LP‑refined vector for next round
                lp_seq = lp_seq.copy()
                break

    # --------------------------------------------------------------
    # Phase D – Simulated annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall Anneal] best={val:.6f}")

    # --------------------------------------------------------------
    # Fallback (should never be needed)
    # --------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    # final normalisation and safety checks
    final_seq = np.maximum(global_best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
    else:
        final_seq = final_seq / total

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(final_seq)})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
