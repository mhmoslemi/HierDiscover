def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` (ignoring the 0‑1000
    clipping, which does not affect probability‑simplex vectors).
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0

    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

            continue
        w /= w_sum

        # gradient of Σ w_k (a*a)_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam moments
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        deadline: float,
                        max_iters: int = 300000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact non‑smooth objective.
    Returns the best sequence encountered.
    """
    n = start_seq.size
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    w = np.zeros(2 * n - 1, dtype=np.float64)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # build sub‑gradient (average over active constraints)
        g = np.zeros_like(a)
        if active.size:
            w.fill(0.0)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # keep numerics stable
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _lp_refine(f: np.ndarray,
               deadline: float,
               t_mix: float = 0.7,
               max_lp_iters: int = 25) -> np.ndarray:
    """
    Repeated cheap LP refinements (normalised to the probability simplex).
    """
    if not CVXPY_AVAILABLE:
        return f

    a = f.copy()
    n = a.size
    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - 1e-9)[0]
        if active.size == 0:
            break

        g = cp.Variable(n)
        constraints = [g >= 0]

        # one linear constraint per active index
        for k in active:
            coeff = np.zeros(n, dtype=np.float64)
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i = np.arange(i_low, i_high + 1)
            j = k - i
            coeff[j] = a[i]                     # (f * g)[k] = Σ_i f[i]·g[k‑i]
            constraints.append(coeff @ g <= M + 1e-12)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=2000, abstol=1e-9, reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if g.value is None:
            break

        g_opt = np.asarray(g.value).flatten()
        # normalise the LP solution to sum = 1
        tot = g_opt.sum()
        if tot <= 0:
            break
        g_opt = g_opt / tot

        a = (1.0 - t_mix) * a + t_mix * g_opt
        a = _project_simplex(a)   # stay on the simplex
    return a


def _lp_grow(a: np.ndarray,
             deadline: float,
             t_mix: float = 0.2,
             max_lp_iters: int = 15) -> np.ndarray:
    """
    LP‑based mass‑growing step.

    Keeps the current max‑convolution bound (active constraints) while
    maximising the total L1 mass.  No renormalisation is performed – the
    L1‑mass is allowed to increase (capped per entry at 1000).
    """
    if not CVXPY_AVAILABLE:
        return a

    n = a.size
    best_obj = _objective(a)
    best_a = a.copy()

    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - 1e-9)[0]
        if active.size == 0:
            break

        g = cp.Variable(n)
        constraints = [g >= 0, g <= 1000]   # per‑coordinate ceiling

        # linear constraints for the *active* convolution positions
        for k in active:
            coeff = np.zeros(n, dtype=np.float64)
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i = np.arange(i_low, i_high + 1)
            j = k - i
            coeff[j] = a[i]               # (a * g)[k] = Σ_i a[i]·g[j]
            constraints.append(coeff @ g <= M + 1e-12)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=2000, abstol=1e-9, reltol=1e-9)
        except Exception:
            break

        if g.value is None:
            break

        g_opt = np.asarray(g.value).flatten()
        # mixing – small step to keep the bound stable
        a = (1.0 - t_mix) * a + t_mix * g_opt
        a = np.clip(a, 0.0, 1000)

        cur_obj = _objective(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[LP‑grow] iteration={lp_it} new best={cur_obj:.6f}")

        # stop early if improvement stalls
        if lp_it >= 1 and cur_obj >= best_obj * 0.9999:
            break

    return best_a


def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk (fine‑tuning).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)   # initial temperature
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, cur.size, size=2)
        while i == j:
            j = rng.integers(0, cur.size)

        delta = rng.random() * 0.02 * cur[i]   # up to 2 % of source mass
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=cur.size)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(cur.size)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


def _make_seeds(n: int,
                previous: np.ndarray | None,
                rng: np.random.Generator) -> list[np.ndarray]:
    """
    Assemble a diverse set of start vectors, all normalised to sum = 1.
    """
    seeds: list[np.ndarray] = []

    # classic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))
    for sigma_factor in np.linspace(0.08, 0.30, 7):
        seeds.append(_gaussian(n, sigma_factor))
    for alpha in np.linspace(0.55, 0.95, 9):
        seeds.append(_beta_window(n, alpha))
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # sparse random seeds – a family that often yields a smaller max‑conv
    for m in range(30, 210, 20):
        for _ in range(4):
            seeds.append(_sparse_random_seed(n, m, rng))

    # deterministic Sidon‑like structures
    for m in (50, 70, 90, 110):
        seeds.append(_sidon_like_seed(n, min(m, n)))
        seeds.append(_quadratic_residue_seed(n, min(m, n)))

    # reuse previous construction if present
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # clean‑up – enforce non‑negativity and renormalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

# Core optimisation driver for a given length `n`
# ----------------------------------------------------------------------


def _optimise_length(n: int,
                    previous: np.ndarray | None,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Runs the complete multi‑phase optimisation for a specific length `n`.
    Returns the best sequence (still on the probability simplex) and its
    objective value.
    """
    rng = np.random.default_rng()
    print(f"\n=== Optimising length {n} ===")
    seeds = _make_seeds(n, previous, rng)

    # quick evaluation of seeds – keep the top few
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 20
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = seed_vals[0][0].copy()
    global_best_val = seed_vals[0][1]
    print(f"[seed] best initial={global_best_val:.6f}")

    # Phase A – Adam on smoothed‑max for each top seed
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(seed_vec, deadline, max_iters=18000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Adam] seed={idx} best={val:.6f}")

    # Phase B – long sub‑gradient refinement on the current best
    if time.time() < deadline:
        seq, val = _subgradient_refine(global_best_seq, deadline,
                                       max_iters=300000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[SubGrad] after long run best={val:.6f}")

    # Phase C – LP refinements (normalised) + short sub‑gradient passes
    while time.time() < deadline - 30.0:
        # LP refinement (keeps simplex)
        seq_lp = _lp_refine(global_best_seq, deadline,
                            t_mix=0.6, max_lp_iters=30)
        if seq_lp is None:
            break

        # short sub‑gradient after LP
        seq, val = _subgradient_refine(seq_lp, deadline,
                                       max_iters=150000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[LP+SubGrad] new best={val:.6f}")

        # perturbation block (helps escape shallow minima)
        if time.time() > deadline - 20.0:
            break
        perturb = global_best_seq.copy()
        noise = rng.normal(scale=0.001, size=n)
        perturb += noise
        perturb = np.clip(perturb, 0.0, None)
        if perturb.sum() == 0.0:
            perturb = _uniform(n)
        else:
            perturb /= perturb.sum()
        seq, val = _subgradient_refine(perturb, deadline,
                                       max_iters=200000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Perturbed] new best={val:.6f}")

    # Phase D – final annealing
    if time.time() < deadline - 5.0:
        seq, val = _anneal_optimize(global_best_seq, deadline, max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Anneal final] best={val:.6f}")

    print(f"\n=== Finished length {n} ===\nBest objective ≈ {global_best_val:.6f}")
    return global_best_seq, global_best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # Retrieve a previous construction if supplied by the grader
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Stage 1 – optimise a moderate size (n=4000) with the full pipeline
    # ------------------------------------------------------------------
    n1 = 4000
    seq1, val1 = _optimise_length(n1, previous, deadline)
    best_seq = seq1.copy()
    best_val = val1

    # ------------------------------------------------------------------
    # Stage 2 – attempt a growth step (mass‑increasing LP) on the best
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n=== LP‑grow phase (mass increase) ===")
        grow_deadline = deadline
        seq_grown = _lp_grow(best_seq, grow_deadline,
                             t_mix=0.25, max_lp_iters=30)
        grown_val = _objective(seq_grown)
        if grown_val < best_val:
            best_val = grown_val
            best_seq = seq_grown.copy()
            print(f"[LP‑grow] improved objective to {grown_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – optional second length (n=8000) if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        n2 = 8000
        # re‑normalise previous for the new length (simply interpolate)
        # we up‑sample by linear interpolation and re‑normalise.
        x_old = np.linspace(0, 1, best_seq.size)
        x_new = np.linspace(0, 1, n2)
        seq_up = np.interp(x_new, x_old, best_seq)
        seq_up = np.clip(seq_up, 0.0, None)
        seq_up /= seq_up.sum()
        seq2, val2 = _optimise_length(n2, seq_up, deadline)
        if val2 < best_val:
            best_val = val2
            best_seq = seq2.copy()
            print(f"[Large‑n] improvement to {val2:.6f}")

        # final growth on the larger vector
        if time.time() < deadline - 5.0:
            print("\n=== LP‑grow on large vector ===")
            seq_grown2 = _lp_grow(best_seq, deadline,
                                  t_mix=0.25, max_lp_iters=20)
            grown2_val = _objective(seq_grown2)
            if grown2_val < best_val:
                best_val = grown2_val
                best_seq = seq_grown2.copy()
                print(f"[LP‑grow‑large] improved to {grown2_val:.6f}")

    # ------------------------------------------------------------------
    # Return the best discovered candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")

    print(f"\n=== Overall best ===\nObjective ≈ {best_val:.6f}  (length {len(best_seq)})")
    globals()["height_sequence_1"] = np.copy(best_seq)
    return best_seq.tolist()
