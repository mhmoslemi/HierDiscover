      2·n·C / m² .
  The theoretical maximum of m is ≈√(8·n·C) (≈179 for n=4000, C=4),
  giving a target <1.2 – well below the current best ~1.511.

* The set is obtained by a fast greedy construction with many random
  restarts (the “low‑collision” set).  The best mask found within the
  time budget is turned into a uniform vector on its support.

* Starting from this vector we run a few optimisation phases:
    – Adam on a smooth‑max surrogate of max(conv(a,a))
    – exact sub‑gradient mirror‑descent
    – an optional tiny LP‑refinement (if cvxpy is available)
    – a final simulated‑annealing walk.

All phases respect the overall time budget (default 1000 s).  The
procedure returns the best sequence discovered as a plain list of
floats, stored globally as ``height_sequence_1`` for possible reuse.
"""

import time
import math
import numpy as np

# Optional cvxpy – used only for the cheap LP refinement
try:
    import cvxpy as cp
    CVXPY_AVAILABLE = True
except Exception:  # pragma: no cover
    CVXPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on a simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Low‑collision B₂[C] set construction
# ----------------------------------------------------------------------


def _low_collision_set_improved(
    n: int,
    max_allowed: int,
    target_m: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray | None, np.ndarray | None]:
    """
    Greedy construction of a set S ⊂ {0,…,n‑1} such that each ordered sum
    i+j (i,j∈S) occurs at most `max_allowed` times.
    Multiple random attempts are performed until the time budget runs out.
    Returns the mask of the best set found together with its sum‑count array.
    """
    best_mask = None
    best_counts = None
    best_size = 0
    attempt = 0

    # stop early if we hit the theoretical limit
    theoretical_limit = int(math.sqrt(2 * max_allowed * (2 * n - 1))) + 1

    while time.time() < deadline - 0.5:
        attempt += 1
        mask = np.zeros(n, dtype=bool)
        sum_counts = np.zeros(2 * n - 1, dtype=np.uint16)

        order = rng.permutation(n)
        for cand in order:
            # self‑pair
            if sum_counts[2 * cand] + 1 > max_allowed:
                continue

            existing = np.nonzero(mask)[0]
            if existing.size:
                sums = cand + existing
                if np.any(sum_counts[sums] + 2 > max_allowed):
                    continue

            # accept candidate
            mask[cand] = True
            sum_counts[2 * cand] += 1
            if existing.size:
                sum_counts[sums] += 2

            # early exit if we already reached the target size
            if mask.sum() >= target_m:
                break

        cur_size = mask.sum()
        if cur_size > best_size:
            best_size = cur_size
            best_mask = mask.copy()
            best_counts = sum_counts.copy()
            print(f"[LowColl] attempt {attempt:4d} size {cur_size}")

        if best_size >= theoretical_limit:
            break

    if best_mask is None:
        return None, None
    return best_mask, best_counts


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 25000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam on a smooth‑max surrogate of max(conv(a,a)).
    Returns the best vector seen and its true objective.
    """
    v = np.log(seed + 1e-30)  # logits for simplex parametrisation

    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 1.5
    beta_mul = 1.007

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·(a*a)_k -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- chain rule through soft‑max -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam moment update -----
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact non‑smooth
    objective.  Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over all active constraints)
        g = np.zeros_like(a)
        if len(active) > 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / len(active)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _lp_refine(
    f: np.ndarray,
    n: int,
    deadline: float,
    t_mix: float = 0.3,
    max_lp_iters: int = 5,
) -> np.ndarray:
    """
    Perform a few cheap linear‑programming refinements.
    For each refinement we:
        * compute the current maximum M of conv(f,f)
        * collect the (few) indices where the max is attained (active set)
        * solve   maximise 1ᵀ g   subject to   f * g ≤ M  and g ≥ 0
          (the constraints are linear because f is fixed)
        * blend the solution g back into f with weight `t_mix`.
    Returns the refined vector (still on the simplex).
    """
    if not CVXPY_AVAILABLE:
        return f

    a = f.copy()
    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - 1e-9)[0]

        g = cp.Variable(n)
        constraints = [g >= 0]

        for k in active:
            coeff = np.zeros(n, dtype=np.float64)
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i = np.arange(i_low, i_high + 1)
            j = k - i
            coeff[j] = a[i]
            constraints.append(coeff @ g <= M + 1e-12)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                       abstol=1e-9, reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if g.value is None:
            break

        g_opt = np.asarray(g.value).flatten()
        tot = g_opt.sum()
        if tot <= 0:
            break
        g_opt = g_opt / tot

        a = (1.0 - t_mix) * a + t_mix * g_opt
        a = _project_simplex(a)

    return a


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk that moves a tiny amount of mass
    between two random positions and occasionally adds a global Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n, size=1)[0]

        delta = rng.random() * 0.02 * cur[i]
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the supplied time budget and returns the best
    sequence discovered as a plain list of floats.
    """
    # --- initialise ---------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    rng = np.random.default_rng(seed + 1)

    # Problem size – 4000 is a good trade‑off
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # --- low‑collision set construction --------------------------------
    max_allowed = 4
    # theoretical bound ~ sqrt(8 * n * max_allowed)
    target_m = int(math.sqrt(8 * n * max_allowed)) + 5
    low_mask, low_counts = _low_collision_set_improved(
        n=n,
        max_allowed=max_allowed,
        target_m=target_m,
        rng=rng,
        deadline=deadline,
    )

    # Build a uniform vector on the best low‑collision set (if any)
    if low_mask is not None and low_mask.any():
        idx = np.where(low_mask)[0]
        m = idx.size
        a_low = np.zeros(n, dtype=np.float64)
        a_low[idx] = 1.0 / m
        # exact max ordered count (integer)
        max_count = int(np.max(_conv_fft(low_mask.astype(np.float64),
                                         low_mask.astype(np.float64))))
        init_obj = 2.0 * n * max_count / (m * m)
        print(f"[seed] low‑collision set size {m}, max ordered count {max_count}, "
              f"obj≈{init_obj:.6f}")
    else:
        a_low = _uniform(n)
        init_obj = _objective(a_low)
        print(f"[seed] falling back to uniform, obj≈{init_obj:.6f}")

    # --- collect other seeds (previous best, a few classics) ----------
    seeds = [a_low]

    # previous best (if any)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev = np.asarray(previous, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            s = prev.sum()
            if s > 0:
                prev = prev / s
                seeds.append(prev)
        except Exception:
            pass

    # a few classic families for safety (very cheap)
    def _add_family(vec):
        if np.isfinite(_objective(vec)):
            seeds.append(vec)

    _add_family(_uniform(n))
    _add_family(_gaussian(n, sigma_factor=0.1))
    _add_family(_hann_window(n))
    _add_family(_blackman_window(n))
    # random softmax seeds
    for _ in range(3):
        logits = rng.standard_normal(n)
        prob = np.exp(logits - np.max(logits))
        prob = prob / prob.sum()
        _add_family(prob)

    # Choose the few best seeds for the expensive phases
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = min(10, len(seed_vals))
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = seed_vals[0][0].copy()
    global_best_val = seed_vals[0][1]
    print(f"[seed] initial best obj = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam (smooth‑max) on the top seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=30000)
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall Adam] seed={idx} best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        # Rough cost per iteration ≈ 0.0015 s
        max_sub_iters = int(min(500000, remaining / 0.0015))
        seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                       max_iters=max_sub_iters)
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – LP refinement (if cvxpy is present)
    # ------------------------------------------------------------------
    if CVXPY_AVAILABLE and time.time() < deadline - 2.0:
        lp_seq = global_best_seq.copy()
        while time.time() + 5.0 < deadline:
            lp_seq = _lp_refine(lp_seq, n, deadline, t_mix=0.2, max_lp_iters=5)
            # a short sub‑gradient clean‑up
            sub_seq, sub_val = _subgradient_refine(lp_seq, n, deadline,
                                                   max_iters=50000)
            if sub_seq is not None and sub_val < global_best_val:
                global_best_seq = sub_seq.copy()
                global_best_val = sub_val
                print(f"[LP+Sub] new best {sub_val:.6f}")
                lp_seq = sub_seq.copy()
            else:
                break

    # ------------------------------------------------------------------
    # Phase D – Simulated annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                    max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall Anneal] best={val:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation
    # ------------------------------------------------------------------
    final_seq = np.maximum(global_best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
    else:
        final_seq = final_seq / total

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(final_seq)})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
