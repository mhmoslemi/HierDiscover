"""
search for a non‑negative step function minimizing
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)^2)

The algorithm runs four time‑aware phases (total budget ≈ 1000 s):

1. **Deterministic analytic seeds** – a variety of closed‑form families
   (uniform, Catalan, Hann, Hamming, Blackman, Gaussian, Beta‑shaped)
   are generated for several lengths.  The best seed is kept.

2. **Heavy random Dirichlet sampling** – many Dirichlet draws with a
   selection of concentration parameters (α∈{0.3,0.5,1.0,2.0}) are taken.
   The overall best sample is retained.

3. **Smooth L‑BFGS refinement** – the current best vector is re‑parameterised
   by logits (soft‑max).  For a sequence of increasing temperatures β the
   smooth approximation  smooth_maxβ(conv) = (1/β)·log Σ exp(β·conv)  is
   minimised with L‑BFGS‑B (SciPy).  After each β the exact objective
   2·n·max(conv) is evaluated and the best solution kept.

4. **Exact sub‑gradient mirror descent** – a final few thousand iterations
   of exponentiated‑gradient (KL‑mirror descent) polish the solution.

All phases respect the global time budget; the routine always returns a
non‑negative list (the evaluator will clip / normalise automatically).
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ---------------------------------------------------------------
# Optional SciPy import (used only in phase 3)
# ---------------------------------------------------------------
try:
    import scipy.optimize  # type: ignore
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# Low‑level convolution utilities
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using real FFT.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the grader for a *normalised* sequence
    (sum(seq) == 1):
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ---------------------------------------------------------------
# Analytic seed families (all returned normalised to sum = 1)
# ---------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2.0 * i) * (2.0 * i - 1.0)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 - np.cos(2.0 * math.pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2.0 * math.pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2.0 * math.pi * i / (n - 1))
         + a2 * np.cos(4.0 * math.pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete Beta(alpha,alpha) density on the points (i+0.5)/n,
    normalised to sum = 1.
    """
    # Avoid division‑by‑zero when alpha < 1: add a tiny epsilon.
    eps = 1e-30
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    coeffs = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, alpha - 1.0)
    total = coeffs.sum()
    if total == 0.0 or not np.isfinite(total):
        return _uniform(n)
    return coeffs / total

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """
    Draw a Dirichlet(alpha,…,alpha) vector.
    """
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s

# Smooth L‑BFGS optimisation (log‑softmax parametrisation)

class _SmoothObj:
    """
    Callable object that returns (value, gradient) of the smoothed objective
    for a fixed length n and temperature β.
    The variable is the log‑softmax logits (unconstrained).
    """
    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, logits: np.ndarray):
        n = self.n
        beta = self.beta

        # soft‑max (stable)
        max_log = np.max(logits)
        exp_logits = np.exp(logits - max_log)
        a = exp_logits / exp_logits.sum()          # a ≥ 0, sum = 1

        # convolution a*a
        conv = _conv_fft(a, a)                     # length = 2n‑1

        # smooth max via log‑sum‑exp (stable)
        m = np.max(conv)
        exp_beta = np.exp(beta * (conv - m))
        sum_exp = exp_beta.sum()
        p = exp_beta / sum_exp                     # soft weights, sum = 1
        smooth_max = m + np.log(sum_exp) / beta

        # objective (smooth version of 2·n·max)
        obj = 2.0 * n * smooth_max

        # gradient w.r.t a (g_a = 2 * conv(p, a[::-1]))
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot) * (2.0 * n)   # include outer factor 2·n

        return obj, grad_logits


def _lbfgs_refine(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Apply a short L‑BFGS‑B schedule on a series of increasing temperatures.
    Returns the best sequence seen and its exact objective value.
    """
    if not _SCIPY_AVAILABLE:                     # fallback – nothing to do
        return start_seq, _objective(start_seq)

    n = len(start_seq)
    logits = np.log(start_seq + 1e-30)

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    # temperature schedule – start smooth, finish sharp
    betas = [5.0, 15.0, 30.0, 60.0, 150.0]

    for beta in betas:
        if time.time() >= deadline:
            break
        obj_fun = _SmoothObj(n, beta)

        # L‑BFGS options – we limit maxiter; ftol is tiny because we rely on time.
        try:
            res = scipy.optimize.minimize(
                obj_fun,
                logits,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 2000, "ftol": 1e-12, "disp": False},
            )
        except Exception as exc:                  # pragma: no cover
            print("[L‑BFGS] optimisation aborted:", exc)
            break

        logits = res.x
        # exact evaluation of the current point
        max_log = np.max(logits)
        a = np.exp(logits - max_log)
        a /= a.sum()
        val = _objective(a)
        print(f"[L‑BFGS] β={beta:.1f} exact={val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = a.copy()

        # early exit if already under the target
        if best_val < 1.505:
            break

    return best_seq, best_val

def _refine_subgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient (KL‑mirror descent) on the simplex,
    using the true sub‑gradient of the active convolution entry.
    Returns the best vector found.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Refine] it={it:d} best={cur_val:.6f}")

        # active sub‑gradient (only the maximal entry participates)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = 0.7 / math.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the supplied time budget and returns the best
    sequence found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Re‑use a previously stored solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                previous = prev_arr / prev_arr.sum()
            else:
                previous = None
        except Exception:                     # pragma: no cover
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore (moderate but not too large)
    # ------------------------------------------------------------------
    lengths = [1500, 2000, 2500, 3000, 3500, 4000, 5000, 6000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic seeds (analytic families)
    # ------------------------------------------------------------------
    seed_families = (
        _uniform,
        _catalan_shape,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.25),
        lambda n: _gaussian(n, sigma_factor=0.35),
        lambda n: _beta_shape(n, 0.3),   # very edge‑heavy
        lambda n: _beta_shape(n, 0.6),
        lambda n: _beta_shape(n, 1.0),   # actually uniform again
        lambda n: _beta_shape(n, 1.5),
        lambda n: _beta_shape(n, 2.0),   # centre‑heavy
    )

    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – random Dirichlet sampling (heavy‑weight)
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.35   # ~35 % of remaining budget
    sample_deadline = now + time_for_sampling
    alphas = [0.3, 0.5, 1.0, 2.0]

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.2f} val={val:.6f}")

        if best_val < 1.505:       # a comfortable early break
            break

    # ------------------------------------------------------------------
    # Phase 3 – smooth L‑BFGS refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        lbfgs_deadline = time.time() + (deadline - time.time()) * 0.30
        seq_lbfgs, val_lbfgs = _lbfgs_refine(best_seq, lbfgs_deadline)
        if val_lbfgs < best_val:
            best_val = val_lbfgs
            best_seq = seq_lbfgs
            print(f"[L‑BFGS‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – exact sub‑gradient mirror descent (final polishing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        final_seq, final_val = _refine_subgradient(best_seq, deadline)
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Final‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Sanity check – the grader will call evaluate_sequence again.
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:                    # pragma: no cover
        print("Warning: evaluate_sequence raised", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )

    # expose the final vector for possible reuse in subsequent runs
    globals()["height_sequence_1"] = best_seq.tolist()

    return best_seq.tolist()
