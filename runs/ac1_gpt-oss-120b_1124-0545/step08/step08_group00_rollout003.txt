"""
A fast heuristic for the “max‑autocorrelation” problem.

The code builds a diverse pool of analytically‑motivated seeds,
keeps the best one (or the previously‑found global best) and then
performs a long run of projected sub‑gradient descent on the true
objective 2·n·max(conv(a,a))/(sum(a)²).  The sub‑gradient of the
active lag is computed analytically and a simple step‑size
schedule  lr = lr0/√t  with projection onto the probability simplex
keeps the vector non‑negative and normalised.  After the bulk of the
budget a short simulated‑annealing polishing phase is run.

The routine respects the global time budget (default 1000 s) and
stores the best vector in the global name ``height_sequence_1``.
"""
import time
import numpy as np

# ------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution)
# ------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used in the optimisation.
    The sequence is assumed to be ℓ₁‑normalised (sum = 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic families (already ℓ₁‑normalised)
# ------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geometric(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geometric(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic(n: int, scale: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(i / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Main optimisation routine
# ------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget and returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # tiny safety margin

    # ------------------------------------------------------------------
    # Length of the candidate.  4000 gives a good speed / quality trade‑off.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1️⃣  Seed generation (diverse analytic families + random draws)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _triangular(n),
        _logistic(n, scale=0.2 * n),
        _logistic(n, scale=0.4 * n),
        _quadratic(n),
    ]

    # geometric families
    for r in np.linspace(0.3, 0.98, 6):
        seeds.append(_geometric(n, r))
        seeds.append(_sym_geometric(n, r))

    # power‑law families
    for p in np.linspace(0.0, 3.0, 7):
        seeds.append(_power_law(n, p))

    # beta families
    for a in (0.5, 1.0, 1.5):
        for b in (0.5, 1.0, 2.0):
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows
    for sigma_frac in (0.15, 0.30, 0.45):
        seeds.append(_gaussian(n, sigma_frac))

    # Random Dirichlet draws (different concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws (heavy‑tailed)
    for _ in range(3):
        seeds.append(_random_exponential(n, rng))

    # Include previously discovered solution if available
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev = np.asarray(previous, dtype=np.float64)
            if prev.shape == (n,) and prev.sum() > 0:
                seeds.append(prev / prev.sum())
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2️⃣  Pick the most promising seed(s) using the cheap surrogate.
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    best_seed, best_val = evaluated[0]
    print(f"[Init] Best seed surrogate value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Heavy projected sub‑gradient descent (the bulk of the budget).
    # ------------------------------------------------------------------
    a = best_seed.copy()                 # current iterate (ℓ₁‑norm = 1)
    global_best_seq = a.copy()
    global_best_val = _objective(a)

    # sub‑gradient parameters
    lr0 = 0.8                             # initial step‑size coefficient

    it = 0
    # reserve ~10 % of the budget for the final SA phase
    sa_reserve = 0.10 * budget_s

    while time.time() < deadline - sa_reserve:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < global_best_val:
            global_best_val = cur_val
            global_best_seq = a.copy()
            print(f"[Sub] it={it:7d} new best={cur_val:.6f}")

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        if it % 200_000 == 0:

    # 4️⃣  Final stochastic polishing (simple simulated annealing).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur = global_best_seq.copy()
        cur_val = _objective(cur)
        best_seq = cur.copy()
        best_val = cur_val

        T = max(0.02 * cur_val, 1e-12)
        min_T = 1e-12
        it_sa = 0
        while time.time() < deadline:
            it_sa += 1
            # exponential cooling
            T = max(T * 0.9995, min_T)

            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue

            new = cur.copy()
            new[idx] = new_alloc

            if delta < 0.0 or rng.random() < np.exp(-delta / T):

                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_seq = new.copy()
                    print(f"[SA] it={it_sa:9d} best={new_val:.6f}")

        global_best_seq = best_seq
        global_best_val = best_val

    # 5️⃣  Final sanity check (optional) and return.
    # ------------------------------------------------------------------

        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

        f"(length {len(global_best_seq)})"
    )
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
