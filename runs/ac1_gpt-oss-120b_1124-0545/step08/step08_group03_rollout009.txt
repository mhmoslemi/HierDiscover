"""
Optimisation of a non‑negative step‑function for the objective

    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear convolution.
The algorithm works in three stages:

1. **Rich seed generation** – a large, diverse pool of candidates is built
   (classical windows, finely‑swept geometric / power families, many random
   sparse supports, Sidon‑like constructions, random softmax).  All candidates
   are normalised to the probability simplex and the *k* best (by the exact
   objective) are kept.

2. **Refinement** – each of the retained seeds is improved with a two‑phase
   optimiser:
      * Adam on a smooth‑max surrogate of the convolution peak.
      * Exponentiated‑gradient (mirror‑descent) on the true non‑smooth
        objective with a long schedule.
   The best result of the refinement is stored as the current global optimum.

3. **Final annealing** – a cheap Metropolis‑Hastings walk (simulated
   annealing) is run for any remaining time to escape shallow minima.

All heavy work respects the global time budget (default 1000 s).  The
function `propose_candidate` returns the best sequence found (as a plain
list of floats) and stores it in the global variable `height_sequence_1`
for possible reuse by later runs.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective (identical to `evaluate_sequence` apart from the 0‑1000
    clipping, which never triggers because we stay on the simplex).
    Returns ``np.inf`` for degenerate inputs.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density (α∈(0,1]) discretised on mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()

def _sparse_random_seed(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random support of size `m` (1 ≤ m ≤ n)."""
    if m >= n:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """Deterministic Sidon‑like support (triangular numbers)."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _quadratic_residue_seed(n: int, m: int) -> np.ndarray:
    """Positions i² (mod n).  Works especially well when `n` is prime."""
    idx = np.mod(np.arange(m) ** 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of max(conv(a, a)).
    Returns the best vector (still on the simplex) and its true objective.
    """
    v = np.log(seed + 1e-30)      # logits for soft‑max parametrisation

    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # start with a modest smoothing temperature
    beta_smooth = 1.0
    beta_mul = 1.005

    best_vec = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_vec = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta_smooth *= beta_mul
        conv = _conv_fft(a, a)
        shift = conv.max()
        w = np.exp(beta_smooth * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of the smoothed max (Σ w_k (a*a)_k)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam moments
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = 0.7 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_vec, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 800000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) directly on the true non‑smooth
    objective.  Returns the best sequence observed.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    w = np.zeros(2 * n - 1, dtype=np.float64)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over all active indices)
        g = np.zeros_like(a)
        if active.size:
            w.fill(0.0)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.95 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                     # stabilise numerics
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis‑Hastings walk that swaps a small amount of mass between
    two positions and occasionally proposes a global Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)    # initial temperature
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # up to 2 % of source mass
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        # occasional “global” jump
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            s = new_seq.sum()
            if s == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= s
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               t_mix: float = 0.6,
               max_lp_iters: int = 15) -> np.ndarray:
    """
    Very cheap (few‑iteration) linear‑programming refinement.
    Reinforces the current convolution bound whilst attempting to increase the
    total mass; the solution is blended back with weight ``t_mix``.
    """
    if not CVXPY_AVAILABLE:
        return f

    a = f.copy()
    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - 1e-9)[0]
        if active.size == 0:
            break

        g = cp.Variable(n)
        constraints = [g >= 0]

        # one linear constraint per active index
        for k in active:
            coeff = np.zeros(n, dtype=np.float64)
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i = np.arange(i_low, i_high + 1)
            j = k - i
            coeff[j] = a[i]                     # (f * g)[k] = Σ_i f[i]·g[k‑i]
            constraints.append(coeff @ g <= M + 1e-12)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=2000, abstol=1e-9, reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if g.value is None:
            break

        g_opt = np.asarray(g.value).flatten()
        tot = g_opt.sum()
        if tot <= 0:
            break
        g_opt = g_opt / tot

        a = (1.0 - t_mix) * a + t_mix * g_opt
        a = _project_simplex(a)

    return a

def _make_seeds(n: int,
                previous: np.ndarray | None,
                rng: np.random.Generator) -> list[np.ndarray]:
    """Construct a very large pool of candidate vectors (all on the simplex)."""
    seeds: list[np.ndarray] = []

    # ---------- deterministic families (coarse + fine sweep) ----------
    seeds.append(_uniform(n))

    # geometric (r in (0,1))
    for r in np.linspace(0.01, 0.99, 31):
        seeds.append(_geom(n, r))

    # power (p ≥ 0)
    for p in np.linspace(0.0, 3.0, 31):
        seeds.append(_power(n, p))

    # geometric * power
    for r in np.linspace(0.01, 0.99, 11):
        for p in np.linspace(0.0, 2.0, 11):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for r in np.linspace(0.01, 0.99, 21):
        seeds.append(_sym_geom(n, r))

    # classical windows
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.9, 15):
        seeds.append(_tukey_window(n, alpha))
    for sigma_factor in np.linspace(0.05, 0.35, 15):
        seeds.append(_gaussian(n, sigma_factor))
    for alpha in np.linspace(0.05, 0.95, 15):
        seeds.append(_beta_window(n, alpha))

    # random softmax draws
    for _ in range(40):
        seeds.append(_random_softmax(n, rng))

    # ----- sparse random seeds (various support sizes) -----
    for m in [20, 30, 40, 50, 60, 80, 100, 120, 150, 180]:
        for _ in range(5):
            seeds.append(_sparse_random_seed(n, m, rng))

    # ----- deterministic Sidon‑like / quadratic residue -----
    for m in [30, 50, 70, 90, 110, 130]:
        seeds.append(_sidon_like_seed(n, min(m, n)))
        seeds.append(_quadratic_residue_seed(n, min(m, n)))

    # reuse previous construction (if supplied)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final sanity‑check
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget and returns the best
    sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # safety margin

    # ------------------------------------------------------------------
    # Load a previous construction if the grader provided one
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    rng = np.random.default_rng(seed + 1)

    # ------------------------------------------------------------------
    # Problem size – we work with a relatively large n (4000) for expressive
    # power, but the code also tries a smaller size (2000) if time permits.
    # ------------------------------------------------------------------
    n_choices = [4000, 2000]    # try the large size first, then fall back
    global_best_seq = None
    global_best_val = float("inf")

    for n in n_choices:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")

        # ---- 1) Build a very diverse seed pool ----
        seeds = _make_seeds(n, previous, rng)

        # ---- 2) Quick evaluation – retain the top few ----
        # We keep only the best 15 seeds to limit later optimisation work.
        top_k = 15
        best_seen = []

        for s in seeds:
            val = _objective(s)
            if len(best_seen) < top_k:
                best_seen.append((val, s))
                best_seen.sort(key=lambda x: x[0])
            else:
                if val < best_seen[-1][0]:
                    best_seen[-1] = (val, s)
                    best_seen.sort(key=lambda x: x[0])

        top_seeds = [s for _, s in best_seen[:top_k]]
        if not top_seeds:
            continue

        # initialise global best from this n
        if best_seen[0][0] < global_best_val:
            global_best_val = best_seen[0][0]
            global_best_seq = best_seen[0][1].copy()
            print(f"[seed] best initial={global_best_val:.6f}")

        # ---- 3) Adam phase on each top seed ----
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, n, deadline,
                                      max_iters=20000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[Adam] n={n} seed={idx} best={val:.6f}")

        # ---- 4) Long sub‑gradient refinement of the best so far ----
        if time.time() + 1.0 < deadline:  # leave a tiny buffer
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=800000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[SubGrad] after long run n={n} best={val:.6f}")

        # ---- 5) Light LP‑mix + sub‑gradient cycles (optional) ----
        while time.time() < deadline - 30.0:
            seq_lp = _lp_refine(global_best_seq, n, deadline,
                                t_mix=0.6, max_lp_iters=15)
            if seq_lp is None:
                break
            seq, val = _subgradient_refine(seq_lp, n, deadline,
                                           max_iters=300000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[LP+SubGrad] n={n} new best={val:.6f}")

            # small random perturbation before next loop iteration
            if time.time() > deadline - 20.0:
                break
            noise = rng.normal(scale=0.001, size=n)
            pert = global_best_seq + noise
            pert = np.clip(pert, 0.0, None)
            s = pert.sum()
            if s == 0.0:
                pert = _uniform(n)
            else:
                pert /= s
            seq, val = _subgradient_refine(pert, n, deadline,
                                           max_iters=250000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[Perturbed] n={n} new best={val:.6f}")

        # ---- 6) Final simulated‑annealing fine‑tuning (if time) ----
        if time.time() < deadline - 5.0:
            seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                        max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[Anneal final] n={n} best={val:.6f}")

        # keep the best across different n‑values
        print(f"\n>>> After length {n}, best objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep a safe guard
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    globals()["height_sequence_1"] = np.copy(global_best_seq)
    return global_best_seq.tolist()
