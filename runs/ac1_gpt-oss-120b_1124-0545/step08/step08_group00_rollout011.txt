def upscale(seq, n_new):
    old_n = len(seq)
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n-1, n_new)
    seq_new = np.interp(x_new, x_old, seq)
    seq_new = np.maximum(seq_new,0)
    seq_new /= seq_new.sum()
    return seq_new

"""
A fast heuristic for the “max–convolution” minimisation problem.

The objective to be minimised is

    2·n·max(conv(a,a)) / (sum(a)**2)

which is invariant under scaling – we may assume sum(a)=1 and then
the objective reduces to the L∞‑norm of the self‑convolution of a
probability vector a.

The programme works in three stages (all respecting a 1000 s wall‑clock
budget).

1. **Seed generation** – a diverse collection of analytically motivated
   shapes (uniform, sine‑window, beta, geometric, power‑law,
   √‑inverse, etc.) together with an up‑sampled copy of the best solution
   found in previous runs.  All seeds are normalised to sum = 1 and the
   best few (according to the exact objective) are kept.

2. **Heavy continuous optimisation** – each seed is refined by
   *Adam* on a smooth log‑sum‑exp surrogate of the max‑convolution.
   The temperature β is increased slowly (up to several thousand) so
   that the surrogate converges to the true max.  After Adam a
   projected (Euclidean) sub‑gradient descent with momentum is applied
   directly to the exact max‑convolution; the gradient of the max is
   cheap to compute because it is just a shifted copy of the current
   vector.

3. **Local polishing** – a tiny stochastic hill‑climber that moves a
   small amount of mass between a few coordinates and keeps the move
   if the exact objective improves.

The final vector is stored in the global name ``height_sequence_1`` and
returned as a plain list of floats ready for ``evaluate_sequence``.

The code is deliberately written with small helper functions that can
be reused and with generous printing for progress monitoring.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Low‑level utilities
# ------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (real vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """Exact objective for a probability vector (sum(seq)=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):          # all entries zero → uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic families (all returned ℓ₁‑normalised)
# ------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_sqrt(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1)  (symmetrised)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _upscale(seq: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    if seq.shape[0] == n_new:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, seq.shape[0])
    x_new = np.linspace(0.0, 1.0, n_new)
    res = np.interp(x_new, x_old, seq)
    res = np.maximum(res, 0.0)
    if res.sum() == 0.0:
        res = _uniform(n_new)
    else:
        res /= res.sum()
    return res


# ------------------------------------------------------------
# Seed collection
# ------------------------------------------------------------


def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_inverse_sqrt(n))

    # Geometric families
    for r in np.linspace(0.2, 0.98, 8):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # Power‑law family (including the crucial p≈0.5)
    for p in np.linspace(0.0, 4.0, 17):
        seeds.append(_power(n, p))

    # Beta family (wide range)
    alphas = np.linspace(0.3, 2.0, 5)
    betas = np.linspace(0.3, 4.0, 5)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Random draws
    for conc in (0.05, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Previous solution, up‑scaled if necessary
    if previous is not None and previous.shape[0] > 0:
        try:
            prev_up = _upscale(previous, n)
            seeds.append(prev_up)
        except Exception:
            pass

    return seeds

def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 0.1,
    beta_mul: float = 1.008,
    beta_max: float = 5_000.0,
    lr0: float = 0.7,
) -> tuple[np.ndarray | None, float]:
    """
    Adam on a log‑sum‑exp surrogate of max(conv).

    Returns (best_sequence, best_surrogate_value) or (None, inf) if
    nothing was evaluated before the deadline.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    s = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    cur_beta = beta_start
    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)    # exact surrogate
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # increase temperature slowly
        cur_beta = min(cur_beta * beta_mul, beta_max)

        # smoothed‑max weights  w_k ∝ exp(beta·(conv_k – max_conv))
        shift = np.max(conv)
        w = np.exp(cur_beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a  (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]        # central n entries

        # convert to logits‑space (log‑softmax)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update (learning rate decays like 1/√t)
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        s = beta2 * s + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        s_hat = s / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(s_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_momentum(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    lr0: float = 0.5,
    beta_m: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution
    with a simple momentum term.
    """
    a = start_seq.copy()
    mom = np.zeros_like(a)

    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # exact sub‑gradient of the max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]

        # momentum update
        mom = beta_m * mom + (1.0 - beta_m) * g
        lr = lr0 / np.sqrt(it)
        a = a - lr * mom
        a = _proj_simplex(a)

    return best_seq, best_val


def _local_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    move_frac: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Tiny hill‑climber that moves a tiny fraction of mass between two
    randomly chosen coordinates.  The move is accepted only if the
    exact objective improves (or with a tiny Metropolis probability).
    """
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2, replace=False)
        eps = move_frac * cur[i]          # small chunk taken from i
        if eps <= 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps

        new_val = _objective(new)
        if new_val < cur_val:
            cur, cur_val = new, new_val
            if new_val < best_val:
                best_seq, best_val = new.copy(), new_val
        else:
            # very low‑temperature occasional acceptance
            if rng.random() < np.exp(-(new_val - cur_val) * 0.01):
                cur, cur_val = new, new_val

        if it % 20_000 == 0:
            print(f"[Refine] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **_kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine respects the supplied
    time budget and returns the best vector found.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # Choose a length – larger n gives a tighter theoretical lower bound.
    # n≈12000 provides a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 12_000

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any) and up‑scale it.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and previous:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0:
                previous = _upscale(prev_arr, n)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a pool of seeds and keep the most promising.
    # ------------------------------------------------------------------
    print(f"\n--- Generating seeds for length {n} ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seeds (exact objective):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – heavy optimisation (Adam + momentum sub‑gradient)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n=== Optimisation run {idx+1}/{len(best_seeds)} ===")

        # Adam phase (smooth surrogate)
        seq_adam, val_adam = _adam_optimize(
            seed_vec,
            n,
            deadline,
            max_iters=200_000,
            beta_start=0.1,
            beta_mul=1.008,
            beta_max=5_000.0,
            lr0=0.7,
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[global‑Adam] new best = {val_adam:.6f}")

        # Exact projected sub‑gradient with momentum
        seq_proj, val_proj = _subgradient_momentum(
            global_best_seq,
            n,
            deadline,
            max_iters=300_000,
            lr0=0.5,
            beta_m=0.9,
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[global‑SubGrad] new best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final stochastic polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_ref, val_ref = _local_refine(
            global_best_seq,
            n,
            deadline,
            max_iters=200_000,
            move_frac=0.001,
        )
        if val_ref < global_best_val:
            global_best_seq = seq_ref.copy()
            global_best_val = val_ref
            print(f"[global‑Refine] final best = {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Sanity check with the provided evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_seq.tolist())
        print(f"evaluate_sequence returned {final_score:.6f}")
    except Exception as e:
        print("Warning: evaluate_sequence raised an exception:", e)

    # expose for possible reuse by later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
