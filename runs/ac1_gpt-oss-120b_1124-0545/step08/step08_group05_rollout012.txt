"""
Hybrid optimiser for the step‑function objective (balanced time allocation).

The optimiser works in three stages, each respecting a share of the total
budget:

1. **Exponentiated‑gradient sub‑gradient descent** – reduces the worst
   convolution entry rapidly.  The number of iterations is limited by a
   fixed time budget (≈50 % of the total) rather than a huge iteration
   count, so that later stages still have time to act.

2. **Smoothed‑max (log‑sum‑exp) gradient descent** – a finer descent on a
   smooth surrogate of the max‑convolution term.  This stage receives
   roughly 30 % of the budget.

3. **Simulated‑annealing walk** – larger stochastic moves that escape
   shallow local minima and polish the best vector.  It uses the final
   20 % of the time.

The code also builds a large, diverse set of seed windows (geometric,
power, classic windows, random soft‑max, etc.) and keeps the best few.
A simple population‑mixing step creates additional candidates from
different seeds before the optimisation pipeline is applied.

All vectors are kept on the probability simplex (non‑negative,
normalised to sum = 1) so that the objective reduces to
    2·n·max(conv(a,a)).
The implementation relies on FFT‑based convolution for speed
(`_conv_fft`).  The final sequence is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real‑valued)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) for a probability vector.
    The input is assumed to be non‑negative and sum(seq) ≈ 1.
    """
    # Guard against degenerate sequences (should never happen here)
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max()


# ----------------------------------------------------------------------
# Simple families – all return a probability vector (sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on (0,1), discretised on mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_general(n: int, alpha: float, beta: float) -> np.ndarray:
    """General discrete Beta(α,β) shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _make_seeds(
    n: int, previous: np.ndarray | None, rng: np.random.Generator
) -> list[np.ndarray]:
    """
    Assemble a diverse set of start vectors, all normalised to sum = 1.
    """
    seeds: list[np.ndarray] = []

    # Basic families (including increasing geometric families)
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # **Increasing** geometric families – new seeds
    for c in np.linspace(0.5, 4.5, 7):
        r = 1.0 + c / n  # > 1
        seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 3.5, 7):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 7):
            seeds.append(_geom_power(n, r, p))

    for c in np.linspace(0.5, 4.0, 7):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # Classic windows
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 7):
        seeds.append(_tukey_window(n, alpha))

    # Gaussian family
    for sigma_factor in np.linspace(0.08, 0.30, 11):
        seeds.append(_gaussian(n, sigma_factor))

    # Beta families
    for alpha in np.linspace(0.05, 0.95, 10):
        seeds.append(_beta_window(n, alpha))
    for alpha in np.linspace(0.05, 0.95, 7):
        for beta in np.linspace(0.05, 0.95, 7):
            if abs(alpha - beta) > 1e-12:
                seeds.append(_beta_general(n, alpha, beta))

    # Random soft‑max vectors
    for _ in range(12):
        seeds.append(_random_softmax(n, rng))

    # A few random permutations of the previous best (if any)
    if previous is not None and previous.shape == (n,):
        for _ in range(6):
            perm = rng.permutation(n)
            seeds.append(previous[perm] / previous[perm].sum())

    # Clean‑up – enforce non‑negativity and normalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex (used by the annealer)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Sub‑gradient mirror‑descent (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the exact objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices that are essentially maximal
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over active constraints)
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1 : n - 1 + n]

        # exponentiated‑gradient step with a simple learning‑rate schedule
        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)  # avoid overflow
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _smooth_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Gradient descent on a log‑sum‑exp surrogate of max(conv)."""
    a = start_seq.copy()
    eps = 1e-30
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)

        # annealing of the smoothness parameter
        if it < 100_000:
            alpha = 30.0
        elif it < 250_000:
            alpha = 120.0
        else:
            alpha = 300.0

        m = conv.max()
        exps = np.exp(alpha * (conv - m))
        w = exps / exps.sum()  # soft‑max weights

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1 : n - 1 + n]

        lr = 0.6 / math.sqrt(it)
        log_a = np.log(a + eps) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum):
            a = _uniform(n)
        else:
            a /= a_sum

        if it % 2000 == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Smooth] it={it:7d} best={val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (larger mass‑transfer moves)
# ----------------------------------------------------------------------


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """Metropolis–Hastings walk with occasional large jumps."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # temperature schedule – start a little higher than in the original code
    T0 = max(1e-6, cur_val * 0.08)
    decay = 0.9993

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay**it)

        # move a small random fraction of mass from i to j
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        delta = rng.random() * 0.07 * cur[i]  # up to 7 % transfer
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
        if accept:
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # every few hundred steps perform a global jitter
        if it % 600 == 0:
            noise = rng.normal(scale=0.0015, size=n)
            jittered = cur + noise
            jittered = np.clip(jittered, 0.0, None)
            if jittered.sum() == 0.0:
                jittered = _uniform(n)
            else:
                jittered /= jittered.sum()
            val = _objective(jittered)
            if val < cur_val or rng.random() < math.exp(-(val - cur_val) / (T + 1e-12)):
                cur = jittered
                cur_val = val
                if val < best_val:
                    best_val = val
                    best = jittered.copy()
                    print(f"[Anneal] it={it:7d} (jitter) best={val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Mixing of several good candidates – population crossover
# ----------------------------------------------------------------------


def _mix_candidates(candidates: list[np.ndarray], rng: np.random.Generator) -> np.ndarray:
    """
    Random convex combination of up to three randomly chosen candidates,
    followed by simplex projection.
    """
    if not candidates:
        raise ValueError("No candidates to mix")
    k = min(3, len(candidates))
    idx = rng.choice(len(candidates), size=k, replace=False)
    weights = rng.dirichlet(np.ones(k))
    mixed = sum(w * candidates[i] for w, i in zip(weights, idx))
    # projection back onto the simplex to keep a proper probability vector
    return _project_simplex(mixed)


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine balances three optimisation phases
    (sub‑gradient, smooth‑max descent, annealing) within the allotted time
    and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 1.5

    # --------------------------------------------------------------
    # Length – 4000 is a good trade‑off (larger hardly improves speed)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Re‑use a previously found construction if available
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # --------------------------------------------------------------
    # 0) evaluate the previous vector (if any)
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    if previous is not None and previous.shape == (n,):
        val = _objective(previous)
        best_seq = previous.copy()
        best_val = val
        print(f"[prev] n={n} val={val:.6f}")

    # --------------------------------------------------------------
    # 1) generate a rich pool of seeds
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed + 123)
    seeds = _make_seeds(n, previous, rng)
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_seeds = [s for s, _ in seed_vals[:5]]
    if seed_vals[0][1] < best_val:
        best_val = seed_vals[0][1]
        best_seq = seed_vals[0][0].copy()
        print(f"[seed] new best={best_val:.6f}")

    # --------------------------------------------------------------
    # 2) allocate time for the three optimisation phases
    # --------------------------------------------------------------
    total_time = deadline - time.time()
    sub_time = total_time * 0.48
    smooth_time = total_time * 0.30
    anneal_time = total_time - sub_time - smooth_time

    # --------------------------------------------------------------
    # Phase 1 – (limited) sub‑gradient mirror descent
    # --------------------------------------------------------------
    phase_deadline = time.time() + sub_time
    seq, val = _subgradient_refine(best_seq, n, phase_deadline, max_iters=500_000)
    if val < best_val:
        best_val = val
        best_seq = seq.copy()
        print(f"[phase SubGrad] best={val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – smooth‑max gradient descent
    # --------------------------------------------------------------
    phase_deadline = time.time() + smooth_time
    seq, val = _smooth_optimize(best_seq, n, phase_deadline, max_iters=250_000)
    if val < best_val:
        best_val = val
        best_seq = seq.copy()
        print(f"[phase Smooth] best={val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    phase_deadline = time.time() + anneal_time
    seq, val = _anneal_optimize(best_seq, n, phase_deadline, max_iters=None)
    if val < best_val:
        best_val = val
        best_seq = seq.copy()
        print(f"[phase Anneal] best={val:.6f}")

    # --------------------------------------------------------------
    # Optional extra pop‑mixing pass (uses any remaining time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print(f"[mix] performing an extra mixing pass (time left {remaining:.1f}s)")
        # keep a small pool of good candidates (including the current best)
        pool = top_seeds + [best_seq]
        for _ in range(6):
            mixed = _mix_candidates(pool, rng)
            # short refinement (few hundred sub‑gradient steps)
            sub_dead = time.time() + min(2.0, remaining / 8.0)
            mixed_ref, val = _subgradient_refine(mixed, n, sub_dead, max_iters=40_000)
            if val < best_val:
                best_val = val
                best_seq = mixed_ref.copy()
                print(f"[mix‑refine] new best={val:.6f}")

    # --------------------------------------------------------------
    # 5) Fallback (unlikely)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    # store for potential subsequent runs
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
