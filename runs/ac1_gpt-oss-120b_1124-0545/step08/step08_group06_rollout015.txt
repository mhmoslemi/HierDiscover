def _refine_smooth_adam(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    # start_seq sum=1
    a = start_seq.copy()
    n = len(a)
    w = np.log(a + 1e-30)  # log-softmax
    # compute best values
    best_seq = a.copy()
    best_val = _objective(a)
    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    lambda_param = 10.0
    # maybe schedule lambda increase
    while time.time() < deadline:
        t += 1
        # compute a via softmax
        # In case we maintain a variable w and compute a as softmax
        # shift for stability
        max_w = np.max(w)
        exp_w = np.exp(w - max_w)
        a = exp_w / exp_w.sum()
        # compute convolution
        conv = _conv_fft(a, a)
        # compute real objective
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # compute smooth weights p
        max_c = np.max(conv)
        # choose lambda scaled maybe based on iteration
        # schedule lambda: increase over time
        # e.g., lambda = min(5000, 10 * sqrt(t))
        # but avoid overflow: we will compute exponent relative to max.
        lam = lambda_param
        exp_term = np.exp(lam * (conv - max_c))
        p = exp_term / exp_term.sum()
        # compute gradient w.r.t a
        # compute a_rev
        a_rev = a[::-1]
        # convolution of a_rev and p
        g_full = _conv_fft(a_rev, p) # length = n + (2n-1) -1 = 3n-2
        g_a = g_full[(n-1):(n-1 + n)]  # slice length n
        grad_a = 4.0 * n * g_a
        # compute avg = dot(a, grad_a)
        avg = np.dot(a, grad_a)
        # gradient w.r.t w
        grad_w = a * (grad_a - avg)
        # Adam updates for w
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        # Update w
        w = w - (0.01 / (np.sqrt(v_hat) + 1e-8)) * m_hat
        # schedule lambda: maybe increase slowly?
        if t % 1000 == 0:
            lambda_param = min(2000.0, lambda_param * 1.2)
            # optional print progress
            print(f"[SmoothAdam] it={t} val={cur_val:.6f} lambda={lambda_param:.2f}")
    # final compute a from final w
    max_w = np.max(w)
    exp_w = np.exp(w - max_w)
    a = exp_w / exp_w.sum()
    return best_seq, best_val

"""
Improved search for the minimum of ``evaluate_sequence``.

The code combines:
* a richer collection of analytic seed families (uniform, Catalan,
  geometric, power‑law and two‑parameter Beta shapes);
* massive random sampling of the seed families;
* a smooth‑max version of Adam (mirrored Adam on the probability simplex
  via a soft‑max parametrisation).  The smooth‑max objective
  ``M_λ(conv) = (1/λ)·log∑exp(λ·conv)`` is a differentiable surrogate for
  ``max(conv)``; its gradient can be written using only two FFT‑based
  convolutions, so each iteration is cheap (~O(n log n)).
* a very small time‑budget manager – the whole routine stops well
  before the allotted 1000 s and returns the best sequence found.

The optimiser works on the log‑probability vector ``w`` (so that
``a = softmax(w)`` is always a non‑negative vector with sum 1).  Adam
updates are performed on ``w``; after each iteration the true
objective (the exact ``evaluate_sequence`` value) is recorded and the
best sequence is kept.

The implementation uses only the standard library, NumPy and SciPy
(only for the optional beta‑function based seeds).  All helper functions
are top‑level, contain no closures, and the final result is a plain
Python list as required by the grader.
"""

import time
import numpy as np
from scipy import special

# ----------------------------------------------------------------------
# Fast convolution (real FFT, full linear convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real 1‑D arrays via FFT.
    Returns ``np.convolve(x, y)`` but much faster for the lengths used.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple objective (the value that ``evaluate_sequence`` would return
# for a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """
    ``2·n·max(conv(seq,seq))`` for a probability vector ``seq``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """a_i ∝ binom(2i,i)/4**i  (≈ 1/√i)."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _geometric_shape(n: int, r: float = 0.999) -> np.ndarray:
    """Truncated geometric law a_i ∝ r**i."""
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


def _powerlaw_shape(n: int, alpha: float = 0.55, offset: float = 0.0) -> np.ndarray:
    """
    a_i ∝ (i+offset+1)**(-alpha)   (alpha>0).
    After normalisation the vector sums to 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = idx ** (-alpha)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


def _beta_shape(n: int, a: float = 0.6, b: float = 1.4) -> np.ndarray:
    """
    a_i ∝ (i+1)^{a-1}·(n-i)^{b-1}   (a,b>0, a+b>1 guarantees finiteness).
    This family includes the Catalan shape (a≈0.5,b≈1) and many
    asymmetric variants.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    left = idx ** (a - 1.0)
    right = (n - idx + 1) ** (b - 1.0)
    a_vec = left * right
    total = a_vec.sum()
    if total == 0.0:
        return _uniform(n)
    return a_vec / total


# ----------------------------------------------------------------------
# Random Dirichlet sampling (used as a cheap non‑parametric baseline)
# ----------------------------------------------------------------------


def _dirichlet(alpha: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet(α,…,α) – returns a probability vector."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s

    v = np.zeros_like(w)

    best_seq = a.copy()
    best_val = _objective(a)

    t = 0
    lam = 10.0          # start temperature
    lam_inc = 1.05      # multiplicative increase every few thousand iters
    lr_base = 0.02      # base learning rate for w

    while time.time() < deadline:
        t += 1

        # ----- compute probability vector -----
        max_w = np.max(w)
        exps = np.exp(w - max_w)        # stabilised softmax
        a = exps / exps.sum()           # a ≥0, Σa = 1

        # ----- exact objective (for bookkeeping) -----
        conv_full = _conv_fft(a, a)      # length 2n‑1
        cur_val = 2.0 * n * conv_full.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 500 == 0:
                print(f"[SmoothAdam] it={t:7d} val={cur_val:.6f} λ={lam:.2f}")

        # ----- smooth‑max weights -----
        # p_i = exp(λ·conv_i) / Σ exp(λ·conv)
        max_c = conv_full.max()
        exp_term = np.exp(lam * (conv_full - max_c))
        p = exp_term / exp_term.sum()    # shape (2n‑1,)

        # ----- gradient w.r.t. a -----
        # grad_a = 4·n· Σ_k p_k a_{k‑i}
        a_rev = a[::-1]                  # reversed a, length n
        # full convolution of a_rev (len n) with p (len 2n‑1)
        g_full = _conv_fft(a_rev, p)      # length n+(2n‑1)-1 = 3n‑2
        # slice the part that corresponds to indices i = 0 … n‑1
        start = n - 1
        grad_a = 4.0 * n * g_full[start:start + n]   # shape (n,)

        # ----- gradient w.r.t. w (softmax) -----
        avg = np.dot(a, grad_a)          # scalar = ⟨a,grad_a⟩
        grad_w = a * (grad_a - avg)      # ∇_w  (softmax Jacobian)

        # ----- Adam update on w -----
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (decreasing)
        lr = lr_base / (np.sqrt(t) + 1.0)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # ----- temperature schedule -----
        if t % 2000 == 0:
            lam = min(5000.0, lam * lam_inc)

    # final evaluation of the best sequence
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point – the function required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **_ignore):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the given time budget
    (default 1000 s) and returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # Keep a margin for the final sanity check
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Lengths that we will try – larger n gives a finer approximation
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic analytic seeds
    # ------------------------------------------------------------------
    for n in lengths:
        for maker, name in (
            (_catalan_shape, "catalan"),
            (_uniform, "uniform"),
            (lambda n: _geometric_shape(n, r=0.999), "geo_0.999"),
            (lambda n: _geometric_shape(n, r=0.995), "geo_0.995"),
            (lambda n: _powerlaw_shape(n, alpha=0.55), "pow_0.55"),
            (lambda n: _powerlaw_shape(n, alpha=0.57), "pow_0.57"),
            (lambda n: _powerlaw_shape(n, alpha=0.6), "pow_0.60"),
            (lambda n: _beta_shape(n, a=0.55, b=1.45), "beta_0.55_1.45"),
            (lambda n: _beta_shape(n, a=0.6, b=1.4), "beta_0.60_1.40"),
        ):
            try:
                seq = maker(n)
            except Exception:          # defensive – a maker may raise
                continue
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={name:12s} val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – random Dirichlet sampling (different concentration)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    sample_deadline = now + remaining * 0.30   # ~30 % of the remaining budget

    alphas = [0.5, 1.0, 2.0]                 # sparse → smooth draws
    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = float(rng.choice(alphas))
        seq = _dirichlet(alpha, n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.1f} val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – smooth‑max Adam refinement (starts from best seed)
    # ------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        refine_deadline = deadline
        seq_ref, val_ref = _refine_smooth_adam(best_seq, refine_deadline)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = seq_ref
            print(f"[Overall‑SmoothAdam] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will call evaluate_sequence again)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:                 # noqa: BLE001
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
