"""
search.py – a fast, time‑budgeted optimiser for the “max‑convolution’’ objective.

The algorithm proceeds in four (time‑aware) stages:

1. **Analytic seeds** – a diverse catalogue of windows (uniform,
   Hann, Hamming, Blackman, Gaussian, Catalan, triangular, quadratic,
   exponential) is generated for several lengths.  The best seed is kept.

2. **Random Dirichlet‑style sampling** – many random vectors are drawn
   from Dirichlet distributions (α∈{0.2,0.5,1,2}) and also from a
   sorted‑Dirichlet (monotone decreasing) distribution.  The overall
   best sample is retained.

3. **Smooth Adam optimisation** – starting from the best vector,
   minimise a smooth approximation of ``max(conv(a,a))`` using a
   log‑sum‑exp (temperature β) surrogate.  β is annealed from 5 to 200
   and Adam’s learning‑rate decays as 1/√t.  The exact objective is
   evaluated periodically, and the best iterate is saved.

4. **Exact sub‑gradient mirror descent** – a few thousand KL‑mirror–
   descent steps polish the solution using the true sub‑gradient of the
   active convolution entry.

All stages respect the global 1000 s budget; the routine finally returns
the best sequence it found (as a Python list).  The helper functions are
top‑level (no closures) and use only NumPy and the provided
``evaluate_sequence``.  Print statements give progress information.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from math import sqrt, pi, log, exp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (fast for large vectors).
    Returns an array of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Fast surrogate of ``evaluate_sequence`` for a *normalised* vector
    (sum(seq) == 1).  Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (all returned normalised to sum == 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    return coeffs / total if total != 0.0 else _uniform(n)


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0 - a1 * np.cos(2 * pi * i / (n - 1))
         + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = w.sum()
    return w / s if s != 0.0 else _uniform(n)


def _triangular_decreasing(n: int) -> np.ndarray:
    """Linear decay from the first element to zero at the last."""
    w = np.arange(n, 0, -1, dtype=np.float64)
    return w / w.sum()


def _triangular_symmetric(n: int) -> np.ndarray:
    """Symmetric triangle peaked in the centre (Fejér‑type)."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    w = 1.0 - np.abs(i - centre) / (centre + 1.0)
    w[w < 0] = 0.0
    return w / w.sum()


def _quadratic_shape(n: int) -> np.ndarray:
    """Quadratic bump: (i+1)*(n‑i).  Symmetric, peaks in the middle."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1) * (n - i)
    return w / w.sum()


def _exponential_decay(n: int, rho: float = 0.999) -> np.ndarray:
    """Strictly decreasing exponential a_i ∝ rho**i."""
    i = np.arange(n, dtype=np.float64)
    w = rho ** i
    return w / w.sum()


# ----------------------------------------------------------------------
# Random Dirichlet sampling
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet(alpha,…,alpha) vector."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s != 0.0 else _uniform(n)


def _random_sorted_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Generate a Dirichlet sample and sort it descending (monotone)."""
    v = _random_dirichlet(n, rng, alpha)
    return np.sort(v)[::-1]


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    max_iter: int = 30_000_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)) using Adam on
    logits (log‑softmax) representation.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999
    lr0 = 0.02

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        a = np.exp(logits)
        a /= a.sum()

        conv = _conv_fft(a, a)                         # length = 2n‑1

        # linear annealing of temperature β
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.5))

        # smooth max via log‑sum‑exp
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()                 # soft‑weights

        # gradient w.r.t. a  (see derivation in analysis)
        conv_pa = _conv_fft(p, a[::-1])               # length = 3n‑2
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]          # slice of length n

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update with bias correction
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # Periodic exact evaluation
        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:8d} n={n:5d} val={val:.6f} β={beta:.1f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient mirror descent (exponentiated gradient)
# ----------------------------------------------------------------------
def _refine_subgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Classical KL‑mirror descent on the simplex using the true sub‑gradient
    of the active convolution entry.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Refine] it={it:8d} n={n:5d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.7 / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)               # stabilisation
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Local random perturbation (hill‑climbing)
# ----------------------------------------------------------------------
def _local_perturbation(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: repeatedly move a tiny amount of mass from a
    random source index to a random target index if the move improves the
    exact objective.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    rng = np.random.default_rng()
    # step size as a fraction of the current total mass at the source
    step_frac = 0.02
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] < 1e-12:
            continue
        eps = step_frac * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            a = cand
            # occasionally print progress
            if (it & 2047) == 0:
                print(f"[Perturb] it={it:8d} n={n:5d} best={val:.6f}")
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective under a 1000 s time budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # Re‑use a previously stored solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore (moderate, still fast enough for many FFTs)
    # ------------------------------------------------------------------
    lengths = [1500, 2000, 2500, 3000, 3500, 4000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic seeds
    # ------------------------------------------------------------------
    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.25),
        _triangular_decreasing,
        _triangular_symmetric,
        _quadratic_shape,
        lambda n: _exponential_decay(n, rho=0.998),
    )

    for n in lengths:
        for fn in seed_families:
            try:
                seq = fn(n)
            except Exception:
                continue
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:20s} val={val:.6f}")

        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – massive random Dirichlet sampling (including sorted)
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.4   # 40 % of remaining budget
    sample_deadline = now + time_for_sampling
    alphas = [0.2, 0.5, 1.0, 2.0]

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = float(rng.choice(alphas))
        # 50 % chance to use a sorted (monotone) Dirichlet sample
        if rng.random() < 0.5:
            seq = _random_sorted_dirichlet(n, rng, alpha=alpha)
        else:
            seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.2f} sorted={rng.random()<0.5} val={val:.6f}")

        if best_val < 1.48:          # early exit if we’re already very good
            break

    # ------------------------------------------------------------------
    # Phase 3 – Smoothed Adam optimisation (refinement)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # allocate ~30 % of the remaining time to Adam
        opt_deadline = time.time() + (deadline - time.time()) * 0.3
        seq_adam, val_adam = _smooth_adam_optimize(
            best_seq,
            opt_deadline,
            beta_start=5.0,
            beta_end=200.0,
            max_iter=30_000_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam
            print(f"[Adam‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Exact sub‑gradient mirror descent (final polishing)
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _refine_subgradient(
            best_seq,
            deadline,
        )
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Final‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Tiny local perturbations (cheap hill‑climbing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # give the last few seconds to a very cheap local search
        perturb_seq, perturb_val = _local_perturbation(
            best_seq,
            deadline,
        )
        if perturb_val < best_val:
            best_val = perturb_val
            best_seq = perturb_seq
            print(f"[Perturb‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check (the grader will call evaluate_sequence again)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
