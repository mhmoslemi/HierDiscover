    values = np.interp(x_new, x_old, seq)

"""
Hybrid Genetic + Sub‑gradient optimiser for the step‑function attack.

Goal
----
Find a non‑negative vector `a` (length n) minimising

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the full linear convolution.  The objective is homogeneous,
so we keep all candidates on the probability simplex (∑ a = 1) – this way
the clipping performed inside ``evaluate_sequence`` never interferes.

Approach
--------
1. **Rich seed pool** – analytic families (geometric, power‑law, windows,
   Dirichlet draws, random soft‑max, etc.) with very generous parameter ranges.
   The best few seeds are kept.

2. **Genetic search** – a small population (≈ 50) evolves under
   elitism, crossover, Gaussian mutation and a “mass‑transfer” move.  After
   each mutation the offspring is projected onto the simplex.  The fitness
   is the exact objective (computed via an FFT‑based convolution).  The
   algorithm runs until the time budget expires (≈ 70 % of the total time).

3. **Local sub‑gradient refinement** – the current best individual is
   polished with exponentiated‑gradient mirror‑descent on the exact
   non‑smooth objective (≈ 20 % of the remaining time).

4. **Fallback** – if nothing is found a uniform distribution is returned.

All timings are printed; the final vector is stored in the global variable
``height_sequence_2`` for possible reuse by later runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Convolution of a with itself."""
    return _conv_general(a, a)


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_self(seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families – all return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1), discretised on mid‑points.
    α ∈ (0,1] gives edge‑heavy shapes; α=1 is uniform.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `alpha`."""
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _upsample_to(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing probability vector to a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return _uniform(new_n)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    vals = np.interp(x_new, x_old, seq)
    vals = np.clip(vals, 0.0, None)
    s = vals.sum()
    return vals / s if s > 0 else _uniform(new_n)


def _make_extended_seeds(
    n: int,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Assemble a very diverse set of start vectors (all normalised to sum = 1).
    The ranges are deliberately generous to increase the chance of finding
    a very good region of the search space.
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # very basic
    seeds.append(_uniform(n))

    # geometric – decay factor up to roughly 1‑12/n
    for c in np.linspace(0.5, 12.0, 24):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power law
    for p in np.linspace(0.0, 4.0, 21):
        seeds.append(_power(n, p))

    # combined geometric‑power
    for c in np.linspace(0.5, 12.0, 12):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 5):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 12.0, 12):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classic windows
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))

    # beta‑pdf family
    for alpha in np.linspace(0.3, 2.5, 12):
        seeds.append(_beta_window(n, alpha))

    # Gaussian family – coarse sweep
    for sigma_factor in np.linspace(0.06, 0.30, 9):
        i = np.arange(n, dtype=np.float64)
        centre = (n - 1) / 2.0
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
        a = a / a.sum()
        seeds.append(a)

    # Dirichlet draws – several concentration values
    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, alpha, rng))

    # random soft‑max vectors
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # up‑sample previous solution if it exists and has a different length
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            seeds.append(_upsample_to(previous, n))
        except Exception:
            pass

    # final clean‑up – clip tiny negatives, normalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned


# ----------------------------------------------------------------------
# Sub‑gradient mirror‑descent (exponentiated‑gradient) refinement
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact non‑smooth objective.
    Returns the best vector seen and its objective value.
    """
    n = start_seq.size
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_self(a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient: average over active constraints
        g = np.zeros_like(a)
        if active.size > 0:
            w = np.zeros(conv.size, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_general(rev_a, w)
            g = 2.0 * full[n - 1 : n - 1 + n]

        # exponentiated‑gradient step
        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # numerical safety
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Genetic optimiser
# ----------------------------------------------------------------------


def _genetic_optimize(
    init_population: list[np.ndarray],
    deadline: float,
    elite_frac: float = 0.20,
    mutation_sigma_start: float = 0.04,
    mutation_sigma_end: float = 0.005,
) -> tuple[np.ndarray, float]:
    """
    Simple steady‑state GA.  The population is a list of probability vectors.
    Returns the best member and its objective value.
    """
    pop = [(a, _objective(a)) for a in init_population]
    pop.sort(key=lambda x: x[1])
    n = pop[0][0].size
    best_seq, best_val = pop[0]

    elite_size = max(2, int(len(pop) * elite_frac))
    sigma = mutation_sigma_start
    decay = (mutation_sigma_end / mutation_sigma_start) ** (1.0 / (deadline - time.time()))

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- parent selection (uniform from elite) ----
        p1 = pop[np.random.randint(0, elite_size)][0]
        p2 = pop[np.random.randint(0, elite_size)][0]

        # ---- crossover (convex blend) ----
        alpha = np.random.rand()
        child = alpha * p1 + (1.0 - alpha) * p2

        # ---- mutation ----
        if np.random.rand() < 0.7:
            # Gaussian jitter
            child += np.random.normal(scale=sigma, size=n)
        else:
            # mass‑transfer move
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                delta = np.random.rand() * 0.2 * child[i]
                child[i] -= delta
                child[j] += delta

        # ---- projection onto simplex ----
        child = _project_simplex(child)

        # ---- evaluation ----
        child_val = _objective(child)

        # ---- replacement (steady‑state) ----
        if child_val < pop[-1][1]:
            pop[-1] = (child, child_val)
            pop.sort(key=lambda x: x[1])
            if child_val < best_val:
                best_val = child_val
                best_seq = child.copy()
                if it % 5000 == 0:
                    print(f"[GA] it={it:6d} best={best_val:.6f}")

        # ---- adapt mutation scale ----
        sigma *= decay

        # occasional injection of fresh random individuals (keep diversity)
        if it % 3000 == 0:
            for _ in range(3):
                fresh = np.random.dirichlet(np.ones(n))
                fresh_val = _objective(fresh)
                if fresh_val < pop[-1][1]:
                    pop[-1] = (fresh, fresh_val)
                    pop.sort(key=lambda x: x[1])
                    if fresh_val < best_val:
                        best_val = fresh_val
                        best_seq = fresh.copy()
                        print(f"[GA] (inject) it={it:6d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function that (approximately) minimises
    `evaluate_sequence`.  Returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Choose a length – 8000 gives a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Load previous construction if it has the right length (or up‑sample)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 0) evaluate previous construction (if any) after up‑sampling
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    if previous is not None:
        prev_up = _upsample_to(previous, n)
        val = _objective(prev_up)
        best_seq = prev_up
        best_val = val
        print(f"[prev] up‑sampled val={val:.6f}")

    # ------------------------------------------------------------------
    # 1) Build a rich seed pool and keep the top candidates
    # ------------------------------------------------------------------
    seeds = _make_extended_seeds(n, previous)
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    # keep best 30 seeds for the GA initial population
    init_pop = [s for s, _ in seed_vals[:30]]

    if seed_vals[0][1] < best_val:
        best_val = seed_vals[0][1]
        best_seq = seed_vals[0][0].copy()
        print(f"[seed] new global best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Genetic optimisation (≈ 70 % of remaining time)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 10.0:
        ga_deadline = time.time() + 0.70 * time_left
        seq, val = _genetic_optimize(init_pop, ga_deadline)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[overall GA] best={val:.6f}")

    # ------------------------------------------------------------------
    # 3) Sub‑gradient refinement (use whatever time is still left)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 5.0 and best_seq is not None:
        refine_deadline = time.time() + time_left
        seq, val = _subgradient_refine(best_seq, refine_deadline, max_iters=500_000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[overall SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # 4) Fallback (very unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    # store for a possible later run
    globals()["height_sequence_2"] = best_seq.copy()
    return best_seq.tolist()
