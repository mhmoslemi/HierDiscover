def _lp_step(f, M, t):
    n = len(f)
    # Build constraint matrix A (size (2n-1, n))
    A = sparse.lil_matrix((2*n-1, n), dtype=np.float64)
    for i in range(n):
        fi = f[i]
        if fi == 0.0:
            continue
        A[i:i+n, i] = fi
    A = A.tocsr()
    c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
    # Upper bounds on each g entry to keep within evaluate_sequence limits
    ub = 1000.0
    bounds = [(0.0, ub) for _ in range(n)]
    b = np.full(2*n-1, M, dtype=np.float64)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 == 0.0:
        return None
    scale = np.sqrt(2.0 * n) / sum_g0
    g = g0 * scale
    # convex combination
    f_new = (1 - t) * f + t * g
    return f_new
```

"""
Search for a non‑negative step‑function that minimises
``evaluate_sequence``.  The algorithm proceeds in several stages:

1.  Deterministic analytical seeds (Catalan, uniform, geometric) for a
    few lengths.  The seed is normalised so that Σ a_i = √(2 n); with
    this normalisation the evaluation reduces to the maximal entry of
    the auto‑convolution.

2.  Massive Dirichlet sampling (different concentration parameters) to
    discover good candidates quickly.

3.  **LP‑based mirror descent** – an iteration of the method described
    in the statement:  for the current vector *f* we solve the linear
    programme

           maximise   Σ g_i
           subject to  (f * g)(k) ≤ M   for all k,
                       g_i ≥ 0 , g_i ≤ 1000,

    where *M* is the current maximal convolution value of *f*.  The
    solution *g* is rescaled to Σ g_i = √(2 n) and mixed with *f*.
    A few such iterations dramatically reduce the bound.

4.  A final projected‑Adam refinement on the normalised vector
    (Σ = 1) – the same routine used in the reference solution – to
    polish the result.

All helper functions are defined at top‑level; no closures or lambdas
are used.  The routine respects the supplied time budget (default
1000 s) and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# -------------------------------------------------------------
# Low‑level utilities
# -------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for the sizes we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N).real[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Compute the value that ``evaluate_sequence`` would return when ``seq``
    is already normalised to sum = 1:  2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# -------------------------------------------------------------
# Analytic seed families
# -------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Coefficients proportional to binom(2i,i)/4**i.
    Normalised to sum = 1.
    """
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2.0 * i) * (2.0 * i - 1.0)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _geometric_shape(n: int, r: float = 0.999) -> np.ndarray:
    """Truncated geometric distribution a_i ∝ r**i, normalised to sum = 1."""
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


# -------------------------------------------------------------
# Random Dirichlet sampling
# -------------------------------------------------------------


def _dirichlet(alpha: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector of length n."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# -------------------------------------------------------------
# Simplex projection – used by Adam refinement
# -------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# Refinement – projected Adam
# -------------------------------------------------------------


def _refine_proj_adam(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimiser with Euclidean projection onto the simplex."""
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        cur_val = float(2.0 * n * np.max(conv))

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 500 == 0:
                print(f"[Adam] it={t:7d} val={cur_val:.6f}")

        # sub‑gradient of the maximal entry (as in the reference code)
        max_idx = int(np.argmax(conv))
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.05 / np.sqrt(t)           # simple diminishing base learning rate
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _proj_simplex(a_new)

        # occasional jitter to escape plateaus
        if t % 5000 == 0:
            noise = rng.normal(scale=1e-5, size=n)
            a = _proj_simplex(a + noise)

    return best_seq, best_val


# -------------------------------------------------------------
# LP‑based mirror descent step
# -------------------------------------------------------------


def _lp_mirror_step(f: np.ndarray, M: float, mix: float) -> np.ndarray | None:
    """
    One LP‑based update.
    Solve  maximise Σ g_i  subject to  f⋆g ≤ M  and 0 ≤ g_i ≤ 1000.
    The solution is rescaled to Σ g_i = √(2 n) and mixed with *f*
    using a convex coefficient *mix* (0 < mix ≤ 1).  Returns the new
    vector or ``None`` if the LP fails.
    """
    n = len(f)
    # Build the (2n‑1)×n constraint matrix A where
    #   (A @ g)[k] = (f ⋆ g)[k] .
    A = sparse.lil_matrix((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        fi = f[i]
        if fi == 0.0:
            continue
        # f_i contributes to rows i … i+n‑1 in column i
        A[i:i + n, i] = fi
    A = A.tocsr()

    # maximise Σ g_i  → minimise -Σ g_i
    c = -np.ones(n, dtype=np.float64)

    # upper bound for each g_i – prevents unboundedness on zero entries of f
    ub = 1000.0
    bounds = [(0.0, ub) for _ in range(n)]

    b = np.full(2 * n - 1, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs", options={"presolve": True})
    if not res.success:
        return None

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 == 0.0:
        return None

    # rescale to keep the invariant Σ g_i = √(2 n)
    scale = np.sqrt(2.0 * n) / sum_g0
    g = g0 * scale

    # convex combination
    f_new = (1.0 - mix) * f + mix * g
    return f_new
