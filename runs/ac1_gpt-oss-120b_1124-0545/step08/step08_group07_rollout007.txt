def _smooth_exponential_descent(start_seq, n, deadline):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    beta = 4.0
    it = 0
    while time.time() < deadline:
        it+=1
        # compute conv
        conv = _conv_fft(a,a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv  # since sum=1
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Smooth] it={it} best={best_val:.6f}")
        # compute weights
        # to avoid overflow, subtract max
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        w = w / w_sum
        # gradient
        rev = a[::-1]
        conv2 = _conv_fft(rev, w)   # conv length 3n-2
        grad_raw = 2.0 * conv2[n-1:n-1+n]
        # step size schedule
        lr = 0.5 / math.sqrt(it)   # can tune
        # update log a
        log_a = np.log(a + 1e-30) - lr * grad_raw
        log_a -= log_a.max()
        a = np.exp(log_a)
        a /= a.sum()
        # occasionally increase beta
        if it % 5000 == 0:
            beta = min(beta * 1.5, 200.0)
            # add small noise
            noise = np.random.normal(scale=0.001, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
    return best_seq, best_val

"""
Hybrid optimiser (smooth exponential‑gradient + annealing).

Goal
----
Minimise 2·n·max(conv(a,a)) / (sum(a)**2) for a non‑negative vector `a`
(the evaluation function `evaluate_sequence` is used only at the very end).

Approach
--------
1. **Rich seed pool** – the same families as before are generated
   (uniform, geometric, power‑law, classic windows, Dirichlet draws,
   random soft‑max).  The best seed (according to the true objective)
   is kept.

2. **Smooth exponential‑gradient descent** – a differentiable
   approximation of the `max` (log‑sum‑exp) is minimised.
   For a temperature β the auxiliary weights are
       w_k ∝ exp(β·conv_k) .
   The gradient of the smoothed objective w.r.t. `a` can be computed
   analytically as a (cross‑) convolution, so each iteration costs
   only a few FFTs.  An exponentiated‑gradient update keeps the vector
   non‑negative and automatically stays on the simplex after a
   normalisation step.  β is increased slowly (homotopy) to concentrate
   the optimisation on the true maximum.

3. **Simulated‑annealing fine‑tuning** – the best vector obtained from
   the gradient phase is further polished with the Metropolis‑Hastings
   walk from the previous version (mass‑transfer, swap, global jitter).

All work is bounded by a 1000‑second budget; the best sequence found
is stored in the global `height_sequence_1` for possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT‑based convolution (full mode)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Families of candidate vectors (all return a probability vector)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1), discretised on mid‑points.
    α∈(0,1] gives edge‑heavy shapes; α=1 is uniform.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `alpha`."""
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _make_extended_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a large, diverse collection of normalised (sum‑1) seeds."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic families
    seeds.append(_uniform(n))

    # geometric – decay factor down to ~0.997 (c up to 12)
    for c in np.linspace(0.5, 12.0, 24):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law
    for p in np.linspace(0.0, 4.0, 21):
        seeds.append(_power(n, p))

    # combined geometric‑power
    for c in np.linspace(0.5, 12.0, 12):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 5):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 12.0, 12):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classic windows
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))

    # Gaussian sweep
    for sigma_factor in np.linspace(0.06, 0.30, 9):
        seeds.append(_gaussian(n, sigma_factor))

    # Beta‑pdf family
    for alpha in np.linspace(0.3, 2.5, 12):
        seeds.append(_beta_window(n, alpha))

    # Dirichlet draws
    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, alpha, rng))

    # random soft‑max vectors
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # reuse previous if shape matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final clean‑up – clip tiny negatives, normalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on a smooth approximation of max
# ----------------------------------------------------------------------


def _smooth_exponential_descent(start_seq: np.ndarray,
                                n: int,
                                deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth log‑sum‑exp approximation of the max‑convolution.
    Returns the best vector seen (normalised to sum = 1) and its exact
    objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    beta = 4.0               # start temperature (smoothness)
    it = 0
    while time.time() < deadline:
        it += 1
        # 1. current convolution
        conv = _conv_fft(a, a)                 # length 2n‑1
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv           # sum(a)=1 ⇒ objective = 2n·max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Smooth] iter={it:7d} best={best_val:.6f}")

        # 2. soft‑max weights over the convolution entries
        #    (subtract max for numerical stability)
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:                      # safety – should not happen
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # 3. gradient of the smoothed objective ∂/∂a  (up to a constant)
        #    g_raw = Σ_k w_k ∂conv_k/∂a = 2 * (a_rev * w)[n‑1 : n‑1+n]
        rev = a[::-1]
        conv2 = _conv_fft(rev, w)               # length 3n‑2
        grad_raw = 2.0 * conv2[n - 1:n - 1 + n]  # size n

        # 4. exponentiated‑gradient update
        #    log a ← log a – η·grad_raw   (η decays like 1/√t)
        lr = 0.5 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad_raw
        #   remove global shift for numerical stability
        log_a -= log_a.max()
        a = np.exp(log_a)
        a /= a.sum()

        # 5. schedule: slowly increase β (making the approximation sharper)
        if it % 4000 == 0:
            beta = min(beta * 1.4, 200.0)   # cap to avoid overflow
            # occasional tiny jitter to escape shallow plateaus
            jitter = np.random.normal(scale=0.001, size=n)
            a = np.clip(a + jitter, 0.0, None)
            a /= a.sum()

    return best_seq, best_val

# Simulated‑annealing fine‑tuning (unchanged from the previous version)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Metropolis–Hastings walk (mass transfer / swap / global jitter)
    with a slowly cooling temperature schedule.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-7, cur_val * 0.08)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        move_type = rng.random()
        if move_type < 0.70:                     # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.random() * 0.20 * cur[i]
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        elif move_type < 0.85:                   # swap
            i, j = rng.integers(0, n, size=2)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:                                    # global jitter
            noise = rng.normal(scale=0.0015, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0:
                new_seq /= total
            else:
                new_seq = _uniform(n)

        # projection back onto simplex
        new_seq = _project_simplex(new_seq)

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:8d} best={new_val:.6f}")

        if it % 2000 == 0:
            # occasional stronger perturbation around the current best
            jitter = rng.normal(scale=0.003, size=n)
            cand = best + jitter
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand)
            cand_val = _objective(cand)
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
                print(f"[Anneal] (jump) it={it:8d} best={cand_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that (approximately)
    minimises ``evaluate_sequence``.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # Length – 4000 offers a good balance between quality and speed
    n = 4000

    # Try to reuse a previous construction if available
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 0) evaluate previous construction (if any) and keep it as a seed
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    if previous is not None and previous.shape == (n,):
        val = _objective(previous)
        best_seq = previous.copy()
        best_val = val
        print(f"[Prev] objective={val:.6f}")

    # ------------------------------------------------------------------
    # 1) Build a very rich seed pool and keep the top few
    # ------------------------------------------------------------------
    seeds = _make_extended_seeds(n, previous)
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    if seed_vals[0][1] < best_val:
        best_val = seed_vals[0][1]
        best_seq = seed_vals[0][0].copy()
        print(f"[Seed] new best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Smooth exponential‑gradient descent (most of the time budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        seq, val = _smooth_exponential_descent(best_seq, n, deadline)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Overall Smooth] best={val:.6f}")

    # ------------------------------------------------------------------
    # 3) Simulated‑annealing fine‑tuning (uses remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(best_seq, n, deadline, max_iters=None)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Overall Anneal] best={val:.6f}")

    # ------------------------------------------------------------------
    # 4) Fallback – should never happen
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[Fallback] using uniform distribution")

    print("\n=== Finished ===")
    print(f"Best objective ≈ {best_val:.6f} (length {len(best_seq)})")
    # store for possible later runs
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
