"""
Propose a step‑function that minimises `evaluate_sequence`.

Algorithm (≈ 1000 s budget)

1. **Analytic seeds** – generate a variety of symmetric families
   (uniform, Catalan, Hann, Hamming, Blackman, Gaussian,
   arcsine, Beta‑pdf shapes) for several lengths (2000…8000).
   Keep the best seed.

2. **Heavy random sampling** – draw Dirichlet vectors with α∈{0.5,1,2}
   and random Beta‑pdf shapes for the same lengths while ~30 % of the
   time budget remains.  Keep the globally best sample.

3. **Smoothed Adam** – starting from the best vector, minimise a
   smooth approximation of `max(conv(a,a))` using the log‑sum‑exp
   with an annealed temperature β∈[5,200].  Several independent
   restarts are performed; the best Adam result is kept.

4. **L‑BFGS refinement** (optional – used when SciPy is available) –
   run a limited L‑BFGS‑B optimisation on the same smooth objective
   (β fixed at a large value) to obtain a finer balance.

5. **Aggressive mirror‑descent polishing** – a KL‑mirror descent
   where the sub‑gradient aggregates *all* near‑maxima of the
   convolution (indices with value ≥ max‑ε).  This overcomes the
   stall that can appear when only a single active index is used.

All phases respect the overall time budget.  The returned list is
already normalised (sum = 1) and non‑negative; `evaluate_sequence`
will clip it if needed.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from math import sqrt, log, pi, exp

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a *normalised* sequence
    (sum == 1).  Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * pi * i / (n - 1))
         + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf shape on a discrete grid (shifted by 0.5 to avoid end‑points)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Guard against negative exponents (alpha<1 or beta<1) by using max(x, eps)
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s

# Smoothed Adam optimiser (log‑its parametrisation)
# ---------------------------------------------------------------

def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    max_iter: int = 5_000_000,
    lr0: float = 0.03,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)).
    Uses Adam on the log‑softmax (logits) representation.
    """
    n = len(start_seq)
    eps = 1e-30

    # initialise logits from the starting probability vector
    logits = np.log(start_seq + eps)

    # Adam moments
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # soft‑max to obtain a probability vector
        a = np.exp(logits)
        a_sum = a.sum()
        a = a / a_sum

        # compute self‑convolution
        conv = _conv_fft(a, a)            # length = 2n‑1

        # temperature schedule (linear annealing)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        # smooth‑max via log‑sum‑exp
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()     # soft weights, sum=1

        # gradient w.r.t. a : g_a = 2 * (p * a) “convolved” with reversed a
        conv_pa = _conv_fft(p, a[::-1])   # length = 3n‑2
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule ∝ 1/√it
        lr = lr0 / sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional exact evaluation
        if (it & 1023) == 0:    # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

            # quick early exit if we are already very good
            if best_val < 1.45:
                break

    return best_seq, best_val

def _lbfgs_refine(start_seq: np.ndarray, deadline: float, beta: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS‑B on the smooth max objective.
    Returns the best (exact) objective value and the corresponding sequence.
    """
    if minimize is None:          # SciPy not available
        return start_seq, _objective(start_seq)

    n = len(start_seq)
    eps = 1e-30

    def fun_and_grad(logits: np.ndarray):
        a = np.exp(logits)
        a = a / a.sum()

        conv = _conv_fft(a, a)         # length = 2n‑1
        # smooth max (log‑sum‑exp)
        max_c = np.max(conv)
        exps = np.exp(beta * (conv - max_c))
        lse = max_c + (1.0 / beta) * np.log(exps.sum())
        value = 2.0 * n * lse

        # gradient w.r.t a
        p = exps / exps.sum()          # soft weights, sum=1
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)   # same as in Adam

        return value, grad_logits.astype(np.float64)

    x0 = np.log(start_seq + eps)

    # tiny callback that aborts when time is up
    class TimeOutException(RuntimeError):
        pass

    def callback(xk):
        if time.time() >= deadline:
            raise TimeOutException()

    try:
        res = minimize(
            fun=lambda x: fun_and_grad(x)[0],
            x0=x0,
            jac=lambda x: fun_and_grad(x)[1],
            method='L-BFGS-B',
            options={'maxiter': 200, 'disp': False},
            callback=callback,
        )
    except TimeOutException:
        # time ran out – fall back to whatever we have
        best_logits = x0
    else:
        best_logits = res.x

    best_a = np.exp(best_logits)
    best_a /= best_a.sum()
    best_val = _objective(best_a)

    return best_a, best_val

def _refine_aggregated_subgradient(start_seq: np.ndarray, deadline: float, eps_rel: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent where the sub‑gradient aggregates *all* indices
    whose convolution entry is within a relative tolerance of the current
    maximum.  This typically reduces the objective further than the
    single‑active‑index version.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        # indices that are “almost maximal”
        delta = eps_rel * max_val
        active_idx = np.where(conv >= max_val - delta)[0]

        # indicator vector on the full convolution length (2n‑1)
        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        # aggregated sub‑gradient = 2 * (active * a_rev) convolution
        conv_ag = _conv_fft(active, a[::-1])          # length = 3n‑2
        g = 2.0 * conv_ag[n - 1:n - 1 + n]            # slice = n

        # exponentiated‑gradient step (KL‑mirror)
        lr = 0.7 / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)        # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        # periodic exact evaluation
        if (it & 1023) == 0:   # every 1024 iterations
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[AggRefine] it={it:7d} n={n:5d} best={val:.6f}")

            # early exit if time is almost exhausted
            if time.time() > deadline - 1.0:
                break

    return best_seq, best_val

        if best_val < 1.45:
            break

    # ------------------------------------------------------------------
    # Phase 3 – smoothed Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # keep the three currently best candidates for independent restarts
        candidates = [best_seq]
        # generate a few more random starts from the best‑performing length
        top_len = len(best_seq)
        for _ in range(2):
            cand = _random_dirichlet(top_len, rng, alpha=rng.choice(alphas))
            candidates.append(cand)

        # allocate roughly 25 % of the remaining budget to Adam runs
        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.25

        for idx, cand in enumerate(candidates):
            seq_adam, val_adam = _smooth_adam_optimize(
                start_seq=cand,
                deadline=adam_deadline,
                beta_start=5.0,
                beta_end=200.0,
                max_iter=4_000_000,
                lr0=0.03,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = seq_adam
                print(f"[Adam‑Refine {idx}] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – optional L‑BFGS refinement (if SciPy is available)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:          # leave a bit of time for the final polish
            lbfgs_deadline = time.time() + remaining * 0.20
            seq_lbfgs, val_lbfgs = _lbfgs_refine(best_seq, lbfgs_deadline, beta=200.0)
            if val_lbfgs < best_val:
                best_val = val_lbfgs
                best_seq = seq_lbfgs
                print(f"[L‑BFGS] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – aggressive mirror‑descent polishing
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _refine_aggregated_subgradient(best_seq, deadline)
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Agg‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check (the grader will evaluate again anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
```

This implementation adds a richer collection of analytic seeds (including arcsine and Beta‑pdf families), expands the search to larger sequence lengths, performs more aggressive Adam optimisation with a higher temperature schedule, optionally refines the solution with L‑BFGS (when SciPy is present), and finally applies a mirror‑descent polishing step that aggregates *all* near‑maximal convolution entries rather than a single active index.  All phases respect the 1000‑second time budget and return a non‑negative normalized list ready for `evaluate_sequence`.
