def _adam_opt_knots(logits_init, n, deadline, max_iters=200000):
    v = logits_init.copy()
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta = 0.5
    beta_mul = 1.0005   # slower increase
    beta_max = 400.0
    best_val = float('inf')
    best_logits = None
    t = 0
    eps = 1e-30
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute sequence and objective
        seq = logits_to_full(v, n)
        cur_obj = _objective(seq)
        if cur_obj < best_val:
            best_val = cur_obj
            best_logits = v.copy()
            print(f"[AdamKnots] it={it} best={cur_obj:.6f}")
        # compute gradient of smoothed objective:
        # compute conv and w as above (w softmax with temperature beta)
        conv = _conv_fft(seq, seq)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0:
            # keep same?
            continue
        w /= w_sum
        # grad a
        rev_seq = seq[::-1]
        full = _conv_fft(rev_seq, w)  # length 3n-2
        grad_a = 2.0 * full[n-1:n-1+n]
        # Now backprop to logits:
        # Compute gradient w.r.t soft (knots)
        K = len(v)
        # compute interpolation mapping
        pos = np.linspace(0, K-1, n)
        left = np.floor(pos).astype(int)
        right = np.clip(left+1, 0, K-1)
        alpha = pos - left
        # compute grad_soft of shape K
        grad_soft = np.zeros(K, dtype=np.float64)
        np.add.at(grad_soft, left, grad_a * (1 - alpha))
        np.add.at(grad_soft, right, grad_a * alpha)
        # compute soft values
        soft = _softmax(v)
        # gradient of logits via softmax chain rule
        dot = np.dot(soft, grad_soft)
        grad_logits = soft * (grad_soft - dot)
        # Adam update
        lr0 = 0.8
        t += 1
        lr = lr0 / np.sqrt(t)
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        m = beta1 * m + (1 - beta1) * grad_logits
        v2 = beta2 * v2 + (1 - beta2) * (grad_logits ** 2)
        m_hat = m / (1 - beta1 ** t)
        v2_hat = v2 / (1 - beta2 ** t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)
        # increase beta
        beta = min(beta * beta_mul, beta_max)
    if best_logits is None:
        best_logits = logits_init.copy()
    return best_logits, best_val

"""
Step‑function optimiser using a low‑dimensional spline parametrisation.
The search proceeds in three phases:

1.  Initialise a small number of “knot” logits (default 80) by taking the
    current best solution (if any) or a uniform start.
2.  Run a long Adam optimisation of a smoothed‑max surrogate.
    The full step‑function is obtained by linearly interpolating the
    softmax of the knot logits and re‑normalising to sum = 1.
3.  Refine the resulting full‑length vector with a mirror‑descent
    sub‑gradient method and a brief simulated‑annealing polishing.
The algorithm respects the 1000 s budget and stores the best sequence in the
global name ``height_sequence_1``.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Low‑level utilities
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _uniform(n: int) -> np.ndarray:
    """Uniform probability vector of length n."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _objective(seq: np.ndarray) -> float:
    """
    True objective used for bookkeeping.
    ``seq`` must be non‑negative and sum≈1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# --------------------------------------------------------------
# Interpolation helpers (K knots → n points)
# --------------------------------------------------------------

def _interp_from_soft(soft: np.ndarray, n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector ``soft`` (size K)
    to a length‑n vector, followed by a normalisation to sum = 1.
    """
    K = soft.shape[0]
    # positions of the n grid points in the knot index space
    pos = np.linspace(0, K - 1, n, dtype=np.float64)
    left = np.floor(pos).astype(np.int64)
    right = np.clip(left + 1, 0, K - 1)
    alpha = pos - left
    a_raw = soft[left] * (1.0 - alpha) + soft[right] * alpha
    s = a_raw.sum()
    if s == 0.0:
        return _uniform(n)
    return a_raw / s


def _seq_from_logits(logits: np.ndarray, n: int) -> np.ndarray:
    """
    Convert raw logits (size K) into a normalised length‑n step function.
    """
    soft = _softmax(logits)
    return _interp_from_soft(soft, n)


# --------------------------------------------------------------
# Adam optimisation on the low‑dimensional knot logits
# --------------------------------------------------------------

def _adam_opt_knots(logits_init: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 300000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed‐max objective.
    Returns the best logits encountered and its true objective value.
    """
    K = logits_init.shape[0]
    v = logits_init.copy()
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5                 # start temperature for the log‑sum‑exp
    beta_mul = 1.001
    beta_max = 300.0

    best_val = float("inf")
    best_logits = None

    eps = 1e-30
    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- current sequence and true objective -----
        seq = _seq_from_logits(v, n)
        cur_val = _objective(seq)
        if cur_val < best_val:
            best_val = cur_val
            best_logits = v.copy()
            print(f"[AdamKnots] it={it:7d} best={cur_val:.6f}")

        # ----- smoothed‑max gradient (log‑sum‑exp) -----
        conv = _conv_fft(seq, seq)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            # numerical failure – skip this step
            beta = min(beta * beta_mul, beta_max)
            continue
        w /= w_sum

        # gradient wrt the normalised sequence (size n)
        rev_seq = seq[::-1]
        full = _conv_fft(rev_seq, w)  # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # ----- back‑prop through normalisation -----
        raw = _interp_from_soft(_softmax(v), n)      # a_raw before the final renorm
        s_raw = raw.sum()
        if s_raw == 0.0:
            s_raw = 1.0
        total_grad = grad_a.sum()
        grad_raw = (grad_a - (total_grad * raw / s_raw)) / s_raw

        # ----- back‑prop through linear interpolation -----
        pos = np.linspace(0, K - 1, n, dtype=np.float64)
        left = np.floor(pos).astype(np.int64)
        right = np.clip(left + 1, 0, K - 1)
        alpha = pos - left

        grad_soft = np.zeros(K, dtype=np.float64)
        np.add.at(grad_soft, left, grad_raw * (1.0 - alpha))
        np.add.at(grad_soft, right, grad_raw * alpha)

        # ----- back‑prop through softmax -----
        soft = _softmax(v)
        dot = np.dot(soft, grad_soft)
        grad_logits = soft * (grad_soft - dot)

        # ----- Adam update -----
        t = it
        lr = lr0 / np.sqrt(t)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_logits ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

        # ----- increase temperature -----
        beta = min(beta * beta_mul, beta_max)

    if best_logits is None:
        best_logits = logits_init.copy()
    return best_logits, best_val


# --------------------------------------------------------------
# Exact sub‑gradient refinement on the full‑length vector
# --------------------------------------------------------------

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution
    objective. Returns the best sequence seen and its objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the max index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# --------------------------------------------------------------
# Simple simulated‑annealing polishing
# --------------------------------------------------------------

def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 150000,
                         init_temp_factor: float = 0.02,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Randomly reshuffle a small subset of coordinates.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)           # size of the perturbed subset 2…6
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small guard margin so that we finish before the hard timeout
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    n = 4000                         # base length
    K = 80                           # number of knots (low‑dimensional)

    # ------------------------------------------------------------------
    # Initialise knot logits (try to reuse the previous solution)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and len(previous) == n:
        prev_arr = np.asarray(previous, dtype=np.float64)
        # down‑sample by simple averaging over equal sized blocks
        block = n // K
        knots = np.empty(K, dtype=np.float64)
        for i in range(K):
            start = i * block
            end = start + block if i < K - 1 else n
            knots[i] = prev_arr[start:end].mean()
        # avoid exact zeros before the log
        knots = np.maximum(knots, 1e-30)
        init_logits = np.log(knots)
        print("[Init] Seeded logits from previous best.")
    else:
        init_logits = np.zeros(K, dtype=np.float64)
        print("[Init] Starting from uniform logits.")

    # ------------------------------------------------------------------
    # Phase 1 – long Adam optimisation on the spline knots
    # ------------------------------------------------------------------
    print("\n--- Phase 1: Adam on knots ---")
    logits_best, val_best = _adam_opt_knots(init_logits, n, deadline,
                                            max_iters=300000)
    best_seq = _seq_from_logits(logits_best, n)
    best_val = _objective(best_seq)
    print(f"[Phase 1] best surrogate objective ≈ {val_best:.6f}, true objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient refinement on the full vector
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 2: Sub‑gradient refinement ---")
        seq_sub, val_sub = _subgradient_refine(best_seq, n, deadline,
                                               max_iters=250000)
        if val_sub < best_val:
            best_seq, best_val = seq_sub, val_sub
            print(f"[Phase 2] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 3: Simulated annealing ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline,
                                              max_iters=150000)
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print(f"[Phase 3] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling (double the resolution) if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_big = n * 2
        print("\n--- Upsampling to longer length ---")
        x_old = np.linspace(0.0, 1.0, n, dtype=np.float64)
        x_big = np.linspace(0.0, 1.0, n_big, dtype=np.float64)
        upsampled = np.interp(x_big, x_old, best_seq)
        upsampled = upsampled / upsampled.sum()
        seq_big, val_big = _subgradient_refine(upsampled, n_big,
                                               deadline, max_iters=150000)
        if val_big < best_val:
            best_seq, best_val = seq_big, val_big
            n = n_big
            print(f"[Upsample] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (call evaluate_sequence once)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

"""
Step‑function optimiser using a low‑dimensional spline parametrisation.

The algorithm proceeds in three stages:

1. Initialise a small set of “knot” logits (default 80).  If a previous
   solution exists it is down‑sampled to provide a warm start,
   otherwise a uniform start is used.

2. Run a long Adam optimisation on a smoothed‑max surrogate.
   The full step‑function is obtained by linearly interpolating the
   softmax of the knot logits and re‑normalising to sum = 1.  The
   gradient is back‑propagated through the interpolation and the
   normalisation.

3. Refine the resulting full‑length vector with a mirror‑descent
   (exponentiated‑gradient) sub‑gradient method, followed by a brief
   simulated‑annealing polishing.  If time permits the vector is
   up‑sampled (doubling the resolution) and refined again.

The best sequence is stored globally as ``height_sequence_1`` and
returned as a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Low‑level utilities
# --------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _uniform(n: int) -> np.ndarray:
    """Uniform probability vector of length n."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _objective(seq: np.ndarray) -> float:
    """
    True objective (scale‑invariant): 2·n·max(conv(seq,seq)).
    The sequence is assumed to be non‑negative and sum≈1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# --------------------------------------------------------------
# Interpolation helpers (K knots → n points)
# --------------------------------------------------------------


def _interp_raw(soft: np.ndarray, n: int, left: np.ndarray,
                right: np.ndarray, alpha: np.ndarray) -> np.ndarray:
    """
    Linear interpolation of ``soft`` (size K) to a length‑n vector,
    *without* final normalisation.
    """
    return soft[left] * (1.0 - alpha) + soft[right] * alpha


def _interp_normalised(soft: np.ndarray, n: int, left: np.ndarray,
                       right: np.ndarray, alpha: np.ndarray) -> np.ndarray:
    """
    Linear interpolation followed by normalisation to sum = 1.
    """
    raw = _interp_raw(soft, n, left, right, alpha)
    s = raw.sum()
    return raw / s if s != 0.0 else _uniform(n)


def _seq_from_logits(logits: np.ndarray, n: int,
                     left: np.ndarray, right: np.ndarray,
                     alpha: np.ndarray) -> np.ndarray:
    """Convert raw logits (size K) into a normalised length‑n step function."""
    soft = _softmax(logits)
    return _interp_normalised(soft, n, left, right, alpha)


# --------------------------------------------------------------
# Adam optimisation on the low‑dimensional knot logits
# --------------------------------------------------------------


def _adam_opt_knots(logits_init: np.ndarray, n: int,
                    deadline: float,
                    max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed‑max surrogate.
    Returns the best logits encountered and its true objective value.
    """
    K = logits_init.shape[0]

    # pre‑compute interpolation indices – they never change
    pos = np.linspace(0.0, K - 1.0, n, dtype=np.float64)
    left = np.floor(pos).astype(np.int64)
    right = np.clip(left + 1, 0, K - 1).astype(np.int64)
    alpha = pos - left

    v = logits_init.copy()
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5                 # start temperature for the log‑sum‑exp
    beta_mul = 1.001
    beta_max = 300.0

    best_val = float("inf")
    best_logits = None

    eps = 1e-30
    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---- current sequence and true objective (for bookkeeping) ----
        seq = _seq_from_logits(v, n, left, right, alpha)
        cur_val = _objective(seq)
        if cur_val < best_val:
            best_val = cur_val
            best_logits = v.copy()
            print(f"[AdamKnots] it={it:7d} best={cur_val:.6f}")

        # ---- smoothed‑max gradient (log‑sum‑exp) ----
        conv = _conv_fft(seq, seq)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            beta = min(beta * beta_mul, beta_max)
            continue
        w /= w_sum

        # gradient wrt the *normalised* sequence (size n)
        rev_seq = seq[::-1]
        full = _conv_fft(rev_seq, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

        # ---- back‑prop through the normalisation ----
        soft = _softmax(v)
        raw = _interp_raw(soft, n, left, right, alpha)   # un‑normalised interpolant
        s_raw = raw.sum()
        if s_raw == 0.0:                              # should never happen
            s_raw = 1.0
        a = raw / s_raw                               # this equals ``seq``

        # scalar needed for the chain rule
        dot = np.dot(grad_a, a)                       # ⟨grad_a , a⟩

        # gradient w.r.t the *raw* interpolated vector
        grad_raw = (grad_a - dot) / s_raw

        # ---- back‑prop through linear interpolation ----
        grad_soft = np.zeros(K, dtype=np.float64)
        np.add.at(grad_soft, left, grad_raw * (1.0 - alpha))
        np.add.at(grad_soft, right, grad_raw * alpha)

        # ---- back‑prop through softmax (logits) ----
        dot2 = np.dot(soft, grad_soft)
        grad_logits = soft * (grad_soft - dot2)

        # ---- Adam update ----
        t = it
        lr = lr0 / np.sqrt(t)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_logits ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

        # increase temperature (sharpen the surrogate)
        beta = min(beta * beta_mul, beta_max)

    if best_logits is None:
        best_logits = logits_init.copy()
    return best_logits, best_val


# --------------------------------------------------------------
# Exact sub‑gradient refinement on the full‑length vector
# --------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray, n: int,
                        deadline: float,
                        max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution
    objective. Returns the best sequence seen and its objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# --------------------------------------------------------------
# Simple simulated‑annealing polishing
# --------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 150_000,
                         init_temp_factor: float = 0.02,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Randomly reshuffle a small subset of coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)           # 2 … 6 coordinates
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123,

    """
    # reproducibility

    start_time = time.time()
    deadline = start_time + budget_s - 5.0        # safety margin

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    n = 4000            # base resolution
    K = 80              # number of spline knots (low‑dimensional)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")

        prev_arr = np.asarray(previous, dtype=np.float64)
        block = n // K
        knots = np.empty(K, dtype=np.float64)
        for i in range(K):
            start = i * block
            end = start + block if i < K - 1 else n
            knots[i] = prev_arr[start:end].mean()
        knots = np.maximum(knots, 1e-30)
        init_logits = np.log(knots)
        print("[Init] Seeded logits from previous best.")
    else:

        print("[Init] Starting from uniform logits.")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    print("\n--- Phase 1: Adam on spline knots ---")
    logits_best, surrogate_best = _adam_opt_knots(init_logits, n,
                                                  deadline,
                                                  max_iters=300_000)
    best_seq = _seq_from_logits(logits_best, n,
                                left=np.floor(np.linspace(0, K - 1, n)).astype(np.int64),
                                right=np.clip(np.floor(np.linspace(0, K - 1, n)).astype(np.int64) + 1, 0, K - 1),
                                alpha=(np.linspace(0, K - 1, n) % 1))
    best_val = _objective(best_seq)

    # Optional up‑sampling (double the resolution) if time permits
