"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

The search consists of five stages:

1. **Analytic seeds** – a handful of deterministic families
   (uniform, Catalan, geometric, power‑law with various exponents and
   shifts).  All candidates are normalised to sum = 1 and the best one
   is kept.

2. **Random Dirichlet sampling** – while time permits many Dirichlet
   vectors are drawn with different concentration parameters and
   lengths.  The best result is retained.

3. **Exponentiated‑gradient mirror descent** – a sub‑gradient method
   (multiplicative updates) starting from the best candidate from stage 2.
   It runs for a fraction of the remaining budget.

4. **Projected Adam** – an adaptive first‑order method with Euclidean
   projection onto the probability simplex.  The best vector seen so
   far is saved.

5. **Fine‑grained hill‑climbing** – tiny random mass transfers between
   two coordinates; an improvement is kept immediately.  This cheap
   local search often squeezes a few extra points out of the current
   optimum.

The routine respects the supplied time budget (default 1000 s) and
returns a plain ``list`` of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Objective that ``evaluate_sequence`` would return when ``seq`` is already
    normalised to sum = 1:  2·n·max(conv(seq,seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Coefficients proportional to binom(2i,i)/4**i (the series of (1‑z)^‑½).
    After truncation they give an almost constant auto‑convolution.
    """
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _geometric_shape(n: int, r: float = 0.999) -> np.ndarray:
    """
    Truncated geometric distribution a_i ∝ r**i, normalised to sum = 1.
    """
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


def _powerlaw_shape(n: int, p: float, shift: float = 0.0) -> np.ndarray:
    """
    Power‑law shape a_i ∝ (i+1+shift)^{-p}.
    ``p`` may be any positive number, ``shift`` ≥ 0 avoids the singularity at i=0.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + shift
    a = idx ** (-p)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    return a / total


# ----------------------------------------------------------------------
# Random Dirichlet sampling
# ----------------------------------------------------------------------


def _dirichlet(alpha: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector and normalise to sum = 1."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by Adam refinement
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Refinement – exponentiated gradient (mirror descent)
# ----------------------------------------------------------------------


def _refine_expgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent on the max‑convolution
    objective.  Returns the best sequence observed before *deadline*.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ExpGrad] it={it:6d} val={cur_val:.6f}")

        # sub‑gradient for the active lag
        tol = 1e-12
        tie_idxs = np.where(conv >= max_val - tol)[0]
        k = int(np.random.choice(tie_idxs))

        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.9 / np.sqrt(it)          # a slightly larger prefactor than before
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # improve numerical stability
        a = np.exp(log_a)
        a /= a.sum()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Refinement – projected Adam (adaptive moments)
# ----------------------------------------------------------------------


def _refine_proj_adam(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with Euclidean projection onto the simplex after each step.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _objective(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 500 == 0:
                print(f"[Adam] it={t:6d} val={cur_val:.6f}")

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.07 / np.sqrt(t)          # a modest but not too aggressive decay
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _proj_simplex(a_new)

        # occasional jitter to escape plateaus
        if t % 4000 == 0:
            noise = rng.normal(scale=1e-5, size=n)
            a = _proj_simplex(a + noise)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Refinement – cheap hill‑climbing (mass transfer)
# ----------------------------------------------------------------------


def _refine_hillclimb(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: repeatedly move a tiny fraction of mass from a
    random index to another one.  Accept only improving moves.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _objective(a)

    rng = np.random.default_rng()
    step_frac = 0.02  # initial relative step size

    while time.time() < deadline:
        # pick two distinct positions
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] == 0.0:
            continue
        delta = step_frac * a[i]          # amount to move
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # sum is preserved, non‑negativity holds by construction
        cur_val = _objective(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()
            a = a_new
            step_frac = max(step_frac * 0.99, 1e-5)   # slowly shrink step
            if best_val < 1.49:      # a tiny early‑stop shortcut
                print(f"[Hill] improved to {best_val:.6f}")
                break
        else:
            # very occasional random acceptance (simulated annealing flavour)
            if rng.random() < 1e-4:
                a = a_new
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given
    time budget and returns the best sequence found (as a plain list).
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final sanity check
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Lengths to be examined (larger length gives a finer theoretical bound)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to update the current best
    # ------------------------------------------------------------------
    def _update(seq_arr: np.ndarray, val: float, tag: str):
        nonlocal best_seq, best_val
        if val < best_val:
            best_val = val
            best_seq = seq_arr.copy()
            print(f"[{tag}] n={len(seq_arr):5d} val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic analytic families
    # ------------------------------------------------------------------
    for n in lengths:
        # uniform
        _update(_uniform(n), _objective(_uniform(n)), "Uniform")
        # Catalan (already a good baseline)
        try:
            cat = _catalan_shape(n)
            _update(cat, _objective(cat), "Catalan")
        except Exception:
            pass

        # geometric shapes with several r
        for r in np.linspace(0.97, 0.999, 7):
            geo = _geometric_shape(n, r)
            _update(geo, _objective(geo), f"Geo r={r:.3f}")

        # power‑law families (exponent p and optional shift)
        for p in np.linspace(0.4, 0.8, 9):
            pl = _powerlaw_shape(n, p, shift=0.0)
            _update(pl, _objective(pl), f"Pow p={p:.2f}")

            for shift in (0.5, 1.0, 2.0):
                pl_shift = _powerlaw_shape(n, p, shift=shift)
                _update(pl_shift, _objective(pl_shift), f"Pow p={p:.2f} s={shift}")

    # ------------------------------------------------------------------
    # Stage 2 – random Dirichlet sampling (different α, different lengths)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    # spend about 40 % of the remaining budget on pure sampling
    sample_deadline = now + remaining * 0.40
    alphas = [0.3, 0.5, 1.0, 2.0, 4.0]

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = float(rng.choice(alphas))
        seq = _dirichlet(alpha, n, rng)
        val = _objective(seq)
        _update(seq, val, f"Dir α={alpha:.1f}")
        # early exit if we already beat the target
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # Stage 3 – exponentiated‑gradient mirror descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + (deadline - time.time()) * 0.25
        seq_exp, val_exp = _refine_expgradient(best_seq, exp_deadline)
        _update(seq_exp, val_exp, "ExpGrad")

    # ------------------------------------------------------------------
    # Stage 4 – projected Adam refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = time.time() + (deadline - time.time()) * 0.30
        seq_adam, val_adam = _refine_proj_adam(best_seq, adam_deadline)
        _update(seq_adam, val_adam, "Adam")

    # ------------------------------------------------------------------
    # Stage 5 – cheap hill‑climbing for a final squeeze
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        seq_hill, val_hill = _refine_hillclimb(best_seq, hill_deadline)
        _update(seq_hill, val_hill, "HillClimb")

    # ------------------------------------------------------------------
    # Final sanity check – the grader will also call evaluate_sequence
    # ------------------------------------------------------------------
    try:
        # the grader works with a plain Python list
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
