"""
Optimiser for the step‑function objective.

The aim is to *minimise*  

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the full linear convolution of a non‑negative vector *a*
with itself.  The algorithm works in three phases:

1. **Seeding** – a rich collection of candidate shapes (uniform,
   geometric / power families, classic windows, random soft‑max vectors,
   plus a few random “skewed’’ families) is generated and the best few
   are retained.

2. **Smooth‑max Adam optimisation** – the non‑smooth max‑norm is
   approximated by a log‑sum‑exp (soft‑max) surrogate with a temperature
   parameter ``α`` that is increased during the run.  Adam is used on the
   logits of a soft‑max parametrisation (so the probability simplex is
   satisfied automatically).  After each update the vector is sorted in
   non‑increasing order (a heuristic that favours low autocorrelation) and
   re‑normalised to the simplex.

3. **Final annealing** – a Metropolis–Hastings walk with relatively
   large mass‑transfer moves finishes the search using any remaining time.

The routine respects the supplied time budget, prints a short progress
log and finally returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (real FFT, full mode)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution (real FFT)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple families (all return a *probability* vector on the simplex)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


def _beta_general(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    e = np.exp(logits - np.max(logits))
    return e / e.sum()


def _skewed_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random mixture of geometric and power decay – encourages asymmetry."""
    r = 1.0 - rng.uniform(0.01, 0.45) / n          # r close to 1
    p = rng.uniform(0.0, 2.5)
    a = (r ** np.arange(n, dtype=np.float64)) / ((np.arange(1, n + 1)) ** p)
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse collection of normalised start vectors."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic families
    seeds.append(_uniform(n))
    for r in np.linspace(0.93, 0.999, 7):
        seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 3.0, 9):
        seeds.append(_power(n, p))
    for r in np.linspace(0.93, 0.999, 5):
        for p in np.linspace(0.0, 2.0, 5):
            seeds.append(_geom_power(n, r, p))

    # beta families (both symmetric and asymmetric)
    for a in np.linspace(0.1, 0.9, 5):
        seeds.append(_beta_general(n, a, a))
    for a in np.linspace(0.1, 0.9, 3):
        for b in np.linspace(0.1, 0.9, 3):
            if abs(a - b) > 1e-12:
                seeds.append(_beta_general(n, a, b))

    # random soft‑max vectors
    for _ in range(12):
        seeds.append(_random_softmax(n, rng))

    # a few hand‑crafted skewed seeds
    for _ in range(8):
        seeds.append(_skewed_seed(n, rng))

    # reuse previous construction if compatible
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final clean‑up – enforce non‑negativity and normalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    return cleaned


# ----------------------------------------------------------------------
# Projection onto the probability simplex (used by the annealer)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_smoothmax(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 800000) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp (smooth‑max) surrogate.
    The vector is kept on the simplex via a soft‑max of logits.
    After each step we also enforce a monotone‑decreasing shape
    (heuristic that tends to reduce the autocorrelation centre).
    """
    eps = 1e-30
    # initialise logits from the start vector (soft‑max inverse)
    logits = np.log(start_seq + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999
    lr0 = 0.08           # base learning rate (will be scaled)
    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    # schedule for smoothness α (starts small, grows to large)
    def _alpha(it):
        # exponential growth until a ceiling
        if it < 150000:
            return 20.0
        elif it < 400000:
            return 200.0
        else:
            # slowly increase toward 2000
            return min(2000.0, 200.0 + 0.001 * (it - 400000))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # 1) current probability vector
        a = np.exp(logits - np.max(logits))  # stable soft‑max
        a /= a.sum()

        # 2) smooth‑max weights
        alpha = _alpha(it)
        conv = _conv_fft(a, a)
        m_conv = conv.max()
        exps = np.exp(alpha * (conv - m_conv))
        w = exps / exps.sum()               # soft‑max over convolution entries

        # 3) gradient of the surrogate wrt a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # 4) back‑prop through the soft‑max parametrisation
        #    dL/dlogits = a * (grad_a - ⟨grad_a, a⟩)
        dot = np.dot(grad_a, a)
        grad_logits = a * (grad_a - dot)

        # 5) Adam update on logits
        lr = lr0 / math.sqrt(it)            # gradual decay
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # 6) optional monotone‑decreasing projection (helps a lot)
        a_new = np.exp(logits - np.max(logits))
        a_new /= a_new.sum()
        a_new = np.sort(a_new)[::-1]        # enforce non‑increasing order
        a_new /= a_new.sum()
        logits = np.log(a_new + eps)

        # 7) occasional exact evaluation
        if it % 2000 == 0:
            val = _objective(a_new)
            if val < best_val:
                best_val = val
                best_seq = a_new.copy()
                print(f"[Adam] it={it:7d} best={val:.6f}")

    # final evaluation of the last iterate
    final_a = np.exp(logits - np.max(logits))
    final_a /= final_a.sum()
    final_a = np.sort(final_a)[::-1]
    final_a /= final_a.sum()
    final_val = _objective(final_a)
    if final_val < best_val:
        best_val = final_val
        best_seq = final_a.copy()

    return best_seq, best_val

def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Metropolis–Hastings walk with generous mass‑transfer steps."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # start temperature a fraction of the current objective (larger than before)
    T0 = max(1e-6, cur_val * 0.08)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick source and destination indices (different)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # transfer a *moderate* fraction of the source mass (up to 20 %)
        delta = rng.random() * 0.20 * cur[i]
        if delta == 0.0:
            continue
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # every few hundred steps propose a small Gaussian jitter
        if it % 400 == 0:
            noise = rng.normal(scale=0.0015, size=n)
            jitter = cur + noise
            jitter = np.clip(jitter, 0.0, None)
            if jitter.sum() == 0.0:
                jitter = _uniform(n)
            else:
                jitter /= jitter.sum()
            jitter_val = _objective(jitter)
            accept = jitter_val < cur_val or rng.random() < math.exp(-(jitter_val - cur_val) / (T + 1e-12))
            if accept:
                cur = jitter
                cur_val = jitter_val
                if jitter_val < best_val:
                    best_val = jitter_val
                    best = jitter.copy()
                    print(f"[Anneal] it={it:7d} (jitter) best={jitter_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  Returns the best sequence discovered as a list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – a few seconds are needed for the grader to collect output
    deadline = start_time + budget_s - 4.0

    # ------------------------------------------------------------------
    # Length – 4000 offers a good trade‑off (roughly 32 MiB of FFT buffers)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Pull a previously‑found construction (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 0) evaluate the previous vector (if it has the right length)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    if previous is not None and previous.shape == (n,):
        val = _objective(previous)
        best_seq = previous.copy()
        best_val = val
        print(f"[prev] n={n} val={val:.6f}")

    # ------------------------------------------------------------------
    # 1) generate a broad pool of seeds and keep the top few
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous)
    seed_evals = [(s, _objective(s)) for s in seeds]
    seed_evals.sort(key=lambda x: x[1])
    top_seeds = [s for s, _ in seed_evals[:6]]

    if seed_evals[0][1] < best_val:
        best_val = seed_evals[0][1]
        best_seq = seed_evals[0][0].copy()
        print(f"[seed] new best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam optimisation on the smooth‑max surrogate for each elite seed
    #    (parallelising over the seeds is cheap – we run them sequentially)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, 1):
        if time.time() > deadline - 8.0:
            break
        print(f"[Adam] start seed #{idx}")
        seq, val = _adam_smoothmax(seed_vec, n, deadline,
                                   max_iters=800000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[overall Adam] best={val:.6f}")

    # ------------------------------------------------------------------
    # 3) Final simulated‑annealing refinement (uses any remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(best_seq, n, deadline,
                                    max_iters=None)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[overall Anneal] best={val:.6f}")

    # ------------------------------------------------------------------
    # 4) Fallback – should never be needed, but we keep it for safety
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    # store for possible subsequent runs
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
