"""
Optimiser for the “step‑function” challenge.

Main idea
---------
The objective is  

    2·n·max(conv(a,a)) / (sum(a)²)

with a non‑negative vector a of length n (the sum is normalised to 1).

If the support S = {i : a_i>0} satisfies that each *ordered* sum
i+j (i,j∈S) occurs at most t times, then

    max(conv(a,a)) ≤ t / |S|²

and therefore  

    objective ≤ 2·n·t / |S|² .

For t=4 the theory predicts that a set of size
≈√(8·n) ≃ 180 (for n=4000) yields an objective around 1.02.
The previous solver stopped the construction at 150 elements,
which kept the bound above the target 1.503.

This version pushes the low‑collision set construction much harder:
* it allows the full theoretical size (no early stop at 150);
* it performs thousands of greedy attempts within the time budget;
* the best set found is used with a uniform distribution (already
  satisfies the bound);
* a short mirror‑descent (exponentiated‑gradient) refinement is then
  applied – it can only improve the objective;
* if time permits a tiny simulated‑annealing fine‑tune is run.

The whole routine respects a 1000‑second wall‑clock budget and returns
the best sequence encountered as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution and objective (identical to the original helper)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple probability vectors (used only for fall‑backs)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform probability vector of length n."""
    return np.ones(n, dtype=np.float64) / n


# ----------------------------------------------------------------------
# Low‑collision (B₂[2]) set construction
# ----------------------------------------------------------------------


def _low_collision_set_extended(
    n: int,
    target_m: int,
    max_ordered: int,
    rng: np.random.Generator,
    deadline: float,
    max_attempts: int = 5000,
) -> np.ndarray | None:
    """
    Greedy construction of a set S⊂{0,…,n‑1} such that every ordered sum
    i+j (i,j∈S) occurs at most ``max_ordered`` times.
    The routine performs many independent random attempts until the
    time budget expires.  The best mask (largest |S|) is returned as an
    index array.
    """
    best_mask = None
    best_size = 0
    attempts = 0

    # upper bound on feasible size (theoretical limit)
    theoretical_limit = int(math.sqrt(max_ordered * (2 * n - 1)))

    while time.time() < deadline - 0.2 and attempts < max_attempts:
        attempts += 1
        mask = np.zeros(n, dtype=bool)
        sum_counts = np.zeros(2 * n - 1, dtype=np.uint8)

        order = rng.permutation(n)
        for cand in order:
            # self‑pair test
            if sum_counts[cand * 2] + 1 > max_ordered:
                continue

            # check sums with already selected indices
            existing = np.nonzero(mask)[0]
            if existing.size:
                sums = cand + existing
                # each new unordered pair introduces two ordered pairs
                if np.any(sum_counts[sums] + 2 > max_ordered):
                    continue

            # accept candidate
            mask[cand] = True
            sum_counts[cand * 2] += 1
            if existing.size:
                sum_counts[sums] += 2

        cur_size = mask.sum()
        if cur_size > best_size:
            best_size = cur_size
            best_mask = mask.copy()
            print(f"[LC‑Build] attempt {attempts:4d} size {cur_size}")

            # stop early if we already hit the (almost) theoretical limit
            if best_size >= theoretical_limit:
                break

        # Stop if we have already reached the user‑specified target
        if best_size >= target_m:
            # keep trying a little longer – maybe we can beat the target
            # but we do not abort immediately to give the algorithm more
            # chances to improve further.
            pass

    if best_mask is None or best_size == 0:
        return None
    return np.where(best_mask)[0]


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact non‑smooth objective
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) directly on the true
    objective.  The method keeps the best vector seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices where the max is (almost) attained
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient: average over all active constraints
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient update
        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # keep numerics stable
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Sub‑Grad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very light simulated‑annealing (optional, runs only if time left)
# ----------------------------------------------------------------------


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """Tiny Metropolis‑Hastings walk – only a few thousand moves."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.05)
    decay = 0.999

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a tiny amount of mass from i to j
        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

    return best, best_val

        print(f"[Info] low‑collision set size = {lc_idx.size}")
        seq = np.zeros(n, dtype=np.float64)
        seq[lc_idx] = 1.0 / lc_idx.size
        seq /= seq.sum()                # safety normalisation

    best_seq = seq.copy()
    best_val = _objective(best_seq)
    print(f"[Init] objective ≈ {best_val:.6f}")

    # --------------------------------------------------------------
    # Optional short mirror‑descent refinement (very cheap)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        # cost of one sub‑gradient iteration is ≈0.001 s
        max_iter = int(min(250000, remaining / 0.0015))
        if max_iter > 0:
            seq_ref, val_ref = _subgradient_refine(best_seq, n, deadline, max_iters=max_iter)
            if val_ref < best_val:
                best_seq, best_val = seq_ref, val_ref
                print(f"[Refine] sub‑gradient improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Tiny simulated‑annealing fine‑tune (uses any remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        seq_ann, val_ann = _anneal_optimize(best_seq, n, deadline, max_iters=None)
        if val_ann < best_val:
            best_seq, best_val = seq_ann, val_ann
            print(f"[Fine‑Tune] annealing improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and return
    # --------------------------------------------------------------
    final_seq = np.maximum(best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
    else:
        final_seq = final_seq / total

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(final_seq)})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
