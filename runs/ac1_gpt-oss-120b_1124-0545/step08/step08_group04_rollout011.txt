def _subgradient_multi(a, tol_rel=1e-8):
    conv = _conv_fft(a, a)
    max_val = conv.max()
    # mask indices where conv >= max_val - rel_tol*max_val - abs_tol?

    w = mask.astype(np.float64)
    conv_w = _conv_fft(w, a[::-1])
    g = 2.0 * conv_w[len(a)-1 : len(a)-1 + len(a)]

import time
import numpy as np
from math import pi, sqrt, exp

# Helper functions

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT. Returned length = len(x)+len(y)-1."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Objective 2·n·max(conv(seq,seq)) for a sequence normalised to sum=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic seed families (each returns a normalised numpy array)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total

def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()

def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()

def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * pi * i / (n - 1))
         + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()

def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total

def _gaussian_sigma15(n: int) -> np.ndarray:
    return _gaussian(n, sigma_factor=0.15)

def _gaussian_sigma20(n: int) -> np.ndarray:
    return _gaussian(n, sigma_factor=0.20)

def _gaussian_sigma25(n: int) -> np.ndarray:
    return _gaussian(n, sigma_factor=0.25)

def _exponential(n: int, decay: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-i / (decay * n))
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total

def _exponential_decay02(n: int) -> np.ndarray:
    return _exponential(n, decay=0.2)

def _exponential_decay03(n: int) -> np.ndarray:
    return _exponential(n, decay=0.3)

def _power_law(n: int, exponent: float = 0.5) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-exponent)
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total

def _power_law_exp05(n: int) -> np.ndarray:
    return _power_law(n, exponent=0.5)

def _power_law_exp10(n: int) -> np.ndarray:
    return _power_law(n, exponent=1.0)

def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.minimum(i + 1, n - i)
    return w / w.sum()

# Random Dirichlet sampling

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s

# Smoothed Adam optimiser (log‑its parametrisation)

def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 2.0,
    beta_end: float = 200.0,
    max_iter: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)) using Adam.
    The smooth max is realised by a log‑sum‑exp with an annealed temperature.
    The vector is parametrised by logits (soft‑max) guaranteeing non‑negativity
    and unit sum.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.02

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        a = np.exp(logits)
        a_sum = a.sum()
        a = a / a_sum

        conv = _conv_fft(a, a)

        # anneal beta linearly
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.5))

        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()

        # gradient w.r.t. a
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} beta={beta:.1f}")
            if best_val < 1.45:   # aggressive early exit, unlikely to happen
                break

    # final projection to simplex
    best_seq = best_seq / best_seq.sum()
    return best_seq, best_val

# Multi‑peak sub‑gradient mirror descent

def _refine_multi(start_seq: np.ndarray, deadline: float, lr0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using a sub‑gradient that aggregates
    *all* currently active convolution entries (those within a tiny
    tolerance of the maximum). This often yields a lower maximum
    autocorrelation than using a single active index.
    """
    a = start_seq.copy()
    n = len(a)
    eps = 1e-30
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = np.max(conv)

        # tolerance – include indices that are virtually maximal
        tol = max(1e-12, max_val * 1e-9)
        mask = conv >= max_val - tol
        w = mask.astype(np.float64)

        # gradient contribution from all active indices
        conv_w = _conv_fft(w, a[::-1])
        g = 2.0 * conv_w[n - 1:n - 1 + n]

        lr = lr0 / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise before exponentiation
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()

import time

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (fast for moderate vectors).
    Returned length = len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to evaluate_sequence for a sequence already normalised to
    sum = 1:
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    """Raised‑cosine window (Hann)."""
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()

    i = np.arange(n, dtype=np.float64)

    w = (a0
         - a1 * np.cos(2 * pi * i / (n - 1))
         + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()

    """Discrete Gaussian centred in the middle, σ = sigma_factor·n."""
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """
    Draw a Dirichlet(alpha,…,alpha) vector.
    For alpha=1 we use exponentials; otherwise Gamma.
    """

        x = rng.exponential(1.0, size=n)
    else:

    return x / s

def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 2.0,
    beta_end: float = 200.0,
    max_iter: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)) using Adam.
    The smooth max is realised by a log‑sum‑exp with a temperature β
    that is slowly increased.  The vector is parametrised by logits
    (soft‑max) guaranteeing non‑negativity and unit sum.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.02

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        a = np.exp(logits)
        a_sum = a.sum()
        a = a / a_sum

        conv = _conv_fft(a, a)

        # linearly anneal the temperature
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.5))

        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()

        # gradient w.r.t. a (smooth max)
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if (it & 1023) == 0:   # periodic exact evaluation

            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} beta={beta:.1f}")

    best_seq = best_seq / best_seq.sum()
    return best_seq, best_val

def _refine_multi(start_seq: np.ndarray, deadline: float, lr0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using a sub‑gradient that aggregates
    *all* currently active convolution entries (those within a tiny
    tolerance of the maximum).  This often yields a lower max
    autocorrelation than using a single active index.
    """

    n = len(a)
    eps = 1e-30

    best_seq = a.copy()
    best_val = _objective(a)
