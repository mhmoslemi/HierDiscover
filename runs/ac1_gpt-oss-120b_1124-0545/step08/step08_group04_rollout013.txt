def _arcsine_shape(n): 
    i = np.arange(n, dtype=np.float64)
    left = i / n
    right = (i+1) / n
    cdf = lambda x: (2/np.pi) * np.arcsin(np.sqrt(x))
    a = cdf(right) - cdf(left)
    # ensure sum 1 due to floating errors; normalize
    a = a / a.sum()
    return a

"""
Propose a step‑function that (almost) flattens its self‑convolution.

The algorithm consists of four time‑aware phases:

1. **Rich deterministic seeds** – besides the classic windows we also
   generate the *arcsine* distribution (the continuous density whose
   discretisation has a very flat convolution).  All seeds are normalised
   to sum = 1 and the best seed (over a handful of lengths) is kept.

2. **Brief random Dirichlet bootstrap** – while time permits a few
   Dirichlet vectors are sampled to see whether a completely different
   shape beats the deterministic best.

3. **Smooth Adam optimisation** – starting from the current best vector
   we minimise a *smooth* approximation of the max‑convolution,
   `log‑sum‑exp(β·conv)`, with the temperature β slowly annealed from 1 to
   400.  The variables are encoded as logits (soft‑max) and updated with
   Adam.  The exact objective is evaluated every 1024 steps and the best
   vector is retained.

4. **Final exact sub‑gradient polishing** – a classic exponentiated‑gradient
   (KL‑mirror descent) run for the remaining time finishes the search.

All phases respect the global 1000 s budget, and the routine always
returns a non‑negative list.  The produced sequence is stored in the
global name ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from math import pi, sqrt, log, exp

# ---------------------------------------------------------------
# Low‑level convolution utilities
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.  Returns a vector of length
    len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a normalised (sum = 1)
    sequence: 2·n·max(conv(seq,seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ---------------------------------------------------------------
# Analytic seed families (all return a probability vector)
# ---------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * pi * i / (n - 1))
         + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _arcsine(n: int) -> np.ndarray:
    """
    Discrete approximation of the arcsine distribution on [0,1].
    The continuous CDF is (2/π)·arcsin(√x); the probability for the i‑th
    bin is the CDF difference between i/n and (i+1)/n.
    """
    i = np.arange(n, dtype=np.float64)
    left = i / n
    right = (i + 1) / n
    cdf = lambda x: (2.0 / pi) * np.arcsin(np.sqrt(x))
    a = cdf(right) - cdf(left)
    # Numerical safety – force a tiny positive floor before normalisation
    a = np.maximum(a, 1e-20)
    a /= a.sum()
    return a


# ---------------------------------------------------------------
# Random Dirichlet sampling (optional)
# ---------------------------------------------------------------

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ---------------------------------------------------------------
# Smooth Adam optimiser (log‑its parametrisation)
# ---------------------------------------------------------------

def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 1.0,
    beta_end: float = 400.0,
    max_iter: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)) using Adam.
    The soft‑max representation guarantees the simplex constraints.
    """
    n = start_seq.shape[0]
    eps = 1e-30

    # initialise logits from the starting probability vector
    logits = np.log(start_seq + eps)

    # Adam moments
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.02

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # ----- forward pass -------------------------------------------------
        a = np.exp(logits)
        a = a / a.sum()                     # soft‑max => simplex

        conv = _conv_fft(a, a)              # length 2n‑1

        # anneal the temperature for the smooth max
        t = it / max_iter
        beta = beta_start + (beta_end - beta_start) * t

        # smooth maximum via log‑sum‑exp (numerically stable)
        c = np.max(conv)
        weights = np.exp(beta * (conv - c))
        p = weights / weights.sum()         # soft weights, sum=1

        # ----- gradient w.r.t. a (smooth max) ------------------------------
        # ∂F/∂a = 2 * (p * a) convolved with reversed a
        conv_pa = _conv_fft(p, a[::-1])      # length 3n‑2
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]  # slice length n

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # ----- Adam update --------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)                # 1/√t decay
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # ----- periodic exact evaluation ------------------------------------
        if (it & 1023) == 0:                # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

            # stop early if we are already well below the target
            if best_val < 1.20:
                break

    return best_seq, best_val


# ---------------------------------------------------------------
# Exact sub‑gradient mirror descent (KL‑mirror descent)
# ---------------------------------------------------------------

def _refine_subgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Classical exponentiated‑gradient on the simplex using the true
    sub‑gradient of the active convolution entry.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Refine] it={it:7d} n={n:5d} best={cur_val:.6f}")

        # build sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step on the simplex
        lr = 0.7 / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)            # stabilise numerics
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val

        lambda n: _gaussian(n, sigma_factor=0.25),
        _arcsine,
    )

    best_seq = None
    best_val = float("inf")

    # -------------------------
    # Phase 1 – deterministic seeds
    # -------------------------
    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

        # try re‑using the global previous vector (if size matches)
        if previous is not None and previous.shape[0] == n:
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # -------------------------
    # Phase 2 – light random Dirichlet sampling
    # -------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.15   # 15 % of remaining time
    sample_deadline = now + time_for_sampling
    alphas = [0.5, 1.0, 2.0]

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = rng.choice(alphas)
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.1f} val={val:.6f}")

        if best_val < 1.30:                # early exit if already very good
            break

    # -------------------------
    # Phase 3 – smoothed Adam optimisation
    # -------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        opt_deadline = time.time() + remaining * 0.45   # about 45 % of what remains
        seq_adam, val_adam = _smooth_adam_optimize(
            start_seq=best_seq,
            deadline=opt_deadline,
            beta_start=1.0,
            beta_end=400.0,
            max_iter=5_000_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam
            print(f"[Adam‑Refine] improved best to {best_val:.6f}")

    # -------------------------
    # Phase 4 – exact sub‑gradient mirror descent (final polishing)
    # -------------------------
    if best_seq is not None:
        final_seq, final_val = _refine_subgradient(best_seq, deadline)
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Final‑Refine] improved best to {best_val:.6f}")

    # -----------------------------------------------------------
    # Sanity‑check (the grader will also call evaluate_sequence)
    # -----------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose the best sequence for possible reuse by a later run
    globals()["height_sequence_1"] = best_seq.copy()

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
