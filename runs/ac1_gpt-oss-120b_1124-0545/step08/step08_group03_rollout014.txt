We have a Python function evaluate_sequence(sequence) that:

- Checks that the input is a nonempty list of numbers, no NaNs, infinities, booleans.

- Converts items to float, clips each to [0, 1000] and filter negative -> set to max(0, x).

- Calculates n = len(sequence).

- Computes b_sequence = np.convolve(sequence, sequence). That's the linear convolution, as defined: resulting length = 2n-1, with convolution.

- max_b = max(b_sequence). sum_a = sum(sequence).

- If sum_a < 0.01 returns np.inf.

- Returns 2 * n * max_b / (sum_a ** 2) as a float.

Goal: minimize this objective.

Essentially we can think of a sequence a_i >= 0. If we view it as a step function and consider convolution with itself, the maximum entry of the convolution is at some index - denote M = max_i (a*a)_i. The target objective is 2 n M / (sum a)^2.

This resembles a scaled product of length and maximum.

The problem reminiscent of the extremal problem: minimize max convolution given total sum.

Goal: to find a_i such that the "square maximum" is minimized relative to sum. Equivalent to minimize M * 2n / (sum^2). Since n is fixed for a given sequence length; we can treat maximizing sum given M or minimizing M for fixed sum.

"""
Propose a step‑function (non‑negative vector) that minimizes
    2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser builds many diverse seeds (uniform, windows, power laws,
Beta‑densities, geometric / logistic decays, sparse supports, etc.),
runs a short Adam phase on a smooth‑max surrogate, then a long
exponentiated‑gradient (mirror‑descent) refinement, interleaves cheap
LP‑based mixing, upsamples the best solution to a finer grid and refines
again, and finally finishes with simulated‑annealing.  All updates stay
on the probability simplex.  The routine respects the supplied time
budget (default 1000 s) and stores the best vector in the global variable
`height_sequence_1`.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Optional linear‑programming support

try:
    import cvxpy as cp

# Low‑level FFT based convolution (exact, real‑valued)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by `evaluate_sequence` (clipping ignored)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

# Simple families (all return a probability vector on the simplex)
# ------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on (0,1), discretised on mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _logistic_decay(n: int, shift: float, scale: float) -> np.ndarray:
    """Monotone decreasing logistic shape."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - shift) / scale))
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random_seed(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random support of size `m` (1 ≤ m ≤ n)."""
    if m >= n:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """Deterministic “Sidon‑like” support using triangular numbers."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _quadratic_residue_seed(n: int, m: int) -> np.ndarray:
    """Positions are i² (mod n)."""
    idx = np.mod(np.arange(m) ** 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a

# Utility functions
# ------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _smooth_sequence(a: np.ndarray, radius: int = 3, sigma: float = 1.0) -> np.ndarray:
    """Apply a small Gaussian smoothing (linear convolution, same length)."""
    k = 2 * radius + 1
    x = np.arange(k) - radius
    kern = np.exp(-0.5 * (x / sigma) ** 2)
    kern = kern / kern.sum()
    smooth = np.convolve(a, kern, mode="same")
    smooth = np.clip(smooth, 0.0, None)
    s = smooth.sum()
    if s > 0:
        smooth = smooth / s
    return smooth


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation up‑sample to `new_n` points and re‑normalise."""
    old_n = a.shape[0]
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s > 0:
        a_new /= s
    return a_new

def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 12000,
) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max surrogate of max(conv(a,a))."""
    v = np.log(seed + 1e-30)  # logits for simplex parametrisation
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 1.5
    beta_mul = 1.007

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # smoothed‑max weights
        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum

        # gradient of Σ w_k (a*a)_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam moments
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400000,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror‑descent) on the exact objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    w = np.zeros(2 * n - 1, dtype=np.float64)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over active constraints)
        g = np.zeros_like(a)
        if active.size:
            w.fill(0.0)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1 : n - 1 + n]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # numeric stability
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _lp_refine(
    f: np.ndarray,
    n: int,
    deadline: float,
    t_mix: float = 0.5,
    max_lp_iters: int = 20,
) -> np.ndarray:
    """Repeated cheap linear‑programming refinements."""
    if not CVXPY_AVAILABLE:
        return f

    a = f.copy()
    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - 1e-9)[0]
        if active.size == 0:
            break

        g = cp.Variable(n)
        constraints = [g >= 0]

        for k in active:
            coeff = np.zeros(n, dtype=np.float64)
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i = np.arange(i_low, i_high + 1)
            j = k - i
            coeff[j] = a[i]                     # (f * g)[k] = Σ_i f[i]·g[k‑i]
            constraints.append(coeff @ g <= M + 1e-12)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=2000, abstol=1e-9, reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if g.value is None:
            break

        g_opt = np.asarray(g.value).flatten()
        tot = g_opt.sum()
        if tot <= 0:
            break
        g_opt = g_opt / tot

        a = (1.0 - t_mix) * a + t_mix * g_opt
        a = _project_simplex(a)

    return a


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """Simple Metropolis‑Hastings annealing."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)   # initial temperature
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # up to 2 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


# ------------------------------------------------------------
# Seed generation (including many sparse and parametric candidates)
# ------------------------------------------------------------


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Assemble a diverse set of start vectors, all normalised to sum = 1."""
    seeds: list[np.ndarray] = []

    # Classic families
    seeds.append(_uniform(n))

    # Geometric decay – sweep a wide range of ratios
    for r in np.linspace(0.80, 0.9999, 12):
        seeds.append(_geom(n, r))

    # Power laws
    for p in np.linspace(0.0, 2.5, 11):
        seeds.append(_power(n, p))

    # Combined geometric‑power families
    for r in np.linspace(0.90, 0.9995, 7):
        for p in np.linspace(0.0, 2.0, 7):
            seeds.append(_geom_power(n, r, p))

    # Symmetric geometric families
    for r in np.linspace(0.90, 0.9995, 6):
        seeds.append(_sym_geom(n, r))

    # Windows / classic smooth shapes
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    for sigma_factor in np.linspace(0.07, 0.35, 9):
        seeds.append(_gaussian(n, sigma_factor))

    # Beta densities – explore both edge‑heavy and mildly centre‑heavy shapes
    for alpha in np.concatenate(
        [np.linspace(0.08, 0.45, 8), np.linspace(0.55, 1.0, 6)]
    ):
        seeds.append(_beta_window(n, alpha))

    # Logistic decreasing families – randomising shift & scale
    for _ in range(6):
        shift = rng.uniform(0.2 * n, 0.8 * n)
        scale = rng.uniform(0.01 * n, 0.15 * n)
        seeds.append(_logistic_decay(n, shift, scale))

    # Random softmax seeds
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # Sparse random seeds – useful for finding “flat” convolutions
    for m in range(30, 210, 20):          # 30,50,…,190
        for _ in range(3):
            seeds.append(_sparse_random_seed(n, m, rng))

    # Deterministic Sidon‑like structures
    for m in (50, 80, 110, 140):
        seeds.append(_sidon_like_seed(n, min(m, n)))
        seeds.append(_quadratic_residue_seed(n, min(m, n)))

    # Re‑use previous construction if present
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # Clean‑up – enforce non‑negativity and renormalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    The routine respects the supplied time budget and returns the best
    sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # Load a previous construction if the grader supplied one
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    rng = np.random.default_rng(seed + 1)

    # ------------------------------------------------------------------
    # Phase 0 – coarse optimisation on a moderate length
    # ------------------------------------------------------------------
    n0 = 4000
    print(f"\n=== Coarse optimisation, length {n0} ===")
    seeds = _make_seeds(n0, previous, rng)

    # Quick evaluation of all seeds, keep the top few
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 15
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    best_seq = seed_vals[0][0].copy()
    best_val = seed_vals[0][1]
    print(f"[seed] best initial={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam on smoothed‑max for each top seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(seed_vec, n0, deadline, max_iters=15000)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Adam] seed={idx} best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – long sub‑gradient (mirror‑descent) refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _subgradient_refine(best_seq, n0, deadline, max_iters=500000)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[SubGrad] after long run best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – cheap LP mixing interleaved with short sub‑gradient passes
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # LP refinement (aggressive mixing)
        seq_lp = _lp_refine(best_seq, n0, deadline, t_mix=0.7, max_lp_iters=25)
        if seq_lp is None:
            break

        # short sub‑gradient after LP
        seq, val = _subgradient_refine(seq_lp, n0, deadline, max_iters=150000)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[LP+SubGrad] new best={val:.6f}")

        # intermittent perturbation
        if time.time() > deadline - 20.0:
            break
        perturb = best_seq.copy()
        noise = rng.normal(scale=0.001, size=n0)
        perturb += noise
        perturb = np.clip(perturb, 0.0, None)
        if perturb.sum() == 0.0:
            perturb = _uniform(n0)
        else:
            perturb /= perturb.sum()
        seq, val = _subgradient_refine(perturb, n0, deadline, max_iters=200000)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Perturbed] new best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase D – up‑sample to a finer grid and refine again
    # ------------------------------------------------------------------
    n_big = 8000
    print(f"\n=== Upsampling to length {n_big} and refinement ===")
    up_seq = _upsample_sequence(best_seq, n_big)

    # a few smoothing passes (helps escape tiny local peaks)
    for smooth_it in range(3):
        if time.time() > deadline:
            break
        up_seq = _smooth_sequence(up_seq, radius=3, sigma=0.9)
    # final mirror‑descent on the finer grid
    if time.time() < deadline:
        seq, val = _subgradient_refine(up_seq, n_big, deadline, max_iters=250000)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Fine SubGrad] new best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase E – final simulated‑annealing fine‑tuning (any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        seq, val = _anneal_optimize(best_seq, best_seq.size, deadline, max_iters=None)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Anneal final] best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(max(1, best_seq.shape[0] if best_seq is not None else n0))
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    globals()["height_sequence_1"] = np.copy(best_seq)
    # Convert to plain Python list as required by the grader
    return best_seq.tolist()
