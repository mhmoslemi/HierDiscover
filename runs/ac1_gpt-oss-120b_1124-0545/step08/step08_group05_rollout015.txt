def _adam_optimize(start_seq, n, deadline, max_iters):
    a = start_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        # compute gradient
        conv = _conv_fft(a, a)
        m_val = conv.max()
        # smoothing parameter schedule
        if it < 50000:
            alpha = 30.0
        elif it < 200000:
            alpha = 100.0
        else:
            alpha = 300.0
        exps = np.exp(alpha * (conv - m_val))
        w = exps / exps.sum()
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 4.0 * n * full[n-1:n-1+n]  # gradient wrt a

        # convert to gradient wrt log domain
        grad_log = a * (grad_a - np.dot(grad_a, a))  # a * (g - <g,a>)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = 0.5 / (np.sqrt(v_hat) + 1e-8)  # learning rate
        # update log_a: track log_a = log a
        # we can maintain log_a directly:
        if it == 1:
            log_a = np.log(a + 1e-30)
        log_a -= step * m_hat
        # recenter
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        # evaluate occasional ...
        if it % 5000 == 0:
            val = _objective(a)
            if val < best_val: update.

"""
Hybrid optimiser for the step‑function problem.

Strategy
--------
1. Build a diverse pool of seed windows (uniform, geometric families,
   power families, classic signal windows, random soft‑max vectors and a few
   fine‑grained geometric families close to the uniform distribution).
2. Keep the best few seeds (by the exact objective) and run a short
   Adam‑style exponentiated‑gradient optimisation on a smooth
   log‑sum‑exp surrogate of the max‑convolution.
   The sequence is kept on the probability simplex via the softmax
   parametrisation x → softmax(x).
3. Run a longer Adam optimisation on the best sequence found so far,
   increasing the smoothness (α) during the run.
4. Finish with a lightweight simulated‑annealing walk to escape possible
   shallow local minima.
5. Return the best sequence discovered within the time budget.

The optimiser respects the 1000 s budget, prints progress and stores the
result in the global variable ``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (real input, full mode)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple families (all return a probability vector on the simplex)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse set of start vectors, all normalised to sum = 1."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic families
    seeds.append(_uniform(n))

    # geometric – coarse sweep
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # geometric – fine sweep near 1 (the region that gave the best bounds)
    for c in np.linspace(0.05, 1.0, 10):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power families
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    # combined geometric–power families
    for c in np.linspace(0.5, 3.5, 7):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 7):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 7):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classic windows
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 7):
        seeds.append(_tukey_window(n, alpha))

    # random soft‑max vectors
    for _ in range(12):
        seeds.append(_random_softmax(n, rng))

    # mixture of geometric and uniform (helps to explore the region close to uniform)
    for c in np.linspace(0.05, 1.0, 5):
        r = 1.0 - c / n
        geom = _geom(n, r)
        for lam in (0.3, 0.5, 0.7):
            mix = lam * geom + (1 - lam) * _uniform(n)
            seeds.append(mix)

    # reuse previous construction if it has the right length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final clean‑up – enforce non‑negativity and normalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned


# ----------------------------------------------------------------------
# Adam optimiser on a smooth log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------


def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (Adam) optimiser on the smooth surrogate:
        obj_smooth(a) = 2·n·(1/α)·log∑_k exp(α·(a*a)_k)

    The variables are kept on the probability simplex via a soft‑max
    parametrisation: a = softmax(x).  The method uses Adam‑style
    moment estimates for the gradient in the log‑domain.
    """
    a = np.clip(start_seq, 1e-30, None)
    a /= a.sum()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and smooth weights
        conv = _conv_fft(a, a)
        max_c = conv.max()

        # smoothness schedule (grows with the iteration count)
        if it < 100000:
            alpha = 30.0
        elif it < 300000:
            alpha = 120.0
        else:
            alpha = 300.0

        exps = np.exp(alpha * (conv - max_c))   # stable because <= 0
        w = exps / exps.sum()                  # soft‑max weights

        # gradient of ∑_k w_k·conv_k w.r.t. a (exact, see derivation)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 4.0 * n * full[n - 1:n - 1 + n]   # ∂obj/∂a

        # chain rule to obtain gradient w.r.t. log‑variables (soft‑max)
        avg = np.dot(grad_a, a)
        grad_log = a * (grad_a - avg)

        # Adam update on the log‑variables
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 / (np.sqrt(v_hat) + eps)   # learning‑rate

        log_a -= step * m_hat

        # re‑centre for numerical stability, then map back to simplex
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum):
            a = _uniform(n)
            log_a = np.log(a + 1e-30)
        else:
            a /= a_sum

        # occasional exact evaluation
        if it % 2000 == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} best={val:.6f}")

    return best_seq, best_val

def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Metropolis–Hastings walk with larger mass‑transfer moves."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.06)
    decay = 0.9992

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.05 * cur[i]
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        if it % 800 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val
