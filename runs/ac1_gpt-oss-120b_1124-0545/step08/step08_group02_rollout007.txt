"""
Propose a step‑function that attains an objective below the target 1.503.
The algorithm focuses on constructing a large *low‑collision* support
(set S ⊂ {0,…,n‑1}) such that every ordered pair‑sum i+j (i,j∈S) occurs at
most 4 times (a B₂[2] set).  For a uniform distribution on such a set,
max⁡(a∗a) ≤ 4/|S|² and the objective equals 2 n max⁡(a∗a) ≤ 8 n/|S|².
For n=4000 a support of size |S|≥146 guarantees an objective ≤1.501,
which beats the target.  The code repeatedly builds random greedy
B₂[2] sets until a sufficiently large one is found, then returns the
uniform vector on that support.  If the time budget is still available
a short Adam + sub‑gradient refinement is performed (optional, never
required for the target).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT based convolution (exact, real‑valued)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The input is assumed to be a non‑negative vector.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple optimisation helpers (Adam & mirror‑descent sub‑gradient)
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 25000) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate of max(conv(a,a)).
    Returns the best vector encountered and its true objective.
    """
    v = np.log(seed + 1e-30)                     # logits for simplex parametrisation
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 1.5
    beta_mul = 1.007

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # ----- smooth max weights -----
        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·(a*a)_k -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- chain rule through soft‑max -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam moment update -----
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 500000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact non‑smooth objective.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over all active constraints)
        g = np.zeros_like(a)
        if len(active) > 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / len(active)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Low‑collision B₂[2] set generation
# ----------------------------------------------------------------------


def _low_collision_set(n: int,
                       max_ordered: int,
                       rng: np.random.Generator,
                       deadline: float,
                       needed_size: int) -> np.ndarray | None:
    """
    Greedy construction of a set S ⊂ {0,…,n‑1} such that each ordered
    pair‑sum i+j occurs at most ``max_ordered`` times.
    Returns the index array of the largest set found (or ``None``).
    The search stops as soon as a set of size ``needed_size`` is found
    (or when the time budget expires).
    """
    best_mask = None
    best_size = 0
    attempt = 0

    # rough absolute upper bound (theoretical B₂[2] limit)
    theoretical_max = int(np.sqrt(2 * n * max_ordered))

    while time.time() < deadline - 0.5:
        attempt += 1
        mask = np.zeros(n, dtype=bool)
        sum_counts = np.zeros(2 * n - 1, dtype=np.uint8)
        existing: list[int] = []

        order = rng.permutation(n)
        for cand in order:
            # self‑pair (i,i) contributes one ordered pair to sum 2*cand
            if sum_counts[2 * cand] + 1 > max_ordered:
                continue

            if existing:
                arr = np.array(existing, dtype=np.int32)
                sums = cand + arr                      # vectorised
                if np.any(sum_counts[sums] + 2 > max_ordered):
                    continue

            # accept the candidate
            mask[cand] = True
            sum_counts[2 * cand] += 1
            if existing:
                sum_counts[sums] += 2
            existing.append(int(cand))

            # early stop if we already reached the theoretical ceiling
            if len(existing) >= theoretical_max:
                break

        cur_size = len(existing)
        if cur_size > best_size:
            best_size = cur_size
            best_mask = mask.copy()
            print(f"[LowColl] attempt {attempt:3d} size {cur_size}")

        # stop once we have a set large enough to beat the target
        if best_size >= needed_size:
            break

    if best_mask is None:
        return None
    return np.where(best_mask)[0]


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and returns the best sequence discovered as a plain list of floats.
    """
    # --------------------------------------------------------------
    # Basic setup
    # --------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin
    rng = np.random.default_rng(seed + 1)

    # --------------------------------------------------------------
    # Problem size – we keep the original 4000 (good trade‑off)
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # --------------------------------------------------------------
    # Target support size that guarantees the desired bound
    #   objective ≤ 8·n / m²  ⇒  m ≥ sqrt(8·n / 1.503) ≈ 146
    # --------------------------------------------------------------
    needed_m = 146

    # --------------------------------------------------------------
    # Build a large low‑collision set (ordered sum count ≤ 4)
    # --------------------------------------------------------------
    low_coll_set = _low_collision_set(
        n=n,
        max_ordered=4,
        rng=rng,
        deadline=deadline,
        needed_size=needed_m,
    )

    if low_coll_set is not None and low_coll_set.size >= needed_m:
        # Uniform distribution on the support
        a = np.zeros(n, dtype=np.float64)
        a[low_coll_set] = 1.0 / low_coll_set.size
        a /= a.sum()
        init_obj = _objective(a)
        print(f"[Init] low‑collision set size {low_coll_set.size}, "
              f"objective {init_obj:.6f}")
        best_seq = a
        best_val = init_obj
    else:
        # Fallback to a simple seed (uniform) – unlikely to be needed
        print("[Fallback] could not find large enough low‑collision set")
        best_seq = np.ones(n, dtype=np.float64) / n
        best_val = _objective(best_seq)

    # --------------------------------------------------------------
    # If we already beat the target, we are done
    # --------------------------------------------------------------
    if best_val <= 1.503:
        print(f"Target achieved without extra optimisation: {best_val:.6f}")
        final_seq = best_seq.copy()
        globals()["height_sequence_1"] = final_seq.copy()
        return final_seq.tolist()

    # --------------------------------------------------------------
    # Optional: a short Adam phase (may shave a few 1e‑3)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        seq_adam, val_adam = _adam_optimize(best_seq, n, deadline, max_iters=20000)
        if seq_adam is not None and val_adam < best_val:
            best_seq = seq_adam
            best_val = val_adam
            print(f"[Overall Adam] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # Optional: exact sub‑gradient refinement (still cheap)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        # Rough cost of one sub‑gradient iteration ≈ 0.001 s
        max_sub_iters = int(min(200000, remaining / 0.0015))
        seq_sub, val_sub = _subgradient_refine(best_seq, n, deadline,
                                               max_iters=max_sub_iters)
        if seq_sub is not None and val_sub < best_val:
            best_seq = seq_sub
            best_val = val_sub
            print(f"[Overall SubGrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and store
    # --------------------------------------------------------------
    final_seq = np.maximum(best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = np.ones(n, dtype=np.float64) / n
    else:
        final_seq /= total

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
          f"(length {len(final_seq)})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
