"""
Fast‐gradient optimizer for the step‑function lower bound.

The algorithm:
  1. Build many diverse seeds (uniform, windows, random Dirichlet,
     sparse supports, Sidon‑like sets …) and keep the few best.
  2. For each good seed run a few thousand Adam steps on a
     smooth‑max surrogate of max(conv(a,a)).  The gradient of the
     surrogate is computed by one forward FFT of `a` and one forward
     FFT of the soft‑max weights – the “reversed” FFT needed for the
     gradient is obtained analytically, cutting the cost in half.
  3. Starting from the best Adam result launch a long exponentiated‑
     gradient (mirror‑descent) run on the exact non‑smooth objective.
     The sub‑gradient of `max(conv)` is obtained from the active set
     (indices whose convolution value is within 1e‑12 of the maximum).
     Each iteration needs only two FFTs (forward of `a` and of the
     weight vector), so millions of steps are affordable.
  4. Occasionally perturb the current best (tiny Gaussian noise) and
     resume the sub‑gradient loop – this helps to escape shallow
     plateaus.
  5. The best sequence found before the time budget expires is stored
     in the global `height_sequence_1` variable and returned as a plain
     list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities – everything works on a fixed length `N` (next power of two)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolution(a: np.ndarray, N: int, L: int) -> np.ndarray:
    """
    Linear convolution a*a (full length L = 2·len(a)‑1) using a real FFT of length N.
    Returns the first L entries.
    """
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return conv


def _fft_rev_convolution(a_rev_fft: np.ndarray,
                         w: np.ndarray,
                         phi: np.ndarray,
                         N: int,
                         L: int) -> np.ndarray:
    """
    Compute convolution rev(a) * w (full length L) using the FFT of rev(a)
    that is obtained from the FFT of a via a simple phase factor `phi`.
    """
    fw = np.fft.rfft(w, n=N)
    # rev(a) FFT = conj(FFT(a)) * phi   (see derivation in the analysis)
    conv_rev = np.fft.irfft(a_rev_fft * fw, n=N)[:L]
    return conv_rev


# ----------------------------------------------------------------------
# Exact objective (same as `evaluate_sequence` for sums that are not clipped)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(seq,seq)) / (sum(seq)^2).
    `seq` may contain any non‑negative numbers; the function is safe for
    zero total mass (it returns +inf).
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    n = seq.size
    N = _next_pow2(2 * n - 1)
    L = 2 * n - 1
    conv = _fft_self_convolution(seq, N, L)
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Seed families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a_sum = a.sum()
    if a_sum == 0:
        return _uniform(n)
    return a / a_sum


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s == 0:
        return _uniform(n)
    return a / s


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse‑ish Dirichlet (α<1 → many near‑zero entries)."""
    x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0:
        return _uniform(n)
    return x / s


def _sparse_random_seed(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random support of size m (1 ≤ m ≤ n)."""
    if m >= n:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """Deterministic Sidon‑like support (triangular numbers)."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate (log‑sum‑exp) of max(conv).
    Returns the best vector seen and its true objective.
    """
    # log‑parameterisation (simplex via soft‑max)
    v = np.log(seed + 1e-30)

    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # smoothing parameter
    beta = 1.0
    beta_mul = 1.005

    N = _next_pow2(2 * n - 1)
    L = 2 * n - 1
    phi = np.exp(-2j * np.pi * np.arange(N // 2 + 1) * (n - 1) / N)

    best_seq = None
    best_val = np.inf
    t = 0

    while t < max_iters and time.time() < deadline:
        t += 1
        a = np.exp(v - np.max(v))  # stable soft‑max
        a /= a.sum()

        # forward FFT of a (used for both self‑conv and rev‑conv)
        fa = np.fft.rfft(a, n=N)

        # self‑convolution a*a
        conv = np.fft.irfft(fa * fa, n=N)[:L]

        shift = conv.max()
        # smoothed‑max weights
        beta = beta * beta_mul
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·(a*a)_k
        fw = np.fft.rfft(w, n=N)
        # rev(a) FFT via phase factor
        f_rev = np.conj(fa) * phi
        conv_rev = np.fft.irfft(f_rev * fw, n=N)[:L]
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]  # centre slice

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam moments
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:5d} best={cur_val:.6f}")

    return best_seq, best_val


def _subgradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 800000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient mirror‑descent on the exact (non‑smooth) objective.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    N = _next_pow2(2 * n - 1)
    L = 2 * n - 1
    phi = np.exp(-2j * np.pi * np.arange(N // 2 + 1) * (n - 1) / N)

    rel_tol = 1e-12
    it = 0
    no_improve = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        # forward FFT and self‑convolution
        fa = np.fft.rfft(a, n=N)
        conv = np.fft.irfft(fa * fa, n=N)[:L]
        max_val = conv.max()

        active = np.where(conv >= max_val - rel_tol)[0]
        if active.size == 0:
            # should never happen, but guard against it
            active = np.array([np.argmax(conv)])

        # uniform weight on the active set
        w = np.zeros(L, dtype=np.float64)
        w[active] = 1.0 / active.size

        # gradient of the max (average over active constraints)
        fw = np.fft.rfft(w, n=N)
        f_rev = np.conj(fa) * phi
        conv_rev = np.fft.irfft(f_rev * fw, n=N)[:L]
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # exponentiated‑gradient step on the simplex
        lr = 0.9 / np.sqrt(it + 1.0)
        log_a = np.log(a + 1e-30) - lr * grad_a
        log_a -= np.max(log_a)                # numerical stability
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")
        else:
            no_improve += 1

        # occasional random jitter to escape plateaus
        if no_improve >= 50000:
            rng = np.random.default_rng()
            noise = rng.normal(scale=0.001, size=n)
            a += noise
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = _uniform(n)
            else:
                a /= s
            no_improve = 0

    return best_seq, best_val


def _perturb_and_refine(seq: np.ndarray,
                        n: int,
                        deadline: float,
                        steps: int = 120000) -> tuple[np.ndarray, float]:
    """
    Small Gaussian perturbation followed by a short sub‑gradient run.
    """
    rng = np.random.default_rng()
    pert = seq + rng.normal(scale=0.001, size=n)
    pert = np.clip(pert, 0.0, None)
    s = pert.sum()
    if s == 0.0:
        pert = _uniform(n)
    else:
        pert /= s
    return _subgradient_optimize(pert, n, deadline, max_iters=steps)


# ----------------------------------------------------------------------
# Seed factory
# ----------------------------------------------------------------------


def _make_seeds(n: int,
                previous: np.ndarray | None,
                rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse pool of candidate vectors (all normalised to sum = 1)."""
    seeds = []

    # classic deterministic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.12))
    seeds.append(_gaussian(n, sigma_factor=0.20))
    seeds.append(_random_softmax(n, rng))
    seeds.append(_random_softmax(n, rng))
    seeds.append(_random_dirichlet(n, alpha=0.3, rng=rng))
    seeds.append(_random_dirichlet(n, alpha=0.5, rng=rng))
    seeds.append(_random_dirichlet(n, alpha=0.8, rng=rng))

    # geometric / power families
    for r in np.linspace(0.7, 0.99, 5):
        seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 5):
        seeds.append(_power(n, p))

    # sparse random supports
    for m in (40, 80, 120, 160, 200):
        for _ in range(2):
            seeds.append(_sparse_random_seed(n, m, rng))

    # Sidon‑like deterministic supports
    for m in (80, 120, 160, 200):
        seeds.append(_sidon_like_seed(n, min(m, n)))

    # reuse previous construction if available
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final sanity pass (clip & renorm)
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence discovered (as a plain list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # Retrieve possible previous construction
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    rng = np.random.default_rng(seed + 1)

    # ------------------------------------------------------------------
    # Problem size – 4000 has proven a sweet spot
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # 1) Build a rich seed pool
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous, rng)

    # ------------------------------------------------------------------
    # 2) Quick evaluation of seeds – keep the best few
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = seed_vals[0][0].copy()
    global_best_val = seed_vals[0][1]
    print(f"[seed] best initial={global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam on the smoothed‑max for each top seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(seed_vec, n, deadline,
                                  max_iters=8000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Adam] seed={idx} best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – long sub‑gradient refinement on the best seed found so far
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _subgradient_optimize(global_best_seq, n, deadline,
                                         max_iters=800000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[SubGrad] after long run best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – occasional perturbations + short refinements
    # ------------------------------------------------------------------
    while time.time() < deadline - 20.0:
        # perturb the current best and run a short refinement
        seq, val = _perturb_and_refine(global_best_seq, n, deadline,
                                       steps=150000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Perturb] new best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase D – final annealing‑style walk (tiny mass swaps) for any left‑over time
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        # Simple random walk that swaps a small fraction of mass between two positions
        cur = global_best_seq.copy()
        cur_val = _objective(cur)
        best = cur.copy()
        best_val = cur_val
        it = 0
        while time.time() < deadline:
            it += 1
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.random() * 0.015 * cur[i]  # move up to 1.5 % of source mass
            delta = min(delta, cur[i])
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            new_val = _objective(new_seq)
            if new_val < cur_val:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} best={new_val:.6f}")
        if best_val < global_best_val:
            global_best_val = best_val
            global_best_seq = best.copy()
            print(f"[Anneal] final best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")

    globals()["height_sequence_1"] = np.copy(global_best_seq)
    # Convert to plain Python list for the grader
    return global_best_seq.tolist()
```

This implementation builds a large, varied seed pool, quickly discards the worst candidates, applies a fast Adam phase on a smoothed max, then performs a very long exponentiated‑gradient sub‑gradient optimisation (each iteration needs only two FFTs thanks to the analytical reversal trick). Periodic random perturbations and a final lightweight annealing walk help the search escape plateaus. The whole procedure respects the 1000 s budget and stores the result in `height_sequence_1` for possible reuse.
