Algorithm
---------
1. Initialise a probability vector of length ``n``.
   * Use the best sequence known from a previous run (global
     ``height_sequence_1``) if it exists.
   * Otherwise fall back to a few inexpensive seeds (uniform,
     random soft‑max, Sidon‑like support) and keep the best of them.

2. **Exact sub‑gradient (exponentiated‑gradient) phase**  
   The objective is the non‑smooth max of the auto‑convolution.
   We keep the sequence on the simplex using the mirror‑descent
   update `log a ← log a – η·g` where `g` is a sub‑gradient obtained
   from the currently active constraints.  This phase runs for as
   many iterations as the remaining time permits (normally several
   hundred thousand steps).

3. **Linearised LP refinement** (optional, requires cvxpy)  
   When the sub‑gradient phase finishes we have a current vector
   `a` with a maximal convolution value `M = max(conv(a,a))`.
   For the (few) indices where this maximum is attained we
   linearise the quadratic constraints:
       2·conv(x, a)_k ≤ M + conv(a,a)_k .
   The resulting problem is a linear program in the new variable
   `x` (still on the simplex) and a scalar `M`.  Solving it reduces
   the true bound.  We repeat this refinement a few times or until
   the bound stops improving.

4. **Simulated annealing** – final fine‑tuning for the tiny amount
   of time left.

5. Return the best vector found, store it in the global
   ``height_sequence_1`` for the next invocation.

All operations respect the supplied time budget (default 1000 s)
and the sequence is guaranteed to be non‑negative and normalised to
sum = 1, which makes the returned objective equal to
``2·n·max(conv)``.
"""

import time
import math
import numpy as np

# optional cvxpy – used only for the LP refinement
try:
    import cvxpy as cp
    CVXPY_AVAILABLE = True
except Exception:  # pragma: no cover
    CVXPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Basic convolution utilities (exact, using real FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple seed families (return a probability vector on the simplex)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """Deterministic Sidon‑like support using triangular numbers."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a / a.sum()


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient descent (exact, non‑smooth)
# ----------------------------------------------------------------------
def _subgradient_refine(start_seq: np.ndarray,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Returns the best sequence encountered and its objective value.
    """
    a = start_seq.copy()
    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    it = 0
    while True:
        it += 1
        if time.time() > deadline:
            break

        # compute convolution and locate active indices (where the max is attained)
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient: average over all active constraints
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = 3n‑2
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                         # keep numbers stable
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:8d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linearised LP refinement (convex‑concave procedure)
# ----------------------------------------------------------------------
def _lp_linearised_refine(a: np.ndarray,
                         deadline: float,
                         max_refine: int = 8,
                         eps_active: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Perform a few LP refinements where the quadratic constraints are
    linearised around the current point `a`.  Returns the best vector
    discovered and its objective.
    """
    if not CVXPY_AVAILABLE:
        return a, _objective(a)

    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)

    # warm start variables for cvxpy
    x_var = cp.Variable(n)
    M_var = cp.Variable()

    # build problem once – constraints will be updated in each iteration
    constraints = [x_var >= 0,
                   cp.sum(x_var) == 1]

    prob = None   # will be created inside the loop when we know the active set

    for it in range(max_refine):
        if time.time() > deadline - 0.3:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - eps_active)[0]
        if active.size == 0:
            break

        # coefficient matrix A where A[row, i] = a[k-i] for k = active[row]
        A = np.zeros((active.size, n), dtype=np.float64)
        for r, k in enumerate(active):
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i_range = np.arange(i_low, i_high + 1, dtype=int)
            A[r, i_range] = a[k - i_range]

        # rebuild the LP with the fresh active constraints
        lin_constraints = [2 * (A @ x_var) <= M_var + conv[active]]
        prob = cp.Problem(cp.Minimize(M_var), constraints + lin_constraints)

        try:
            prob.solve(solver=cp.ECOS,
                       warm_start=True,
                       max_iters=2000,
                       abstol=1e-9,
                       reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if x_var.value is None or M_var.value is None:
            break

        a_new = np.asarray(x_var.value).flatten()
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new /= s

        val_new = _objective(a_new)
        print(f"[LP] iter={it+1:2d} M={M:.6f} new_obj={val_new:.6f}")

        if val_new < best_val - 1e-9:
            best_val = val_new
            best_seq = a_new.copy()
            a = a_new  # continue refining from the improved point
        else:
            # no improvement – stop early
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing fine‑tuner
# ----------------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis‑Hastings walk that swaps a small amount of mass
    between two random positions and occasionally makes a global
    Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, cur.size, size=2)
        while i == j:
            j = rng.integers(0, cur.size)

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # occasional global perturbation
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=cur.size)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot > 0:
                new_seq /= tot
            else:
                new_seq = _uniform(cur.size)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The best sequence found (as a plain list of
    floats) is returned.  The routine respects the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    rng = np.random.default_rng(seed + 12345)

    # ------------------------------------------------------------------
    # Problem size – keep a balance between quality and speed
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising step function of length {n} ===")

    # ------------------------------------------------------------------
    # Initialise a decent starting point
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.shape == (n,):
                seed_vec = previous / previous.sum()
            else:
                seed_vec = None
        except Exception:
            seed_vec = None
    else:
        seed_vec = None

    # If we have no usable previous vector, create a few cheap seeds and keep the best
    if seed_vec is None:
        simple_seeds = [
            _uniform(n),
            _random_softmax(n, rng),
            _sidon_like_seed(n, min(120, n)),
        ]
        best_val = float("inf")
        best_seq = None
        for s in simple_seeds:
            val = _objective(s)
            if val < best_val:
                best_val = val
                best_seq = s
        seed_vec = best_seq
        print(f"[Init] best initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Exact sub‑gradient (exponentiated gradient) optimisation
    # ------------------------------------------------------------------
    sub_deadline = deadline - 30.0   # keep ~30 s for later refinements
    if sub_deadline > time.time():
        print("[Phase 1] Sub‑gradient refinement start")
        seq_sub, val_sub = _subgradient_refine(seed_vec, sub_deadline)
        print(f"[Phase 1] best objective after sub‑gradient = {val_sub:.6f}")
    else:
        seq_sub, val_sub = seed_vec, _objective(seed_vec)

    # ------------------------------------------------------------------
    # Phase 2 – Linearised LP refinement (if cvxpy is available)
    # ------------------------------------------------------------------
    lp_deadline = deadline - 10.0   # reserve a few seconds for annealing
    if CVXPY_AVAILABLE and lp_deadline > time.time():
        print("[Phase 2] LP linearised refinement start")
        seq_lp, val_lp = _lp_linearised_refine(seq_sub, lp_deadline)
        if val_lp < val_sub:
            seq_sub, val_sub = seq_lp, val_lp
            print(f"[Phase 2] improved objective = {val_lp:.6f}")
        else:
            print("[Phase 2] no improvement")
    else:
        print("[Phase 2] skipped (no cvxpy or insufficient time)")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing fine‑tuning with whatever time is left
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("[Phase 3] Simulated annealing fine‑tuning")
        seq_final, val_final = _anneal_optimize(seq_sub, deadline)
        if val_final < val_sub:
            seq_sub, val_sub = seq_final, val_final
            print(f"[Phase 3] final objective = {val_final:.6f}")
        else:
            print("[Phase 3] annealing did not improve")
    else:
        print("[Phase 3] no time left for annealing")

    # ------------------------------------------------------------------
    # Final safety normalisation and export
    # ------------------------------------------------------------------
    final_seq = np.maximum(seq_sub, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
        total = final_seq.sum()
    final_seq = final_seq / total

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
