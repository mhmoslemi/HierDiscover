"""
Propose a low‑value step‑function.

Algorithm
---------
1. Build a large heterogeneous pool of ℓ₁‑normalised vectors (analytic families,
   many Dirichlet draws with various concentrations, plus the previous best).
   Keep the top few according to the cheap surrogate
   `2·n·max(conv(seq,seq))` (sum=1 ⇒ same as the true objective).

2. For each of those seeds run a two‑phase continuous optimiser:
   *Adam* on a smoothed‑max surrogate (log‑sum‑exp) with an aggressive
   temperature schedule (β grows quickly to ≈2000).  The parameters are
   chosen for larger steps than the reference implementation.
   Then a projected sub‑gradient descent on the exact max‑convolution
   (step size = η/√t, η≈0.8) to polish the result.  The best vector seen so
   far is kept globally.

3. With the remaining time run a long projected sub‑gradient descent on the
   current global best (many thousands of iterations) – this is cheap
   (one FFT per iteration) and drives the max‑convolution down further.

4. Finish with a simple simulated‑annealing that randomly redistributes the
   mass of a small random subset of coordinates; the move is accepted if it
   improves the surrogate or with a temperature‑dependent probability.
   Occasionally a symmetrisation move `a←(a+a[::-1])/2` is tried.

All steps respect the global time budget (default 1000 s).  The final
vector is stored in the global name ``height_sequence_1`` and returned as a
plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Low‑level utilities
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N).real[:L]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Surrogate: 2·n·max(conv(seq,seq)) (assumes Σseq = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

# ------------------------------------------------------------
# Analytic shape families (already ℓ₁‑normalised)
# ------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `conc` (scalar)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# ------------------------------------------------------------
# Seed collection
# ------------------------------------------------------------

def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.15))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))

    # Geometric families
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # Power‑law family
    for p in np.linspace(0.0, 4.0, 18):
        seeds.append(_power(n, p))

    # Beta family
    alphas = np.linspace(0.3, 2.0, 8)
    betas = np.linspace(0.3, 4.0, 8)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows
    for sigma_frac in (0.12, 0.18, 0.25, 0.33, 0.40):
        seeds.append(_gaussian(n, sigma_frac))

    # Tukey windows
    for alpha in (0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws – a denser sweep of concentration values
    conc_vals = (0.01, 0.02, 0.03, 0.05, 0.1, 0.2, 0.5, 1.0)
    for conc in conc_vals:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws (heavy‑tailed)
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 1_200_000,
                   lr0: float = 0.8,
                   beta0: float = 5.0,
                   beta_mul: float = 1.02,
                   beta_max: float = 2000.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    Returns the best sequence seen (as a probability vector) and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                     # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = beta0
    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex, sum = 1
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # increase smoothing temperature
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp on the convolution entries
        conv = _conv_fft(a, a)               # length 2n‑1
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)            # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # convert to log‑softmax space (chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (Σ=1, v≥0).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 5_000_000,
                          init_lr: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective,
    with Euclidean projection onto the simplex after each additive step.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]   # == _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the max at index max_idx:
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step size
        lr = init_lr / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 1_200_000,
                         init_temp_factor: float = 0.03,
                         min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Simple SA that reshuffles a random subset of coordinates.
    Occasionally proposes a symmetrisation move.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-14)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9995, min_temp)

        # choose move type
        if rng.random() < 0.02:          # symmetrisation move (2 % of steps)
            new = (cur + cur[::-1]) * 0.5
            new = _proj_simplex(new)      # keep on simplex
        else:
            k = rng.integers(2, 15)       # size of perturbed subset
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget and returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # Retrieve previous solution if available
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------
    # Choose length – 4000 is a good trade‑off (fast enough, good quality)
    # --------------------------------------------------------
    n = 4000

    # --------------------------------------------------------
    # Stage 1 – generate a large pool of seeds and keep the most promising
    # --------------------------------------------------------
    print(f"\n--- Generating seeds (n={n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    # Evaluate all seeds with the cheap surrogate (sum is 1 ⇒ value = 2·n·max(conv))
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 6
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------
    # Stage 2 – heavy optimisation (Adam + projected sub‑gradient)
    # --------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Optimisation run {idx+1}/{len(best_seeds)} ---")

        # Adam phase – aggressive temperature schedule
        seq_adam, val_adam = _adam_optimize(
            seed_vec,
            n,
            deadline,
            lr0=0.8,
            beta0=5.0,
            beta_mul=1.02,
            beta_max=2000.0,
            max_iters=1_200_000,
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # Sub‑gradient phase – a quick polishing pass (limited by remaining time)
        seq_sub, val_sub = _subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            init_lr=0.8,
            max_iters=1_000_000,
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[overall‑SubGrad] new global best = {val_sub:.6f}")

    # --------------------------------------------------------
    # Stage 3 – prolonged sub‑gradient descent on the current best
    # --------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long sub‑gradient refinement ---")
        seq_refine, val_refine = _subgradient_optimize(
            global_best_seq,
            n,
            deadline,
            init_lr=0.6,          # a slightly smaller step for the fine‑tuning phase
            max_iters=5_000_000,
        )
        if val_refine < global_best_val:
            global_best_seq = seq_refine.copy()
            global_best_val = val_refine
            print(f"[Refine] new global best = {val_refine:.6f}")

    # --------------------------------------------------------
    # Stage 4 – final stochastic polishing (simulated annealing)
    # --------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq,
            n,
            deadline,
            max_iters=1_500_000,
            init_temp_factor=0.03,
            min_temp=1e-14,
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # --------------------------------------------------------
    # Sanity check (optional)
    # --------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # store for potential reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
