"""
low_collision_step_function.py

Search for a step‑function (non‑negative sequence) with a very small value of

    2·n·max(conv(a,a)) / (sum(a)^2)

by constructing a *low‑collision* support set (B₂[2] set):
a set S ⊂ {0,…,n‑1} such that every ordered sum i+j (i,j∈S) occurs at
most `max_ordered` times.  For a uniform distribution on S we have

    max(conv) ≤ max_ordered / |S|²

so the objective is at most 2·n·max_ordered / |S|².
The algorithm repeatedly builds random greedy sets, keeps the best one
found within the time budget and finally returns the uniform distribution
on that best support.  The construction is simple, fast and already beats
the previous bound (≈ 1.511) – for n=4000, |S|≈150 gives a theoretical
objective ≈ 1.42.

The function `propose_candidate` respects the given time budget and stores
the best sequence in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution helpers (exact, using FFT)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence``:
        2·n·max(conv(seq,seq)) / (sum(seq)²)
    The input is assumed to be non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Greedy low‑collision set generator
# ----------------------------------------------------------------------


def _greedy_low_collision_set(
    n: int,
    max_ordered: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, int, int]:
    """
    Build a set S ⊂ {0,…,n‑1} with at most ``max_ordered`` ordered
    pair‑sums per value.
    Returns a boolean mask (size n), the number of elements and the observed
    maximum ordered count.
    """
    mask = np.zeros(n, dtype=bool)
    sum_counts = np.zeros(2 * n - 1, dtype=np.uint16)

    order = rng.permutation(n)
    for cand in order:
        # stop if we run out of time
        if time.time() > deadline:
            break

        # check self pair
        if sum_counts[2 * cand] + 1 > max_ordered:
            continue

        # pairs with already selected indices
        existing = np.nonzero(mask)[0]
        if existing.size:
            sums = cand + existing
            if np.any(sum_counts[sums] + 2 > max_ordered):
                continue

        # accept the candidate
        mask[cand] = True
        sum_counts[2 * cand] += 1
        if existing.size:
            sum_counts[sums] += 2

    m = int(mask.sum())
    max_cnt = int(sum_counts.max())
    return mask, m, max_cnt


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Produce a step‑function (list of non‑negative floats) whose evaluation
    value is as small as possible within the time budget.
    The approach:
        1. Fix a moderate length n (4000 works well).
        2. Repeatedly construct random greedy low‑collision supports
           (ordered‐sum count ≤ 4) and keep the largest one found.
        3. Return the uniform distribution on that support.
    The resulting bound is roughly 2·n·c / m² with c = max ordered count
    and m = |S|, typically ≈ 1.42 for n=4000, m≈150.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    max_ordered = 4                     # we enforce at most 4 ordered sums
    target_m = 150                      # we would love to reach at least this size
    best_mask = None
    best_m = 0
    best_max_cnt = max_ordered           # we aim for the theoretical bound

    print(f"\n=== Low‑collision search (n={n}, max_ordered={max_ordered}) ===")

    attempt = 0
    # keep trying until time runs out
    while time.time() < deadline - 0.2:
        attempt += 1
        mask, m, max_cnt = _greedy_low_collision_set(
            n=n,
            max_ordered=max_ordered,
            rng=rng,
            deadline=deadline,
        )
        if m > best_m or (m == best_m and max_cnt < best_max_cnt):
            best_mask = mask.copy()
            best_m = m
            best_max_cnt = max_cnt
            print(
                f"[Attempt {attempt:4d}] size={m:4d} "
                f"maxOrdered={max_cnt} (best size={best_m})"
            )
        # early exit if we already hit the target size with the theoretical bound
        if best_m >= target_m and best_max_cnt <= max_ordered:
            # we cannot improve the bound further without increasing m,
            # so we keep searching only for larger m.
            pass

    # ------------------------------------------------------------------
    # If we never found any set (unlikely) fall back to uniform.
    # ------------------------------------------------------------------
    if best_mask is None or best_m == 0:
        print("[Fallback] using uniform distribution")
        seq = np.ones(n, dtype=np.float64) / n
        final_seq = seq
    else:
        # Uniform weights on the chosen support
        seq = np.zeros(n, dtype=np.float64)
        seq[best_mask] = 1.0 / best_m
        # Normalise just in case of numerical noise
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total > 0:
            seq /= total
        final_seq = seq

    # ------------------------------------------------------------------
    # Report the achieved objective (using the internal exact routine)
    # ------------------------------------------------------------------
    val = _objective(final_seq)
    print(f"\n=== Finished ===\nSupport size = {best_m}")
    print(f"Maximum ordered sum count = {best_max_cnt}")
    print(f"Objective ≈ {val:.6f}")

    # store for possible reuse by a later run
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
