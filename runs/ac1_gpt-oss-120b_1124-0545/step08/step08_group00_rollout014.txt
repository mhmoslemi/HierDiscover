def _subgradient_opt(seq: np.ndarray, n: int, deadline: float, max_iters: int):
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-12
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # find indices where conv is within eps of max
        mask = (conv >= max_val - 1e-12)
        w = mask.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0:
            # fallback to single argmax
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum
        # compute gradient of weighted sum conv w.r.t a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n-1:n-1+n]
        # step size
        lr = 0.1 / np.sqrt(it)
        a = a - lr * grad
        # project onto simplex
        a = _proj_simplex(a)
    return best_seq, best_val
```

"""
Optimiser for the ``evaluate_sequence`` objective.

The goal is to minimise  

    2 * n * max(conv(seq, seq)) / (sum(seq) ** 2)

where ``conv`` is the full linear convolution (numpy.convolve).  
All entries must be non‑negative and the evaluation routine clips
to [0, 1000] – we therefore keep every element in this range.

Strategy
--------
1. **Rich seed pool** – a collection of analytic families (uniform,
   sine, Tukey, Gaussian, Beta, geometric, power‑law) together with a
   large number of random Dirichlet draws.  Every candidate is
   ℓ₁‑normalised (∑a_i = 1) – the objective is scaling invariant.

2. **Exact sub‑gradient descent** – the true objective is
   ``2*n*max(conv)`` (the denominator is constant =1).  The gradient
   of ``max(conv)`` is a sub‑gradient that can be written as a
   convolution; we compute it with FFTs and take a step on the
   simplex (Euclidean projection).  A decreasing step‑size
   ``ηₖ = η₀ / sqrt(k)`` is used.  The algorithm is run for as many
   iterations as the remaining time allows; we keep the best vector
   seen.

3. **Heavy‑weight refinement** – the top‑K seeds are first refined with
   a short Adam run on the smooth log‑sum‑exp surrogate (helps escape
   plateaus) and then handed to the exact sub‑gradient optimiser above.

4. **Final simulated‑annealing polishing** – random Dirichlet
   re‑allocations of a small subset of coordinates are accepted
   according to a cooling schedule, again projected onto the simplex.

All stages respect the global 1000 s budget; the best sequence is stored
in the global name ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective (the denominator is 1 because we keep the
    ℓ₁‑norm equal to 1).  Returns 2·n·max(conv(seq,seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries non‑positive – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    a /= a.sum()
    return a


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    a /= a.sum()
    return a


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (
                1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1))
            )
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    a /= a.sum()
    return a


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    a /= a.sum()
    return a


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    a /= a.sum()
    return a

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a pool of ℓ₁‑normalised candidate vectors."""
    seeds = []

    # Deterministic analytic families (small variations)
    seeds.append(_uniform(n))
    for gamma in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))

    for sigma in (0.12, 0.18, 0.24, 0.30):
        seeds.append(_gaussian(n, sigma))

    for alpha in (0.05, 0.15, 0.35, 0.55, 0.85):
        seeds.append(_tukey_window(n, alpha))

    for a in (0.4, 0.8, 1.2, 1.6):
        for b in (0.4, 1.0, 2.5, 4.0):
            seeds.append(_beta_shape(n, a, b))

    for r in np.linspace(0.1, 0.95, 9):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power_law(n, p))

    # Random Dirichlet draws (diverse concentration)
    for conc in (0.02, 0.07, 0.15, 0.4, 0.8):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

def _adam_on_surrogate(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp (soft‑max) surrogate.
    The iterate lives in the log‑domain (v) and is turned into a
    probability vector with softmax each step.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.2            # start temperature (smooth surrogate)
    beta_mul = 1.003
    beta_max = 200.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # probability vector (sum=1)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # cool the surrogate – higher β → closer to true max
        beta = min(beta * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

        # map gradient to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # simple norm clipping for numerical stability
        norm = np.linalg.norm(grad_v)
        if norm > 10.0:
            grad_v = grad_v * (10.0 / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr = 0.6 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_exact(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true objective
    2·n·max(conv(seq,seq)), with projection onto the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubG] it={it:7d} best={cur_val:.6f}")

        # indices of (approximately) maximal entries – improves stability
        mask = (conv >= max_val - 1e-14)
        w = mask.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad = 2.0 * full[n - 1:n - 1 + n]   # gradient of Σ w·conv

        # decaying step‑size
        lr = 0.25 / np.sqrt(it)
        a = a - lr * grad
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.03,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Simple SA that reshuffles the probability mass of a random subset
    of coordinates while staying on the simplex.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9998, min_temp)

        # pick a random subset (2–10 indices) and re‑distribute its mass
        k = rng.integers(2, 11)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best list of heights found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # small safety margin

    # possibly re‑use the solution from a previous run
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length of the step function – a power of two yields the fastest FFT.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Stage 1 – diverse seeds
    # ------------------------------------------------------------------
    print(f"\n=== Stage 1 – seed generation (n={n}) ===")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} raw seeds.")

    # evaluate surrogate objective (all seeds are ℓ₁‑normalised)
    scored = [(vec, _objective(vec)) for vec in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 5
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best initial surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – main optimisation (Adam → exact sub‑gradient)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n=== Optimisation round {idx + 1}/{len(best_seeds)} ===")

        # Adam on surrogate – quick, helps to escape bad plateaus
        seq_adam, val_adam = _adam_on_surrogate(
            seed_vec, n, deadline, max_iters=250_000
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[global] Adam improved best to {val_adam:.6f}")

        # Exact sub‑gradient descent – long, drives objective down
        seq_sub, val_sub = _subgradient_exact(
            global_best_seq, n, deadline, max_iters=1_200_000
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[global] Sub‑gradient improved best to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final stochastic polishing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n=== Stage 3 – simulated annealing polishing ===")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=600_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[global] SA improved best to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (calls the provided evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_seq.tolist())
        print(f"\nFinal evaluate_sequence score = {final_score:.6f}")
    except Exception as exc:
        print("Warning – evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
