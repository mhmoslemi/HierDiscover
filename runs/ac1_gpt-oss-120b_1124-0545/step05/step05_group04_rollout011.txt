"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The implementation below builds on the earlier “analytic → smooth‑max → sub‑gradient → local‑search”
pipeline but adds a handful of new ingredients that typically improve the final bound:

* **More diverse seeds** – in addition to the analytical families we draw many
  Dirichlet‑distributed random vectors with different concentration parameters.
* **Adam with adaptive learning‑rate schedule** – a longer continuation on λ and a
  learning‑rate that shrinks as λ grows, which yields a more accurate approach to the
  true (non‑smoothed) max‑objective.
* **Polyak step‑size sub‑gradient descent** – instead of a simple 1/√t schedule we use
  step = (f(x)‑f_best) / (‖g‖²+ε) (with a floor to stay safe).  This often gives
  faster reduction of the exact objective.
* **Simulated‑annealing mass‑transfer moves** – stochastic moves are accepted with a
  temperature schedule, allowing the search to escape shallow local minima.
* **Population based crossover** – a tiny population (size = 4) of the current best
  solutions is maintained; new candidates are generated by convex combinations and
  occasional random perturbations.

All heavy‑lifting (convolution, soft‑max, projection) is performed with NumPy‑FFT;
the code respects the 1000 s time‑budget and returns a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a.
    The vector is first normalised to unit sum (the evaluation routine does the same).
    Returns 2·n·max(conv(a,a)).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    return float(2.0 * a.shape[0] * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = u - cssv / (np.arange(n) + 1.0)
    rho_idx = np.where(rho > 0)[0]
    if rho_idx.size == 0:
        theta = 0.0
    else:
        rho = rho_idx[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one index to another.
    The amount transferred is `scale * a[i] * U(0,1)`.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families (same as before, kept for stability)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exp_decay(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - center))
    a = a / a.sum()
    return a


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max utilities (Φ_λ and its gradient)
# ----------------------------------------------------------------------


def _phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    Compute the smooth approximation Φ_λ and its gradient w.r.t. the log‑probability
    vector `x`.  Returns (phi, grad_x).
    """
    # a = softmax(x)  (always on the simplex)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    conv = conv_fft_self(a)                 # linear convolution, length 2n‑1
    max_c = np.max(conv)

    # smooth‑max over all convolution entries
    shift = conv - max_c                    # ≤ 0
    exp_term = np.exp(lam * shift)         # ≤ 1
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam

    w = exp_term / sum_exp                  # soft‑max weights (sum to 1)

    # gradient of Φ_λ w.r.t. a (see paper notes)
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]          # length n
    grad_a = 2.0 * slice_part[::-1]                # ∂Φ/∂a

    # map through Jacobian of soft‑max
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return phi, grad_x


def adam_optimize(x: np.ndarray, lam: float, n: int,
                  steps: int, lr: float, deadline: float) -> np.ndarray:
    """Adam optimiser for the smooth‑max objective."""
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for t in range(1, steps + 1):
        if time.time() >= deadline:
            break
        _, grad = _phi_and_grad(x, lam, n)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step
    return x


# ----------------------------------------------------------------------
# Exact sub‑gradient utilities (Polyak step‑size)
# ----------------------------------------------------------------------


def active_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv(a,a)) at the current point.
    Returns a vector of the same shape as a.
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    # binary mask of active (maximal) indices – tolerance to avoid numerical noise
    active = (conv >= max_val - 1e-12).astype(np.float64)
    cnt = active.sum()
    if cnt == 0:
        return np.zeros_like(a)

    w = active / cnt                          # uniform weight over the active set
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]                # ∂max/∂a
    return grad


def polyak_subgrad_step(a: np.ndarray, best_val: float, eps: float = 1e-12) -> np.ndarray:
    """
    One projected Polyak step:  step = (f(a)-f_best) / (||g||²+ε).
    """
    n = a.shape[0]
    f_val = compute_score(a)                     # exact objective (2·n·max)
    g = active_subgradient(a)
    norm2 = np.dot(g, g)
    if norm2 < eps:
        return a
    step = (f_val - best_val) / (norm2 + eps)
    # Avoid exploding steps – cap to a modest value
    step = np.clip(step, 1e-6, 0.5)
    a_new = a - step * g
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Simulated annealing utilities
# ----------------------------------------------------------------------


def anneal_move(a: np.ndarray, scale: float, temp: float) -> np.ndarray:
    """
    Perform a random mass‑transfer move and decide acceptance by the
    Metropolis rule with temperature `temp`.
    Returns the (potentially) new vector.
    """
    cand = random_move(a, scale)
    cur_score = compute_score(a)
    cand_score = compute_score(cand)

    if cand_score < cur_score:
        return cand, cand_score
    # otherwise accept with probability exp(-(Δ)/temp)
    prob = np.exp(-(cand_score - cur_score) / max(temp, 1e-12))
    if np.random.rand() < prob:
        return cand, cand_score
    return a, cur_score


# ----------------------------------------------------------------------
# Population crossover utilities
# ----------------------------------------------------------------------


def crossover(parents, alpha=0.5):
    """
    Generate a child vector as a convex combination of two randomly chosen parents.
    """
    i, j = np.random.choice(len(parents), size=2, replace=False)
    p1, p2 = parents[i], parents[j]
    child = alpha * p1 + (1.0 - alpha) * p2
    # small random jitter to keep diversity
    child = random_move(child, scale=0.01)
    return child

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimises the step‑function using a multi‑phase strategy:
    1. Analytic + Dirichlet seeds → best initial point.
    2. Continuation of smooth‑max (Adam) with a richer λ‑schedule.
    3. Polyak‑step sub‑gradient descent.
    4. Simulated‑annealing exploration with a tiny population.
    5. Light random‑mass transfers for final polishing.

    The function respects the time budget and always returns the best sequence
    found before the deadline (as a plain Python list).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5               # safety margin

    # ------------------------------------------------------------------
    # Problem size (default 2000, can be overridden via kwargs)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 0) Initialise with a diverse set of seeds
    # ------------------------------------------------------------------
    best_a = np.ones(n) / n          # uniform (objective = 2)
    best_score = compute_score(best_a)

    # Use any previously known construction if present
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n and ref.sum() > 0:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            if sc < best_score:
                best_score, best_a = sc, ref.copy()

    # Analytic families (same grids as before, but a bit tighter)
    p_vals = np.linspace(0.30, 0.70, 81)
    off_vals = np.linspace(0.0001, 2.0, 81)
    c_vals = np.linspace(0.0005, 0.0200, 41)
    sigma_vals = np.linspace(80.0, 800.0, 41)

    for mode in ("product", "sum"):
        for p in p_vals:
            for off in off_vals:
                if time.time() >= deadline:
                    break
                cand = analytic_product(n, p, off) if mode == "product" else analytic_sum(n, p, off)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score, best_a = sc, cand.copy()

    for c in c_vals:
        if time.time() >= deadline:
            break
        cand = analytic_exp_decay(n, c)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        cand = analytic_gaussian(n, sigma)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    # Random Dirichlet seeds – spread concentration from 0.1 to 10.0
    dirichlet_alphas = np.concatenate([np.logspace(-1, 1, 12), [0.5, 1.0, 2.0, 5.0]])
    for alpha in dirichlet_alphas:
        if time.time() >= deadline:
            break
        cand = np.random.dirichlet([alpha] * n)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    print(f"[seed] best score after seeding = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Smooth‑max optimisation (Adam) – richer λ schedule
    # ------------------------------------------------------------------
    eps = 1e-12
    x = np.log(np.maximum(best_a, eps))

    # λ values start tiny and climb up to 1e5 (smooth → true max)
    lam_schedule = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0,
                    2000.0, 8000.0, 20000.0, 80000.0, 150000.0, 300000.0]

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        # heuristics for step count – we allocate a few thousand steps
        # but never exceed the remaining time budget.
        remaining = deadline - time.time()
        max_iters = int(remaining / 0.001)          # rough 1 ms / iteration
        iters = min(4000, max_iters)               # cap to keep it manageable
        if iters <= 0:
            break
        # learning‑rate shrinks as λ grows
        lr = 0.1 / np.sqrt(lam + 1.0)
        x = adam_optimize(x, lam, n, iters, lr, deadline)
        a = softmax(x)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[adam λ={lam:.0f}] new best = {best_score:.6f} (t={time.time() - start:.1f}s)")

    # ------------------------------------------------------------------
    # 2) Polyak sub‑gradient descent on the exact (non‑smoothed) objective
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        a = polyak_subgrad_step(a, best_score)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[polyak iter={it}] new best = {best_score:.6f} (t={time.time() - start:.1f}s)")
        # occasional restart to avoid stagnation
        if it % 5000 == 0:
            # blend current best with a fresh Dirichlet draw
            fresh = np.random.dirichlet([0.5] * n)
            a = 0.7 * best_a + 0.3 * fresh
            a = project_simplex(a)

    # ------------------------------------------------------------------
    # 3) Simulated‑annealing with a tiny population
    # ------------------------------------------------------------------
    # initialise a population of five good candidates (including best)
    pop = [best_a.copy()]
    while len(pop) < 5 and time.time() < deadline:
        cand = np.random.dirichlet([0.7] * n)
        pop.append(cand)

    temp = 1.0
    scale = 0.02
    sa_iter = 0
    while time.time() < deadline:
        sa_iter += 1
        # slowly cool down
        temp = max(0.01, temp * 0.99995)

        # pick a parent from population (biased towards better scores)
        scores = np.array([compute_score(p) for p in pop])
        probs = np.exp(-scores) / np.sum(np.exp(-scores))
        parent_idx = np.random.choice(len(pop), p=probs)
        parent = pop[parent_idx]

        # propose a move
        cand, cand_score = anneal_move(parent, scale, temp)

        # keep it in population if it improves the local slot
        # replace the worst member if better
        worst_idx = np.argmax(scores)
        if cand_score < scores[worst_idx]:
            pop[worst_idx] = cand
            scores[worst_idx] = cand_score

        # update global best
        if cand_score < best_score:
            best_score, best_a = cand_score, cand.copy()
            print(f"[SA iter={sa_iter}] new global best = {best_score:.6f} (t={time.time() - start:.1f}s)")

        # occasional crossover to inject fresh material
        if sa_iter % 500 == 0:
            child = crossover(pop, alpha=0.5)
            child_score = compute_score(child)
            if child_score < best_score:
                best_score, best_a = child_score, child.copy()
                print(f"[crossover] new best = {best_score:.6f} (t={time.time() - start:.1f}s)")
            # replace worst if child is good enough
            if child_score < scores[worst_idx]:
                pop[worst_idx] = child
                scores[worst_idx] = child_score

    # ------------------------------------------------------------------
    # 4) Final fine‑grained random mass‑transfer local search
    # ------------------------------------------------------------------
    a = best_a.copy()
    scale = 0.015
    while time.time() < deadline:
        cand = random_move(a, scale)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[final] new best = {best_score:.6f} (t={time.time() - start:.1f}s)")
            a = cand
            # tighten step size after an improvement
            scale = max(0.001, scale * 0.9)
        else:
            # slowly relax step size to keep exploring
            scale = min(0.03, scale * 1.001)
            a = cand

    total = time.time() - start
    print(f"[done] best score = {best_score:.6f}, total time = {total:.1f}s")
    # Return a plain Python list as required by the external harness
    return best_a.tolist()
