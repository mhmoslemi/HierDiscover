"""
Search for a non‑negative step function that minimizes
    2·n·max(conv(a,a))/(sum(a)**2)

Algorithm:
1. Analytic sweep over a rich family of edge‑heavy shapes
   (product‑law, sum‑law) including mixture with a uniform component.
2. Simple stochastic hill‑climbing on the three parameters
   (mode, p, offset, α) to refine the best candidate.
3. Continuation with a smoothed‑max objective and Adam optimisation.
4. Projected sub‑gradient descent + random local mass‑transfer moves.
All phases respect the time budget and keep the best sequence found.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def conv_fft_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]

def compute_score(a: np.ndarray) -> float:
    """Exact objective for a non‑negative vector a (already normalised to sum=1)."""
    n = a.shape[0]
    # Normalise to avoid rounding issues
    s = a.sum()
    if s <= 0:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))

def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    # Avoid overflow: compute on log scale if needed
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a

def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a

def analytic_exp_edge(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c * min(i, n-1-i)) – heavy at the edges, decaying towards the centre."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-c * d)
    a = a / a.sum()
    return a

def softmax(x: np.ndarray) -> np.ndarray:
    m = np.max(x)
    e = np.exp(x - m)
    e_sum = e.sum()
    if e_sum == 0:
        return np.full_like(x, 1.0 / x.shape[0])
    return e / e_sum

def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / exp_x.sum()
    conv = conv_fft_self(a)                # length 2n‑1
    max_c = np.max(conv)
    shift = conv - max_c
    exp_term = np.exp(lam * shift)
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam

    w = exp_term / sum_exp                 # weights for each lag
    # gradient of Φ w.r.t a via convolution with reversed w
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]      # length n
    grad_a = 2.0 * slice_part[::-1]            # dΦ/da

    # map gradient to x-space (softmax Jacobian)
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return phi, grad_x

def adam_optimize(x: np.ndarray, lam: float, n: int, steps: int,
                  lr: float, deadline: float) -> np.ndarray:
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for t in range(1, steps + 1):
        if time.time() >= deadline:
            break
        _, grad = compute_phi_and_grad(x, lam, n)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step
    return x

def project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

def subgrad_step(a: np.ndarray, step: float) -> np.ndarray:
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    active = (conv >= max_val - 1e-12).astype(np.float64)
    cnt = active.sum()
    if cnt == 0:
        return a
    w = active / cnt
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]
    a_new = a - step * grad
    return project_simplex(a_new)

            best_score, best_a = sc, a.copy()
            print(f"[adam λ={lam:.1f}] new best = {best_score:.6f} (t={time.time()-start_time:.1f}s)")

            print(f"[subgrad iter={it}] new best = {best_score:.6f} (t={time.time()-start_time:.1f}s)")

    a = best_a.copy()

    print(f"[done] final best score = {best_score:.6f}, total time = {total_time:.1f}s")

"""
Search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. **Extensive analytic sweep** – builds edge‑heavy candidates from the
   product‑law and sum‑law families, each possibly mixed with a uniform
   component.  The sweep covers a wide range of exponent p, offset δ and
   mixing weight α.

2. **Stochastic refinement** – a tiny simulated‑annealing / hill‑climbing
   loop perturbs the (mode, p, δ, α) parameters and keeps any improvement.
   The perturbations shrink slowly, preventing premature stagnation.

3. **Smooth‑max continuation** – the objective is approximated by the
   smooth function Φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k).  An Adam optimiser
   runs over a sequence of increasing λ values, starting from the best
   discrete candidate.

4. **Exact sub‑gradient descent** – after smoothing the true max‑convolution
   objective is tackled directly with a projected sub‑gradient method.

5. **Light random mass transfer** – occasional stochastic moves (transfer
   a tiny amount of probability mass from one index to another) explore the
   neighbourhood of the current best point.

All phases respect the supplied time budget; the best sequence found when
the deadline is reached is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """Linear (zero‑padded) convolution of a with itself using FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a (assumed normalised to sum=1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    s = a.sum()
    if s <= 0.0:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))

# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ ((i+δ)*(n‑1‑i+δ))^(‑p)   (U‑shaped)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+δ)^(‑p) + (n‑1‑i+δ)^(‑p)   (two tails)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exp_edge(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(‑c·min(i, n‑1‑i)) – heavy at the edges, exponential decay."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-c * d)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max utilities for the smoothed max objective
# ----------------------------------------------------------------------


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(x, 1.0 / x.shape[0])
    return e / s


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    For a log‑probability vector `x`, smoothing `λ` and length `n`,
    return the smooth max Φ_λ(a) and its gradient w.r.t. `x`.
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / exp_x.sum()

    conv = conv_fft_self(a)               # length 2·n‑1
    max_c = np.max(conv)

    # smooth‑max over the convolution entries
    shift = conv - max_c                  # ≤ 0
    exp_term = np.exp(lam * shift)       # ≤ 1
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam

    w = exp_term / sum_exp                # weights, sum to 1
    # gradient w.r.t. a via convolution with reversed w
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad_a = 2.0 * slice_part[::-1]       # ∂Φ/∂a

    # map to x‑space using the Jacobian of soft‑max
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return phi, grad_x


def adam_optimize(x: np.ndarray, lam: float, n: int, steps: int,
                  lr: float, deadline: float) -> np.ndarray:
    """Adam optimiser for the smooth‑max objective."""
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for t in range(1, steps + 1):
        if time.time() >= deadline:
            break
        _, grad = compute_phi_and_grad(x, lam, n)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step
    return x


# ----------------------------------------------------------------------
# Projection utilities for the simplex (used by sub‑gradient & random moves)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def subgrad_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step for the exact max‑convolution objective.
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    active = (conv >= max_val - 1e-12).astype(np.float64)
    cnt = active.sum()
    if cnt == 0:
        return a
    w = active / cnt               # uniform over active indices
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]
    a_new = a - step * grad
    return project_simplex(a_new)

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the supplied evaluation function.  The routine respects the given time
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = int(kwargs.get("n", 2000))

    uniform = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # 0) Initialise from a previously discovered construction (if any)
    # --------------------------------------------------------------
    best_a = uniform.copy()
    best_score = compute_score(best_a)

    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n and ref.sum() > 0.0:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            if sc < best_score:
                best_score, best_a = sc, ref.copy()
                print(f"[init] loaded previous best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1) Extensive analytic sweep (product, sum) with uniform mixing
    # --------------------------------------------------------------
    # Parameter grids
    p_vals = np.concatenate([np.linspace(0.05, 0.5, 20),
                             np.linspace(0.5, 2.0, 30)])          # ~50 values
    offset_vals = np.logspace(-12, 0, 40)                        # 1e‑12 … 1
    alpha_vals = np.linspace(0.0, 1.0, 11)                       # mix weight for edge‑heavy part

    best_params = {"mode": "product", "p": 0.0, "off": 1.0, "alpha": 0.0}

    for mode in ("product", "sum"):
        if time.time() >= deadline:
            break
        for p in p_vals:
            if time.time() >= deadline:
                break
            for off in offset_vals:
                if time.time() >= deadline:
                    break
                # base shape (edge‑heavy)
                base = analytic_product(n, p, off) if mode == "product" else analytic_sum(n, p, off)
                for alpha in alpha_vals:
                    a = alpha * base + (1.0 - alpha) * uniform
                    a = a / a.sum()               # protect against rounding drift
                    sc = compute_score(a)
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        best_params = {"mode": mode, "p": p, "off": off, "alpha": alpha}
                        print(f"[grid] new best = {best_score:.6f} "
                              f"(mode={mode}, p={p:.4f}, off={off:.2e}, α={alpha:.2f})")

    # --------------------------------------------------------------
    # 1b) Tiny extra family: exponential edge decay (optional)
    # --------------------------------------------------------------
    c_vals = np.linspace(0.001, 0.05, 20)
    for c in c_vals:
        if time.time() >= deadline:
            break
        base = analytic_exp_edge(n, c)
        for alpha in alpha_vals:
            a = alpha * base + (1.0 - alpha) * uniform
            a = a / a.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                best_params = {"mode": "exp_edge", "c": c, "alpha": alpha}
                print(f"[exp] new best = {best_score:.6f} (c={c:.4f}, α={alpha:.2f})")

    print(f"[grid] best after analytic sweep = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Stochastic refinement of the parametric family
    # --------------------------------------------------------------
    cur_params = best_params.copy()
    cur_a = best_a.copy()
    cur_score = best_score

    sigma_p = 0.12
    sigma_off = 0.25
    sigma_alpha = 0.07
    sigma_c = 0.15      # only used when mode == "exp_edge"

    it_refine = 0
    while time.time() < deadline:
        it_refine += 1
        # possibly switch between product/sum/exp_edge
        mode_new = cur_params["mode"]
        if np.random.rand() < 0.03:
            mode_new = np.random.choice(["product", "sum", "exp_edge"])
        # perturb parameters
        if mode_new in ("product", "sum"):
            p_new = max(1e-6, cur_params["p"] * np.exp(np.random.randn() * sigma_p))
            off_new = max(1e-12, cur_params["off"] * np.exp(np.random.randn() * sigma_off))
            alpha_new = np.clip(cur_params["alpha"] + np.random.randn() * sigma_alpha, 0.0, 1.0)
            base_new = analytic_product(n, p_new, off_new) if mode_new == "product" else analytic_sum(n, p_new, off_new)
        else:   # exp_edge
            c_new = max(1e-6, cur_params["c"] * np.exp(np.random.randn() * sigma_c))
            alpha_new = np.clip(cur_params["alpha"] + np.random.randn() * sigma_alpha, 0.0, 1.0)
            base_new = analytic_exp_edge(n, c_new)

        cand = alpha_new * base_new + (1.0 - alpha_new) * uniform
        cand = cand / cand.sum()
        sc = compute_score(cand)

        # global update
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            best_params = {"mode": mode_new}
            if mode_new in ("product", "sum"):
                best_params.update({"p": p_new, "off": off_new, "alpha": alpha_new})
            else:
                best_params.update({"c": c_new, "alpha": alpha_new})
            print(f"[refine] new global best = {best_score:.6f} (iter={it_refine})")

        # local walk update
        if sc < cur_score:
            cur_score = sc
            cur_a = cand.copy()
            cur_params = {"mode": mode_new}
            if mode_new in ("product", "sum"):
                cur_params.update({"p": p_new, "off": off_new, "alpha": alpha_new})
            else:
                cur_params.update({"c": c_new, "alpha": alpha_new})

        # decay perturbations slowly
        sigma_p   *= 0.9999
        sigma_off *= 0.9999
        sigma_alpha *= 0.9999
        sigma_c   *= 0.9999

        # occasional reset of step sizes to keep exploration alive
        if sigma_p < 1e-5 and sigma_off < 1e-5 and sigma_alpha < 1e-4:
            sigma_p, sigma_off, sigma_alpha, sigma_c = 0.12, 0.25, 0.07, 0.15

    print(f"[refine] finished – best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Smooth‑max optimisation (continuation on λ)
    # --------------------------------------------------------------
    eps = 1e-12
    x = np.log(np.maximum(best_a, eps))

    lam_schedule = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0,
                    2000.0, 8000.0, 20000.0, 80000.0]

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # estimate affordable iterations (≈0.001 s per Adam step)
        max_steps = int(remaining / 0.001)
        steps = min(3000, max_steps)
        if steps <= 0:
            break
        x = adam_optimize(x, lam, n, steps, lr=0.05, deadline=deadline)
        a = softmax(x)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[adam λ={lam:.1f}] new best = {best_score:.6f} (t={time.time()-start_time:.1f}s)")

    # --------------------------------------------------------------
    # 4) Exact sub‑gradient descent on the true max‑convolution
    # --------------------------------------------------------------
    a = best_a.copy()
    it_sg = 0
    while time.time() < deadline:
        it_sg += 1
        step = 0.5 / np.sqrt(it_sg)
        a = subgrad_step(a, step)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[subgrad iter={it_sg}] new best = {best_score:.6f} (t={time.time()-start_time:.1f}s)")

    # --------------------------------------------------------------
    # 5) Light random‑mass‑transfer local search
    # --------------------------------------------------------------
    a = best_a.copy()
    scale = 0.02
    while time.time() < deadline:
        cand = random_move(a, scale)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[random] new best = {best_score:.6f} (t={time.time()-start_time:.1f}s)")
            a = cand
            scale = max(0.001, scale * 0.9)
        else:
            scale = min(0.05, scale * 1.001)
            a = cand

    total = time.time() - start_time
    print(f"[done] final best score = {best_score:.6f}, total time = {total:.1f}s")
    return best_a.tolist()
