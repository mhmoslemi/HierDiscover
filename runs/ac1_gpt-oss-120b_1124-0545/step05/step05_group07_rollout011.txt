"""
Propose a non‑negative step function that (approximately) minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Coarse grid search** over several analytic families (product, sum,
   exponential, cosine, beta) with a broad range of parameters.  Each
   candidate is normalised to the probability simplex and scored with a
   fast FFT based exact evaluation.

2. **Local stochastic refinement** – a simple hill‑climber that repeatedly
   transfers a small random amount of mass between two positions (or adds
   a tiny Gaussian jitter) and keeps the change only when it strictly
   improves the exact score.  The step size is gradually reduced.

3. **Optional short Adam‑style polishing** (used only if time remains) that
   minimises a smooth approximation of the max‑convolution and projects
   onto the simplex after each update.

All helper functions are defined at the top level; there are no closures.
The routine respects the 1000 s wall‑time budget and prints progress
messages.  The returned list is a valid candidate for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (zero‑padded) using FFT.  Length = 2·len(a)‑1."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two arbitrary 1‑D arrays using FFT."""
    size = _next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: len(a) + len(b) - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a.
    The vector is first normalised to sum‑one, then
    2·n·max(conv(a,a)) is returned.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {w | w_i ≥ 0, Σ w_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny random amount of mass from a random source to a random target."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families (all are already normalised to sum‑one)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """U‑shaped product family: a_i ∝ ((i+off)*(n‑1‑i+off))^{‑p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """Sum of two power‑laws from the ends."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """Exponential tails: a_i ∝ exp(‑lam·min(i,n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a = a / a.sum()
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """Cosine‑power family: a_i ∝ |cos(π·i/(n‑1))|^{p}."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    a = a / a.sum()
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped family: a_i ∝ (i+1)^{α}·(n‑i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Smoothed‑max gradient (used only for a short optional polishing phase)
# ----------------------------------------------------------------------


def smooth_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)."""
    n = a.shape[0]
    conv = conv_fft(a)                         # length 2n‑1
    maxc = conv.max()
    exp_alpha = np.exp(alpha * (conv - maxc))
    w = exp_alpha / exp_alpha.sum()           # soft‑max weights, Σ w = 1

    rev_a = a[::-1]
    g_full = conv_fft_general(rev_a, w)       # length 3n‑2
    start = n - 1
    g = 2.0 * g_full[start:start + n]
    return g

# Main entry point
# ----------------------------------------------------------------------

    Search for a step‑function (list of non‑negative floats) that minimises
    the given evaluation function within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a small safety margin

    # --------------------------------------------------------------
    # Determine length of the sequence
    # --------------------------------------------------------------
    if "n" in kwargs:
        n = int(kwargs["n"])
    elif "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        n = 2000

    # --------------------------------------------------------------
    # Initial candidate (global if available)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        seq0 = np.maximum(seq0, 0.0)
        seq0 = project_simplex(seq0)
        sc0 = compute_score(seq0)
        best_seq = seq0
        best_score = sc0
        print("[init] global candidate score =", best_score)

    # --------------------------------------------------------------
    # Coarse grid search over analytic families
    # --------------------------------------------------------------
    # helper to evaluate and possibly replace the best sequence
    def try_candidate(cand: np.ndarray, tag: str):
        nonlocal best_seq, best_score
        sc = compute_score(cand)
        if sc < best_score:
            best_seq = cand
            best_score = sc
            print(f"[grid] {tag} -> new best {best_score:.6f}")

    # Uniform baseline
    try_candidate(np.ones(n) / n, "uniform")

    # 1) product family
    p_vals = np.linspace(0.3, 1.8, 30)
    off_vals = np.linspace(0.05, 1.5, 30)
    for p in p_vals:
        for off in off_vals:
            cand = analytic_product(n, p, off)
            try_candidate(cand, f"product p={p:.3f} off={off:.3f}")

    # 2) sum family
    for p in np.linspace(0.2, 2.0, 30):
        for off in np.linspace(0.05, 1.5, 30):
            cand = analytic_sum(n, p, off)
            try_candidate(cand, f"sum p={p:.3f} off={off:.3f}")

    # 3) exponential tails
    for lam in np.linspace(0.0005, 0.02, 30):
        cand = analytic_exponential(n, lam)
        try_candidate(cand, f"exp lam={lam:.5f}")

    # 4) cosine power
    for p in np.linspace(0.5, 5.0, 30):
        cand = analytic_cosine(n, p)
        try_candidate(cand, f"cosine p={p:.3f}")

    # 5) beta family (symmetric α=β)
    for a in np.linspace(-1.5, 0.5, 30):
        cand = analytic_beta(n, a, a)
        try_candidate(cand, f"beta a={a:.3f}")

    print("[grid] best after coarse search =", best_score)

    # --------------------------------------------------------------
    # Local stochastic refinement (simple hill‑climber)
    # --------------------------------------------------------------
    # The step size slowly decays so we first explore large moves,
    # then fine‑tune.
    move_scale = 0.3
    it = 0
    while time.time() < deadline:
        if np.random.rand() < 0.7:
            cand = random_move(best_seq, scale=move_scale)
        else:
            # Gaussian jitter
            cand = best_seq + np.random.normal(scale=1e-3, size=n)
            cand = np.clip(cand, 0.0, None)
            cand = project_simplex(cand)

        sc = compute_score(cand)
        if sc + 1e-12 < best_score:
            best_seq = cand
            best_score = sc
            print(f"[refine] iter {it:07d} new best {best_score:.6f}")

        # decay the scale very gently
        move_scale = max(1e-4, move_scale * 0.9995)
        it += 1

    # --------------------------------------------------------------
    # Optional short Adam polishing (used only if time remains)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                     # allocate at most ~30 s
        a = best_seq.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        alpha_min, alpha_max = 1.0, 200.0
        alpha = alpha_min
        growth_factor = (alpha_max / alpha_min) ** (1.0 / 150000.0)
        step = 0
        while time.time() < deadline:
            grad = smooth_max_gradient(a, alpha)

            step += 1
            lr = 0.05 / (1.0 + step * 1e-5)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** step)
            v_hat = v / (1.0 - beta2 ** step)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = np.clip(a, 0.0, None)
            a = project_simplex(a)

            alpha = min(alpha_max, alpha * growth_factor)

            if step % 200 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_seq = a.copy()
                    best_score = sc
                    print(f"[adam] step {step} improved best {best_score:.6f}")

            if step >= 150000:   # safety cap
                break

    total = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, n = {n}, time = {total:.1f}s")
    return best_seq.tolist()
