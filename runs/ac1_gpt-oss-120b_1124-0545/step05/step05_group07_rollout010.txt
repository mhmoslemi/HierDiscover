    a = np.exp(z)
    s = a.sum()

"""
Propose a non‑negative step function that (approximately) minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Broad analytic sweep** – generate many candidate vectors from a
   collection of simple analytic families (product, sum‑power,
   beta‑shaped, exponential, cosine) together with a handful of random
   Dirichlet draws.  The exact score is computed for each and the best
   few are kept.

2. **L‑BFGS optimisation of a smooth surrogate** – each kept candidate
   is optimised in the log‑space of the simplex (z_i = log a_i) using
   a smooth approximation of the max‑convolution
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k) .
   A short L‑BFGS run is performed for a sequence of increasing
   temperatures α = 10 → 30 → 80.  The gradient of the surrogate is
   obtained analytically via FFTs, and the chain rule converts it to
   the gradient w.r.t. the log‑variables.  The best result among all
   starts is retained.

3. **Final stochastic refinement** – on the remaining time budget a
   simple Gaussian perturbation (projected back onto the probability
   simplex) is applied repeatedly; any improvement in the exact score
   is kept.

All heavy work uses FFT‑based convolutions, so the routine comfortably
fits into the 1000 s budget for vectors of length ≈2000.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (zero‑padded) using FFT.
    Returns an array of length 2·len(a)‑1."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two arbitrary 1‑D arrays using FFT.
    Returns an array of length len(a)+len(b)-1."""
    size = _next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: len(a) + len(b) - 1]


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a non‑negative vector a.
    The vector is assumed to lie on the simplex (Σa = 1); a safety
    normalisation is performed nevertheless.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex:
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def smooth_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of the smooth approximation
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    using FFTs.  Returns a vector of length len(a)."""
    n = a.shape[0]
    conv = conv_fft(a)                      # length 2n‑1
    maxc = conv.max()
    exp_alpha = np.exp(alpha * (conv - maxc))
    w = exp_alpha / exp_alpha.sum()         # soft‑max weights, Σ w = 1
    rev_a = a[::-1]
    g_full = conv_fft_general(rev_a, w)     # length 3n‑2
    start = n - 1
    g = 2.0 * g_full[start:start + n]
    return g


# ----------------------------------------------------------------------
# Analytic families (non‑negative, summed to 1)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """U‑shaped family: a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """Sum of power‑laws from both ends."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """Exponential tails: a_i ∝ exp(-lam·min(i,n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a = a / a.sum()
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """Cosine power family: a_i ∝ |cos(πi/(n‑1))|^{p}."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    a = a / a.sum()
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped family: a_i ∝ (i+1)^{α}·(n-i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Smooth‑max surrogate (value + gradient) for L‑BFGS
# ----------------------------------------------------------------------


def smooth_obj(z: np.ndarray, n: int, alpha: float) -> float:
    """Smooth surrogate 2·n·f_α(a) where a = softmax(z)."""
    a = np.exp(z)
    s = a.sum()
    if s == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / s
    conv = conv_fft(a)
    maxc = conv.max()
    # stable log‑sum‑exp
    exp_shift = np.exp(alpha * (conv - maxc))
    log_sum = np.log(exp_shift.sum())
    f_alpha = maxc + log_sum / alpha
    return 2.0 * n * f_alpha


def smooth_obj_grad(z: np.ndarray, n: int, alpha: float) -> np.ndarray:
    """Gradient of smooth_obj w.r.t. log‑variables z."""
    a = np.exp(z)
    s = a.sum()
    a = a / s
    # gradient w.r.t. a of the smoothed max
    grad_a = 2.0 * n * smooth_max_gradient(a, alpha)
    # chain rule: dz = a * (grad_a - <a,grad_a>)
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    return grad_z

            print(f"[refine] improved to {best_score:.6f}")
        iter_count += 1

    total = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, length = {n}, time = {total:.1f}s")
    return best_seq.tolist()
