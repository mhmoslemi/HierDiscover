def mirror_descent_step(a, step):
    # Compute subgrad of max conv: grad_a as in subgrad_step but not subtract step
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    active = (conv >= max_val - 1e-12).astype(np.float64)
    cnt = active.sum()
    w = active / cnt
    # gradient = 2 * convweight with a
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    grad = 2.0 * conv_wa[n - 1: 2 * n - 1][::-1]
    # Exponentiated gradient update
    a_new = a * np.exp(-step * grad)
    a_new = a_new / a_new.sum()
    return a_new
```

def gradient_move(a, scale):
    grad = get_max_conv_subgrad(a)  # returns grad
    # pick i with probability proportional to grad positive values
    pos = grad > 0
    if not np.any(pos):
        return a
    prob_i = grad[pos] / grad[pos].sum()
    i = np.random.choice(np.where(pos)[0], p=prob_i)
    # pick j with probability proportional to -grad (negative)
    neg = grad < 0
    if np.any(neg):
        prob_j = (-grad[neg]) / (-grad[neg]).sum()
        j = np.random.choice(np.where(neg)[0], p=prob_j)
    else:
        # pick random among all not i
        j = np.random.randint(0, len(a)-1)
        while j == i:
            j = np.random.randint(0, len(a)-1)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = project_simplex(a_new)
    return a_new

def lp_step(f, M):
    # solve maximize sum(b) subject to conv_f_b <= M, b>=0
    n = len(f)
    # build A (2n-1, n)
    A = np.zeros((2*n - 1, n), dtype=np.float64)
    for i in range(n):
        rows = i + np.arange(n)
        A[rows, np.arange(n)] = f[i]
    b_ub = np.full(2*n - 1, M, dtype=np.float64)
    c = -np.ones(n)  # since linprog does min
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        b = res.x
    else:
        # fallback: use f itself
        b = f.copy()
    return b

"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm (≈ 1000 s budget)
--------------------------------
1. Analytic families (product‑law, sum‑law, exponential, Gaussian) – a
   fast deterministic sweep to obtain a good initial shape.

2. Smooth‑max continuation – minimise a log‑sum‑exp approximation of
   the max‑convolution using L‑BFGS‑B on the soft‑max parametrisation
   (x → softmax(x)).  The smoothing parameter λ is increased gradually.

3. Linear‑programming refinement – given the current distribution f,
   solve a LP that finds g maximising its total mass under the
   convolution bound  conv(f,g) ≤ M (M = max(conv(f,f))).  Mix f and g
   (convex combination) to strictly lower the bound.

4. Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution
   with a line‑search over a few step sizes; this uses the sub‑gradient
   of the true max objective and respects the simplex via a softmax‑like
   update.

5. Light stochastic mass‑transfer local search (biased by the current
   sub‑gradient) for final polishing.

The routine respects the supplied time budget, prints intermediate
scores and returns the best sequence found (as a plain Python list).
"""

import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (zero‑padded) via FFT. Length = 2·len(a)‑1."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a non‑negative vector a."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the initial sweep)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^(‑p)  (edge‑heavy)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^(‑p) + (n‑1‑i+off)^(‑p)  (two tails)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exp_decay(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(‑c·|i‑center|)  (center‑peaked)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - center))
    a = a / a.sum()
    return a


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(‑0.5·((i‑center)/σ)²)  (smooth bump)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max utilities (log‑sum‑exp approximation)
# ----------------------------------------------------------------------


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    For a log‑probability vector x, λ>0, return smooth max Φ_λ(a)
    and its gradient w.r.t. x.  a = softmax(x) is guaranteed ∑a=1.
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # convolution a*a
    conv = conv_fft_self(a)                # length 2n‑1
    max_c = np.max(conv)

    # soft‑max over conv entries
    shift = conv - max_c                   # ≤ 0
    exp_term = np.exp(lam * shift)        # ∈ (0,1]
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam    # Φ_λ

    w = exp_term / sum_exp                 # weights, sum to 1

    # gradient of Φ_λ w.r.t. a : 2 * (w reversed * a) convolution
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]      # length n
    grad_a = 2.0 * slice_part[::-1]            # ∂Φ/∂a

    # map gradient to x‑space via Jacobian of soft‑max
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)                # ∇_x Φ
    return phi, grad_x


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smoothed objective
# ----------------------------------------------------------------------


def _lbfgs_optimize(x0: np.ndarray, lam: float, n: int,
                    deadline: float, maxiter: int = 60) -> np.ndarray:
    """Run a limited L‑BFGS‑B minimisation of Φ_λ (softmax‑parametrised)."""
    try:
        from scipy.optimize import minimize
    except Exception:          # scipy not available – fall back to identity
        return x0

    class TimeOut(RuntimeError):
        pass

    def fun(x):
        if time.time() >= deadline:
            raise TimeOut()
        phi, _ = compute_phi_and_grad(x, lam, n)
        return phi

    def jac(x):
        if time.time() >= deadline:
            raise TimeOut()
        _, grad = compute_phi_and_grad(x, lam, n)
        return grad

    try:
        res = minimize(fun, x0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'disp': False})
        return res.x
    except TimeOut:
        return x0
    except Exception:
        # any other numeric issue – keep current
        return x0


# ----------------------------------------------------------------------
# Linear‑programming refinement (max‑convolution bound)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Build dense matrix A (2n‑1 × n) with rows = conv(f, b)."""
    n = f.shape[0]
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    # row k, column j gets f[k-j] if 0 ≤ k‑j < n
    # equivalent to: for each i (source index) fill f[i] on diagonal i
    for i in range(n):
        rows = i + np.arange(n)
        A[rows, np.arange(n)] = f[i]
    return A


def _lp_improve(f: np.ndarray, M: float, deadline: float):
    """
    One LP iteration: maximise sum(b) s.t. conv(f, b) ≤ M, b ≥ 0.
    Returns the new mixed vector (1‑t)·f + t·b_norm.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        return f  # cannot improve without LP solver

    n = f.shape[0]
    A = _build_conv_matrix(f)            # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)    # maximise sum(b) → minimise -sum(b)

    # tiny safety: abort if we run out of time
    if time.time() >= deadline:
        return f

    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})

    if not res.success or res.x is None:
        return f

    b = res.x
    s = b.sum()
    if s <= 1e-12:
        return f
    b_norm = b / s
    # mixing weight – a modest 0.2 works well in practice
    t = 0.2
    new_f = (1.0 - t) * f + t * b_norm
    return new_f


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient utilities
# ----------------------------------------------------------------------


def max_conv_and_subgrad(a: np.ndarray):
    """
    Returns (max_val, grad) where max_val = max(conv(a,a))
    and grad = ∂/∂a max(conv) (any sub‑gradient) .
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    # binary mask of active (maximal) lags
    active = (conv >= max_val - 1e-12).astype(np.float64)
    cnt = active.sum()
    if cnt == 0:
        w = np.zeros_like(active)
    else:
        w = active / cnt
    # gradient = 2 * (w_rev * a) convolution
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    grad = 2.0 * conv_wa[n - 1: 2 * n - 1][::-1]
    return max_val, grad


def exp_gradient_update(a: np.ndarray, grad: np.ndarray, step: float) -> np.ndarray:
    """Mirror‑descent (exponentiated‑gradient) update on the simplex."""
    # avoid overflow – clip the exponent
    upd = np.exp(-step * grad)
    a_new = a * upd
    s = a_new.sum()
    if s <= 0.0:
        # numerical safeguard – fall back to uniform
        a_new = np.ones_like(a) / a.shape[0]
    else:
        a_new = a_new / s
    return a_new


def gradient_biased_move(a: np.ndarray, grad: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from a high‑gradient index to a low‑gradient one."""
    n = a.shape[0]
    # indices with large positive gradient (mass tends to increase the max)
    pos = grad > np.percentile(grad, 70)  # top 30 %
    neg = grad < np.percentile(grad, 30)  # bottom 30 %
    if pos.sum() == 0 or neg.sum() == 0:
        # fall back to pure random move
        i, j = np.random.choice(n, 2, replace=False)
    else:
        i = np.random.choice(np.where(pos)[0])
        j = np.random.choice(np.where(neg)[0])
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # ensure non‑negativity and simplex
    a_new = project_simplex(a_new)
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the supplied evaluation function.  The routine respects the given time
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 0) Optional starting point from a previous run
    # ------------------------------------------------------------------
    best_a = np.ones(n) / n
    best_score = compute_score(best_a)

    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n and ref.sum() > 0.0:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            if sc < best_score:
                best_score, best_a = sc, ref.copy()

    # ------------------------------------------------------------------
    # 1) Analytic families – deterministic baseline
    # ------------------------------------------------------------------
    p_vals = np.linspace(0.30, 0.70, 81)
    offset_vals = np.linspace(0.0001, 2.0, 81)
    c_vals = np.linspace(0.0005, 0.0200, 41)
    sigma_vals = np.linspace(100.0, 800.0, 41)

    for mode in ("product", "sum"):
        for p in p_vals:
            for off in offset_vals:
                if time.time() >= deadline:
                    break
                cand = analytic_product(n, p, off) if mode == "product" else analytic_sum(n, p, off)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score, best_a = sc, cand.copy()

    for c in c_vals:
        if time.time() >= deadline:
            break
        cand = analytic_exp_decay(n, c)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        cand = analytic_gaussian(n, sigma)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    print(f"[grid] best score after analytic sweep = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Smoothed‑max continuation with L‑BFGS‑B
    # ------------------------------------------------------------------
    eps = 1e-12
    x = np.log(np.maximum(best_a, eps))

    lam_schedule = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0,
                    2000.0, 8000.0, 20000.0, 80000.0]

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        # adapt iteration budget to remaining time
        remaining = deadline - time.time()
        max_iters = int(min(80, max(5, remaining / 0.01)))   # generous but safe
        x = _lbfgs_optimize(x, lam, n, deadline, maxiter=max_iters)
        a = softmax(x)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[lbfgs λ={lam:.1f}] new best = {best_score:.6f}  (t={time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # 3) Linear‑programming refinement (few iterations)
    # ------------------------------------------------------------------
    f = best_a.copy()
    lp_iters = 0
    while time.time() < deadline and lp_iters < 12:
        conv_f = conv_fft_self(f)
        M = conv_f.max()
        f_new = _lp_improve(f, M, deadline)
        sc = compute_score(f_new)
        if sc < best_score:
            best_score, best_a = sc, f_new.copy()
            print(f"[LP iter={lp_iters}] new best = {best_score:.6f}  (t={time.time() - start_time:.1f}s)")
        # always move forward even if not improved – LP guarantees non‑increase of M
        f = f_new
        lp_iters += 1

    # ------------------------------------------------------------------
    # 4) Mirror‑descent on the true max‑convolution (exponentiated‑gradient)
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        max_val, grad = max_conv_and_subgrad(a)

        improved = False
        # try a handful of step sizes (larger steps early, finer later)
        for step in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
            cand = exp_gradient_update(a, grad, step)
            sc = compute_score(cand)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                a = cand
                improved = True
                print(f"[expgrad it={it} step={step:.3f}] new best = {best_score:.6f}  (t={time.time() - start_time:.1f}s)")
                break
        if not improved:
            # if no improvement, perform a cheap biased random move
            cand = gradient_biased_move(a, grad, scale=0.02)
            sc = compute_score(cand)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                a = cand
                print(f"[biased move it={it}] new best = {best_score:.6f}  (t={time.time() - start_time:.1f}s)")
            else:
                a = cand   # keep exploring

    # ------------------------------------------------------------------
    # 5) Light random‑mass‑transfer local search (pure exploration)
    # ------------------------------------------------------------------
    a = best_a.copy()
    scale = 0.02
    while time.time() < deadline:
        # simple random transfer
        n_idx = a.shape[0]
        i, j = np.random.randint(0, n_idx, size=2)
        while i == j:
            j = np.random.randint(0, n_idx)
        delta = scale * a[i] * np.random.rand()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_simplex(cand)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            a = cand
            scale = max(0.001, scale * 0.9)
            print(f"[random it] new best = {best_score:.6f}  (t={time.time() - start_time:.1f}s)")
        else:
            # slowly relax the move size
            scale = min(0.05, scale * 1.001)
            a = cand

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    # Return a plain Python list (the harness expects a list, not a NumPy array)
    return best_a.tolist()
