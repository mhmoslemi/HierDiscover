def subgradient_optimize(a, max_time, initial_lr=0.1):
    start = time.time()
    best = a.copy()
    best_score = compute_score(best)
    lr = initial_lr
    step = 0
    while time.time() < max_time:
        conv = conv_fft(a)
        max_val = conv.max()
        # Identify active indices
        idxs = np.where(conv >= max_val - 1e-12)[0]  # near max
        # Compute average gradient
        grad = np.zeros_like(a)
        for k in idxs:
            low = max(0, k-(n-1))
            high = min(n-1, k)
            js = np.arange(low, high+1)
            grad[js] += 2*a[k - js]
        grad /= len(idxs)
        # Update a
        a = a - lr * grad
        a = np.clip(a, 0, None)
        a = project_simplex(a)
        # Decay learning rate
        step += 1
        lr = initial_lr / np.sqrt(step+1)
        if step % 100 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best = a.copy()
        # Optionally random jitter if no improvement after some time
    return best

"""
Propose a step‑function (list of non‑negative floats) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Broad analytic exploration** – enumerate several parametrised families
   (product, exponential, power‑sum, Beta, cosine and simple geometric
   decay) on a coarse grid and keep the best candidate.

2. **Deterministic sub‑gradient descent** – start from the best analytic
   vector and iteratively minimise the exact ℓ∞‑norm of the convolution.
   The sub‑gradient of max(conv) is obtained analytically from the active
   convolution indices, and each step is projected onto the probability
   simplex.

3. **Stochastic polishing** – tiny Gaussian perturbations are added and
   accepted whenever they improve the exact score.

The whole routine respects the 1000 s budget, uses only NumPy (no external
solvers) and works for any length `n` (default 2000). The returned list is
already normalised (`sum == 1`), so `evaluate_sequence` will compute the
desired objective directly.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding)."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT. Returns length 2·len(a)-1."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Simple probability‑simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """U‑shaped family a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential tails a_i ∝ exp(-lam·min(i,n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a = a / a.sum()
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^{-p} + (n‑1‑i+off)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped family a_i ∝ (i+1)^{α}·(n‑i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = a / a.sum()
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """Cosine‑power family a_i ∝ |cos(π i/(n‑1))|^{p}."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    a = a / a.sum()
    return a


def analytic_geometric(n: int, r: float) -> np.ndarray:
    """Purely decreasing geometric a_i ∝ r^{i}, 0<r<1 (left‑hand side)."""
    i = np.arange(n, dtype=np.float64)
    a = (1 - r) * (r ** i)          # already normalised
    return a


# ----------------------------------------------------------------------
# Exact objective (sum already normalised to 1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)


# ----------------------------------------------------------------------
# Sub‑gradient optimiser for the ℓ∞‑norm of the convolution
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray, max_seconds: float) -> np.ndarray:
    """
    Deterministic sub‑gradient descent on  max(conv(a,a)).
    Returns the best vector seen within the time budget.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)

    start = time.time()
    step = 0
    # start with a relatively aggressive learning rate
    lr = 0.5

    while time.time() - start < max_seconds:
        conv = conv_fft(a)
        max_val = conv.max()
        # active set (indices attaining the max, tolerance for numerics)
        act = np.where(conv >= max_val - 1e-14)[0]

        # accumulate gradient from all active indices
        grad = np.zeros_like(a)
        for k in act:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(act))

        # simple Adam‑like decay (no moments, just 1/√t)
        step += 1
        a = a - lr * grad
        a = np.clip(a, 0.0, None)
        a = project_simplex(a)

        # learning‑rate schedule
        lr = 0.5 / np.sqrt(step + 1.0)

        # periodic evaluation
        if step % 200 == 0:
            sc = compute_score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                print(f"[subgrad {step:06d}] new best = {best_score:.6f}")

        # simple stagnation restart
        if step % 2000 == 0:
            # jitter a little around the current best
            jitter = np.random.rand(n) * 5e-4
            a = best + jitter
            a = project_simplex(a)

    return best


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that approximately minimises
    2·n·max(conv(a,a)) / (sum(a)**2).  The method combines a broad analytic
    grid search, a sub‑gradient optimiser and a final stochastic polishing.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the harness

    # ------------------------------------------------------------------
    # vector length
    # ------------------------------------------------------------------
    if "n" in kwargs:
        n = int(kwargs["n"])
    elif "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        n = 2000

    # ------------------------------------------------------------------
    # start from a known global candidate if it exists
    # ------------------------------------------------------------------
    best_seq = np.ones(n) / n
    best_score = compute_score(best_seq)

    if "height_sequence_1" in globals() and len(globals()["height_sequence_1"]) == n:
        cand = np.maximum(np.asarray(globals()["height_sequence_1"], dtype=float), 0.0)
        s = cand.sum()
        if s > 1e-12:
            cand = cand / s
        else:
            cand = np.ones(n) / n
        sc = compute_score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"[init] global seed score = {best_score:.6f}")

    print(f"[init] uniform baseline = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – systematic exploration of analytic families
    # ------------------------------------------------------------------
    def try_update(name: str, vec: np.ndarray):
        nonlocal best_seq, best_score
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            return
        vec = vec / vec.sum()
        sc = compute_score(vec)
        if sc < best_score:
            best_seq, best_score = vec, sc
            print(f"[grid] new best = {best_score:.6f} from {name}")

    # product family (U‑shaped)
    for p in np.linspace(0.3, 3.0, 15):
        for off in np.linspace(0.0, 5.0, 15):
            try_update(f"product(p={p:.2f},off={off:.2f})",
                       analytic_product(n, p, off))

    # exponential tails (symmetric)
    for lam in np.geomspace(0.001, 4.0, 15):
        try_update(f"exp(lam={lam:.3f})", analytic_exponential(n, lam))

    # sum of two power laws (both ends)
    for p in np.linspace(0.3, 3.0, 12):
        for off in np.linspace(0.0, 5.0, 12):
            try_update(f"sum(p={p:.2f},off={off:.2f})",
                       analytic_sum(n, p, off))

    # beta‑shaped (asymmetric powers)
    for a_val in np.linspace(-2.0, 2.0, 11):
        for b_val in np.linspace(-2.0, 2.0, 11):
            try_update(f"beta(a={a_val:.2f},b={b_val:.2f})",
                       analytic_beta(n, a_val, b_val))

    # cosine power family
    for p in np.linspace(0.0, 10.0, 13):
        try_update(f"cosine(p={p:.2f})", analytic_cosine(n, p))

    # pure geometric decay (left‑hand side heavy)
    for r in np.linspace(0.5, 0.99, 12):
        try_update(f"geom(r={r:.3f})", analytic_geometric(n, r))

    # random Dirichlet‑like draws (small fraction of the budget)
    for _ in range(200):
        vec = np.random.rand(n)
        try_update("rand_dirichlet", vec)

    # ------------------------------------------------------------------
    # Phase 2 – deterministic sub‑gradient optimisation
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                                   # leave a few seconds for polishing
        opt_time = remaining * 0.85
        print(f"[opt] starting sub‑gradient optimisation for {opt_time:.1f}s")
        best_seq = subgradient_optimize(best_seq, opt_time)
        best_score = compute_score(best_seq)
        print(f"[opt] finished, best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final stochastic refinement
    # ------------------------------------------------------------------
    while time.time() < deadline:
        perturb = np.random.normal(scale=5e-4, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        sc = compute_score(cand)
        if sc < best_score:

            print(f"[refine] improved to {best_score:.6f}")

    total = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, length = {n}, time = {total:.1f}s")
    return best_seq.tolist()
