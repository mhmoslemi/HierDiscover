"""
Propose a non‑negative step function that (approximately) minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. Quick analytic grid‑search over several families (product, sum,
   exponential, cosine‑power, beta) and keep the ten best candidates.
2. Pairwise convex mixing of those top candidates.
3. Greedy “mass‑transfer” local search:
   - Work with the self‑convolution obtained by FFT.
   - In each iteration move a small amount of probability mass from an
     index that most influences the current maximal convolution entry
     (sub‑gradient) to an index that influences it the least.
   - Update the convolution exactly by slice operations (O(n) per step)
     and accept the move only if the maximum improves.
   - A simple linear step‑size schedule, random exploratory moves,
     periodic re‑computation of the convolution (to avoid drift) and
     jitter restarts are used to escape plateaus.
4. Final short random‑perturbation phase.

The routine respects the wall‑time budget (default 1000 s) and always
returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog  # not used but kept for possible extensions

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_self(a: np.ndarray) -> np.ndarray:
    """Exact linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """Exact objective (scale‑invariant) for a probability vector."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_self(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families (return normalised probability vectors)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    return a / a.sum()


def analytic_cosine(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    return a / a.sum()


def analytic_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta_
    return a / a.sum()

# Analytic grid + mixing (small search to obtain a decent start)

def _add_candidate(
    cand: np.ndarray,
    best_list: list[tuple[float, np.ndarray]],
    max_len: int = 10,
) -> None:
    sc = compute_score(cand)
    inserted = False
    for idx, (s, _) in enumerate(best_list):
        if sc < s:
            best_list.insert(idx, (sc, cand.copy()))
            inserted = True
            break
    if not inserted and len(best_list) < max_len:
        best_list.append((sc, cand.copy()))
    while len(best_list) > max_len:
        best_list.pop()


def analytic_grid_search(n: int) -> tuple[np.ndarray, list[tuple[float, np.ndarray]]]:
    top: list[tuple[float, np.ndarray]] = []

    # product family
    p_vals = np.arange(0.1, 3.01, 0.15)
    off_vals = np.concatenate(
        (np.geomspace(1e-3, 0.01, 5), np.arange(0.01, 5.01, 0.3))
    )
    for p in p_vals:
        for off in off_vals:
            cand = analytic_product(n, p, off)
            _add_candidate(cand, top)

    # sum family
    for p in p_vals:
        for off in off_vals:
            cand = analytic_sum(n, p, off)
            _add_candidate(cand, top)

    # exponential tails
    lam_vals = np.logspace(-3, 1, num=20)
    for lam in lam_vals:
        cand = analytic_exponential(n, lam)
        _add_candidate(cand, top)

    # cosine‑power
    p_cos = np.arange(0.5, 8.1, 0.5)
    for p in p_cos:
        cand = analytic_cosine(n, p)
        _add_candidate(cand, top)

    # beta family
    alpha_vals = np.arange(-0.5, 2.51, 0.5)
    beta_vals = np.arange(-0.5, 2.51, 0.5)
    for a in alpha_vals:
        for b in beta_vals:
            cand = analytic_beta(n, a, b)
            _add_candidate(cand, top)

    # include an optional global reference if present

        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:

            ref = ref / ref.sum()
            _add_candidate(ref, top)

    best_seq = min(top, key=lambda x: x[0])[1].copy()
    return best_seq, top


def mix_top_candidates(
    top: list[tuple[float, np.ndarray]],
    n: int,
) -> np.ndarray:
    best_seq = top[0][1].copy()
    best_score = top[0][0]
    alphas = np.linspace(0.0, 1.0, 21)
    for i in range(len(top)):
        for j in range(i + 1, len(top)):
            a_i = top[i][1]
            a_j = top[j][1]
            for alpha in alphas:
                cand = alpha * a_i + (1.0 - alpha) * a_j
                cand = project_simplex(cand)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
    return best_seq


# ----------------------------------------------------------------------
# Greedy mass‑transfer local optimisation
# ----------------------------------------------------------------------


def greedy_mass_transfer(
    f: np.ndarray,
    deadline: float,
    start_time: float,
    stagnation_time: float = 20.0,
) -> tuple[np.ndarray, float]:
    """
    Perform a deterministic greedy local search based on moving small
    amounts of probability mass from indices that increase the current
    maximum of the self‑convolution to indices that decrease it.
    Returns the best vector found and its score.
    """
    n = f.shape[0]
    conv = conv_self(f)                     # current self‑convolution
    best_f = f.copy()
    best_score = 2.0 * n * conv.max()
    best_M = conv.max()
    last_improve = time.time()
    iter_cnt = 0

    # step‑size schedule (large at start → small later)
    max_step = 0.20
    min_step = 0.001

    while time.time() < deadline:
        # sub‑gradient at the current maximal entry
        max_idx = int(conv.argmax())
        shift = max_idx - np.arange(n)
        mask = (shift >= 0) & (shift < n)
        g = np.zeros(n)
        g[mask] = 2.0 * f[shift[mask]]

        # choose source (high influence) with non‑zero mass
        src_order = np.argsort(-g)  # descending
        src = None
        for idx in src_order:
            if f[idx] > 1e-16:
                src = idx
                break
        if src is None:
            src = np.random.randint(n)

        # choose destination (low influence) – avoid picking src
        dst_order = np.argsort(g)   # ascending
        dst = None
        for idx in dst_order:
            if idx != src:
                dst = idx
                break
        if dst is None:
            dst = (src + 1) % n

        # linear schedule for step size
        elapsed = time.time() - start_time
        total = deadline - start_time
        step_frac = elapsed / total
        step_factor = max_step * (1.0 - step_frac) + min_step * step_frac

        # propose a transfer amount
        eps = step_factor * f[src] * np.random.rand()
        if eps <= 0.0:
            eps = step_factor * np.min([f[src], 1.0]) * np.random.rand()

        # exact convolution update (linear + quadratic term)
        conv_new = conv.copy()
        delta = 2.0 * eps * f
        conv_new[src : src + n] -= delta
        conv_new[dst : dst + n] += delta
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_M = conv_new.max()
        new_score = 2.0 * n * new_M

        if new_score < best_score - 1e-12:
            # accept move
            f[src] -= eps
            f[dst] += eps
            conv = conv_new
            best_f = f.copy()
            best_score = new_score
            best_M = new_M
            last_improve = time.time()
            print(f"[greedy {iter_cnt:07d}] new best = {best_score:.6f}")

        # occasional random exploratory transfer (independent of gradient)
        if np.random.rand() < 0.02:
            i, j = np.random.randint(n, size=2)
            while i == j:
                j = np.random.randint(n)
            eps_r = step_factor * f[i] * np.random.rand()
            if eps_r > 0.0:
                conv_tmp = conv.copy()
                delta_r = 2.0 * eps_r * f
                conv_tmp[i : i + n] -= delta_r
                conv_tmp[j : j + n] += delta_r
                conv_tmp[2 * i] += eps_r * eps_r
                conv_tmp[2 * j] += eps_r * eps_r
                conv_tmp[i + j] -= 2.0 * eps_r * eps_r
                new_M_r = conv_tmp.max()
                new_score_r = 2.0 * n * new_M_r
                if new_score_r < best_score - 1e-12:
                    f[i] -= eps_r
                    f[j] += eps_r
                    conv = conv_tmp
                    best_f = f.copy()
                    best_score = new_score_r
                    best_M = new_M_r
                    last_improve = time.time()
                    print(f"[greedy {iter_cnt:07d}] random move improved to {best_score:.6f}")

        # periodic exact recomputation to fight floating‑point drift
        if iter_cnt % 5000 == 0 and iter_cnt > 0:
            conv = conv_self(f)

        # stagnation handling – jitter restart
        if time.time() - last_improve > stagnation_time:
            sigma = 0.001
            jitter = np.random.randn(n) * sigma
            f = best_f + jitter
            f = project_simplex(f)
            conv = conv_self(f)
            last_improve = time.time()
            print(f"[greedy {iter_cnt:07d}] jitter restart")

        # early exit if we have already reached the target bound
        if best_score <= 1.503:
            break

        iter_cnt += 1

    return best_f, best_score

# Main entry point

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the evaluation function.  The routine respects a wall‑time budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------- 1) Analytic grid + mixing (fast, provides a good start) -------
    n = int(kwargs.get("n", 2000))
    print("[0] Analytic grid search, n =", n)
    best_seq, top10 = analytic_grid_search(n)
    best_seq = mix_top_candidates(top10, n)
    best_score = compute_score(best_seq)
    print(f"[0] after mixing: best_score = {best_score:.6f}")

    # ------- 2) Greedy mass‑transfer local optimisation --------------------
    # Allocate the bulk of the budget to this phase.
    # Leave a few seconds for the final random refinement.
    greedy_deadline = deadline - 5.0
    if time.time() < greedy_deadline:
        print("[1] Starting greedy mass‑transfer optimisation")
        best_seq, best_score = greedy_mass_transfer(
            best_seq.copy(),
            deadline=greedy_deadline,
            start_time=start_time,
        )
        print(f"[1] greedy finished, best_score = {best_score:.6f}")

    # ------- 3) Very short final random perturbation ----------------------
    # (few seconds left)
    remaining = deadline - time.time()
    if remaining > 0.5:
        print("[2] Final random perturbation phase")
        n = best_seq.shape[0]
        while time.time() < deadline:
            perturb = np.random.normal(scale=5e-4, size=n)
            cand = best_seq + perturb
            cand = np.clip(cand, 0.0, None)
            cand = project_simplex(cand)
            cand_score = compute_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[2] improved to {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] best_score = {best_score:.6f}, length = {n}, elapsed = {total_time:.1f}s")
    return best_seq.tolist()
