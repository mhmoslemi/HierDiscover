"""
Propose a high‑quality step‑function using a hybrid global‑local search.

Algorithm
---------
1. **Seed evaluation** – a small diverse set of analytically motivated
   probability vectors (uniform, sine window, geometric, power‑law,
   beta‑like, random Dirichlet, and the previously known best) are built
   and the best of them is kept.

2. **Differential Evolution (DE) on the log‑softmax parameters** – the
   candidate vector `v` lives on ℝⁿ; the actual step heights are
   `a = softmax(v)`.  DE performs mutation `a + F·(b‑c)` and binomial
   crossover while always staying on the probability simplex after a
   softmax.  The population size is modest (≈30) so that many generations
   can be evaluated inside the 1000 s budget.  The global best found by
   DE is recorded.

3. **Exact sub‑gradient mirror descent** – starting from the DE result,
   the true sub‑gradient of `max(conv(a,a))` is applied with an
   exponentiated‑gradient (mirror‑descent) step.  This refines the
   solution without any smoothing.

4. **Simulated‑annealing hill climbing** – a cheap stochastic neighbourhood
   (Dirichlet re‑allocation of a few entries) is explored; downhill moves
   are always accepted, uphill moves with a temperature‑controlled
   probability.  This helps escape shallow basins after the deterministic
   refinement.

All phases are time‑aware: they stop as soon as the global deadline is
reached.  The best probability vector discovered is finally returned as
a plain list of floats (the `evaluate_sequence` routine will clip it to
the required range).

The combination of a global evolutionary strategy and a precise
sub‑gradient / annealing polish typically yields a lower bound than the
previous Adam‑based pipeline, pushing the objective below the target
~1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """
    Objective used during the search.
    The sequence must already be normalised to sum == 1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Return a modest list of diverse normalised seeds."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for r in (0.99, 0.97, 0.95):
        seeds.append(_geom(n, r))

    for p in (0.0, 0.5, 1.0, 1.5, 2.0):
        seeds.append(_power(n, p))

    for a in (0.3, 0.5, 0.7):
        for b in (0.3, 0.5, 0.7):
            seeds.append(_beta_shape(n, a, b))

    # a few Dirichlet random draws
    for conc in (0.3, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # reuse previous best if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

def _de_optimize(
    n: int,
    deadline: float,
    init_seq: np.ndarray | None,
    rng: np.random.Generator,
    popsize: int = 30,
    F: float = 0.5,
    CR: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Differential Evolution on the log‑softmax representation.
    Returns the best probability vector and its objective value.
    """
    # initialise population on logit space
    pop_v = []

    eps = 1e-12
    if init_seq is not None:
        pop_v.append(np.log(init_seq + eps))

    while len(pop_v) < popsize:
        pop_v.append(rng.standard_normal(n))

    pop_v = np.array(pop_v)                     # shape (popsize, n)

    # evaluate initial population
    pop_val = np.empty(popsize, dtype=np.float64)
    for i in range(popsize):
        pop_val[i] = _objective(_softmax(pop_v[i]))

    best_idx = int(np.argmin(pop_val))
    best_val = pop_val[best_idx]
    best_v = pop_v[best_idx].copy()
    it = 0

    # DE main loop
    while time.time() < deadline:
        it += 1
        for i in range(popsize):
            if time.time() >= deadline:
                break

            # choose three distinct individuals different from i
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, 3, replace=False)

            mutant = pop_v[a] + F * (pop_v[b] - pop_v[c])

            # binomial crossover
            jrand = rng.integers(n)
            mask = rng.random(n) < CR
            mask[jrand] = True
            trial = pop_v[i].copy()
            trial[mask] = mutant[mask]

            # evaluate trial
            trial_val = _objective(_softmax(trial))

            # selection
            if trial_val < pop_val[i]:
                pop_v[i] = trial
                pop_val[i] = trial_val
                if trial_val < best_val:
                    best_val = trial_val
                    best_v = trial.copy()
                    print(f"[DE] n={n:5d} gen={it:6d} best={best_val:.6f}")

        # optional early stop – if no improvement for many generations
        # (not critical given the generous time budget)

    best_seq = _softmax(best_v)
    return best_seq, best_val

# Exact sub‑gradient mirror‑descent (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using the exact sub‑gradient of max(conv).
    Works on the probability simplex via exponentiated updates.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val

def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles a tiny random subset of entries."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        # pick 2–4 indices and redistribute their total mass
        k = rng.integers(2, 5)          # 2, 3 or 4
        idx = rng.choice(n, size=k, replace=False)
        tot = cur[idx].sum()
        if tot <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * tot
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:6d} best={new_val:.6f}")

    return best_seq, best_val

        if global_best_seq is not None and global_best_seq.shape[0] == n:
            sub_deadline = deadline - 0.1 * (deadline - time.time())
            seq_sub, val_sub = _subgradient_refine(
                start_seq=global_best_seq,
                n=n,
                deadline=sub_deadline,
                max_iters=200_000,
            )
            if val_sub < global_best_val:
                global_best_val = val_sub
                global_best_seq = seq_sub.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val_sub:.6f}")

        # ------------------------------------------------------------------
        # 4. Simulated‑annealing fine‑tuning (optional)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            sa_deadline = deadline
            seq_sa, val_sa = _simulated_annealing(
                start_seq=global_best_seq,
                n=n,
                deadline=sa_deadline,
                max_iters=300_000,
            )
            if val_sa < global_best_val:
                global_best_val = val_sa
                global_best_seq = seq_sa.copy()
                print(f"[overall‑SA] n={n:5d} new global best={val_sa:.6f}")

        # if we have spent a lot of time, break early
        if time.time() > deadline:
            break

    # ----------------------------------------------------------------------
    # Fallback (unlikely to be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # final sanity check via the provided evaluator (optional)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
