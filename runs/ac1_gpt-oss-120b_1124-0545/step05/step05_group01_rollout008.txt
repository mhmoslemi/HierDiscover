    i_start = max(0, k_max - (n - 1))

    i_down_candidates = np.where(g == np.min(g))[0]

"""
Search for a non‑negative step‑function that minimises the
self‑convolution ∞‑norm (the objective used by ``evaluate_sequence``).

Algorithm
---------

* Build a diverse pool of seeds (uniform, geometric, power‑law, Dirichlet,
  previous best, and their mirrors).
* For each seed run a *mass‑transfer* mirror‑descent:
  - Compute the convolution a∗a.
  - Identify the lag k* where the convolution attains its maximum.
  - The sub‑gradient of this maximal entry is g_i = 2·a_{k*‑i}.
  - Transfer a fraction of mass from the index with the largest g_i
    to the index with the smallest g_i (step size η_t = η0/√t).
  - Occasionally add a tiny Gaussian jitter to escape plateaus.
* Keep the best sequence seen.
* If time permits, polish the best candidate with the classic exponentiated‑gradient
  sub‑gradient refinement (as in the reference implementation) and a light
  simulated‑annealing walk.
* Return the best sequence (plain list of floats).  All numbers are kept
  non‑negative and normalised to sum = 1, which is safe because the
  objective is scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to harmless clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq))², seq assumed non‑negative."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Seed factories
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse or dense Dirichlet draw (Gamma(α,1) normalised)."""
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _make_seeds(n: int, previous: np.ndarray | None, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Assemble a diverse set of start‑vectors (already normalised to sum = 1).
    """
    seeds: list[np.ndarray] = []

    # basic families
    seeds.append(_uniform(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # Dirichlet draws – from very sparse to fairly dense
    for alpha in [0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # some extra random Dirichlet draws with random α
    for _ in range(5):
        alpha = rng.uniform(0.05, 4.0)
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # mirror each seed (helps when the optimum is asymmetric)
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # reuse previously best construction if it exists and matches length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # guarantee non‑empty list
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Mirror‑descent based on the true max‑convolution (mass‑transfer)
# ----------------------------------------------------------------------


def _mass_transfer_refine(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 120000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Simple projected mirror‑descent:
    move a fraction of mass from the index that contributes most to the
    maximal convolution entry to the index that contributes least.
    """
    rng = np.random.default_rng()
    a = start_seq.astype(np.float64, copy=True)
    # normalise to sum = 1 (scale‑invariant)
    total = a.sum()
    if total <= 0.0:
        a = _uniform(n)
    else:
        a /= total

    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear convolution and index of the maximal entry
        conv = _conv_fft(a, a)
        k_max = int(np.argmax(conv))

        # current objective (for tracking)
        cur_val = 2.0 * n * conv[k_max] / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[MassTransfer] n={n} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the maximal convolution entry:
        # g_i = 2 * a_{k_max - i} for overlapping indices, else 0
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_max - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # indices to move mass between
        i_up = int(np.argmax(g))          # contributes most to the max entry
        i_down = int(np.argmin(g))        # contributes least

        if i_up == i_down:
            # pick the second‑smallest gradient if a tie occurs
            sorted_idx = np.argsort(g)
            for cand in sorted_idx:
                if cand != i_up:
                    i_down = cand
                    break

        # step‑size schedule (diminishing)
        eta = eta0 / np.sqrt(it)
        delta = eta * a[i_up]  # move a fraction of the mass at i_up
        # keep a[i_up] non‑negative
        if delta > a[i_up]:
            delta = a[i_up]

        if delta <= 0.0:
            continue

        a[i_up] -= delta
        a[i_down] += delta

        # numerical housekeeping
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s != 0.0:
            a /= s

        # occasional tiny global jitter to avoid getting stuck
        if it % 5000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            a = np.clip(a + noise, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = _uniform(n)
            else:
                a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent) – kept from the reference
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_max = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[k_max] / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] n={n} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_max - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update on the simplex
        lr = 0.75 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (light version)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Cheap Metropolis‑Hastings walk that transfers a tiny amount of mass
    between two positions and occasionally adds Gaussian jitter."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)   # small fraction of current objective
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move a tiny fraction from i to j
        delta = rng.random() * 0.025 * cur[i]
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                if it % 1000 == 0:
                    print(f"[Anneal] n={n} it={it:6d} best={new_val:.6f}")

        # periodic global Gaussian jitter
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found
    as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # 1 s safety margin

    # Re‑use a previously discovered construction if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – short runs first, enlarge if time permits
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]   # larger lengths only if time remains
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        rng = np.random.default_rng(seed + n)  # reproducible per length
        seeds = _make_seeds(n, previous, rng)

        # 1) Evaluate raw seeds, keep a few promising ones
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(8, len(seed_vals))]]

        # Record the best raw seed globally
        if seed_vals and seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Mass‑transfer mirror‑descent (fast, aggressive)
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _mass_transfer_refine(seed_vec, n, deadline,
                                             max_iters=120000, eta0=0.8)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall MassTransfer] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement (optional, if time left)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=40000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Light simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                        max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(2000)
        print("[fallback] using uniform distribution")

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
