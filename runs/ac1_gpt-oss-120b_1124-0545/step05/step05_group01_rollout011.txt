                      beta0=2.0, beta_mul=1.06, lr0=1.0,
                      lambda0=1.0, lambda_decay=0.999,
                      lambda2=1.0):
    # similar to earlier but includes extra uniform penalty

    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val=cur_val
            best_seq=a.copy()
            print(f"[AdamV2] n={n} it={it} best={cur_val:.6f}")
        # compute softened max
        beta*=beta_mul
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        w /= w_sum
        # gradient of smoothed max
        rev_a = a[::-1]
        full_max = _conv_fft(rev_a, w)
        grad_max = 2.0 * full_max[n-1:n-1+n]
        # gradient of sum(conv^2)
        full_sq = _conv_fft(rev_a, conv)
        grad_sq = 2.0 * full_sq[n-1:n-1+n]
        # gradient of uniform deviation: use tanh for soft sign
        dev = np.tanh(5.0 * (conv - avg)) # beta2 fixed 5.0
        full_dev = _conv_fft(rev_a, dev)
        grad_dev = 2.0 * full_dev[n-1:n-1+n]
        grad_a = grad_max + lam * grad_sq + lambda2 * grad_dev
        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # Adam update
        m = 0.9 * m + 0.1 * grad_v

"""
Hybrid gradient + uniformity optimisation for the step‑function problem.

The goal is to minimise
    2·n·max(conv(a,a)) / (sum(a))²
with a_i ≥ 0, sum(a)=1.
The algorithm consists of three stages:

1. A diverse family of seed vectors (uniform, geometric, power‑law,
   Dirichlet draws, previous best, etc.).
2. Adam optimisation on a smooth surrogate:
      L(a) = soft‑max(conv)  + λ·∑ conv² + λ₂·∑ tanh(β₂·(conv‑μ))·conv
   where μ = 1/(2n‑1) is the uniform target.  The extra λ₂‑term
   penalises deviation from a flat convolution.
   Parameters (β, λ, λ₂, learning‑rate) are annealed.
3. Exact sub‑gradient refinement (exponentiated‑gradient) on the true
   max‑convolution term, followed by a cheap simulated‑annealing walk.

The routine respects the time budget (default 1000 s) and returns the
best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Objective – identical to evaluate_sequence up to the normalisation
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq))², seq assumed non‑negative."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Seed families (unchanged from the previous implementation)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a sparse/dense Dirichlet vector."""
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse pool of normalised start vectors."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic families
    seeds.append(_uniform(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # Dirichlet draws – cover sparse to dense regimes
    for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # mirrored copies
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # reuse a previously best construction if length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # safety – never empty
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation – smoothed max + variance + uniformity penalty
# ----------------------------------------------------------------------


def _adam_optimize_v2(seed: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 25000,
                      beta0: float = 3.0,
                      beta_mul: float = 1.06,
                      lr0: float = 1.0,
                      lambda0: float = 1.5,
                      lambda_decay: float = 0.998,
                      lambda2: float = 1.0,
                      tanh_beta: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the surrogate loss

        L = (1/β)·log∑exp(β·conv)  + λ·∑conv²
            + λ₂·∑ tanh(β₂·(conv‑μ))·conv

    The extra λ₂‑term penalises deviation from a perfectly flat
    convolution (μ = 1/(2n‑1)).  The vector stays on the simplex via
    a log‑softmax parametrisation.
    """
    # initialise log‑probability vector
    v = np.log(seed + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0
    lam = lambda0
    mu = 1.0 / (2 * n - 1)          # uniform target for the convolution

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current point on the simplex
        a = _softmax(v)

        # evaluate true objective (for bookkeeping)
        conv = _conv_fft(a, a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[AdamV2] n={n} it={it:5d} best={cur_val:.6f}")

        # ---------- smoothed max ----------
        beta *= beta_mul
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of smoothed max
        rev_a = a[::-1]
        full_max = _conv_fft(rev_a, w)
        grad_max = 2.0 * full_max[n - 1:n - 1 + n]

        # ---------- variance regulariser ----------
        full_sq = _conv_fft(rev_a, conv)
        grad_sq = 2.0 * full_sq[n - 1:n - 1 + n]

        # ---------- uniformity (L1‑like) penalty ----------
        # tanh approximates a sign; we use a modest β for smoothness.
        dev = np.tanh(tanh_beta * (conv - mu))
        full_dev = _conv_fft(rev_a, dev)
        grad_dev = 2.0 * full_dev[n - 1:n - 1 + n]

        # combine all gradient contributions
        grad_a = grad_max + lam * grad_sq + lambda2 * grad_dev

        # chain‑rule through the softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on logits
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

        # slowly decay the variance regulariser
        lam *= lambda_decay

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 100000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true max‑convolution term."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirrored simplex update)
        lr = 0.7 / np.sqrt(it)   # slightly larger prefactor than before
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)    # stabilise numerics
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Cheap Metropolis‑Hastings walk with occasional global jitter."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initialise temperature relative to current objective
    T0 = max(1e-6, cur_val * 0.05)
    decay = 0.9993

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct positions
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # propose moving a tiny fraction of mass from i to j
        delta = rng.random() * 0.03 * cur[i]   # up to 3 % of source
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # every 1500 steps perform a tiny global jitter
        if it % 1500 == 0:
            noise = rng.normal(scale=0.0015, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising evaluate_sequence.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # try to reuse a previously discovered construction if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – start medium, then larger if time permits
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]   # larger ones only if time remains
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # 1) generate a rich seed set
        seeds = _make_seeds(n, previous)

        # 2) quick evaluation of raw seeds – keep the most promising ones
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(12, len(seed_vals))]]

        # record the best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation with enhanced uniformity regulariser
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize_v2(
                seed_vec,
                n,
                deadline,
                max_iters=25000,
                beta0=3.0,
                beta_mul=1.06,
                lr0=1.0,
                lambda0=1.5,
                lambda_decay=0.998,
                lambda2=1.0,
                tanh_beta=5.0,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=80000
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(
                global_best_seq, n, deadline, max_iters=None
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

        # keep ability to upscale the best vector to the next length
        previous = global_best_seq

    # ------------------------------------------------------------------
    # Fallback – should never happen but safe‑guard
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
