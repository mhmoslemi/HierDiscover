def local_optimize(seed, n, deadline):
    # Ensure sum=1
    a = seed.astype(np.float64)
    a /= max(a.sum(), 1e-12)
    best_seq = a.copy()
    best_val = compute_obj(a)

    # Mirror descent subgradient improvement early
    seq, val = subgradient_refine(a, n, deadline, max_iters=20000)
    if val < best_val: best_seq, best_val = seq, val

    # L-BFGS smoothed max with beta schedule
    v = np.log(seq + 1e-12)   # parameters
    betas = [5.0, 15.0, 50.0, 150.0, 500.0]
    for beta in betas:
        if time.time() > deadline: break
        # define closure for objective+grad
        def fun(v):
            a = _softmax(v)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_term = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_term)
            smooth_max = shift + np.log(sum_exp) / beta
            obj = 2.0 * n * smooth_max
            # gradient
            w = exp_term / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v
        # SciPy L-BFGS
        res = minimize(fun, v, method='L-BFGS-B', jac=True,
                       options={'maxiter': 200, 'disp': False})
        v = res.x
        # evaluate exact objective
        a = _softmax(v)
        val = compute_obj(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[L-BFGS] n={n} beta={beta} val={val:.6f}")

    # final subgradient refine
    seq, val = subgradient_refine(best_seq, n, deadline, max_iters=20000)
    if val < best_val:
        best_val, best_seq = val, seq

    # final hill climb
    seq, val = _hill_climb(best_seq, n, deadline, max_iters=20000)
    if val < best_val:
        best_seq, best_val = seq, val

    return best_seq, best_val

"""
Hybrid optimisation for the “max‑auto‑convolution” problem.

* Fast analytical seeds – a richer family (beta, power, geometric, sine,
  uniform, Dirichlet, random softmax).  The best few seeds are retained.

* Local polishing – for every retained seed we run three deterministic
  refinements:
    – Exact sub‑gradient mirror‑descent (the true max‑convolution),
    – L‑BFGS on a smooth log‑sum‑exp surrogate of the max (beta is
      increased gradually), and
    – A cheap stochastic hill‑climb that swaps a tiny mass between two
      positions.

* A small DE run is used as a last‑ditch global search if time allows.

All steps respect a global time‑budget (default 1000 s).  The routine
returns the best non‑negative step‑function (as a plain list of floats)
found so far.  The sequence is L₁‑normalised; because the objective is
scale‑invariant this does not affect the final score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a simplex vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Analytic seed families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_extended_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of seeds, using broader parameter ranges."""
    rng = np.random.default_rng(2024)
    seeds: list[np.ndarray] = []

    # very small families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric (r close to 1 as well as small)
    for r in np.linspace(0.99, 0.9999, 5):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # power laws – go up to p=5
    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, p))

    # beta shapes – explore extreme α,β < 1 as well as >1
    alphas = np.concatenate([np.logspace(-2, -0.3, 5), np.linspace(0.3, 2.0, 5)])
    betas = np.concatenate([np.logspace(-2, -0.3, 5), np.linspace(0.3, 2.0, 5)])
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws with a variety of concentration parameters
    for conc in (0.1, 0.5, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random softmax vectors
    for _ in range(10):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # include previous best if compatible
    if previous is not None and isinstance(previous, np.ndarray) and previous.shape == (n,):
        tmp = previous.copy()
        tmp = np.maximum(tmp, 0.0)
        s = tmp.sum()
        if s > 0:
            tmp = tmp / s
        else:
            tmp = _uniform(n)
        seeds.append(tmp)

    return seeds

# Objective utilities (sum‑normalised ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Exact objective for a simplex‑normalised sequence."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 60000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val

# Smooth‑max L‑BFGS optimisation (requires SciPy)
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Smoothed max (log‑sum‑exp) objective and gradient w.r.t. the softmax
    parameters v.  Returns (obj, grad_v).
    """
    a = _softmax(v)                     # stays on the simplex
    conv = _conv_fft(a, a)

    shift = np.max(conv)
    # safe exponentiation
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_term)
    smooth_max = shift + np.log(sum_exp) / beta
    obj = 2.0 * n * smooth_max

    # gradient w.r.t. a of the smoothed max: Σ w_k·∂conv_k/∂a
    w = exp_term / sum_exp               # weights that sum to 1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    return float(obj), grad_v.astype(np.float64)


def _lbfgs_optimize(seed: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule: list[float] = (5.0, 15.0, 50.0, 150.0, 500.0),
                    maxiter_per_stage: int = 250) -> tuple[np.ndarray, float]:
    """
    L‑BFGS optimisation on a sequence using a continuation over the
    smoothing parameter `beta`.  Returns the best sequence and its exact
    objective (max‑convolution).
    """
    # initialise the softmax parameters
    v = np.log(seed + 1e-30)
    best_seq = seed.copy()
    best_val = _objective(seed)

    # SciPy may be unavailable – guard the import
    try:
        from scipy.optimize import minimize
    except Exception as exc:
        print("[L‑BFGS] SciPy not available – skipping this stage")
        return best_seq, best_val

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # wrap the objective/gradient for SciPy (extra args are passed via `args`)
        def fun(vv, n_local, beta_local):
            return _smooth_obj_and_grad(vv, n_local, beta_local)

        # minimise – we ask for the gradient (`jac=True`)
        res = minimize(fun,
                       v,
                       args=(n, beta),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_stage,
                                'disp': False})
        v = res.x

        # evaluate the *exact* max‑convolution of the resulting point
        a = _softmax(v)
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[L‑BFGS] n={n:5d} beta={beta:6.1f} val={val:.6f}")

    return best_seq, best_val

def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """Random pair‑wise mass transfer, keeping improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.01  # initial relative step size

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new)

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:6d} best={val:.6f}")

        if it % 1000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val

def _de_optimize(n: int,
                 deadline: float,
                 popsize: int = 30,
                 K: int | None = None,
                 max_gens: int = 200) -> tuple[np.ndarray, float]:
    """
    Differential Evolution on a piece‑wise‑linear representation.
    Only invoked if time permits after the deterministic stages.
    """
    rng = np.random.default_rng()

    if K is None:
        K = max(20, int(np.sqrt(n)))      # ≈ √n knots
    knots = np.linspace(0, n - 1, K)

    # initialise population on the simplex (Dirichlet draws)
    pop = [rng.dirichlet(np.ones(K)) for _ in range(popsize)]

    pop_vals = np.empty(popsize, dtype=np.float64)
    pop_seq = [None] * popsize
    for i in range(popsize):
        a = np.interp(np.arange(n, dtype=np.float64), knots, pop[i])
        a = a / a.sum()
        pop_vals[i] = _objective(a)
        pop_seq[i] = a

    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_seq = pop_seq[best_idx].copy()

    F = 0.8
    CR = 0.9
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a_vec = pop[a_idx]
            b_vec = pop[b_idx]
            c_vec = pop[c_idx]

            donor = a_vec + F * (b_vec - c_vec)
            donor = np.maximum(donor, 0.0)
            ds = donor.sum()
            donor = donor / ds if ds > 0 else _uniform(K)

            # binomial crossover
            trial = pop[i].copy()
            jrand = rng.integers(K)
            for j in range(K):
                if rng.random() < CR or j == jrand:
                    trial[j] = donor[j]

            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            trial = trial / ts if ts > 0 else _uniform(K)

            a = np.interp(np.arange(n, dtype=np.float64), knots, trial)
            a = a / a.sum()
            val_trial = _objective(a)

            if val_trial < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val_trial
                pop_seq[i] = a
                if val_trial < best_val:
                    best_val = val_trial
                    best_seq = a.copy()
        if gen % 20 == 0:
            print(f"[DE] n={n:5d} gen={gen:3d} best={best_val:.6f}")

    return best_seq, best_val

            if len(best_seed_list) < top_k:
                best_seed_list.append((val, s.copy()))
                best_seed_list.sort(key=lambda x: x[0])
            else:
                if val < best_seed_list[-1][0]:
                    best_seed_list[-1] = (val, s.copy())
                    best_seed_list.sort(key=lambda x: x[0])

        # --------------------------------------------------------------
        # 2. Deterministic local polishing of every promising seed
        # --------------------------------------------------------------
        for seed_val, seed_seq in best_seed_list:
            if time.time() > deadline:
                break
            print(f"[seed] start local optimisation (init val={seed_val:.6f})")

            # a) exact sub‑gradient refinement (fast, cheap)
            seq_sg, val_sg = _subgradient_refine(seed_seq, n, deadline,
                                                 max_iters=25000)
            # b) L‑BFGS on smoothed max (continuation on beta)
            seq_lbfgs, val_lbfgs = _lbfgs_optimize(seq_sg, n, deadline)

            # c) a final sub‑gradient polish (to settle any remaining wiggle)
            seq_final, val_final = _subgradient_refine(seq_lbfgs, n, deadline,
                                                       max_iters=20000)

            # d) cheap hill‑climb for last‑minute tweaks
            seq_hc, val_hc = _hill_climb(seq_final, n, deadline, max_iters=20000)

            # pick the best among the four
            candidates = [(val_sg, seq_sg), (val_lbfgs, seq_lbfgs),
                          (val_final, seq_final), (val_hc, seq_hc)]
            for v, seq in candidates:
                if v < global_best_val:
                    global_best_val = v
                    global_best_seq = seq.copy()
                    print(f"[improved] n={n:5d} new global best={v:.6f}")

        # --------------------------------------------------------------
        # 3. If time permits, run a compact DE as a safety net
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0:
            seq_de, val_de = _de_optimize(n, deadline, popsize=25,
                                          K=None, max_gens=150)
            if val_de < global_best_val:
                global_best_val = val_de
                global_best_seq = seq_de.copy()
                print(f"[DE] n={n:5d} new global best={val_de:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # --------------------------------------------------------------
    # Final verification with the provided evaluate_sequence()
    # --------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("[verify] evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")

    # expose the best we found for possible reuse in a subsequent call
    globals()["height_sequence_1"] = global_best_seq.tolist()

    return global_best_seq.tolist()
