"""
Propose a step‑function (non‑negative vector) with a very low
`evaluate_sequence` value.

The algorithm is deliberately simple and fast:
1.  Choose a moderately large size (default 5000).  For this size the
    arcsine (Beta(½,½)) density is already close to the asymptotic optimum.
2.  Build the exact discretisation of the continuous arcsine law by
    integrating the density over each bin; the resulting vector already
    sums to 1.
3.  Perform a tiny projected sub‑gradient descent (a few thousand
    iterations) on the true `max(conv(a,a))` objective to shave off any
    remaining slack.  The step size follows a 1/√t schedule and the
    iterate is projected back onto the probability simplex.
4.  Return the best vector found (as a plain Python list).  The whole
    routine respects the 1000 s budget and prints intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  O(n log n) algorithm (the classic “sorting” method)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # numeric safety
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the linear convolution a*a using an FFT of length
    `fft_len`.  The convolution length is 2*len(a)-1, the function returns
    the maximal value as a plain float."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv.max())


def arcsine_distribution(n: int) -> np.ndarray:
    """
    Exact discretisation of the continuous arcsine density
        f(x) = 1 / (π √(x(1‑x))) ,   x∈(0,1)
    by integrating over each bin [i/n,(i+1)/n).  The resulting vector
    already sums to 1.
    """
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    # arcsin(2x‑1) maps [0,1] → [‑π/2,π/2]
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:                         # should never happen
        return np.full(n, 1.0 / n)
    return a / total


def subgradient_max_conv(a: np.ndarray,
                         conv_vals: np.ndarray,
                         max_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max(conv(a,a))  at the current iterate.
    For a maximising index s the gradient is 2·a shifted by s.
    The function averages the contributions of all maximising indices.
    """
    n = a.size
    g = np.zeros_like(a)
    for s in max_idx:
        i_start = max(0, s - (n - 1))
        i_end = min(n - 1, s)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = s - i
        g[i] += 2.0 * a[j]
    if max_idx.size > 0:
        g /= float(max_idx.size)
    return g


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The strategy is:

        * build the arcsine distribution for a moderately large n;
        * run a short projected sub‑gradient descent on the exact
          max‑convolution objective;
        * return the best vector found.

    The routine respects the allocated time budget (default 1000 s) and
    prints progress information.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a tiny safety margin for the outer evaluator
    overall_deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a better asymptotic constant but
    # still keeps the O(n²) final evaluation tractable.
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 5000))          # default length = 5000
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two ≥ 2n‑1

    # ------------------------------------------------------------------
    # 0) initialise with the exact arcsine discretisation
    # ------------------------------------------------------------------
    a = arcsine_distribution(n)
    conv_max = max_convolution(a, fft_len)
    best_score = 2.0 * n * conv_max
    best_a = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) short projected sub‑gradient descent (few thousand iterations)
    # ------------------------------------------------------------------
    it = 0
    # we stop either when we run out of time or after a modest number of steps
    max_iters = 6000
    while time.time() < overall_deadline and it < max_iters:
        # linear convolution via FFT (fast, O(n log n))
        fa = np.fft.fft(a, fft_len)
        conv_vals = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        cur_max = float(conv_vals.max())
        # indices where the maximum is attained (within floating tolerance)
        max_idx = np.where(np.abs(conv_vals - cur_max) < 1e-14)[0]

        # sub‑gradient (average over all active indices)
        g = subgradient_max_conv(a, conv_vals, max_idx)

        # step size – classic 𝛼 / √t schedule, 𝛼 tuned empirically
        it += 1
        lr = 0.5 / np.sqrt(it)

        # gradient step and projection back onto the simplex
        a = proj_simplex(a - lr * g)

        # evaluate the exact objective (no extra overhead, uses FFT)
        conv_max = max_convolution(a, fft_len)
        cur_score = 2.0 * n * conv_max
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_a = a.copy()
            print(f"[pgd {it:05d}] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) final normalisation and return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    elapsed = time.time() - start_time
    print(f"[done] elapsed={elapsed:.1f}s final estimated score={best_score:.6f}")
    return best_a.tolist()
