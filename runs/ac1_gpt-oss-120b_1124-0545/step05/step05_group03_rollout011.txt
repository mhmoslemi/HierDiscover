def lp_step(a, t):
    conv = _conv_fft(a, a)
    M = conv.max()
    # Build A and solve LP
    A = sp.diags([np.full(len(a), a[d]) for d in range(len(a))], offsets=np.arange(len(a)),
                 shape=(2*len(a)-1, len(a)), format='csr')
    res = opt.linprog(-np.ones(len(a)), A_ub=A, b_ub=np.full(2*len(a)-1, M),
                     bounds=[(0, None)]*len(a), method='highs')
    if not res.success:
        return a
    b = res.x
    if b.sum() == 0:
        return a
    g = b / b.sum()
    new_a = (1 - t) * a + t * g
    new_a = np.maximum(new_a, 0)
    new_a = new_a / new_a.sum()
    return new_a

"""
search for a non‑negative step function that minimises
evaluate_sequence().  The algorithm combines:

* a lightweight differential‑evolution (DE) on a piece‑wise linear
  parametrisation (≈√n parameters);
* an LP “mass‑redistribution” step that solves
      max Σ b_i   s.t.  a * b ≤ M   (a fixed, M = max(a*a))
  and mixes the result back into a;
* intensive local polishing (Adam on a smooth‑max, exact sub‑gradient
  mirror descent, and a stochastic hill‑climber).

All phases respect a 1000 s global budget.  The best sequence (length 4000
or 8000) is returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Objective utility
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a)  (a is assumed to sum to 1)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# LP “mass‑redistribution” step
# ----------------------------------------------------------------------


def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """Toeplitz matrix K with K_{k,j}=a_{k-j} (k‑j in 0..n‑1)."""
    n = a.shape[0]
    # each diagonal offset d (0 ≤ d < n) contains the constant a[d]
    data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=np.int64)
    K = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return K


def _lp_step(a: np.ndarray, t: float) -> np.ndarray:
    """
    Solve  max Σ b_i   s.t.  a * b ≤ M,  b ≥ 0.
    Return the convex combination (1‑t)·a + t·(b/Σb).
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    M = np.max(conv)

    # build the linear constraints a * b ≤ M
    K = _build_conv_matrix(a)
    ub = np.full(2 * n - 1, M, dtype=np.float64)

    # linear programme: minimise -Σb   (i.e. maximise Σb)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = opt.linprog(c, A_ub=K, b_ub=ub, bounds=bounds,
                      method="highs", options={"presolve": True})

    if not res.success or res.x.sum() == 0.0:
        # LP failed – fall back to the original vector
        return a.copy()

    b = res.x
    g = b / b.sum()                 # renormalise to L1 = 1
    new_a = (1.0 - t) * a + t * g
    new_a = np.maximum(new_a, 0.0)
    new_a /= new_a.sum()
    return new_a


# ----------------------------------------------------------------------
# Adam on a smoothed max (as in the reference code)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑softmax parametrisation, using a smooth‑max surrogate."""
    v = np.log(seed + 1e-30)            # log‑softmax parameters
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta_smooth = 5.0
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                        # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta_smooth *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta_smooth * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # back‑prop to the softmax parameters
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 100000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # avoid overflow
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """Random pair‑wise mass transfer, keeping improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.01  # initial relative step size

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # randomly decide who donates
        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new)

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 1000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val

# Minimal DE (piece‑wise‑linear) – small population, few generations
# ----------------------------------------------------------------------


def _eval_param(p: np.ndarray, knots: np.ndarray, n: int) -> tuple[float, np.ndarray]:
    """Interpolate low‑dimensional simplex vector p to length‑n, normalise, eval."""
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    if a.sum() == 0.0:
        a = _uniform(n)
    a /= a.sum()
    val = _objective(a)
    return val, a


def _de_optimize(n: int,
                 deadline: float,
                 popsize: int = 20,
                 max_gens: int = 60) -> tuple[np.ndarray, float]:
    """Very small‑scale DE on the piecewise‑linear parametrisation."""
    rng = np.random.default_rng()
    K = max(30, int(2 * np.sqrt(n)))  # a few dozen knots
    knots = np.linspace(0, n - 1, K)

    # initialise population on the simplex
    pop = [rng.dirichlet(np.ones(K)) for _ in range(popsize)]
    pop_vals = np.empty(popsize, dtype=np.float64)
    pop_seq = [None] * popsize
    for i in range(popsize):
        val, seq = _eval_param(pop[i], knots, n)
        pop_vals[i] = val
        pop_seq[i] = seq

    best_idx = int(np.argmin(pop_vals))
    best_val = float(pop_vals[best_idx])
    best_seq = pop_seq[best_idx].copy()
    best_p = pop[best_idx].copy()

    F = 0.7
    CR = 0.9

    gen = 0
    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a_vec, b_vec, c_vec = pop[a_idx], pop[b_idx], pop[c_idx]

            donor = a_vec + F * (b_vec - c_vec)
            donor = np.clip(donor, 0.0, None)
            ds = donor.sum()
            if ds == 0.0:
                donor = _uniform(K)
            else:
                donor /= ds

            # binomial crossover
            trial = pop[i].copy()
            jrand = rng.integers(K)
            for j in range(K):
                if rng.random() < CR or j == jrand:
                    trial[j] = donor[j]
            trial = np.clip(trial, 0.0, None)
            ts = trial.sum()
            if ts == 0.0:
                trial = _uniform(K)
            else:
                trial /= ts

            val_trial, seq_trial = _eval_param(trial, knots, n)

            if val_trial < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val_trial
                pop_seq[i] = seq_trial
                if val_trial < best_val:
                    best_val = val_trial
                    best_seq = seq_trial.copy()
                    best_p = trial.copy()

        if gen % 10 == 0:
            print(f"[DE] n={n:5d} gen={gen:3d} best={best_val:.6f}")

    return best_seq, best_val

    else:
        final_seq = best_seq_small
        final_val = best_val_small
        final_n = n_small

    print(f"\n=== Finished ===\nBest objective ≈ {final_val:.6f} (length {final_n})")

    # store for possible future runs
    globals()["height_sequence_1"] = final_seq.tolist()

    # quick sanity check with the provided evaluate_sequence()
    try:
        _ = evaluate_sequence(final_seq.tolist())
    except Exception:
        pass

    return final_seq.tolist()
