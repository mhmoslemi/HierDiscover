def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    n_choices = [1500, 2000, 3000, 5000, 8000]   # list
    best_score = float('inf')
    best_a = None
    best_n = None

    # Phase A: param families across n
    for n in n_choices:
        fft_len = 1 << ((2*n-1).bit_length())
        # candidate families
        cand_list = []
        cand_list.append(arcsine_distribution(n))
        cand_list.append(np.full(n, 1.0/n))  # uniform
        # Beta symmetric alphas grid
        for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            cand_list.append(beta_distribution(n, alpha, alpha))
        # mixture arcsine/uniform
        arcs = cand_list[0]; uni = cand_list[1]
        for lam in [0.2, 0.4, 0.6, 0.8]:
            mix = lam*arcs + (1-lam)*uni
            mix /= mix.sum()
            cand_list.append(mix)
        # random dirichlet seeds few
        for _ in range(3):
            rand = rng.dirichlet(np.ones(n))
            cand_list.append(rand)

        for a in cand_list:
            # evaluate
            fft_a = np.fft.rfft(a, fft_len)
            conv = np.fft.irfft(fft_a * fft_a, fft_len)[:2*n-1]
            maxconv = conv.max()
            score = 2.0 * n * maxconv
            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_n = n
                # print progress
                print(f"[init n={n}] new best {best_score:.6f}")
        # stop early if time elapsed near deadline
        if time.time() > deadline*0.9:
            break
    # Phase B: Exponentiated gradient refinement
    # using best_n
    n = best_n if best_n is not None else 2000
    fft_len = 1 << ((2*n-1).bit_length())
    a = best_a.copy()
    it = 0
    base_eta = 0.5
    last_improve_time = time.time()
    while time.time() < deadline:
        # compute conv and max
        fft_a = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fft_a * fft_a, fft_len)[:2*n-1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_a = a.copy()
            last_improve_time = time.time()
            print(f"[EG {it:07d}] improved to {best_score:.6f}")
        # compute subgradient
        # find indices where conv close to max (relative tolerance)
        tol = 1e-12
        max_idxs = np.where(conv >= max_val - tol)[0]
        # compute gradient
        g = np.zeros_like(a)
        for s in max_idxs:
            i_start = max(0, s-(n-1))
            i_end = min(n-1, s)
            i = np.arange(i_start, i_end+1)
            j = s - i
            g[i] += 2.0 * a[j]
        if max_idxs.size > 0:
            g /= float(max_idxs.size)
        # step size schedule
        it += 1
        eta = base_eta / np.sqrt(it)
        # exponentiated gradient update
        a = a * np.exp(-eta * g)
        # add small random noise occasionally
        if it % 2000 == 0:
            # add jitter to avoid stagnation
            noise = rng.normal(scale=0.001, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
        # renormalize to sum 1
        total = a.sum()
        if total <= 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / total
        # restart if no improvement for long time
        if time.time() - last_improve_time > 30.0:
            # random perturbation of best
            a = best_a.copy()
            # add small jitter
            noise = rng.normal(scale=0.01, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            last_improve_time = time.time()
            print(f"[EG {it}] restart")
    # Phase C: final simulated annealing refinement for remaining seconds
    remaining = deadline - time.time()
    if remaining > 5:
        # use simple SA moves
        # same parameters
        T_start = best_score * 0.005
        T_end = best_score * 1e-6
        sa_start = time.time()
        a = best_a.copy()
        best_conv = best_score/(2.0*n)
        it_sa = 0
        while time.time() < deadline:
            elapsed = time.time() - sa_start
            frac = elapsed / remaining
            T = T_start * (1-frac) + T_end * frac
            # random i,j, move delta
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            max_move = min(a[i], 0.05*a[i] + 1e-12)
            if max_move <= 0:
                continue
            delta = rng.random() * max_move
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # evaluate
            fft_new = np.fft.rfft(a_new, fft_len)
            conv_new = np.fft.irfft(fft_new * fft_new, fft_len)[:2*n-1]
            maxconv_new = conv_new.max()
            new_score = 2.0*n*maxconv_new
            if new_score < best_score or rng.random() < np.exp(-(new_score - best_score)/ (T+1e-12)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    best_conv = maxconv_new
                    print(f"[SA {it_sa}] new best {best_score:.6f}")
            it_sa += 1

    # final normalization, ensure nonnegative
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = np.full(best_n, 1.0/best_n)
    elapsed = time.time() - start
    print(f"[done] total time = {elapsed:.1f}s, best score = {best_score:.6f}, n={best_n}")
    return best_a.tolist()

"""
Propose a step‑function (non‑negative probability vector) that minimises
`evaluate_sequence`.  

The algorithm has three phases:

1. **Fast parametric sweep** – a handful of families (arcsine, symmetric
   Beta, mixtures and a few random Dirichlet draws) are tried for several
   problem sizes.  The best candidate is kept as a warm start.

2. **Exponentiated‑gradient mirror descent** – the true objective
   `2·n·max(conv(a,a))` is minimised directly on the simplex.
   For the current vector the sub‑gradient of the maximal convolution
   entry is computed and an exponential update
   `a_i ← a_i·exp(−η·g_i)` is performed, followed by a simple
   ℓ₁‑normalisation (no O(n log n) simplex projection!).  A
   diminishing stepsize `η = η₀/√t` is used, occasional random
   perturbations and restarts avoid stagnation.

3. **Local simulated‑annealing refinement** – when the time budget is
   almost exhausted a cheap stochastic hill‑climber with a linear
   temperature schedule performs tiny mass‑transfer moves and accepts
   them with the Metropolis rule.

All work is done with FFT‑based convolution (real FFT) which is
O(N log N) and very fast.  The code respects the 1000 s budget and
prints intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def arcsine_distribution(n: int) -> np.ndarray:
    """
    Exact discretisation of the arcsine density f(x)=1/(π√(x(1−x))) on [0,1].
    The vector is normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    tot = a.sum()
    if tot == 0.0:
        return np.full(n, 1.0 / n)
    return a / tot


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric (or asymmetric) Beta(α,β) density on a uniform grid,
    normalised to sum = 1.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    tot = a.sum()
    if tot == 0.0:
        return np.full(n, 1.0 / n)
    return a / tot


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """
    Linear convolution a*a via a real FFT and return its maximum entry.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return float(conv.max())


def compute_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)−1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def subgradient_max_conv(a: np.ndarray, conv: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max(conv(a,a))  w.r.t. a.
    For each active shift s we add 2·a shifted by s.
    The result is the average over all active shifts.
    """
    n = a.size
    g = np.zeros_like(a)
    for s in max_idxs:
        i_start = max(0, s - (n - 1))
        i_end = min(n - 1, s)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = s - i
        g[i] += 2.0 * a[j]
    if max_idxs.size:
        g /= float(max_idxs.size)
    return g

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine consists of a cheap parametric sweep,
    an exponentiated‑gradient mirror‑descent phase and a final simulated‑
    annealing refinement.  It respects the given time budget and prints
    progress information.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin before forced termination
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Phase 0 – quick sweep over a few families and several lengths
    # ------------------------------------------------------------------
    candidate_lengths = [1500, 2000, 3000, 5000, 8000]
    best_a = None
    best_score = float("inf")
    best_n = None

    # a few alphas for the Beta family (symmetric case)
    beta_alphas = [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]

    for n in candidate_lengths:
        fft_len = 1 << ((2 * n - 1).bit_length())
        # ---- build a small pool of candidates for this n -----------------
        candidates = []

        # arcsine (usually very strong)
        candidates.append(arcsine_distribution(n))

        # uniform baseline
        candidates.append(np.full(n, 1.0 / n))

        # symmetric Beta
        for a in beta_alphas:
            candidates.append(beta_distribution(n, a, a))

        # mixtures arcsine / uniform
        arcs = candidates[0]
        uni = candidates[1]
        for lam in [0.2, 0.4, 0.6, 0.8]:
            mix = lam * arcs + (1.0 - lam) * uni
            mix /= mix.sum()
            candidates.append(mix)

        # a few random Dirichlet draws (spread out mass)
        for _ in range(3):
            candidates.append(rng.dirichlet(np.ones(n)))

        # ---- evaluate each candidate ------------------------------------
        for idx, a in enumerate(candidates):
            conv = compute_conv(a, fft_len)
            maxc = conv.max()
            score = 2.0 * n * maxc
            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_n = n
                print(f"[init n={n:5d} cand={idx:02d}] new best = {best_score:.6f}")

        # stop early if we are close to the deadline
        if time.time() > deadline * 0.9:
            break

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient (mirror descent) on the simplex
    # ------------------------------------------------------------------
    if best_a is None:                     # fallback (should never happen)
        best_n = 2000
        best_a = np.full(best_n, 1.0 / best_n)
        best_score = 2.0 * best_n * (1.0 / best_n)   # =2

    n = best_n
    fft_len = 1 << ((2 * n - 1).bit_length())
    a = best_a.copy()
    it = 0
    eta0 = 0.5
    last_improve = time.time()

    while time.time() < deadline:
        # convolution and max entry
        conv = compute_conv(a, fft_len)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # record improvement
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[EG {it:07d}] improved score = {best_score:.6f}")

        # sub‑gradient of the maximal entry (average over active shifts)
        # tolerance is absolute – 1e‑12 works well for double precision
        tol = 1e-12
        max_idxs = np.where(conv >= max_val - tol)[0]
        g = subgradient_max_conv(a, conv, max_idxs)

        # step size – classical  η_t = η0 / √t
        it += 1
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update + ℓ₁‑normalisation
        a = a * np.exp(-eta * g)

        # occasional jitter to escape flat plateaus
        if it % 2000 == 0:
            a += rng.normal(scale=0.001, size=n)

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # if we have been idle for a while, restart from the current best
        if time.time() - last_improve > 30.0:
            a = best_a.copy()
            # add a modest perturbation before continuing
            a += rng.normal(scale=0.01, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            last_improve = time.time()
            print(f"[EG {it}] restart after stagnation")

    # ------------------------------------------------------------------
    # Phase 2 – short simulated‑annealing fine‑tuning (if time left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a = best_a.copy()
        best_conv = best_score / (2.0 * n)

        # linear temperature schedule
        T_start = best_score * 0.004
        T_end = best_score * 1e-6
        sa_start = time.time()
        it_sa = 0

        while time.time() < deadline:
            elapsed = time.time() - sa_start
            frac = min(1.0, elapsed / remaining)
            T = T_start * (1.0 - frac) + T_end * frac

            # propose a random elementary mass transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            # move up to 5 % of a[i] (or a tiny epsilon)
            max_move = min(a[i], 0.05 * a[i] + 1e-12)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # evaluate proposal
            conv_new = compute_conv(a_new, fft_len)
            new_score = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    print(f"[SA {it_sa}] new best = {best_score:.6f} (t={time.time() - start:.1f}s)")

            it_sa += 1

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(best_n, 1.0 / best_n)
    else:
        best_a /= total

    elapsed = time.time() - start
    print(f"[done] total time = {elapsed:.1f}s, best score = {best_score:.6f}, length = {best_n}")
    return best_a.tolist()
