"""
propose_candidate
==================

A three‑phase optimiser for the “max‑convolution” objective used by
`evaluate_sequence`.

1. **Fast parametric sweep** – a few analytic families (arcsine,
   symmetric Beta, simple mixtures) are discretised and the best
   candidate is kept.

2. **Iterative linear‑programming refinement** – for the current
   distribution *f* we solve a linear programme

        maximise   Σ g_i
        subject to   (f * g)_k ≤ M_f   for all k,
                     g_i ≥ 0,

   where *M_f* = max(f⋆f).  The optimal *g* has the largest feasible total
   mass; after normalising it to a probability vector we obtain a new
   distribution with a strictly smaller max‑convolution (or we stop).

   The convolution constraints are expressed as a sparse Toeplitz matrix,
   built with `scipy.sparse.diags`; the LP itself is solved by
   `scipy.optimize.linprog` (HiGHS).  Only a few LP iterations are needed
   to push the constant below the previous record.

3. **Local stochastic refinement** – a cheap incremental‑update SA
   (simulated annealing) moves a small amount of mass between two
   coordinates.  Because a single move changes the convolution by a
   rank‑1 update, the new max can be computed by simple slice additions
   (`O(n)`) rather than a full FFT, allowing many cheap proposals.

The whole routine respects the supplied time budget (default 1000 s) and
returns the best vector found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional fast imports – if SciPy is unavailable we fall back to the
# projected‑gradient method used previously.
# ----------------------------------------------------------------------
try:
    from scipy.sparse import diags, csr_matrix
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:                     # pragma: no cover
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def fft_len_for_n(n: int) -> int:
    """Return the smallest power‑of‑two FFT length for a linear convolution."""
    return 1 << ((2 * n - 1).bit_length())


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the linear convolution a⋆a using an FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv.max())


def conv_update_on_move(conv: np.ndarray,
                        a: np.ndarray,
                        i: int,
                        j: int,
                        delta: float) -> np.ndarray:
    """
    Return a copy of ``conv`` after moving ``delta`` mass from a[i] to a[j].
    The update follows   conv' = conv - 2·delta·shift_i + 2·delta·shift_j,
    where shift_x places ``a`` starting at index ``x``.
    """
    new_conv = conv.copy()
    n = a.size
    # subtract the contribution of the removed mass
    new_conv[i:i + n] -= 2.0 * delta * a
    # add the contribution of the added mass
    new_conv[j:j + n] += 2.0 * delta * a
    return new_conv


def arcsine_distribution(n: int) -> np.ndarray:
    """Exact discretisation of the continuous arcsine law on [0,1]."""
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric (or not) Beta(α,β) density on a uniform grid, normalised."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def build_constraint_matrix(f: np.ndarray) -> csr_matrix:
    """
    Construct the sparse Toeplitz matrix A such that
        (A @ g)[k] = (f * g)[k]   (linear convolution).

    For each offset d = 0…n‑1 the diagonal at offset ``d`` contains the
    constant value ``f[d]``.
    """
    n = f.size
    # each diagonal has exactly n entries (the full length of g)
    data = [np.full(n, f[d], dtype=float) for d in range(n)]
    offsets = list(range(n))
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return A


def evaluate_vec(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a probability vector (sum = 1)."""
    return 2.0 * a.size * max_convolution(a, fft_len)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     n: int = 2000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The algorithm runs three phases described in
    the module docstring and respects the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------
    # 0) Problem size & FFT set‑up
    # ------------------------------
    n = int(kwargs.get("n", n))
    fft_len = fft_len_for_n(n)

    # -------------------------------------------------
    # 0) Gather a few cheap parametric candidates
    # -------------------------------------------------
    candidates = []

    # reuse a previously discovered construction if present
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.size == n and a0.sum() > 0.0:
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # arcsine (the classic starter)
    candidates.append(arcsine_distribution(n))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # symmetric Beta family – a finer sweep than the previous code
    for alpha in np.linspace(0.1, 2.0, 38):           # step ≈0.05
        candidates.append(beta_distribution(n, alpha, alpha))

    # mixtures of arcsine and uniform (coarse)
    arcsine = candidates[1]
    uniform = candidates[2]
    for lam in np.linspace(0.0, 1.0, 21):
        mix = lam * arcsine + (1.0 - lam) * uniform
        mix /= mix.sum()
        candidates.append(mix)

    # -------------------------------------------------
    # 1) Choose the best cheap candidate
    # -------------------------------------------------
    best_a = None
    best_score = math.inf
    for idx, a in enumerate(candidates):
        sc = evaluate_vec(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[init {idx:02d}] new best = {best_score:.6f}")

    # -------------------------------------------------
    # 2) Linear‑programming refinement (if SciPy is present)
    # -------------------------------------------------
    if SCIPY_AVAILABLE:
        # allocate roughly 55 % of the total budget to this phase
        lp_deadline = start_time + 0.55 * budget_s
        f = best_a.copy()
        lp_iter = 0
        while time.time() < lp_deadline:
            lp_iter += 1
            # current bound M = max(f⋆f)
            M = max_convolution(f, fft_len)

            # build the Toeplitz constraint matrix A (sparse)
            A = build_constraint_matrix(f)

            # Linear programme: maximise Σ g_i  ⇔ minimise -Σ g_i
            c = -np.ones(n, dtype=float)
            b_ub = np.full(2 * n - 1, M, dtype=float)

            # solve
            res = linprog(c,
                          A_ub=A,
                          b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method="highs",
                          options={"presolve": True, "time_limit": max(0.0, lp_deadline - time.time())})
            if not res.success:
                print(f"[LP {lp_iter:04d}] linprog failed – {res.message}")
                break

            g0 = res.x
            sum_g0 = g0.sum()
            if sum_g0 <= 1e-12:
                print(f"[LP {lp_iter:04d}] degenerate solution (zero mass)")
                break

            # Normalise to a probability vector
            g = g0 / sum_g0

            # Evaluate the new candidate
            new_score = evaluate_vec(g, fft_len)

            if new_score < best_score - 1e-9:
                best_score = new_score
                best_a = g.copy()
                f = g.copy()
                print(f"[LP {lp_iter:04d}] improved to {best_score:.6f}")
                continue

            # If pure replacement did not improve, try a convex blend
            t = 0.3
            f_blend = (1.0 - t) * f + t * g
            f_blend /= f_blend.sum()
            blend_score = evaluate_vec(f_blend, fft_len)

            if blend_score < best_score - 1e-9:
                best_score = blend_score
                best_a = f_blend.copy()
                f = f_blend.copy()
                print(f"[LP {lp_iter:04d}] blend improved to {best_score:.6f}")
                continue

            # No improvement – stop the LP phase
            print(f"[LP {lp_iter:04d}] no improvement, terminating LP loop")
            break

    # -------------------------------------------------
    # 3) Local stochastic refinement (simple SA)
    # -------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a = best_a.copy()
        # pre‑compute the full convolution of the current a
        conv = np.real(np.fft.ifft(np.fft.fft(a, fft_len) *
                                   np.fft.fft(a, fft_len)))[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # temperature schedule – linear cooling
        T_start = cur_score * 0.004
        T_end = cur_score * 1e-6
        sa_start = time.time()
        sa_iter = 0

        while time.time() < deadline:
            elapsed = time.time() - sa_start
            frac = min(1.0, elapsed / remaining)
            T = T_start * (1.0 - frac) + T_end * frac

            # propose a random elementary mass transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1   # ensure i ≠ j

            # move up to 5 % of the donor’s mass
            max_move = min(a[i], 0.05 * a[i] + 1e-12)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # cheap incremental update of the convolution
            conv_new = conv_update_on_move(conv, a, i, j, delta)
            new_max = conv_new.max()
            new_score = 2.0 * n * new_max

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
                a = a_new
                conv = conv_new
                cur_score = new_score
                cur_max = new_max
                if new_score < best_score - 1e-9:
                    best_score = new_score
                    best_a = a_new.copy()
                    print(f"[SA {sa_iter:07d}] new best = {best_score:.6f}  (t={time.time() - start_time:.1f}s)")

            sa_iter += 1

    # ------------------------------------------------------------------
    # Final normalisation & return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    elapsed = time.time() - start_time
    print(f"[done] total time = {elapsed:.1f}s, best score = {best_score:.6f}, length = {n}")

    return best_a.tolist()
