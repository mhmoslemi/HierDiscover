"""
Proposes a step function minimizing the objective 2·n·max(conv(a,a))/(sum a)^2.
The algorithm:
1. Exhaustive grid search over several parametric families (product,
   sum, exponential, beta, cosine) to obtain promising seeds.
2. For the best few seeds runs a projected Adam optimisation on a
   smooth approximation of the max‑convolution (log‑sum‑exp).  The learning
   rate and temperature schedule are tuned for fast convergence.
3. A short sub‑gradient phase using the exact max‑convolution refines the
   solution.
4. Final stochastic refinement (Gaussian perturbations) explores the
   neighbourhood of the current best.
All operations respect the non‑negative simplex constraint; the sequence
is projected onto the probability simplex after each update.
"""

import time
import numpy as np

# Helper: smallest power of two >= x
def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def conv_fft(a):
    """Linear convolution a*a using FFT. Returns length 2·len(a)-1 array."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

def conv_fft_general(a, b):
    """Linear convolution of a and b using FFT."""
    size = _next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: len(a) + len(b) - 1]

def compute_score(a):
    """Exact objective for a (treated as a probability vector)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)

def project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_move(a, scale):
    """Transfer a small amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    # The amount transferred is proportional to the source mass
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

# Analytic families ---------------------------------------------------------

def analytic_product(n, p, offset):
    """U‑shaped family: a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total <= 0.0 or not np.isfinite(total):
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / total
    return a

def analytic_sum(n, p, offset):
    """Sum of two power‑laws from the ends."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total <= 0.0 or not np.isfinite(total):
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / total
    return a

def analytic_exponential(n, lam):
    """Exponential tails: a_i ∝ exp(-lam·min(i,n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    total = a.sum()
    if total <= 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / total
    return a

def analytic_cosine(n, p):
    """Cosine‑power family."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    total = a.sum()
    if total <= 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / total
    return a

def analytic_beta(n, alpha, beta):
    """Beta‑shaped family: a_i ∝ (i+1)^{α}·(n‑i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total <= 0.0 or not np.isfinite(total):
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / total
    return a

# Smoothed max gradient -----------------------------------------------------

def smooth_max_gradient(a, alpha):
    """Gradient of f_α(a) = (1/α) log Σ exp(α·(a∗a)_k)."""
    n = a.shape[0]
    conv = conv_fft(a)  # length 2n-1
    maxc = conv.max()
    # numerically stable soft‑max weights
    exp_alpha = np.exp(alpha * (conv - maxc))
    w = exp_alpha / exp_alpha.sum()  # Σ w = 1
    rev_a = a[::-1]
    g_full = conv_fft_general(rev_a, w)  # length 3n-2
    start = n - 1
    g = 2.0 * g_full[start:start + n]
    return g

# Sub‑gradient of the true max‑convolution (used in fine tuning) ----------
def max_convolution_subgradient(a):
    """Return a sub‑gradient of max_k (a*a)_k."""
    conv = conv_fft(a)
    kmax = int(conv.argmax())
    n = a.shape[0]
    grad = np.zeros_like(a)
    # For each i such that 0 <= kmax-i < n we have derivative 2*a_{kmax-i}
    # This is vectorised:
    idx = np.arange(n)
    j = kmax - idx
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]
    return grad, conv.max()

# Main entry ---------------------------------------------------------------

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimizing the evaluation
    function.  The routine performs:
        * a dense grid search over several analytic families,
        * a projected Adam optimisation of a smooth max‑convolution,
        * a short sub‑gradient refinement using the exact max,
        * a final stochastic Gaussian perturbation stage.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Determine length
    if "n" in kwargs:
        n = int(kwargs["n"])
    elif "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        n = 2000

    # ------------------------------------------------------------------
    # Initialise with any provided global solution
    best_seq = None
    best_score = np.inf
    if "height_sequence_1" in globals():
        seq = np.asarray(globals()["height_sequence_1"], dtype=float)
        seq = np.clip(seq, 0.0, None)
        seq = seq / seq.sum()
        sc = compute_score(seq)
        best_seq = seq.copy()
        best_score = sc
        print(f"[init] global seed score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # ---- 1) Grid search over analytic families -----------------------
    MAX_CAND = 12    # number of top candidates we keep for refinement
    candidates = []  # list of (score, array)

    def store_candidate(arr):
        nonlocal candidates, best_seq, best_score
        sc = compute_score(arr)
        if sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            print(f"[grid] new global best = {best_score:.6f}")
        # keep only top MAX_CAND candidates
        if len(candidates) < MAX_CAND:
            candidates.append((sc, arr.copy()))
        else:
            # find worst among stored
            worst_idx = np.argmax([c[0] for c in candidates])
            if sc < candidates[worst_idx][0]:
                candidates[worst_idx] = (sc, arr.copy())

    # product family
    p_vals = np.linspace(0.4, 1.9, 27)        # finer around 1
    off_vals = np.logspace(-4, 0.5, 27)      # 1e-4 .. 3.16
    for p in p_vals:
        for off in off_vals:
            cand = analytic_product(n, p, off)
            store_candidate(cand)

    # sum family
    for p in np.linspace(0.4, 2.0, 25):
        for off in np.logspace(-4, 0.5, 25):
            cand = analytic_sum(n, p, off)
            store_candidate(cand)

    # exponential family
    for lam in np.logspace(-3, 0.5, 30):    # 0.001 .. 3.16
        cand = analytic_exponential(n, lam)
        store_candidate(cand)

    # cosine power family
    for p in np.linspace(0.5, 12.0, 20):
        cand = analytic_cosine(n, p)
        store_candidate(cand)

    # beta family (allow small negative exponents)
    al_vals = np.arange(-0.8, 2.05, 0.2)
    be_vals = np.arange(-0.8, 2.05, 0.2)
    for a_ in al_vals:
        for b_ in be_vals:
            cand = analytic_beta(n, a_, b_)
            store_candidate(cand)

    # ------------------------------------------------------------------
    # Sort candidates by score (ascending) and keep the best few
    candidates.sort(key=lambda x: x[0])
    top_candidates = [c[1] for c in candidates[:max(1, len(candidates)//2)]]

    # ------------------------------------------------------------------
    # ---- 2) Adam optimisation on smoothed max for each top seed -----
    # Allocate remaining time roughly equally among candidates
    remaining = deadline - time.time()
    if remaining <= 0:
        return best_seq.tolist() if best_seq is not None else [1.0]

    time_per_candidate = remaining / max(1, len(top_candidates))
    print(f"[opt] remaining time {remaining:.1f}s, "
          f"time per candidate ≈ {time_per_candidate:.1f}s")

    for idx, seed_seq in enumerate(top_candidates):
        # allocate sub‑deadline
        sub_deadline = time.time() + time_per_candidate
        a = seed_seq.copy()
        # ensure feasibility
        a = np.clip(a, 0.0, None)
        a = project_simplex(a)

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        beta1, beta2, eps = 0.9, 0.999, 1e-8
        step = 0
        # temperature schedule for soft‑max
        alpha = 1.0
        alpha_target = 200.0
        growth = (alpha_target / alpha) ** (1.0 / 50000.0)

        # track best for this seed
        local_best_seq = a.copy()
        local_best_score = compute_score(a)

        while time.time() < sub_deadline:
            step += 1
            grad = smooth_max_gradient(a, alpha)

            # Adam update
            lr = 0.07 / (1.0 + step * 1e-5)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** step)
            v_hat = v / (1.0 - beta2 ** step)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection onto simplex
            a = np.clip(a, 0.0, None)
            a = project_simplex(a)

            # increase temperature
            alpha = min(alpha_target, alpha * growth)

            # occasional exact evaluation
            if step % 150 == 0:
                cur_score = compute_score(a)
                if cur_score < local_best_score - 1e-12:
                    local_best_score = cur_score
                    local_best_seq = a.copy()
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[adam] new global best = {best_score:.6f} (step {step})")

            # occasional cheap random mass‑transfer move
            if np.random.rand() < 0.01:
                a = random_move(a, scale=0.4)

        # after budget for this candidate, keep its local best
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.copy()
            print(f"[seed {idx}] improved global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # ---- 3) Sub‑gradient refinement using the exact max ------------
    # Use the current best sequence as starting point
    a = best_seq.copy()
    m = np.zeros_like(a)

"""
Proposes a step function minimizing the objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm proceeds in four stages:

1. **Dense grid search** over several parametric families (product, sum,
   exponential, cosine‑power, beta) to obtain promising seeds.
2. **Projected Adam optimisation** on a smooth approximation of the
   max‑convolution (log‑sum‑exp).  The learning‑rate schedule and the
   temperature α are tuned for rapid convergence; each seed receives a
   fair share of the remaining time budget.
3. **Sub‑gradient refinement** using the exact max‑convolution to tighten
   the solution further.
4. **Final stochastic refinement** with shrinking Gaussian perturbations
   exploring the neighbourhood of the current best.

All iterates are projected onto the probability simplex (non‑negative,
unit‑sum) after each update, guaranteeing validity for the
`evaluate_sequence` harness.  The function returns the best sequence found
before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of a and b using FFT."""
    size = _next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: len(a) + len(b) - 1]


def compute_score(a: np.ndarray) -> float:
    """Exact objective (assumes a is non‑negative)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    return float(2.0 * a.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {w | w_i ≥ 0, Σ w_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families (used for the initial grid search)
# ----------------------------------------------------------------------
def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """U‑shaped: a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total <= 0.0 or not np.isfinite(total):
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """Sum of two power‑laws from the ends."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total <= 0.0 or not np.isfinite(total):
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """Exponential tails: a_i ∝ exp(-lam·min(i,n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    total = a.sum()
    if total <= 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / total
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """Cosine‑power family."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    total = a.sum()
    if total <= 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / total
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped: a_i ∝ (i+1)^α·(n‑i)^β."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total <= 0.0 or not np.isfinite(total):
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Smoothed‑max gradient (log‑sum‑exp)
# ----------------------------------------------------------------------
def smooth_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """∇ of (1/α)·log Σ exp(α·(a∗a)_k)."""
    n = a.shape[0]
    conv = conv_fft(a)                     # length 2n‑1
    maxc = conv.max()
    exp_alpha = np.exp(alpha * (conv - maxc))
    w = exp_alpha / exp_alpha.sum()       # soft‑max weights, Σw = 1
    rev_a = a[::-1]
    g_full = conv_fft_general(rev_a, w)   # length 3n‑2
    start = n - 1
    return 2.0 * g_full[start:start + n]


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution (used for fine tuning)
# ----------------------------------------------------------------------
def max_convolution_subgradient(a: np.ndarray):
    """Return a sub‑gradient of max_k (a*a)_k and the max value."""
    conv = conv_fft(a)
    kmax = int(conv.argmax())
    n = a.shape[0]
    grad = np.zeros_like(a)
    idx = np.arange(n)
    j = kmax - idx
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]
    return grad, conv.max()
