   for it in range(max_iters):
       if time.time()>deadline: break
       # pick two indices i!=j

"""
Hybrid heuristic for minimizing

    2·n·max(conv(a,a)) / (∑a)²

where ``conv`` is the full linear convolution.  The method consists of

1.  a large pool of analytic seeds (uniform, geometric, power‑law, beta,
    sine‑window, random Dirichlet, sparse Sidon‑set, etc.);
2.  a fast Adam optimiser on a smooth ‑log‑sum‑exp approximation of the
    max‑operator (the previous implementation used an incorrect shift;
    here we use the true max of the convolution);
3.  an exact sub‑gradient mirror descent that works on the true max;
4.  a cheap simulated‑annealing hill‑climber;
5.  a final coordinate‑descent style mass‑swap refinement.

All phases are time‑aware and stop as soon as the global deadline is hit.
The best sequence found (as a plain ``list`` of ``float``) is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (length = len(x)+len(y)-1)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming ∑seq == 1.
    This matches the behaviour of ``evaluate_sequence`` when the same
    normalisation is applied.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric: weight decays from both ends."""
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _random_logits(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    return _softmax(logits)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed by a Dirichlet draw (conc < 1 makes it sparse)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _sidon_set(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Greedy construction of a Sidon (B₂) set of size `m` inside {0,…,n‑1}.
    Returns a probability vector with mass 1/m on each chosen index.
    """
    while True:  # restart until we succeed
        chosen = []
        sums = set()
        attempts = 0
        max_attempts = n * 10
        while len(chosen) < m and attempts < max_attempts:
            cand = rng.integers(0, n)
            conflict = False
            for s in chosen:
                if (cand + s) in sums:
                    conflict = True
                    break
                if (2 * cand) == (cand + cand) and (2 * cand) in sums:
                    conflict = True
                    break
            if conflict:
                attempts += 1
                continue
            # add new element
            for s in chosen:
                sums.add(cand + s)
                sums.add(s + cand)      # ordered pair count (redundant but safe)
            sums.add(2 * cand)            # self‑pair
            chosen.append(cand)
        if len(chosen) == m:
            a = np.zeros(n, dtype=np.float64)
            a[chosen] = 1.0 / m
            return a
        # restart with new random order
        attempts = 0


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Assemble a diverse set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # baseline families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families (wide range of r)
    for r in np.linspace(0.2, 0.9, 8):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 5.0, 7):
        seeds.append(_power(n, p))

    # beta‑like shapes (including sharper edges)
    for a in (0.2, 0.3, 0.5, 0.8, 1.2):
        for b in (0.2, 0.3, 0.5, 0.8, 1.2):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random soft‑max vectors (logits)
    for _ in range(5):
        seeds.append(_random_logits(n, rng))

    # sparse random patterns (few non‑zeros)
    for k in (10, 20, 30, 50):
        for conc in (0.1, 0.3):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # Sidon‑set based sparse seeds
    max_m = int(np.sqrt(2 * n))  # safe upper bound for Sidon size
    for m in (max_m // 2, max_m, max_m + 5):
        if 2 <= m < n:
            seeds.append(_sidon_set(n, m, rng))

    # reuse previous solution if length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    seed_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 20000,
) -> tuple[np.ndarray | None, float]:
    """Adam optimiser on a smooth log‑sum‑exp approximation of max(conv)."""
    eps = 1e-30
    v = np.log(seed_vec + eps)  # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.9

    # smoothing schedule (log‑sum‑exp temperature)
    beta = 5.0
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)               # stays on the simplex
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)           # correct shift (previous code used max(a) – bug)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w_k·conv_k w.r.t. a (linear convolution) ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)  # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]  # central n entries

        # ---- convert to gradient w.r.t. v (softmax Jacobian) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional gradient clipping for stability
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ---- Adam update on v ----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 80000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror descent on max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of maxₖ convₖ w.r.t a (only the active lag)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                 # avoid overflow
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250000,
    init_temp_factor: float = 0.03,
    min_temp: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """Stochastic hill‑climbing with a temperature schedule."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    # initial temperature proportional to the current objective
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9996, min_temp)

        # pick a small random subset (2–5 indices) to reshuffle
        k = rng.integers(2, 6)                # 2,3,4 or 5 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # fresh allocation respecting the total mass
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:6d} best={new_val:.6f}")

    return best_seq, best_val


def _coordinate_descent(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 120000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-10,
) -> tuple[np.ndarray, float]:
    """
    Very cheap mass‑swap local search.  Randomly moves a fraction of the mass
    from one index to another; downhill moves are always accepted, uphill
    moves with a temperature‐controlled probability.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cooling
        T = max(T * 0.9997, min_temp)

        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] == 0.0:
            continue

        # move a random fraction (up to 50%) of mass from i to j
        frac = rng.random() * 0.5
        delta = cur[i] * frac
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        new_val = _objective(new)
        delta_obj = new_val - cur_val
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[CD] n={n:5d} it={it:6d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget and returns the best sequence found (as a list of ``float``).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # retrieve a previously found construction if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # lengths to try – we focus on one medium length that gives a good
    # trade‑off between tractability and expressive power.
    lengths = [4000, 5000, 6000]

    global_best_seq = None
    global_best_val = float("inf")
    rng = np.random.default_rng(seed)

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous, rng)

        # ------------------------------------------------------------------
        # Quick scan of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # Keep only the top few seeds for the expensive phases
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:10]]

        # ------------------------------------------------------------------
        # Phase 1 – Adam on smoothed max (more iterations)
        # ------------------------------------------------------------------
        for s in top_seeds:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline, max_iters=25000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=100000)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Simulated‑annealing hill climbing
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _simulated_annealing(global_best_seq, n, deadline,
                                            max_iters=300000)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SA] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – Final coordinate‑descent refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _coordinate_descent(global_best_seq, n, deadline,
                                           max_iters=150000)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑CD] n={n:5d} new global best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Verify with the provided evaluate_sequence (optional)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    # Convert to plain list as required by the outer harness
    return global_best_seq.tolist()
