def propose_candidate(seed=42,budget_s=1000,**kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety
    
    n = int(kwargs.get('n', 2000))
    # 1. Analytic grid search (expanded)
    best_seq, top10 = analytic_grid_search(n)
    best_seq = mix_top_candidates(top10, n)
    best_score = compute_score(best_seq)
    print('[init] best after mixing:', best_score)

    # 2. Simulated annealing
    a = best_seq.copy()
    curr_score = best_score
    # SA hyperparameters
    T0 = 0.02   # initial temp
    T_end = 1e-7
    step_max = 0.5
    step_min = 1e-4
    iteration = 0
    last_improve_time = start

    while time.time() < deadline - 150:  # keep 150s for final refinement
        elapsed = time.time() - start
        progress = elapsed / (budget_s - 150)
        T = T0 * (1 - progress) + T_end * progress
        step_scale = step_max * (1 - progress) + step_min * progress

        i = np.random.randint(n)
        if a[i] == 0.0:
            continue
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # delta up to step_scale * a[i]
        delta = step_scale * a[i] * np.random.rand()
        # ensure positive
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # conv compute
        new_score = compute_score(a_new)
        # accept
        if new_score < curr_score or np.random.rand() < np.exp(-(new_score - curr_score) / max(T,1e-12)):
            a = a_new
            curr_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                last_improve_time = time.time()
                print(f"[SA {iteration:07d}] new best = {best_score:.6f}")
        # occasional perturbations
        if iteration % 5000 == 0:
            # jitter around best_seq
            jitter = np.random.randn(n) * step_scale * 0.1
            a_jit = best_seq + jitter
            a_jit = np.clip(a_jit, 0, None)
            a_jit = a_jit / a_jit.sum()
            score_jit = compute_score(a_jit)
            if score_jit < best_score:
                best_score = score_jit
                best_seq = a_jit.copy()
                a = a_jit.copy()
                curr_score = score_jit
                print(f"[SA jitter {iteration:07d}] improved to {best_score:.6f}")
        # occasional Dirichlet restart
        if np.random.rand() < 0.0005:
            rand_vec = np.random.gamma(1.0, 1.0, size=n)
            rand_vec = rand_vec / rand_vec.sum()
            rand_score = compute_score(rand_vec)
            if rand_score < best_score:
                best_score = rand_score
                best_seq = rand_vec.copy()
                a = rand_vec.copy()
                curr_score = rand_score
                print(f"[SA restart] improved to {best_score:.6f}")

        iteration += 1

    # 3. Fine gradient: Adam on smoothed max (like before)
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_adam = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    beta_start = 0.5
    beta_end = 400.0

    while time.time() < deadline - 5:
        conv = conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[Adam] new best = {best_score:.6f}")

        elapsed = time.time() - start
        progress = elapsed / (budget_s - 5)
        beta = beta_start * (1 - progress) + beta_end * progress
        # softmax weights (stable)
        exponents = beta * (conv - max_val)
        w_raw = np.exp(exponents - np.max(exponents))  # for numerical stability
        w = w_raw / w_raw.sum()

        a_rev = a[::-1]
        weighted_shift = conv_two(w, a_rev)[n-1:n-1+n]
        g = a + weighted_shift

        t_adam += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step_vec
        a_candidate = np.clip(a_candidate, 0, None)
        a_candidate = a_candidate / a_candidate.sum()
        a = a_candidate

    total_time = time.time() - start
    print(f"[done] best score = {best_score:.6f}, length = {n}, time = {total_time:.1f}s")
    return best_seq.tolist()

"""
Hybrid stochastic‑annealing + smoothed‑max Adam optimizer.

* Phase 0 – a quick analytic grid search over several parametric families.
  The 10 best vectors (on the simplex) are kept and pairwise mixed.
* Phase 1 – simulated‑annealing on the exact max‑convolution objective.
  One mass‑transfer move (i→j) is proposed each iteration; the move is
  accepted with a Metropolis probability that slowly cools.
  Occasional Dirichlet restarts and Gaussian jitter prevent stagnation.
* Phase 2 – Adam optimisation of a soft‑max (log‑sum‑exp) surrogate of the
  max‑convolution.  The temperature is increased linearly so that the
  surrogate starts very smooth and becomes the true max at the end.
* Phase 3 – a short random‑perturbation refinement.

All vectors are kept on the probability simplex (∑a_i = 1, a_i≥0) – the
objective 2·n·max(conv(a,a)) is scale‑invariant, therefore the simplex
projection is sufficient.  The routine respects a wall‑time budget (default
1000 s) and returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of a and b via FFT (full length)."""
    size = _next_pow2(a.shape[0] + b.shape[0] - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (linear, length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Objective (scale‑invariant)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a on the simplex: 2·n·max(conv(a,a)).
    The function normalises a internally (sum may drift slightly).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_self(a)
    return float(2.0 * a.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families (return normalised probability vectors)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, off: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + off) * (n - 1 - i + off)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, off: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + off) ** (-p) + (n - 1 - i + off) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    return a / a.sum()


def analytic_cosine(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    return a / a.sum()


def analytic_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta_
    return a / a.sum()


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


# ----------------------------------------------------------------------
# Helpers for the analytic phase
# ----------------------------------------------------------------------


def _add_candidate(
    cand: np.ndarray,
    best_list: list[tuple[float, np.ndarray]],
    max_len: int = 10,
) -> None:
    """Insert a candidate in a sorted top‑list (lower score = better)."""
    sc = compute_score(cand)
    inserted = False
    for idx, (s, _) in enumerate(best_list):
        if sc < s:
            best_list.insert(idx, (sc, cand.copy()))
            inserted = True
            break
    if not inserted and len(best_list) < max_len:
        best_list.append((sc, cand.copy()))
    while len(best_list) > max_len:
        best_list.pop()


def analytic_grid_search(n: int):
    """
    Exhaustively scan a set of parametric families.
    Returns the overall best vector and a list of the top‑10 candidates.
    """
    top = []  # (score, vector)

    # ----- Product family -------------------------------------------------
    p_vals = np.arange(0.1, 5.01, 0.2)
    off_vals = np.concatenate(
        (np.geomspace(1e-3, 0.01, 6), np.arange(0.01, 5.01, 0.3))
    )
    for p in p_vals:
        for off in off_vals:
            _add_candidate(analytic_product(n, p, off), top)

    # ----- Sum family ----------------------------------------------------
    for p in p_vals:
        for off in off_vals:
            _add_candidate(analytic_sum(n, p, off), top)

    # ----- Exponential tails ---------------------------------------------
    lam_vals = np.logspace(-4, 1, 30)  # 1e‑4 … 10
    for lam in lam_vals:
        _add_candidate(analytic_exponential(n, lam), top)

    # ----- Cosine‑power ---------------------------------------------------
    p_cos = np.arange(0.5, 10.1, 0.5)
    for p in p_cos:
        _add_candidate(analytic_cosine(n, p), top)

    # ----- Beta family ----------------------------------------------------
    alpha_vals = np.arange(-0.5, 3.01, 0.5)
    beta_vals = np.arange(-0.5, 3.01, 0.5)
    for a in alpha_vals:
        for b in beta_vals:
            _add_candidate(analytic_beta(n, a, b), top)

    # ----- Gaussian family ------------------------------------------------
    sigma_vals = np.linspace(0.3, n / 2.0, 25)  # from narrow to very wide
    for sigma in sigma_vals:
        _add_candidate(analytic_gaussian(n, sigma), top)

    # ----- Global reference (if provided) --------------------------------
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            _add_candidate(ref, top)

    best_vec = min(top, key=lambda x: x[0])[1].copy()
    return best_vec, top


def mix_top_candidates(top: list[tuple[float, np.ndarray]], n: int) -> np.ndarray:
    """
    Explore convex combinations of the best candidates.
    21 mixing ratios (0..1 step 0.05) are tried for each pair.
    """
    best_seq = top[0][1].copy()
    best_score = top[0][0]
    alphas = np.linspace(0.0, 1.0, 21)
    for i in range(len(top)):
        for j in range(i + 1, len(top)):
            a_i = top[i][1]
            a_j = top[j][1]
            for alpha in alphas:
                cand = alpha * a_i + (1.0 - alpha) * a_j
                cand = project_simplex(cand)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
    return best_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
        2·n·max(conv(a,a)) / (sum(a)**2)
    The routine respects the wall‑time budget and returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 0) Analytic families + mixing (fast, gives a solid baseline)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))
    print("[0] Analytic grid search (n =", n, ")")
    best_seq, top10 = analytic_grid_search(n)
    best_seq = mix_top_candidates(top10, n)
    best_score = compute_score(best_seq)
    print(f"[0] After mixing best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Simulated annealing on the exact max‑convolution
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score
    iteration = 0
    last_improve = start

    # cooling / step schedule (linear in elapsed time)
    T0, T_end = 0.025, 1e-7               # temperature (Metropolis)
    step_start, step_end = 0.5, 1e-4      # maximal relative mass transfer

    # allocate ~70 % of the budget to SA
    sa_deadline = start + 0.70 * budget_s

    while time.time() < sa_deadline and time.time() < deadline - 200:
        # progress in [0,1]
        prog = (time.time() - start) / (budget_s * 0.70)
        T = T0 * (1.0 - prog) + T_end * prog
        step_scale = step_start * (1.0 - prog) + step_end * prog

        # ----- propose a single mass‑transfer move -----
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = step_scale * a[i] * np.random.rand()   # 0 ≤ δ ≤ step_scale·a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # renormalise to fight accumulation of round‑off error
        a_new /= a_new.sum()

        new_score = compute_score(a_new)

        # ----- Metropolis acceptance -----
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                last_improve = time.time()
                print(f"[SA {iteration:07d}] new best = {best_score:.6f}")

        # ----- occasional diversifying actions -----
        if iteration % 5000 == 0 and iteration > 0:
            # small Gaussian jitter around the current best
            jitter = np.random.randn(n) * step_scale * 0.1
            a_jit = best_seq + jitter
            a_jit = np.clip(a_jit, 0, None)
            a_jit = a_jit / a_jit.sum()
            score_jit = compute_score(a_jit)
            if score_jit < best_score:
                best_score = score_jit
                best_seq = a_jit.copy()
                a = a_jit.copy()
                cur_score = score_jit
                print(f"[SA jitter {iteration:07d}] improved to {best_score:.6f}")

        if np.random.rand() < 0.0007:   # rare Dirichlet restart
            rand_vec = np.random.gamma(1.0, 1.0, size=n)
            rand_vec = rand_vec / rand_vec.sum()
            rs = compute_score(rand_vec)
            if rs < best_score:
                best_score = rs
                best_seq = rand_vec.copy()
                a = rand_vec.copy()
                cur_score = rs
                print(f"[SA restart] improved to {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 2) Adam optimisation of soft‑max surrogate (smooth max)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_adam = 0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12
    beta_start, beta_end = 0.5, 400.0

    # allocate ~25 % of the budget to Adam
    adam_deadline = start + 0.95 * budget_s

    while time.time() < adam_deadline and time.time() < deadline - 30:
        conv = conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[Adam {t_adam:07d}] new best = {best_score:.6f}")

        # temperature schedule (smooth → hard)
        prog = (time.time() - start) / (budget_s * 0.95)
        beta = beta_start * (1.0 - prog) + beta_end * prog

        # soft‑max over all convolution entries
        exps = beta * (conv - max_val)            # max exponent = 0
        w = np.exp(exps - np.max(exps))           # numerical stabilisation
        w /= w.sum()

        # weighted shifted sum (correlation of w with reversed a)
        a_rev = a[::-1]
        corr = conv_two(w, a_rev)                # length = 3n‑2
        weighted_shift = corr[n - 1 : n - 1 + n]

        g = a + weighted_shift                    # gradient of smooth objective

        # Adam update
        t_adam += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate = a_candidate / a_candidate.sum()

        # occasional tiny random perturbations to escape tiny plateaus
        if np.random.rand() < 0.012:
            a_candidate = random_move(a_candidate, lr * 0.3)
        if np.random.rand() < 0.004:
            jitter = np.random.randn(n) * 5e-4
            a_candidate = a_candidate + jitter
            a_candidate = np.clip(a_candidate, 0.0, None)
            a_candidate = a_candidate / a_candidate.sum()

        a = a_candidate

    # ------------------------------------------------------------------
    # 3) Final random refinement (few seconds)
    # ------------------------------------------------------------------
    print("[refine] final random perturbation phase")
    while time.time() < deadline:
        perturb = np.random.normal(scale=5e-4, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total = time.time() - start
    print(f"[done] best score = {best_score:.6f}, n = {n}, time = {total:.1f}s")
    return best_seq.tolist()


# ----------------------------------------------------------------------
# Small helper used in the Adam phase (mass‑transfer move)
# ----------------------------------------------------------------------


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from a random index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = np.clip(a_new, 0.0, None)
    a_new = a_new / a_new.sum()
    return a_new
