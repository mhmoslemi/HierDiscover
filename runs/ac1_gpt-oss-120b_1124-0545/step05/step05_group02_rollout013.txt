"""
Search for a step‑function that minimises

    2·n·max(conv(a,a)) / (∑ a)²

by a multi‑scale optimisation: many analytic seeds, a long Adam phase on a
small problem (n≈512), up‑sampling the best shape to larger n and
refining it with exponentiated‑gradient sub‑steps and a short simulated‑annealing
hill‑climber.  The routine respects a global time‑budget (default 1000 s) and
returns the best sequence found as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution using a real FFT."""
    L = x.size + y.size - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) assuming ∑seq = 1."""
    n = seq.size
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families of seeds (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed by a Dirichlet draw (conc < 1 makes it sparse)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(seed_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 15000,
                   lr0: float = 0.8,
                   beta_init: float = 5.0,
                   beta_mul: float = 1.02) -> tuple[np.ndarray | None, float]:
    """Adam on a smoothed version of the ``max`` objective."""
    eps = 1e-30
    v = np.log(seed_vec + eps)               # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = beta_init
    best_seq = None
    best_val = float("inf")
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ---- smoothed max (log‑sum‑exp) ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w_k·conv_k w.r.t. a (linear convolution) ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # keep central n entries

        # ---- convert to gradient w.r.t. v (softmax Jacobian) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional gradient clipping
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ---- Adam update on v ----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 100000,
                        lr_coeff: float = 0.6) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror‑descent on max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = lr_coeff / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                 # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300000,
                         init_temp_factor: float = 0.02,
                         min_temp: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing with a temperature schedule.
    Randomly selects a few indices and reshuffles their total mass via a
    Dirichlet draw.  Improvements are always accepted; uphill moves are
    accepted with probability exp(−Δ/T).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 6)                # move 2–5 positions
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:6d} best={new_val:.6f}")

    return best_seq, best_val

"""
Search for a step‑function that minimises

    2·n·max(conv(a,a)) / (∑ a)²

using a multi‑scale scheme:

* a rich collection of analytic seeds,
* a long Adam optimisation on a smoothed max (log‑sum‑exp) for a small
  problem (n≈512),
* progressive up‑sampling of the best shape to larger n,
* several passes of exponentiated‑gradient / sub‑gradient mirror descent,
* a final short simulated‑annealing hill‑climb.

The routine respects a global time budget (default 1000 s) and returns the
best sequence found as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution using a real FFT."""
    L = x.size + y.size - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) assuming ∑seq = 1."""
    n = seq.size
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families of seeds (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed by a Dirichlet draw (conc < 1 makes it sparse)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(seed_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 15000,
                   lr0: float = 0.8,
                   beta_init: float = 5.0,
                   beta_mul: float = 1.02) -> tuple[np.ndarray | None, float]:
    """Adam on a smoothed version of the ``max`` objective."""
    eps = 1e-30
    v = np.log(seed_vec + eps)               # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = beta_init
    best_seq = None
    best_val = float("inf")
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ---- smoothed max (log‑sum‑exp) ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w_k·conv_k w.r.t. a (linear convolution) ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

        # ---- convert to gradient w.r.t. v (softmax Jacobian) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional gradient clipping
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ---- Adam update on v ----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 100000,
                        lr_coeff: float = 0.6) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror‑descent on max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = lr_coeff / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                 # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300000,
                         init_temp_factor: float = 0.02,
                         min_temp: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing with a temperature schedule.
    Randomly selects a few indices and reshuffles their total mass via a
    Dirichlet draw.  Improvements are always accepted; uphill moves are
    accepted with probability exp(−Δ/T).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 6)                # 2–5 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:6d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine runs a multi‑scale
    optimisation (Adam → sub‑gradient → simulated annealing) and respects the
    supplied time‑budget.  The best sequence found is returned as a plain list
    of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Small base problem (fast, many iterations)
    # ------------------------------------------------------------------
    base_n = 512
    print(f"\n=== Optimising base size {base_n} ===")
    seeds = [_uniform(base_n), _sine_window(base_n)]

    # Wide range of beta shapes
    for _ in range(12):
        alpha = 10 ** rng.uniform(-3, 1)   # 0.001 … 10
        beta = 10 ** rng.uniform(-3, 1)
        seeds.append(_beta_shape(base_n, alpha, beta))

    # Power‑law and geometric families
    for p in np.linspace(0.0, 3.0, 7):
        seeds.append(_power(base_n, p))
    for r in np.linspace(0.1, 0.9, 7):
        seeds.append(_geom(base_n, r))

    # Sparse Dirichlet draws
    for conc in (0.01, 0.1, 0.5):
        for _ in range(3):
            seeds.append(_random_dirichlet(base_n, conc, rng))

    # Re‑use any previously discovered construction (down‑sample if needed)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        prev_arr = np.asarray(previous, dtype=np.float64)
        if prev_arr.size > 0:
            factor = max(1, prev_arr.size // base_n)
            down = prev_arr[:factor * base_n].reshape(base_n, factor).mean(axis=1)
            if down.sum() > 0:
                seeds.append(down / down.sum())

    # Quick scan of raw seeds
    global_best_seq = None

        if seq is not None and val < global_best_val:

    # Sub‑gradient refinement on the best base sequence
    if global_best_seq is not None:

    # ------------------------------------------------------------------
    # Progressive up‑sampling to larger problems
    # ------------------------------------------------------------------
    current_seq = global_best_seq
    current_n = base_n
