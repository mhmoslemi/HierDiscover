"""
Improved optimisation of a non‑negative step‑function.

Goal
----
Minimise   2·n·max(conv(a,a)) / (∑a)²

Key ideas
---------
*   The dominant bottleneck in the previous implementation was a
    tiny bug in the smooth‑max approximation: the shift used for the
    log‑sum‑exp was taken from ``a`` instead of the convolution values.
    Using ``max(conv)`` makes the soft‑max numerically stable and
    yields much sharper gradients.
*   Introduce a new analytically‑motivated seed – the *square‑root of the
    uniform polynomial* – which is known to give a rather flat
    autocorrelation.
*   Run a longer Adam phase (with the corrected smooth‑max) and a final
    fine‑tuning Adam pass after the stochastic hill‑climber.
*   Allow a larger problem size (up to 8000 points) – the objective
    improves with finer discretisation, but the time‑budget is respected.
*   Keep the original sub‑gradient and simulated‑annealing phases, but
    add a few extra random restarts for robustness.

The routine is fully time‑aware: each phase stops as soon as the global
deadline (≈1000 s) is reached.  The best sequence discovered (a plain
list of floats) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax producing a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq, seq)) assuming ∑seq == 1.
    This matches ``evaluate_sequence`` for a normalised sequence.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (all normalised to sum == 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _sym_geom(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _random_logits(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    return _softmax(logits)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed by a Dirichlet draw (conc < 1 makes it sparse)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _sqrt_uniform(n: int) -> np.ndarray:
    """
    Coefficients of the power‑series sqrt(1 + z + … + z^{2n‑2}),
    truncated to degree n‑1.
    The recursion:
        a0 = 1
        a_m = (1 - Σ_{i=1}^{m‑1} a_i a_{m‑i}) / 2  for m ≥ 1
    After construction the vector is normalised to sum 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for m in range(1, n):
        s = 0.0
        # Σ_{i=1}^{m‑1} a_i * a_{m‑i}
        for i in range(1, m):
            s += a[i] * a[m - i]
        a[m] = (1.0 - s) / 2.0
        # Numerical safety – clip tiny negatives that may appear due to round‑off
        if a[m] < 0.0:
            a[m] = 0.0
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _make_seeds(n: int, previous: np.ndarray | None, rng: np.random.Generator) -> list[np.ndarray]:
    """Assemble a diverse set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.5, 11):
        seeds.append(_power(n, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # beta‑like shapes
    for a in (0.3, 0.5, 0.7):
        for b in (0.3, 0.5, 0.7):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random soft‑max vectors (logits)
    for _ in range(7):
        seeds.append(_random_logits(n, rng))

    # sparse random patterns
    for k in (3, 5, 8, 12):
        for conc in (0.2, 0.5):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # the new analytic seed based on sqrt of the uniform polynomial
    seeds.append(_sqrt_uniform(n))

    # reuse previously discovered sequence if length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

# Optimisation primitives

def _adam_optimize(seed_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000,
                   beta0: float = 0.5,
                   beta_mul: float = 1.05,
                   lr0: float = 0.9) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a smooth (log‑sum‑exp) approximation of the max‑convolution.
    The smoothing parameter ``beta`` (inverse temperature) is increased
    geometrically each iteration.  The bug present in the reference code
    (using ``max(a)`` for centring) is fixed – we now use ``max(conv)``.
    """
    eps = 1e-30
    v = np.log(seed_vec + eps)           # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = beta0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                     # stays on the simplex, ∑a = 1
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ---- smooth‑max: softmax weights over the convolution vector ----
        conv = _conv_fft(a, a)               # length = 2n‑1
        shift = np.max(conv)                 # *** correct centre ***
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            # numerical glitch – skip this iteration
            beta *= beta_mul
            continue
        w /= w_sum                           # now w is a probability vector (smooth max)

        # ---- gradient of weighted sum conv w.r.t. a (linear convolution) ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)           # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

        # ---- convert to gradient w.r.t. v (softmax Jacobian) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional gradient clipping for stability
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ---- Adam update on pre‑softmax parameters v ----
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** t)
        v2_hat = v2 / (1.0 - 0.999 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

        # increase the smoothness (lower temperature)
        beta = min(beta * beta_mul, 150.0)   # cap to avoid overflow

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 300000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror‑descent on the true max‑conv objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.7 / np.sqrt(it)          # a slightly larger step than the reference
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 400000,
                         init_temp_factor: float = 0.025,
                         min_temp: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing with a geometric cooling schedule.
    Each move reshuffles the mass of a small random subset of indices
    using a Dirichlet proposal.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    # initialise temperature proportional to the current objective
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9993, min_temp)

        k = rng.integers(2, 6)                # 2‑5 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:6d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative heights) that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time‑budget and returns the best sequence found as a list of floats.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Global time‑budget in seconds (default 1000 s).
    """
    # deterministic reproducibility
    np.random.seed(seed)
    start_time = time.time()
    # we leave a small safety margin for the caller to fetch the result
    deadline = start_time + budget_s - 2.0

    # re‑use a previously discovered construction if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # problem sizes – larger n gives a finer discretisation, but is slower.
    # We will dynamically prune the list if the deadline is close.
    lengths = [2000, 4000, 6000, 8000]

    global_best_seq = None
    global_best_val = float("inf")
    rng = np.random.default_rng(seed)

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous, rng)

        # ------------------------------------------------------------------
        # Quick scan of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam on smoothed max
        # ------------------------------------------------------------------
        for s in seeds:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline,
                                      max_iters=18000,
                                      beta0=0.5,
                                      beta_mul=1.05,
                                      lr0=0.85)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement (high‑resolution)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=250000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Simulated‑annealing hill climbing
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _simulated_annealing(global_best_seq, n, deadline,
                                            max_iters=500000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SA] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – Final fine‑tune Adam (high beta, few iterations)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _adam_optimize(global_best_seq, n, deadline,
                                      max_iters=8000,
                                      beta0=10.0,      # start already sharp
                                      beta_mul=1.02,
                                      lr0=0.4)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[final‑Adam] n={n:5d} new global best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Verify the final sequence with the provided evaluate_sequence (optional)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
