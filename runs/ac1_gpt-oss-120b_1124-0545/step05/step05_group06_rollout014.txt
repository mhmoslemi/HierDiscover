def full_convolution(a, fft_len):
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[:2*a.size-1]

def softmax_objects(a, beta, fft_len):
    conv = full_convolution(a, fft_len)
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)   # stable
    sum_exp = np.sum(exp_scaled)
    w = exp_scaled / sum_exp
    # compute gradient

"""
Propose a step‑function (non‑negative simplex vector) that minimises
`evaluate_sequence`.  The search consists of four stages:

1. **Parametric seeding** – generate a small pool of simple analytic
   families (arcsine, Beta, mixtures) and keep the best one.
2. **Smooth projected gradient descent** – minimise a smooth
   approximation of `max(conv(a,a))` obtained by a soft‑max (log‑sum‑exp)
   with a temperature that is slowly increased.  After each gradient
   step the iterate is clipped to non‑negative values and renormalised
   onto the probability simplex; symmetry `a[i]=a[n‑1‑i]` is enforced.
3. **Linear‑programming refinement** – solve a LP that maximises the
   total mass of a vector `g≥0` under the current convolution bound
   `a∗g ≤ M`.  Mixing the current best vector with the normalised LP
   solution often reduces the true `L∞` objective.
4. **Final stochastic annealing** – very cheap random mass transfers
   (simulated‑annealing) are performed until the time budget expires.

All phases respect the 1000 s budget, print intermediate scores and
return the best vector found (as a plain Python list)."""

import time
import numpy as np

def proj_simplex_clip(v: np.ndarray) -> np.ndarray:
    """
    Simple projection onto the probability simplex:
    clip negatives to zero and renormalise to sum = 1.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        return v / s
    # fallback – uniform distribution
    return np.full_like(v, 1.0 / v.shape[0])


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """
    Linear convolution a∗a (full length 2·len(a)−1) via FFT and return
    its maximum value.
    """
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv_full.max())


def evaluate_vec(a: np.ndarray, fft_len: int) -> float:
    """
    Exact value of `evaluate_sequence` for a vector already normalised
    to sum = 1 (the factor 2·n·max(conv) is returned).
    """
    mx = max_convolution(a, fft_len)
    return float(2.0 * a.size * mx)


def arcsine_distribution(n: int) -> np.ndarray:
    """Discretised arcsine density on a uniform grid; sums to 1."""
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric or asymmetric Beta(α,β) density on a grid, normalised."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def softmax_gradient(a: np.ndarray, beta: float, fft_len: int):
    """
    Compute (conv, conv_max, grad) where conv = a∗a, conv_max = max(conv),
    and grad is the gradient of the soft‑max approximation
        (1/β)·log Σ exp(β·conv_k)
    with respect to a.
    """
    n = a.size
    # --- convolution a∗a ---
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    conv_max = float(conv_full.max())

    # --- soft‑max weights ---
    scaled = beta * conv_full
    shift = scaled.max()                # numerical stability
    exp_scaled = np.exp(scaled - shift)
    w = exp_scaled / exp_scaled.sum()   # length 2·n‑1, sums to 1

    # --- gradient = 2·(rev_a ∗ w) restricted to n entries ---
    rev_a = a[::-1]
    rev_f = np.fft.fft(rev_a, fft_len)
    w_f = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(rev_f * w_f))
    # entries indices n‑1 … n‑1+n‑1 correspond to gradient components
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    return conv_full, conv_max, grad


def linear_program_refine(a: np.ndarray, M: float, n: int, max_seconds: float = 5.0):
    """
    Solve the LP
        maximise   Σ g_j
        subject to a∗g ≤ M   (for every shift)
                  g ≥ 0
    Returns the optimal g (or None on failure).
    """
    try:
        from scipy.optimize import linprog
    except Exception:                # scipy not available
        return None

    # Build the (2n‑1)×n constraint matrix:  A[k,i] = a_i  if  k∈[i, i+n‑1]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = a[i]

    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise –Σ g_j  ⇔ maximise Σ g_j
    bounds = [(0, None)] * n

    # Give the solver a modest time budget (the outer routine already
    # guarantees that we have at most a few seconds left).
    options = {"disp": False, "time_limit": max_seconds}
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs", options=options)
    if not res.success:
        return None
    return res.x


def simulated_annealing(a: np.ndarray,
                        best_score: float,
                        fft_len: int,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Very cheap local stochastic refinement that transfers a small amount
    of mass between two randomly chosen coordinates.  Uses a linear
    temperature schedule.
    """
    n = a.size
    best_a = a.copy()
    best_sc = best_score
    start = time.time()
    total_time = deadline - start
    if total_time <= 0:
        return best_a, best_sc

    T_start = best_sc * 0.004
    T_end = best_sc * 1e-6

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = min(1.0, elapsed / total_time)
        T = T_start * (1.0 - frac) + T_end * frac

        # propose elementary move
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_move = min(a[i], 0.05 * a[i] + 1e-12)
        if max_move <= 0.0:
            continue
        delta = rng.random() * max_move

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate proposal
        new_score = evaluate_vec(a_new, fft_len)

        # Metropolis accept/reject
        if new_score < best_sc or rng.random() < np.exp(-(new_score - best_sc) / (T + 1e-12)):
            a = a_new
            if new_score < best_sc:
                best_sc = new_score
                best_a = a_new.copy()
                print(f"[SA {it:07d}] new best = {best_sc:.6f} (t={time.time() - start:.1f}s)")

        it += 1

    return best_a, best_sc

    `evaluate_sequence`.  The routine proceeds through seeding,
    smooth projected gradient descent, an LP refinement, and a final
    stochastic annealing phase.  Progress is printed and the best
    vector found (as a Python list) is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    overall_deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))
    # FFT length must accommodate convolutions up to length 3·n‑2 (gradient step)
    fft_len = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # 0) Build a small pool of simple analytic candidates
    # ------------------------------------------------------------------
    candidates = []

    # 0a) reuse a previously discovered construction if available
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.size == n and a0.sum() > 0.0:
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # 0b) arcsine distribution
    candidates.append(arcsine_distribution(n))

    # 0c) uniform distribution (baseline)
    candidates.append(np.full(n, 1.0 / n))

    # 0d) symmetric Beta families (α = β) – a modest sweep
    for alpha in np.linspace(0.3, 2.0, 12):
        candidates.append(beta_distribution(n, alpha, alpha))

    # 0e) mixtures of arcsine and uniform
    arcsine = candidates[1]
    uniform = candidates[2]
    for lam in np.linspace(0.0, 1.0, 11):
        mix = lam * arcsine + (1.0 - lam) * uniform
        mix = np.maximum(mix, 0.0)
        mix = mix / mix.sum()
        candidates.append(mix)

    # ------------------------------------------------------------------
    # 1) Pick the best among cheap candidates
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    for idx, a in enumerate(candidates):
        sc = evaluate_vec(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[init {idx:02d}] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Smooth projected gradient descent (soft‑max approximation)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.60 * budget_s
    a = best_a.copy()
    it = 0
    momentum = np.zeros_like(a)

    # temperature schedule (soft‑max β) – exponential increase
    beta_start = 1.0
    beta_end = 5000.0

    last_improvement = time.time()
    while time.time() < phase2_deadline:
        it += 1
        fraction = (time.time() - start_time) / (phase2_deadline - start_time)
        beta = beta_start * (beta_end / beta_start) ** fraction

        # gradient of smooth surrogate
        conv_full, conv_max, grad = softmax_gradient(a, beta, fft_len)

        # step size (standard 1/√t schedule)
        lr = 0.5 / np.sqrt(it)

        # classic momentum update
        momentum = 0.9 * momentum - lr * grad

        a_candidate = a + momentum
        a_candidate = proj_simplex_clip(a_candidate)

        # enforce symmetry (a[i] = a[n‑1‑i])
        a_candidate = (a_candidate + a_candidate[::-1]) * 0.5
        a = a_candidate

        cur_score = evaluate_vec(a, fft_len)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improvement = time.time()
            print(f"[GD {it:07d}] improved to {best_score:.6f}")

        # occasional restart if stagnating for a while
        if time.time() - last_improvement > 25.0:
            noise = rng.normal(scale=0.02, size=n)
            a = proj_simplex_clip(best_a + noise)
            a = (a + a[::-1]) * 0.5
            momentum.fill(0.0)
            last_improvement = time.time()
            print(f"[GD {it:07d}] restart after stagnation")

    # ------------------------------------------------------------------
    # 3) Linear‑programming refinement (single step)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 10.0:
        M_current = best_score / (2.0 * n)          # current max (conv)
        g = linear_program_refine(best_a, M_current, n,
                                  max_seconds=min(4.0, remaining / 2.0))
        if g is not None and g.sum() > 1e-12:
            g_norm = g / g.sum()
            # Mix a bit of g into the current best vector
            a_mix = 0.5 * best_a + 0.5 * g_norm
            a_mix = proj_simplex_clip(a_mix)
            a_mix = (a_mix + a_mix[::-1]) * 0.5
            mix_score = evaluate_vec(a_mix, fft_len)
            if mix_score < best_score:
                best_score = mix_score
                best_a = a_mix.copy()
                print(f"[LP] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Local stochastic refinement (simulated annealing)
    # ------------------------------------------------------------------
    if overall_deadline - time.time() > 5.0:
        best_a, best_score = simulated_annealing(best_a,
                                                 best_score,
                                                 fft_len,
                                                 overall_deadline,
                                                 rng)

    # ------------------------------------------------------------------
    # Finalisation
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / s

    elapsed_total = time.time() - start_time
    print(f"[done] total time = {elapsed_total:.1f}s, best score = {best_score:.6f}, length = {n}")
    return best_a.tolist()
