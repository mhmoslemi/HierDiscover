def linear_program_step(f, M):
    n = f.shape[0]
    # Build A matrix
    # Use sp.diags with each column j representing constant f_i across rows offset i?
    # Actually A[k,j] = f_{k-j} if 0<=k-j<n else 0.
    # This matches diagonal offset i = (k - j) = i, coefficient f_i.
    A = sp.diags([np.full(n, val) for val in f], offsets=range(n), shape=(2*n-1,n), format='csc')
    # constraints be A @ g <= M
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum(g) => minimize -sum(g)
    bounds = [(0, None)]*n
    solution = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if solution.success:
        g = solution.x
    else:
        g = np.zeros(n)
    return g

    M = np.max(conv_fft_self(f))
    # Build constraint matrix

    offsets = np.arange(n)
    A = sp.diags(data, offsets, shape=(2*n-1, n), format='csc')

    c = -np.ones(n)  # maximize sum(g)

    g = res.x

def directed_move(a, scale):
    # compute subgrad for max conv
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    active = (conv >= max_val - 1e-12).astype(np.float64)
    cnt = active.sum()
    if cnt == 0:
        return a
    w = active / cnt
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]  # gradient

    # Identify indices
    i_max = np.argmax(grad)
    i_min = np.argmin(grad)
    # Transfer some mass
    delta = scale * a[i_max] * np.random.rand()
    a_new = a.copy()
    a_new[i_max] = max(0.0, a_new[i_max] - delta)
    a_new[i_min] = a_new[i_min] + delta
    # Project onto simplex
    return project_simplex(a_new)

"""
Hybrid optimizer for the step‑function objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The search proceeds in four stages

1. **Analytic sweep** – evaluate several parametric families (product,
   sum, exponential, Gaussian and a simple triangular shape) on a fine
   grid and keep the best candidate.

2. **Linear‑programming mixing** – given a current distribution `f`,
   solve the LP

       max   Σ_j g_j
       s.t.  (f * g)_k ≤ M   for all k
             g_j ≥ 0

   where `M = max(f*f)`.  The solution is rescaled to the probability
   simplex and mixed with `f` (small convex step).  Repeating a few
   times rapidly flattens the autocorrelation.

3. **Smooth‑max Adam** – minimise a smooth approximation of the max
   (soft‑max with growing λ) using Adam on the log‑probability vector.
   The continuation on λ drives the solution close to a true local
   minimum of the original non‑smooth objective.

4. **Exact sub‑gradient + directed local search** – a projected
   sub‑gradient descent on the exact max‑convolution, augmented with a
   cheap “move mass from the most harmful index to the least harmful”
   heuristic (gradient‑guided random walk).  The loop runs until the
   time budget expires.

The routine respects the supplied `budget_s` (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a (the sum is ignored;
    the function normalises internally).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def directed_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Gradient‑guided mass transfer: move a fraction of the mass from the
    index that contributes most to the current maximal convolution to
    the index that contributes the least.
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    active = (conv >= max_val - 1e-12).astype(np.float64)
    cnt = active.sum()
    if cnt == 0:
        return a
    w = active / cnt
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]      # sub‑gradient of max(conv) w.r.t. a

    i_max = np.argmax(grad)
    i_min = np.argmin(grad)
    if a[i_max] <= 0:
        return a
    delta = scale * a[i_max] * np.random.rand()
    a_new = a.copy()
    a_new[i_max] = max(0.0, a_new[i_max] - delta)
    a_new[i_min] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^(‑p)   (edge‑heavy shape)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^(‑p) + (n‑1‑i+off)^(‑p)   (two tails)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exp_decay(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(‑c·|i‑center|)   (center‑peaked shape)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - center))
    a = a / a.sum()
    return a


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(‑0.5·((i‑center)/σ)²)   (smooth bump)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = a / a.sum()
    return a


def analytic_triangle(n: int) -> np.ndarray:
    """Symmetric triangular shape (peak in the middle)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 - np.abs(i - center) / center
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth‑max utilities
# ----------------------------------------------------------------------


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    Smooth max Φ_λ and its gradient w.r.t. the log‑probability vector x.
    """
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    conv = conv_fft_self(a)                # length 2n‑1
    max_c = np.max(conv)

    shift = conv - max_c                   # ≤ 0
    exp_term = np.exp(lam * shift)        # ≤ 1
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam    # Φ_λ

    w = exp_term / sum_exp                 # weights, sum to 1
    w_rev = w[::-1]

    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad_a = 2.0 * slice_part[::-1]            # ∂Φ/∂a

    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)                # ∇_x Φ = J^T ∇_a Φ
    return phi, grad_x


def adam_optimize(x: np.ndarray, lam: float, n: int,
                  steps: int, lr: float,
                  deadline: float) -> np.ndarray:
    """Simple Adam optimiser for the smooth‑max objective."""
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for t in range(1, steps + 1):
        if time.time() >= deadline:
            break
        _, grad = compute_phi_and_grad(x, lam, n)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step
    return x


# ----------------------------------------------------------------------
# Linear‑programming mixing (LP step)
# ----------------------------------------------------------------------


def lp_mix_step(f: np.ndarray, step_alpha: float) -> np.ndarray:
    """
    One LP‑based mixing step.
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M   for all k,   g ≥ 0.
    The solution g is rescaled to the simplex and mixed with f:
        f ← (1‑α) f + α g.
    """
    n = f.shape[0]
    M = np.max(conv_fft_self(f))

    # Build the convolution constraint matrix:
    # (A @ g)[k] = Σ_i f_i * g_{k‑i}   (zero‑padding outside the range)
    data = [np.full(n, v, dtype=np.float64) for v in f]
    offsets = np.arange(n, dtype=np.int64)
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')

    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)               # maximise Σ g_j  ⇔ minimise –Σ g_j
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success or np.allclose(res.x, 0):
        # fallback – keep the current distribution
        return f

    g = res.x
    g_sum = g.sum()
    if g_sum == 0:
        return f
    g = g / g_sum                     # normalise to a probability vector
    f_new = (1.0 - step_alpha) * f + step_alpha * g
    return project_simplex(f_new)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max(conv))
# ----------------------------------------------------------------------


def subgrad_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step for the exact max‑convolution.
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    active = (conv >= max_val - 1e-12).astype(np.float64)   # binary mask
    cnt = active.sum()
    if cnt == 0:
        return a
    w = active / cnt
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]            # sub‑gradient w.r.t. a
    a_new = a - step * grad
    return project_simplex(a_new)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The function respects the time budget and returns
    the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Analytic grid search – deterministic baseline
    # ------------------------------------------------------------------
    best_a = np.ones(n) / n
    best_score = compute_score(best_a)

    # use the global construction, if supplied
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n and ref.sum() > 0:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            if sc < best_score:
                best_score, best_a = sc, ref.copy()

    # parameter grids
    p_vals = np.linspace(0.30, 0.70, 81)
    off_vals = np.linspace(0.0001, 2.0, 81)
    c_vals = np.linspace(0.0005, 0.0200, 41)
    sigma_vals = np.linspace(100.0, 800.0, 41)

    # product‑law / sum‑law families
    for mode in ("product", "sum"):
        for p in p_vals:
            for off in off_vals:
                if time.time() >= deadline:
                    break
                if mode == "product":
                    cand = analytic_product(n, p, off)
                else:
                    cand = analytic_sum(n, p, off)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score, best_a = sc, cand.copy()

    # exponential decay
    for c in c_vals:
        if time.time() >= deadline:
            break
        cand = analytic_exp_decay(n, c)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    # Gaussian bump
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        cand = analytic_gaussian(n, sigma)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    # simple triangular shape (often competitive)
    cand = analytic_triangle(n)
    sc = compute_score(cand)
    if sc < best_score:
        best_score, best_a = sc, cand.copy()

    print(f"[grid] best score after analytic sweep = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) LP‑based mixing – a few aggressive flattening steps
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lp_iters = 12                      # modest number – cheap but powerful
        for it in range(lp_iters):
            if time.time() >= deadline:
                break
            step_alpha = max(0.05, 0.25 / (it + 1))   # decreasing step size
            best_a = lp_mix_step(best_a, step_alpha)
            sc = compute_score(best_a)
            if sc < best_score:
                best_score, best_a = sc, best_a.copy()
                print(f"[LP iter={it}] new best = {best_score:.6f} (t={time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # 3) Smooth‑max optimisation with Adam (λ‑continuation)
    # ------------------------------------------------------------------
    eps = 1e-12
    x = np.log(np.maximum(best_a, eps))

    lam_vals = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0,
                2000.0, 8000.0, 20000.0, 80000.0, 200000.0]

    for lam in lam_vals:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # Rough cost estimate: one Adam step ≈ 0.002 s for n≈2000
        max_steps = int(remaining / 0.002)

        if lam >= 2000:
            steps = 4000
        if lam >= 20000:
            steps = 6000
        steps = min(steps, max_steps)
        if steps <= 0:

        x = adam_optimize(x, lam, n, steps, lr=0.06, deadline=deadline)
        a = softmax(x)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[Adam λ={lam:.0f}] new best = {best_score:.6f} (t={time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # 4) Exact sub‑gradient descent + directed local search
    # ------------------------------------------------------------------
    a = best_a.copy()

    # ------------------------------------------------------------------
    # 5) Light random‑mass‑transfer / directed search (final polishing)
    # ------------------------------------------------------------------
    a = best_a.copy()
    scale = 0.02
    while time.time() < deadline:
        # 50 % probability for the gradient‑directed move
        if np.random.rand() < 0.5:
            cand = directed_move(a, scale)
        else:

            a = cand
            scale = max(0.001, scale * 0.9)        # tighten after improvement
        else:
            scale = min(0.05, scale * 1.001)       # slowly relax
            a = cand

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    return best_a.tolist()
