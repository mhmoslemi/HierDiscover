    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    if n is None: n = int(kwargs.get('n', 2000))

    # Phase 1: analytic grid search
    best_seq, top = analytic_grid_search(n)  # will get top 10
    # refine top around best (we might re-sample param families with fine limit)
    # Actually we can also add random Dirichlet seeds
    for _ in range(30):
        rand = np.random.gamma(1.0, 1.0, n)
        rand /= rand.sum()
        _add_candidate(rand, top, max_len=20)
    
    # sort top, pick top K
    top.sort(key=lambda x: x[0])
    K = min(15, len(top))
    population = [cand for (_, cand) in top[:K]]

    best_seq = population[0].copy()
    best_score = compute_score(best_seq)

    # Phase 2: evolutionary SA
    # schedule for temperature: start T0 ~ 0.02, end Tf ~ 1e-6
    # use iteration time loop until about 300s left for final phases.

    # Determine time marks:
    # let t_phase2_end = start + 0.45*budget_s (approx 450s)
    # let t_phase3_end = start + 0.85*budget_s (approx 850s)
    # t_phase4_end = deadline

    while time.time() < t_phase2_end:
        # Choose random parent
        i = np.random.randint(len(population))
        parent = population[i]
        # decide mutation type
        if np.random.rand() < 0.5:
            child = random_move(parent, scale=0.2) # scale fraction of mass
        else:
            # Blend with another random parent
            j = np.random.randint(len(population))
            while j == i:
                j = np.random.randint(len(population))
            alpha = np.random.rand()
            child = alpha * parent + (1-alpha) * population[j]
            child = project_simplex(child)
            # add small Gaussian
            child += np.random.normal(scale=0.001, size=n)
            child = np.clip(child, 0, None)
            child = project_simplex(child)

        child_score = compute_score(child)
        # Accept if better or with prob exp(-(child-score-best_score)/T)
        T = max(1e-6, (t_phase2_end - time.time()) / (t_phase2_end - start)) * 0.02
        if child_score < best_score:
            best_score = child_score
            best_seq = child.copy()
            # replace parent
            population[i] = child
        else:
            # uphill acceptance
            if np.random.rand() < np.exp(-(child_score - best_score)/T):
                population[i] = child

        # occasionally replace worst in population with global best mutated
        if np.random.rand() < 0.01:
            idx_worst = np.argmax([compute_score(v) for v in population])
            mutated = random_move(best_seq, scale=0.1)
            population[idx_worst] = mutated

    # Phase 3: Adam on softmax
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:

            best_seq = a.copy()

        progress = (time.time() - start) / (budget_s - 20.0)
        beta = 0.5 + (400.0 - 0.5) * progress  # as before but maybe extend to 800

        weighted_shift = conv_two(w, a_rev)[n-1:n-1+n]
        g = a + weighted_shift

        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t_adam)
        v_hat = v / (1 - beta2**t_adam)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

            best_seq = cand.copy()

"""
Propose a non‑negative step function that (approximately) minimizes

        2·n·max(conv(a,a)) / (sum(a)**2)

The method consists of four stages, each respecting the wall‑time budget:

0. Analytic grid search over several parametric families (including refined
   sweeps of the classic families and two new families).  The best few
   candidates are kept and enriched with random Dirichlet samples.

1. A simple evolutionary / simulated‑annealing loop that maintains a small
   population of high‑scoring vectors.  Mutations are mass‑transfer moves
   and convex blends of two parents; occasional uphill moves are accepted
   with a temperature schedule.

2. A soft‑max (log‑sum‑exp) optimisation of the exact max‑convolution using
   Adam (first‑order momentum).  The temperature is raised from a smooth
   regime (β≈0.5) to a very sharp one (β≈400) while the learning rate is
   kept adaptive.  The current best vector is continuously updated.

3. A final short random‑perturbation phase that tries Gaussian jitter around
   the best vector found so far.

All vectors are kept on the probability simplex (∑a_i = 1, a_i ≥ 0) by an
Euclidean projection, therefore the objective becomes
    score(a) = 2·n·max(conv_self(a))
which is exactly what ``evaluate_sequence`` would compute after the
normalisation step.  The routine respects the 1000 s wall‑time limit and
returns the best sequence discovered.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of a and b using FFT (real‑valued)."""
    size = _next_pow2(a.shape[0] + b.shape[0] - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum = 1.
    Returns 2·n·max(conv_self(a)).
    """
    # safety – avoid division by (almost) zero
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # a is assumed to be already on the simplex; normalise tiny
    a = a / s
    conv = conv_self(a)
    return float(2.0 * a.shape[0] * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    return a / a.sum()


def analytic_cosine(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    return a / a.sum()


def analytic_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta_
    return a / a.sum()


def analytic_logistic(n: int, steep: float, mid: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(steep * (i - mid)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def analytic_double_exp(n: int, lam_l: float, lam_r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.where(i <= mid,
                 np.exp(-lam_l * (mid - i)),
                 np.exp(-lam_r * (i - mid)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _add_candidate(cand: np.ndarray,
                   best_list: list[tuple[float, np.ndarray]],
                   max_len: int = 20) -> None:
    """Insert a candidate into a sorted best‑list (lower score is better)."""
    sc = compute_score(cand)
    inserted = False
    for idx, (s, _) in enumerate(best_list):
        if sc < s:
            best_list.insert(idx, (sc, cand.copy()))
            inserted = True
            break
    if not inserted and len(best_list) < max_len:
        best_list.append((sc, cand.copy()))
    # keep only the best ``max_len`` entries
    while len(best_list) > max_len:
        best_list.pop()


def analytic_grid_search(n: int) -> tuple[np.ndarray, list[tuple[float, np.ndarray]]]:
    """
    Exhaustively scan a set of analytic families (including finer grids).
    Returns the overall best vector and a list of the top candidates.
    """
    top: list[tuple[float, np.ndarray]] = []

    # ----- Product family -------------------------------------------------
    p_vals = np.arange(0.1, 3.01, 0.1)               # finer than before
    off_vals = np.concatenate((
        np.geomspace(1e-4, 0.02, 6),
        np.arange(0.02, 5.01, 0.2)
    ))
    for p in p_vals:
        for off in off_vals:
            cand = analytic_product(n, p, off)
            _add_candidate(cand, top)

    # ----- Sum family ----------------------------------------------------
    for p in p_vals:
        for off in off_vals:
            cand = analytic_sum(n, p, off)
            _add_candidate(cand, top)

    # ----- Exponential tails ---------------------------------------------
    lam_vals = np.logspace(-3, 1, num=30)
    for lam in lam_vals:
        cand = analytic_exponential(n, lam)
        _add_candidate(cand, top)

    # ----- Cosine‑power ---------------------------------------------------
    p_cos = np.arange(0.5, 8.1, 0.4)
    for p in p_cos:
        cand = analytic_cosine(n, p)
        _add_candidate(cand, top)

    # ----- Beta family ----------------------------------------------------
    alpha_vals = np.arange(-0.5, 2.51, 0.5)
    beta_vals = np.arange(-0.5, 2.51, 0.5)
    for a in alpha_vals:
        for b in beta_vals:
            cand = analytic_beta(n, a, b)
            _add_candidate(cand, top)

    # ----- Logistic family ------------------------------------------------
    steep_vals = np.linspace(0.01, 10.0, 25)          # positive -> decreasing
    mid_vals = np.linspace(0.25 * n, 0.75 * n, 15)
    for steep in steep_vals:
        for mid in mid_vals:
            cand = analytic_logistic(n, steep, mid)
            _add_candidate(cand, top)

    # ----- Double‑exponential (asymmetric) -------------------------------
    lam_l_vals = np.geomspace(1e-3, 5.0, 12)
    lam_r_vals = np.geomspace(1e-3, 5.0, 12)
    for lam_l in lam_l_vals:

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The algorithm proceeds through several
    stages (analytic families → evolutionary SA → Adam on a softened max
    → final jitter) while respecting the wall‑time limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 0) Analytic grid + random Dirichlet seeds
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))
    print(f"[0] Analytic grid (n={n})")
    best_seq, top10 = analytic_grid_search(n)

    # add a few random Dirichlet starters to the top‑list
    for _ in range(30):
        rnd = np.random.gamma(1.0, 1.0, size=n)
        rnd /= rnd.sum()
        _add_candidate(rnd, top10, max_len=30)

    # keep the best few as an initial population
    top10.sort(key=lambda x: x[0])
    pop_size = min(15, len(top10))
    population = [vec.copy() for (_, vec) in top10[:pop_size]]
    best_seq = population[0].copy()
    best_score = compute_score(best_seq)
    print(f"[0] initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Evolutionary / simulated‑annealing loop
    # ------------------------------------------------------------------
    # allocate ~45 % of the budget to this phase
    phase2_end = start_time + 0.45 * budget_s

    # temperature schedule: linear decay from T0 to Tf (tiny)
    T0 = 0.02
    Tf = 1e-6

    it = 0
    while time.time() < phase2_end:
        it += 1
        # pick a parent at random
        i = np.random.randint(pop_size)
        parent = population[i]

        # decide on a mutation strategy
        if np.random.rand() < 0.55:                     # pure mass‑transfer
            child = random_move(parent, scale=0.25)
        else:                                          # convex blend + jitter
            j = np.random.randint(pop_size)
            while j == i:
                j = np.random.randint(pop_size)
            alpha = np.random.rand()
            child = alpha * parent + (1.0 - alpha) * population[j]
            # small Gaussian noise
            child += np.random.normal(scale=5e-4, size=n)
            child = np.clip(child, 0.0, None)
            child = project_simplex(child)

        child_score = compute_score(child)

        # temperature based on remaining time
        remaining = max(phase2_end - time.time(), 0.0)
        total = max(phase2_end - start_time, 1e-9)
        T = T0 * (remaining / total) + Tf * (1 - remaining / total)

        improved = False
        if child_score < best_score - 1e-12:
            best_score = child_score
            best_seq = child.copy()
            improved = True
            # replace the parent with the better child
            population[i] = child.copy()
        else:
            # occasional uphill acceptance (simulated annealing)
            if np.random.rand() < math.exp(-(child_score - best_score) / T):
                population[i] = child.copy()

        # replace the worst individual now and then with a jittered best
        if np.random.rand() < 0.02:
            worst_idx = np.argmax([compute_score(v) for v in population])
            population[worst_idx] = random_move(best_seq, scale=0.12)

        if improved:
            print(f"[evo {it:07d}] new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam optimisation on a softened max (log‑sum‑exp)
    # ------------------------------------------------------------------
    # keep ~35 % of the budget for Adam
    phase3_end = start_time + 0.85 * budget_s

    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_adam = 0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12
    # temperature schedule for the log‑sum‑exp
    beta_start, beta_end = 0.5, 400.0
    last_improve_time = time.time()
    stagnation_limit = 20.0

    while time.time() < phase3_end:
        conv = conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve_time = time.time()
            print(f"[adam {t_adam:07d}] new best = {best_score:.6f}")

        # temperature rising linearly with overall progress
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 20.0)
        beta = beta_start * (1.0 - progress) + beta_end * progress

        # soft‑max weights over all convolution entries
        w_raw = np.exp(beta * (conv - max_val))      # max exponent = 0
        w = w_raw / w_raw.sum()

        # weighted shift = Σ_k w_k * a_{k-i}
        a_rev = a[::-1]
        corr_full = conv_two(w, a_rev)
        weighted_shift = corr_full[n - 1:n - 1 + n]

        # gradient of the smooth objective
        g = a + weighted_shift

        # Adam update
        t_adam += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step_vec
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate = project_simplex(a_candidate)

        # occasional random mass‑transfer move
        if np.random.rand() < 0.015:
            a_candidate = random_move(a_candidate, lr * 0.4)

        # occasional pure jitter
        if np.random.rand() < 0.005:
            a_candidate = a_candidate + np.random.randn(n) * 5e-4
            a_candidate = np.clip(a_candidate, 0.0, None)
            a_candidate = project_simplex(a_candidate)

        # stagnation handling – jitter around the best
        if time.time() - last_improve_time > stagnation_limit:
            jitter = np.random.rand(n) * 1e-3
            a_candidate = best_seq + jitter
            a_candidate = project_simplex(a_candidate)
            last_improve_time = time.time()
            print(f"[adam {t_adam:07d}] jitter restart")

        a = a_candidate

    # ------------------------------------------------------------------
    # 3) Fine random perturbation phase (last few seconds)
    # ------------------------------------------------------------------
    print("[refine] final random perturbation phase")
    while time.time() < deadline:
        cand = best_seq + np.random.normal(scale=5e-4, size=n)
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, length = {n}, time = {total_time:.1f}s")
    return best_seq.tolist()
