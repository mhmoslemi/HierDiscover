"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm (≈1000 s budget)
------------------------------------------------------------
1. Analytic sweep + a few random Dirichlet starts.
   – product, sum, exponential‑decay and Gaussian families.
   – dense grids for the shape parameters.
   – keep the best candidate.

2. Smoothed‑max optimisation.
   – work in the log‑probability (softmax) space so the simplex
     constraints are satisfied automatically.
   – continuation on the smoothing parameter λ from 0.5 up to 2·10⁵.
   – Adam optimiser with a learning‑rate that shrinks with λ and a
     generous number of iterations (the budget is split dynamically).

3. Exact sub‑gradient descent with momentum (projected onto the simplex).
   – the true objective max(conv) is used; the active set (indices at
     the current maximum) supplies the sub‑gradient.
   – a simple momentum term accelerates the noisy sub‑gradient steps.

4. Simulated‑annealing random‑mass‑transfer.
   – repeatedly pick two indices i≠j and move a tiny amount of probability
     from i to j (projected onto the simplex).
   – accept worsening moves with probability exp(−Δ/τ) where the temperature τ
     decays exponentially.
   – the scale of the transferred mass is adapted on the fly.

All stages respect the global time budget; the best sequence found
before the deadline is returned as a plain Python list.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    exp_x = np.exp(x - m)
    return exp_x / exp_x.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a (sum is ignored –
    the function normalises internally).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a tiny amount of probability mass from one index to another,
    then project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """
    a_i ∝ ((i+off)*(n‑1‑i+off))^(‑p)   (edge‑heavy shape).
    """
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """
    a_i ∝ (i+off)^(‑p) + (n‑1‑i+off)^(‑p)   (two tails).
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exp_decay(n: int, c: float) -> np.ndarray:
    """
    a_i ∝ exp(‑c·|i‑center|)   (center‑peaked shape).
    """
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - center))
    a = a / a.sum()
    return a


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    """
    a_i ∝ exp(‑0.5·((i‑center)/σ)²)   (smooth bump).
    """
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max utilities (Φ_λ and its gradient)
# ----------------------------------------------------------------------


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    For a given log‑probability vector x, smoothing λ and length n,
    return the smooth max Φ_λ(a) and its gradient w.r.t. x.
    a = softmax(x)
    """
    # soft‑max
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / exp_x.sum()

    # convolution a*a
    conv = conv_fft_self(a)                # length 2n‑1
    max_c = np.max(conv)

    # smooth max over the convolution entries
    shift = conv - max_c                   # ≤ 0
    exp_term = np.exp(lam * shift)        # ≤ 1
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam    # Φ_λ

    w = exp_term / sum_exp                 # weights, sum to 1
    w_rev = w[::-1]

    # gradient of Φ_λ w.r.t. a
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = rfft(w_rev, n=size)
    fft_a = rfft(a, n=size)
    conv_wa = irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]      # length n
    grad_a = 2.0 * slice_part[::-1]            # ∂Φ/∂a

    # map to x‑space (soft‑max Jacobian)
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return phi, grad_x


def adam_optimize(x: np.ndarray, lam: float, n: int,
                  steps: int, lr: float, deadline: float) -> np.ndarray:
    """
    Simple Adam optimiser for the smooth‑max objective.
    Stops early if the global deadline is reached.
    """
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for t in range(1, steps + 1):
        if time.time() >= deadline:
            break
        _, grad = compute_phi_and_grad(x, lam, n)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step
    return x


# ----------------------------------------------------------------------
# Exact sub‑gradient descent with momentum
# ----------------------------------------------------------------------


def subgrad_step_with_momentum(a: np.ndarray, v: np.ndarray,
                               beta: float, step_coef: float,
                               t_iter: int) -> tuple[np.ndarray, np.ndarray]:
    """
    One projected sub‑gradient step on the exact max‑convolution objective.
    Returns the new vector a and the updated momentum v.
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    active = (conv >= max_val - 1e-12).astype(np.float64)   # binary mask
    cnt = active.sum()
    if cnt == 0:
        return a, v
    w = active / cnt
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = rfft(w_rev, n=size)
    fft_a = rfft(a, n=size)
    conv_wa = irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]            # sub‑gradient wrt a

    # classic momentum update
    v_new = beta * v + (1.0 - beta) * grad
    step = step_coef / np.sqrt(t_iter)
    a_new = a - step * v_new
    a_proj = project_simplex(a_new)
    return a_proj, v_new

    # leave the remaining ~30 % of the budget for this stage
    remaining = global_deadline - time.time()
    subgrad_deadline = time.time() + 0.30 * remaining

    while time.time() < subgrad_deadline:
        t_iter += 1
        # step coefficient (tuned empirically)
        step_coef = 0.45
        a, v = subgrad_step_with_momentum(a, v, beta, step_coef, t_iter)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[subgrad‑mom iter={t_iter}] new best = {best_score:.6f} "
                  f"(t={time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing random‑mass‑transfer local search
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_score = compute_score(a)
    scale = 0.02                    # typical transferred fraction
    T = 0.012                       # initial temperature
    decay = 0.9999                  # exponential cooling

    while time.time() < global_deadline:
        cand = random_move(a, scale)
        cand_score = compute_score(cand)

        if cand_score < cur_score:
            # always accept improvements
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score, best_a = cand_score, cand.copy()
                print(f"[anneal] new best = {best_score:.6f} "
                      f"(t={time.time() - start_time:.1f}s)")
        else:
            # Metropolis‑type acceptance for worsening moves
            if np.random.rand() < np.exp(-(cand_score - cur_score) / T):
                a = cand
                cur_score = cand_score

        # adapt temperature and move scale
        T *= decay
        if cur_score < best_score:
            # when we are improving, shrink the scale a bit faster
            scale = max(0.0005, scale * 0.995)
        else:
            # otherwise keep exploring more broadly
            scale = min(0.05, scale * 1.001)

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    # return plain Python list as required by the external harness
    return best_a.tolist()
