def full_convolution_fft(x, y, fft_len):
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    Z = X * Y

"""
Propose a non‑negative step function that minimises the
`evaluate_sequence` objective.

The search proceeds in three stages:

1. **Broad parametric sweep** – generate a library of simple
   families (arcsine, uniform, triangular, many Beta densities and
   mixtures).  The best few candidates are retained.

2. **Smooth Adam optimisation** – minimise a smooth surrogate of the
   L∞‑norm, namely
       G(a)= (1/λ)·log∑ exp(λ·(a*a)),
   whose gradient can be written in closed‑form using an extra
   convolution.  An Adam optimiser with Euclidean projection onto the
   probability simplex (∑a=1, a≥0) is run for the allotted time.  The
   temperature parameter λ is increased from a small value (≈5) to a
   larger one (≈50) so that the method gradually focuses on the true
   maximum.

3. **Final refinement** – a short projected sub‑gradient descent
   (exact max‑convolution gradient) followed by a lightweight
   simulated‑annealing style mass‑transfer local search.

The routine respects the 1000 s budget, prints intermediate scores
and finally returns the best vector found (as a Python list).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                 # all zero – fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def real_convolution(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT of length `fft_len`."""
    A = np.fft.rfft(a, fft_len)
    B = np.fft.rfft(b, fft_len)
    C = A * B
    conv = np.fft.irfft(C, fft_len)
    return conv[: a.size + b.size - 1]


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the linear self‑convolution of `a`."""
    conv = real_convolution(a, a, fft_len)
    return float(conv.max())


def conv_full(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution (no slicing)."""
    return real_convolution(a, b, fft_len)


def arcsine_distribution(n: int) -> np.ndarray:
    """Discretised arcsine density on [0,1] (exact integration per bin)."""
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric or asymmetric Beta(α,β) density evaluated at bin centre."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangular shape peaking at the centre."""
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    a = (center + 1) - np.abs(i - center)
    a = np.maximum(a, 0.0)
    total = a.sum()
    return a / total


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Stable soft‑max on the convolution values, scaled by λ."""
    c = conv - conv.max()                 # shift for numerical stability
    ex = np.exp(lam * c)
    return ex / ex.sum()


def grad_from_softmax(a: np.ndarray, p: np.ndarray,
                     rev_fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed objective.
    g = 2 * (a_rev * p) slice_{n-1..n-1+n-1}.
    """
    a_rev = a[::-1]
    full = real_convolution(a_rev, p, rev_fft_len)   # length = 3n‑2
    start = a.size - 1
    return 2.0 * full[start:start + a.size]


def evaluate_obj(a: np.ndarray, conv_fft_len: int) -> float:
    """Exact `evaluate_sequence` score for a normalised vector (sum=1)."""
    mx = max_convolution(a, conv_fft_len)
    return 2.0 * a.size * mx


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The algorithm follows the three‑stage plan
    described in the module docstring.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 2500))                  # default length
    conv_fft_len = 1 << ((2 * n - 1).bit_length())  # for a*a
    grad_fft_len = 1 << ((3 * n - 2).bit_length())  # for a_rev * p

    # --------------------------------------------------------------
    # 0) build a rich pool of cheap candidates
    # --------------------------------------------------------------
    cand = []

    # reuse a known construction if present
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.size == n and a0.sum() > 0.0:
            a0 = a0 / a0.sum()
            cand.append(a0)

    # basic families
    cand.append(arcsine_distribution(n))
    cand.append(np.full(n, 1.0 / n))                # uniform
    cand.append(triangular_distribution(n))

    # symmetric beta sweep
    for a in np.linspace(0.25, 2.0, 16):
        cand.append(beta_distribution(n, a, a))

    # asymmetric beta sweep (coarse grid)
    alphas = np.linspace(0.25, 2.0, 6)
    betas = np.linspace(0.25, 2.0, 6)
    for alpha in alphas:
        for beta in betas:
            if abs(alpha - beta) > 1e-12:          # skip the symmetric ones already added
                cand.append(beta_distribution(n, alpha, beta))

    # mixtures of arcsine with every beta already in the pool (including asym.)
    arcs = cand[1]  # arcsine
    for other in cand[2:]:
        for lam in np.linspace(0.0, 1.0, 7):
            mix = lam * arcs + (1.0 - lam) * other
            mix = np.maximum(mix, 0.0)
            mix_sum = mix.sum()
            if mix_sum > 0.0:
                cand.append(mix / mix_sum)

    # --------------------------------------------------------------
    # 1) cheap pre‑screen – keep the best few vectors
    # --------------------------------------------------------------
    best_score = float("inf")
    best_vec = None
    scores = []
    for idx, a in enumerate(cand):
        sc = evaluate_obj(a, conv_fft_len)
        scores.append(sc)
        if sc < best_score:
            best_score = sc
            best_vec = a.copy()
        print(f"[init {idx:03d}] score = {sc:.6f}")

    # keep the top‑K promising starters
    K = 5
    top_idx = np.argsort(scores)[:K]
    starters = [cand[i].copy() for i in top_idx]

    # --------------------------------------------------------------
    # 2) Smooth Adam optimisation (soft‑max surrogate)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.55 * budget_s
    lam_start, lam_end = 5.0, 50.0               # anneal softness
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    base_lr = 0.25                               # base learning‑rate for Adam

    for start_a in starters:
        if time.time() >= phase1_deadline:
            break

        a = proj_simplex(start_a)                # ensure exact simplex
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        it = 0
        last_imp = time.time()

        while time.time() < phase1_deadline:
            it += 1
            # full self‑convolution (needed for the surrogate)
            conv = real_convolution(a, a, conv_fft_len)[: 2 * n - 1]

            # update λ linearly over the first 4000 iterations
            lam = lam_start + (lam_end - lam_start) * min(1.0, it / 4000.0)

            # soft‑max weights over the convolution entries
            p = softmax_weights(conv, lam)

            # gradient of the smooth surrogate
            g = grad_from_softmax(a, p, grad_fft_len)

            # Adam moments
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

            a = proj_simplex(a - step)

            # evaluate true objective
            cur_score = evaluate_obj(a, conv_fft_len)
            if cur_score < best_score - 1e-9:
                best_score = cur_score
                best_vec = a.copy()
                last_imp = time.time()
                print(f"[Adam {it:07d}] new best = {best_score:.6f}")

            # stagnation handling – tiny random perturbation
            if time.time() - last_imp > 30.0:
                noise = rng.normal(scale=0.01, size=n)
                a = proj_simplex(best_vec + noise)
                m.fill(0.0)
                v.fill(0.0)
                last_imp = time.time()
                print(f"[Adam {it:07d}] restart after stagnation")

    # --------------------------------------------------------------
    # 3) Projected sub‑gradient descent (max‑convolution only)
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.80 * budget_s
    a = best_vec.copy()
    it = 0
    last_imp = time.time()
    while time.time() < phase2_deadline:
        it += 1
        conv = real_convolution(a, a, conv_fft_len)[: 2 * n - 1]
        conv_max = conv.max()
        # indices achieving the maximum (within tolerance)
        max_idx = np.where(conv >= conv_max - 1e-15)[0]

        # sub‑gradient: average over all active shifts
        g = np.zeros_like(a)
        for k in max_idx:
            start = max(0, k - (n - 1))
            stop = min(n - 1, k)
            i = np.arange(start, stop + 1, dtype=int)
            j = k - i
            g[i] += 2.0 * a[j]
        if max_idx.size > 0:
            g /= float(max_idx.size)

        lr = 0.12 / np.sqrt(it)          # classic diminishing step‑size
        a = proj_simplex(a - lr * g)

        cur_score = 2.0 * n * conv_max
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_vec = a.copy()
            last_imp = time.time()
            print(f"[PGD {it:07d}] improved to {best_score:.6f}")

        if time.time() - last_imp > 30.0:
            noise = rng.normal(scale=0.015, size=n)
            a = proj_simplex(best_vec + noise)
            last_imp = time.time()
            print(f"[PGD {it:07d}] restart after stagnation")

    # --------------------------------------------------------------
    # 4) Local stochastic refinement (simulated‑annealing style)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a = best_vec.copy()
        best_conv_max = best_score / (2.0 * n)

        T_start = best_score * 0.005
        T_end = best_score * 1e-6
        sa_start = time.time()
        sa_it = 0

        while time.time() < deadline:
            elapsed = time.time() - sa_start
            frac = min(1.0, elapsed / (deadline - sa_start))
            T = T_start * (1.0 - frac) + T_end * frac

            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            max_move = min(a[i], 0.05 * a[i] + 1e-12)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            new_conv_max = max_convolution(a_new, conv_fft_len)
            new_score = 2.0 * n * new_conv_max

            if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_vec = a_new.copy()
                    print(f"[SA {sa_it:07d}] new best = {best_score:.6f} (t={time.time() - start_time:.1f}s)")

            sa_it += 1

    # --------------------------------------------------------------
    # Finalisation
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n)
    else:
        best_vec = best_vec / total

    elapsed = time.time() - start_time
    print(f"[done] total time = {elapsed:.1f}s, best score = {best_score:.6f}, length = {n}")

    return best_vec.tolist()
