def subgradient_descent(p, deadline, max_iters):
    best = p, best_val = objective(p)
    it = 0
    while time.time() < deadline and it < max_iters:
        it+=1
        conv = _conv_fft(p,p)
        max_val = conv.max()
        active = np.where(conv >= max_val - 1e-12)[0]
        g = np.zeros_like(p)
        for k in active:
            i_start = max(0, k-(n-1))
            i_end = min(n-1, k)
            idx_i = np.arange(i_start, i_end+1)
            rev_idx = k - idx_i
            g[idx_i] += 2 * p[rev_idx]
        g /= len(active)
        # line search:
        step = 1.0 / np.sqrt(it)
        for _ in range(10):
            cand = p - step * g
            cand = _project_simplex(cand)
            val = _objective(cand)
            if val < best_val:  # accept if better than current.
                p = cand
                best_seq = cand.copy()

"""
Propose a step‑function (non‑negative list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where conv is the full linear convolution.  The objective is convex on the
simplex, so a simple (sub‑)gradient method works well.  The implementation
here differs from the reference code in three ways:

1.  **Much richer seed set** – geometric, power‑law, symmetric
    power‑law, logistic, piecewise‑linear “U‑shaped”, and many random
    Dirichlet draws.  The seeds are evaluated cheaply and the best few are
    kept.

2.  **Heavy sub‑gradient descent** – the best seed is refined for the
    majority of the time budget using exponentiated‑gradient (mirror
    descent on the simplex).  The learning‑rate schedule is a little more
    aggressive early on (prefactor 5) and the tolerance for the active set
    is tightened to 1e‑12.

3.  **Linear‑programming boost** – after the long sub‑gradient phase we
    solve a tiny LP (only the currently active convolution constraints) to
    add a fresh direction that strictly decreases the maximal autocorrelation.
    The LP is solved with SciPy’s high‑performance “highs” interior‑point
    solver.

The routine respects the supplied time budget (default 1000 s) and returns
the best sequence found as a plain list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy – needed for the LP boost
# ----------------------------------------------------------------------
try:
    from scipy import optimize
    SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full mode)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    seq must be non‑negative (the caller guarantees this).
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple families that return a probability vector (sum == 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r^i, r∈(0,1)."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric: high at both ends, low in the centre."""
    a = r ** np.arange(n, dtype=np.float64)
    a = a + a[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_power(n: int, p: float) -> np.ndarray:
    """Symmetric power law a_i ∝ ((dist to nearest edge)+1)^{-p}."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(np.float64)
    a = 1.0 / ((d + 1) ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic(n: int, slope: float, offset: float) -> np.ndarray:
    """Monotone decreasing logistic shape."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(slope * (i - offset)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _piecewise_u(n: int, edge_frac: float) -> np.ndarray:
    """
    Simple U‑shaped distribution:
    constant value on the two edge blocks of width edge_frac·n,
    zero in the middle.
    The edge value is chosen so that the vector sums to 1.
    """
    edge = int(max(1, round(edge_frac * n)))
    a = np.zeros(n, dtype=np.float64)
    if edge * 2 >= n:
        a[:] = 1.0 / n
        return a
    val = 1.0 / (2 * edge)
    a[:edge] = val
    a[-edge:] = val
    return a / a.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration parameter alpha (scalar)."""
    x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    """Standard‑normal logits → soft‑max probabilities."""
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build an extensive list of start vectors (all sum to 1, non‑negative).
    """
    rng = np.random.default_rng(123456)
    seeds: list[np.ndarray] = []

    # classic families
    seeds.append(_uniform(n))
    for r in np.linspace(0.90, 0.9999, 12):
        seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    for r in np.linspace(0.90, 0.9999, 8):
        for p in np.linspace(0.0, 2.0, 5):
            a = r ** np.arange(n, dtype=np.float64) / ((np.arange(1, n + 1, dtype=np.float64)) ** p)
            s = a.sum()
            seeds.append(a / s if s > 0 else _uniform(n))

    for r in np.linspace(0.90, 0.999, 6):
        seeds.append(_sym_geom(n, r))

    for p in np.linspace(0.2, 3.0, 9):
        seeds.append(_sym_power(n, p))

    # logistic shapes – sweep slope and offset
    for slope in np.linspace(0.005, 0.050, 6):
        for offset in np.linspace(0.2 * n, 0.8 * n, 5):
            seeds.append(_logistic(n, slope, offset))

    # piecewise “U” – different edge widths
    for ef in [0.02, 0.05, 0.10, 0.15, 0.20]:
        seeds.append(_piecewise_u(n, ef))

    # random Dirichlet draws – very low concentration gives sparse vectors
    for a in [0.02, 0.04, 0.08, 0.15, 0.30]:
        seeds.append(_random_dirichlet(n, a, rng))

    # random soft‑max vectors
    for _ in range(12):
        seeds.append(_random_softmax(n, rng))

    # reuse a previously found construction if available
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # clean & normalise (protect against negative zeros)
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    return cleaned


# ----------------------------------------------------------------------
# Sub‑gradient (exponentiated‑gradient / mirror descent) refinement
# ----------------------------------------------------------------------
def _subgradient_expgrad(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the true sub‑gradient of
    max(conv(p,p)).  The update works in log‑space (exponentiated gradient).
    """
    a = start_seq.copy()
    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12          # active‑set tolerance
    prefactor = 5.0          # larger prefactor for faster early progress

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # ----- build sub‑gradient (average over active shifts) -----
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        if len(active) > 0:
            g /= float(len(active))

        # exponentiated‑gradient update
        lr = prefactor / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        # stability: shift to keep numbers modest
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ExpGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# LP boost – solve a tiny LP using only the currently active constraints
# ----------------------------------------------------------------------
def _lp_boost(current: np.ndarray,
              max_active: int = 8,
              max_iters: int = 20,
              t: float = 0.15) -> np.ndarray | None:
    """
    Given a current probability vector `current`, identify the active indices
    of its autocorrelation and solve a tiny linear program that maximises
    the total mass of a new vector `g` while respecting the active constraints.
    The returned vector is renormalised to sum = 1 (or None if LP fails).
    """
    if not SCIPY_AVAILABLE:
        return None

    n = current.size
    conv = _conv_fft(current, current)
    M = conv.max()
    # very tiny tolerance – we only keep a few most binding constraints
    active = np.where(conv >= M - 1e-12)[0]
    if active.size == 0:
        return None
    # optionally truncate the active set to keep the LP tiny
    if active.size > max_active:
        # keep the most central (largest M) indices – they are usually a few
        active = active[:max_active]

    # Build matrix A (|active| × n) where row k corresponds to shift = active[k]
    rows = []
    for shift in active:
        row = np.zeros(n, dtype=np.float64)
        i_start = max(0, shift - (n - 1))
        i_end = min(n - 1, shift)
        # fill the row: coefficient of g_j = current_i where i + j = shift
        for i in range(i_start, i_end + 1):
            j = shift - i
            row[j] = current[i]
        rows.append(row)
    A = np.vstack(rows)

    # maximise sum(g)  <=>  minimise -1·g
    c = -np.ones(n, dtype=np.float64)

    # constraints: A @ g ≤ M
    bounds = [(0.0, None)] * n
    res = optimize.linprog(c,
                           A_ub=A,
                           b_ub=np.full(A.shape[0], M, dtype=np.float64),
                           bounds=bounds,
                           method='highs',
                           options={'presolve': True, 'time_limit': 5.0})
    if res.success and res.x is not None:
        g = res.x
        if g.sum() == 0.0:
            return None
        # blend a little with the current vector to keep the walk stable
        g = (1 - t) * current + t * (g / g.sum())
        g = np.clip(g, 0.0, None)
        if g.sum() == 0.0:
            return None
        g /= g.sum()
        return g
    return None

# Simple Metropolis‑Hastings annealing (tiny mass swaps)
# ----------------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Very light annealing that moves a few percent of mass between two sites."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, cur.size, size=2)
        while i == j:
            j = rng.integers(0, cur.size)

        # move at most 2 % of the source mass
        delta = rng.random() * 0.02 * cur[i]
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # occasional global Gaussian "jump"
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=cur.size)
            tmp = cur + noise
            tmp = np.clip(tmp, 0.0, None)
            if tmp.sum() == 0.0:
                tmp = _uniform(cur.size)
            else:
                tmp /= tmp.sum()
            new_val = _objective(tmp)
            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = tmp
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = tmp.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val

    return global_best_seq.tolist()
