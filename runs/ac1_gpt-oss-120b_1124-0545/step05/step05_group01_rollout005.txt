"""
Propose a step‑function that minimises the self‑convolution ∞‑norm.

The algorithm combines several ideas:

* A rich pool of seeds (uniform, geometric, power‑law, Dirichlet, etc.).
* Adam on a smooth surrogate of the max‑convolution (log‑sum‑exp + variance)
  with a log‑softmax parametrisation (keeps the vector on the simplex).
* Exact projected sub‑gradient descent (mirror‑descent) on the true objective.
* A cheap linear‑programming “gap‑filler”: maximise Σg under the linear
  constraints (f * g) ≤ M, where M is the current maximal convolution value.
  The solution g is blended back into f – this often flattens the peak.
* A short simulated‑annealing walk for a final fine‑tune.

The routine respects the time budget (default 1000 s) and explores a few
different lengths (2000 → 4000 → 8000 → 12000).  The best sequence found
over all lengths is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# Optional linear‑programming imports – fall back gracefully if unavailable.
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
    _HAS_LP = True
except Exception:  # pragma: no cover
    _HAS_LP = False


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – maps ℝⁿ → simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Objective identical to `evaluate_sequence` (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq))²  –  seq assumed non‑negative."""
    s = seq.sum()
    if s < 1e-12:
        return math.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse collection of start‑vectors (already normalised)."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    for alpha in [0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # mirrors – sometimes the optimum is asymmetric
    seeds.extend([s[::-1].copy() for s in seeds])

    # reuse a previous construction if size matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    if not seeds:
        seeds.append(_uniform(n))
    return seeds


# ----------------------------------------------------------------------
# Adam optimiser on a smooth surrogate of the true objective
# ----------------------------------------------------------------------
def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000,
                   beta0: float = 2.0,
                   beta_mul: float = 1.03,
                   lr0: float = 0.7,
                   lambda0: float = 0.8,
                   lambda_decay: float = 0.997) -> tuple[np.ndarray, float]:
    """
    Adam on  L(a) = (1/β)·log∑exp(β·convₖ(a)) + λ·∑ convₖ(a)² .
    Returns the best sequence found and its *true* objective.
    """
    v = np.log(seed + 1e-30)                # logits → softmax → simplex
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = math.inf
    beta = beta0
    lam = lambda0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                      # current point on the simplex
        conv = _conv_fft(a, a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ---- log‑sum‑exp weights ----
        beta *= beta_mul
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of smoothed max ----
        rev_a = a[::-1]
        full_max = _conv_fft(rev_a, w)
        grad_max = 2.0 * full_max[n - 1:n - 1 + n]

        # ---- gradient of variance term (∑ conv²) ----
        full_sq = _conv_fft(rev_a, conv)
        grad_sq = 2.0 * full_sq[n - 1:n - 1 + n]

        grad_a = grad_max + lam * grad_sq

        # ---- chain rule through softmax ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---- Adam update on logits ----
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / math.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

        lam *= lambda_decay
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact projected sub‑gradient (mirror‑descent) on the true objective
# ----------------------------------------------------------------------
def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 60000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using the true sub‑gradient
    of max(conv).  Returns the best sequence seen and its objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step on the simplex
        lr = 0.75 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)            # stabilise
        a = np.exp(log_a)
        a /= a.sum()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming “gap‑filler”
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               max_outer: int = 2) -> tuple[np.ndarray, float]:
    """
    Solve  max Σg  s.t.  (f * g) ≤ M, g ≥ 0   (M = max(f*f)).
    The (potentially unbounded) solution is normalised to the same sum as f
    and blended back.  Repeated a few times if improvement is seen.
    """
    if not _HAS_LP:
        return f, _objective(f)

    best_f = f.copy()
    best_val = _objective(best_f)

    for outer in range(max_outer):
        if time.time() > deadline:
            break

        conv = _conv_fft(best_f, best_f)
        M = conv.max()
        m = 2 * n - 1

        # Build sparse Toeplitz matrix A with diagonals equal to f[d]
        offsets = np.arange(n, dtype=int)          # 0 … n‑1
        diag_data = [np.full(n, best_f[d], dtype=np.float64) for d in range(n)]
        A = _sp.diags(diag_data, offsets, shape=(m, n), format='csc')
        b = np.full(m, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise Σg ⇒ minimise –Σg

        try:
            res = _opt.linprog(c, A_ub=A, b_ub=b,
                               bounds=[(0, None)] * n,
                               method='highs')
        except Exception as exc:                    # pragma: no cover
            print("[LP] linprog failed:", exc)
            break

        if not res.success:
            # LP could be infeasible (should not happen for strictly positive f)
            print("[LP] linprog did not converge, status:", res.status)
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            break

        # Normalise g to the same ℓ₁‑mass as f (which is 1 after our normalisation)
        g_norm = g / sum_g

        # Try a few blending factors; keep the best improvement.
        improved = False
        for alpha in (0.5, 0.3, 0.1):
            cand = (1.0 - alpha) * best_f + alpha * g_norm
            cand = np.clip(cand, 0.0, None)
            cand /= cand.sum()
            val = _objective(cand)
            if val < best_val:
                best_val = val
                best_f = cand
                improved = True
                print(f"[LP] n={n:5d} step={outer} α={alpha:.2f} val={val:.6f}")
                break
        if not improved:               # no further progress – stop early
            break
    return best_f, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (cheap Metropolis–Hastings)
# ----------------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.025 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n:5d} it={it:6d} best={new_val:.6f}")

        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n:5d} it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

# Main entry point (used by the grader)
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    # optionally reuse a construction discovered in a previous run
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # Lengths – start modest, enlarge if time permits.
    lengths = [2000, 4000, 8000, 12000]

    global_best_seq = None
    global_best_val = math.inf

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1) Build a diverse set of seeds (already normalised)
        # --------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # --------------------------------------------------------------
        # 2) Quick evaluation of raw seeds – keep a handful of the best
        # --------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(12, len(seed_vals))]]

        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # --------------------------------------------------------------
        # 3) Adam optimisation on the smooth surrogate
        # --------------------------------------------------------------
        # Allocate a slice of the remaining time to Adam – more for larger n.
        time_per_n = (deadline - time.time()) / (len(lengths) - lengths.index(n))
        max_adam_iters = int(min(15000, max(2000, time_per_n * 0.3)))
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=max_adam_iters,
                beta0=2.0,
                beta_mul=1.03,
                lr0=0.7,
                lambda0=0.8,
                lambda_decay=0.997,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # --------------------------------------------------------------
        # 4) Exact sub‑gradient refinement (mirror descent)
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            max_subgrad_iters = int(min(60000, max(10000,
                                                   (deadline - time.time()) * 20)))
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=max_subgrad_iters)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # --------------------------------------------------------------
        # 5) Linear‑programming gap‑filler (if SciPy is available)
        # --------------------------------------------------------------
        if _HAS_LP and global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _lp_refine(global_best_seq, n, deadline, max_outer=2)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall LP] n={n} best={val:.6f}")

        # --------------------------------------------------------------
        # 6) Simulated‑annealing fine‑tuning
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            # number of annealing steps loosely proportional to remaining time
            remaining = deadline - time.time()
            max_anneal = int(min(200000, max(20000, remaining * 150)))
            seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                        max_iters=max_anneal)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(2000)
        print("[fallback] using uniform distribution")

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    # The grader expects a plain Python list.
    return global_best_seq.tolist()
