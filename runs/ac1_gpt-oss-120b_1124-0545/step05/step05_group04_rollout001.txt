def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = int(kwargs.get('n', 2000))

    # define utils: conv_fft_self, compute_score similar to earlier.

    # load initial best from height_sequence_1 if available.
    best_a = np.ones(n) / n
    best_score = compute_score(best_a)

    if 'height_sequence_1' in globals():
        try:
            seq = np.asarray(globals()['height_sequence_1'], dtype=float)
            if seq.shape[0] == n and seq.sum() > 0:
                seq = np.maximum(seq, 0.0)
                seq /= seq.sum()
                sc = compute_score(seq)
                if sc < best_score:
                    best_a, best_score = seq, sc
        except Exception: pass

    # Function to generate candidates and evaluate.
    def try_candidate(a):
        nonlocal best_a, best_score
        sc = compute_score(a)
        if sc < best_score:
            best_a = a.copy()
            best_score = sc
            print(f"[new best] score={best_score:.6f}, t={time.time()-start:.1f}s")
        return sc

    # 1. Deterministic grid search (expanded ranges)
    # produce list of families
    def family_product(p, off):
        i = np.arange(n, dtype=np.float64)
        a = ((i+off)*(n-1-i+off))**(-p)
        a[~np.isfinite(a)] = 0.0
        total = a.sum()
        if total == 0:

"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. Load a reference sequence if available, otherwise start from uniform.
2. **Expanded analytic sweep** – systematic grid over several symmetric
   families (product‑law, sum‑law, exponential, Gaussian, cosine) with
   broader parameter ranges than before.  The best candidate becomes the
   current best.
3. **Random parametric search** – uniform sampling of the same families
   (10‑20 k draws) to explore beyond the grid.
4. **Smooth‑max continuation** – Adam optimisation of a log‑softmax
   parametrisation using the smooth approximation
   Φ_λ(a) = (1/λ)·log Σ exp(λ·(a*a)_k).  λ is increased geometrically up
   to 2·10⁵, each stage performing a few thousand Adam steps.
5. **Exact sub‑gradient descent** – projected sub‑gradient steps on the
   true objective (max autoconvolution).  Step size follows the classic
   1/√t schedule.
6. **Simulated‑annealing local moves** – tiny random mass transfers
   between two coordinates, accepted either if improving or with a
   temperature‑dependent probability.  Temperature is exponential‑decay.

All phases are time‑boxed by the supplied `budget_s`.  The best sequence
found before the deadline is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself (zero‑padded) using FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector a (sum(a)=1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    if a.sum() < 1e-12:
        return np.inf
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

def analytic_product(n: int, p: float, off: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^(‑p)   (edge‑heavy)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + off) * (n - 1 - i + off)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_product_pos(n: int, p: float, off: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^(p)   (center‑heavy)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + off) * (n - 1 - i + off)) ** p
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, off: float) -> np.ndarray:
    """a_i ∝ (i+off)^(‑p) + (n‑1‑i+off)^(‑p)   (two tails)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + off) ** (-p) + (n - 1 - i + off) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum_pos(n: int, p: float, off: float) -> np.ndarray:
    """a_i ∝ (i+off)^(p) + (n‑1‑i+off)^(p)   (center‑heavy two‑tails)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + off) ** p + (n - 1 - i + off) ** p
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exp_decay(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(‑c·|i‑center|)   (center‑peaked)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - center))
    a = a / a.sum()
    return a


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(‑0.5·((i‑center)/σ)²)   (smooth bump)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = a / a.sum()
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """a_i ∝ (1 + cos(π·(i‑center)/center))^p   (zero at ends, peak centre)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = (1.0 + np.cos(np.pi * (i - center) / center)) ** p
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max utilities
# ----------------------------------------------------------------------


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    Given log‑probability vector x, smoothing λ and length n,
    return Φ_λ(a) and gradient w.r.t. x (where a = softmax(x)).
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # convolution a*a
    conv = conv_fft_self(a)                       # length 2n‑1
    max_c = np.max(conv)

    # smooth‑max over conv entries
    shift = conv - max_c                         # ≤0
    exp_term = np.exp(lam * shift)               # in [0,1]
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam          # Φ_λ

    w = exp_term / sum_exp                        # weights, sum to 1

    # gradient of Φ w.r.t. a (using FFT‑based correlation)
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]       # length n
    grad_a = 2.0 * slice_part[::-1]              # ∂Φ/∂a

    # map gradient to x‑space via Jacobian of soft‑max
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)                  # ∇_x Φ
    return phi, grad_x


def adam_optimize(x: np.ndarray, lam: float, n: int,
                  steps: int, lr: float, deadline: float) -> np.ndarray:
    """Adam optimiser for the smooth‑max objective."""
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for t in range(1, steps + 1):
        if time.time() >= deadline:
            break
        _, grad = compute_phi_and_grad(x, lam, n)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step
    return x


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max‑convolution)
# ----------------------------------------------------------------------


def subgrad_step(a: np.ndarray, step: float) -> np.ndarray:
    """One projected sub‑gradient step for the exact max‑conv objective."""
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    active = (conv >= max_val - 1e-12).astype(np.float64)
    cnt = active.sum()
    if cnt == 0:
        return a
    w = active / cnt
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]
    a_new = a - step * grad
    return project_simplex(a_new)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the supplied evaluation function, respecting the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    n = int(kwargs.get("n", 2000))

    # --------------------------------------------------------------
    # Initialise best candidate (uniform or provided reference)
    # --------------------------------------------------------------
    best_a = np.ones(n, dtype=np.float64) / n
    best_score = compute_score(best_a)

    if "height_sequence_1" in globals():
        try:
            ref = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if ref.shape[0] == n and ref.sum() > 0:
                ref = np.maximum(ref, 0.0)
                ref = ref / ref.sum()
                sc = compute_score(ref)
                if sc < best_score:
                    best_a, best_score = ref, sc
                    print(f"[init] loaded reference, score={best_score:.6f}")
        except Exception:
            pass

    def try_candidate(a_vec: np.ndarray):
        """Evaluate a candidate, keep it if it improves the best score."""
        nonlocal best_a, best_score
        sc = compute_score(a_vec)
        if sc < best_score:
            best_a = a_vec.copy()
            best_score = sc
            print(f"[new best] score={best_score:.6f}  t={time.time()-start_time:.1f}s")
        return sc

    # --------------------------------------------------------------
    # 1) Expanded analytic grid search
    # --------------------------------------------------------------
    print("[grid] starting expanded analytic sweep")
    # Parameter grids (coarse enough to stay fast)
    p_grid = np.linspace(0.10, 2.00, 191)          # step ≈0.01
    off_grid = np.linspace(0.001, 10.0, 101)      # step ≈0.1
    c_grid = np.linspace(0.0005, 0.0300, 126)    # exponential decay
    sigma_grid = np.linspace(30.0, 1500.0, 121)   # Gaussian width
    cos_p_grid = np.linspace(0.10, 6.00, 121)

    # product‑law (edge‑heavy)
    for p in p_grid:
        if time.time() >= deadline: break
        for off in off_grid:
            if time.time() >= deadline: break
            cand = analytic_product(n, p, off)
            try_candidate(cand)

    # product‑law (center‑heavy)
    for p in p_grid:
        if time.time() >= deadline: break
        for off in off_grid:
            if time.time() >= deadline: break
            cand = analytic_product_pos(n, p, off)
            try_candidate(cand)

    # sum‑law (edge‑heavy)
    for p in p_grid:
        if time.time() >= deadline: break
        for off in off_grid:
            if time.time() >= deadline: break
            cand = analytic_sum(n, p, off)
            try_candidate(cand)

    # sum‑law (center‑heavy)
    for p in p_grid:
        if time.time() >= deadline: break
        for off in off_grid:
            if time.time() >= deadline: break
            cand = analytic_sum_pos(n, p, off)
            try_candidate(cand)

    # exponential decay
    for c in c_grid:
        if time.time() >= deadline: break
        cand = analytic_exp_decay(n, c)
        try_candidate(cand)

    # Gaussian bump
    for sigma in sigma_grid:
        if time.time() >= deadline: break
        cand = analytic_gaussian(n, sigma)
        try_candidate(cand)

    # Cosine bump
    for p in cos_p_grid:
        if time.time() >= deadline: break
        cand = analytic_cosine(n, p)
        try_candidate(cand)

    print(f"[grid] best after sweep = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Random parametric search (lightweight)
    # --------------------------------------------------------------
    print("[rand] random parametric search")
    families = [
        ("product_neg", analytic_product,   {"p": (0.1, 2.0), "off": (0.001, 10.0)}),
        ("product_pos", analytic_product_pos, {"p": (0.1, 2.0), "off": (0.001, 10.0)}),
        ("sum_neg",    analytic_sum,       {"p": (0.1, 2.0), "off": (0.001, 10.0)}),
        ("sum_pos",    analytic_sum_pos,   {"p": (0.1, 2.0), "off": (0.001, 10.0)}),
        ("exp",        analytic_exp_decay, {"c": (0.0005, 0.03)}),
        ("gauss",      analytic_gaussian,  {"sigma": (30.0, 1500.0)}),
        ("cos",        analytic_cosine,    {"p": (0.1, 6.0)})
    ]

    max_rand_iters = 20000
    for it in range(max_rand_iters):
        if time.time() >= deadline:
            break
        fam_name, gen_func, param_ranges = families[np.random.randint(len(families))]
        params = {}
        for key, (lo, hi) in param_ranges.items():
            params[key] = np.random.rand() * (hi - lo) + lo
        try:
            if fam_name.startswith("product") or fam_name.startswith("sum"):
                cand = gen_func(n, params["p"], params["off"])
            elif fam_name == "exp":
                cand = gen_func(n, params["c"])
            elif fam_name == "gauss":
                cand = gen_func(n, params["sigma"])
            elif fam_name == "cos":
                cand = gen_func(n, params["p"])
            else:
                continue
        except Exception:
            continue
        try_candidate(cand)

    # --------------------------------------------------------------
    # 3) Smooth‑max Adam optimisation (λ‑continuation)
    # --------------------------------------------------------------
    print("[adam] smooth‑max continuation")
    eps = 1e-12
    x = np.log(np.maximum(best_a, eps))

    lam_schedule = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0,
                    2000.0, 8000.0, 20000.0, 80000.0, 200000.0]

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # cheap estimate: ~0.003 s per iteration at n≈2000
        max_iters = max(1, int(remaining / 0.003))
        steps = min(6000, max_iters)  # cap to keep it reasonable
        if lam <= 10.0:
            steps = max(1000, int(steps * 0.5))
        if lam > 20000.0:
            steps = max(500, int(steps * 0.3))
        x = adam_optimize(x, lam, n, steps, lr=0.05, deadline=deadline)
        a = softmax(x)
        try_candidate(a)

    # --------------------------------------------------------------
    # 4) Exact sub‑gradient descent (projected)
    # --------------------------------------------------------------
    print("[subgrad] exact sub‑gradient descent")
    a = best_a.copy()
    iter_sub = 0
    while time.time() < deadline:
        iter_sub += 1
        step = 0.5 / np.sqrt(iter_sub)          # classic 1/√t schedule
        a = subgrad_step(a, step)
        try_candidate(a)

    # --------------------------------------------------------------
    # 5) Simulated‑annealing random mass transfers
    # --------------------------------------------------------------
    print("[anneal] stochastic local search")
    a = best_a.copy()
    # start temperature – roughly the scale of the objective
    T0 = 0.02
    decay = 0.9995
    scale = 0.02
    itr = 0
    while time.time() < deadline:
        itr += 1
        T = T0 * (decay ** itr)
        cand = random_move(a, scale)
        sc_cand = compute_score(cand)
        delta = sc_cand - best_score
        # accept if better or with prob exp(-Δ/T)
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-12)):
            a = cand
            if sc_cand < best_score:
                best_a, best_score = cand, sc_cand
                print(f"[anneal] new best = {best_score:.6f}  t={time.time()-start_time:.1f}s")
        # slowly shrink the move scale to fine‑tune
        scale = max(1e-4, scale * 0.999)

    total = time.time() - start_time
    print(f"[done] final best score = {best_score:.6f}, total time = {total:.1f}s")
    return best_a.tolist()
