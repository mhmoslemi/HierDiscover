def propose_candidate(seed=42,budget_s=1000,**kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Determine candidate lengths to try
    possible_n = [2000, 3000, 4000]
    best_seq=None
    best_score=np.inf
    for n in possible_n:
        # generate some start candidates
        # product family grid search
        for p in np.linspace(0.4,1.6,6):
            for off in np.linspace(0.05,1.0,6):
                cand = analytic_product(n,p,off)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score=sc; best_seq=cand.copy(); best_n=n
        # try exponential family
        for lam in np.linspace(0.001,0.01,5):
            cand=analytic_exponential(n,lam)
            sc=compute_score(cand)
            if sc<best_score: best_seq=cand.copy(); best_score=sc; best_n=n
        # try cosine
        for p in np.linspace(0.5,4.0,5):
            cand=analytic_cosine(n,p)
            sc=compute_score(cand)
            if sc<best_score: best_seq=cand.copy(); best_score=sc; best_n=n
    # output initial best
    a=best_seq; n=best_n
    print(f"[init] best initial n={n} score={best_score:.6f}")
    best_seq=a.copy()
    best_score=compute_score(a)
    # Adam with smoothed gradient
    m=np.zeros_like(a); v=np.zeros_like(a)
    beta1,beta2,eps=0.9,0.999,1e-8
    alpha_min,alpha_max=1.0,200.0
    alpha=alpha_min
    growth_factor = (alpha_max/alpha_min)**(1.0/200000.0) # 200k steps
    step=0
    last_improve=time.time()
    while time.time()<deadline and step<200000:
        grad = smooth_max_gradient(a, alpha)
        step+=1
        lr = 0.05/(1.0+step*1e-5)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1-beta1**step)
        v_hat = v/(1-beta2**step)
        a = a - lr * m_hat/(np.sqrt(v_hat)+eps)
        a = np.clip(a,0,None)
        a = project_simplex(a)
        alpha = min(alpha_max, alpha*growth_factor)
        if step%1000==0:
            cur = compute_score(a)
            if cur<best_score-1e-12:
                best_score=cur; best_seq=a.copy(); last_improve=time.time()
                print(f"[smooth] step {step} new best {best_score:.6f}")
        if time.time() - last_improve > 30.0:
            # jitter restart
            jitter = np.random.rand(n)*1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            last_improve=time.time()
            print(f"[smooth] jitter restart at step {step}")
        # occasional random move
        if np.random.rand()<0.01:
            a = random_move(a, step*0.2)
    # Switch to subgradient Adam
    # Reset Adam parameters maybe
    m=np.zeros_like(a); v=np.zeros_like(a)
    step_sub=0
    while time.time()<deadline:
        # compute conv and gradient
        conv = conv_fft(a)
        max_val = conv.max()
        # tolerance for max indices
        max_idxs = np.where(np.abs(conv - max_val) < 1e-12)[0]
        # compute avg_rev_shift
        grad = np.zeros_like(a)
        for k in max_idxs:
            start = max(0, k - (n-1))
            end = min(k, n-1)
            idxs = k - np.arange(start, end+1)
            grad[start:end+1] += a[idxs]
        grad = grad / len(max_idxs)
        # scale gradient: objective gradient = 4*n*grad
        grad = grad * (4.0 * n)
        # Adam update
        step_sub+=1
        lr = 0.05/(1.0+step_sub*1e-5)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1-beta1**step_sub)
        v_hat = v/(1-beta2**step_sub)
        a = a - lr * m_hat/(np.sqrt(v_hat)+eps)
        a = np.clip(a,0,None)
        a = project_simplex(a)
        if step_sub % 500 == 0:
            cur = compute_score(a)
            if cur<best_score-1e-12:
                best_score=cur; best_seq=a.copy(); last_improve=time.time()
                print(f"[subgrad] step {step_sub} new best {best_score:.6f}")
        if time.time() - last_improve > 30.0:
            jitter = np.random.rand(n)*1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            last_improve=time.time()
            print(f"[subgrad] jitter restart at step {step_sub}")
        if np.random.rand()<0.01:
            a = random_move(a, step_sub * 0.2)
    # final refinement
    while time.time()<deadline:
        perturb = np.random.normal(scale=5e-4,size=n)
        cand = best_seq + perturb
        cand = np.clip(cand,0,None)
        cand = project_simplex(cand)
        sc = compute_score(cand)
        if sc < best_score:
            best_score=sc
            best_seq=cand.copy()
            print(f"[refine] improved to {best_score:.6f}")
    print(f"[done] best {best_score:.6f}, n={n}, time={time.time()-start:.1f}s")
    return best_seq.tolist()

"""
Propose a non‑negative step‑function that (approximately) minimises

    2·n·max(conv(a,a)) / (sum(a)**2).

The algorithm works in three phases:

1. **Broad initialisation** – evaluate a handful of analytic families
   (product, sum‑of‑powers, exponential, cosine, beta) on several
   candidate lengths (default n=4000).  The best candidate is kept.

2. **Hybrid optimisation** – first run an Adam optimiser on the smooth
   max‑approximation (log‑sum‑exp) with a slowly increasing temperature
   α.  When α reaches its maximum we switch to a projected Adam optimiser
   that uses the exact sub‑gradient of the max‑convolution (much cheaper).

3. **Stochastic refinement** – tiny Gaussian perturbations are accepted
   when they improve the exact score.

All updates keep the vector on the probability simplex (Σa=1, a_i≥0).
The routine respects the 1000 s budget and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (zero‑padded) via FFT.  Length = 2·len(a)-1."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two arbitrary 1‑D arrays via FFT."""
    size = _next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: len(a) + len(b) - 1]


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w≥0, Σw=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families (used for a quick grid search)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """U‑shaped product family: a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """Sum‑of‑powers family: a_i ∝ (i+off)^{-p} + (n‑1‑i+off)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """Exponential tail family: a_i ∝ exp(-lam·min(i,n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a = a / a.sum()
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """Cosine‑power family: a_i ∝ |cos(πi/(n‑1))|^{p}."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    a = a / a.sum()
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped family: a_i ∝ (i+1)^{α}·(n‑i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max gradient (soft‑max of the convolution)
# ----------------------------------------------------------------------


def smooth_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k).
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    conv = conv_fft(a)                         # length 2·n‑1
    maxc = conv.max()
    exp_alpha = np.exp(alpha * (conv - maxc))
    w = exp_alpha / exp_alpha.sum()           # soft‑max weights, Σ w = 1

    rev_a = a[::-1]
    g_full = conv_fft_general(rev_a, w)       # length 3·n‑2
    start = n - 1
    g = 2.0 * g_full[start:start + n]         # factor 2 from symmetry
    return g


# ----------------------------------------------------------------------
# Exact sub‑gradient of max‑convolution
# ----------------------------------------------------------------------


def subgradient_max(a: np.ndarray) -> np.ndarray:
    """
    Compute the (unnormalised) sub‑gradient of max(conv(a,a)).
    Returns a vector of length len(a).  The gradient points in the
    direction of increase; the optimiser will subtract it.
    """
    n = a.shape[0]
    conv = conv_fft(a)
    max_val = conv.max()
    # consider all indices attaining the maximum (within a tiny tolerance)
    idxs = np.where(np.abs(conv - max_val) < 1e-12)[0]

    grad = np.zeros_like(a)
    for k in idxs:
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        # indices of a involved in this term
        idx_range = k - np.arange(start, end + 1)
        grad[start:end + 1] += a[idx_range]

    if len(idxs) > 0:
        grad = grad / len(idxs)               # average over the active set
    # The true gradient of max(conv) is 2·grad, but the factor is
    # irrelevant for a projected Adam update (it only rescales the step size).
    return grad


# ----------------------------------------------------------------------
# Exact objective (used for bookkeeping)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a (sum normalized to 1).
    Returns 2·n·max(conv(a,a)).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)
