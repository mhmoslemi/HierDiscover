"""
Propose a step‑function (list of non‑negative floats) that minimizes

        2·n·max(conv(a,a)) / (sum(a)**2)

The implementation combines several complementary strategies
within the 1000 s budget:

1. **Analytic sweep** – a quick scan of a few classic families
   (product‑law, sum‑law, exponential, Gaussian) to obtain a decent
   starting point.

2. **Mixture‑parameter hill‑climbing** – the candidate is taken as a
   convex combination of a uniform distribution and a product‑law
   shape ``a_i ∝ ((i+off)*(n‑1‑i+off))^{‑p}``.  The three parameters
   (p, off, α) are explored with simulated‑annealing‑style random
   moves; the best mixture is kept throughout.

3. **Smooth‑max Adam optimisation** – the objective is approximated by
   a log‑sum‑exp (smooth max) with a continuation on the smoothing
   parameter λ.  Adam updates are performed on the unconstrained
   log‑probability vector (soft‑max parametrisation).  The resulting
   point is projected onto the probability simplex.

4. **Exact sub‑gradient descent** – after smoothing we directly
   minimise the true max‑convolution by a projected sub‑gradient method
   (step size ∝ 1/√t).

5. **Annealed random mass‑transfer** – a final stochastic local search
   that repeatedly transfers a tiny amount of probability mass between
   two coordinates.  Worse moves are occasionally accepted with a
   temperature schedule to escape shallow local minima.

All phases respect the global deadline; the best sequence found so far
is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a (the sum is ignored;
    the function internally normalises).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families (used for the initial sweep)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^(‑p)   (edge‑heavy shape)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^(‑p) + (n‑1‑i+off)^(‑p)   (two tails)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exp_decay(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(‑c·|i‑center|)   (center‑peaked shape)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - center))
    a = a / a.sum()
    return a


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(‑0.5·((i‑center)/σ)²)   (smooth bump)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Mixture of product‑law and uniform (parameterised family)
# ----------------------------------------------------------------------


def uniform_array(n: int) -> np.ndarray:
    return np.ones(n) / n


def mix_product_uniform(n: int, p: float, off: float, alpha: float) -> np.ndarray:
    """
    Convex blend  a = (1‑α)·uniform + α·product‑law(p,off).
    Both components are already normalised, so the mixture is as well.
    """
    a_prod = analytic_product(n, p, off)
    a_uni = uniform_array(n)
    a = (1.0 - alpha) * a_uni + alpha * a_prod
    # safeguard against numerical drift
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth‑max utilities for Adam optimisation
# ----------------------------------------------------------------------


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    Smooth approximation Φ_λ and its gradient w.r.t. the log‑probability
    vector x (a = softmax(x)).  Returns (phi, grad_x).
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # convolution a*a
    conv = conv_fft_self(a)                # length 2n‑1
    max_c = np.max(conv)

    # smooth max (log‑sum‑exp) over convolution entries
    shift = conv - max_c                   # ≤ 0
    exp_term = np.exp(lam * shift)        # ≤ 1
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam    # Φ_λ

    w = exp_term / sum_exp                 # weights, sum to 1

    # gradient of Φ_λ w.r.t. a: convolution of w (reversed) with a, scaled by 2
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad_a = 2.0 * slice_part[::-1]            # ∂Φ/∂a

    # map gradient to x‑space via Jacobian of soft‑max
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)                # ∇_x Φ
    return phi, grad_x


def adam_optimize(x: np.ndarray, lam: float, n: int, steps: int,
                  lr: float, deadline: float) -> np.ndarray:
    """Adam optimiser for the smooth‑max objective."""
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for t in range(1, steps + 1):
        if time.time() >= deadline:
            break
        _, grad = compute_phi_and_grad(x, lam, n)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step
    return x


# ----------------------------------------------------------------------
# Exact sub‑gradient descent for the true objective
# ----------------------------------------------------------------------


def subgrad_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step for the exact objective
    max(conv(a,a)).  Uses an FFT‑based computation of the sub‑gradient.
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    active = (conv >= max_val - 1e-12).astype(np.float64)   # binary mask
    cnt = active.sum()
    if cnt == 0:
        return a
    w = active / cnt                                         # uniform over active
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]            # sub‑gradient w.r.t. a
    a_new = a - step * grad
    return project_simplex(a_new)

    cur_alpha = 0.92          # weight on the product‑law component
    cur_a = mix_product_uniform(n, cur_p, cur_off, cur_alpha)
    cur_score = compute_score(cur_a)

    # keep the best seen so far (might already be better than the grid)
    if cur_score < best_score:
        best_score, best_a = cur_score, cur_a.copy()
        print(f"[mix] new best from mixture start = {best_score:.6f}")

    # annealing schedule
    T = 0.02                # initial temperature for Metropolis acceptance
    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        # propose new parameters (global jump vs small perturbation)
        if np.random.rand() < 0.3:
            # global random draw
            p_new = np.random.uniform(0.05, 1.5)
            off_new = np.random.uniform(0.0005, 5.0)
            alpha_new = np.random.uniform(0.60, 0.99)
        else:
            # local Gaussian perturbation
            p_new = cur_p + np.random.normal(scale=0.02)
            off_new = cur_off + np.random.normal(scale=0.02)
            alpha_new = cur_alpha + np.random.normal(scale=0.02)

        # clip to admissible ranges
        p_new = np.clip(p_new, 0.01, 3.0)
        off_new = np.clip(off_new, 0.0001, 10.0)
        alpha_new = np.clip(alpha_new, 0.0, 1.0)

        cand_a = mix_product_uniform(n, p_new, off_new, alpha_new)
        cand_score = compute_score(cand_a)

        # keep global best
        if cand_score < best_score:
            best_score, best_a = cand_score, cand_a.copy()
            print(f"[mix] iter={iter_cnt} new best = {best_score:.6f} (t={time.time()-start_time:.1f}s)")

        # Metropolis acceptance for the *current* state
        delta = cand_score - cur_score
        if delta < 0 or np.exp(-delta / max(T, 1e-12)) > np.random.rand():
            cur_p, cur_off, cur_alpha = p_new, off_new, alpha_new
            cur_a, cur_score = cand_a, cand_score

        # exponential cooling (very slow)
        T *= 0.99995
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3) Smooth‑max optimisation with Adam (continuation on λ)
    # ------------------------------------------------------------------
    # initialise log‑probability vector from the best mixture found
    eps = 1e-12
    x = np.log(np.maximum(best_a, eps))

    # schedule of smoothing parameters (from gentle to almost exact)
    lam_schedule = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0,
                    2000.0, 8000.0, 20000.0, 80000.0, 200000.0, 800000.0]

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        # dynamic step budget: roughly 0.001 s per Adam iteration for n≈2000
        remaining = deadline - time.time()
        max_iter = int(min(8000, max(500, remaining / 0.0015)))
        x = adam_optimize(x, lam, n, max_iter, lr=0.05, deadline=deadline)
        a = softmax(x)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[adam λ={lam:.0f}] new best = {best_score:.6f} (t={time.time()-start_time:.1f}s)")

    # ------------------------------------------------------------------
    # 4) Exact sub‑gradient descent on the true max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        step = 0.5 / np.sqrt(it)          # classic diminishing stepsize
        a = subgrad_step(a, step)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[subgrad iter={it}] new best = {best_score:.6f} (t={time.time()-start_time:.1f}s)")
        # tiny safety break if progress stalls (optional)
        if it % 500000 == 0 and time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 5) Annealed random‑mass‑transfer local search (simulated annealing)
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_score = best_score
    scale = 0.02                     # relative amount moved per trial
    T_move = 0.02                    # temperature for accepting worse moves
    min_scale = 0.001
    max_scale = 0.06
    while time.time() < deadline:
        cand = random_move(a, scale)
        cand_score = compute_score(cand)
        delta = cand_score - cur_score
        if delta < 0 or np.exp(-delta / max(T_move, 1e-12)) > np.random.rand():
            a, cur_score = cand, cand_score
        if cand_score < best_score:
            best_score, best_a = cand_score, cand.copy()
            print(f"[random] new best = {best_score:.6f} (t={time.time()-start_time:.1f}s)")
            # tighten the move size after improvement
            scale = max(min_scale, scale * 0.85)
        else:
            # slowly relax step size when no improvement
            scale = min(max_scale, scale * 1.0015)
        # cooling of the Metropolis temperature
        T_move *= 0.9999

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    return best_a.tolist()
