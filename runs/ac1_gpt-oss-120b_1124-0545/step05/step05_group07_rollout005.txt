def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    # Determine length candidates
    n_candidates = [2000, 3000, 4000, 5000]
    best_seq = None
    best_score = np.inf
    # Grid search initial families
    for n in n_candidates:
        # product family
        p_vals = np.linspace(0.2, 0.9, 15)
        off_vals = np.concatenate((np.linspace(0.001, 0.1, 5), np.linspace(0.2,1.0,10)))
        for p in p_vals:
            for off in off_vals:
                cand = analytic_product(n,p,off)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc; best_seq = cand.copy()
        # beta family (symmetric)
        alpha_vals = np.linspace(-1.5, 0, 13)
        for a in alpha_vals:
            cand = analytic_beta(n, a, a)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc; best_seq = cand.copy()
        # exponential tails
        lam_vals = np.linspace(0.01, 2.0, 15)
        for lam in lam_vals:
            cand = analytic_exponential(n, lam)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc; best_seq = cand.copy()
    print("[init] best init score {:.6f}, length {}".format(best_score, len(best_seq)))
    # Run main optimizer
    a = best_seq.copy()
    n = a.shape[0]
    # Setup adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0
    last_improve = time.time()
    best_seq_opt = a.copy()
    best_score_opt = best_score
    alpha = 1.0
    alpha_max = 1000.0
    # growth factor per iteration, approximate up to 500k steps
    growth_factor = 1.00002
    # main loop
    while time.time() < deadline:
        # gradient
        grad = smooth_max_gradient(a, alpha)
        step += 1
        lr = 0.1 / (1.0 + step * 1e-5)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        a = project_simplex(a)
        # increase alpha
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * growth_factor)
        # periodically evaluate exact score
        if step % 200 == 0:
            cur = compute_score(a)
            if cur < best_score_opt - 1e-14:
                best_score_opt = cur
                best_seq_opt = a.copy()
                last_improve = time.time()
                print("[{:07d}] new best {:.6f} α={:.2f}".format(step, cur, alpha))
        # jitter restart if stuck
        if time.time() - last_improve > 12.0:
            jitter = np.random.rand(n) * 1e-4
            a = best_seq_opt + jitter
            a = project_simplex(a)
            last_improve = time.time()
            print("[{:07d}] jitter restart".format(step))
        # occasional random mass transfer
        if np.random.rand() < 0.02:
            a = random_move(a, scale=0.5)
        # exit condition
        if time.time() >= deadline:
            break
    # final refinement
    while time.time() < deadline:
        pert = np.random.normal(scale=2e-4, size=n)
        cand = best_seq_opt + pert

"""
Propose a non‑negative step function that (approximately) minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Broad analytic grid search** – evaluates several families (product,
   beta‑shaped and exponential tails) on a range of lengths (2000‑5000)
   and selects the best normalised vector.  This gives a good
   “U‑shaped” start (close to the continuous arcsine law), which is
   known to be near‑optimal for the L∞‑self‑convolution problem.

2. **Smoothed‑max Adam optimisation** – minimise the smooth
   approximation `f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)` using Adam.
   The temperature `α` is increased very slowly (up to 1000) so that
   the optimiser gradually focuses on the true max.  After each update
   the vector is projected onto the probability simplex.

3. **Final stochastic refinement** – small Gaussian perturbations
   around the best vector are accepted when they improve the exact
   (non‑smoothed) score.

All heavy work (convolutions and gradients) is carried out with FFTs,
so the routine comfortably fits into the 1000 s budget for `n≈4000–5000`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (zero‑padded) using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two arbitrary 1‑D arrays using FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    size = _next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: len(a) + len(b) - 1]


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """U‑shaped family a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped family a_i ∝ (i+1)^{α}·(n‑i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """Exponential tails a_i ∝ exp(-lam·min(i,n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a = a / a.sum()
    return a


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a.
    The vector is normalised to Σa = 1 before evaluation.
    Returns 2·n·max(conv(a,a)).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)


def smooth_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed‑max objective
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    using FFTs.  Returns a vector of length len(a).
    """
    n = a.shape[0]
    conv = conv_fft(a)                         # length 2n‑1
    maxc = conv.max()
    # stable soft‑max weights
    exp_alpha = np.exp(alpha * (conv - maxc))
    w = exp_alpha / exp_alpha.sum()           # Σ w = 1
    # gradient = 2·(rev(a) ∗ w)[n‑1 : n‑1+n]
    rev_a = a[::-1]
    g_full = conv_fft_general(rev_a, w)       # length 3n‑2
    start = n - 1
    g = 2.0 * g_full[start:start + n]
    return g

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the given evaluation function.  The algorithm combines a fast analytic
    grid search with Adam optimisation on a smoothed max‑convolution
    objective, followed by a short stochastic refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) Broad analytic grid search over several lengths and families
    # ------------------------------------------------------------------
    n_candidates = [2000, 3000, 4000, 5000]
    best_seq = None
    best_score = np.inf

    for n in n_candidates:
        # product (U‑shaped) family
        p_vals = np.linspace(0.2, 0.9, 13)
        offset_vals = np.concatenate((
            np.linspace(0.001, 0.1, 5),
            np.linspace(0.2, 1.0, 9)
        ))
        for p in p_vals:
            for off in offset_vals:
                cand = analytic_product(n, p, off)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()

        # symmetric beta family (negative exponents stress the ends)
        alpha_vals = np.linspace(-1.5, 0.0, 13)
        for a in alpha_vals:
            cand = analytic_beta(n, a, a)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()

        # exponential tails
        lam_vals = np.linspace(0.01, 2.0, 13)
        for lam in lam_vals:
            cand = analytic_exponential(n, lam)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()

    if best_seq is None:
        # fallback – uniform distribution
        n = 3000
        best_seq = np.ones(n) / n
        best_score = compute_score(best_seq)
    else:
        n = best_seq.shape[0]

    print(f"[init] best initial score = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 2) Smoothed‑max Adam optimisation
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    alpha = 1.0
    alpha_max = 1000.0
    # increase α geometrically – factor tuned for a few hundred thousand steps
    growth_factor = 1.000025
    last_improve = time.time()
    best_seq_opt = a.copy()
    best_score_opt = best_score

    while time.time() < deadline:
        # gradient of the smooth max‑convolution
        grad = smooth_max_gradient(a, alpha)

        step += 1
        # learning‑rate schedule (slow decay)
        lr = 0.12 / (1.0 + step * 1e-5)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep feasible (non‑negative, sum = 1)
        a = np.clip(a, 0.0, None)
        a = project_simplex(a)

        # temperature schedule
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * growth_factor)

        # occasional exact evaluation
        if step % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score_opt - 1e-14:
                best_score_opt = cur_score
                best_seq_opt = a.copy()
                last_improve = time.time()
                print(f"[{step:07d}] new best = {cur_score:.6f} (α={alpha:.2f})")

        # restart with tiny jitter if stalled
        if time.time() - last_improve > 12.0:
            jitter = np.random.rand(n) * 5e-5
            a = best_seq_opt + jitter
            a = project_simplex(a)
            last_improve = time.time()
            print(f"[{step:07d}] jitter restart")

        # cheap random mass‑transfer move
        if np.random.rand() < 0.02:
            a = random_move(a, scale=0.5)

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3) Final stochastic refinement (Gaussian perturbations)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        perturb = np.random.normal(scale=2e-4, size=n)
        cand = best_seq_opt + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        sc = compute_score(cand)
        if sc < best_score_opt:
            best_score_opt = sc
            best_seq_opt = cand.copy()
            print(f"[refine] improved to {sc:.6f}")

    total = time.time() - start_time
    print(f"[done] best score = {best_score_opt:.6f}, length = {n}, time = {total:.1f}s")
    return best_seq_opt.tolist()
