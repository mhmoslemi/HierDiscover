"""
Propose a non‑negative step‑function that minimizes the evaluation function.

The algorithm works in three stages:

1. **Seed creation** – a large, diverse pool (uniform, geometric,
   power‑law, triangular, cosine, Gaussian, Dirichlet draws, mirrors and the
   previously found best sequence).

2. **Smooth‐max L‑BFGS optimisation** – the loss
       L(v) = (1/β)·log∑ₖexp(β·convₖ(a)) + λ·∑ₖconvₖ(a)²,
   where a = softmax(v) (so a ≥ 0 and ∑a = 1).  L is a smooth proxy for the
   true objective `max(conv)`.  We run a few L‑BFGS phases with increasing
   β and decreasing λ, using analytic gradients obtained via one FFT per
   term.

3. **Exact sub‑gradient refinement + cheap annealing** – after the smooth
   optimisation we perform a mirror‑descent sub‑gradient walk on the true
   max‑convolution and finish with a lightweight simulated‑annealing
   perturbation.  All phases respect the global time budget.

The best sequence observed over all explored lengths is returned as a plain
list of floats.  The code contains no closures, only top‑level helpers,
and uses only NumPy and SciPy (already available in the execution environment).
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the grader (scale‑invariant)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0, None)
    return a / a.sum()


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    a = np.clip(a, 0, None)
    return a / a.sum()


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse or dense Dirichlet draw."""
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a rich list of start vectors (all normalised to sum = 1)."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic deterministic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 7):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 7):
        seeds.append(_power(n, p))

    # mixed geometric·power
    for c in np.linspace(0.5, 3.5, 4):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 4):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 4):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # Dirichlet draws – cover a wide sparsity spectrum
    for alpha in [0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # mirrors (helps when optimum is asymmetric)
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # re‑use previously discovered best construction if length matches
    if previous is not None and previous.shape == (n,):
        try:
            prev = np.asarray(previous, dtype=np.float64)
            if prev.sum() > 0.0:
                seeds.append(prev / prev.sum())
        except Exception:
            pass

    # guarantee non‑empty pool
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Smooth‑max loss used by L‑BFGS
# ----------------------------------------------------------------------


def _smooth_max_loss(v: np.ndarray, n: int, beta: float, lam: float) -> tuple[float, np.ndarray]:
    """
    Returns (loss, grad_v) for the smoothed max objective.
    - a = softmax(v)   (∑a = 1)
    - loss = (1/β)·log∑exp(β·conv(a,a)) + λ·∑conv(a,a)²
    - grad_v is the gradient w.r.t. the logits v.
    """
    a = _softmax(v)                      # probability vector
    conv = _conv_fft(a, a)               # full linear convolution, length 2n‑1

    # ---- smooth max (log‑sum‑exp) ----
    shift = conv.max()
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    loss = (np.log(sum_exp) + beta * shift) / beta

    # optional variance regulariser
    if lam != 0.0:
        loss += lam * np.sum(conv ** 2)

    # ---- gradient of the smooth max ----
    w = exp_term / sum_exp               # softmax‑weights over convolution entries
    rev_a = a[::-1]                      # reversed a for correlation‑style convolution
    full_w = _conv_fft(rev_a, w)         # length 2n‑1
    grad_max = 2.0 * full_w[n - 1:n - 1 + n]

    # ---- gradient of the variance term (if any) ----
    if lam != 0.0:
        full_sq = _conv_fft(rev_a, conv)
        grad_sq = 2.0 * full_sq[n - 1:n - 1 + n]
    else:
        grad_sq = 0.0

    grad_a = grad_max + lam * grad_sq     # chain rule through convolution

    # ---- chain rule through the softmax parametrisation ----
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    return float(loss), grad_v.astype(np.float64)


def _lbfgs_fun(v: np.ndarray, n: int, beta: float, lam: float) -> float:
    loss, _ = _smooth_max_loss(v, n, beta, lam)
    return loss


def _lbfgs_jac(v: np.ndarray, n: int, beta: float, lam: float) -> np.ndarray:
    _, grad = _smooth_max_loss(v, n, beta, lam)
    return grad


def _lbfgs_optimize(seed: np.ndarray, n: int, deadline: float) -> np.ndarray:
    """
    Runs a few L‑BFGS phases on the smooth‑max surrogate.
    Returns the best sequence found (still on the simplex, i.e. ∑a = 1).
    """
    # initialise logits (log‑softmax parameters)
    v = np.log(seed + 1e-30)

    best_a = _softmax(v)
    best_val = _objective(best_a)

    # schedule of (beta, lambda, maxiter) – later phases focus more on the true max
    schedule = [
        (2.0, 0.5, 80),
        (5.0, 0.2, 80),
        (12.0, 0.0, 200),
    ]

    for beta, lam, maxiter in schedule:
        if time.time() > deadline:
            break

        res = optimize.minimize(
            fun=_lbfgs_fun,
            x0=v,
            args=(n, beta, lam),
            jac=_lbfgs_jac,
            method="L-BFGS-B",
            options={"maxiter": maxiter, "disp": False},
        )
        v = res.x

        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a
            print(f"[L‑BFGS] n={n}, β={beta:.1f}, λ={lam:.2f}, best={cur_val:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the true max)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray, n: int, deadline: float,
                        max_iters: int = 120000) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror‑descent) on the genuine maximal
    convolution entry.  Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirrored simplex update)
        lr = 0.8 / np.sqrt(it)               # simple 1/√t schedule
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # stabilisation
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq


# ----------------------------------------------------------------------
# Very light simulated‑annealing (used only if time permits)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray, n: int, deadline: float,
                    max_iters: int | None = None) -> np.ndarray:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)   # start temperature
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.025 * cur[i]   # up to 2.5 % of source mass
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:6d} best={new_val:.6f}")

        if it % 1000 == 0:
            # occasional global Gaussian jitter
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:6d} (jump) best={new_val:.6f}")

    return best


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns the
    best sequence found (as a Python list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # keep a safety margin

    # reuse previously discovered best construction if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – start small, enlarge when time permits
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000]   # 8000 will be attempted only if time remains
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # 1) Build a rich seed set
        # ------------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # ------------------------------------------------------------------
        # 2) Quick evaluation of raw seeds, keep the most promising few
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(10, len(seed_vals))]]

        # update global best based on raw seeds
        if seed_vals and seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # 3) L‑BFGS optimisation of the smooth max surrogate
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            # ensure sum = 1 (the optimiser works on the simplex)
            seed_vec = seed_vec / seed_vec.sum()
            seq = _lbfgs_optimize(seed_vec, n, deadline)
            cur_val = _objective(seq)
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = seq.copy()
                print(f"[overall L‑BFGS] n={n} seed={idx} best={cur_val:.6f}")

        # ------------------------------------------------------------------
        # 4) Exact sub‑gradient refinement on the best candidate for this length
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq = _subgradient_refine(global_best_seq, n, deadline)
            cur_val = _objective(seq)
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={cur_val:.6f}")

        # ------------------------------------------------------------------
        # 5) Light simulated‑annealing fine‑tuning (if time left)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
            cur_val = _objective(seq)
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={cur_val:.6f}")

        # keep the best sequence for the next length (interpolate if needed)
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            previous = global_best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback (very unlikely)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(2000)
        print("[fallback] using uniform distribution")

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    # Convert to plain Python list as required by the grader
    return global_best_seq.tolist()
