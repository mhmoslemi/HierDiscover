"""
Propose a step‑function (non‑negative vector) that minimises
`evaluate_sequence`.  The algorithm consists of three stages:

1. **Warm‑start** – take a simple analytic shape (arcsine) and a few
   cheap candidates, keep the best.
2. **Smoothed projected Adam optimisation** – minimise a smooth
   approximation of the max‑convolution (log‑sum‑exp) on the simplex
   using Adam and Euclidean projection.  The smoothing parameter β is
   increased gradually, turning the smooth objective into the true
   max as optimisation proceeds.
3. **Local stochastic refinement** – simulated‑annealing style random
   mass transfers between two coordinates, accepting moves that lower
   the exact objective or with a temperature‑dependent probability.

The code respects a 1000 s budget, prints progress information and
returns the best vector found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of a linear convolution a*a using an FFT of length `fft_len`."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv.max())


def evaluate_vec(a: np.ndarray, fft_len: int) -> float:
    """Exact score used by `evaluate_sequence` for a normalised vector (sum = 1)."""
    mx = max_convolution(a, fft_len)
    return float(2 * a.size * mx)


def arcsine_distribution(n: int) -> np.ndarray:
    """Discretisation of the continuous arcsine density on a uniform grid."""
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density evaluated at bin centres and normalised."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total

    prints intermediate scores and finally returns the best vector
    (as a Python list).

    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 4000))

    conv_fft_len = 1 << ((2 * n - 1).bit_length())
