"""
Improved step‑function search.

The algorithm proceeds in four stages:
1. **Analytic + random parametric exploration** – systematic grid over a few
   families (product, sum, exponential, cosine‑power, beta) plus a large
   random search over the same families.  The 20 best candidates are kept.
2. **Convex mixing** – exhaustive convex combinations of the top candidates
   (21 mixing ratios) to enlarge the search space without extra gradient work.
3. **Projected sub‑gradient descent with a targeted mass‑transfer move** –
   the true max‑convolution term is used as sub‑gradient; a small amount of
   probability mass is moved from the index with the largest gradient to the
   index with the smallest gradient.  Nesterov momentum, jitter restarts and
   occasional random moves keep the iteration from stagnating.
4. **Soft‑max (log‑sum‑exp) optimisation with Adam** – the max is replaced by a
   temperature‑controlled smooth approximation; the temperature is ramped up
   making the method first explore globally and later focus on the true peak.
5. **Final random refinement** – Gaussian perturbations of the best vector for
   the last few seconds.

All vectors are projected onto the probability simplex after each update,
so the evaluation function `evaluate_sequence` (which is scale‑invariant) can
be called directly on the un‑scaled vector.  The routine respects the
wall‑time budget (default 1000 s) and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using FFT (size = next power of two)."""
    size = _next_pow2(a.shape[0] + b.shape[0] - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (linear convolution)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector on the simplex (∑a = 1, a_i ≥ 0):
        2·n·max(conv(a,a))
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_self(a)
    return float(2.0 * a.shape[0] * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

# Analytic families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    return a / a.sum()


def analytic_cosine(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    return a / a.sum()


def analytic_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta_
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a

def _add_candidate(cand: np.ndarray,
                   best_list: list[tuple[float, np.ndarray]],
                   max_len: int = 20) -> None:
    """Insert a candidate into a bounded best‑list (sorted by score)."""
    sc = compute_score(cand)
    inserted = False
    for idx, (s, _) in enumerate(best_list):
        if sc < s:
            best_list.insert(idx, (sc, cand.copy()))
            inserted = True
            break
    if not inserted and len(best_list) < max_len:
        best_list.append((sc, cand.copy()))
    while len(best_list) > max_len:
        best_list.pop()


def analytic_grid_search(n: int) -> list[tuple[float, np.ndarray]]:
    """Deterministic scan over the five analytic families."""
    top = []
    # product
    p_vals = np.arange(0.1, 3.01, 0.15)
    off_vals = np.concatenate(
        (np.geomspace(1e-3, 0.01, 5), np.arange(0.01, 5.01, 0.3))
    )
    for p in p_vals:
        for off in off_vals:
            _add_candidate(analytic_product(n, p, off), top)
    # sum
    for p in p_vals:
        for off in off_vals:
            _add_candidate(analytic_sum(n, p, off), top)
    # exponential
    lam_vals = np.logspace(-3, 1, num=20)
    for lam in lam_vals:
        _add_candidate(analytic_exponential(n, lam), top)
    # cosine‑power
    p_cos = np.arange(0.5, 8.1, 0.5)
    for p in p_cos:
        _add_candidate(analytic_cosine(n, p), top)
    # beta
    alpha_vals = np.arange(-0.5, 2.51, 0.5)
    beta_vals = np.arange(-0.5, 2.51, 0.5)
    for a in alpha_vals:
        for b in beta_vals:
            _add_candidate(analytic_beta(n, a, b), top)
    # include previous global solution if shape matches
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            if ref.sum() > 0:
                ref = ref / ref.sum()
            else:
                ref = np.ones_like(ref) / n
            _add_candidate(ref, top)
    return top


def mix_top_candidates(top: list[tuple[float, np.ndarray]],
                       n: int) -> np.ndarray:
    """Try convex combinations of the best candidates (21 ratios)."""
    best_seq = top[0][1].copy()
    best_score = top[0][0]
    alphas = np.linspace(0.0, 1.0, 21)
    for i in range(len(top)):
        for j in range(i + 1, len(top)):
            a_i = top[i][1]
            a_j = top[j][1]
            for alpha in alphas:
                cand = alpha * a_i + (1.0 - alpha) * a_j
                cand = project_simplex(cand)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
    return best_seq


def random_parameter_candidate(n: int) -> np.ndarray:

    family = np.random.choice(["product", "sum", "exp", "cos", "beta"])
    if family == "product":
        p = 10 ** np.random.uniform(-1, 0.5)          # ~0.1–3
        off = 10 ** np.random.uniform(-3, 0.7)       # ~1e‑3–5
        return analytic_product(n, p, off)
    if family == "sum":
        p = 10 ** np.random.uniform(-1, 0.5)
        off = 10 ** np.random.uniform(-3, 0.7)
        return analytic_sum(n, p, off)
    if family == "exp":
        lam = 10 ** np.random.uniform(-3, 1)         # log‑space
        return analytic_exponential(n, lam)
    if family == "cos":
        p = np.random.uniform(0.1, 8.0)
        return analytic_cosine(n, p)
    # beta
    a = np.random.uniform(-0.4, 2.0)
    b = np.random.uniform(-0.4, 2.0)

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

    The routine respects a time budget and returns the best vector found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger than the default 2000 often helps)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 3000))
    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # 0) Build an initial vector (up‑sample previous solution if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.asarray(globals()["height_sequence_1"], dtype=float)
        if old.shape[0] != n:
            # linear interpolation to new length
            x_old = np.linspace(0, 1, old.shape[0])
            x_new = np.linspace(0, 1, n)
            init = np.interp(x_new, x_old, old)
        else:
            init = old.copy()
        init = np.maximum(init, 0.0)
        if init.sum() == 0.0:
            init = np.ones(n) / n
        else:
            init = init / init.sum()
    else:
        init = np.ones(n) / n

    # ------------------------------------------------------------------
    # 1) Systematic grid + random sampling over analytic families
    # ------------------------------------------------------------------
    top = analytic_grid_search(n)               # deterministic scan (≈10‑15 s)
    # random sampling (aim for ~200 s)
    while time.time() < deadline - 600:          # reserve >10 min for later phases
        cand = random_parameter_candidate(n)
        _add_candidate(cand, top, max_len=30)
        # occasional mixture with current best to diversify
        if np.random.rand() < 0.05:
            mix = 0.7 * cand + 0.3 * top[0][1]
            mix = project_simplex(mix)
            _add_candidate(mix, top, max_len=30)

    # best after mixing the (up‑to‑30) top candidates
    best_seq = mix_top_candidates(top, n)
    best_score = compute_score(best_seq)
    print(f"[0] best score after mixing = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Projected sub‑gradient descent with targeted mass transfer
    # ------------------------------------------------------------------
    a = best_seq.copy()
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    last_improve = time.time()
    stagnation_limit = 12.0   # seconds without improvement → jitter
    iter_cnt = 0

    # step schedule (large → very small)
    step_start, step_end = 0.8, 1e-5

    while time.time() < deadline - 300:   # keep ~5 min for Adam + refinement
        conv = conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[sg {iter_cnt:07d}] new best = {best_score:.6f}")

        # sub‑gradient of the active max term
        max_idx = int(conv.argmax())
        shift = max_idx - np.arange(n)
        mask = (shift >= 0) & (shift < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[shift[mask]]

        # targeted mass transfer (move from largest gradient to smallest)
        if np.any(g > 0):
            i_big = int(np.argmax(g))
            i_small = int(np.argmin(g))
            if i_big != i_small:
                delta = 0.5 * step_start * a[i_big] * np.random.rand()
                a_temp = a.copy()
                a_temp[i_big] = max(a_temp[i_big] - delta, 0.0)
                a_temp[i_small] += delta
                a_candidate = project_simplex(a_temp)
            else:
                a_candidate = a.copy()
        else:
            a_candidate = a.copy()

        # Nesterov‑style momentum update
        elapsed = time.time() - start_time
        frac = (elapsed) / (budget_s - 300)
        step = step_start * (1.0 - frac) + step_end * frac
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        a_candidate = a_candidate - step * momentum
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate = project_simplex(a_candidate)

        # occasional random move to add diversity
        if np.random.rand() < 0.02:
            a_candidate = random_move(a_candidate, step * 0.5)

        # jitter restart after stagnation
        if time.time() - last_improve > stagnation_limit:
            jitter = np.random.rand(n) * 1e-3
            a_candidate = best_seq + jitter
            a_candidate = project_simplex(a_candidate)
            last_improve = time.time()
            print(f"[sg {iter_cnt:07d}] jitter restart")

        a = a_candidate
        iter_cnt += 1

    # ------------------------------------------------------------------
    # 3) Adam optimisation of a smooth (soft‑max) surrogate
    # ------------------------------------------------------------------
    b = a.copy()
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    t_adam = 0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12
    beta_start, beta_end = 0.5, 500.0
    last_improve = time.time()
    stagnation_limit = 20.0

    while time.time() < deadline - 20:
        conv = conv_self(b)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = b.copy()
            last_improve = time.time()
            print(f"[adam {t_adam:07d}] new best = {best_score:.6f}")

        # temperature schedule (smooth → sharp)
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 20)
        beta = beta_start * (1.0 - progress) + beta_end * progress

        # soft‑max weights over all convolution indices
        w_raw = np.exp(beta * (conv - max_val))   # max exponent = 0
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # weighted shift = Σ_k w_k * a_{k-i}  (correlation)
        b_rev = b[::-1]
        corr = conv_two(w, b_rev)                # length = 3n‑2
        weighted_shift = corr[n - 1:n - 1 + n]    # central n entries

        # gradient of the smooth objective
        g = b + weighted_shift

        # Adam update
        t_adam += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

        b_candidate = b - step_vec
        b_candidate = np.clip(b_candidate, 0.0, None)
        b_candidate = project_simplex(b_candidate)

        # occasional random perturbations
        if np.random.rand() < 0.015:
            b_candidate = random_move(b_candidate, lr * 0.3)
        if np.random.rand() < 0.005:
            jitter = np.random.randn(n) * 5e-4
            b_candidate = b_candidate + jitter
            b_candidate = np.clip(b_candidate, 0.0, None)
            b_candidate = project_simplex(b_candidate)

        # jitter restart on stagnation
        if time.time() - last_improve > stagnation_limit:
            jitter = np.random.rand(n) * 1e-3
            b_candidate = best_seq + jitter
            b_candidate = project_simplex(b_candidate)
            last_improve = time.time()
            print(f"[adam {t_adam:07d}] jitter restart")

        b = b_candidate

    # ------------------------------------------------------------------
    # 4) Final random refinement (Gaussian perturbations)
    # ------------------------------------------------------------------
    print("[refine] starting final random perturbation phase")
    while time.time() < deadline:
        perturb = np.random.normal(scale=5e-4, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, length = {n}, time = {total:.1f}s")
    return best_seq.tolist()
