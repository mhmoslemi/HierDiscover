"""
Step‑function optimiser for the objective

    2·n·max(conv(a,a)) / (sum(a)**2)

Goal
----
* minimise the evaluation function defined in the test harness.
* The algorithm combines:
    1. Broad analytic families (including a new beta‑type family) and
       random parametric sampling.
    2. A continuation of a smooth‑max (log‑sum‑exp) optimiser using Adam,
       with a much longer λ‑schedule than the reference code.
    3. A final short exact sub‑gradient phase for polishing.
    4. Light stochastic mass–transfer moves to escape shallow minima.

The optimiser respects the supplied time budget (default 1000 s) and
prints progress information.  The best sequence found is returned as a
plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a (the sum is ignored,
    the function normalises internally).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families (including a new beta‑type family)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^(‑p)   (edge‑heavy shape)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^(‑p) + (n‑1‑i+off)^(‑p)   (two‑tail shape)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exp_decay(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(‑c·|i‑center|)   (centre‑peaked shape)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - center))
    a = a / a.sum()
    return a


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(‑0.5·((i‑center)/σ)²)   (smooth bump)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = a / a.sum()
    return a


def analytic_beta(n: int, alpha: float, beta: float, offset: float) -> np.ndarray:
    """
    a_i ∝ (i+off)^(α‑1)·(n‑1‑i+off)^(β‑1)   (Beta‑type density, edge‑heavy
    when α,β<1).  The offset keeps the ends finite.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (alpha - 1.0) * (n - 1 - i + offset) ** (beta - 1.0)
    a = np.where(np.isfinite(a) & (a > 0), a, 0.0)
    tot = a.sum()
    if tot == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / tot
    return a


# ----------------------------------------------------------------------
# Smoothed‑max utilities (Φ_λ and its gradient)
# ----------------------------------------------------------------------


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    For a given log‑probability vector x, smoothing λ and length n,
    return the smooth max Φ_λ(a) and its gradient w.r.t. x.
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # convolution a*a
    conv = conv_fft_self(a)                # length 2n‑1
    max_c = np.max(conv)

    # soft‑max over convolution entries (log‑sum‑exp)
    shift = conv - max_c                   # non‑positive
    exp_term = np.exp(lam * shift)        # ≤ 1
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam    # Φ_λ

    w = exp_term / sum_exp                 # (2n‑1) weights, sum to 1

    # gradient of Φ_λ w.r.t. a
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]      # length n
    grad_a = 2.0 * slice_part[::-1]            # ∂Φ/∂a

    # map gradient to x‑space using the Jacobian of soft‑max
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)                # ∇_x Φ = J^T ∇_a Φ
    return phi, grad_x


def adam_optimize(x: np.ndarray, lam: float, n: int, steps: int,
                  lr: float, deadline: float) -> np.ndarray:
    """Simple Adam optimiser for the smooth‑max objective."""
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for t in range(1, steps + 1):
        if time.time() >= deadline:
            break
        _, grad = compute_phi_and_grad(x, lam, n)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step
    return x


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max(conv))
# ----------------------------------------------------------------------


def subgrad_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step for the exact objective
    max(conv(a,a)).  Uses an FFT‑based computation of the sub‑gradient.
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    active = (conv >= max_val - 1e-12).astype(np.float64)   # binary mask
    cnt = active.sum()
    if cnt == 0:
        return a
    w = active / cnt                                         # uniform over active indices
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]            # sub‑gradient w.r.t. a
    a_new = a - step * grad
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the supplied evaluation function.  The routine respects the given time
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # length of the candidate; user may override via kwargs
    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Analytic families + random parametric sampling
    # ------------------------------------------------------------------
    best_a = np.ones(n) / n
    best_score = compute_score(best_a)

    # possibly reuse a previously discovered construction
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n and ref.sum() > 0:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            if sc < best_score:
                best_score, best_a = sc, ref.copy()

    # deterministic grids (product, sum, beta)
    p_vals = np.linspace(0.30, 0.70, 71)            # product / sum power
    offset_vals = np.linspace(0.001, 2.0, 71)       # avoid zero
    alpha_vals = np.linspace(0.10, 0.90, 31)       # beta‑family α
    beta_vals = np.linspace(0.10, 0.90, 31)        # beta‑family β

    # product and sum families
    for mode in ("product", "sum"):
        for p in p_vals:
            for off in offset_vals:
                if time.time() >= deadline:
                    break
                if mode == "product":
                    cand = analytic_product(n, p, off)
                else:
                    cand = analytic_sum(n, p, off)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score, best_a = sc, cand.copy()

    # beta‑type family (asymmetric edge‑heavy)
    for a_param in alpha_vals:
        for b_param in beta_vals:
            if time.time() >= deadline:
                break
            off = np.random.uniform(0.001, 1.0)
            cand = analytic_beta(n, a_param, b_param, off)
            sc = compute_score(cand)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()

    # exponential decay (centre‑peaked) – kept for completeness
    for c in np.linspace(0.0005, 0.0200, 41):
        if time.time() >= deadline:
            break
        cand = analytic_exp_decay(n, c)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    # Gaussian bump family
    for sigma in np.linspace(100.0, 800.0, 41):
        if time.time() >= deadline:
            break
        cand = analytic_gaussian(n, sigma)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    # random sampling of beta parameters (adds diversity)
    rnd_trials = 2000
    for _ in range(rnd_trials):
        if time.time() >= deadline:
            break
        a_param = np.random.uniform(0.05, 1.2)
        b_param = np.random.uniform(0.05, 1.2)
        off = np.random.uniform(0.001, 2.0)
        cand = analytic_beta(n, a_param, b_param, off)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    print(f"[grid] best score after analytic sweep = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Smooth‑max optimisation with Adam (continuation on λ)
    # ------------------------------------------------------------------
    eps = 1e-12
    x = np.log(np.maximum(best_a, eps))

    # λ grows geometrically; we go up to ~1e6 (far larger than previous 8e4)
    lam_vals = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0,
                2000.0, 8000.0, 20000.0, 80000.0, 200000.0,
                800000.0]
    # allocate a generous but safe number of steps per λ (depends on remaining time)
    for lam in lam_vals:
        if time.time() >= deadline:
            break
        # Allow more iterations for the larger λ values where the landscape is
        # sharper.
        remaining = deadline - time.time()
        # Rough estimate of 0.0015 s per Adam iteration (FFT of length ~4 k)
        max_possible = int(remaining / 0.0015)
        # base step budget; we increase it for the last few λ values
        base_steps = 3000
        if lam >= 20000:
            base_steps = 8000
        if lam >= 200000:
            base_steps = 15000
        steps = min(base_steps, max_possible)
        if steps <= 0:
            break

        # small random jitter before each new λ to encourage escaping flat plateaus
        x += np.random.randn(n) * 0.02

        x = adam_optimize(x, lam, n, steps, lr=0.05, deadline=deadline)
        a = softmax(x)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[adam λ={lam:.0e}] new best = {best_score:.6f} "
                  f"(t={time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # 3) Exact sub‑gradient descent (polishing)
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        # step size decreases as 1/√t (standard for sub‑gradients)
        step = 0.8 / np.sqrt(it)
        a = subgrad_step(a, step)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[subgrad iter={it}] new best = {best_score:.6f} "
                  f"(t={time.time() - start_time:.1f}s)")
        # occasional random reshuffle to explore new valleys
        if it % 2500 == 0:
            cand = random_move(a, 0.03)
            sc2 = compute_score(cand)
            if sc2 < best_score:
                best_score, best_a = sc2, cand.copy()
                print(f"[random after {it}] new best = {best_score:.6f} "
                      f"(t={time.time() - start_time:.1f}s)")
                a = cand

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    # expose the best sequence for possible reuse in a later run
    globals()["height_sequence_1"] = best_a.tolist()
    return best_a.tolist()
