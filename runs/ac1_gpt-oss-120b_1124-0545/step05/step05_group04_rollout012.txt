"""
A lightweight multi‑phase optimizer for the step‑function objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The routine works on the probability simplex (the objective is scale‑invariant)
and combines

1. an extensive deterministic sweep over several analytic families
   (product‑law, sum‑law, exponential, Gaussian, Beta, uniform‑mix),

2. a fast random‑parameter search that samples the same families with
   continuous parameters on a logarithmic scale,

3. a short Adam optimisation of a smooth‑max surrogate (helps to
   escape shallow local minima),

4. a projected sub‑gradient descent on the exact max‑convolution
   objective, and finally

5. a light random‑mass‑transfer local search (simulated‑annealing style).

All phases respect a user supplied time budget (default 1000 s) and the
best sequence found is returned as a plain Python list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a tiny amount of probability mass from one index to another.
    Returns a new vector (already projected onto the simplex).
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^(‑p)   (edge‑heavy shape)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.ones(n) / n


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^(‑p) + (n‑1‑i+off)^(‑p)   (two‑tail shape)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.ones(n) / n


def analytic_exp_decay(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(‑c·|i‑center|)   (center‑peaked shape)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - center))
    return a / a.sum()


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(‑0.5·((i‑center)/σ)²)   (smooth bump)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    return a / a.sum()


def analytic_beta(n: int, alpha: float, beta: float, offset: float) -> np.ndarray:
    """
    a_i ∝ (i+off)^{α‑1}·(n‑1‑i+off)^{β‑1}   (Beta‑distribution shape).
    Offsets avoid singularities at the boundaries.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (alpha - 1) * (n - 1 - i + offset) ** (beta - 1)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.ones(n) / n


def analytic_mix_product_uniform(
    n: int, p: float, offset: float, lam: float
) -> np.ndarray:
    """
    Convex mixture of a product‑law shape and the uniform distribution:
        a = λ·product + (1‑λ)·uniform
    """
    prod = analytic_product(n, p, offset)
    uni = np.ones(n) / n
    a = lam * prod + (1.0 - lam) * uni
    return a / a.sum()


# ----------------------------------------------------------------------
# Smoothed‑max utilities (Φ_λ) – same as in the reference code
# ----------------------------------------------------------------------


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    Smooth‑max Φ_λ and its gradient w.r.t. the log‑probability vector `x`.
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    conv = conv_fft_self(a)                # length 2·n‑1
    max_c = np.max(conv)

    shift = conv - max_c                   # ≤ 0
    exp_term = np.exp(lam * shift)        # ≤ 1
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam    # Φ_λ

    w = exp_term / sum_exp                 # weights, sum to 1
    # gradient of Φ_λ w.r.t. a
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad_a = 2.0 * slice_part[::-1]

    # map gradient to the log‑probability space (soft‑max Jacobian)
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return phi, grad_x


def adam_optimize(x: np.ndarray, lam: float, n: int, steps: int,
                  lr: float, deadline: float) -> np.ndarray:
    """Simple Adam optimiser for the smooth‑max surrogate."""
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for t in range(1, steps + 1):
        if time.time() >= deadline:
            break
        _, grad = compute_phi_and_grad(x, lam, n)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step
    return x


# ----------------------------------------------------------------------
# Exact sub‑gradient utilities
# ----------------------------------------------------------------------


def subgrad_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step for max(conv(a,a)).
    Returns a new simplex‑projected vector.
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    active = (conv >= max_val - 1e-12).astype(np.float64)   # binary mask
    cnt = active.sum()
    if cnt == 0:
        return a
    w = active / cnt
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]
    a_new = a - step * grad
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Exact objective (used for internal ranking – the external harness
# calls ``evaluate_sequence`` which is scale invariant).
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a (the sum is ignored).
    Returns 2·n·max(conv(a,a)) after normalising a to sum = 1.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    n = a.shape[0]
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))

        for p in [0.4, 0.5, 0.6]:
            for off in [0.01, 0.1, 1.0]:
                if time.time() >= deadline:
                    break
                cand = analytic_mix_product_uniform(n, p, off, lam)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score, best_a = sc, cand.copy()
                    print(f"[mix] new best = {best_score:.6f}")

    print(f"[grid] best score after analytic sweep = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) randomised parametric search (large number of cheap samples)
    # --------------------------------------------------------------
    max_rand_iters = 8000
    rand_iter = 0
    while time.time() < deadline and rand_iter < max_rand_iters:
        rand_iter += 1
        # choose a family uniformly at random
        kind = np.random.choice(
            ["product", "sum", "exp", "gau", "beta", "mix"]
        )
        if kind == "product":
            p = np.random.uniform(0.2, 1.5)
            off = 10 ** np.random.uniform(-5, 0.5)
            cand = analytic_product(n, p, off)
        elif kind == "sum":
            p = np.random.uniform(0.2, 1.5)
            off = 10 ** np.random.uniform(-5, 0.5)
            cand = analytic_sum(n, p, off)
        elif kind == "exp":
            c = np.random.uniform(0.0002, 0.0300)
            cand = analytic_exp_decay(n, c)
        elif kind == "gau":
            sigma = np.random.uniform(30.0, 1200.0)
            cand = analytic_gaussian(n, sigma)
        elif kind == "beta":
            alpha = np.random.uniform(0.1, 2.0)
            beta = np.random.uniform(0.1, 2.0)
            off = 10 ** np.random.uniform(-5, 0.0)
            cand = analytic_beta(n, alpha, beta, off)
        else:  # mix
            lam = np.random.rand()
            p = np.random.uniform(0.2, 1.5)
            off = 10 ** np.random.uniform(-5, 0.5)
            cand = analytic_mix_product_uniform(n, p, off, lam)

        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[random] new best = {best_score:.6f}")

        # occasional tiny local refinement
        if rand_iter % 50 == 0:
            cand_ref = random_move(cand, scale=0.15)
            sc_ref = compute_score(cand_ref)
            if sc_ref < best_score:
                best_score, best_a = sc_ref, cand_ref.copy()
                print(f"[random‑ref] new best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) smooth‑max optimisation using Adam (continuation on λ)
    # --------------------------------------------------------------
    eps = 1e-12
    x = np.log(np.maximum(best_a, eps))
    lam_schedule = [0.5, 2.0, 10.0, 50.0, 200.0,
                    800.0, 2000.0, 8000.0, 20000.0, 80000.0]
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # very rough estimate of cost per iteration (≈0.001 s)
        max_iters = max(1, int(remaining / 0.001))
        steps = min(2000, max_iters)
        x = adam_optimize(x, lam, n, steps, lr=0.05, deadline=deadline)
        a = softmax(x)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[adam λ={lam:.1f}] new best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) exact sub‑gradient descent on max(conv) (main fine‑tuner)
    # --------------------------------------------------------------
    a = best_a.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        step = 0.5 / np.sqrt(it)           # classic diminishing stepsize
        a = subgrad_step(a, step)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[subgrad iter={it}] new best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) final light random‑mass‑transfer local search (simulated annealing)
    # --------------------------------------------------------------
    a = best_a.copy()
    scale = 0.02
    while time.time() < deadline:
        cand = random_move(a, scale)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[final random] new best = {best_score:.6f}")
            a = cand
            scale = max(0.001, scale * 0.9)
        else:
            scale = min(0.05, scale * 1.001)
            a = cand

    total = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total:.1f}s")
    # Return plain Python list as required by the external harness
    return best_a.tolist()
