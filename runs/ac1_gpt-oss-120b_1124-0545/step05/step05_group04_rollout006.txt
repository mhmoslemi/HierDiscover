import cvxpy as cp

def lp_update(f, M):
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    # construct constraints
    constraints = []
    # For each k compute linear combination
    # Convolution of f with b: (f * b)[k] = sum_i f[i] * b[k-i] for i where 0 <= i < n and 0 <= k-i < n
    # we can precompute index matrix mapping.
    # Use for loop.
    for k in range(2*n-1):
        # compute sum
        indices_i = np.arange(max(0, k-(n-1)), min(n-1, k)+1)
        # corresponding j = k-i
        # linear expression = sum f[i] * b[j]
        expr = cp.sum(cp.multiply(f[indices_i], b[k - indices_i]))
        constraints.append(expr <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    return b.value
```

def solve_lp(f, M):
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    constraints = []
    # Precompute f reversed? Actually constraint: sum_i f[i] * b[k-i] <= M for each k.
    # Using convolution.
    # For each k we can compute indices i where 0<=i<n and 0<=k-i<n => i in [max(0, k-(n-1)), min(n-1,k)]
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np, cvxpy as cp

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    n = kwargs.get('n', 2000)

    # init a
    if 'height_sequence_1' in globals():
        a = np.asarray(globals()['height_sequence_1'], dtype=float)
        if a.shape[0] != n:
            # adapt: if longer, truncate; if shorter, pad with uniform
            if a.shape[0] > n:
                a = a[:n]
            else:
                # pad with zeros then renormalize
                pad = np.zeros(n - a.shape[0])
                a = np.concatenate([a, pad])
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = compute_score(best_a)
    print(f"[init] score = {best_score:.6f}")

    # Phase LP descent
    iteration = 0
    t = 0.2
    while time.time() < deadline and iteration < 15:
        iteration += 1
        # compute M for current best_a
        M = conv_fft_self(best_a).max()
        # solve LP
        b = solve_lp_max_sum(best_a, M, deadline)   # pass deadline to limit solve
        if b is None:

            break
        sum_b = b.sum()
        if sum_b <= 0:
            break
        g = b / sum_b
        # convex combination
        new_a = (1 - t) * best_a + t * g
        new_a = project_simplex(new_a)   # ensure sum=1
        new_score = compute_score(new_a)
        print(f"[lp iter={iteration}] score = {new_score:.6f}")
        if new_score < best_score:
            best_score = new_score
            best_a = new_a
            # try larger step next time?
            t = min(0.5, t * 1.1)
        else:
            # reduce step
            t = max(0.01, t * 0.7)

    # Phase 3: local random search
    a = best_a.copy()
    scale = 0.02
    while time.time() < deadline:

"""
A fast iterative LP‑based descent for the max‑convolution objective.

Idea
----
Given a non‑negative probability vector `a` (∑a = 1) we compute
M = max_k (a*a)_k.  The linear programme

    maximise   1ᵀ b
    subject to  a * b ≤ M      (component‑wise)
                b ≥ 0

produces a new vector `b` whose total mass is as large as possible while
still respecting the same convolution bound.  After normalising `b`
to a probability vector `g` we move a little towards `g`

    a ← (1‑t)·a + t·g ,   t∈(0,1),

which provably never increases the bound `M`.  Repeating this step a few
times drives the max‑convolution down.  The LP is tiny (≈2000 variables,
≈4000 linear constraints) and solved with CVXPY/ECOS.  After the LP
phase a cheap random‑mass‑transfer local search finishes the job.

The whole routine respects a 1000 s budget and returns the best sequence
found.
"""
import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a with itself using FFT.
    Result length = 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a probability vector.
    Returns np.inf for degenerate inputs.
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    O(n log n) algorithm of Duchi et al. (2008).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(len(v)) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one index to another and
    project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def subgrad_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step for the exact max‑convolution
    objective.  Uses an FFT to compute the sub‑gradient.
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    active = (conv >= max_val - 1e-12).astype(np.float64)
    cnt = active.sum()
    if cnt == 0:
        return a
    w = active / cnt
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]
    a_new = a - step * grad
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Linear‑programming step
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1)×n) such that (A @ b)_k = (f * b)_k.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    j = np.arange(n, dtype=np.int64)
    for i, val in enumerate(f):
        A[i + j, j] += val          # adds val to row i+j, column j
    return A


def solve_lp_max_sum(f: np.ndarray, M: float, time_deadline: float):
    """
    Solve   maximise   1ᵀ b   subject to   f * b ≤ M,   b ≥ 0.
    Returns the optimal b as a NumPy array, or None on failure.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = _build_conv_matrix(f)

    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M * np.ones(rows)]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        # ECOS is fast for this size; warm_start helps if called repeatedly
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   eps_abs=1e-9, eps_rel=1e-9, verbose=False)
    except Exception as e:
        print("[LP] solver exception:", e)
        return None

    if b.value is None or prob.status not in ("optimal", "optimal_inaccurate"):
        print("[LP] no feasible solution, status =", prob.status)
        return None
    return np.asarray(b.value, dtype=np.float64)

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the supplied evaluation function.  The routine respects the given time
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # Initialisation – use the best known global if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=float)
        if a.shape[0] != n:
            if a.shape[0] > n:
                a = a[:n]
            else:
                a = np.concatenate([a, np.zeros(n - a.shape[0])])
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    else:
        a = np.ones(n, dtype=float) / n

    best_a = a.copy()
    best_score = compute_score(best_a)
    print(f"[init] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP descent (the core novelty)
    # ------------------------------------------------------------------
    t = 0.20                 # initial blending factor
    lp_iter = 0
    max_lp_iters = 12       # keep total LP time modest

    while time.time() < deadline and lp_iter < max_lp_iters:
        lp_iter += 1
        M = conv_fft_self(best_a).max()
        b = solve_lp_max_sum(best_a, M, deadline)
        if b is None:
            print("[LP] failed – aborting LP phase")
            break
        sum_b = b.sum()
        if sum_b <= 0:
            break
        g = b / sum_b                     # normalise to a probability vector
        # convex combination with adaptive step size
        new_a = (1 - t) * best_a + t * g
        new_a = project_simplex(new_a)    # guarantees sum = 1, non‑negative
        new_score = compute_score(new_a)

        print(f"[LP iter={lp_iter}] score = {new_score:.6f}")

        if new_score < best_score:
            best_score, best_a = new_score, new_a.copy()
            # after an improvement we can be a bit bolder
            t = min(0.5, t * 1.1)
        else:
            # otherwise back off to stay stable
            t = max(0.01, t * 0.7)

    # ------------------------------------------------------------------
    # Phase 2 – short sub‑gradient refinement (optional, cheap)
    # ------------------------------------------------------------------
    a = best_a.copy()
    sub_it = 0
    while time.time() < deadline and sub_it < 8000:
        sub_it += 1
        step = 0.6 / np.sqrt(sub_it)        # diminishing step
        a = subgrad_step(a, step)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[subgrad it={sub_it}] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – light random‑mass‑transfer local search
    # ------------------------------------------------------------------
    a = best_a.copy()
    scale = 0.02
    while time.time() < deadline:
        cand = random_move(a, scale)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            a = cand
            scale = max(0.001, scale * 0.9)
            print(f"[random] new best = {best_score:.6f}")
        else:
            scale = min(0.05, scale * 1.001)
            a = cand

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    # Return a plain Python list as required by the external harness
    return best_a.tolist()
