"""
Propose a non‑negative step function (list of heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1.  Fast analytic grid search over several families (product, sum,
    exponential, cosine‑power, beta) and keep the 10 best candidates.
2.  Convex mixing of the top candidates.
3.  Sub‑gradient descent on the exact max‑convolution.  The sub‑gradient
    is taken as the average of all *active* constraints (all shift
    indices whose value is within a tiny tolerance of the maximum).  After
    each projection onto the probability simplex a tiny 3‑point Gaussian
    smoothing is applied – this encourages smoother shapes and helps to
    lower the true maximum.
4.  Adam optimisation of the smooth soft‑max surrogate with the same
    smoothing step.
5.  One‑dimensional refinement of the beta‑family parameters
    (α,β) by a cheap Nelder‑Mead optimiser – the beta shape usually
    sits very close to the optimum.
6.  Final short random‑perturbation phase.
All phases respect a user‑provided wall‑time budget (default 1000 s)
and the best sequence found so far is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using FFT (fast for many calls)."""
    size = _next_pow2(a.shape[0] + b.shape[0] - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Objective helpers (scale‑invariant – we normalise to sum = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector a (sum = 1)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_self(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex ∑v=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from a random source index to a target."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def smooth_vec(v: np.ndarray) -> np.ndarray:
    """Three‑point Gaussian smoothing (kernel [0.25,0.5,0.25])."""
    kernel = np.array([0.25, 0.5, 0.25])
    # pad with reflection to keep length unchanged
    padded = np.pad(v, (1, 1), mode='reflect')
    smoothed = np.convolve(padded, kernel, mode='valid')
    return smoothed

# Analytic families (return normalised probability vectors)

def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    return a / a.sum()


def analytic_cosine(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    return a / a.sum()


def analytic_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta_
    return a / a.sum()

def _add_candidate(cand: np.ndarray,
                   best_list: list[tuple[float, np.ndarray]],
                   max_len: int = 10) -> None:
    """Insert vector into sorted best‑list (ascending score)."""
    sc = compute_score(cand)
    inserted = False
    for idx, (s, _) in enumerate(best_list):
        if sc < s:
            best_list.insert(idx, (sc, cand.copy()))
            inserted = True
            break
    if not inserted and len(best_list) < max_len:
        best_list.append((sc, cand.copy()))
    while len(best_list) > max_len:
        best_list.pop()


def analytic_grid_search(n: int) -> tuple[np.ndarray, list[tuple[float, np.ndarray]]]:
    """Exhaustive sweep over the five parametric families."""
    top: list[tuple[float, np.ndarray]] = []

    # ----- product family ------------------------------------------------
    p_vals = np.arange(0.1, 3.01, 0.15)
    off_vals = np.concatenate(
        (np.geomspace(1e-3, 0.01, 5), np.arange(0.01, 5.01, 0.3))
    )
    for p in p_vals:
        for off in off_vals:
            cand = analytic_product(n, p, off)
            _add_candidate(cand, top)

    # ----- sum family ----------------------------------------------------
    for p in p_vals:

            _add_candidate(cand, top)

    # ----- exponential tails ---------------------------------------------

        cand = analytic_exponential(n, lam)
        _add_candidate(cand, top)

    # ----- cosine‑power --------------------------------------------------
    p_cos = np.arange(0.5, 8.1, 0.5)

    beta_vals = np.arange(-0.5, 2.51, 0.5)

    # ----- any previously found global reference -------------------------

            ref = ref / ref.sum()
            _add_candidate(ref, top)

    best_seq = min(top, key=lambda x: x[0])[1].copy()
    return best_seq, top


def mix_top_candidates(top: list[tuple[float, np.ndarray]],
                       n: int) -> np.ndarray:
    """Exhaustively try convex mixes of the top‑10 candidates."""
    best_seq = top[0][1].copy()
    best_score = top[0][0]
    alphas = np.linspace(0.0, 1.0, 21)    # 0,0.05,…,1

                sc = compute_score(cand)
                if sc < best_score:

                    best_seq = cand.copy()
    return best_seq

def subgradient_step(a: np.ndarray,
                     step: float,
                     tol_rel: float = 1e-9) -> np.ndarray:
    """
    One projected sub‑gradient step.
    The sub‑gradient is the average of all active max‑convolution constraints.
    """
    n = a.shape[0]
    conv = conv_self(a)
    max_val = conv.max()
    # indices whose value is within tol of the max
    tol = max(tol_rel * max_val, 1e-12)
    active = np.where(conv >= max_val - tol)[0]

    g = np.zeros_like(a)
    for k in active:
        shift = k - np.arange(n)
        mask = (shift >= 0) & (shift < n)
        # contribution 2*a[shift] for each active k
        g[mask] += 2.0 * a[shift[mask]]
    if active.size > 0:
        g /= float(active.size)

    # Nesterov‑style momentum (simple heavy‑ball)
    a_new = a - step * g
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_simplex(a_new)
    # tiny Gaussian smoothing to encourage smoothness
    a_new = smooth_vec(a_new)
    a_new = project_simplex(a_new)
    return a_new

# Adam optimisation of the soft‑max surrogate (with smoothing)
# ----------------------------------------------------------------------


def adam_step(a: np.ndarray,
              m: np.ndarray,
              v: np.ndarray,
              t: int,
              lr: float,
              beta1: float,
              beta2: float,
              eps: float,
              beta_temp: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Perform one Adam optimisation step on the log‑sum‑exp surrogate.
    Returns (new_a, new_m, new_v).
    """
    n = a.shape[0]
    conv = conv_self(a)
    max_val = conv.max()

    # soft‑max weights (temperature beta_temp)
    w_raw = np.exp(beta_temp * (conv - max_val))
    w_sum = w_raw.sum()
    w = w_raw / w_sum if w_sum != 0 else np.full_like(w_raw, 1.0 / w_raw.size)

    # weighted shift (correlation of w with reversed a)
    a_rev = a[::-1]
    corr_full = conv_two(w, a_rev)               # length = 3n‑2
    weighted_shift = corr_full[n - 1:n - 1 + n]   # length n

    # gradient of the surrogate: g = a + weighted_shift
    g = a + weighted_shift

    # Adam moments
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

    a_new = a - step_vec
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_simplex(a_new)
    a_new = smooth_vec(a_new)            # same smoothing as sub‑grad
    a_new = project_simplex(a_new)
    return a_new, m, v

def refine_beta_params(a: np.ndarray, n: int, budget_s: float) -> np.ndarray:
    """
    Fit a beta distribution (α,β) to the current vector a (via log‑linear
    regression) and then locally optimise the two parameters with Nelder‑Mead.
    The optimisation is bounded to a reasonable rectangle.
    """
    # ------------------------------------------------------------------
    # 1) obtain a cheap LS estimate for (α,β)
    # ------------------------------------------------------------------
    idx = np.where(a > 0)[0]
    if idx.size < 2:
        return a  # nothing to improve
    log_a = np.log(a[idx])
    X = np.column_stack((
        np.log(idx + 1.0),
        np.log(n - idx)
    ))
    # solve log a ≈ c + α*log(i+1) + β*log(n-i)
    coeff, *_ = np.linalg.lstsq(X, log_a, rcond=None)
    alpha0, beta0 = coeff[0], coeff[1]

    # ------------------------------------------------------------------
    # 2) local optimisation (Nelder‑Mead)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import minimize
    except Exception:          # pragma: no cover
        return a                # scipy unavailable – skip this phase

    def obj_xy(xy):
        alpha, beta_ = xy
        cand = analytic_beta(n, alpha, beta_)
        return compute_score(cand)

    bounds = [(-1.5, 5.0), (-1.5, 5.0)]

    # A very modest budget – we stop once the wall‑time budget is consumed
    local_start = time.time()
    res = minimize(obj_xy,
                   x0=np.array([alpha0, beta0]),
                   method='Nelder-Mead',
                   bounds=bounds,
                   options={'maxfev': 200, 'xatol': 1e-5, 'fatol': 1e-5})
    # If optimisation succeeded, return the refined beta shape
    if res.success:
        refined = analytic_beta(n, res.x[0], res.x[1])
        if compute_score(refined) < compute_score(a):
            return refined
    return a

        cur_score = 2.0 * n * max_val

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[sg {iter_cnt:07d}] new best = {best_score:.6f}")

        # sub‑gradient with active set
        tol = max(1e-9 * max_val, 1e-12)
        active = np.where(conv >= max_val - tol)[0]
        g = np.zeros_like(a)
        for k in active:
            shift = k - np.arange(n)
            mask = (shift >= 0) & (shift < n)
            g[mask] += 2.0 * a[shift[mask]]
        if active.size > 0:
            g /= float(active.size)

        # momentum & step‑size schedule
        step_frac = (time.time() - start_time) / (budget_s - 260.0)
        step = step_start * (1.0 - step_frac) + step_end * step_frac
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        a_candidate = a - step * momentum
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate = project_simplex(a_candidate)
        a_candidate = smooth_vec(a_candidate)
        a_candidate = project_simplex(a_candidate)

        # occasional random mass‑transfer move
        if np.random.rand() < 0.02:
            a_candidate = random_move(a_candidate, step * 0.5)

        # jitter restart on stagnation
        if time.time() - last_improve > stagnation_time:
            jitter = np.random.rand(n) * 1e-3
            a_candidate = best_seq + jitter
            a_candidate = project_simplex(a_candidate)
            a_candidate = smooth_vec(a_candidate)
            a_candidate = project_simplex(a_candidate)
            last_improve = time.time()
            print(f"[sg {iter_cnt:07d}] jitter restart")

        a = a_candidate
        iter_cnt += 1

    # --------------------------------------------------------------
    # 2) Adam on soft‑max surrogate (with smoothing)
    # --------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_adam = 0
    lr = 0.12
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta_start, beta_end = 0.5, 400.0
    last_improve = time.time()
    stagnation_time = 18.0

    while time.time() < deadline - 40:
        conv = conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[adam {t_adam:07d}] new best = {best_score:.6f}")

        # temperature schedule
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 45.0)
        beta_temp = beta_start * (1.0 - progress) + beta_end * progress

        a, m, v = adam_step(a, m, v, t_adam + 1, lr,
                            beta1, beta2, eps, beta_temp)
        t_adam += 1

        # occasional jitter / random move
        if np.random.rand() < 0.015:
            a = random_move(a, lr * 0.3)
        if np.random.rand() < 0.005:
            jitter = np.random.randn(n) * 5e-4
            a = np.clip(a + jitter, 0.0, None)
            a = project_simplex(a)
            a = smooth_vec(a)
            a = project_simplex(a)

        if time.time() - last_improve > stagnation_time:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            a = smooth_vec(a)
            a = project_simplex(a)
            last_improve = time.time()
            print(f"[adam {t_adam:07d}] jitter restart")

    # --------------------------------------------------------------
    # 3) Local refinement of the beta‑family (if applicable)
    # --------------------------------------------------------------
    a = refine_beta_params(best_seq, n, budget_s)
    cand_score = compute_score(a)
    if cand_score < best_score:
        best_score = cand_score
        best_seq = a.copy()
        print(f"[beta‑refine] improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final random perturbation phase (last few seconds)
    # --------------------------------------------------------------
    print("[refine] final random perturbation phase")
    while time.time() < deadline:
        perturb = np.random.normal(scale=5e-4, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, length = {n}, time = {total:.1f}s")
    return best_seq.tolist()
