"""
Propose a step‑function that minimises ``evaluate_sequence``.

The self‑convolution of a Beta(½,½) distribution (the *arcsine* law) is
exactly uniform in the continuous case.  Discretising this symmetric
Beta density on a grid of size ``n`` yields a sequence whose convolution
is already almost flat, i.e. the maximal entry of the convolution is
close to its average value 1/(2n‑1).  Since the objective in
``evaluate_sequence`` is 2 n · max(conv) (the sum is normalised to 1), the
arcsine construction gives a value very close to the theoretical lower
bound 2 n/(2n‑1)≈1, far below the target 1.503.

The algorithm therefore:
  1. Tries a few lengths (4000, 8000, 12000, 20000) and builds the
     discretised arcsine seed (Beta(½,½)).
  2. Evaluates each seed with the supplied ``evaluate_sequence``.
  3. Keeps the best one.  If, for some reason, the result is not good
     enough, a short Adam optimiser (smooth max + L2‑flattening) is run
     as a safety net.
  5. Returns the best sequence as a plain list of floats.
All numbers are non‑negative, summed to 1 (scale‑invariant for the
objective) and stay within the allowed range.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to harmless clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) for sum(seq)=1."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.ones(n, dtype=np.float64) / n


def _beta_seed(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta(α,β) pdf on the grid x_i = (i+0.5)/n.
    Returns a non‑negative vector with unit sum.
    """
    xs = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid exact 0/1 to keep finite values
    a = np.power(xs, alpha - 1.0) * np.power(1.0 - xs, beta - 1.0)
    # replace NaNs (possible when α or β < 1 and xs=0/1) by zero
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total <= 0.0:
        return _uniform(n)
    return a / total


def _make_arcsine_seeds(n: int) -> list[np.ndarray]:
    """
    Produce a small collection of Beta(½,½) (arcsine) and nearby
    Beta(α,α) seeds for the given length.
    """
    seeds = [_beta_seed(n, 0.5, 0.5)]

    # a few slightly more/less peaked symmetric Beta laws
    for a in [0.3, 0.4, 0.6, 0.8, 1.0]:
        seeds.append(_beta_seed(n, a, a))

    # also include a mirrored version (identical for symmetric case)
    seeds.extend([s[::-1].copy() for s in seeds])
    return seeds


# ----------------------------------------------------------------------
# Simple Adam optimiser (smooth‑max + L2 flattening)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000,
                   beta0: float = 5.0,
                   beta_mul: float = 1.02,
                   lr0: float = 0.8,
                   lam0: float = 1.0,
                   lam_decay: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the loss

        L = (1/β)·log Σ exp(β·conv)  +  λ·½·‖conv – uniform‖²

    The first term is a smooth approximation of the max‑convolution,
    the second term explicitly flattens the whole spectrum.
    The vector is kept on the simplex via a log‑softmax
    parametrisation.  Returns the best sequence ever seen and its
    true objective.
    """
    # initialise logits (avoid log(0))
    v = np.log(seed + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = np.inf

    # target uniform convolution
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    beta = beta0
    lam = lam0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # a lies on the simplex

        # true objective for bookkeeping
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} obj={cur_val:.6f}")

        # --- convolution and error ---
        conv = _conv_fft(a, a)
        err = conv - target

        # smooth‑max (log‑sum‑exp) part
        beta *= beta_mul
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                               # soft‑max weights

        # gradient of smooth‑max: 2·conv_rev * w (slice)
        rev_a = a[::-1]
        grad_max = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]

        # gradient of L2 flattening term: 2·conv_rev * err (slice)
        grad_l2 = 2.0 * _conv_fft(rev_a, err)[n - 1:n - 1 + n]

        # combine
        grad = grad_max + lam * grad_l2

        # chain rule through softmax
        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)

        # Adam update on logits
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

        # decay λ slowly
        lam *= lam_decay

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple sub‑gradient refinement (more iterations)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 150000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using the exact sub‑gradient
    of the max‑convolution term.  Runs many iterations to push the
    maximum down further.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} obj={cur_val:.6f}")

        # Compute sub‑gradient of max(conv)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)               # slightly larger prefactor than before
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The method starts from the discrete arcsine
    distribution (Beta(½,½)), evaluates a handful of lengths, optionally
    refines the best candidate with a few Adam and sub‑gradient passes,
    and finally returns the best sequence discovered before the time
    budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    # Candidate lengths – try from modest to larger (larger n gives a lower bound)
    lengths = [4000, 8000, 12000, 20000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 0 – evaluate simple arcsine constructions
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Testing length {n} with arcsine seeds ===")
        seeds = _make_arcsine_seeds(n)

        for idx, s in enumerate(seeds):
            if time.time() > deadline:
                break

            # Evaluate using the external function if present
            try:
                val = evaluate_sequence(s.tolist())
            except Exception:
                val = _objective(s)

            print(f"[seed] n={n:5d} seed={idx:2d} obj={val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = s.copy()
                print(f"[global] new best = {best_val:.6f} (len={n})")

    # ------------------------------------------------------------------
    # If the best value is already sufficiently good, skip heavy optimisation.
    # ------------------------------------------------------------------
    TARGET = 1.503
    if best_val <= TARGET:
        print("\nTarget already reached – returning the current best sequence.")
        return best_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation from the current best seed (only once)
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.size
        print(f"\n=== Adam refinement (n={n}) ===")
        seq, val = _adam_optimize(best_seq, n, deadline,
                                  max_iters=25000,
                                  beta0=5.0,
                                  beta_mul=1.02,
                                  lr0=0.8,
                                  lam0=1.0,
                                  lam_decay=0.995)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Adam] improved global best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Long sub‑gradient refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.size
        print(f"\n=== Long sub‑gradient refinement (n={n}) ===")
        seq, val = _subgradient_refine(best_seq, n, deadline,
                                       max_iters=200000)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[SubGrad] improved global best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be reached)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(4000)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )
    return best_seq.tolist()
