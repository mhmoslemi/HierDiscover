def solve_lp_for_f(f, M):
    n = len(f)
    cobj = -np.ones(n)  # maximize sum(g) => minimize -sum(g)
    # Build constraint matrix
    c = np.concatenate([f, np.zeros(n-1)])  # length 2n-1
    r = np.concatenate([f[:1], np.zeros(n-1)])
    A = toeplitz(c, r)  # shape (2n-1, n)
    # upper bound: A @ g <= M
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(cobj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    return g

"""
Improved search for a non‑negative step function minimizing

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1.  Analytic grid search over several parametric families (product, sum,
    exponential, cosine‑power, beta) – the 10 best candidates are kept.
2.  Randomised fine‑search of the product and sum families (hill‑climbing
    on the two parameters) to improve the best grid candidates.
3.  Alternating linear‑program (LP) improvement:
        given the current f, solve   max Σ g_j
        s.t.          f * g ≤ M   (M = max(f*f))
        g ≥ 0.
    The LP is linear in g, thus easily solvable with scipy.optimize.linprog.
    If the scaled solution ĝ = g / Σg improves the objective it replaces f.
    A few iterations (≈5–8) are performed.
4.  Sub‑gradient descent with Nesterov momentum (few hundred iterations) to
    polish the solution.
5.  Final short‑run of random Gaussian perturbations (simple local search)
    until the wall‑time budget expires.

All vectors are kept on the probability simplex (∑a_i = 1), because the
objective is scale‑invariant.  The routine respects a 1000 s wall‑time
budget and returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using FFT (full output)."""
    size = _next_pow2(a.shape[0] + b.shape[0] - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length, no padding)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector a (not necessarily normalised).
    Returns 2·n·max(conv(a,a)) / (sum(a)**2).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_self(a)
    return float(2.0 * a.shape[0] * conv.max())   # denominator = 1²


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families (return normalised probability vectors)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    return a / a.sum()


def analytic_cosine(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    return a / a.sum()


def analytic_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta_
    return a / a.sum()

def _add_candidate(
    cand: np.ndarray,
    best_list: list[tuple[float, np.ndarray]],
    max_len: int = 10,
) -> None:
    """Insert cand into a sorted best‑list (ascending by score)."""
    sc = compute_score(cand)
    inserted = False
    for idx, (s, _) in enumerate(best_list):
        if sc < s:
            best_list.insert(idx, (sc, cand.copy()))
            inserted = True
            break
    if not inserted and len(best_list) < max_len:
        best_list.append((sc, cand.copy()))
    while len(best_list) > max_len:
        best_list.pop()


def analytic_grid_search(n: int) -> tuple[np.ndarray, list[tuple[float, np.ndarray]]]:
    """Exhaustive grid scan over all analytic families; keep top‑10."""
    top: list[tuple[float, np.ndarray]] = []

    # ---- product family -------------------------------------------------
    p_vals = np.arange(0.1, 3.01, 0.15)
    off_vals = np.concatenate(
        (np.geomspace(1e-3, 0.01, 5), np.arange(0.01, 5.01, 0.3))
    )
    for p in p_vals:
        for off in off_vals:
            cand = analytic_product(n, p, off)
            _add_candidate(cand, top)

    # ---- sum family -----------------------------------------------------
    for p in p_vals:
        for off in off_vals:
            cand = analytic_sum(n, p, off)
            _add_candidate(cand, top)

    # ---- exponential tails ------------------------------------------------
    lam_vals = np.logspace(-3, 1, num=20)
    for lam in lam_vals:
        cand = analytic_exponential(n, lam)
        _add_candidate(cand, top)

    # ---- cosine‑power ----------------------------------------------------
    p_cos = np.arange(0.5, 8.1, 0.5)
    for p in p_cos:
        cand = analytic_cosine(n, p)
        _add_candidate(cand, top)

    # ---- beta family -----------------------------------------------------
    alpha_vals = np.arange(-0.5, 2.51, 0.5)
    beta_vals = np.arange(-0.5, 2.51, 0.5)
    for a in alpha_vals:
        for b in beta_vals:
            cand = analytic_beta(n, a, b)
            _add_candidate(cand, top)

    # ---- include the previously found sequence if available ------------
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            _add_candidate(ref, top)

    best_seq = min(top, key=lambda x: x[0])[1].copy()
    return best_seq, top


def mix_top_candidates(
    top: list[tuple[float, np.ndarray]],
    n: int,
) -> np.ndarray:
    """Convexly mix the best few candidates (grid search of alphas)."""
    best_seq = top[0][1].copy()
    best_score = top[0][0]
    alphas = np.linspace(0.0, 1.0, 21)  # 0,0.05,…,1
    for i in range(len(top)):
        for j in range(i + 1, len(top)):
            a_i = top[i][1]
            a_j = top[j][1]
            for alpha in alphas:
                cand = alpha * a_i + (1.0 - alpha) * a_j
                cand = project_simplex(cand)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
    return best_seq

def random_family_search(
    family_func,
    n: int,
    iters: int,
    param_bounds: list[tuple[float, float]],
    best_seq: np.ndarray,
) -> np.ndarray:
    """
    Simple random search over the 2‑parameter families.
    param_bounds = [(p_min, p_max), (off_min, off_max)].
    """
    best_score = compute_score(best_seq)
    best_vec = best_seq.copy()
    p_min, p_max = param_bounds[0]
    off_min, off_max = param_bounds[1]

    for _ in range(iters):
        p = np.random.uniform(p_min, p_max)
        off = np.random.uniform(off_min, off_max)
        cand = family_func(n, p, off)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
    return best_vec

def solve_lp_for_f(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ g_j   s.t.   f * g ≤ M   ,   g ≥ 0.
    Returns the optimal g (unscaled) or None on failure.
    """
    if not SCIPY_AVAILABLE:
        return None

    n = f.shape[0]
    # Build convolution matrix: rows = (f*g)_k, cols = g_j
    c = np.concatenate([f, np.zeros(n - 1)])          # first column of Toeplitz
    r = np.concatenate([f[:1], np.zeros(n - 1)])     # first row of Toeplitz
    A = toeplitz(c, r)                                # shape (2n‑1, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Objective: maximize sum(g)  → minimize -sum(g)
    obj = -np.ones(n, dtype=np.float64)

    res = linprog(
        obj,
        A_ub=A,
        b_ub=b,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    return res.x

def local_mass_transfer(
    a: np.ndarray,
    time_deadline: float,
    max_moves: int = 5000,
    scale: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Perform cheap random mass‑transfer moves while time permits.
    Returns the best vector found and its score.
    """
    best = a.copy()
    best_score = compute_score(best)
    n = a.shape[0]

    while time.time() < time_deadline and max_moves > 0:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = scale * best[i] * np.random.rand()
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0.0:  # keep feasibility
            continue
        sc = compute_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
        max_moves -= 1
    return best, best_score

            if cand_score < best_score:
                best_seq = g_scaled
                best_score = cand_score
                print(f"[lp {it}] improved to {best_score:.6f}")
                continue
            # try a few convex mixes if the pure g does not improve
            improved = False
            for t in (0.2, 0.5, 0.8):
                cand = (1 - t) * best_seq + t * g_scaled
                cand = project_simplex(cand)
                sc = compute_score(cand)
                if sc < best_score:
                    best_seq = cand
                    best_score = sc
                    improved = True
                    print(f"[lp mix {it}] t={t:.2f} → {best_score:.6f}")
                    break
            if not improved:
                print(f"[lp {it}] no improvement – stop LP loop")
                break

    # -------------------------------------------------------------
    # 4. Sub‑gradient descent with Nesterov momentum (short run)
    # -------------------------------------------------------------
    a = best_seq.copy()
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    step_start, step_end = 0.5, 1e-6
    sub_start = time.time()
    sub_deadline = start + budget_s * 0.75
    it_cnt = 0
    last_improve = sub_start

    while time.time() < sub_deadline:
        conv = conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[sg {it_cnt:07d}] new best = {best_score:.6f}")

        # sub‑gradient corresponding to the (possibly multiple) active max entries
        active = np.where(conv >= max_val - 1e-12)[0]
        g = np.zeros_like(a)
        for idx in active:
            shift = idx - np.arange(n)
            mask = (shift >= 0) & (shift < n)
            g[mask] += 2.0 * a[shift[mask]]
        g /= max(1, len(active))  # average if several active points

        # Nesterov step schedule (linear decay)
        elapsed = time.time() - start
        progress = elapsed / (budget_s - 50.0)
        step = step_start * (1.0 - progress) + step_end * progress

        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        a_candidate = a - step * momentum
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate = project_simplex(a_candidate)

        # occasional random jitter to avoid stagnation
        if np.random.rand() < 0.02:
            a_candidate = random_move(a_candidate, step * 0.5)

        a = a_candidate
        it_cnt += 1

        if time.time() - last_improve > 12.0:
            # small jitter restart
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            last_improve = time.time()
            print(f"[sg {it_cnt}] jitter restart")

    # -------------------------------------------------------------
    # 5. Final random Gaussian perturbation (local search)
    # -------------------------------------------------------------
    print("[refine] entering final perturbation phase")
    while time.time() < deadline:
        sigma = 5e-4
        perturb = np.random.normal(scale=sigma, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total = time.time() - start
    print(f"[done] best score = {best_score:.6f}, n = {n}, time = {total:.1f}s")
    return best_seq.tolist()
