function solve_lp(f, M):
    n = len(f)
    # create sparse matrix A (2n-1, n)
    # Use lists to accumulate
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs from i_start to i_end inclusive
        # j = k - i
        # coefficient f[i]
        # Append row repeated for each j
        # Could vectorize: generate array of i's.
        # Simpler: loop.
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The optimiser works in three stages:

1. **Initialisation** – start from the best known construction
   (the global ``height_sequence_1`` if present, otherwise a small
   collection of classic window functions).  The vector is normalised
   to sum = 1 and a tiny floor (1e‑12) is added so that every entry is
   strictly positive.

2. **Linear‑programming refinement** – for a fixed sequence ``f`` we
   compute ``M = max(conv(f,f))`` and solve the LP

        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   for all k = 0…2n‑2
                    g_i ≥ 0

   (the convolution is linear in ``g``).  The LP is solved with
   ``scipy.optimize.linprog`` on a sparse‐Toeplitz matrix that represents
   the convolution.  The solution ``g`` is normalised back to sum = 1,
   then mixed with the current vector:

        f ← (1‑t)·f + t·g                      (t ≈ 0.2, reduced if no improvement)

   The process is repeated while time permits (typically a handful of
   iterations).  The mixing step always keeps the sum constant and
   a symmetrisation ``f ← (f+f[::-1])/2`` is applied to stay in the
   symmetric subspace where the optimum is believed to lie.

3. **Optional polishing** – if any time is left a few steps of the
   exact sub‑gradient mirror‑descent (the code used previously) are run
   to squeeze out a final discount.

All phases respect the supplied 1000 s budget.  The routine returns the
best vector found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (LP + sparse matrices)
# ----------------------------------------------------------------------
try:
    import scipy.sparse as sp
    import scipy.optimize as opt
    SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Low‑level convolution utilities (real FFT, full linear convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used in the competition (no clipping)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Classic window families – used as seeds if no previous result exists
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _make_initial_seq(n: int) -> np.ndarray:
    """Return a good starting vector (previous result if available)."""
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            seq = np.asarray(previous, dtype=np.float64)
            if seq.shape == (n,):
                seq = np.clip(seq, 0.0, None)
                if seq.sum() > 0:
                    seq = seq / seq.sum()
                    return seq
        except Exception:
            pass

    # otherwise pick the best among a tiny family of windows
    candidates = [
        _uniform(n),
        _hann_window(n),
        _hamming_window(n),
        _blackman_window(n),
        _tent_window(n),
    ]

    best = None
    best_val = np.inf
    for cand in candidates:
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best = cand
    return best


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine (convolution constraints)
# ----------------------------------------------------------------------


def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n sparse matrix A such that A @ g  =  f * g
    (full linear convolution).  The matrix has constant diagonals
    equal to the entries of ``f``.
    """
    n = f.shape[0]
    # Offsets are negative – each diagonal k = -d contains value f[d]
    offsets = -np.arange(n, dtype=np.int64)          # 0, -1, -2, … , -(n-1)
    # Each diagonal is a constant array of length n (the number of columns)
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP  maximise Σ g_i  s.t.  (f * g)_k ≤ M,  g_i ≥ 0.
    Returns the optimal ``g`` (may be all‑zero in pathological cases).
    """
    if not SCIPY_AVAILABLE:
        raise RuntimeError("SciPy is required for the LP refinement.")

    n = f.shape[0]
    A = _conv_matrix(f)                     # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ g_i  → minimise −Σ g_i
    c = -np.ones(n, dtype=np.float64)

    # No explicit upper bounds – the convolution constraints keep
    # each variable finite.  A tiny bound (1.0) is added to keep the
    # problem well‑posed when some f_i are extremely small.
    bounds = [(0.0, 1.0) for _ in range(n)]

    # HiGHS (default) works well with large sparse systems.
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method="highs", options={"presolve": True})
    # Cleanup the matrix as soon as we are done – helps the GC.
    del A

    if not res.success:
        # Fall back to a uniform vector if the LP failed.
        return np.full(n, 1.0 / n, dtype=np.float64)

    g = res.x
    # Normalise to sum = 1 (the LP maximises the sum, but we will
    # subsequently mix with the current f which already has sum = 1).
    s = g.sum()
    if s > 0:
        g = g / s
    else:
        g = np.full_like(g, 1.0 / n)
    return g


# ----------------------------------------------------------------------
# Mirror‑descent (sub‑gradient) refinement – short polishing step
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        deadline: float,
                        max_iters: int = 200_000) -> np.ndarray:
    """
    Exact mirror‑descent on the true non‑smooth objective.
    Used only as a final polish when a few seconds remain.
    """
    a = start_seq.copy()
    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        if len(active) > 0:
            g /= float(len(active))

        lr = 0.9 / math.sqrt(it + 1.0)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Main entry point – the function that the grader calls
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  Returns the best sequence found as a plain
    list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # keep a safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Choose a length and an initial sequence
    # ------------------------------------------------------------------
    n = 4000                      # a good trade‑off between expressivity and speed
    f = _make_initial_seq(n)      # sum(f) == 1, all entries > 0

    # tiny floor to avoid exact zeros (needed for the LP)
    eps = 1e-12
    f = np.clip(f, eps, None)
    f = f / f.sum()

    best_seq = f.copy()
    best_val = _objective(best_seq)
    print(f"[Init] length={n} objective={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  LP‑based refinement loop
    # ------------------------------------------------------------------
    if SCIPY_AVAILABLE:
        t = 0.2                         # initial mixing coefficient
        lp_iters = 0
        while time.time() < deadline and lp_iters < 12:
            lp_iters += 1
            # Compute current max convolution (M)
            conv_f = _conv_fft(f, f)
            M = conv_f.max()

            # Solve LP for a direction g
            g = _lp_max_sum(f, M)

            # Mix and enforce symmetry + simplex
            mixed = (1.0 - t) * f + t * g
            mixed = (mixed + mixed[::-1]) * 0.5          # symmetrise
            mixed = np.clip(mixed, eps, None)
            mixed /= mixed.sum()

            new_val = _objective(mixed)

            if new_val < best_val:
                best_val = new_val
                best_seq = mixed.copy()
                f = mixed
                print(f"[LP] iter={lp_iters:2d} t={t:.3f} best={new_val:.6f}")
                # If things improve we can be a bit bolder with t
                t = min(0.4, t * 1.1)
            else:
                # No improvement – back off the step size
                t *= 0.5
                if t < 1e-3:
                    # step became too tiny, quit the LP phase
                    break
                # keep the old f and try again
                print(f"[LP] iter={lp_iters:2d} no improvement, shrinking t→{t:.3f}")
    else:
        print("[Info] SciPy not available – skipping LP refinement.")

    # ------------------------------------------------------------------
    # 3️⃣  Short sub‑gradient polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Polish] running sub‑gradient refinement")
        best_seq = _subgradient_refine(best_seq, deadline, max_iters=100_000)

    # ------------------------------------------------------------------
    # 4️⃣  Return the best sequence discovered
    # ------------------------------------------------------------------
    final_obj = _objective(best_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {len(best_seq)})")
    # Store for potential reuse by a later run
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
