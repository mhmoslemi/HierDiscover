def _lp_step(f):
    n = len(f)
    M = np.max(np.convolve(f, f))
    # build A
    rows = 2*n - 1
    A = np.empty((rows, n), dtype=np.float64)
    # fill zeros
    A.fill(0.0)
    for j in range(n):
        A[j:j+n, j] = f
    # constraints
    b_ub = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return None
    g = b / sum_b
    return g
```

"""
search_step_function.py

A fast, multi‑phase optimiser for the “max‑convolution” objective
    2·n·max(conv(a,a)) / (∑a)²
with a≥0.  The routine works inside the given time budget (default 1000 s) and
returns the best sequence found.

Main ideas
-----------
1. **Start from a reasonable seed** – a random Dirichlet draw (≈‑uniform mass
   distribution) or the previously discovered global sequence if it exists.

2. **Exact sub‑gradient mirror descent** – the objective after normalising
   ∑a=1 is 2·n·max(conv).  The sub‑gradient of the max‑convolution is cheap
   (only the pair of indices that realise the current maximum).  An
   exponentiated‑gradient (mirror‑descent) update keeps the iterate on the
   simplex.  We run many iterations (≈ 2·10⁵) because each step is O(n).

3. **Linear‑programming acceleration (optional)** – given the current f,
   maximise ∑b subject to f∗b ≤ M (M = max(f∗f)) and b≥0.  The LP has
   2·n‑1 linear constraints and n variables and can be solved with
   ``scipy.optimize.linprog`` (HiGHS).  The solution b is normalised and
   convex‑combined with f.  This step quickly drives the sequence toward a
   lower peak.

4. **Long simulated‑annealing (SA) hill‑climb** – finally we perform a very
   thorough SA that repeatedly redistributes the total mass of a small random
   subset of indices (2‑4 positions) using a Dirichlet proposal.  The
   temperature is decayed very slowly (factor 0.9999) so that the search can
   escape shallow minima even near the end of the budget.

All phases respect the global deadline; whichever phase finishes last simply
returns the best sequence recorded so far.  The objective is evaluated with
``_objective`` which assumes a normalised ℓ₁‑norm – this matches
``evaluate_sequence`` after the same normalisation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming ∑seq == 1.
    This is exactly the value returned by ``evaluate_sequence`` after the same
    normalisation (the denominator cancels because the objective is scale‑invariant).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Sub‑gradient mirror descent (exact max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the true max‑convolution.
    Returns the best sequence and its objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

        # --- sub‑gradient of max_k conv_k w.r.t. a --------------------
        # Only the pairs (i, j) with i+j = max_idx contribute.
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # --- mirror‑descent update (exponentiated gradient) ----------
        lr = 0.6 / np.sqrt(it)                 # step‑size decays ∝ 1/√t
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                 # stabilise numerics
        a = np.exp(log_a)
        a /= a.sum()                           # stay on the simplex

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill climbing
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.03,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Stochastic neighbourhood search.
    In each move a random subset of 2‑4 indices reshuffles its total mass using
    a Dirichlet draw (the overall ℓ₁‑norm stays constant).  The temperature
    decays geometrically with factor 0.9999 (very slow cooling) so the search
    can keep exploring late in the budget.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)   # start temperature

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # very slow geometric cooling
        T = max(T * 0.9999, min_temp)

        # pick a tiny random subset to reshuffle
        k = rng.integers(2, 5)                 # 2,3 or 4 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # fresh allocation respecting the total mass
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:6d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional Linear‑Programming acceleration (requires SciPy)
# ----------------------------------------------------------------------


def _lp_step(current: np.ndarray, max_conv: float):
    """
    Solve the LP:
        maximise   ∑ b_j
        subject to  current * b ≤ max_conv   (component‑wise)
                    b_j ≥ 0
    Returns a normalised vector g (∑g = 1) or None if the LP fails.
    """
    try:
        from scipy.optimize import linprog
    except Exception as e:
        # SciPy not available – signal failure
        return None

    n = len(current)
    rows = 2 * n - 1

    # Build the constraint matrix A where each column j is the
    # convolution kernel formed by shifting ``current`` by j.
    A = np.empty((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = current

    b_ub = np.full(rows, max_conv, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise sum(b)  →  minimise –sum(b)
    bounds = [(0.0, None)] * n

    # HiGHS (default) handles large dense problems efficiently.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        return None

    b = res.x
    s = b.sum()
    if s <= 0.0:
        return None
    return b / s                               # normalise to ℓ₁ = 1

    best_val = _objective(best_seq)

    # Phase 1 – exact sub‑gradient mirror descent (many iterations)
    # ------------------------------------------------------------------
    seq, val = _subgradient_refine(best_seq, n, deadline,
                                   max_iters=200_000)
    if val < best_val:
        best_val, best_seq = val, seq.copy()
        print(f"[Phase‑1] New global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP‑based improvement (if SciPy is present)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # we keep re‑using the current best as the starting point
        cur = best_seq.copy()
        for lp_iter in range(6):
            if time.time() > deadline:
                break
            conv = _conv_fft(cur, cur)
            M = np.max(conv)
            g = _lp_step(cur, M)
            if g is None:
                print("[LP] Linear program failed – skipping remaining LP steps")
                break
            # convex combination (t = 0.5) and renormalisation
            cur = (0.5) * cur + (0.5) * g
            cur /= cur.sum()
            cur_val = _objective(cur)
            if cur_val < best_val:
                best_val, best_seq = cur_val, cur.copy()
                print(f"[LP] iter {lp_iter} new global best = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _simulated_annealing(best_seq, n, deadline,
                                        max_iters=800_000,
                                        init_temp_factor=0.03,
                                        min_temp=1e-12)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Phase‑3‑SA] New global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity (clipping, normalisation) and reporting
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)        # enforce non‑negativity
    if best_seq.sum() == 0.0:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_seq /= best_seq.sum()             # ensure ℓ₁ = 1 (scale‑invariant)

    # Optional: double‑check with the provided evaluate_sequence
    try:
        reported = evaluate_sequence(best_seq.tolist())
        print(f"[Check] evaluate_sequence reports {reported:.6f}")
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f}  (length {n})")
    return best_seq.tolist()
```
