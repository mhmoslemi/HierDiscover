"""
Hybrid optimization for the max‑auto‑convolution problem.

* Phase 1 – cheap analytic seeds + a focused Monte‑Carlo search over a 3‑parameter
  family (Beta‑shape + uniform mixture).  The best parameters give a very good
  starting point.

* Phase 2 – deterministic polishing:
    – Adam on a smooth max‑approximation (log‑sum‑exp) to obtain a high‑precision
      local minimum.
    – Exact mirror‑descent sub‑gradient descent (the “sub‑gradient refine” step).
    – A light stochastic hill‑climb that moves a small amount of mass between
      two positions.

All steps respect the global time budget (default 1000 s).  The routine returns
a list of non‑negative floats that minimises ``evaluate_sequence``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (on the simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay (already normalised)."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p (already normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric shape."""
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+1)/(n+1))."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ i^{α‑1}·(n‑i+1)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _inv_edge(n: int) -> np.ndarray:
    """a_i ∝ 1/((i+1)*(n‑i)) – larger near the boundaries."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / ((i + 1.0) * (n - i))
    return a / a.sum()


def _inv_sqrt_edge(n: int) -> np.ndarray:
    """a_i ∝ 1/√((i+1)*(n‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 1.0) * (n - i))
    return a / a.sum()


def _triangle(n: int) -> np.ndarray:
    """Symmetric triangular (tent) shape."""
    half = (n + 1) // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    dec = inc[: n // 2][::-1]          # drop the middle element if n is odd
    a = np.concatenate([inc, dec])
    return a / a.sum()


def _make_initial_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of analytically defined start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_triangle(n))
    seeds.append(_inv_edge(n))
    seeds.append(_inv_sqrt_edge(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # symmetric geometric families
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # Beta‑shape grid (including symmetric and asymmetric)
    for a in np.linspace(0.2, 2.0, 9):
        for b in np.linspace(0.2, 2.0, 9):
            seeds.append(_beta_shape(n, a, b))

    rng = np.random.default_rng(2023)
    # Dirichlet draws – different concentrations
    for conc in (0.5, 1.0, 3.0):
        for _ in range(4):
            seeds.append(rng.dirichlet(np.full(n, conc)))
    # Random softmax vectors
    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # ensure non‑negativity after numeric noise
    cleaned = []
    for s in seeds:
        s = np.maximum(s, 0.0)
        s_sum = s.sum()
        if s_sum == 0.0:
            continue
        cleaned.append(s / s_sum)
    return cleaned


# ----------------------------------------------------------------------
# Objective helper (exact, but faster than evaluate_sequence)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a sequence already normalised to sum=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Monte‑Carlo search over the 3‑parameter family:
#   a = (1‑λ)·Beta(α,β) + λ·uniform
# ----------------------------------------------------------------------


def _search_beta_mixture(n: int,
                         deadline: float,
                         max_evals: int = 3000) -> tuple[np.ndarray, float]:
    """Randomly explore (α,β,λ) and keep the best sequence."""
    rng = np.random.default_rng()
    best_seq = _uniform(n)
    best_val = _objective(best_seq)

    # parameter ranges
    log_alpha_low, log_alpha_high = -2.0, 1.0   # α ∈ [10⁻², 10¹]
    log_beta_low,  log_beta_high  = -2.0, 1.0
    for _ in range(max_evals):
        if time.time() > deadline:
            break
        # sample α,β on a log‑scale; λ on a linear scale
        log_a = rng.uniform(log_alpha_low, log_alpha_high)
        log_b = rng.uniform(log_beta_low, log_beta_high)
        a = 10 ** log_a
        b = 10 ** log_b
        lam = rng.uniform(0.0, 0.5)          # we rarely need >0.5 mixing

        beta_seq = _beta_shape(n, a, b)
        seq = (1.0 - lam) * beta_seq + lam * _uniform(n)
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()

        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """Adam on a log‑softmax parametrisation, using a smoothed‑max surrogate."""
    n = len(seed)
    v = np.log(seed + 1e-30)           # parameters on the simplex (softmax)

    # Adam state
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # smoothing schedule for the soft‑max approximation of the max
    beta = 5.0
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1

        a = _softmax(v)                     # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report (lightweight)
            if it % 2000 == 0:
                print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # ----- smooth‑max weights -----
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·conv_k w.r.t. a -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

        # ----- map to gradient w.r.t. v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        deadline: float,
                        max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * len(a) * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (len(a) - 1))
        i_end = min(len(a) - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(len(a))
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (pairwise mass transfer)
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                deadline: float,
                max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """Random pair‑wise mass transfer, keeping improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02  # relative step size at the beginning

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, len(cur), size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new)

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[Hill] it={it:5d} best={val:.6f}")

        if it % 1000 == 0:
            step_scale = max(step_scale * 0.9, 1e-5)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Combines an analytic‑seed scan, a focused Monte‑Carlo search over a
    three‑parameter family and deterministic refinements (Adam, sub‑gradient,
    hill‑climb).  Respects the supplied time budget.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    margin = 2.0                      # safety margin for housekeeping
    deadline = start_time + budget_s - margin

    # ------------------------------------------------------------------
    # Choose a problem size – 4000 gives a good trade‑off between speed
    # and quality; 8000 can be used if time permits.
    # ------------------------------------------------------------------
    lengths = [4000, 8000] if budget_s > 200 else [4000]

    # if a previous solution exists, use it as a seed
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1. Analytic seed scan
        # --------------------------------------------------------------
        seeds = _make_initial_seeds(n, previous)
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # --------------------------------------------------------------
        # 2. Focused Monte‑Carlo search over Beta + uniform mixture
        # --------------------------------------------------------------
        seq_mc, val_mc = _search_beta_mixture(
            n,
            deadline,
            max_evals=2500 if n == 4000 else 1500,
        )
        if val_mc < global_best_val:
            global_best_val = val_mc
            global_best_seq = seq_mc.copy()
            print(f"[MC‑mix] n={n:5d} new global best={val_mc:.6f}")

        # --------------------------------------------------------------
        # 3. Deterministic polishing – Adam (smooth max)
        # --------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq_adam, val_adam = _adam_optimize(
                global_best_seq,
                deadline,
                max_iters=15000,
            )
            if seq_adam is not None and val_adam < global_best_val:
                global_best_val = val_adam
                global_best_seq = seq_adam.copy()
                print(f"[Adam‑fine] n={n:5d} new global best={val_adam:.6f}")

        # --------------------------------------------------------------
        # 4. Exact sub‑gradient refinement
        # --------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq_sg, val_sg = _subgradient_refine(
                global_best_seq,
                deadline,
                max_iters=40000,
            )
            if seq_sg is not None and val_sg < global_best_val:
                global_best_val = val_sg
                global_best_seq = seq_sg.copy()
                print(f"[SubGrad‑fine] n={n:5d} new global best={val_sg:.6f}")

        # --------------------------------------------------------------
        # 5. Stochastic hill‑climbing
        # --------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq_hc, val_hc = _hill_climb(
                global_best_seq,
                deadline,
                max_iters=30000,
            )
            if seq_hc is not None and val_hc < global_best_val:
                global_best_val = val_hc
                global_best_seq = seq_hc.copy()
                print(f"[Hill‑fine] n={n:5d} new global best={val_hc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # --------------------------------------------------------------
    # Verify with the provided evaluate_sequence (it includes clamping).
    # --------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
