"""
Optimisation of the step‑function objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The objective is invariant under scaling; we therefore keep all vectors on the
probability simplex (∑a_i = 1, a_i ≥ 0).  The algorithm works in three phases:

1.  **Analytic seeding** – a quick grid search over several parametric families
    (product, sum, exponential, cosine‑power, beta).  The 10 best vectors are
    kept as elite seeds.

2.  **Population‑based simulated annealing** – a small population (≈30) is
    evolved for the bulk of the time‑budget.  Moves consist of a
    *mass‑transfer* (pick i, j and move a fraction of a_i to a_j) and an
    occasional *global mixing* with a random Dirichlet draw.  Acceptance follows
    a Metropolis rule with a linearly decreasing temperature.  After each
    move the vector is projected onto the simplex.

3.  **Fine‑grained gradient polishing** – the best solution from phase 2 is
    refined with a few hundred iterations of sub‑gradient descent (the active
    shift of the max‑convolution provides an exact sub‑gradient).  A final
    Gaussian jitter stage extracts any remaining improvement.

All operations use FFT‑based convolution for speed; the final returned vector
has sum 1 (any scaling would give the same score in the evaluator).  The
routine respects a 1000 s wall‑time limit and prints intermediate scores.
"""

import time
import math

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT (real input → real output).
    Returns a vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def objective_fft(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1) using FFT convolution.
    The value is 2·n·max(conv_self(a)).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_b = conv.max()
    return 2.0 * n * max_b


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_move(a: np.ndarray, rng: np.random.Generator, scale: float) -> np.ndarray:
    """
    Transfer a random fraction of the mass from one index to another.
    """
    n = a.shape[0]
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    delta = scale * a[i] * rng.random()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # sum unchanged → only need to clip possible negative rounding errors
    a_new = np.maximum(a_new, 0.0)
    return a_new


# ----------------------------------------------------------------------
# Analytic families (all return normalised probability vectors)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    return a / a.sum()


def analytic_cosine(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    return a / a.sum()


def analytic_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta_
    return a / a.sum()


def analytic_piecewise_linear(n: int, slope: float) -> np.ndarray:
    """
    Simple U‑shaped piecewise linear profile:
        a_i ∝ 1 + slope * |i - (n‑1)/2|
    The slope>0 pulls mass to the ends.
    """
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + slope * np.abs(i - centre)
    return a / a.sum()

# Analytic grid search – returns the best vector and the top‑10 list

def _add_candidate(cand: np.ndarray,
                   best_list: list[tuple[float, np.ndarray]],
                   max_len: int = 10) -> None:
    """Insert a candidate into the bounded best‑list (sorted by score)."""
    sc = objective_fft(cand)
    inserted = False
    for idx, (s, _) in enumerate(best_list):
        if sc < s:
            best_list.insert(idx, (sc, cand.copy()))
            inserted = True
            break
    if not inserted and len(best_list) < max_len:
        best_list.append((sc, cand.copy()))
    while len(best_list) > max_len:
        best_list.pop()


def analytic_grid_search(n: int) -> tuple[np.ndarray, list[tuple[float, np.ndarray]]]:
    """
    Exhaustively scan a small set of parametric families.
    Returns the overall best vector and the sorted list of the 10 best
    candidates (score, vector).
    """
    top: list[tuple[float, np.ndarray]] = []

    # Product family
    for p in np.arange(0.1, 3.01, 0.15):
        for off in np.concatenate(
            (np.geomspace(1e-3, 0.01, 5), np.arange(0.01, 5.01, 0.3))
        ):
            cand = analytic_product(n, p, off)
            _add_candidate(cand, top)

    # Sum family
    for p in np.arange(0.1, 3.01, 0.15):
        for off in np.concatenate(
            (np.geomspace(1e-3, 0.01, 5), np.arange(0.01, 5.01, 0.3))
        ):
            cand = analytic_sum(n, p, off)
            _add_candidate(cand, top)

    # Exponential tails
    for lam in np.logspace(-3, 1, num=20):
        cand = analytic_exponential(n, lam)
        _add_candidate(cand, top)

    # Cosine‑power
    for p in np.arange(0.5, 8.1, 0.5):
        cand = analytic_cosine(n, p)
        _add_candidate(cand, top)

    # Beta family
    for a in np.arange(-0.5, 2.51, 0.5):
        for b in np.arange(-0.5, 2.51, 0.5):
            cand = analytic_beta(n, a, b)
            _add_candidate(cand, top)

    # Simple linear U‑shape
    for s in np.arange(0.0, 5.05, 0.5):
        cand = analytic_piecewise_linear(n, s)
        _add_candidate(cand, top)

    # If a previous construction is present, seed it as well
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            _add_candidate(ref, top)

    best_seq = min(top, key=lambda x: x[0])[1].copy()
    return best_seq, top


# ----------------------------------------------------------------------
# Evolutionary / Simulated‑Annealing optimiser
# ----------------------------------------------------------------------


def anneal_optimize(start_vec: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated annealing with simple mass‑transfer moves.
    Returns the best vector found before `deadline`.
    """
    n = start_vec.shape[0]
    cur = start_vec.copy()
    cur_score = objective_fft(cur)
    best = cur.copy()
    best_score = cur_score

    # annealing schedule parameters
    start_temp = 0.5 * cur_score   # temperature of the same order as the score
    min_temp = 1e-4
    move_scale = 0.2               # fraction of source mass moved per step
    last_improve = time.time()
    stagnation_limit = 30.0        # seconds without improvement → jitter

    while time.time() < deadline:
        # linear cooling
        remaining = deadline - time.time()
        total = deadline - (deadline - (deadline - start_time))  # dummy to keep expr simple
        # simpler: proportion of elapsed time from start of this phase
        elapsed = time.time() - (deadline - (deadline - start_time))  # not used

        # temperature decays linearly
        fraction = (time.time() - (deadline - (deadline - start_time))) / (deadline - (deadline - start_time) + 1e-12)
        # to avoid negative values when we overshoot, clamp
        fraction = max(0.0, min(1.0, fraction))
        temp = max(min_temp, start_temp * (1.0 - fraction))

        # propose a mass‑transfer move
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        delta = move_scale * cur[i] * rng.random()
        if delta > cur[i]:
            delta = cur[i]
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # numeric safety
        cand = np.maximum(cand, 0.0)

        cand_score = objective_fft(cand)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / (temp + 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                last_improve = time.time()
                print(f"[anneal] new best = {best_score:.6f}")

        # occasional global mixing with a random Dirichlet draw
        if rng.random() < 0.008:
            rand = rng.gamma(1.0, 1.0, size=n)
            rand = rand / rand.sum()
            alpha = rng.random()
            mixed = alpha * cur + (1.0 - alpha) * rand
            mixed = project_simplex(mixed)
            mixed_score = objective_fft(mixed)
            if mixed_score < cur_score or rng.random() < math.exp(-(mixed_score - cur_score) / (temp + 1e-12)):
                cur = mixed
                cur_score = mixed_score
                if mixed_score < best_score:
                    best = mixed.copy()
                    best_score = mixed_score
                    last_improve = time.time()
                    print(f"[anneal] new best (global mix) = {best_score:.6f}")

        # jitter restart if stagnated
        if time.time() - last_improve > stagnation_limit:
            jitter = rng.random(n) * 1e-3
            cur = best + jitter
            cur = project_simplex(cur)
            cur_score = objective_fft(cur)
            last_improve = time.time()
            print("[anneal] jitter restart")

    return best, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------


def subgradient_descent(start_vec: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent with Nesterov momentum on the exact max‑convolution.
    """
    a = start_vec.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    best = a.copy()
    best_score = objective_fft(a)

    # step size schedule (log‑linear)
    step_start, step_end = 0.7, 1e-5
    while time.time() < deadline:
        conv = conv_self_fft(a)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[sgd] new best = {best_score:.6f}")

        # sub‑gradient for the active shift
        shift = max_idx - np.arange(n)
        mask = (shift >= 0) & (shift < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[shift[mask]]

        # step size (decreases with elapsed fraction)
        elapsed = time.time() - start_time
        fraction = elapsed / (budget_s - 1e-12)
        step = step_start * (1.0 - fraction) + step_end * fraction

        # Nesterov momentum update
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        a_candidate = a - step * momentum
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate = project_simplex(a_candidate)

        # occasional random move to escape plateaus
        if rng.random() < 0.015:
            a_candidate = random_move(a_candidate, rng, step * 0.5)

        a = a_candidate

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
        2·n·max(conv(a,a)) / (sum(a)**2)

    The routine respects a wall‑time budget (default 1000 s) and returns the
    best sequence discovered.
    """
    global start_time, budget_s  # needed for sub‑functions
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    n = int(kwargs.get("n", 2000))
    print("[0] Analytic grid search (n =", n, ")")
    best_seq, top10 = analytic_grid_search(n)

    # keep the top‑10 candidates and also create a few random Dirichlet seeds
    population: list[np.ndarray] = [vec for _, vec in top10]
    while len(population) < 30:
        vec = rng.gamma(1.0, 1.0, size=n)
        vec = vec / vec.sum()
        population.append(vec)

    # ------------------------------------------------------------------
    # 1) Simulated annealing on a small population
    # ------------------------------------------------------------------
    phase1_deadline = deadline - 300.0   # reserve ~5 min for polishing
    print("[1] Simulated‑annealing phase, time left ≈",
          int(phase1_deadline - time.time()), "s")

    # keep the best individual from the initial pool as a starting point
    cur_best = min(population, key=objective_fft)
    cur_best, cur_best_score = anneal_optimize(cur_best, phase1_deadline, rng)
    print("[1] after annealing best score =", cur_best_score)

    # ------------------------------------------------------------------
    # 2) Sub‑gradient polishing (exact max‑conv)
    # ------------------------------------------------------------------
    phase2_deadline = deadline - 60.0   # leave last minute for final jitter
    print("[2] Sub‑gradient polishing, time left ≈",
          int(phase2_deadline - time.time()), "s")
    cur_best, cur_best_score = subgradient_descent(cur_best, phase2_deadline, rng)
    print("[2] after polishing best score =", cur_best_score)

    # ------------------------------------------------------------------
    # 3) Final Gaussian jitter (last few seconds)
    # ------------------------------------------------------------------
    print("[3] Final jitter phase")
    best = cur_best.copy()
    best_score = cur_best_score
    while time.time() < deadline:
        pert = rng.normal(scale=4e-4, size=n)
        cand = best + pert
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        cand_score = objective_fft(cand)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            print(f"[3] improved to {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, length = {n}, time = {total_time:.1f}s")
    return best.tolist()
