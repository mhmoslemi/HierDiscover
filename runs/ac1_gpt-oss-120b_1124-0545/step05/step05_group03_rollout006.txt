"""
Hybrid global–local optimisation with extended differential evolution and
multi‑stage refinement.

* Coarse DE on sqrt(n) knots, many generations.
* Fine DE on twice‑as‑many knots, seeded by the coarse best.
* Adam optimisation on a smooth‑max surrogate (more iterations).
* Exact sub‑gradient mirror descent (more iterations, multiple‑active‑set).
* Stochastic hill‑climbing with adaptive step size.
* Optional symmetrisation after each phase.

All stages respect the supplied time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# low‑level helpers: FFT convolution, softmax, etc.
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n

# ----------------------------------------------------------------------
# Analytic seeds (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()

def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))
    for a in (0.3, 0.5, 0.7):
        for b in (0.3, 0.5, 0.7):
            seeds.append(_beta_shape(n, a, b))
    rng = np.random.default_rng(2023)
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())
    return seeds

# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max
# ----------------------------------------------------------------------
def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray | None, float]:
    v = np.log(seed + 1e-30)          # log‑softmax parameters
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    beta = 5.0            # smoothing coeff
    beta_mul = 1.03
    best_seq = None
    best_val = float("inf")
    t = 0
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                         # stays on simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")
        # smooth‑max weighting
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum
        # gradient w.r.t a of Σ w_k·conv_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries
        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)
        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)
    return best_seq, best_val

# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror descent) with multi‑active set
# ----------------------------------------------------------------------
def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 80000,
                        eps_active: float = 1e-4) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:5d} best={cur_val:.6f}")
        # active set: indices whose value is within eps_active of the max
        mask = conv >= max_val - eps_active * max_val
        # weights for active constraints (uniform)
        # compute gradient = 2 * sum_{k active} a shifted by k
        # This can be obtained by convolution of a with mask_rev
        mask_rev = mask[::-1].astype(np.float64)
        full = _conv_fft(a, mask_rev)                     # length = n + (2n-1) -1 = 3n-2
        grad = 2.0 * full[n - 1:n - 1 + n]                # central n entries
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)    # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s
    return best_seq, best_val

# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 80000) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i
        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta
        val = 2.0 * n * np.max(_conv_fft(new, new))
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] it={it:5d} best={val:.6f}")
        if it % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)
    return best_seq, best_val

# Symmetrisation helper
# ----------------------------------------------------------------------
def _symmetrize(seq: np.ndarray) -> np.ndarray:
    s = (seq + seq[::-1]) * 0.5
    s_sum = s.sum()
    return s / s_sum if s_sum > 0 else _uniform(len(seq))

# ----------------------------------------------------------------------
# Differential Evolution (global search) – general version
# ----------------------------------------------------------------------
def _eval_param(p: np.ndarray, knots: np.ndarray, n: int) -> tuple[float, np.ndarray]:
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    if s == 0.0:
        a = _uniform(n)
        s = a.sum()
    a = a / s
    conv = _conv_fft(a, a)
    val = 2.0 * n * np.max(conv)      # sum(a)=1 ⇒ objective = 2·n·max(conv)
    return val, a

def _de_optimize(n: int,
                 deadline: float,
                 popsize: int = 30,
                 K: int | None = None,
                 max_gens: int = 300,
                 init_pop: list[np.ndarray] | None = None,
                 seed: int = 123) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng(seed)
    if K is None:
        K = max(20, int(np.sqrt(n)))
    knots = np.linspace(0, n - 1, K)
    if init_pop is not None:
        pop = init_pop
        if len(pop) != popsize:
            # pad or truncate to required size
            while len(pop) < popsize:
                pop.append(rng.dirichlet(np.ones(K)))
            pop = pop[:popsize]
    else:
        pop = [rng.dirichlet(np.ones(K)) for _ in range(popsize)]
    pop_vals = np.empty(popsize, dtype=np.float64)
    pop_seq = [None] * popsize
    for i in range(popsize):
        val, seq = _eval_param(pop[i], knots, n)
        pop_vals[i] = val
        pop_seq[i] = seq
    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_seq = pop_seq[best_idx].copy()
    best_p = pop[best_idx].copy()
    F = 0.8
    CR = 0.9
    gen = 0
    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a_vec = pop[a_idx]
            b_vec = pop[b_idx]
            c_vec = pop[c_idx]
            donor = a_vec + F * (b_vec - c_vec)
            donor = np.maximum(donor, 0.0)
            ds = donor.sum()
            if ds == 0.0:
                donor = _uniform(K)
            else:
                donor = donor / ds
            trial = pop[i].copy()
            jrand = rng.integers(K)
            for j in range(K):
                if rng.random() < CR or j == jrand:
                    trial[j] = donor[j]
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts == 0.0:
                trial = _uniform(K)
            else:
                trial = trial / ts
            val_trial, seq_trial = _eval_param(trial, knots, n)
            if val_trial < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val_trial
                pop_seq[i] = seq_trial
                if val_trial < best_val:
                    best_val = val_trial
                    best_seq = seq_trial.copy()
                    best_p = trial.copy()
        if gen % 20 == 0:
            print(f"[DE] n={n:5d} gen={gen:3d} best={best_val:.6f}")
    return best_seq, best_val

def _interpolate_to_knots(seq: np.ndarray, K: int) -> np.ndarray:
    """Sample a full‑length sequence at K equally spaced points."""
    n = seq.shape[0]
    idxs = np.linspace(0, n - 1, K)
    p = np.interp(idxs, np.arange(n, dtype=np.float64), seq)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s == 0.0:
        return _uniform(K)
    return p / s

import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.ones(n, dtype=np.float64) / n


# ----------------------------------------------------------------------
# Analytic seeds (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a modest set of diverse normalised start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    for a in (0.3, 0.5, 0.7):
        for b in (0.3, 0.5, 0.7):
            seeds.append(_beta_shape(n, a, b))

    rng = np.random.default_rng(2023)
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max (same as reference code, longer run)
# ----------------------------------------------------------------------
def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑softmax parametrisation, using a smooth‑max surrogate."""
    v = np.log(seed + 1e-30)          # log‑softmax parameters
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0            # initial smoothing coefficient
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # smooth‑max weighting
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient w.r.t a of Σ w_k·conv_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # convert to gradient w.r.t v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror descent) with multi‑active set
# ----------------------------------------------------------------------
def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 80000,
                        eps_active: float = 1e-3) -> tuple[np.ndarray, float]:
    """Exact mirror‑descent using all convolution entries close to the current max."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        cur_val = 2.0 * n * max_val

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:5d} best={cur_val:.6f}")

        # active set: all indices within a relative tolerance of the maximum
        mask = conv >= max_val - eps_active * max_val
        mask_rev = mask[::-1].astype(np.float64)

        # gradient = 2 * Σ_{k∈active} a shifted by k
        full = _conv_fft(a, mask_rev)                # length = 3n‑2
        grad = 2.0 * full[n - 1:n - 1 + n]            # central n entries

        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)                        # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """Random pair‑wise mass transfer, keeping improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02  # initial relative step size

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor],
                           step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] it={it:5d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Symmetrisation helper
# ----------------------------------------------------------------------
def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Replace a by (a + a[::-1]) / 2 and re‑normalise."""
    s = (seq + seq[::-1]) * 0.5
    s_sum = s.sum()
    return s / s_sum if s_sum > 0 else _uniform(len(seq))


# ----------------------------------------------------------------------
# Differential Evolution (global search) – generic version
# ----------------------------------------------------------------------
def _eval_param(p: np.ndarray, knots: np.ndarray, n: int) -> tuple[float, np.ndarray]:
    """
    Interpolate a low‑dimensional simplex vector `p` (sum=1)
    to a full length‑`n` sequence, normalise, and evaluate the exact objective.
    Returns (objective, full_sequence).
    """
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    if s == 0.0:
        a = _uniform(n)
        s = a.sum()
    a = a / s
    conv = _conv_fft(a, a)
    val = 2.0 * n * np.max(conv)      # sum(a)=1 ⇒ objective = 2·n·max(conv)
    return val, a


def _de_optimize(n: int,
                 deadline: float,
                 popsize: int = 30,
                 K: int | None = None,
                 max_gens: int = 300,
                 init_pop: list[np.ndarray] | None = None,
                 seed: int = 123) -> tuple[np.ndarray, float]:
    """Differential Evolution on a piece‑wise‑linear representation."""
    rng = np.random.default_rng(seed)

    if K is None:
        K = max(20, int(np.sqrt(n)))
    knots = np.linspace(0, n - 1, K)

    # initialise population
    if init_pop is not None:
        pop = init_pop[:]
        while len(pop) < popsize:
            pop.append(rng.dirichlet(np.ones(K)))
        pop = pop[:popsize]
    else:
        pop = [rng.dirichlet(np.ones(K)) for _ in range(popsize)]

    pop_vals = np.empty(popsize, dtype=np.float64)
    pop_seq = [None] * popsize
    for i in range(popsize):
        val, seq = _eval_param(pop[i], knots, n)
        pop_vals[i] = val
        pop_seq[i] = seq

    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_seq = pop_seq[best_idx].copy()
    best_p = pop[best_idx].copy()

    F = 0.8
    CR = 0.9
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a_vec = pop[a_idx]
            b_vec = pop[b_idx]
            c_vec = pop[c_idx]

            donor = a_vec + F * (b_vec - c_vec)
            donor = np.maximum(donor, 0.0)
            ds = donor.sum()
            if ds == 0.0:
                donor = _uniform(K)
            else:
                donor = donor / ds

            # binomial crossover
            trial = pop[i].copy()
            jrand = rng.integers(K)
            for j in range(K):
                if rng.random() < CR or j == jrand:
                    trial[j] = donor[j]

            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts == 0.0:
                trial = _uniform(K)
            else:
                trial = trial / ts

            val_trial, seq_trial = _eval_param(trial, knots, n)

            if val_trial < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val_trial
                pop_seq[i] = seq_trial
                if val_trial < best_val:
                    best_val = val_trial
                    best_seq = seq_trial.copy()
                    best_p = trial.copy()

        if gen % 20 == 0:
            print(f"[DE] n={n:5d} gen={gen:3d} best={best_val:.6f}")

    return best_seq, best_val


def _interpolate_to_knots(seq: np.ndarray, K: int) -> np.ndarray:
    """Sample a full‑length sequence at K equally spaced points and re‑normalise."""
    n = seq.shape[0]
    idxs = np.linspace(0, n - 1, K)
    p = np.interp(idxs, np.arange(n, dtype=np.float64), seq)
    p = np.maximum(p, 0.0)
    s = p.sum()
    return p / s if s != 0.0 else _uniform(K)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine combines analytic seeds, a two‑stage differential‑evolution,
    Adam optimisation, mirror‑descent sub‑gradient refinement with a multi‑active
    set, stochastic hill‑climbing and several symmetrisation steps.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Fixed length (4000 offers a good trade‑off between quality and speed)
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # 0. Load previous result (if any) to use as an extra seed
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # 1. Analytic seeds
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")
    seeds = _make_seeds(n, previous)
    for s in seeds:
        if time.time() > deadline:
            break
        conv = _conv_fft(s, s)
        val = 2.0 * n * np.max(conv)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = s.copy()
            print(f"[seed] new global best={val:.6f}")

    # ------------------------------------------------------------------
    # 2. Coarse DE (√n knots)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_coarse, val_coarse = _de_optimize(
            n, deadline,
            popsize=30,
            K=None,
            max_gens=300,
            seed=seed + 1
        )
        if val_coarse < global_best_val:
            global_best_val = val_coarse
            global_best_seq = seq_coarse.copy()
            print(f"[DE‑coarse] new global best={val_coarse:.6f}")

    # ------------------------------------------------------------------
    # 3. Fine DE (2·√n knots) seeded by coarse best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        K_fine = max(30, int(2 * np.sqrt(n)))
        init_pop = [_interpolate_to_knots(global_best_seq, K_fine)]
        rng = np.random.default_rng(seed + 2)
        while len(init_pop) < 30:
            init_pop.append(rng.dirichlet(np.ones(K_fine)))
        seq_fine, val_fine = _de_optimize(
            n, deadline,
            popsize=30,
            K=K_fine,
            max_gens=200,
            init_pop=init_pop,
            seed=seed + 3
        )
        if val_fine < global_best_val:
            global_best_val = val_fine
            global_best_seq = seq_fine.copy()
            print(f"[DE‑fine] new global best={val_fine:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam optimisation (smooth‑max surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline and global_best_seq is not None:
        seq_adam, val_adam = _adam_optimize(
            global_best_seq, n,
            deadline,
            max_iters=20000
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_val = val_adam
            global_best_seq = seq_adam.copy()
            print(f"[Adam] new global best={val_adam:.6f}")

    # ------------------------------------------------------------------
    # 5. Symmetrisation checkpoint
    # ------------------------------------------------------------------
    if time.time() < deadline and global_best_seq is not None:
        sym = _symmetrize(global_best_seq)
        conv = _conv_fft(sym, sym)
        val_sym = 2.0 * n * np.max(conv)
        if val_sym < global_best_val:
            global_best_val = val_sym
            global_best_seq = sym.copy()
            print(f"[sym] new global best={val_sym:.6f}")

    # ------------------------------------------------------------------
    # 6. Sub‑gradient refinement (multi‑active set)
    # ------------------------------------------------------------------
    if time.time() < deadline and global_best_seq is not None:
        seq_sg, val_sg = _subgradient_refine(
            global_best_seq, n,
            deadline,
            max_iters=80000,
            eps_active=1e-3
        )
        if val_sg < global_best_val:
            global_best_val = val_sg
            global_best_seq = seq_sg.copy()
            print(f"[SubGrad] new global best={val_sg:.6f}")

    # ------------------------------------------------------------------
    # 7. Second symmetrisation (after sub‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline and global_best_seq is not None:
        sym = _symmetrize(global_best_seq)
        conv = _conv_fft(sym, sym)
        val_sym = 2.0 * n * np.max(conv)
        if val_sym < global_best_val:
            global_best_val = val_sym
            global_best_seq = sym.copy()
            print(f"[sym‑post] new global best={val_sym:.6f}")

    # ------------------------------------------------------------------
    # 8. Stochastic hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline and global_best_seq is not None:
        seq_hc, val_hc = _hill_climb(
            global_best_seq, n,
            deadline,
            max_iters=80000
        )
        if val_hc < global_best_val:
            global_best_val = val_hc
            global_best_seq = seq_hc.copy()
            print(f"[Hill] new global best={val_hc:.6f}")

    # ------------------------------------------------------------------
    # 9. Final symmetrisation (cheap)
    # ------------------------------------------------------------------
    if global_best_seq is not None:
        sym = _symmetrize(global_best_seq)
        conv = _conv_fft(sym, sym)
        val_sym = 2.0 * n * np.max(conv)
        if val_sym < global_best_val:
            global_best_val = val_sym
            global_best_seq = sym.copy()
            print(f"[final‑sym] new global best={val_sym:.6f}")

    # ------------------------------------------------------------------
    # 10. Verify with the provided evaluation function
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")

    # store for the next run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
