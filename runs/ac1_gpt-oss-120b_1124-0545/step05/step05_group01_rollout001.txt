"""
Improved search for a non‑negative step function minimising the
self‑convolution ∞‑norm used by ``evaluate_sequence``.

Algorithm
---------

1. **Rich seed pool** – uniform, geometric, power‑law, symmetric geometric,
   sine‑window, Gaussian, Dirichlet draws, **Beta(α,α)** distributions
   (including the arcsine case α=0.5) and the previously discovered best
   construction (if any).  All seeds are clipped to [0,1000] and re‑normalised
   to sum = 1 (the objective is scale‑invariant).

2. **Adam optimisation** on a smooth surrogate of the true objective:
        L(a)= (1/β)·log∑ₖexp(β·convₖ(a)) + λ·∑ₖconvₖ(a)² .
   The vector is kept on the simplex via a log‑softmax parametrisation.
   β starts small (≈1) and is multiplied by 1.03 each iteration,
   λ starts at 0.8 and decays slowly.  The best iterate seen is recorded.

3. **Exact sub‑gradient refinement** – mirror‑descent (exponentiated‑gradient)
   on the genuine maximal convolution entry.  A 1/√t step‑size schedule is
   used and the best point is kept.

4. **Simulated‑annealing fine‑tuning** – cheap Metropolis–Hastings walk
   moving a tiny fraction of mass between two positions, with occasional
   global Gaussian jitters.

The routine respects the global time budget (default 1000 s) and iterates
over three increasing lengths (8000, 12000, 20000).  The best sequence
encountered across all lengths is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _clip_and_normalise(x: np.ndarray) -> np.ndarray:
    """Clip to the range accepted by ``evaluate_sequence`` and renormalise."""
    y = np.clip(x, 0.0, 1000.0)
    s = y.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return y / s


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to harmless clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq))²."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    return _clip_and_normalise(a)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _clip_and_normalise(a)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    return _clip_and_normalise(a)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return _clip_and_normalise(a)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return _clip_and_normalise(a)


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return _clip_and_normalise(a)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse or dense Dirichlet draw (Gamma(α,1) normalised)."""
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return _clip_and_normalise(vec)


def _beta_seed(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) on the discrete grid i+0.5 over n points.
    For α=0.5 this is the arcsine distribution.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n               # lies strictly in (0,1)
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    # Guard against NaNs for extreme α
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    return _clip_and_normalise(a)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse set of start‑vectors (already normalised to sum = 1).
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # Basic families
    seeds.append(_uniform(n))

    # Geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # Power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # Combined geometric × power
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # Symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # Window functions
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # Dirichlet draws covering sparse ↔ dense regimes
    for alpha in [0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # Symmetric Beta families (including arcsine at α=0.5)
    for alpha in [0.4, 0.5, 0.6, 0.8, 1.0]:
        seeds.append(_beta_seed(n, alpha))

    # Mirror each seed (helps when optimum is asymmetric)
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # Re‑use previous best construction if length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(_clip_and_normalise(previous))

    # Guarantee at least one seed
    if not seeds:
        seeds.append(_uniform(n))

    return seeds

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000,
                   beta0: float = 1.0,
                   beta_mul: float = 1.03,
                   lr0: float = 0.9,
                   lambda0: float = 0.8,
                   lambda_decay: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the loss
        L(a) = (1/β)·log∑ₖexp(β·convₖ(a)) + λ·∑ₖconvₖ(a)² .
    The vector is kept on the simplex via a log‑softmax parametrisation.
    Returns the best sequence encountered and its *true* objective.
    """
    v = np.log(seed + 1e-30)          # initialise logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0
    lam = lambda0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # simplex point

        # evaluate true objective for book‑keeping
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ---------- smoothed‑max weights ----------
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = conv.max()
        w_max = np.exp(beta * (conv - shift))
        w_sum = w_max.sum()
        if w_sum == 0.0:
            continue
        w_max /= w_sum

        # ---------- gradient of smoothed max ----------
        rev_a = a[::-1]
        full_max = _conv_fft(rev_a, w_max)
        grad_max = 2.0 * full_max[n - 1:n - 1 + n]

        # ---------- gradient of Σ conv² ----------
        full_sq = _conv_fft(rev_a, conv)
        grad_sq = 2.0 * full_sq[n - 1:n - 1 + n]

        # combine gradients
        grad_a = grad_max + lam * grad_sq

        # ----- chain rule through the softmax -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on logits -----
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

        # decay variance regularizer
        lam *= lambda_decay

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using the true sub‑gradient of
    max(conv).  Returns the best sequence seen and its objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.75 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis–Hastings walk that transfers a tiny amount of mass
    between two random positions and, every 1000 steps, adds a global
    Gaussian jitter.  Returns the best sequence discovered.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature (a few % of the current objective)
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move a small fraction of mass from i to j
        delta = rng.random() * 0.025 * cur[i]
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n:5d} it={it:5d} best={new_val:.6f}")

        # occasional global jitter
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n:5d} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # previous best (if the environment already stored one)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # Lengths – start modest, enlarge if time permits
    lengths = [8000, 12000, 20000]
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # 1) Seed generation
        # ------------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # ------------------------------------------------------------------
        # 2) Quick evaluation of raw seeds – keep the most promising
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(10, len(seed_vals))]]

        # record best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # 3) Adam optimisation on the top seeds
        # ------------------------------------------------------------------
        # adapt iteration budget to length (larger n → fewer iterations)
        max_adam_iters = 8000 if n <= 12000 else 5000
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=max_adam_iters,
                beta0=1.0,
                beta_mul=1.03,
                lr0=0.9,
                lambda0=0.8,
                lambda_decay=0.996,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # 4) Exact sub‑gradient refinement (mirror‑descent)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=60000 if n <= 12000 else 40000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # 5) Simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                        max_iters=None)  # run until time runs out
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

        # keep reference for the next length (mirrored and normalised)
        previous = global_best_seq

    # ------------------------------------------------------------------
    # Fallback (unlikely)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(8000)
        print("[fallback] using uniform distribution")

    # final normalisation (just in case)
    global_best_seq = np.clip(global_best_seq, 0.0, 1000.0)
    s = global_best_seq.sum()
    if s > 0:
        global_best_seq = global_best_seq / s

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
