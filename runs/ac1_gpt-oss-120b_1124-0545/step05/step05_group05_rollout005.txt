    # Stage 1: short subgradient descent (maybe 100 iterations)
    a = best_seq.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    last_improve = time.time()
    stag1_end = start_time + (budget_s * 0.15)  # allocate 15% for subgradient
    iter_cnt = 0
    while time.time() < min(deadline - 300, stag1_end):
        # similar subgradient as before
        conv = conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[sg {iter_cnt}] new best {best_score:.6f}")

        shift = max_idx - np.arange(n)

        a_candidate = np.clip(a_candidate, 0.0, None)

def param_to_vector(params, n):
    # params is array length K
    K = len(params)
    # Compute half length
    half_len = (n + 1) // 2  # number of indices in left half (including middle for odd)
    # Knot positions on half interval:
    x_k = np.linspace(0, half_len - 1, K)
    # Interpolate over integer indices 0..half_len-1
    left = np.interp(np.arange(half_len), x_k, params)
    left = np.maximum(left, 0.0)
    # Mirror to full length
    if n % 2 == 0:
        right = left[::-1]
    else:
        right = left[-2::-1]  # exclude middle element duplicated
    full = np.concatenate([left, right])
    # Normalize to sum 1
    total = full.sum()
    if total <= 0:
        full = np.ones_like(full) / n
    else:
        full = full / total
    return full

"""
Propose a non‑negative step function (list of floats) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1.  Analytic grid search over several parametric families (product, sum,
    exponential, cosine‑power, beta) and keep the best 10 candidates.
2.  Pairwise convex combinations of those top candidates.
3.  A short sub‑gradient descent (Nesterov momentum) to move away from
    obvious local minima.
4.  Differential‑evolution (SciPy) on a low‑dimensional symmetric
    piece‑wise‑linear parametrisation of the sequence (12 parameters for
    the whole vector).  The DE runs until the remaining time budget is
    low, guaranteeing a global‑search phase that the previous solver
    lacked.
5.  Adam optimisation of a softened max (log‑sum‑exp) objective.
6.  Final random‑perturbation refinement.

All vectors are projected onto the probability simplex (∑a_i = 1, a_i≥0),
so the objective is scale‑invariant.  The routine respects the wall‑time
budget (default 1000 s) and returns the best sequence discovered.
"""

import time
import math
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of a and b using FFT."""
    size = _next_pow2(a.shape[0] + b.shape[0] - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """Exact objective (scale invariant) for a vector on the simplex."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_self(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    return a / a.sum()


def analytic_cosine(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    return a / a.sum()


def analytic_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta_
    return a / a.sum()

# Search utilities

def _add_candidate(cand: np.ndarray,
                   best_list: list[tuple[float, np.ndarray]],
                   max_len: int = 10) -> None:
    """Insert a candidate into a bounded best‑list (sorted by score)."""
    sc = compute_score(cand)
    inserted = False
    for idx, (s, _) in enumerate(best_list):
        if sc < s:
            best_list.insert(idx, (sc, cand.copy()))
            inserted = True
            break
    if not inserted and len(best_list) < max_len:
        best_list.append((sc, cand.copy()))
    while len(best_list) > max_len:
        best_list.pop()


def analytic_grid_search(n: int) -> tuple[np.ndarray, list[tuple[float, np.ndarray]]]:
    """Exhaustively scan a set of families; return best vector and top‑10."""
    top = []   # (score, vector) sorted by score

    # Product family
    p_vals = np.arange(0.1, 3.01, 0.15)
    off_vals = np.concatenate(
        (np.geomspace(1e-3, 0.01, 5), np.arange(0.01, 5.01, 0.3))
    )
    for p in p_vals:
        for off in off_vals:
            cand = analytic_product(n, p, off)
            _add_candidate(cand, top)

    # Sum family
    for p in p_vals:
        for off in off_vals:
            cand = analytic_sum(n, p, off)
            _add_candidate(cand, top)

    # Exponential tails
    lam_vals = np.logspace(-3, 1, num=20)
    for lam in lam_vals:
        cand = analytic_exponential(n, lam)
        _add_candidate(cand, top)

    # Cosine‑power
    p_cos = np.arange(0.5, 8.1, 0.5)
    for p in p_cos:
        cand = analytic_cosine(n, p)
        _add_candidate(cand, top)

    # Beta family
    alpha_vals = np.arange(-0.5, 2.51, 0.5)
    beta_vals = np.arange(-0.5, 2.51, 0.5)
    for a in alpha_vals:
        for b in beta_vals:
            cand = analytic_beta(n, a, b)
            _add_candidate(cand, top)

    # If a reference sequence exists, include it
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            _add_candidate(ref, top)

    best_seq = min(top, key=lambda x: x[0])[1].copy()
    return best_seq, top


def mix_top_candidates(top: list[tuple[float, np.ndarray]],
                       n: int) -> np.ndarray:
    """Convexly combine the best few candidates (21 mixing ratios)."""
    best_seq = top[0][1].copy()
    best_score = top[0][0]
    alphas = np.linspace(0.0, 1.0, 21)
    for i in range(len(top)):
        for j in range(i + 1, len(top)):
            a_i = top[i][1]
            a_j = top[j][1]
            for alpha in alphas:
                cand = alpha * a_i + (1.0 - alpha) * a_j
                cand = project_simplex(cand)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
    return best_seq


def param_to_vector(params: np.ndarray, n: int) -> np.ndarray:
    """
    Convert a low‑dimensional symmetric parameter vector into a full
    non‑negative sequence of length n.
    The shape is enforced to be symmetric around the centre.
    """
    K = len(params)
    half_len = (n + 1) // 2               # include middle element for odd n
    # knot positions on the left half (0 … half_len‑1)
    knot_x = np.linspace(0, half_len - 1, K)
    # linear interpolation on integer grid
    left = np.interp(np.arange(half_len), knot_x, params)
    left = np.maximum(left, 0.0)

    if n % 2 == 0:
        right = left[::-1]
    else:
        right = left[-2::-1]               # drop duplicated centre point
    full = np.concatenate([left, right])
    # normalise to sum = 1 (simplex projection)
    s = full.sum()
    if s <= 0.0:
        full = np.ones_like(full) / n
    else:
        full = full / s
    return full

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the evaluation function.  The routine respects a wall‑time budget and
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 0) Analytic grid + mixing of top candidates
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))
    print("[0] Analytic grid search, n =", n)
    best_seq, top10 = analytic_grid_search(n)
    best_seq = mix_top_candidates(top10, n)
    best_score = compute_score(best_seq)
    print(f"[0] Score after mixing: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Short sub‑gradient descent (Nesterov momentum)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    last_improve = time.time()
    stag1_end = start_time + 0.15 * budget_s      # ~15 % of total time
    it = 0
    while time.time() < min(deadline - 250, stag1_end):
        conv = conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[sg {it:06d}] new best = {best_score:.6f}")

        max_idx = int(conv.argmax())
        shift = max_idx - np.arange(n)
        mask = (shift >= 0) & (shift < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[shift[mask]]

        # simple linear schedule for step size
        frac = (time.time() - start_time) / budget_s
        step = 0.8 * (1.0 - frac) + 1e-5
        momentum = beta_mom * momentum + (1 - beta_mom) * g
        a_next = a - step * momentum
        a_next = np.clip(a_next, 0.0, None)
        a_next = project_simplex(a_next)

        # occasional stochastic move
        if np.random.rand() < 0.01:
            a_next = random_move(a_next, step * 0.5)

        a = a_next
        it += 1

    # ------------------------------------------------------------------
    # 2) Differential‑evolution on a symmetric piece‑wise‑linear shape
    # ------------------------------------------------------------------
    K = 12                                 # number of spline knots
    bounds = [(0.0, 5.0)] * K

    # initialise DE around the current best sequence
    half_len = (n + 1) // 2
    left_half = best_seq[:half_len]
    knot_x = np.linspace(0, half_len - 1, K)
    init_params = np.interp(knot_x, np.arange(half_len), left_half)

    def de_obj(params):
        vec = param_to_vector(params, n)
        return compute_score(vec)

    def de_callback(xk, convergence):
        # stop DE when we are close to the global deadline
        if time.time() > deadline - 120:     # keep ~2 min for later phases
            return True
        return False

    print("[DE] Starting differential evolution (K =", K, ")")
    result = differential_evolution(
        de_obj,
        bounds,
        strategy='best1bin',
        maxiter=200,
        popsize=15,
        tol=1e-7,
        polish=False,
        init='latinhypercube',
        seed=seed,
        callback=de_callback,
        updating='deferred',
    )
    de_params = result.x
    de_seq = param_to_vector(de_params, n)
    de_score = compute_score(de_seq)
    print(f"[DE] Finished, score = {de_score:.6f}")

    if de_score < best_score:
        best_score = de_score
        best_seq = de_seq.copy()
        print("[DE] Updated global best to", best_score)

    # ------------------------------------------------------------------
    # 3) Adam optimisation on a soft‑max approximation of the max‑conv
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t_adam = 0
    lr = 0.12
    beta_start, beta_end = 0.5, 400.0
    last_improve = time.time()
    stagnation = 20.0

    while time.time() < deadline - 30:    # leave ~30 s for final refinement
        conv = conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[adam {t_adam:06d}] new best = {best_score:.6f}")

        # temperature schedule – slowly turn the soft‑max into a hard max
        elapsed = time.time() - start_time
        prog = elapsed / (budget_s - 30.0)
        beta = beta_start * (1.0 - prog) + beta_end * prog

        # soft‑max weights over all convolution entries
        w_raw = np.exp(beta * (conv - max_val))   # max exponent = 0
        w = w_raw / w_raw.sum()

        # weighted shift = Σ_k w_k * a_{k-i}
        a_rev = a[::-1]
        corr_full = conv_two(w, a_rev)            # length = 3n‑2
        weighted_shift = corr_full[n - 1:n - 1 + n]

        # gradient of the soft‑max objective
        g = a + weighted_shift

        # Adam update
        t_adam += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_next = a - step_vec
        a_next = np.clip(a_next, 0.0, None)
        a_next = project_simplex(a_next)

        # occasional random perturbations
        if np.random.rand() < 0.015:
            a_next = random_move(a_next, lr * 0.3)
        if np.random.rand() < 0.005:
            jitter = np.random.randn(n) * 5e-4
            a_next = a_next + jitter
            a_next = np.clip(a_next, 0.0, None)
            a_next = project_simplex(a_next)

        # stagnation handling
        if time.time() - last_improve > stagnation:
            jitter = np.random.rand(n) * 1e-3
            a_next = best_seq + jitter
            a_next = project_simplex(a_next)
            last_improve = time.time()
            print(f"[adam {t_adam:06d}] jitter restart")

        a = a_next

    # ------------------------------------------------------------------
    # 4) Final random perturbation phase (last few seconds)
    # ------------------------------------------------------------------
    print("[refine] random perturbation phase")
    while time.time() < deadline:
        pert = np.random.normal(scale=5e-4, size=n)
        cand = best_seq + pert
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print("[refine] improved to", best_score)

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, length = {n}, time = {total_time:.1f}s")
    return best_seq.tolist()
