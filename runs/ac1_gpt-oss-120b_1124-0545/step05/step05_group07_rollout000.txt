"""
Propose a non‑negative step function that (approximately) minimizes

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1️⃣  **Rich analytic initialisation** – we evaluate several simple
    parametric families (product, sum‑of‑powers, exponential tails,
    beta‑shaped, cosine‑power, Gaussian) on the chosen length `n`
    and keep the best one (lowest exact score).

2️⃣  **Exact‑max sub‑gradient descent** – at each iteration we compute the
    self‑convolution, locate the (possibly multiple) active indices
    where the maximum is attained and form a sub‑gradient as the
    (equal‑weight) convex combination of the exact gradients of those
    active convolution entries.  A heavy‑ball momentum update is applied,
    followed by projection onto the probability simplex.

    The learning rate decays slowly, and a jitter restart is performed
    whenever no improvement has been seen for a while.

3️⃣  **Stochastic refinement** – after the main loop we add tiny Gaussian
    perturbations and keep any improvement.

All heavy operations use FFT‑based convolutions (O(n log n)).  The
implementation respects the 1000 s time budget and works for
`n` up to a few thousand.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (zero‑padded) using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two arbitrary 1‑D arrays using FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    size = _next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: len(a) + len(b) - 1]


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a whose sum is 1.
    Returns 2·n·max(conv(a,a)).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)


# ----------------------------------------------------------------------
# Analytic families for a quick warm‑start
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """U‑shaped family: a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """Sum of power‑laws from both ends."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """Exponential tails: a_i ∝ exp(-lam·min(i,n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a = a / a.sum()
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """Cosine power family: a_i ∝ |cos(πi/(n‑1))|^{p}."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    a = a / a.sum()
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped family: a_i ∝ (i+1)^{α}·(n‑i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = a / a.sum()
    return a


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian: a_i ∝ exp(- (i‑center)²/(2σ²))."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent using the exact max
# ----------------------------------------------------------------------


def subgradient_step(a: np.ndarray,
                     momentum: np.ndarray,
                     lr: float,
                     eps_active: float) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Perform one sub‑gradient iteration.
    Returns (new_a, new_momentum, max_conv).
    """
    n = a.shape[0]

    # exact convolution
    conv = conv_fft(a)                     # length 2n‑1
    max_val = conv.max()

    # indices where the max is attained (active set)
    # use a relative tolerance to collect near‑max entries
    tol = max(eps_active, max_val * 1e-12)
    active = np.where(conv >= max_val - tol)[0]

    # sub‑gradient = 2·(rev(a) ∗ w) on the active set,
    # where w is a uniform distribution on the active indices.
    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    rev_a = a[::-1]
    g_full = conv_fft_general(rev_a, w)   # length 3n‑2
    start = n - 1
    grad = 2.0 * g_full[start:start + n]

    # heavy‑ball momentum update
    momentum = 0.9 * momentum + grad
    a_new = a - lr * momentum

    # enforce feasibility
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_simplex(a_new)

    return a_new, momentum, max_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the supplied evaluation function.  The method first scans a few
    handcrafted analytic families, then refines the best candidate with an
    exact‑max sub‑gradient descent (heavy‑ball momentum + simplex projection)
    and finally applies a short stochastic polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose length n
    # ------------------------------------------------------------------
    if "n" in kwargs:
        n = int(kwargs["n"])
    elif "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        n = 2000                       # default used in earlier runs

    # ------------------------------------------------------------------
    # Warm‑start: scan several analytic families
    # ------------------------------------------------------------------
    best_seq = np.ones(n) / n          # fallback uniform
    best_score = compute_score(best_seq)

    # Product family
    for p in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5]:
        for off in [0.1, 0.3, 0.5, 0.7, 0.9, 1.1]:
            cand = analytic_product(n, p, off)
            sc = compute_score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()

    # Sum of power‑law family
    for p in [0.5, 0.8, 1.0, 1.2, 1.5]:
        for off in [0.1, 0.5, 1.0]:
            cand = analytic_sum(n, p, off)
            sc = compute_score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()

    # Exponential tails
    for lam in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3]:
        cand = analytic_exponential(n, lam)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()

    # Cosine power
    for p in [1, 2, 4, 8]:
        cand = analytic_cosine(n, p)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()

    # Beta‑shaped
    for a in [-0.5, 0.0, 0.5, 1.0, 2.0]:
        for b in [-0.5, 0.0, 0.5, 1.0, 2.0]:
            cand = analytic_beta(n, a, b)
            sc = compute_score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()

    # Gaussian centre
    for sigma in [n / 8.0, n / 6.0, n / 4.0, n / 3.0]:
        cand = analytic_gaussian(n, sigma)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()

    # ------------------------------------------------------------------
    # If a previous global sequence is available, try it as well
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] == n:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = compute_score(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print("[init] Started from global height_sequence_1, score =", sc)

    print(f"[init] best analytic start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient descent with exact max
    # ------------------------------------------------------------------
    a = best_seq.copy()
    momentum = np.zeros_like(a)

    # learning‑rate schedule
    lr0 = 0.4
    lr = lr0
    decay = 1e-5

    last_improve = time.time()
    it = 0
    eps_active = 1e-12

    while time.time() < deadline:
        it += 1

        a, momentum, cur_max = subgradient_step(a, momentum, lr, eps_active)
        cur_score = 2.0 * n * cur_max          # sum(a) == 1 by construction

        # record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{it:07d}] new best = {best_score:.6f}")

        # decay learning rate slowly
        lr = lr0 / (1.0 + decay * it)

        # jitter restart if we have been stagnant for a while
        if time.time() - last_improve > 30.0:
            jitter = np.random.randn(n) * 1e-4
            a = np.clip(best_seq + jitter, 0.0, None)
            a = project_simplex(a)
            momentum = np.zeros_like(a)
            last_improve = time.time()
            print(f"[{it:07d}] jitter restart")

        # occasional cheap random mass‑transfer move (helps escape tiny plateaux)
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            while i == j:
                j = np.random.randint(0, n)
            delta = a[i] * 0.1 * np.random.rand()
            a[i] -= delta
            a[j] += delta
            a = project_simplex(a)

        # safety break
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final stochastic refinement (Gaussian perturbations)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        perturb = np.random.normal(scale=5e-4, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, length = {n}, time = {total:.1f}s")
    return best_seq.tolist()
