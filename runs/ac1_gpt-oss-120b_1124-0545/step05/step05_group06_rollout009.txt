"""
Propose a non‑negative step‑function (probability vector) that minimises
`evaluate_sequence`.  

The search consists of three stages:

1. **Cheap parametric sweep** – generate a large pool of candidate
   distributions (uniform, arcsine, many Beta(α,β) draws and mixtures) and
   keep the best one according to the exact objective
   2·n·max(conv(a,a)) (the sum of the vector is normalised to 1).

2. **Projected gradient descent with a smooth max surrogate** – the
   objective  max(conv) is approximated by the log‑sum‑exp
   `f_p(a)= (1/p)·log Σ exp(p·conv_k)`.  Its gradient is a weighted
   combination of the usual sub‑gradients and can be evaluated
   efficiently by FFTs.  After every step the iterate is projected back
   onto the probability simplex (Euclidean projection).  The “temperature”
   `p` is increased from a small value to a large one, turning the smooth
   surrogate gradually into the true max.

3. **Local stochastic refinement** – a simulated‑annealing style
   neighbourhood search that moves a random amount of mass between two
   coordinates.  Moves are accepted if they improve the exact objective
   or with a Metropolis probability that depends on a linearly
   decreasing temperature.

All work respects a 1000 s budget, prints progress and finally returns
the best vector found (as a plain Python list).  The implementation uses
only NumPy and runs on a single thread; the FFT sizes are powers of two
so the convolution steps are fast even for a few‑thousand‑element
vectors.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are (numerically) non‑positive – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_distribution(n: int) -> np.ndarray:
    """
    Exact discretisation of the continuous arcsine density
    f(x)=1/(π√(x(1‑x))) on [0,1] by integrating over each bin.
    The resulting vector sums to 1.
    """
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta(α,β) density evaluated at the centre of each of the n bins.
    Returned vector is normalised to sum = 1.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def max_convolution_fft(a: np.ndarray, fft_len: int) -> float:
    """Linear convolution a*a using FFT, returning its maximum entry."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    # length of linear convolution = 2*len(a)-1
    L = 2 * a.size - 1
    return float(conv[:L].max())


def gradient_from_softmax_weights(a: np.ndarray, w: np.ndarray,
                                 fft_len: int) -> np.ndarray:
    """
    Gradient of  (1/p)·log Σ exp(p·conv) w.r.t. a,
    where w_k = exp(p·conv_k) / Σ exp(p·conv_k) .
    The formula is 2·a * w (cross‑correlation).  Implemented via FFT.
    """
    n = a.size
    # reverse w for the cross‑correlation
    w_rev = w[::-1]
    # FFT of both vectors (zero‑padded to fft_len)
    Fa = np.fft.fft(a, fft_len)
    Fw = np.fft.fft(w_rev, fft_len)
    # convolution a * w_rev  (full length)
    conv_full = np.real(np.fft.ifft(Fa * Fw))
    # the needed slice is indices n‑1 … 2n‑2  (inclusive)
    # this slice yields g_desc where g_desc[i] = Σ a_j w_{j + (n‑1‑i)}
    g_desc = conv_full[n - 1: 2 * n - 1]          # length n
    # reverse to obtain the proper order of the gradient
    g = 2.0 * g_desc[::-1]
    return g


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,

    # --------------------------------------------------------------
    n = int(kwargs.get("n", 4000))          # number of steps
    L = 2 * n - 1                           # length of linear convolution

    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    candidates = []

        candidates.append(mix)

    # --------------------------------------------------------------

        # ----- record improvement -----
