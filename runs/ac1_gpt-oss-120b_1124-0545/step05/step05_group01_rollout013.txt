def _lp_optimize(f, active_tol=0.005, max_outer=10, time_deadline):

   For iteration in range(max_outer):
        # compute conv
        conv_f_f = _conv_fft(f, f)
        M = conv_f_f.max()
        # choose active indices
        active_idx = np.where(conv_f_f >= M * (1 - active_tol))[0]
        # ensure at least 1.
        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv_f_f)])

        # shrink: maybe keep top 30 indices (largest conv values)
        if active_idx.size > 30:
            # get top 30 values
            top_order = np.argsort(-conv_f_f[active_idx])[:30]
            active = active_idx[top_order]
        else:
            active = active_idx

        # Build A matrix
        A = _build_constraint_matrix(f, active)

        # Solve LP: maximize sum(g) => minimize -sum(g)
        c = -np.ones(f.shape[0])
        bounds = [(0, None)] * f.shape[0]
        res = linprog(c, A_ub=A, b_ub=np.full(len(active), M), bounds=bounds, method='highs')
        if not res.success:
            # fallback: break
            break
        g_raw = res.x
        s = g_raw.sum()
        if s <= 1e-12:
            break
        g = g_raw / s

        # compute convs
        conv_f_g = _conv_fft(f, g)
        conv_g_g = _conv_fft(g, g)

        # line search minimize max conv
        def max_conv_of_t(t):
            mixed = (1 - t) ** 2 * conv_f_f + 2 * t * (1 - t) * conv_f_g + t ** 2 * conv_g_g
            return mixed.max()

        # golden section
        a, b = 0.0, 1.0
        phi = (np.sqrt(5) - 1) / 2  # conjugate
        # points
        x1 = a + (1 - phi) * (b - a)
        x2 = a + phi * (b - a)
        f1 = max_conv_of_t(x1)
        f2 = max_conv_of_t(x2)
        for _ in range(25):
            if time.time() > deadline: break
            if f1 > f2:
                a = x1
                x1 = x2
                f1 = f2
                x2 = a + phi * (b - a)
                f2 = max_conv_of_t(x2)
            else:
                b = x2
                x2 = x1
                f2 = f1
                x1 = a + (1 - phi) * (b - a)
                f1 = max_conv_of_t(x1)
        t_opt = (a + b) / 2

        # if improvement small break
        new_M = max_conv_of_t(t_opt)
        if new_M >= M * (1 - 1e-6):
            # no significant improvement
            break
        f = (1 - t_opt) * f + t_opt * g
        # renormalize
        f /= f.sum()
        # optional: check convergence
    return f

"""
Propose a step‑function with a very low self‑convolution ∞‑norm.

Algorithm
---------

1. **Seed generation** – a rich pool of starting vectors:
   uniform, geometric, power‑law, mixed geometric × power, symmetric
   geometric, sine‐window, Gaussian, Dirichlet draws, a few hand‑crafted
   “spike” and “triangular’’ shapes, and the previously best construction
   (if any).

2. **Sub‑gradient refinement** – exponentiated‑gradient (mirror‑descent)
   directly on the true objective  max(conv(a,a)).  This quickly drives the
   maximum convolution down while staying on the probability simplex.

3. **Linear‑program mixing** – for the current vector `f` we solve
   ``max Σ gᵢ  s.t. (f∗g) ≤ M, g≥0`` where ``M = max(f∗f)``.
   The LP is built only for the *active* convolution indices
   (those whose value is within a few per‑cent of ``M``); this keeps the
   matrix tiny (≤ 30 rows).  The optimal `g` is renormalised to a
   probability vector, then a line‑search chooses the best convex
   combination ``f←(1‑t) f + t g``.  Repeating this step (≈ 10 times) reduces
   the peak further.

4. **Simulated‑annealing fine‑tuning** – a cheap Metropolis walk that
   moves a tiny amount of mass between two positions; occasional global
   jitter helps escape shallow plateaus.

All steps respect a global time budget (default 1000 s).  The best
sequence found is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(seq,seq)) / (sum(seq))²."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = v.max()
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _two_spike(n: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[0] = 0.5
    a[-1] = 0.5
    return a


def _triangle(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1, dtype=np.float64)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2, dtype=np.float64)
        right = left[:-1][::-1]
    a = np.concatenate([left, right])
    return a / a.sum()


def _make_seeds(
    n: int, previous: np.ndarray | None
) -> list[np.ndarray]:
    """
    Assemble a diverse set of start vectors (already normalised to sum = 1).
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power laws
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # mixed geometric×power
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # Dirichlet draws
    for alpha in [0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # hand‑crafted shapes
    seeds.append(_two_spike(n))
    seeds.append(_triangle(n))

    # mirror each seed (helps when the optimum is asymmetric)
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # reuse previous best if length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # guarantee non‑empty list
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Sub‑gradient (exponentiated‑gradient) refinement
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 80000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the genuine sub‑gradient
    of max(conv).  Returns the best sequence seen and its objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.75 / np.sqrt(it)               # 1/√t schedule
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑program based mixing (active‑set)
# ----------------------------------------------------------------------


def _build_active_constraint_matrix(
    a: np.ndarray, active_idxs: np.ndarray
) -> sparse.csr_matrix:
    """
    Construct a sparse matrix A (|active| × n) such that (A @ g)[k] = (a∗g)[active_idxs[k]].
    Only the rows indexed by ``active_idxs`` are built.
    """
    n = a.size
    rows, cols, data = [], [], []

    for row_id, k in enumerate(active_idxs):
        # column indices that contribute to shift k
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        cols_i = np.arange(i_start, i_end + 1, dtype=int)
        vals = a[k - cols_i]                     # coefficients f_{k-i}
        rows.extend([row_id] * len(cols_i))
        cols.extend(cols_i.tolist())
        data.extend(vals.tolist())

    return sparse.csr_matrix((data, (rows, cols)), shape=(len(active_idxs), n))


def _lp_mix_refine(
    f: np.ndarray,
    deadline: float,
    max_outer: int = 12,
    active_tol: float = 0.005,
    max_active: int = 30,
) -> np.ndarray:
    """
    Repeatedly (i) solve   max Σ g  s.t. (f∗g) ≤ M
    (only the active convolution entries are kept) and
    (ii) mix the current f with the optimal g using a line‑search on t∈[0,1].
    Returns the refined vector (still normalised to sum = 1).
    """
    n = f.size
    for outer in range(max_outer):
        if time.time() > deadline:
            break

        conv_ff = _conv_fft(f, f)
        M = conv_ff.max()

        # pick the “active’’ constraints – those within a few per‑cent of M
        active = np.where(conv_ff >= M * (1.0 - active_tol))[0]
        if active.size == 0:
            active = np.array([np.argmax(conv_ff)])

        if active.size > max_active:
            # keep the top‑max_active values only
            idx = np.argsort(-conv_ff[active])[:max_active]
            active = active[idx]

        A = _build_active_constraint_matrix(f, active)

        # LP: maximise 1ᵀg  ⇔ minimise -1ᵀg
        c = -np.ones(n, dtype=np.float64)
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0, None)] * n

        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(1.0, deadline - time.time())},
        )
        if not res.success:
            # couldn't improve – abort
            break

        g_raw = res.x
        s = g_raw.sum()
        if s <= 1e-12:
            break
        g = g_raw / s

        # Pre‑compute the three convolutions needed for the line‑search
        conv_fg = _conv_fft(f, g)
        conv_gg = _conv_fft(g, g)

        # ----- golden‑section line‑search on t ∈ [0,1] -----
        phi = (np.sqrt(5) - 1) / 2  # conjugate golden ratio
        a_t, b_t = 0.0, 1.0
        x1 = a_t + (1 - phi) * (b_t - a_t)
        x2 = a_t + phi * (b_t - a_t)

        def _max_conv(t):
            mixed = (1 - t) ** 2 * conv_ff + 2 * t * (1 - t) * conv_fg + t ** 2 * conv_gg
            return mixed.max()

        f1 = _max_conv(x1)
        f2 = _max_conv(x2)

        for _ in range(25):
            if time.time() > deadline:
                break
            if f1 > f2:
                a_t = x1
                x1, f1 = x2, f2
                x2 = a_t + phi * (b_t - a_t)
                f2 = _max_conv(x2)
            else:
                b_t = x2
                x2, f2 = x1, f1
                x1 = a_t + (1 - phi) * (b_t - a_t)
                f1 = _max_conv(x1)

        t_opt = (a_t + b_t) / 2.0
        new_M = _max_conv(t_opt)

        # stop if no appreciable reduction
        if new_M >= M * (1 - 1e-8):
            break

        # mix and re‑normalise (mixing already preserves sum=1, but numerical drift may appear)
        f = (1 - t_opt) * f + t_opt * g
        f /= f.sum()

    return f


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (identical to the previous version)
# ----------------------------------------------------------------------


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis‑Hastings walk that moves a tiny fraction of mass
    between two random positions, with exponential cooling.
    Returns the best sequence discovered.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.025 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val
