"""
Propose a non‑negative step function (list of floats) that minimizes

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of several complementary phases carried out within the
given wall‑time budget.

1. **Rich analytic families** – grid‑search over several parametric families
   (product, sum, exponential, cosine‑power, beta) on the probability simplex.
   The 20 best vectors are retained.

2. **Convex mixing & permutation tricks** – all pairs of the retained vectors
   are mixed, a few three‑way random mixtures are tried, and a simple
   “alternating‑ends” permutation of the weights is evaluated (large weights
   are placed far apart to reduce collision in the convolution).

3. **Mirror‑descent on the simplex** – a sub‑gradient of the exact
   max‑convolution is used in an exponentiated‑gradient (mirror descent) step.
   The update is `a ← a * exp(-η·g)`, followed by renormalisation.  The step size
   η decays from 2.0 to 1e‑6.  Stagnation is broken by occasional random mass‑transfer
   moves and jitter restarts.

4. **Soft‑max (log‑sum‑exp) optimisation with Adam** – the non‑smooth max is
   approximated by a temperature‑controlled soft‑max.  The temperature β is
   increased from 0.5 to 800, and Adam updates are applied to the same simplex‑projected
   variables.

5. **Fine random refinement** – a short final phase adds Gaussian perturbations
   to the current best vector.

All intermediate vectors are kept on the probability simplex, therefore the
objective is invariant to scaling.  The routine respects the wall‑time budget
(≤ 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def conv_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of a and b using FFT."""
    size = _next_pow2(a.shape[0] + b.shape[0] - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1, a ≥ 0):
        2·n·max(conv(a,a))
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s                     # project onto simplex
    conv = conv_self(a)
    return float(2.0 * a.shape[0] * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small random amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def alternating_permutation(v: np.ndarray) -> np.ndarray:
    """
    Place sorted weights alternating from the left and right ends:
    largest → 0, second → n‑1, third → 1, fourth → n‑2, …
    """
    n = v.shape[0]
    order = np.argsort(v)[::-1]          # indices of descending weights
    new = np.empty_like(v)
    left, right = 0, n - 1
    turn = True
    for idx in order:
        if turn:
            new[left] = v[idx]
            left += 1
        else:
            new[right] = v[idx]
            right -= 1
        turn = not turn
    return new


# ----------------------------------------------------------------------
# Analytic families (each returns a probability vector)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    return a / a.sum()


def analytic_cosine(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    return a / a.sum()


def analytic_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta_
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Helper for top‑k maintenance
# ----------------------------------------------------------------------


def _add_candidate(cand: np.ndarray,
                   best_list: list[tuple[float, np.ndarray]],
                   max_len: int = 20) -> None:
    """Insert cand into the sorted best‑list (by score)."""
    sc = compute_score(cand)
    inserted = False
    for idx, (s, _) in enumerate(best_list):
        if sc < s:
            best_list.insert(idx, (sc, cand.copy()))
            inserted = True
            break
    if not inserted and len(best_list) < max_len:
        best_list.append((sc, cand.copy()))
    while len(best_list) > max_len:
        best_list.pop()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimizes the evaluation function.
    Returns a list of non‑negative floats (a probability vector);
    the evaluation routine is scale‑invariant, so we keep the sum = 1.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # Length of the sequence (default 2000, can be overridden)
    n = int(kwargs.get("n", 2000))

    print("[0] ==== Analytic grid search (n = %d) ====" % n)

    top = []   # list of (score, vector)

    # ---------- product family ----------
    p_vals = np.linspace(0.1, 3.0, 30)               # 30 values
    off_vals_geom = np.geomspace(1e-4, 0.01, 5)
    off_vals_lin = np.arange(0.01, 5.01, 0.3)
    off_vals = np.concatenate((off_vals_geom, off_vals_lin))
    for p in p_vals:
        for off in off_vals:
            cand = analytic_product(n, p, off)
            _add_candidate(cand, top)

    # ---------- sum family ----------
    for p in p_vals:
        for off in off_vals:
            cand = analytic_sum(n, p, off)
            _add_candidate(cand, top)

    # ---------- exponential tail ----------
    lam_vals = np.logspace(-3, 1, 30)               # 30 values between 0.001 and 10
    for lam in lam_vals:
        cand = analytic_exponential(n, lam)
        _add_candidate(cand, top)

    # ---------- cosine power ----------
    p_cos = np.linspace(0.5, 8.0, 16)               # 16 values
    for p in p_cos:
        cand = analytic_cosine(n, p)
        _add_candidate(cand, top)

    # ---------- beta family ----------
    alpha_vals = np.linspace(-0.5, 2.5, 9)           # 9 values
    beta_vals = np.linspace(-0.5, 2.5, 9)
    for a in alpha_vals:
        for b in beta_vals:
            cand = analytic_beta(n, a, b)
            _add_candidate(cand, top)

    # ---------- incorporate provided reference ----------
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            _add_candidate(ref, top)

    best_seq, best_score = top[0][1].copy(), top[0][0]
    print("[0] Best after grid search: score = %.6f" % best_score)

    # ------------------------------------------------------------------
    # 2) Pairwise and triple convex mixing, plus permutation tricks
    # ------------------------------------------------------------------
    print("[1] Convex mixing of top candidates and permutation heuristics")
    # pairwise mixing
    alphas = np.linspace(0.0, 1.0, 21)   # 0,0.05,…,1
    for i in range(len(top)):
        for j in range(i + 1, len(top)):
            a_i = top[i][1]
            a_j = top[j][1]
            for alpha in alphas:
                cand = alpha * a_i + (1.0 - alpha) * a_j
                cand = project_simplex(cand)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
    # a few random three‑way mixtures
    for _ in range(2000):
        idxs = np.random.choice(len(top), size=3, replace=False)
        w = np.random.dirichlet([1.0, 1.0, 1.0])
        cand = (w[0] * top[idxs[0]][1] +
                w[1] * top[idxs[1]][1] +
                w[2] * top[idxs[2]][1])
        cand = project_simplex(cand)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
    # permutation: alternating‑ends ordering of current best
    alt = alternating_permutation(best_seq)
    sc_alt = compute_score(alt)
    if sc_alt < best_score:
        best_score, best_seq = sc_alt, alt.copy()
    print("[1] Score after mixing & permute: %.6f" % best_score)

    # ------------------------------------------------------------------
    # 3) Mirror‑descent (exponentiated gradient) on the exact max‑conv
    # ------------------------------------------------------------------
    print("[2] Mirror‑descent on simplex (exponentiated gradient)")
    a = best_seq.copy()
    step_start, step_end = 2.0, 1e-6
    last_improve = time.time()
    stagnation_time = 12.0   # seconds without improvement → jitter

    while time.time() < deadline - 300:   # reserve ~5 min for later phases
        conv = conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score, best_seq = cur_score, a.copy()
            last_improve = time.time()
            print("[sg] new best = %.6f" % best_score)

        # sub‑gradient of the active max term
        max_idx = int(conv.argmax())
        shift = max_idx - np.arange(n)
        mask = (shift >= 0) & (shift < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[shift[mask]]

        # step size schedule
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 300.0)
        step = step_start * (1.0 - progress) + step_end * progress

        # exponentiated (mirror) update
        a = a * np.exp(-step * g)
        a = a / a.sum()                     # stays on simplex

        # random mass‑transfer move (occasionally)
        if np.random.rand() < 0.02:
            a = random_move(a, step * 0.5)

        # jitter restart on stagnation
        if time.time() - last_improve > stagnation_time:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            last_improve = time.time()
            print("[sg] jitter restart")

    # ------------------------------------------------------------------
    # 4) Soft‑max (log‑sum‑exp) optimisation with Adam
    # ------------------------------------------------------------------
    print("[3] Adam optimisation on softened max (temperature schedule)")
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_adam = 0
    lr = 0.12
    beta_start, beta_end = 0.5, 800.0
    last_improve = time.time()
    stagnation_time = 20.0

    while time.time() < deadline - 30:   # keep ~30 s for final refinement
        conv = conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score, best_seq = cur_score, a.copy()
            last_improve = time.time()
            print("[adam] new best = %.6f" % best_score)

        # temperature schedule (smooth → sharp)
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 30.0)
        beta = beta_start * (1.0 - progress) + beta_end * progress

        # soft‑max weights over all shift indices
        w_raw = np.exp(beta * (conv - max_val))   # max exponent = 0
        w = w_raw / w_raw.sum()

        # weighted shift: correlation of w with reversed a
        a_rev = a[::-1]
        corr = conv_two(w, a_rev)                # length = 3n‑2
        weighted = corr[n - 1:n - 1 + n]         # same length as a

        # gradient of the soft‑max objective
        g = a + weighted

        # Adam update
        t_adam += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step_vec
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate = project_simplex(a_candidate)

        # occasional exploratory moves
        if np.random.rand() < 0.015:
            a_candidate = random_move(a_candidate, lr * 0.3)
        if np.random.rand() < 0.005:
            jitter = np.random.randn(n) * 5e-4
            a_candidate = a_candidate + jitter
            a_candidate = np.clip(a_candidate, 0.0, None)
            a_candidate = project_simplex(a_candidate)

        # jitter restart on stagnation
        if time.time() - last_improve > stagnation_time:
            jitter = np.random.rand(n) * 1e-3
            a_candidate = best_seq + jitter
            a_candidate = project_simplex(a_candidate)
            last_improve = time.time()
            print("[adam] jitter restart")

        a = a_candidate

    # ------------------------------------------------------------------
    # 5) Final fine‑grained random refinement
    # ------------------------------------------------------------------
    print("[4] Final random perturbation phase")
    while time.time() < deadline:
        perturb = np.random.normal(scale=5e-4, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand.copy()
            print("[refine] improved to %.6f" % best_score)

    total_time = time.time() - start_time
    print("[done] best score = %.6f, length = %d, time = %.1fs"
          % (best_score, n, total_time))

    # Return as a plain Python list (the evaluation function accepts this)
    return best_seq.tolist()
