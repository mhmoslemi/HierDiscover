"""
Propose a non‑negative step function that (approximately) minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------

* Phase 0 – basic utilities (FFT convolution, simplex projection, etc.).
* Phase 1 – random search over several low‑dimensional analytic families
  (product, sum‑of‑powers, exponential, cosine‑power, Beta‑shaped,
  and mixtures).  For each sampled parameter set we build a normalised
  vector and evaluate the exact objective.  The best vector found is kept.
* Phase 2 – local refinement of that best vector with Adam optimisation of a
  smooth approximation of the “max” (soft‑max of the convolution).  The
  temperature α is increased gradually from 0.5 up to ~500 while the vector
  stays projected onto the probability simplex.
* Phase 3 – final stochastic fine‑tuning: tiny Gaussian perturbations are
  accepted only when they improve the exact score.

The whole procedure stops after the given time budget (default 1000 s) and
returns the best sequence discovered.  All numbers are non‑negative and
the algorithm works for any length `n` (default 2000, a power of two for
FFT speed).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (zero‑padded) using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two arbitrary 1‑D arrays using FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    size = _next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: len(a) + len(b) - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a (sum normalised to 1).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = project_simplex(a_new)
    return a_new


# ----------------------------------------------------------------------
# Analytic families (all return normalised vectors)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, off: float) -> np.ndarray:
    """U‑shaped product family: a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + off) * (n - 1 - i + off)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, off: float) -> np.ndarray:
    """Sum‑of‑powers family: a_i ∝ (i+off)^{-p} + (n‑1‑i+off)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + off) ** (-p) + (n - 1 - i + off) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """Exponential‑tail family: a_i ∝ exp(-lam·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a = a / a.sum()
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """Cosine‑power family: a_i ∝ |cos(π i/(n‑1))|^{p}."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    a = a / a.sum()
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped family: a_i ∝ (i+1)^{α}·(n‑i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Smoothed‑max gradient (soft‑max of the convolution)
# ----------------------------------------------------------------------


def smooth_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    using FFTs.  Returns a vector of length len(a).
    """
    n = a.shape[0]
    conv = conv_fft(a)                         # length 2n‑1
    maxc = conv.max()
    exp_alpha = np.exp(alpha * (conv - maxc))
    w = exp_alpha / exp_alpha.sum()           # Σ w = 1

    # gradient = 2·(rev(a) ∗ w)[n‑1 : n‑1+n]
    rev_a = a[::-1]
    g_full = conv_fft_general(rev_a, w)       # length 3n‑2
    start = n - 1
    g = 2.0 * g_full[start:start + n]
    return g


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the objective.
    The search consists of a quick random search over several analytic
    families followed by a smooth‑max Adam refinement and a final
    stochastic polishing.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # slight safety margin

    # ------------------------------------------------------------------
    # 1. Determine length n
    # ------------------------------------------------------------------
    if "n" in kwargs:
        n = int(kwargs["n"])
    elif "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        # use a power of two for FFT speed; 2000 is the historic baseline
        n = 2048

    # ------------------------------------------------------------------
    # 2. Initialise the best known vector (use global if available)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=float)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s >= 1e-12:
            cand = cand / s
        else:
            cand = np.ones_like(cand) / cand.shape[0]
        sc = compute_score(cand)
        best_seq = cand.copy()
        best_score = sc
        print(f"[init] Using global height_sequence_1: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Random search over analytic families
    # ------------------------------------------------------------------
    families = ["product", "sum", "exponential", "cosine", "beta", "mixture"]
    param_deadline = deadline - 200.0   # reserve ~200 s for later phases
    trials = 0
    while time.time() < param_deadline:
        f = np.random.choice(families)
        if f == "product":
            # log‑uniform sampling for p and off
            p = 10 ** np.random.uniform(-2, 1)          # 0.01 .. 10
            off = 10 ** np.random.uniform(-2, 1)        # 0.01 .. 10
            cand = analytic_product(n, p, off)
        elif f == "sum":
            p = 10 ** np.random.uniform(-2, 1)
            off = 10 ** np.random.uniform(-2, 1)
            cand = analytic_sum(n, p, off)
        elif f == "exponential":
            lam = np.random.uniform(0.0, 5.0)
            cand = analytic_exponential(n, lam)
        elif f == "cosine":
            p = np.random.uniform(0.0, 12.0)
            cand = analytic_cosine(n, p)
        elif f == "beta":
            alpha = np.random.uniform(-2.0, 5.0)
            beta = np.random.uniform(-2.0, 5.0)
            cand = analytic_beta(n, alpha, beta)
        elif f == "mixture":
            # mixture of two product families
            p1 = 10 ** np.random.uniform(-2, 1)
            off1 = 10 ** np.random.uniform(-2, 1)
            p2 = 10 ** np.random.uniform(-2, 1)
            off2 = 10 ** np.random.uniform(-2, 1)
            w = np.random.rand()
            cand = w * analytic_product(n, p1, off1) \
                 + (1.0 - w) * analytic_product(n, p2, off2)
            cand = cand / cand.sum()
        else:
            continue

        # Normalise (protect against nan/inf)
        if np.any(np.isnan(cand)) or np.any(np.isinf(cand)):
            continue
        cand = np.maximum(cand, 0.0)
        total = cand.sum()
        if total < 1e-12:
            continue
        cand = cand / total

        sc = compute_score(cand)
        trials += 1
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[param] trial {trials:06d} new best {best_score:.6f}  family={f}")

    # If nothing was found (unlikely), fall back to uniform.
    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = compute_score(best_seq)
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 4. Adam optimisation of the smoothed‑max objective
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # smoothing temperature α schedule
    alpha = 0.5
    alpha_max = 500.0
    # aim for a geometric growth that reaches α_max after ~150 k steps
    growth_factor = (alpha_max / alpha) ** (1.0 / 150000.0)

    step = 0
    last_improve = time.time()
    grad_deadline = deadline - 40.0   # keep ~40 s for final refinement

    while time.time() < grad_deadline:
        step += 1

        # gradient of smooth‑max
        grad = smooth_max_gradient(a, alpha)

        # Adam update
        lr = 0.12 / (1.0 + step * 1e-5)   # slowly decaying learning rate
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # feasibility projection
        a = np.clip(a, 0.0, None)
        a = project_simplex(a)

        # temperature schedule
        alpha = min(alpha_max, alpha * growth_factor)

        # occasional true‑score evaluation
        if step % 250 == 0:
            cur = compute_score(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[adam] step {step:07d} new best {best_score:.6f} α={alpha:.2f}")

        # jitter restart if stagnation > 30 s
        if time.time() - last_improve > 30.0:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            last_improve = time.time()
            print(f"[adam] step {step:07d} jitter restart")

        # occasional random mass‑transfer move to escape tiny plateaux
        if np.random.rand() < 0.02:
            a = random_move(a, scale=0.02)

    # ------------------------------------------------------------------
    # 5. Final stochastic refinement (cheap Gaussian perturbations)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        perturb = np.random.normal(scale=5e-4, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, length = {n}, time = {total_time:.1f}s")
    return best_seq.tolist()
