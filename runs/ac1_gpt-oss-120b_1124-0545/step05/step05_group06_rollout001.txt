    # compute arcsine distribution
    a_arc = arcsine_distribution(n)

    uniform = np.full(n, 1.0 / n)

    # store candidates
    candidates = [a_arc, uniform]

    # fine grid for beta alpha near 0.5
    for alpha in np.linspace(0.3, 0.8, 26):
        candidates.append(beta_distribution(n, alpha, alpha))

    # mixture arcsine--uniform fine grid
    for lam in np.linspace(0.0, 1.0, 21):
        mix = lam * a_arc + (1 - lam) * uniform
        mix = mix / mix.sum()
        candidates.append(mix)

    # evaluate each candidate via max_convolution function (using FFT length)
    fft_len = 1 << ((2*n-1).bit_length())
    best_score = float('inf')
    best_a = None
    for idx, a in enumerate(candidates):
        # ensure nonneg and sum to 1
        a = np.maximum(a, 0.0)
        a_sum = a.sum()

        a = a / a_sum

        score = 2.0 * n * conv_max
        if score < best_score:

            best_a = a.copy()

    # Optional projected gradient descent
    if time.time() < deadline - 5.0:

"""
Propose a non‑negative step function (a probability vector) that minimises
`evaluate_sequence`.  The strategy is:

1.  Choose a large problem size (default 12 000) – the objective error
    decays like O(1/n) and a larger n brings the value close to the
    theoretical optimum ≈1.5.
2.  Build a pool of simple analytic candidates:
        * the exact arcsine distribution (Beta(½,½));
        * the uniform distribution;
        * symmetric Beta(α,α) for a fine grid of α around 0.5;
        * mixtures of arcsine and uniform (λ·arcsine+(1‑λ)·uniform).
   The best of these is kept.
3.  Run a short projected sub‑gradient descent on the true L∞‑objective
    (the gradient is the average of the two shifted copies of *a* at the
    maximising convolution index).  This removes any tiny remaining
    irregularities.
4.  Finish with a light simulated‑annealing style refinement that moves a
    small amount of mass between two random coordinates.
5.  Return the best normalised vector found.

The whole routine respects the 1000 s budget (a safety margin of 0.5 s is
kept).  All helper functions are top‑level and avoid closures.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """
    Return max(conv(a,a)) where conv is linear convolution.
    Computed via FFT of length `fft_len` (power of two ≥ 2*len(a)-1).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv.max())


def subgradient_max_conv(a: np.ndarray, conv: np.ndarray,
                         max_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at the set of maximising indices.
    For each maximiser s we add 2·a shifted by s; the result is averaged.
    """
    n = a.size
    g = np.zeros_like(a)
    for s in max_idx:
        i_start = max(0, s - (n - 1))
        i_end = min(n - 1, s)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = s - i
        g[i] += 2.0 * a[j]
    if max_idx.size > 0:
        g /= float(max_idx.size)
    return g


def arcsine_distribution(n: int) -> np.ndarray:
    """
    Discretisation of the arcsine density f(x)=1/(π√(x(1‑x))) on [0,1]
    by integrating over each bin [i/n,(i+1)/n).  Returns a probability vector.
    """
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric Beta(α,β) density evaluated at the centre of each bin and
    normalised to sum = 1.  Works for any α,β>0.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    The algorithm builds analytic candidates, refines them with a few
    projected‑gradient steps and a tiny simulated‑annealing phase.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # a tiny margin so that the caller does not kill us exactly at the limit
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # problem size (larger n → smaller asymptotic error)
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 12000))
    # keep n reasonable in case the caller gave a huge number
    n = max(2, min(n, 30000))

    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two

    # --------------------------------------------------------------
    # 0) construct a small pool of analytic candidates
    # --------------------------------------------------------------
    candidates = []

    # previously discovered construction (if any)
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.size == n and a0.sum() > 0.0:
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # arcsine – the known strong candidate
    arcsine = arcsine_distribution(n)
    candidates.append(arcsine)

    # uniform – baseline
    uniform = np.full(n, 1.0 / n)
    candidates.append(uniform)

    # fine sweep of symmetric Beta(α,α) around the arcsine exponent 0.5
    for alpha in np.linspace(0.30, 0.80, 26):      # 0.30,0.32,…,0.80
        candidates.append(beta_distribution(n, alpha, alpha))

    # mixtures of arcsine and uniform – scan λ∈[0,1]
    for lam in np.linspace(0.0, 1.0, 21):         # step 0.05
        mix = lam * arcsine + (1.0 - lam) * uniform
        mix /= mix.sum()
        candidates.append(mix)

    # --------------------------------------------------------------
    # 1) pick the best among the cheap candidates
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    for idx, a in enumerate(candidates):
        a = np.maximum(a, 0.0)
        tot = a.sum()
        if tot == 0.0:
            continue
        a = a / tot
        mx = max_convolution(a, fft_len)
        score = 2.0 * n * mx
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[init {idx:02d}] score={score:.6f}")
        else:
            print(f"[init {idx:02d}] score={score:.6f}")

    # --------------------------------------------------------------
    # 2) short projected sub‑gradient descent (≈2000 iterations)
    # --------------------------------------------------------------
    # leave a small buffer for the final annealing phase
    pgd_deadline = deadline - 5.0
    if time.time() < pgd_deadline and best_a is not None:
        a = best_a.copy()
        it = 0
        last_improve = time.time()
        while time.time() < pgd_deadline:
            # convolution via FFT
            fa = np.fft.fft(a, fft_len)
            conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
            conv_max = float(conv.max())
            max_idxs = np.where(np.abs(conv - conv_max) < 1e-14)[0]

            # sub‑gradient
            g = subgradient_max_conv(a, conv, max_idxs)

            # diminishing step size
            it += 1
            lr = 0.15 / np.sqrt(it)

            a = proj_simplex(a - lr * g)

            cur_score = 2.0 * n * conv_max
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                last_improve = time.time()
                print(f"[PGD {it:06d}] improved to {best_score:.6f}")

            # simple restart if we have been idle for ~30 s
            if time.time() - last_improve > 30.0:
                a = proj_simplex(best_a + rng.normal(scale=0.01, size=n))
                last_improve = time.time()
                print(f"[PGD {it:06d}] restart after stagnation")

    # --------------------------------------------------------------
    # 3) very light simulated‑annealing style refinement
    # --------------------------------------------------------------
    if time.time() < deadline and best_a is not None:
        a = best_a.copy()
        best_sa_score = best_score
        # linear temperature schedule
        T_start = best_sa_score * 0.005
        T_end = best_sa_score * 1e-6
        sa_start = time.time()
        while time.time() < deadline:
            elapsed = time.time() - sa_start
            frac = min(1.0, elapsed / (deadline - sa_start))
            T = T_start * (1.0 - frac) + T_end * frac

            # pick two distinct indices
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            # transfer up to 5 % of a[i] to a[j]
            max_move = min(a[i], 0.05 * a[i] + 1e-12)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            new_mx = max_convolution(a_new, fft_len)
            new_score = 2.0 * n * new_mx

            if new_score < best_sa_score or rng.random() < np.exp(-(new_score - best_sa_score) / (T + 1e-12)):
                a = a_new
                if new_score < best_sa_score:
                    best_sa_score = new_score
                    best_a = a_new.copy()
                    print(f"[SA] new best = {best_sa_score:.6f} (t={time.time() - start_time:.1f}s)")

    # --------------------------------------------------------------
    # final normalisation and return
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    elapsed = time.time() - start_time
    print(f"[done] time={elapsed:.1f}s best_score={best_score:.6f} n={n}")
    return best_a.tolist()
