"""
Propose a step function (non‑negative vector) that minimizes the
`evaluate_sequence` objective.

Algorithm
---------
1. **Initial pool** – a collection of simple parametric families
   (uniform, arcsine, symmetric / asymmetric Beta, random Dirichlet
   draws, mixtures) is built and the best member (lowest exact score)
   is kept.

2. **Smooth projected gradient descent** – the true objective
   `2·n·max(conv(a,a))` is replaced by a smooth approximation
   `2·n· (1/γ)·log Σ exp(γ·conv_i)`.  Its gradient can be written
   analytically as a convolution, allowing an O(n log n) update.
   A momentum term and an adaptive learning‑rate (`lr = lr0/√t`) are
   used, together with a Euclidean projection onto the probability
   simplex after each step.  The smoothing parameter γ is increased
   slowly from 1 to 20, turning the smooth problem into the original
   nonsmooth one.

3. **Local stochastic refinement** – a simulated‑annealing style
   mover swaps a small amount of probability mass between two random
   coordinates.  Moves that improve the exact objective are kept,
   otherwise they are accepted with a temperature‑dependent probability.

The whole routine respects the 1000 s budget, prints intermediate
scores and returns the best vector found (as a plain Python list).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                 # numerically all zeros
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the linear convolution a*a (real FFT)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv.max())


def arcsine_distribution(n: int) -> np.ndarray:
    """Exact discretisation of the arcsine density on [0,1]."""
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density evaluated at bin centres and normalised."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point from Dirichlet(conc,…,conc)."""
    x = rng.gamma(conc, 1.0, size=n)
    total = x.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return x / total


def smooth_gradient(a: np.ndarray,
                    fft_len_full: int,
                    len_w: int,
                    gamma: float) -> tuple[np.ndarray, float]:
    """
    Compute the weighted gradient of `max(conv(a,a))` using a smooth
    soft‑max approximation with parameter `gamma`.  Returns the gradient
    (already multiplied by 2) and the current maximal convolution value.
    """
    # FFT of a (zero‑padded to length fft_len_full)
    a_fft = np.fft.fft(a, fft_len_full)

    # ----- convolution a*a (only needed for the weight) -----
    conv_full = np.real(np.fft.ifft(a_fft * a_fft))[:len_w]
    conv_max = float(conv_full.max())

    # ----- smooth weights (softmax) -----
    # subtract conv_max to avoid overflow
    w = np.exp(gamma * (conv_full - conv_max))
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / len_w)
    else:
        w /= w_sum

    # ----- compute gradient via two convolutions -----
    # a * w  (term2)
    w_fft = np.fft.fft(w, fft_len_full)
    conv_a_w = np.real(np.fft.ifft(a_fft * w_fft))

    term2 = conv_a_w[:a.size]                     # Σ a_j·w_{i‑j}

    # a * w_rev  (term1)
    w_rev = w[::-1]
    w_rev_fft = np.fft.fft(w_rev, fft_len_full)
    conv_a_w_rev = np.real(np.fft.ifft(a_fft * w_rev_fft))

    # indices for term1: len_w‑1‑i
    idx = len_w - 1 - np.arange(a.size)
    term1 = conv_a_w_rev[idx]                     # Σ a_j·w_{i+j}

    grad = 2.0 * (term1 + term2)                  # full sub‑gradient
    return grad, conv_max


def evaluate_vec(a: np.ndarray, fft_len: int) -> float:
    """
    Fast exact evaluation of `evaluate_sequence` for a normalised vector.
    (sum(a)=1).  Returns 2·n·max(conv).
    """
    mx = max_convolution(a, fft_len)
    return 2.0 * a.size * mx


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the
    `evaluate_sequence` score.  The routine follows the three‑phase
    scheme described in the module docstring and respects the given
    time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # tiny safety margin so the platform does not kill us
    overall_deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 2000))               # length of the vector
    len_w = 2 * n - 1
    # FFT size that can hold a convolution of length up to 3·n‑2
    fft_len_full = 1 << ((3 * n - 2).bit_length())
    # FFT size for the plain max‑convolution (only a*a)
    fft_len_conv = 1 << ((len_w).bit_length())

    # --------------------------------------------------------------
    # 0) build an initial pool of cheap candidates
    # --------------------------------------------------------------
    candidates = []

    # reuse a previously discovered construction if available
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.size == n and a0.sum() > 0.0:
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # basic families
    candidates.append(arcsine_distribution(n))            # arcsine
    candidates.append(np.full(n, 1.0 / n))                # uniform

    # symmetric beta (α=β) sweep
    for alpha in np.linspace(0.25, 2.0, 16):
        candidates.append(beta_distribution(n, alpha, alpha))

    # asymmetric beta – a modest grid
    for alpha in [0.3, 0.6, 1.0, 1.4]:
        for beta in [0.3, 0.6, 1.0, 1.4]:
            if (alpha, beta) != (1.0, 1.0):
                candidates.append(beta_distribution(n, alpha, beta))

    # random Dirichlet draws (different concentration)
    for conc in [0.3, 0.7, 1.5, 3.0]:
        for _ in range(5):
            candidates.append(random_dirichlet(n, conc, rng))

    # mixtures of arcsine and uniform
    arcsine = candidates[1] if len(candidates) > 1 else arcsine_distribution(n)
    uniform = candidates[2] if len(candidates) > 2 else np.full(n, 1.0 / n)
    for lam in np.linspace(0.0, 1.0, 11):
        mix = lam * arcsine + (1.0 - lam) * uniform
        mix = np.maximum(mix, 0.0)
        mix /= mix.sum()
        candidates.append(mix)

    # --------------------------------------------------------------
    # 1) pick the best among the cheap candidates
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    for idx, a in enumerate(candidates):
        sc = evaluate_vec(a, fft_len_conv)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[init {idx:02d}] new best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) smooth projected gradient descent (major part of the budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.70 * budget_s   # 70 % for GD
    a = best_a.copy()
    m = np.zeros_like(a)          # momentum term
    momentum = 0.9
    it = 0
    last_improve = time.time()
    gamma = 1.0
    gamma_inc = 0.015             # growth per iteration (≈ to reach ~15)
    max_gamma = 15.0
    lr0 = 0.2                     # base learning rate

    while time.time() < phase1_deadline:
        it += 1
        # compute smooth gradient and current max‑convolution
        grad, conv_max = smooth_gradient(a, fft_len_full, len_w, gamma)

        # learning‑rate schedule (inverse square‑root)
        lr = lr0 / np.sqrt(it)

        # momentum update & gradient step
        m = momentum * m + (1.0 - momentum) * grad
        a = a - lr * m
        a = proj_simplex(a)                     # stay on the simplex

        cur_score = 2.0 * n * conv_max
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[GD {it:07d}] improved to {best_score:.6f}")

        # increase smoothing parameter slowly, but never exceed max_gamma
        gamma = min(max_gamma, gamma + gamma_inc)

        # restart if we have been stagnant for a while
        if time.time() - last_improve > 25.0:
            # small random perturbation around the current best
            noise = rng.normal(scale=0.01, size=n)
            a = proj_simplex(best_a + noise)
            m = np.zeros_like(a)
            last_improve = time.time()
            print(f"[GD {it:07d}] restart after stagnation")

    # --------------------------------------------------------------
    # 3) local stochastic refinement (simulated‑annealing style)
    # --------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 5.0:
        a = best_a.copy()
        best_conv_max = best_score / (2.0 * n)
        # temperature schedule
        T_start = best_score * 0.005
        T_end = best_score * 1e-6
        sa_start = time.time()
        sa_it = 0

        while time.time() < overall_deadline:
            elapsed = time.time() - sa_start
            frac = min(1.0, elapsed / remaining)
            T = T_start * (1.0 - frac) + T_end * frac

            # pick two distinct indices
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            # move up to 5 % of a[i] to a[j]
            max_move = min(a[i], 0.05 * a[i] + 1e-12)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # exact evaluation of the proposal (fast FFT version)
            new_score = evaluate_vec(a_new, fft_len_conv)

            # Metropolis acceptance
            if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    print(f"[SA {sa_it:07d}] new best = {best_score:.6f} (t={time.time() - start_time:.1f}s)")

            sa_it += 1

    # --------------------------------------------------------------
    # Finalisation
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    elapsed_total = time.time() - start_time
    print(f"[done] total time = {elapsed_total:.1f}s, best score = {best_score:.6f}, length = {n}")
    return best_a.tolist()
