"""
Hybrid LP‑guided refinement + gradient polishing for the max‑auto‑convolution
problem.

The algorithm works for each candidate length `n` (e.g. 2000, 4000, 8000):

1. Generate a modest collection of analytic seed shapes (uniform, sine‑window,
   geometric, power‑law, beta‑like, random Dirichlet).  Keep the best seed.

2. **LP‑guided improvement** – starting from the current best shape `f`,
   repeat:
      * compute M = max(conv(f,f))
      * pick the `k` largest convolution entries (they are the active
        constraints)
      * build a dense matrix `A` that encodes the linear constraints
        `conv(f,g) ≤ M` for those indices
      * solve the linear program  max ∑ g   s.t.   g≥0, A g ≤ M
        (the LP is tiny – only `k` rows, `n` columns)
      * normalise the solution to a probability vector `ĝ`
      * line‑search a mixing coefficient `t∈[0,1]` for the convex combination
        `f₊ = (1‑t) f + t ĝ` that minimises
        `max(conv(f₊,f₊))`.
      * replace `f←f₊` and keep the globally best sequence.
   The loop stops when the time budget is exhausted or no improvement
   appears.

3. Polish the best sequence with three deterministic refinements that were
   already used in the reference code:
   * Adam optimisation on a smooth approximation of the max,
   * exact sub‑gradient mirror descent,
   * cheap stochastic hill‑climbing.

All phases respect a global 1000 s budget.  The final sequence is stored in
the global variable ``height_sequence_1`` and returned as a plain list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Analytic seed families (ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a modest set of diverse normalised start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    for a in (0.3, 0.5, 0.7):
        for b in (0.3, 0.5, 0.7):
            seeds.append(_beta_shape(n, a, b))

    rng = np.random.default_rng(2023)
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None:
        # up‑sample / down‑sample previous best to the new length
        if previous.shape[0] != n:
            x_old = np.linspace(0, 1, previous.shape[0])
            x_new = np.linspace(0, 1, n)
            up = np.interp(x_new, x_old, previous)
            up = np.maximum(up, 0.0)
            up = up / up.sum()
            seeds.append(up)
        else:
            seeds.append(previous / previous.sum())

    return seeds

# Adam optimisation on a smoothed max (same as reference code)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑softmax parametrisation, using a smooth‑max surrogate."""
    v = np.log(seed + 1e-30)          # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0            # initial smoothing coefficient
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient w.r.t. a of Σ w_k·conv_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val

def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """Random pair‑wise mass transfer, keeping improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.01  # initial relative step size

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 1000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# LP‑guided refinement (core new idea)
# ----------------------------------------------------------------------


def _lp_refine(start_seq: np.ndarray,
               n: int,
               deadline: float,
               max_iters: int = 150,
               top_k: int = 30) -> tuple[np.ndarray, float]:
    """
    Iteratively improve `start_seq` by solving a tiny LP that enforces
    conv(start,g) ≤ M for the `top_k` largest entries of conv(start,start).

    Returns the best sequence found and its objective value.
    """
    # Try importing cvxpy – if unavailable we simply skip LP refinement.
    try:
        import cvxpy as cp
    except Exception as exc:          # pragma: no cover
        print("[LP] cvxpy not available – skipping LP phase")
        return start_seq, 2.0 * n * np.max(_conv_fft(start_seq, start_seq))

    f = start_seq.copy()
    f = np.maximum(f, 0.0)
    f = f / f.sum()                     # enforce ℓ₁‑norm = 1

    best_seq = f.copy()
    best_val = 2.0 * n * np.max(_conv_fft(f, f))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv_ff = _conv_fft(f, f)
        M = conv_ff.max()

        # select the top_k indices (active constraints)
        if top_k >= conv_ff.size:
            active = np.arange(conv_ff.size)
        else:
            active = np.argpartition(conv_ff, -top_k)[-top_k:]

        # build the (len(active) × n) constraint matrix
        A = np.zeros((len(active), n), dtype=np.float64)
        for r, k_idx in enumerate(active):
            # coefficients for g_j are f_{k_idx‑j} when the index is valid
            j_start = max(0, k_idx - (n - 1))
            j_end = min(k_idx, n - 1)
            js = np.arange(j_start, j_end + 1)
            is_ = k_idx - js
            A[r, js] = f[is_]

        # linear program: maximise sum(g)  s.t.  g≥0 , A g ≤ M
        g_var = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)),
                          [g_var >= 0,
                           A @ g_var <= M])
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
        except Exception as exc:  # pragma: no cover
            print(f"[LP] solve error: {exc}")
            break

        if g_var.value is None:
            print("[LP] no solution – terminating LP loop")
            break

        g_opt = np.maximum(np.array(g_var.value, dtype=np.float64), 0.0)
        s = g_opt.sum()
        if s <= 1e-12:
            print("[LP] zero solution – terminating")
            break
        g_norm = g_opt / s

        # pre‑compute mixed convolutions
        cfg = _conv_fft(f, g_norm)
        cgg = _conv_fft(g_norm, g_norm)

        # line search on mixing weight t∈[0,1]
        best_t = 0.0
        best_local_val = best_val
        # a modest grid; the optimum is usually at a small t
        t_grid = np.linspace(0.0, 0.6, 31)[1:]   # exclude t=0
        for t in t_grid:
            conv_t = ((1 - t) ** 2) * conv_ff + 2 * t * (1 - t) * cfg + (t ** 2) * cgg
            val_t = 2.0 * n * np.max(conv_t)
            if val_t < best_local_val:
                best_local_val = val_t
                best_t = t

        if best_t == 0.0:
            # no further improvement possible – stop early
            break

        # update the current shape
        f = (1 - best_t) * f + best_t * g_norm
        f = np.maximum(f, 0.0)
        f = f / f.sum()

        # keep global best
        if best_local_val < best_val:
            best_val = best_local_val
            best_seq = f.copy()
            print(f"[LP] n={n:5d} it={it:3d} new best={best_val:.6f}  t={best_t:.3f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine combines a lightweight LP‑guided
    global phase with deterministic polishing (Adam, sub‑gradient, hill‑climb)
    and respects the given time budget.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Lengths to explore – smaller length is cheaper, larger length gives
    # more freedom.  We start with 2000, then 4000, finally 8000 if time permits.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1. Analytic seed scan
        # --------------------------------------------------------------
        seeds = _make_seeds(n,
                            previous=global_best_seq if global_best_seq is not None else None)
        for s in seeds:
            val = 2.0 * n * np.max(_conv_fft(s, s))
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} val={val:.6f}")

        # --------------------------------------------------------------
        # 2. LP‑guided refinement (new ingredient)
        # --------------------------------------------------------------
        if time.time() > deadline:
            break

        seq_lp, val_lp = _lp_refine(global_best_seq, n, deadline,
                                    max_iters=150, top_k=30)
        if val_lp < global_best_val:
            global_best_val = val_lp
            global_best_seq = seq_lp.copy()
            print(f"[LP‑final] n={n:5d} improved global best to {val_lp:.6f}")

        # --------------------------------------------------------------
        # 3. Adam polishing (same as reference)
        # --------------------------------------------------------------
        if time.time() > deadline:
            break

        seq_adam, val_adam = _adam_optimize(global_best_seq, n, deadline,
                                            max_iters=12000)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_val = val_adam
            global_best_seq = seq_adam.copy()
            print(f"[Adam] n={n:5d} new global best={val_adam:.6f}")

        # --------------------------------------------------------------
        # 4. Exact sub‑gradient refinement
        # --------------------------------------------------------------
        if time.time() > deadline:
            break

        seq_sg, val_sg = _subgradient_refine(global_best_seq, n,
                                            deadline, max_iters=60000)
        if val_sg < global_best_val:
            global_best_val = val_sg
            global_best_seq = seq_sg.copy()
            print(f"[SubGrad] n={n:5d} new global best={val_sg:.6f}")

        # --------------------------------------------------------------
        # 5. Stochastic hill‑climbing (lightweight)
        # --------------------------------------------------------------
        if time.time() > deadline:
            break

        seq_hc, val_hc = _hill_climb(global_best_seq, n,
                                     deadline, max_iters=40000)
        if val_hc < global_best_val:
            global_best_val = val_hc
            global_best_seq = seq_hc.copy()
            print(f"[Hill] n={n:5d} new global best={val_hc:.6f}")

    # ----------------------------------------------------------------------
    # Final sanity check with the provided evaluate_sequence()
    # ----------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    # expose the result for possible re‑use in later runs
    globals()["height_sequence_1"] = global_best_seq.copy()

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
