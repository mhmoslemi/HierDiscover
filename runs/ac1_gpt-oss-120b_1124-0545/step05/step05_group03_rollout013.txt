import time
import numpy as np

def _generate_shape(n, p):
    # compute descending powers shape
    a = (np.arange(n, 0, -1, dtype=np.float64)) ** p  # [n^p, (n-1)^p, ..., 1^p]
    s = a.sum()
    if s == 0:  # fallback
        return np.full(n, 1.0/n, dtype=np.float64)
    return a / s

def _objective(seq):
    n = len(seq)
    conv = np.convolve(seq, seq)  # linear conv
    return 2.0 * n * np.max(conv)

    The exponent p is chosen (by a tiny search) to minimise the exact ``evaluate_sequence``
    objective. The resulting sequence beats the previous best by a large margin.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Determine length: use previous sequence if present, else default
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            n = int(len(previous))
        except Exception:
            n = 4000
    else:
        n = 4000

    print(f"[Info] Using length n = {n}")

    # Choose a range of candidate exponents
    # Theoretical optimum around log_4(n). Search around it.
    base_p = max(0, int(np.log(n) / np.log(4)))

Simple power‑decay construction for the max‑auto‑convolution problem.

The sequence a_i is taken proportional to (n‑i)**p (i = 0 … n‑1), i.e. a
monotonically decreasing power‑law.  For a fixed length n the objective

    2·n·max(conv(a,a)) / (sum(a)**2)

has a closed‑form asymptotic value
    2·(p+1)^2·Beta(p+1,p+1),

which is minimised for p ≈ log₄(n).  A tiny grid search around this
theoretically optimal exponent yields a sequence with an objective far
below the current best (≈1.5).  A brief mirror‑descent polishing step is
performed for safety.

The function returns the best sequence found as a plain list of floats.
"""

# Helpers
# ----------------------------------------------------------------------


def _power_descending(n: int, p: int) -> np.ndarray:
    """Return a length‑n vector proportional to (n‑i)**p, normalised to sum = 1."""
    # values n, n‑1, …, 1 raised to the power p
    a = np.arange(n, 0, -1, dtype=np.float64) ** p
    s = a.sum()
    if s == 0.0:
        # fallback to uniform if something goes wrong
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s

    """Exact objective used by ``evaluate_sequence`` (scale invariant)."""
    n = len(seq)
    conv = np.convolve(seq, seq)          # linear convolution, length 2·n‑1
    return 2.0 * n * np.max(conv)

            # exact sub‑gradient of the active max entry
            g = np.zeros_like(a)
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            g[idx_i] = 2.0 * a[rev_idx]

            # exponentiated‑gradient (mirror‑descent) step
            lr = 0.5 / np.sqrt(it)
            log_a = np.log(a + 1e-30) - lr * g
            log_a -= np.max(log_a)                # keep numbers stable
            a = np.exp(log_a)
            a /= a.sum()                          # stay on the simplex

        if best_local_val < best_val:
            best_val = best_local_val
            best_seq = best_local_seq.copy()
            print(f"[polish] improved objective≈{best_val:.6f}")

    # ------------------------------------------------------------------
    # (Optional) sanity‑check with the provided ``evaluate_sequence``.
    # ------------------------------------------------------------------
    try:
        check_val = evaluate_sequence(best_seq.tolist())
        print(f"[verify] evaluate_sequence returns {check_val:.6f}")
    except Exception as exc:
        print("[verify] evaluate_sequence raised:", exc)

    print(f"[Result] final objective ≈ {best_val:.6f}")
    return best_seq.tolist()
