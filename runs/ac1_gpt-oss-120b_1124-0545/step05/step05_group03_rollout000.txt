def _project_to_simplex(v):
    # Input v as np.ndarray, may contain negatives.
    # Output w with sum=1, w>=0.
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‐function (non‑negative sequence) that minimizes
the evaluation function ``evaluate_sequence``.

Algorithm outline
-----------------
1. **Initial seeds** – a diverse set of analytic shapes (uniform,
   sinusoidal, geometric, power‑law, beta‑like, Dirichlet samples,
   random soft‑max vectors).  The best seed becomes the starting point.
2. **Global search** – Differential Evolution on a piece‑wise‑linear
   representation (≈√n knots).  Uses a current‑to‑pbest mutation
   (more robust than classic DE) and adapts the number of generations
   to the remaining budget.
3. **Local polishing** – several deterministic refinements are applied
   repeatedly until time runs out:
      * Adam on a smooth max (log‑sum‑exp) surrogate.
      * Exact mirror‑descent on the true max‑convolution.
      * Stochastic hill‑climbing (pair‑wise mass transfers).
      * Simulated annealing with occasional uphill moves.
4. **Verification** – the final sequence is passed through the provided
   ``evaluate_sequence`` to ensure it is accepted.

All stages respect a global time budget (default 1000 s).  The function
returns the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if len(rho) == 0:
        # all entries will be zero; return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray, n: int) -> float:
    """Exact evaluation of 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a modest set of diverse normalised start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    for a in (0.3, 0.5, 0.7):
        for b in (0.3, 0.5, 0.7):
            seeds.append(_beta_shape(n, a, b))

    rng = np.random.default_rng(2023)
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000,
                   lr0: float = 0.5,
                   beta_start: float = 5.0,
                   beta_mul: float = 1.02) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑softmax parametrisation, using a smooth‑max surrogate."""
    v = np.log(seed + 1e-30)          # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = beta_start

    best_seq = None
    best_val = np.inf
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights (log‑sum‑exp approximation)
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** t)
        v2_hat = v2 / (1.0 - 0.999 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 120000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:5d} best={cur_val:.6f}")

        # Sub‑gradient of the active max entry (exact)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Exponentiated‑gradient (mirror descent) step
        lr = 0.8 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        a = _project_to_simplex(log_a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """Random pair‑wise mass transfer, keeping improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.05  # initial relative step size

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] it={it:5d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing – occasional uphill moves
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 120000,
                         T_start: float = 0.05,
                         T_end: float = 1e-6) -> tuple[np.ndarray, float]:
    """Very simple annealing with random mass transfers."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear temperature schedule
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        # allow up‑to‑10 % of donor mass to move (larger moves early)
        max_transfer = min(cur[donor], 0.1 * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)

        accept = False
        if val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            if np.exp(-(val - cur_val) / max(T, 1e-12)) > rng.random():
                accept = True

        if accept:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[SA] it={it:5d} best={val:.6f}")

        if it % 3000 == 0:
            # gradually shrink allowed transfer size
            pass  # (kept constant for simplicity)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Low‑dimensional Differential Evolution (global search)
# ----------------------------------------------------------------------


def _eval_param(p: np.ndarray, knots: np.ndarray, n: int) -> tuple[float, np.ndarray]:
    """
    Interpolate a low‑dimensional simplex vector `p` (sum=1)
    to a full length‑`n` sequence, normalise, and evaluate the exact objective.
    Returns (objective, full_sequence).
    """
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    if s == 0.0:
        a = _uniform(n)
        s = a.sum()
    a = a / s
    val = _objective(a, n)
    return val, a


def _de_optimize(n: int,
                 deadline: float,
                 popsize: int = 30,
                 K: int | None = None,
                 max_gens: int = 500) -> tuple[np.ndarray, float]:
    """
    Differential Evolution using a current‑to‑pbest mutation.
    The search space is a piece‑wise‑linear representation with `K` knots.
    """
    rng = np.random.default_rng()
    if K is None:
        K = max(20, int(np.sqrt(n)))      # default ≈ √n
    knots = np.linspace(0, n - 1, K)

    # initialise on the simplex with Dirichlet draws
    pop = [rng.dirichlet(np.ones(K)) for _ in range(popsize)]
    pop_vals = np.empty(popsize, dtype=np.float64)
    pop_seq = [None] * popsize

    for i in range(popsize):
        val, seq = _eval_param(pop[i], knots, n)
        pop_vals[i] = val
        pop_seq[i] = seq

    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_seq = pop_seq[best_idx].copy()
    best_p = pop[best_idx].copy()

    F = 0.8
    CR = 0.9
    p_best_frac = 0.2          # fraction of the population considered as pbest

    gen = 0
    while time.time() < deadline and gen < max_gens:
        gen += 1

        # order population for pbest selection
        order = np.argsort(pop_vals)
        pbest_size = max(2, int(p_best_frac * popsize))

        for i in range(popsize):
            # select pbest from the top pbest_size individuals
            pbest_idx = order[rng.integers(pbest_size)]
            pbest = pop[pbest_idx]

            # select two random distinct vectors different from i
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx = rng.choice(idxs, size=2, replace=False)
            a_vec = pop[a_idx]
            b_vec = pop[b_idx]

            # current‑to‑pbest mutation
            donor = pop[i] + F * (pbest - pop[i]) + F * (a_vec - b_vec)
            donor = np.maximum(donor, 0.0)
            donor_sum = donor.sum()
            donor = donor / donor_sum if donor_sum > 0 else _uniform(K)

            # binomial crossover
            trial = pop[i].copy()
            jrand = rng.integers(K)
            for j in range(K):
                if rng.random() < CR or j == jrand:
                    trial[j] = donor[j]

            # ensure feasibility
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            trial = trial / ts if ts > 0 else _uniform(K)

            val_trial, seq_trial = _eval_param(trial, knots, n)

            if val_trial < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val_trial
                pop_seq[i] = seq_trial
                if val_trial < best_val:
                    best_val = val_trial
                    best_seq = seq_trial.copy()
                    best_p = trial.copy()

        if gen % 20 == 0:
            print(f"[DE] n={n:5d} gen={gen:3d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine combines a global DE stage with several deterministic
    refinements, all respecting the allocated time budget.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a small safety margin

    # ------------------------------------------------------------------
    # Choose problem size (4000 is a sweet spot for the budget)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load any previous best from the global namespace (if present)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Initial diverse seeds
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous)
    best_seq = None
    best_val = np.inf

    print("\n=== Evaluating analytic seeds ===")
    for s in seeds:
        val = _objective(s, n)
        if val < best_val:
            best_val = val
            best_seq = s.copy()
            print(f"[seed] new best={val:.6f}")

    # ------------------------------------------------------------------
    # Global Differential Evolution
    # ------------------------------------------------------------------
    print("\n=== Global DE ===")
    seq_de, val_de = _de_optimize(n, deadline, popsize=30, K=None, max_gens=500)
    if val_de < best_val:
        best_val = val_de
        best_seq = seq_de.copy()
        print(f"[DE] new global best={val_de:.6f}")

    # ------------------------------------------------------------------
    # Local polishing loop – repeat refinements while time permits
    # ------------------------------------------------------------------
    # We will run several cycles of (Adam → Sub‑grad → Hill → SA)
    # Each cycle respects the remaining budget.
    cycle = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:    # not enough time for another full cycle
            break
        cycle += 1
        print(f"\n=== Local refinement cycle {cycle} (remaining {remaining:.1f}s) ===")

        # 1) Adam (smooth max)
        seq_adam, val_adam = _adam_optimize(best_seq, n, deadline,
                                            max_iters=20000,
                                            lr0=0.6, beta_start=5.0, beta_mul=1.02)
        if seq_adam is not None and val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam.copy()
            print(f"[Adam] new best={val_adam:.6f}")

        # 2) Exact sub‑gradient mirror descent
        seq_sg, val_sg = _subgradient_refine(best_seq, n, deadline,
                                             max_iters=120000)
        if val_sg < best_val:
            best_val = val_sg
            best_seq = seq_sg.copy()
            print(f"[SubGrad] new best={val_sg:.6f}")

        # 3) Stochastic hill‑climb
        seq_hc, val_hc = _hill_climb(best_seq, n, deadline,
                                     max_iters=80000)
        if val_hc < best_val:
            best_val = val_hc
            best_seq = seq_hc.copy()
            print(f"[Hill] new best={val_hc:.6f}")

        # 4) Simulated annealing (allows escaping shallow minima)
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline,
                                              max_iters=120000,
                                              T_start=0.05, T_end=1e-6)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()
            print(f"[SA] new best={val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final verification (optional)
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(best_seq.tolist())
    except Exception:
        pass

    print("\n=== Finished ===")
    print(f"Best objective ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
