    a = best_seq.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    last_improve = time.time()
    stagnation_time = 15.0

            best_seq = a.copy()
            last_improve = time.time()

        shift = max_idx - np.arange(n)

"""
Propose a non‑negative step function that minimizes
    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm (≈1000 s budget)
--------------------------------
1.  Analytic families (product, sum, exponential, cosine‑power, beta)
    are exhaustively sampled; the 10 best (on the simplex) are kept.
2.  Convex mixtures of those top candidates are examined.
3.  A soft‑max (log‑sum‑exp) surrogate of the max‑convolution is
    minimised with Adam.  The gradient is the **exact** gradient of the
    surrogate (2·weighted‑shift), unlike the previous version that used
    an incorrect gradient.  The temperature β is annealed from a smooth
    value (≈0.5) to a very sharp value (≈400) so the optimisation first
    explores globally and then focusses on the true maximum.
4.  Simple Nesterov sub‑gradient descent is run before the Adam phase
    to give a solid starting point.
5.  Stagnation is handled by jitter‑restarts and occasional random mass‑
    transfer moves; the current best vector is always kept.
6.  In the last few seconds a fine‑grained Gaussian perturbation is
    applied to the best solution found.

All updates are projected onto the probability simplex (∑a_i=1, a_i≥0)
so the objective is scale‑invariant.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of a and b using FFT."""
    size = _next_pow2(a.shape[0] + b.shape[0] - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (linear) using FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Objective helpers (working on the simplex)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective (scale invariant) for a on the simplex."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_self(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    tot = a.sum()
    if tot == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / tot
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    tot = a.sum()
    if tot == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / tot
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    return a / a.sum()


def analytic_cosine(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    return a / a.sum()


def analytic_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta_
    return a / a.sum()

# Candidate handling utilities

def _add_candidate(
    cand: np.ndarray,
    best_list: list[tuple[float, np.ndarray]],
    max_len: int = 10,
) -> None:
    """Insert a candidate into a bounded best‑list (sorted by score)."""
    sc = compute_score(cand)
    inserted = False
    for idx, (s, _) in enumerate(best_list):
        if sc < s:
            best_list.insert(idx, (sc, cand.copy()))
            inserted = True
            break
    if not inserted and len(best_list) < max_len:
        best_list.append((sc, cand.copy()))
    while len(best_list) > max_len:
        best_list.pop()


def analytic_grid_search(n: int) -> tuple[np.ndarray, list[tuple[float, np.ndarray]]]:
    """Exhaustively scan parametric families; return overall best and top‑10."""
    top: list[tuple[float, np.ndarray]] = []

    # product family
    p_vals = np.arange(0.1, 3.01, 0.15)
    off_vals = np.concatenate(
        (np.geomspace(1e-3, 0.01, 5), np.arange(0.01, 5.01, 0.3))
    )
    for p in p_vals:
        for off in off_vals:
            _add_candidate(analytic_product(n, p, off), top)

    # sum family
    for p in p_vals:
        for off in off_vals:
            _add_candidate(analytic_sum(n, p, off), top)

    # exponential tails
    lam_vals = np.logspace(-3, 1, 20)
    for lam in lam_vals:
        _add_candidate(analytic_exponential(n, lam), top)

    # cosine‑power
    p_cos = np.arange(0.5, 8.1, 0.5)
    for p in p_cos:
        _add_candidate(analytic_cosine(n, p), top)

    # beta family
    alpha_vals = np.arange(-0.5, 2.51, 0.5)
    beta_vals = np.arange(-0.5, 2.51, 0.5)
    for a in alpha_vals:
        for b in beta_vals:
            _add_candidate(analytic_beta(n, a, b), top)

    # global reference sequence (if present)
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            _add_candidate(ref, top)

    best_seq = min(top, key=lambda x: x[0])[1].copy()
    return best_seq, top


def mix_top_candidates(
    top: list[tuple[float, np.ndarray]],
    n: int,
) -> np.ndarray:
    """Convexly blend the best few candidates and return the best mixture."""
    best_seq = top[0][1].copy()
    best_score = top[0][0]
    alphas = np.linspace(0.0, 1.0, 21)  # 0,0.05,…,1
    for i in range(len(top)):
        for j in range(i + 1, len(top)):
            a_i = top[i][1]
            a_j = top[j][1]
            for alpha in alphas:
                cand = alpha * a_i + (1.0 - alpha) * a_j
                cand = project_simplex(cand)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
    return best_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the evaluation function.  The routine respects a wall‑time budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))
    print("[0] Analytic grid search, n =", n)
    best_seq, top10 = analytic_grid_search(n)
    best_seq = mix_top_candidates(top10, n)
    best_score = compute_score(best_seq)
    print(f"[0] initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Sub‑gradient descent with Nesterov momentum (quick polishing)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    momentum = np.zeros_like(a)

    last_improve = time.time()
    stagnation_time = 15.0
    step_start, step_end = 0.7, 1e-5

    while time.time() < deadline - 300:   # reserve ~5 min for later phases
        conv = conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[sg] new best = {best_score:.6f}")

        # gradient from the active maximal entry
        max_idx = int(conv.argmax())
        shift = max_idx - np.arange(n)

        g[mask] = 2.0 * a[shift[mask]]

        progress = (time.time() - start_time) / (budget_s - 310.0)
        step = step_start * (1.0 - progress) + step_end * progress

        a_candidate = a - step * momentum
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate = project_simplex(a_candidate)

            if rs < best_score:

            a_candidate = best_seq + jitter
            a_candidate = project_simplex(a_candidate)
            last_improve = time.time()
            print("[sg] jitter restart")

    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_adam = 0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # ------------------------------------------------------------------

            best_score = cand_score
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, n = {n}, time = {total:.1f}s")
    return best_seq.tolist()
