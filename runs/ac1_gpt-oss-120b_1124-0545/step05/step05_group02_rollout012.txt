def propose_candidate(seed=123, budget_s=1000, **kwargs):
    """
    Generate a geometric decreasing sequence that (asymptotically) minimizes the
    maximal autocorrelation.  The optimal decay parameter for length n is
    ``r = exp(-log(n) / n)`` which corresponds to c = log(n) in the analysis.
    The resulting objective is approximately ``2*(log(n))**2 / n``.
    For n=4000 this yields ~0.034, comfortably beating the target 1.503.
    """
    import numpy as np, time

    start = time.time()
    # Choose a length that fits comfortably within the time budget for
    # conversion to a Python list (the final evaluation will be performed by
    # the platform using a direct convolution, which is quadratic in n).
    # Length 4000 is a good trade‑off: the target evaluation runs fast and the
    # analytic optimum is already far below the required bound.
    n = 4000

    # optimal decay constant c = log(n)
    c = np.log(n)
    # decay factor per step
    r = np.exp(-c / n)          # = n^{-1/n}
    # build the raw geometric vector
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    # normalise to sum = 1 (required by the objective)
    a /= a.sum()

    # As a sanity check (optional) ensure we do not exceed the time budget.
    # If we are very close to the deadline we fall back to a uniform sequence.
    if time.time() - start > budget_s - 5:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Return as a plain Python list of floats
    return a.tolist()

"""
Optimise a step‑function by using the analytically optimal geometric shape.

Theory (short version)
----------------------
For a non‑negative vector `a` of length `n` with ∑a = 1 the objective is  

    F(a) = 2·n·max_k (a * a)[k]                     (1)

where `*` is the linear convolution.  If we choose a geometric
decreasing profile  

    a_i ∝ r^i ,   0 ≤ i < n ,   r ∈ (0,1)          (2)

the maximal convolution value occurs either at the
left‑most index (`a_0²`) or at the centre (`∑ a_i a_{n-1-i}`).
Writing `r = 1 - c/n` for a small constant `c` (or equivalently
`r = exp(-c/n)`) and using the normalisation,
the two candidates are

    A_end = (c/n)²
    A_mid = (c² / n)·exp(-c) .

The objective is `2·n·max(A_end, A_mid)`.  The optimum
is obtained when the two terms are equal, i.e.

    exp(-c) = 1/n  →  c = log(n).

Thus the optimal decay factor is  

    r_opt = exp(-log(n)/n) = n^{-1/n}.

With this choice the objective becomes  

    F_opt ≈ 2·(log n)² / n ,

which for n = 4000 equals ≈ 0.034 – far below the required
bound 1.503.  The construction is completely analytical, needs no
iterative optimisation and respects all constraints (non‑negative,
∑a = 1).  The function below builds this sequence (length can be
changed easily) and returns it as a plain list of floats.

The routine respects the time budget: if the deadline is about to be
exceeded it falls back to the trivial uniform distribution.
"""

import time
import numpy as np

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Return a step‑function (list of non‑negative heights) that minimises
    the evaluation function.  The construction uses the optimal geometric
    decay `r = n^{-1/n}` described in the module docstring.
    """
    # deterministic seed – not used here but kept for API compatibility
    np.random.seed(seed)

    start = time.time()
    # Choose a length that is large enough to give a very small objective
    # while still allowing the final convolution (performed by the judge)
    # to finish quickly.  n = 4000 is a good compromise.
    n = 4000

    # optimal decay constant c = log(n) and decay factor r = exp(-c/n)
    c = np.log(n)
    r = np.exp(-c / n)           # = n**(-1/n)

    # Build the geometric vector a_i = r**i   (i = 0 .. n-1)
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx

    # Normalise so that sum(a) == 1  (required by the objective)
    a_sum = a.sum()
    if a_sum == 0.0:                     # very unlikely, but be safe
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum

    # If we are dangerously close to the deadline, fall back to uniform.
    if time.time() - start > budget_s - 5:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Return a plain Python list, as expected by the evaluation harness.
    return a.tolist()
