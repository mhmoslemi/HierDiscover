1. Initialise a non‑negative probability vector a of length n (default 2000).
   If a previous best vector is available in the global variable
   ``height_sequence_1``, it is used, otherwise a uniform distribution.
   Tiny epsilon is added to keep all entries strictly positive.

2. Linear‑programming “mass‑inflation’’ loop.
   For the current a we compute M = max(conv(a,a)).
   We solve the LP
       max   1ᵀ b
       s.t.  conv(a,b) ≤ M,   b ≥ 0 .
   The linear map conv(a,·) is represented by a sparse Toeplitz matrix.
   If the LP yields a total mass Σb larger than 1 we normalise b to sum 1
   and replace a by a convex combination a ← (1−t)a + t·b̂ .
   The loop stops when the mass cannot be increased or the time budget is
   exhausted.  This step strictly reduces the objective whenever it makes
   progress.

3. Sub‑gradient refinement on the true non‑smooth objective.
   The vector a lives on the simplex; we apply an exponentiated‑gradient
   (mirror‑descent) update using the sub‑gradient of the max‑convolution.
   The step‑size follows 1/√t and we run as many iterations as time permits.

4. Simple Metropolis‑Hastings annealing as a final polish.

The best sequence encountered during the whole run is returned as a plain
list of floats; it is also cached in the global variable `height_sequence_1`
for possible reuse by subsequent runs.
"""

import time, math, numpy as np

# optional SciPy imports
try:
    import scipy.sparse as sp
    import scipy.optimize as opt
    SCIPY_OK = True
except Exception:  # pragma: no cover
    SCIPY_OK = False

def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    n = a.size + b.size - 1
    N = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(A * B, n=N)[:n]
    return conv

def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(a, a)
    return 2.0 * a.size * conv.max() / (s * s)

def _build_conv_matrix(a: np.ndarray, row_idx: np.ndarray, col_idx: np.ndarray) -> sp.csr_matrix:
    """Return sparse matrix representing the linear map b ↦ conv(a,b)."""
    # data = a[i] for each (i,j) pair
    data = np.repeat(a, a.size)
    return sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * a.size - 1, a.size))

def _lp_improve(a: np.ndarray, deadline: float, max_iters: int = 6,
                mix: float = 0.8, eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Perform up to max_iters of the “mass‑inflation’’ LP update.
    Returns (best_a, best_obj) seen during the loop.
    """
    n = a.size
    # Pre‑compute static row/col index arrays (length n²)
    rows = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)).ravel()
    cols = np.tile(np.arange(n, dtype=np.int32), n)
    best_a = a.copy()
    best_obj = _objective(a)

    for it in range(max_iters):
        if time.time() > deadline:
            break
        M = _conv_fft(a, a).max()
        # Build sparse matrix for the linear map
        A = _build_conv_matrix(a, rows, cols)
        # Linear program: maximize sum(b)  ⇔ minimize -(1ᵀ b)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        ub = np.full(2 * n - 1, M, dtype=np.float64)

        # solve
        res = opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            # If the LP fails we abort the LP phase
            print(f"[LP] iteration {it+1} failed: {res.message}")
            break
        b = res.x
        mass = b.sum()
        # No progress?
        if mass <= 1.0 + 1e-9:
            print(f"[LP] iteration {it+1} no mass increase (mass={mass:.6f})")
            break
        # Updated candidate (convex combination)
        b_norm = b / mass
        a_new = (1 - mix) * a + mix * b_norm
        a_new = np.clip(a_new, eps, None)
        a_new /= a_new.sum()
        a = a_new
        cur_obj = _objective(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[LP] iter {it+1} mass={mass:.6f} obj={cur_obj:.6f}")
        else:
            # continue anyway – sometimes objective may fluctuate
            pass
    return best_a, best_obj

def _subgradient_refine(a0: np.ndarray, deadline: float,
                        max_iters: int = 200000, eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact objective.
    Uses the true sub‑gradient of the max‑convolution.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over active shifts)
        g = np.zeros_like(a)
        # For each active shift k we add 2*a shifted by k
        # i.e. g[i] += 2*a[k-i] for all i where 0≤k-i<n
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        if active.size > 0:
            g /= float(active.size)

        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # keep numbers stable
        a = np.exp(log_a)
        a /= a.sum()
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[SubGrad] it={it:6d} val={cur_val:.6f}")
    return best_a, best_val

def _anneal_optimize(a0: np.ndarray, deadline: float,
                    max_iters: int | None = None, eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk that transfers a small amount of mass
    between random indices and occasionally performs a global Gaussian jump.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    cur_val = _objective(a)
    best = a.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0
    n = a.size

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * a[i]   # up to 2 % of source mass
        if delta > a[i]:
            delta = a[i]

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_val = _objective(new_a)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_a.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        if it % 1000 == 0:
            # occasional Gaussian perturbation
            noise = rng.normal(scale=0.001, size=n)
            new_a = a + noise
            new_a = np.clip(new_a, 0.0, None)
            if new_a.sum() == 0.0:
                new_a = np.full(n, 1.0 / n)
            else:
                new_a /= new_a.sum()
            new_val = _objective(new_a)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                a = new_a
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_a.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")
    return best, best_val

"""
Algorithm
---------

1. **Initial vector** – start from the previously best known construction
   (global variable ``height_sequence_1``) if it exists, otherwise from a
   uniform distribution.  A tiny epsilon (`1e-12`) is added to keep every
   entry strictly positive and the vector is normalised to sum‑to‑1.

2. **Linear‑programming “mass‑inflation’’ loop** –  
   For the current vector `a` compute `M = max(conv(a,a))`.  
   Solve the linear program  

        maximise   1ᵀ b
        subject to  conv(a,b) ≤ M ,   b ≥ 0

   (`conv(a,b)` is linear in `b`).  If the optimal total mass
   `Σb` is larger than 1 we normalise `b` and replace  

        a ← (1−t) a + t (b/Σb)      (t≈0.8)

   This step strictly reduces the objective whenever it makes progress.
   The convolution map is built once per iteration as a sparse Toeplitz
   matrix; the LP is solved with SciPy’s HiGHS interior‑point solver.
   At most a few iterations (default 6) are performed – they are cheap
   but often give a noticeable drop.

3. **Exact sub‑gradient mirror‑descent** –  
   The true non‑smooth objective  

        F(a) = 2·n·max(conv(a,a)) / (∑a)²

   is minimised on the simplex using exponentiated‑gradient updates.
   The sub‑gradient of the max‑convolution is computed by averaging the
   gradients of all active (tight) convolution constraints.

    # Phase 2 – Exact sub‑gradient mirror‑descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:                       # need at least a few seconds
        sub_deadline = time.time() + remaining * 0.55
        a_sub, val_sub = _subgradient_refine(a, sub_deadline,
                                             max_iters=200_000,
                                             eps=eps)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            a = a_sub
            print(f"[phase‑SubGrad] best after sub‑grad = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Annealing polish (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ann, val_ann = _anneal_optimize(a, deadline,
                                          max_iters=None,
                                          eps=eps)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[phase‑Anneal] final best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never trigger)
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(n, 1.0 / n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    # Store the result for possible reuse by a later run
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
