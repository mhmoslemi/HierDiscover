from scipy.optimize import linprog

def lp_max_sum_under_conv(a, M):
    n = len(a)
    # Build constraint matrix A (shape (2n-1, n))
    # Use sparse.
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    data = []
    # For each row k from 0 to 2n-2:
    for k in range(2*n-1):
        # i range: max(0, k-(n-1)) <= i <= min(n-1, k)
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # For each i, a_i contributes to column (k-i)
        # Wait column index j = k-i, with i fixed. j in [0,n-1]
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    
    # Solve maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)  # objective: min c^T b = -sum(b)
    bounds = [(0, None)] * n
    # constraints: A @ b <= M
    # Use linprog with inequality A_ub * b <= b_ub
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        # fallback: return current a (or uniform)
        return a
    b_opt = res.x
    return b_opt
```

"""
Propose a non‑negative step function that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. Initialise the sequence from the previous best solution (global
   variable ``height_sequence_1``) or from a uniform distribution.
   The vector is scaled to the convenient sum √(2·n) – the objective
   is scale‑invariant, this normalisation makes the following linear
   program well‑conditioned.

2. LP‑based improvement (the “mirror‑descent” idea from the paper):
   * Let M = max (f * f) for the current vector f.
   * Solve the linear program
         maximise   1ᵀ b
         subject to f * b ≤ M  (elementwise)
                    b ≥ 0
     (convolution f * b is linear in b, therefore the constraints are
      linear).  The optimal b has a larger ℓ₁‑norm than f.
   * Rescale b to the target sum ⇒ g = (√(2·n)/∑b)·b.  Then f * g ≤ M.
   * Form a convex combination h = (1‑t) f + t g (t≈0.3, slowly reduced).
     Because both f and g respect the same convolution bound, the new
     max‑convolution never exceeds M, so the objective is monotone.
   * Repeat until the time budget for this phase expires.

3. Sub‑gradient descent with Nesterov momentum (same as the reference
   implementation) to polish the solution further.

4. Final random‑perturbation refinement for the last few seconds.

All steps respect the wall‑time budget (default 1000 s).  If SciPy is
unavailable the LP phase is skipped and only the sub‑gradient part is
run.  The returned list is normalised to sum = 1 (any scale works for the
evaluation function).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, fast for n≈2000)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (real, zero‑padded)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Scoring (scale‑invariant)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence``.  The vector is
    normalised to ℓ₁‑norm = 1 before the convolution.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = conv_self(a_norm)
    return float(2.0 * a.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Linear‑program helper (requires SciPy)
# ----------------------------------------------------------------------

    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    SCIPY_AVAILABLE = False


def _build_conv_matrix(a: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1) × n matrix  A  such that  (A @ b)_k = (a * b)_k .
    (a is fixed, b is the variable.)
    """
    n = a.shape[0]
    # rows = i + j  (i = index of a, j = index of b)
    rows = (np.arange(n)[:, None] + np.arange(n)).ravel()
    cols = np.tile(np.arange(n), n)
    data = np.repeat(a, n)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def lp_optimize(a: np.ndarray, M: float, sum_target: float) -> np.ndarray:
    """
    Solve  max 1ᵀb   s.t. a * b ≤ M, b ≥ 0.
    Return the scaled feasible vector g with sum = ``sum_target``.
    If the LP fails, return ``a`` (no change).
    """
    n = a.shape[0]
    A = _build_conv_matrix(a)
    c = -np.ones(n)                     # maximise
    bounds = [(0, None)] * n
    b_ub = np.full(2 * n - 1, M)

    # Solve with HiGHS (fast for medium‑size dense problems)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # fallback – keep the current vector
        return a.copy()

    b = res.x
    sum_b = b.sum()
    if sum_b <= 1e-12:
        return a.copy()

    scale = sum_target / sum_b          # ≤ 1  (satisfies a * g ≤ M)
    g = b * scale
    g = np.clip(g, 0.0, None)
    # Renormalise numerically (tiny drift may appear)
    g = g * (sum_target / g.sum())
    return g


# ----------------------------------------------------------------------
# Simple random mass‑transfer move (used in the sub‑gradient phase)
# ----------------------------------------------------------------------


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:

    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return a_new

            # convex blend; keep the sum fixed to sum_target
            a_new = (1.0 - t) * a + t * g
            a_new = a_new * (sum_target / a_new.sum())

            cur_score = compute_score(a_new)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a_new.copy()
                print(f"[LP {it_lp:04d}] improved score = {best_score:.6f}")
                # gradually shrink the step size after a success
                t = max(0.05, t * 0.9)
            else:
                # if no improvement, gently increase t to explore more
                t = min(0.5, t * 1.05)

            a = a_new
            it_lp += 1
    else:
        print("[info] SciPy not available – skipping LP phase.")

    # ------------------------------------------------------------------
    # 2) Sub‑gradient descent with Nesterov momentum (fine tuning)
    # ------------------------------------------------------------------
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    step_start, step_end = 0.5, 1e-5
    last_improve = time.time()
    it_sg = 0
    while time.time() < deadline - 30:
        conv = conv_self(a)
        cur_score = compute_score(a)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[SG {it_sg:06d}] new best = {best_score:.6f}")

        # sub‑gradient of the active max entry
        max_idx = int(conv.argmax())
        shift = max_idx - np.arange(n)
        mask = (shift >= 0) & (shift < n)
        g_sub = np.zeros_like(a)
        g_sub[mask] = 2.0 * a[shift[mask]]

        # Nesterov‑style step with a simple time‑decay schedule
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s - 30.0)
        step = step_start * (1.0 - frac) + step_end * frac

        momentum = beta_mom * momentum + (1.0 - beta_mom) * g_sub
        a_candidate = a - step * momentum
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate = a_candidate * (sum_target / a_candidate.sum())

        # occasional random mass‑transfer move
        if np.random.rand() < 0.02:
            a_candidate = random_move(a_candidate, step * 0.5)
            a_candidate = a_candidate * (sum_target / a_candidate.sum())

        # jitter restart on stagnation
        if time.time() - last_improve > 15.0:
            jitter = np.random.rand(n) * 1e-3
            a_candidate = best_a + jitter
            a_candidate = a_candidate * (sum_target / a_candidate.sum())
            last_improve = time.time()
            print(f"[SG {it_sg:06d}] jitter restart")

        a = a_candidate
        it_sg += 1

    # ------------------------------------------------------------------
    # 3) Final random perturbation phase (last few seconds)
    # ------------------------------------------------------------------
    print("[refine] starting final random perturbation")
    while time.time() < deadline:
        perturb = np.random.normal(scale=5e-4, size=n)
        cand = best_a + perturb
        cand = np.clip(cand, 0.0, None)
        cand = cand * (sum_target / cand.sum())
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, length = {n}, time = {total:.1f}s")

    # Return a version normalised to ℓ₁ = 1 (any scale is fine for the evaluator)
    final_seq = best_a / best_a.sum()
    return final_seq.tolist()
