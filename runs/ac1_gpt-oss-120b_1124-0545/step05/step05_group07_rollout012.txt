    n = kwargs.get('n', len(globals().get('height_sequence_1', [])) or 2000)

    best_score = np.inf
    best_p = None
    # coarse grid
    for p in np.linspace(0.5, 10.0, 96):   # step 0.1 maybe
        a = np.arange(1, n+1, dtype=np.float64) ** p
        a /= a.sum()
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_p = p
    # refine around best_p
    delta = 0.5
    while delta > 1e-4 and time.time() < deadline:
        improved = False
        for dp in (-delta, delta):
            p = best_p + dp
            if p <= 0:
                continue
            a = np.arange(1, n+1, dtype=np.float64) ** p
            a /= a.sum()
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_p = p
                improved = True
                print(f"[refine] p={p:.6f} score={score:.6f}")
                break
        if not improved:
            delta *= 0.5
    print(f"Best p={best_p:.6f} score={best_score:.6f}")
    best_seq = np.arange(1, n+1) ** best_p
    best_seq /= best_seq.sum()
    return best_seq.tolist()

"""
Propose a non‑negative step function that (approximately) minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The objective is scale‑invariant, so we fix Σa = 1 and minimise
the maximum entry of the (linear) self‑convolution.
A simple monotone power law a_i ∝ (i+1)^p gives an analytically
tractable family: for p ≈ 5 the maximum convolution is
≈ (p+1)² / n², yielding a score ~ 0.04 for n≈2000 – far better
than the previous 1.5‑level solutions.

The algorithm:
1. Choose a length n (default 2000 or the length of a previously
   supplied `height_sequence_1`).
2. Perform a coarse grid search over the exponent p.
3. Refine the best p with a shrinking‑step local search.
4. Return the normalised power‑law vector.

All heavy work is O(n log n) via FFT, and the whole search finishes
well within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (zero‑padded) using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a.
    The vector is assumed to be normalised (Σa = 1); the function
    renormalises just in case.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    return float(2.0 * a.shape[0] * conv.max())   # denominator = 1²


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative floats that (approximately) minimises the
    evaluation function.  The search is confined to power‑law families
    a_i ∝ (i+1)^p and refines the exponent p.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Determine the length of the vector
    # --------------------------------------------------------------
    if "n" in kwargs:
        n = int(kwargs["n"])
    elif "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        n = 2000

    # --------------------------------------------------------------
    # Coarse grid over exponent p
    # --------------------------------------------------------------
    best_score = np.inf
    best_p = None

    # grid from 0.5 to 10.0 (step ≈ 0.1)
    p_grid = np.linspace(0.5, 10.0, 96)

    for p in p_grid:
        i = np.arange(1, n + 1, dtype=np.float64)
        a = i ** p
        a /= a.sum()
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_p = p

    print(f"[grid] best exponent after coarse search: p={best_p:.6f}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # Local refinement of the exponent (simple coordinate descent)
    # --------------------------------------------------------------
    delta = 0.5
    while delta > 1e-5 and time.time() < deadline:
        improved = False
        for direction in (-1, 1):
            p = best_p + direction * delta
            if p <= 0:
                continue
            i = np.arange(1, n + 1, dtype=np.float64)
            a = i ** p
            a /= a.sum()
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_p = p
                improved = True
                print(f"[refine] p={p:.6f} score={score:.6f} (delta={delta:.5f})")
                break  # restart the inner loop with the new best_p
        if not improved:
            delta *= 0.5     # shrink step size

    print(f"[final] chosen exponent p={best_p:.6f}, final score={best_score:.6f}")

    # --------------------------------------------------------------
    # Build the final sequence (normalised to sum‑1)
    # --------------------------------------------------------------
    final = (np.arange(1, n + 1, dtype=np.float64) ** best_p)
    final /= final.sum()

    # Safety check: ensure we stay within the evaluation limits
    final = np.clip(final, 0.0, 1000.0)
    total = time.time() - start_time
    print(f"[done] length={n}, time={total:.1f}s, best score={best_score:.6f}")

    return final.tolist()
