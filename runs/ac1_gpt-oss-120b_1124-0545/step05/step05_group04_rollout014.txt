def subgrad_part(a):
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = conv.max()
    active = (conv >= max_val - 1e-12).astype(float)
    cnt = active.sum()

    w = active / cnt

"""
Propose a non‑negative step function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. **Extensive analytic sweep** – dense grids for the two classic families
   (product and sum of power‑laws).  The best analytic candidate becomes the
   starting point.

2. **Multiple Adam continuations** – smooth‑max approximation
   Φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k) is minimised.
   * Several random initialisations (Dirichlet draws) are tried.
   * λ is increased geometrically up to 1e5 and the learning‑rate is
     scaled as 0.1/√λ.
   * Adam updates are performed on the log‑probability vector `x`
     (a = softmax(x)) and after each λ the candidate is projected back
     onto the simplex.

3. **Accelerated Adam on the exact sub‑gradient** – after the smoothing
   phase the true objective max(conv) is tackled directly.
   * The sub‑gradient of max(conv) is obtained via FFT.
   * Adam’s adaptive steps are applied to the probability vector itself,
     with Euclidean projection onto the simplex after every update.

4. **Nesterov‑accelerated sub‑gradient descent** – a final phase of
   classical sub‑gradient steps with Nesterov momentum.  The step size
   follows 0.5/√t and the iterate is always projected onto the simplex.

5. **Light stochastic local search** – random transfer of a tiny amount of
   mass between two indices, with a simple simulated‑annealing acceptance
   rule to escape shallow local minima.

All phases respect the supplied time budget (default 1000 s); the best
sequence found before the deadline is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a (the sum is ignored,
    the function normalises internally).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^(‑p)   (edge‑heavy shape)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^(‑p) + (n‑1‑i+off)^(‑p)   (two tails)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exp_decay(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(‑c·|i‑center|)   (center‑peaked shape)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - center))
    a = a / a.sum()
    return a


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(‑0.5·((i‑center)/σ)²)   (smooth bump)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = a / a.sum()
    return a

def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    For a given log‑probability vector x, smoothing λ and length n,
    return Φ_λ(a) and its gradient w.r.t. x (a = softmax(x)).
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    conv = conv_fft_self(a)                # length 2n‑1
    max_c = np.max(conv)

    # soft‑max over convolution entries
    shift = conv - max_c                   # ≤ 0
    exp_term = np.exp(lam * shift)        # ≤ 1
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam    # Φ_λ

    w = exp_term / sum_exp                 # weights, sum to 1

    # gradient w.r.t. a
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]      # length n
    grad_a = 2.0 * slice_part[::-1]            # ∂Φ/∂a

    # map gradient to x‑space using the Jacobian of soft‑max
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)                # ∇_x Φ = J^T ∇_a Φ
    return phi, grad_x


def adam_optimize(x: np.ndarray, lam: float, n: int, steps: int,
                  lr: float, deadline: float) -> np.ndarray:
    """Adam optimiser for the smooth‑max objective on the log‑space."""
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for t in range(1, steps + 1):
        if time.time() >= deadline:
            break
        _, grad = compute_phi_and_grad(x, lam, n)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step
    return x

def subgrad(a: np.ndarray) -> np.ndarray:
    """
    Return a sub‑gradient of max(conv(a,a)) w.r.t. a.
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    active = (conv >= max_val - 1e-12).astype(np.float64)   # binary mask
    cnt = active.sum()
    if cnt == 0:
        return np.zeros_like(a)
    w = active / cnt
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]
    return grad


def adam_subgrad_optimize(a0: np.ndarray, steps: int, lr: float,
                          deadline: float) -> np.ndarray:
    """
    Adam‑style optimiser directly on the probability vector using the
    exact sub‑gradient of max(conv).  Projection onto the simplex follows
    each update.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for t in range(1, steps + 1):
        if time.time() >= deadline:
            break
        g = subgrad(a)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_simplex(a)
    return a


def nesterov_subgrad_optimize(a0: np.ndarray, max_iters: int,
                              init_step: float, deadline: float):
    """
    Nesterov‑accelerated sub‑gradient descent on max(conv).  The step size
    decays as 0.5/√t; each iterate is projected onto the simplex.
    """
    a_prev = a0.copy()
    y = a0.copy()
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        step = 0.5 / np.sqrt(t)
        g = subgrad(y)
        a_new = project_simplex(y - step * g)
        beta = (t - 1) / (t + 2)          # classic Nesterov coefficient
        y = a_new + beta * (a_new - a_prev)
        a_prev = a_new
        a0 = a_new
    return a0

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the supplied evaluation function.  The routine respects the given time
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 0) Helper to keep track of the current best
    # ------------------------------------------------------------------
    best_a = np.ones(n) / n
    best_score = compute_score(best_a)

    # If a previous construction is available, use it as a seed.
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n and ref.sum() > 0:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            if sc < best_score:
                best_score, best_a = sc, ref.copy()

    # ------------------------------------------------------------------
    # 1) Analytic families (coarse + fine sweep)
    # ------------------------------------------------------------------
    # coarse sweep
    p_vals = np.linspace(0.30, 0.70, 81)          # 0.30 … 0.70
    offset_vals = np.linspace(0.0001, 2.0, 81)    # 0.0001 … 2.0
    for mode in ("product", "sum"):
        for p in p_vals:
            for off in offset_vals:
                if time.time() >= deadline:
                    break
                cand = analytic_product(n, p, off) if mode == "product" else analytic_sum(n, p, off)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score, best_a = sc, cand.copy()

    # fine sweep around the best parameters found above
    # (extract parameters from the best analytic candidate)
    # We simply redo the grid with a finer resolution on a smaller window.
    # For simplicity we reuse the same ranges but with more points.
    p_fine = np.linspace(0.30, 0.70, 151)          # finer step
    off_fine = np.linspace(0.0001, 2.0, 151)
    for mode in ("product", "sum"):
        for p in p_fine:
            for off in off_fine:
                if time.time() >= deadline:
                    break
                cand = analytic_product(n, p, off) if mode == "product" else analytic_sum(n, p, off)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score, best_a = sc, cand.copy()

    # exponential decay (center‑peaked) – moderate grid
    c_vals = np.linspace(0.0005, 0.0200, 41)
    for c in c_vals:
        if time.time() >= deadline:
            break
        cand = analytic_exp_decay(n, c)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    # Gaussian bump family
    sigma_vals = np.linspace(100.0, 800.0, 41)
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        cand = analytic_gaussian(n, sigma)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    print(f"[grid] best score after analytic sweep = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Multiple Adam runs on the smooth‑max approximation
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 2.0, 5.0, 10.0, 20.0, 50.0,
        100.0, 200.0, 500.0, 1000.0,
        2000.0, 5000.0, 10000.0,
        20000.0, 50000.0, 80000.0, 100000.0
    ]

    n_restarts = 4                         # a few random restarts
    for restart in range(n_restarts):
        if time.time() >= deadline:
            break

        # random Dirichlet start (alpha=0.5 gives a bit of sparsity)
        a_start = np.random.dirichlet(alpha=np.full(n, 0.5))
        # blend a little with the current best to stay in a promising region
        a_start = 0.3 * best_a + 0.7 * a_start
        a_start = a_start / a_start.sum()
        x = np.log(np.maximum(a_start, 1e-12))

        for lam in lam_schedule:
            if time.time() >= deadline:
                break
            # allocate steps: more steps for larger λ
            remaining = deadline - time.time()
            # cheap estimate: ~0.0015 s per Adam iteration
            max_possible = int(remaining / 0.0015)
            steps = 2000
            if lam > 5000:
                steps = 8000
            elif lam > 2000:
                steps = 5000
            steps = min(steps, max_possible)
            if steps <= 0:
                break

            lr = 0.1 / np.sqrt(lam + 1.0)   # decrease learning‑rate with λ
            x = adam_optimize(x, lam, n, steps, lr=lr, deadline=deadline)

            a = softmax(x)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[adam restart={restart} λ={lam:.1f}] new best = {best_score:.6f} "
                      f"(t={time.time() - start_time:.1f}s)")

        # keep the result of this restart as a new seed for later phases
        best_a = best_a.copy()

    # ------------------------------------------------------------------
    # 3) Adam on the exact sub‑gradient of max(conv)
    # ------------------------------------------------------------------
    a = best_a.copy()
    while time.time() < deadline:
        # decide how many steps we can afford in this batch
        remaining = deadline - time.time()
        # conservative estimate: ~0.00035 s per Adam‑subgrad step
        max_steps = int(remaining / 0.0004)
        if max_steps <= 0:
            break
        steps = min(4000, max_steps)
        a = adam_subgrad_optimize(a, steps=steps, lr=0.15, deadline=deadline)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[adam‑subgrad] new best = {best_score:.6f} "
                  f"(t={time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # 4) Nesterov‑accelerated sub‑gradient descent (fine tuning)
    # ------------------------------------------------------------------
    a = best_a.copy()
    max_nesterov_iters = 3000000          # large cap; will stop by deadline
    a = nesterov_subgrad_optimize(a, max_iters=max_nesterov_iters,
                                  init_step=0.5, deadline=deadline)
    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[nesterov] new best = {best_score:.6f} (t={time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # 5) Light stochastic local search with simulated‑annealing acceptance
    # ------------------------------------------------------------------
    a = best_a.copy()
    scale = 0.02               # initial relative transfer amount
    temperature = 0.001        # start temperature for annealing
    decay = 0.9999             # exponential cooling per iteration
    while time.time() < deadline:
        cand = random_move(a, scale)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[random] new best = {best_score:.6f} "
                  f"(t={time.time() - start_time:.1f}s)")
            a = cand
            scale = max(0.001, scale * 0.9)
        else:
            # accept with a small probability to escape plateaus
            if np.random.rand() < np.exp(-(sc - best_score) / max(temperature, 1e-12)):
                a = cand
        temperature *= decay
        scale = min(0.05, scale * 1.001)   # slowly expand search radius if nothing improves

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    # return a plain Python list as required by the external harness
    return best_a.tolist()
