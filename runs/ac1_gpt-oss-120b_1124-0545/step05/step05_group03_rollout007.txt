def _lp_improve(a, n, M, deadline, max_iter=5, t=0.2):
    best_seq = a.copy()
    best_val = 2*n*np.max(_conv_fft(a,a))
    cur = a.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # build A_ub
        A = _build_conv_matrix(cur) # sparse
        # bound vector
        b_ub = np.full(2*n-1, M)
        # linear programming
        res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=b_ub,
                                    bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print("[LP] linprog failed")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            # no improvement
            break
        # normalize
        g = b / sum_b
        # convex combination
        cur = (1 - t) * cur + t * g
        # re-normalize sum to 1
        cur = cur / cur.sum()
        # compute new val
        val = 2*n*np.max(_conv_fft(cur,cur))
        if val < best_val:
            best_val = val
            best_seq = cur.copy()
            print(f"[LP] iter {it+1} val {val:.6f}")
        # update M maybe to new maxconv (smaller)
        M = np.max(_conv_fft(cur,cur))
    return best_seq, best_val
```

"""
A fast multi‑phase optimizer for the max‑auto‑convolution problem.

Phases
------
1.  Analytic seed sweep (uniform, sine window, geometric, power,
    beta‑shaped families with a wide range of α,β, random Dirichlet,
    random softmax).  The best seed is kept.
2.  Mirror‑descent (exact sub‑gradient) refinement on a moderate size
    problem (n≈2000).  This is cheap and drives the max‑convolution
    down quickly.
3.  Linear‑programming “exchange” step (the technique described in the
    reference paper).  For the current distribution f we solve

       maximise   1ᵀ b
       subject to  f * b ≤ M    (full convolution)
                  b ≥ 0 .

    The solution b has a larger ℓ₁‑norm while respecting the same
    max‑convolution bound M.  A convex combination f←(1‑t)f+t·b/‖b‖
    strictly reduces the objective.  A handful of these LP steps give
    a sizable boost over the pure sub‑gradient method.
4.  Final up‑sampling to the target length (4000) with linear
    interpolation, a short hill‑climb polish and a final evaluation.

All phases respect the global time budget (default 1000 s) and keep the
best sequence found.  The returned sequence is a plain list of floats
and satisfies the constraints of ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Basic convolution helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N).real[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Analytic seed families (all ℓ₁‑normed)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Collect a diverse set of normalized seed vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric (r close to 1 gives slow decay)
    for r in np.linspace(0.90, 0.999, 5):
        seeds.append(_geom(n, r))

    # power law
    for p in np.linspace(0.0, 3.0, 7):
        seeds.append(_power(n, p))

    # beta shapes – very wide grid (including extreme U‑shapes)
    alphas = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]
    betas = alphas
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random softmax (logits)
    for _ in range(8):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    return seeds


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent sub‑gradient descent on the true max‑convolution.
    Returns the best sequence encountered and its objective value.
    """
    a = np.copy(start_seq)
    # keep a strictly positive to avoid log(0) later
    eps = 1e-12
    a = np.maximum(a, eps)
    a = a / a.sum()

    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry (exact)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a) - lr * g
        log_a -= np.max(log_a)      # improve numerical stability
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming exchange step
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the (2n‑1)×n convolution matrix C_f such that
        C_f @ b  =  f * b   (full linear convolution).
    All entries of f are assumed > 0.
    """
    n = f.shape[0]
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    pos = 0
    for j in range(n):
        rows[pos:pos + n] = np.arange(j, j + n, dtype=np.int32)
        cols[pos:pos + n] = j
        data[pos:pos + n] = f
        pos += n

    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_improve(a: np.ndarray,
                n: int,
                deadline: float,
                max_iter: int = 6,
                t0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Perform up to ``max_iter`` LP exchange steps.
    Each step solves

        maximise   1ᵀ b
        subject to  C_a @ b ≤ M ,  b ≥ 0

    and updates a ← (1‑t)·a + t·b/‖b‖ .
    Returns the best sequence found and its objective.
    """
    cur = np.copy(a)
    # tiny epsilon to keep the matrix full rank
    eps = 1e-12
    best_seq = cur.copy()
    best_val = 2.0 * n * np.max(_conv_fft(cur, cur))

    for it in range(1, max_iter + 1):
        if time.time() > deadline:
            break

        M = np.max(_conv_fft(cur, cur))

        # ensure strictly positive entries for the matrix
        f = cur + eps
        f = f / f.sum()

        # build convolution matrix (sparse)
        A = _build_conv_matrix(f)

        # RHS of the inequality constraints
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # linear program: maximise sum(b)  ⇔ minimise –sum(b)
        res = scipy.optimize.linprog(
            c=-np.ones(n, dtype=np.float64),
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': max(0, deadline - time.time())}
        )

        if not res.success:
            print("[LP] linprog failed (iter=%d, status=%s)" % (it, res.message))
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-10:
            # no improvement possible
            break

        # normalise the new candidate
        g = b / sum_b

        # convex combination (step size decays)
        t = max(0.05, t0 / it)
        cur = (1.0 - t) * cur + t * g
        cur = np.maximum(cur, eps)
        cur = cur / cur.sum()

        # evaluate
        val = 2.0 * n * np.max(_conv_fft(cur, cur))
        if val < best_val:
            best_val = val
            best_seq = cur.copy()
            print(f"[LP] iter={it:2d} best={val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple hill‑climb (lightweight final polish)
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 15000) -> tuple[np.ndarray, float]:
    """Random pairwise mass transfer, keeping improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor, receiver = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] it={it:5d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val
