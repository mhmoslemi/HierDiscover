"""
Propose a step‑function (non‑negative vector) that minimises
`evaluate_sequence`.  The algorithm consists of three phases:

1. **Broad parametric sweep** – generate many candidates from
   Beta‑type families (including extreme α values and random mixtures) and
   keep the best one.

2. **Smooth projected Adam optimisation** – minimise the exact
   objective `2·n·max(conv(a,a))` on the probability simplex.
   The non‑smooth ∞‑norm is approximated by a soft‑max with a
   temperature that is slowly reduced; Adam with Euclidean simplex
   projection and an enforced symmetry (a_i = a_{n‑1‑i}) steers the
   search.

3. **Local stochastic refinement** – a simulated‑annealing style
   random mass‑transfer that fine‑tunes the solution.

The routine respects the given time‑budget (default 1000 s) and prints
intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum x = 1}. Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerically all entries are zero – fall back to uniform.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """
    Return max(conv(a, a)) using an FFT of length `fft_len`
    (linear convolution, i.e. length 2*len(a)-1).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv.max())


def full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Return the whole linear convolution a*a (length 2*len(a)-1) using FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return conv


def enforce_symmetry(a: np.ndarray) -> np.ndarray:
    """Make the vector symmetric: a_i = a_{n-1-i}."""
    return (a + a[::-1]) / 2.0


def arcsine_distribution(n: int) -> np.ndarray:
    """
    Exact discretisation of the continuous arcsine density
    f(x)=1/(π√(x(1‑x))) on [0,1] by integrating over each bin
    [i/n,(i+1)/n).  The resulting vector sums to 1.
    """
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta(α,β) density evaluated at the centre of each bin
    ((i+0.5)/n).  Returns a non‑negative vector that is normalised to sum = 1.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def evaluate_vec(a: np.ndarray, fft_len: int) -> float:
    """
    Compute the exact score used by `evaluate_sequence` for a normalised
    vector (sum = 1).  This is faster than calling the external function.
    """
    mx = max_convolution(a, fft_len)
    return float(2 * a.size * mx)


def smooth_gradient(a: np.ndarray, conv: np.ndarray, mu: float,
                    fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed objective
        f_mu(a) = mu * log( sum_k exp(conv_k / mu) )
    with respect to a, evaluated at the current vector a.
    The result is 2 · cross‑correlation(softmax(conv/ mu), a).
    """
    # Soft‑max weights (stable)
    max_val = conv.max()
    # exponentials are ≤1 because we subtract max_val
    exp_w = np.exp((conv - max_val) / mu)
    w = exp_w / exp_w.sum()

    # Cross‑correlation via FFT: g = 2 * corr(w, a)
    w_fft = np.fft.fft(w, fft_len)
    a_fft = np.fft.fft(a, fft_len)
    corr = np.real(np.fft.ifft(np.conj(w_fft) * a_fft))[: a.size]
    return 2.0 * corr

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The search proceeds through a parametric sweep,
    a smooth projected Adam optimisation (with symmetry) and a final
    stochastic refinement.  It respects the time budget and prints progress.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    overall_deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two

    # ------------------------------------------------------------------
    # 0) Broad parametric sweep
    # ------------------------------------------------------------------
    candidates = []

    # 0a) reuse previously discovered vector if available
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.size == n and a0.sum() > 0.0:
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # 0b) arcsine (known good baseline)
    candidates.append(arcsine_distribution(n))

    # 0c) uniform
    candidates.append(np.full(n, 1.0 / n))

    # 0d) symmetric Beta with a wide range of α
    for alpha in np.linspace(0.05, 2.0, 31):  # 0.05 … 2.0
        candidates.append(beta_distribution(n, alpha, alpha))

    # 0e) asymmetric Beta pairs (randomly sampled)
    for _ in range(100):
        a = 10 ** rng.uniform(-2, 0.5)  # α in approx [0.01, 3]
        b = 10 ** rng.uniform(-2, 0.5)  # β in same range
        candidates.append(beta_distribution(n, a, b))

    # 0f) random mixtures of two Beta components
    for _ in range(80):
        α1 = 10 ** rng.uniform(-2, 0.5)
        β1 = 10 ** rng.uniform(-2, 0.5)
        α2 = 10 ** rng.uniform(-2, 0.5)
        β2 = 10 ** rng.uniform(-2, 0.5)
        w = rng.random()
        mix = w * beta_distribution(n, α1, β1) + (1.0 - w) * beta_distribution(n, α2, β2)
        mix = mix / mix.sum()
        candidates.append(mix)

    # ------------------------------------------------------------------
    # 1) Pick the best cheap candidate
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    for idx, a in enumerate(candidates):
        # evaluate quickly using the FFT‑based objective
        sc = evaluate_vec(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[init {idx:03d}] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Smooth projected Adam optimisation (≈60 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.60 * budget_s
    a = enforce_symmetry(best_a)
    a = proj_simplex(a)  # guarantee feasibility

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr = 0.08
    t = 0

    # initialise smoothing temperature μ
    conv_initial = full_convolution(a, fft_len)
    mu = max(conv_initial.max() * 0.6, 1e-7)  # start relatively warm

    last_improve = time.time()
    while time.time() < phase1_deadline:
        t += 1

        # ---------- compute convolution ----------
        conv_full = full_convolution(a, fft_len)
        conv_max = conv_full.max()

        # ---------- gradient of smoothed max ----------
        grad = smooth_gradient(a, conv_full, mu, fft_len)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ---------- projection & symmetry ----------
        a = proj_simplex(a)
        a = enforce_symmetry(a)
        a = proj_simplex(a)

        # ---------- evaluate current solution ----------
        conv_full_new = full_convolution(a, fft_len)
        cur_score = 2.0 * n * conv_full_new.max()
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam {t:07d}] improved to {best_score:.6f}")

        # ---------- adapt smoothing temperature ----------

Propose a step‑function (non‑negative vector) that minimises

1. **Broad parametric sweep** – generate many candidates from

   and keep the best one.

2. **Smooth projected Adam optimisation** – minimise the exact
   objective `2·n·max(conv(a,a))` on the probability simplex.
   The non‑smooth ∞‑norm is approximated by a soft‑max with a
   temperature that is slowly reduced; Adam with Euclidean simplex
   projection and an enforced symmetry (a_i = a_{n‑1‑i}) steers the
   search.

3. **Local stochastic refinement** – a simulated‑annealing style
   random mass‑transfer that fine‑tunes the solution.

The routine respects the given time‑budget (default 1000 s) and prints
intermediate scores.
"""

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum x = 1}. Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerically all entries are zero – fall back to uniform.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return max(conv(a, a)) using an FFT of length `fft_len`."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv.max())


def full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the full linear convolution a*a (length 2*len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return conv


def enforce_symmetry(a: np.ndarray) -> np.ndarray:
    """Force a_i = a_{n-1-i} (mirror symmetry)."""
    return (a + a[::-1]) / 2.0


def arcsine_distribution(n: int) -> np.ndarray:
    """
    Discretise the arcsine density f(x)=1/(π√(x(1‑x))) on [0,1] by integrating
    over each bin [i/n,(i+1)/n).  Returns a probability vector of length n.
    """
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta(α,β) density evaluated at bin centres ((i+0.5)/n).  Returns a
    normalised probability vector.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def evaluate_vec(a: np.ndarray, fft_len: int) -> float:
    """
    Exact score used by `evaluate_sequence` for a normalised vector
    (sum = 1).  Saves the overhead of the external call.
    """
    mx = max_convolution(a, fft_len)
    return float(2 * a.size * mx)


def smooth_gradient(a: np.ndarray, conv: np.ndarray, mu: float,
                    fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed objective
        f_mu(a) = mu * log( Σ_k exp(conv_k / mu) )
    with respect to a.  It equals
        2 * cross_correlation( softmax(conv/mu), a ).
    """
    max_val = conv.max()
    # Soft‑max (stable)
    exp_w = np.exp((conv - max_val) / mu)  # ≤ 1
    w = exp_w / exp_w.sum()

    # Cross‑correlation via FFT (g = 2·corr(w, a))
    w_fft = np.fft.fft(w, fft_len)
    a_fft = np.fft.fft(a, fft_len)
    corr = np.real(np.fft.ifft(np.conj(w_fft) * a_fft))[: a.size]
    return 2.0 * corr
