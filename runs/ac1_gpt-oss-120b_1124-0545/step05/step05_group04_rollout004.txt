def compute_subgradient(a):
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = conv.max()
    active = (conv >= max_val - 1e-12).astype(np.float64)
    cnt = active.sum()
    if cnt == 0:
        return np.zeros_like(a)
    w = active / cnt
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n-1: 2*n-1]
    grad = 2.0 * slice_part[::-1]
    return grad
```

"""
Improved search for a step‑function that minimises

    2·n·max(conv(a,a)) / (∑a)²

The algorithm consists of:

1. **Extended analytic sweep** – explore a larger parameter space for
   power‑law families (product‑law & sum‑law) and add a *dual‑Gaussian*
   family (two symmetric bumps).  The best candidate becomes the starting
   point.

2. **Fine local refinement of the analytic parameters** – a quick
   hill‑climbing around the best (p,offset) found by the sweep.

3. **Smoothed‑max optimisation** – minimise a smooth log‑sum‑exp
   approximation Φ_λ with Adam while increasing λ up to 2·10⁵.
   The optimiser works in the log‑probability (soft‑max) space, keeping the
   vector on the probability simplex automatically.

4. **Exact sub‑gradient descent** – after the continuation phase, run a
   projected sub‑gradient method on the true objective with a 1/√t step‑size.

5. **Guided mass‑transfer annealing** – finally perform a cheap
   simulated‑annealing style local search that moves probability mass
   from indices with large sub‑gradient (i.e. those that increase the
   max‑convolution) to indices with small sub‑gradient.

All phases are time‑boxed by the supplied `budget_s`.  The routine prints
progress messages and returns the best sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a, normalised internally.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from a random index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^(‑p)   (edge‑heavy shape)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^(‑p) + (n‑1‑i+off)^(‑p)   (two tails)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_dual_gaussian(n: int, L: int, sigma: float) -> np.ndarray:
    """
    Two symmetric Gaussian bumps.
    L – distance from the left edge to the centre of the left bump (integer).
    sigma – standard deviation of each bump (in index units).
    """
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i - L) / sigma) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1 - L)) / sigma) ** 2)
    a = left + right
    a = a / a.sum()
    return a


def analytic_dual_exponential(n: int, L: int, c: float) -> np.ndarray:
    """
    Two symmetric exponential decays.
    """
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-c * np.abs(i - L))
    right = np.exp(-c * np.abs(i - (n - 1 - L)))
    a = left + right
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max utilities (Φ_λ and its gradient)
# ----------------------------------------------------------------------


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    For a given log‑probability vector x, smoothing λ and length n,
    return the smooth max Φ_λ(a) and its gradient w.r.t. x.
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # convolution a*a
    conv = conv_fft_self(a)                # length 2n‑1
    max_c = np.max(conv)

    # smooth max over convolution entries: log‑sum‑exp trick
    shift = conv - max_c                   # ≤ 0
    exp_term = np.exp(lam * shift)        # ≤ 1
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam    # Φ_λ

    w = exp_term / sum_exp                 # (2n‑1) weights, sum to 1

    # gradient of Φ_λ w.r.t. a
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]      # length n
    grad_a = 2.0 * slice_part[::-1]            # ∂Φ/∂a

    # map gradient to x‑space using Jacobian of soft‑max
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)                # ∇_x Φ = J^T ∇_a Φ
    return phi, grad_x


def adam_optimize(x: np.ndarray, lam: float, n: int, steps: int,
                  lr: float, deadline: float) -> np.ndarray:
    """Simple Adam optimiser for the smooth max objective."""
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for t in range(1, steps + 1):
        if time.time() >= deadline:
            break
        _, grad = compute_phi_and_grad(x, lam, n)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step
    return x


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max(conv))
# ----------------------------------------------------------------------


def subgrad_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step for the exact objective
    max(conv(a,a)).  Uses an FFT‑based computation of the sub‑gradient.
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    # binary mask for all entries that are (almost) maximal
    active = (conv >= max_val - 1e-12).astype(np.float64)
    cnt = active.sum()
    if cnt == 0:
        return a
    w = active / cnt
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]
    a_new = a - step * grad
    return project_simplex(a_new)


def compute_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Return the sub‑gradient of max(conv(a,a)) (same computation as in
    `subgrad_step` but without the projection step).
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    active = (conv >= max_val - 1e-12).astype(np.float64)
    cnt = active.sum()
    if cnt == 0:
        return np.zeros_like(a)
    w = active / cnt
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]
    return grad


def guided_mass_transfer(a: np.ndarray, grad: np.ndarray, scale: float) -> np.ndarray:
    """
    Move a small amount of probability mass from the index with the
    largest sub‑gradient (i.e. most harmful) to the index with the
    smallest sub‑gradient.
    """
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return a
    delta = scale * a[i]
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

def refine_product_sum(a0: np.ndarray, mode: str, n: int,
                      time_deadline: float) -> np.ndarray:
    """
    Small hill climbing around a starting (p,off) pair for the chosen mode.
    Returns a potentially improved vector (or the original if time runs out).
    """
    # recover the (p,off) that produced a0 by simple line‑search on the grid
    # (we just reuse the current a0 as a seed and perform local perturbations)
    best = a0.copy()
    best_score = compute_score(best)
    # initialise a log‑parameter vector for numeric stability
    log_a = np.log(np.maximum(best, 1e-12))
    # simple gradient‑ascent in (p,off) space is hard, so we just try a few
    # random perturbations of the parameters and keep improvements.
    for _ in range(200):
        if time.time() >= time_deadline:
            break
        # propose a random perturbation of the log‑vector then project
        pert = log_a + 0.02 * np.random.randn(n)
        cand = softmax(pert)
        if mode == "product":
            tmp = analytic_product(n, 0.5, 0.1)  # dummy to keep shape, not used
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best = sc, cand.copy()
            log_a = np.log(np.maximum(best, 1e-12))

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    print(f"[grid] best after extended power‑law sweep = {best_score:.6f}")

    if time.time() < deadline:

            temp *= 0.995
        else:

            prob = np.exp(-(sc_new - best_score) / max(temp, 1e-8))
            if np.random.rand() < prob:
                a = cand
            # slowly anneal temperature and step size
            scale = min(0.03, scale * 1.0015)
            temp *= 0.999
    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    return best_a.tolist()
