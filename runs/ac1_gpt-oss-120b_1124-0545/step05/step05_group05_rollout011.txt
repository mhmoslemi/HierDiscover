    for lam in lam_vals:
        cand = analytic_exponential(n, lam)
        _add_candidate(cand, top, max_top)

    if "height_sequence_1" in globals():

"""
Improved search for a step‑function minimizing

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Extended analytic grid** – systematic scan of several parametric families
   (beta, cosine‑power, product, sum, exponential) on a finer parameter grid
   and a small top‑10 list of candidates.

2. **Adam optimisation of a soft‑max surrogate** – starting from the best grid
   candidate (and later from a few random Dirichlet restarts) we minimise the
   smooth objective  log‑sum‑exp(β·conv) using Adam with a temperature schedule.
   The iterates are projected onto the probability simplex after each step.

3. **Final random refinement** – Gaussian jitter around the current best vector
   for the remaining wall‑time.

All updates respect the non‑negativity and simplex constraints, and the
procedure keeps the best sequence ever seen.  The total wall‑time is limited
to the supplied budget (default 1000 s).  The returned list can be passed
directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic operations
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (full length a+b-1)."""
    size = _next_pow2(a.shape[0] + b.shape[0] - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a vector on the probability simplex."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_self(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def _add_candidate(cand: np.ndarray,
                   best_list: list[tuple[float, np.ndarray]],
                   max_len: int = 10) -> None:
    """Insert a candidate into a sorted (by score) bounded list."""
    sc = compute_score(cand)
    inserted = False
    for idx, (s, _) in enumerate(best_list):
        if sc < s:
            best_list.insert(idx, (sc, cand.copy()))
            inserted = True
            break
    if not inserted and len(best_list) < max_len:
        best_list.append((sc, cand.copy()))
    while len(best_list) > max_len:
        best_list.pop()

# Analytic families (return probability vectors, sum to 1)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta_
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Adam optimiser (soft‑max surrogate)
# ----------------------------------------------------------------------


def adam_optimize(a_init: np.ndarray,
                  n: int,
                  start_time: float,
                  end_time: float,
                  budget_s: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the smooth objective
        log‑sum‑exp(β·conv(a,a))
    Returns the best vector seen in the interval and its exact score.
    """
    a = a_init.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_local = a.copy()
    best_local_score = compute_score(a)
    # hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.08
    beta_start, beta_end = 2.0, 400.0   # temperature schedule

    while time.time() < end_time:
        conv = conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local = a.copy()

        # temperature (slowly increase to focus on the true max)
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 10.0) if (budget_s - 10.0) > 0 else elapsed
        beta = beta_start * (1.0 - progress) + beta_end * progress

        # soft‑max weights
        w_raw = np.exp(beta * (conv - max_val))
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # gradient of the smoothed objective:
        #   ∂/∂a_j  ( a·a ) = 2·a_{i-j}
        a_rev = a[::-1]
        corr_full = conv_two(w, a_rev)                 # length = 3n‑2
        weighted_shift = corr_full[n - 1:n - 1 + n]    # length n
        g = a + weighted_shift                         # ∇f(a)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step_vec
        a = np.clip(a, 0.0, None)
        a = project_simplex(a)

        # occasional random mass‑transfer move to escape plateaus
        if np.random.rand() < 0.02:
            a = random_move(a, lr * 0.5)

    return best_local, best_local_score
