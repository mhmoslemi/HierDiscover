def _adam_optimize(seed_vec, n, deadline, max_iters=15000):
    eps = 1e-30
    v = np.log(seed_vec + eps)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    lr0 = 0.8
    eps_adam = 1e-12

    beta = 0.5
    beta_mul = 1.02
    beta_cap = 200.0

    best_seq = None
    best_val = float('inf')
    t = 0

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # smoothing
        max_conv = np.max(conv)
        beta = min(beta * beta_mul, beta_cap)
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            continue
        w /= w_sum

        # gradient of weighted conv w.r.t a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1:n-1+n]

        # gradient wrt v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v**2)
        m_hat = m / (1.0 - beta1**t)
        v2_hat = v2 / (1.0 - beta2**t)
        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)
    return best_seq, best_val
```

def _project_onto_simplex(v):
    # Projects onto simplex {x: sum x = 1, x >= 0}
    # Implementation from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # largest i satisfying condition
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
```

def _subgradient_refine(start_seq, n, deadline, max_iters=200000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    tol = 1e-12

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        cur_val = 2.0 * n * max_val
        # update best
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # find all max indices within tolerance
        max_idxs = np.where(conv >= max_val - tol)[0]
        # construct w as uniform over maxima
        w = np.zeros_like(conv)
        w[max_idxs] = 1.0 / len(max_idxs)

        # gradient of weighted conv w.r.t a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n-1:n-1+n]

        # step size
        lr = 0.6 / np.sqrt(it)

        # projected gradient descent
        a_new = a - lr * grad
        # projection onto simplex
        a = _project_onto_simplex(a_new)

    return best_seq, best_val
```

"""
Propose a step‑function with a low 2·n·max(conv)/∑a² value.

Algorithm
---------
1. **Seed generation** – a diverse collection of analytically‑motivated shapes
   (uniform, geometric, power‑law, sine, Hann, semi‑circular, beta‑like,
   random Dirichlet, sparse patterns).  The previously discovered sequence
   is also added when it has the same length.

2. **Fast Adam optimisation** – the sequence is parametrised by log‑softmax
   variables `v`.  A log‑sum‑exp with a temperature that grows up to a fixed
   cap approximates the max‑operator.  The gradient of the smooth max is
   obtained by a single FFT‑based convolution.  Adam (with standard
   hyper‑parameters) minimise the smooth objective; the best iterate is kept.

3. **Exact sub‑gradient mirror descent** – starting from the best Adam
   result we perform a projected‑gradient descent on the true
   `max(conv)` (no smoothing).  All active indices (those attaining the
   current maximum) are averaged to obtain a genuine sub‑gradient.
   The iterate stays on the probability simplex by a simplex‑projection.

4. **Simulated‑annealing hill‑climbing** – a cheap stochastic neighbourhood
   that redistributes the total mass among a few randomly chosen positions.
   Improvements are always accepted; uphill moves are accepted with a
   temperature that decays geometrically.

All phases respect a global time budget (default 1000 s).  The routine
returns the best sequence found as a plain ``list`` of ``float`` values.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming ∑seq == 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _decreasing_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (1 - i/(n-1))**p  (monotone decreasing)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - i / (n - 1)) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _semicircle(n: int) -> np.ndarray:
    x = np.linspace(-1.0, 1.0, n, dtype=np.float64)
    a = np.sqrt(np.clip(1.0 - x ** 2, 0.0, None))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed by a Dirichlet draw."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None, rng: np.random.Generator) -> list[np.ndarray]:
    """Assemble a diverse set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_semicircle(n))

    # geometric families (both fast and slow decay)
    for r in (0.2, 0.4, 0.6, 0.8, 0.9):
        if r > 0.0:
            seeds.append(_geom(n, r))

    # decreasing power‑law families
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        seeds.append(_decreasing_power(n, p))

    # pure power‑law (inverse)
    for p in (0.2, 0.5, 1.0, 1.5, 2.0):
        seeds.append(_power(n, p))

    # beta‑like shapes
    for a in (0.3, 0.5, 0.7):
        for b in (0.3, 0.5, 0.7):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse random patterns
    for k in (3, 5, 8, 12):
        for conc in (0.2, 0.5):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse previously discovered sequence (if length matches)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

def _adam_optimize(seed_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 15000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a smooth approximation of max(conv).
    ``seed_vec`` is already normalised to sum = 1.
    """
    eps = 1e-30
    v = np.log(seed_vec + eps)                      # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 0.5               # initial smoothness
    beta_mul = 1.02
    beta_cap = 200.0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        max_conv = np.max(conv)
        beta = min(beta * beta_mul, beta_cap)
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w_k·conv_k w.r.t. a (linear convolution) ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # keep central n entries

        # ---- convert to gradient w.r.t. v (softmax Jacobian) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional gradient clipping for stability
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ---- Adam update on v ----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on max(conv) with a simplex projection.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        cur_val = 2.0 * n * max_val

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # find all (near) maximal indices
        max_idxs = np.where(conv >= max_val - tol)[0]
        w = np.zeros_like(conv)
        w[max_idxs] = 1.0 / max_idxs.size   # uniform weighting over active set

        # gradient of weighted conv w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1:n - 1 + n]

        # step size (standard decreasing schedule)
        lr = 0.6 / np.sqrt(it)

        # projected gradient step on the simplex
        a_new = a - lr * grad
        a = _project_onto_simplex(a_new)

        # avoid pathological zero‑vector (numerical safety)
        if np.sum(a) == 0.0:
            a = _uniform(n)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200000,
                         init_temp_factor: float = 0.02,
                         min_temp: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing with exponential cooling.  The move picks a
    small random subset of indices and redistributes its total mass using a
    Dirichlet proposal.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9995, min_temp)

        # pick a small random subset (2‑4 indices)
        k = rng.integers(2, 5)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # fresh allocation respecting the subset's total mass
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:6d} best={new_val:.6f}")

    return best_seq, best_val
