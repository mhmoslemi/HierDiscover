Evolutionary + sub‑gradient search for a low‑max‑auto‑correlation step function.

The objective (after normalising the sum to 1) is

    2·n·max(conv(a,a))

which is exactly the value returned by ``evaluate_sequence``.
The algorithm works in three stages:

        * Gaussian perturbation in the log‑softmax (soft) space,
        * Simple averaging crossover of two parents.
   Children are evaluated with an FFT‑based convolution; the best
   individuals are kept, and the global best is reported continuously.

3. **Exact sub‑gradient refinement** – the current global best is fed
   into a long mirror‑descent run (the same exponentiated‑gradient step
   used in the previous state‑of‑the‑art).  This sharpens the solution
   without any smoothing.

The whole routine is time‑aware (default budget = 1000 s) and stops
as soon as the deadline is reached.  The returned list contains only
non‑negative floats and sums to 1, so ``evaluate_sequence`` will give the
same objective value as the internal one.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (fast even for large n).
    The result length is ``len(x)+len(y)-1``.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(a,a)) for a probability vector ``a`` (sum(a)=1).
    """
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Assemble a modest but diverse list of normalised seeds.
    """
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for r in (0.2, 0.4, 0.6, 0.8):
        seeds.append(_geom(n, r))

    # beta‑like shapes
    for a in (0.3, 0.6, 0.9):
        for b in (0.3, 0.6, 0.9):
            seeds.append(_beta_shape(n, a, b))

    # a few random Dirichlet draws
    for conc in (0.2, 0.8, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # reuse previous construction if lengths match
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # shuffle to avoid systematic bias
    rng.shuffle(seeds)
    return seeds


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient mirror descent on the true max‑conv objective.
    Keeps the iterate on the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} new best={cur_val:.6f}")

        # sub‑gradient of max‑conv w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent (exponentiated) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Evolutionary optimiser
# ----------------------------------------------------------------------


def _evolutionary_optimize(seeds: list[np.ndarray],
                           n: int,
                           deadline: float,
                           pop_size: int = 15) -> tuple[np.ndarray, float]:
    """
    Simple population based search.
    Starts from ``seeds`` (all are normalised), keeps the best ``pop_size``
    individuals and evolves them until the deadline.
    """
    rng = np.random.default_rng()
    # evaluate all seeds and keep the best ``pop_size`` of them
    pop: list[tuple[np.ndarray, float]] = []
    for s in seeds:
        val = _objective(s)
        pop.append((s, val))
    pop.sort(key=lambda x: x[1])
    pop = pop[:pop_size]

    global_best_seq, global_best_val = pop[0][0].copy(), pop[0][1]

    it = 0
    while time.time() < deadline:
        it += 1
        # pick an operator
        op = rng.random()
        # always draw a parent from the current population (uniform)
        parent = pop[rng.integers(len(pop))][0]

        if op < 0.6:
            # -------- Dirichlet reshuffle of a few indices ----------
            k = rng.integers(2, 7)                # 2–6 indices
            idx = rng.choice(n, size=k, replace=False)
            total = parent[idx].sum()
            if total > 0.0:
                new_alloc = total * rng.dirichlet(np.full(k, 1.0))
                child = parent.copy()
                child[idx] = new_alloc
            else:
                child = parent.copy()
        elif op < 0.9:
            # -------- Gaussian perturbation in log‑softmax ----------
            # scale of the noise shrinks as we approach the deadline
            progress = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - time.time()))
            # simple linear schedule: start at 0.2, end at 0.01
            sigma = 0.2 * (1.0 - progress) + 0.01
            v = np.log(parent + 1e-30)
            v += rng.normal(scale=sigma, size=n)
            child = _softmax(v)
        else:
            # -------- Crossover (average of two parents) ----------
            p2 = pop[rng.integers(len(pop))][0]
            child = (parent + p2) * 0.5
            child = child / child.sum()

        val = _objective(child)

        # keep a record of the global best
        if val < global_best_val:
            global_best_val = val
            global_best_seq = child.copy()
            print(f"[EA] it={it:6d} new global best={val:.6f}")

        # insert child into population if it is good enough
        if len(pop) < pop_size:
            # insert keeping the list sorted
            pos = 0
            while pos < len(pop) and pop[pos][1] <= val:
                pos += 1
            pop.insert(pos, (child, val))
        else:
            # replace the worst individual if the child is better
            if val < pop[-1][1]:
                # find insertion point
                pos = 0
                while pos < len(pop) and pop[pos][1] <= val:
                    pos += 1
                pop.insert(pos, (child, val))
                pop.pop()   # discard the previous worst

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget and returns the best sequence found (as a ``list`` of
    ``float``).  The algorithm combines an analytic seed phase, a
    population‑based stochastic optimisation, and a final exact
    sub‑gradient refinement.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a tiny safety buffer

    # ------------------------------------------------------------------
    # Determine a working length and obtain a previous solution if any
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # we aim for a reasonably large n (larger → finer discretisation)
    # but still cheap enough for many iterations.
    n_desired = 5000
    if prev_arr is not None and prev_arr.size > 0:
        # keep the same length if it is already large enough
        if prev_arr.shape[0] >= 3000:
            n_desired = prev_arr.shape[0]

    n = int(n_desired)

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Build the initial seed pool
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, prev_arr, rng)

    # ------------------------------------------------------------------
    # Evolutionary optimisation (major part of the budget)
    # ------------------------------------------------------------------
    # allocate ~80 % of the overall time to the EA
    ea_deadline = start_time + 0.80 * (deadline - start_time)
    best_seq, best_val = _evolutionary_optimize(seeds, n, ea_deadline, pop_size=15)

    # ------------------------------------------------------------------
    # Final exact sub‑gradient refinement (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_val = _subgradient_refine(best_seq, n, deadline, max_iters=200_000)

    # ------------------------------------------------------------------
    # Normalise to sum = 1 (the evaluator uses the same normalisation)
    # ------------------------------------------------------------------
    best_seq = best_seq / best_seq.sum()

    # sanity check – optional
    try:
        _ = evaluate_sequence(best_seq.tolist())
    except Exception:
        # ignore – the very tight input checks inside ``evaluate_sequence`` will
        # only raise for non‑numeric or NaN/Inf entries which cannot happen here.
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
