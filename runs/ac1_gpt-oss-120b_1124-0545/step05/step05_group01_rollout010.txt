function proj_simplex(v):
    # returns projection onto {x >=0, sum x = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

from scipy.optimize import linprog
from scipy.sparse import diags, csr_matrix

def _lp_refine(a, n, deadline, max_iters=15, t=0.5):
    # a: numpy array, sum 1
    a = a.copy()
    best_seq = a
    best_val = _objective(a)
    for it in range(max_iters):
        if time.time() > deadline: break
        conv = _conv_fft(a, a)
        M = conv.max()
        # Build sparse convolution matrix
        offsets = np.arange(n)
        # each diagonal constant a_i
        diag_data = [np.full(n, a_i) for a_i in a]
        A = diags(diag_data, offsets, shape=(2*n-1, n), format='csr')
        # linear programming:
        c = -np.ones(n)   # maximize sum(g) => minimize -sum
        b_ub = np.full(2*n-1, M + 1e-12)   # right-hand side
        bounds = [(0, None)] * n
        # solve
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # if fails, break
            break
        g = res.x
        # scale to sum=1
        if g.sum() > 0:
            g_norm = g / g.sum()
        else:
            g_norm = a  # fallback
        # convex combination
        a = (1 - t) * a + t * g_norm
        # enforce simplex
        a = a / a.sum()
        # compute value
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[LP] n={n} it={it+1} best={val:.6f}")
        # optionally reduce t gradually
        t = max(0.1, t * 0.9)
    return best_seq, best_val

"""
Non‑negative step‑function optimizer.

The original objective (scale‑invariant) is
    2·n·max(conv(a,a)) / (sum(a))² .
We fix sum(a)=1 and minimise   2·n·max(conv(a,a)) .

The optimiser is a hybrid:
  1️⃣  Diverse deterministic seeds (uniform, geometric, power‑law,
      symmetric‑geometric, sine, Gaussian, Dirichlet, triangular, parabolic,
      random mixtures of Gaussians).
  2️⃣  Adam optimisation of a smooth log‑sum‑exp surrogate of the max‑convolution.
  3️⃣  Exact sub‑gradient descent on the true max, using Euclidean projection
      onto the simplex (faster than pure exponentiated‑gradient).
  4️⃣  A linear‑programming “active‑set” step:
      for the current a we maximise Σg_i subject to   a∗g ≤ M,
      where M = max(a∗a).  The optimiser then moves a towards g.
  5️⃣  Light simulated‑annealing for final polishing.

All phases respect the global time budget (default 1000 s).  The best
sequence found (list of floats) is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the grader (sum(seq)=1 is assumed)."""
    n = seq.size
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex helpers
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a point on the simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto {x ≥ 0, Σx = 1}.
    Returns the projected vector (same shape as v).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed families (all return vectors normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _triangle(n: int) -> np.ndarray:
    """Symmetric linear tent (peak in the middle)."""
    m = n // 2
    # left side 1..m, right side m..1 (for even n the peak is duplicated)
    left = np.arange(1, m + 1, dtype=np.float64)
    if n % 2 == 0:
        right = left[::-1]
    else:
        right = np.arange(m, 0, -1, dtype=np.float64)
    a = np.concatenate([left, right])
    return a / a.sum()


def _parabolic(n: int) -> np.ndarray:
    """Quadratic bump: (1 - ((i-(n-1)/2)/(n/2))**2) clipped to ≥0."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    width = n / 2.0
    a = 1.0 - ((i - centre) / width) ** 2
    a[a < 0] = 0.0
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse or dense Dirichlet draw."""
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _mixture_gaussian(n: int, rng: np.random.Generator, comps: int = 2) -> np.ndarray:
    """Random mixture of a few Gaussians (positive weights, normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.zeros_like(i)
    weights = rng.random(comps)
    weights /= weights.sum()
    for w in weights:
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.03 * n, 0.25 * n)
        a += w * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse pool of starting vectors (already normalised to sum = 1).
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))
    seeds.append(_triangle(n))
    seeds.append(_parabolic(n))

    # Dirichlet draws – cover sparse to dense regimes
    for alpha in [0.05, 0.1, 0.3, 0.6, 1.0, 2.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # random mixtures of Gaussians
    for _ in range(12):
        seeds.append(_mixture_gaussian(n, rng, comps=rng.integers(2, 4)))

    # mirror each seed – helpful because the optimum may be asymmetric
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # reuse previously found best construction (if length matches)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # guarantee a non‑empty list
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation of smoothed max (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 12000,
                   beta0: float = 2.0,
                   beta_mul: float = 1.04,
                   lr0: float = 0.9,
                   lambda0: float = 0.0,
                   lambda_decay: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth loss
        L = (1/β)·log Σ exp(β·conv)  + λ·‖conv‖²₂ .
    Returns the best vector found (as a simplex point) and its exact objective.
    """
    # logits for the simplex
    v = np.log(seed + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0
    lam = lambda0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # point on the simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # increase smoothing (approach the true max)
        beta *= beta_mul

        # ----- smooth‑max weights (log‑sum‑exp) -----
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of smooth max (convolution with the weight vector)
        rev_a = a[::-1]
        grad_max = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]

        # optional L2 regularisation of the convolution
        if lam != 0.0:
            grad_sq = 2.0 * _conv_fft(rev_a, conv)[n - 1:n - 1 + n]
        else:
            grad_sq = 0.0

        grad = grad_max + lam * grad_sq

        # ----- chain rule through softmax -----
        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)

        # ----- Adam update on logits -----
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

        # decay the regulariser (if used)
        lam *= lambda_decay

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent with Euclidean projection onto simplex
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on max(conv) with Euclidean projection.
    Returns the best vector seen and its exact objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the max‑convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size ~ 1/√t
        eta = 0.75 / np.sqrt(it)
        a_new = a - eta * g
        a = _proj_simplex(a_new)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming active‑set refinement
# ----------------------------------------------------------------------


def _lp_refine(a0: np.ndarray,
               n: int,
               deadline: float,
               max_iters: int = 12,
               mix: float = 0.4) -> tuple[np.ndarray, float]:
    """
    LP‑based refinement.
    For the current a we solve:
        maximise   Σ g_i
        subject to  a * g ≤ M   (elementwise)
                    g ≥ 0
    where M = max(a * a).  After solving we move a towards the normalised
    solution ĝ and project back onto the simplex.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()

        # Build the dense convolution matrix (n×(2n‑1) non‑zero entries per column)
        # A has shape (2n‑1, n) and satisfies A @ g = a * g
        offsets = np.arange(n)
        # each diagonal i contains the constant a[i]
        diag_vals = [np.full(n, a_i, dtype=np.float64) for a_i in a]
        A = diags(diag_vals, offsets, shape=(2 * n - 1, n), format="csr")

        c = -np.ones(n)                     # maximise Σ g_i  → minimise –Σ g_i
        b_ub = np.full(2 * n - 1, M + 1e-14)  # tiny tolerance to avoid numerical issues
        bounds = [(0.0, None)] * n

        # Solve the linear programme (HiGHS is fast and robust)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
        if not res.success:
            # If the LP fails we abort the refinement
            print(f"[LP] n={n} it={it} linprog failed ({res.message})")
            break

        g = res.x
        if g.sum() == 0.0:
            g_norm = a
        else:
            g_norm = g / g.sum()

        # convex combination and projection back onto the simplex
        a = (1.0 - mix) * a + mix * g_norm
        a = _proj_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LP] n={n:5d} it={it:3d} best={cur_val:.6f}")

        # slowly shrink the mixing factor to stabilise
        mix = max(0.05, mix * 0.85)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Cheap simulated‑annealing fine‑tuning (as in the previous code)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # start temperature as a small fraction of the current objective
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.025 * cur[i]   # move up to 2.5 % of source mass
        delta = min(delta, cur[i])

        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        new_val = _objective(new)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new.copy()
                print(f"[Anneal] n={n:5d} it={it:5d} best={new_val:.6f}")

        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new = cur + noise
            new = np.clip(new, 0.0, None)
            if new.sum() == 0.0:
                new = _uniform(n)
            else:
                new /= new.sum()
            new_val = _objective(new)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new.copy()
                    print(f"[Anneal] n={n:5d} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found
    as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # optionally reuse a previously discovered construction
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – we concentrate on a medium size (4000) which
    # gives a good speed/quality trade‑off and also try a larger one if
    # time permits.
    # ------------------------------------------------------------------
    lengths = [4000, 8000]   # we will attempt the second only when time allows
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------- 1️⃣  seed generation -------------------
        seeds = _make_seeds(n, previous)

        # quick evaluation of raw seeds, keep a handful of promising ones
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(12, len(seed_vals))]]

        # record the best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------- 2️⃣  Adam optimisation -----------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=12000,
                beta0=2.0,
                beta_mul=1.05,          # slightly faster temperature increase
                lr0=0.9,
                lambda0=0.0,            # we omit the variance regulariser (empirically better)
                lambda_decay=1.0,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # ------------------- 3️⃣  Exact sub‑gradient refinement -----
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline, max_iters=50000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------- 4️⃣  LP‑based active‑set refinement -----
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _lp_refine(global_best_seq, n, deadline, max_iters=12, mix=0.4)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall LP] n={n} best={val:.6f}")

        # ------------------- 5️⃣  Simulated‑annealing fine‑tune ------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
