def _lp_refine(a, n, deadline, max_iters=5):
    best_seq = a.copy()
    best_val = 2*n*np.max(_conv_fft(best_seq, best_seq))

    curr = a.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        M = np.max(_conv_fft(curr, curr))

        # Build constraint matrix
        A = sp.diags(curr, offsets=np.arange(n), shape=(2*n-1, n), format='csc')
        # linear program maximize sum(b) s.t. A b <= M
        c = -np.ones(n)  # minimize -sum => maximize sum

        bounds = [(0, None)] * n
        res = sp_linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
        if not res.success:
            print("[LP] iteration {}: linprog failed".format(it))
            break
        b = res.x
        # Normalize b to sum=1
        sum_b = b.sum()
        if sum_b == 0:
            break
        b_norm = b / sum_b

        # Now try mixing with current:
        # objective for t in [0,1] mixing f=(1-t)*curr + t*b_norm
        # Since both have sum 1 (curr sum=1, b_norm sum=1), mixing also sum=1.
        # So we can evaluate conv for t in some grid (maybe 0,0.25,0.5,0.75,1)
        # Or we can perform line search by golden-section.
        # We'll evaluate few points to find best.
        t_vals = np.linspace(0, 1, 6) # 0,0.2,...1
        best_local_seq = curr
        best_local_val = best_val
        for t in t_vals:
            new_seq = (1-t)*curr + t*b_norm
            val = 2*n*np.max(_conv_fft(new_seq, new_seq))
            if val < best_local_val:
                best_local_val = val
                best_local_seq = new_seq.copy()
        curr = best_local_seq
        if best_local_val < best_val - 1e-12:
            best_val = best_local_val
            best_seq = best_local_seq.copy()
            print(f"[LP] {n=} iter={it} best={best_val:.6f}")
    return best_seq, best_val
```

"""
Hybrid optimisation for the max‑auto‑convolution problem.

* A fast low‑dimensional Differential Evolution (DE) runs on a
  piece‑wise‑linear parametrisation at a modest size (≈2000).  This
  gives a good “shape” while keeping the number of expensive
  convolutions low.

* The best low‑dimensional candidate is up‑sampled to the final
  length (4000) and polished with three deterministic refinements:
    – Adam on a smooth‑max surrogate (many iterations, slow cooling);
    – Exact sub‑gradient descent on the true max‑convolution;
    – A tiny linear‑programming step that maximises the ℓ₁‑norm
      under the current max‑convolution bound (optional, falls back
      gracefully if SciPy is unavailable).

* A cheap stochastic hill‑climb finishes the search.

All phases respect a global 1000 s budget and keep the best sequence
found so far.  The final sequence is returned as a plain list of
non‑negative floats.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, stays on the simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.ones(n, dtype=np.float64) / n


# ----------------------------------------------------------------------
# Analytic seed families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a modest set of diverse normalised start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    for a in (0.3, 0.5, 0.7):
        for b in (0.3, 0.5, 0.7):
            seeds.append(_beta_shape(n, a, b))

    rng = np.random.default_rng(2023)
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max (same spirit as the reference)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑softmax parametrisation, using a smooth‑max surrogate."""
    v = np.log(seed + 1e-30)          # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    # start smoothing (a low “temperature”) and increase slowly
    beta = 0.5
    beta_mul = 1.025

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0 or it == 1:
                print(f"[SubGrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (cheap)
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """Random pair‑wise mass transfer, keeping improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.01  # relative step size

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                if it % 5000 == 0 or it == 1:
                    print(f"[Hill] n={n:5d} it={it:6d} best={val:.6f}")

        if it % 1000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Low‑dimensional Differential Evolution (global search)
# ----------------------------------------------------------------------


def _eval_param(p: np.ndarray, knots: np.ndarray, n: int) -> tuple[float, np.ndarray]:
    """
    Interpolate a low‑dimensional simplex vector `p` (sum=1)
    to a full length‑`n` sequence, normalise, and evaluate the exact objective.
    Returns (objective, full_sequence).
    """
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    if s == 0.0:
        a = _uniform(n)
        s = a.sum()
    a = a / s
    conv = _conv_fft(a, a)
    val = 2.0 * n * np.max(conv)      # sum(a)=1 ⇒ objective = 2·n·max(conv)
    return val, a


def _de_optimize(n: int,
                 deadline: float,
                 popsize: int = 30,
                 K: int | None = None,
                 max_gens: int = 500) -> tuple[np.ndarray, float]:
    """
    Differential Evolution on a piece‑wise‑linear representation.
    The candidate population lives on the simplex.
    """
    rng = np.random.default_rng()

    if K is None:
        K = max(20, int(np.sqrt(n)))      # ≈ √n knots
    knots = np.linspace(0, n - 1, K)

    # initialise population with Dirichlet draws
    pop = [rng.dirichlet(np.ones(K)) for _ in range(popsize)]

    pop_vals = np.empty(popsize, dtype=np.float64)
    pop_seq = [None] * popsize
    for i in range(popsize):
        val, seq = _eval_param(pop[i], knots, n)
        pop_vals[i] = val
        pop_seq[i] = seq

    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_seq = pop_seq[best_idx].copy()
    best_p = pop[best_idx].copy()

    F = 0.8
    CR = 0.9
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a_vec = pop[a_idx]
            b_vec = pop[b_idx]
            c_vec = pop[c_idx]

            donor = a_vec + F * (b_vec - c_vec)
            donor = np.maximum(donor, 0.0)
            ds = donor.sum()
            if ds == 0.0:
                donor = _uniform(K)
            else:
                donor = donor / ds

            # binomial crossover
            trial = pop[i].copy()
            jrand = rng.integers(K)
            for j in range(K):
                if rng.random() < CR or j == jrand:
                    trial[j] = donor[j]

            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts == 0.0:
                trial = _uniform(K)
            else:
                trial = trial / ts

            val_trial, seq_trial = _eval_param(trial, knots, n)

            if val_trial < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val_trial
                pop_seq[i] = seq_trial
                if val_trial < best_val:
                    best_val = val_trial
                    best_seq = seq_trial.copy()
                    best_p = trial.copy()

        if gen % 20 == 0:
            print(f"[DE]  n={n:5d} gen={gen:3d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (optional, requires SciPy)
# ----------------------------------------------------------------------


def _lp_refine(start_seq: np.ndarray,
               n: int,
               deadline: float,
               max_iters: int = 5) -> tuple[np.ndarray, float]:
    """
    Iteratively solve a linear program:
        maximise   Σ b_j
        subject to a * b ≤ M   (component‑wise),
                   b ≥ 0.
    After each LP we mix the result back toward the current sequence.
    """
    try:
        from scipy.optimize import linprog
        import scipy.sparse as sp
    except Exception as exc:
        print("[LP] SciPy not available – skipping LP refinement.", exc)
        return start_seq, 2.0 * n * np.max(_conv_fft(start_seq, start_seq))

    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(max_iters):
        if time.time() > deadline:
            break

        M = np.max(_conv_fft(a, a))

        # Build the convolution matrix A (size (2n‑1) × n) as a sparse diagonal stack.
        # A[i, j] = a[i‑j]  ⇔ row – col = offset = i‑j.
        offsets = np.arange(n, dtype=int)                     # 0 … n‑1
        A = sp.diags(a, offsets=offsets,
                     shape=(2 * n - 1, n), format='csc')

        # Linear program: maximise sum(b)  ⇔ minimise –1ᵀb
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': max(deadline - time.time(), 0)})

        if not res.success:
            print(f"[LP] iteration {it}: linprog failed – {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        b_norm = b / sum_b                       # re‑scale to sum = 1

        # Try a few convex mixes (t ∈ [0,1]) and keep the best.
        candidates = [(b_norm, 0.0), (a, 1.0)]
        # add a few intermediate t values
        for t in (0.2, 0.5, 0.8):
            candidates.append(((1 - t) * a + t * b_norm, t))

        for cand, t in candidates:
            val = 2.0 * n * np.max(_conv_fft(cand, cand))
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                a = cand.copy()
                print(f"[LP] n={n:5d} iter={it} t={t:.2f} best={val:.6f}")

        # if none of the mixes improved, keep current a unchanged and continue.
    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    A coarse DE run (n≈2000) is followed by up‑sampling, Adam,
    long sub‑gradient descent, optional LP refinement and a final hill‑climb.
    The best sequence (as a Python list) is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Coarse global optimisation (n≈2000)
    # ------------------------------------------------------------------
    n_coarse = 2000
    print(f"\n=== Coarse DE (n={n_coarse}) ===")
    seeds = _make_seeds(n_coarse, None)

    best_seq = None
    best_val = float("inf")
    for s in seeds:
        val = 2.0 * n_coarse * np.max(_conv_fft(s, s))
        if val < best_val:
            best_val = val
            best_seq = s.copy()
            print(f"[seed] n={n_coarse:5d} new best={val:.6f}")

    seq_de, val_de = _de_optimize(n_coarse, deadline,
                                  popsize=30, K=None, max_gens=500)
    if val_de < best_val:
        best_val = val_de
        best_seq = seq_de.copy()
        print(f"[DE]   n={n_coarse:5d} new global best={val_de:.6f}")

    # ------------------------------------------------------------------
    # 2. Upsample to the final length
    # ------------------------------------------------------------------
    n_final = 4000
    print(f"\n=== Upsampling to n={n_final} ===")
    if best_seq.shape[0] != n_final:
        # linear interpolation on the integer grid
        x_old = np.arange(best_seq.shape[0], dtype=np.float64)
        x_new = np.linspace(0, best_seq.shape[0] - 1, n_final)
        best_seq = np.interp(x_new, x_old, best_seq)
        best_seq = best_seq / best_seq.sum()
    cur_val = 2.0 * n_final * np.max(_conv_fft(best_seq, best_seq))
    print(f"[up]   initial objective ≈ {cur_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam refinement (large iteration budget)
    # ------------------------------------------------------------------
    seq_adam, val_adam = _adam_optimize(best_seq, n_final,
                                        deadline, max_iters=20000)
    if seq_adam is not None and val_adam < cur_val:
        best_seq = seq_adam.copy()
        cur_val = val_adam
        print(f"[Adam] n={n_final:5d} improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4. Exact sub‑gradient descent (many iterations)
    # ------------------------------------------------------------------
    seq_sg, val_sg = _subgradient_refine(best_seq, n_final,
                                         deadline, max_iters=200000)
    if val_sg < cur_val:
        best_seq = seq_sg.copy()
        cur_val = val_sg
        print(f"[SubGrad] n={n_final:5d} improved to {val_sg:.6f}")

    # ------------------------------------------------------------------
    # 5. (Optional) Linear‑programming refinement
    # ------------------------------------------------------------------
    seq_lp, val_lp = _lp_refine(best_seq, n_final, deadline, max_iters=5)
    if val_lp < cur_val:
        best_seq = seq_lp.copy()
        cur_val = val_lp
        print(f"[LP]   n={n_final:5d} improved to {val_lp:.6f}")

    # ------------------------------------------------------------------
    # 6. Final cheap hill‑climb
    # ------------------------------------------------------------------
    seq_hc, val_hc = _hill_climb(best_seq, n_final,
                                 deadline, max_iters=50000)
    if val_hc < cur_val:
        best_seq = seq_hc.copy()
        cur_val = val_hc
        print(f"[Hill] n={n_final:5d} improved to {val_hc:.6f}")

    # ------------------------------------------------------------------
    # 7. Store the result for possible warm‑starts in later runs
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = best_seq.tolist()

    # ------------------------------------------------------------------
    # 8. Verify with the provided evaluate_sequence (best effort)
    # ------------------------------------------------------------------
    try:
        check = evaluate_sequence(best_seq.tolist())
        print("[verify] evaluate_sequence returned", check)
    except Exception as exc:
        print("[verify] evaluation failed:", exc)

    print(f"\n=== Finished ===\nBest objective ≈ {cur_val:.6f} "
          f"(length {len(best_seq)})")
    return best_seq.tolist()
