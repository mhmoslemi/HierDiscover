def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = int(kwargs.get('n', 2000))
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())
    
    best_a = None
    best_score = float('inf')
    
    # Phase 0: parametric families
    # arcsine, uniform, Beta variations, trimmed uniform
    candidates = []
    # arcsine
    candidates.append(arcsine_distribution(n))
    # uniform
    candidates.append(np.full(n, 1.0/n))
    # Beta param sweep
    for _ in range(200):
        alpha = 10**rng.uniform(-2, 1)   # range [0.01, 10]
        beta = 10**rng.uniform(-2, 1)
        candidates.append(beta_distribution(n, alpha, beta))
    # trimmed uniform near ends
    for _ in range(50):
        a = rng.uniform(0.02, 0.5)
        cut = int(np.round(a * n))
        if cut == 0: continue
        vec = np.zeros(n)
        # allocate mass equally on both ends
        mass = 0.5 / cut
        vec[:cut] = mass
        vec[-cut:] = mass
        candidates.append(vec / vec.sum())
    
    # evaluate each candidate
    for idx, a in enumerate(candidates):
        a = np.maximum(a, 0)
        if a.sum() == 0: continue
        a = a / a.sum()
        conv_max = max_convolution(a, fft_len_conv)
        score = 2 * n * conv_max
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[cand {idx}] score {score:.6f}")
    
    # Phase 1: smooth gradient descent (Adam) with softmax weights
    # hyperparameters
    max_iter = 500000   # large bound but time limited
    beta_start = 0.0
    # compute approximate convex spread for choosing beta_end
    beta_end = 20000.0
    lr0 = 0.2
    eps = 1e-8
    m = np.zeros_like(best_a)
    v = np.zeros_like(best_a)
    beta1 = 0.9
    beta2 = 0.999
    it = 0
    last_improve_time = time.time()
    a = best_a.copy()
    while time.time() < deadline - 30:  # keep some time for SA
        # compute convolution
        conv_vals = np.real(np.fft.ifft(np.fft.fft(a, fft_len_conv) *
                                        np.fft.fft(a, fft_len_conv)))[:2 * n - 1]
        conv_max = conv_vals.max()
        # compute beta schedule linear
        t_frac = it / max_iter
        beta = beta_start + (beta_end - beta_start) * t_frac
        # softmax weights
        w = np.exp(beta * (conv_vals - conv_max))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / len(w))
        else:
            w = w / w_sum
        # compute gradient via convolution with w
        grad_full = np.real(np.fft.ifft(np.fft.fft(a, fft_len_grad) *
                                        np.fft.fft(w, fft_len_grad)))[:n]
        grad = 2.0 * grad_full
        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr0 / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a - step * m_hat)
        # evaluate objective with actual max (not soft)
        conv_vals_new = np.real(np.fft.ifft(np.fft.fft(a_new, fft_len_conv) *
                                            np.fft.fft(a_new, fft_len_conv)))[:2 * n - 1]
        new_score = 2 * n * conv_vals_new.max()
        if new_score < best_score - 1e-9:
            best_score = new_score
            best_a = a_new.copy()
            last_improve_time = time.time()
            print(f"[grad {it}] new best {best_score:.6f}")
        # accept move if improves else revert (but we keep a_new perhaps). Analytical: we can accept only if improving or with probability? Simulated annealing not needed now.
        if new_score <= best_score + 1e-12:
            a = a_new
        else:
            # optionally keep a if not improving? For Adam we might still move, but to avoid worse we revert
            a = a_new  # still move, rely on Adam adaptation; but we might want to revert.
            # maybe better to accept anyway.
            a = a_new
        # if stagnated for 20 sec, try small random perturbation
        if time.time() - last_improve_time > 20.0:
            noise = rng.normal(scale=0.01, size=n)
            a = proj_simplex(best_a + noise)
            last_improve_time = time.time()
            print(f"[grad {it}] restart")
        # termination after big number of iter but time limited
        if time.time() > deadline - 30:
            break
    
    # Phase 2: local stochastic refinement (simulated annealing)
    remaining = deadline - time.time()
    if remaining > 5.0:
        a = best_a.copy()
        best_conv = best_score / (2.0 * n)
        T_start = best_score * 0.005
        T_end = best_score * 1e-6
        sa_start = time.time()
        sa_it = 0
        while time.time() < deadline:
            elapsed = time.time() - sa_start
            frac = elapsed / remaining
            T = T_start * (1 - frac) + T_end * frac
            # choose i and j randomly
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_move = min(a[i], 0.03 * a[i] + 1e-12)
            if max_move <= 0:

            a_new[i] -= delta
            a_new[j] += delta
            # evaluate
            conv_new = max_convolution(a_new, fft_len_conv)
            new_score = 2 * n * conv_new
            if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):

                if new_score < best_score:

                    best_a = a_new.copy()
                    print(f"[SA {sa_it}] best {best_score:.6f} t={time.time() - start:.1f}s")
            sa_it += 1
    
    # final scaling and return
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0:

    else:
        best_a = best_a / total
    elapsed = time.time() - start
    print(f"[done] total {elapsed:.1f}s best {best_score:.6f} n={n}")
    return best_a.tolist()

"""propose_candidate
A three‑phase optimizer for the step‑function evaluation problem.

1. **Exploratory families** – fast evaluation of a large pool of
   candidate vectors (arcsine, Beta, mixtures, edge‑uniform blocks).
   The best of these seeds the later phases.

2. **Smoothed projected Adam** – we minimise the *soft‑max* of the
   autocorrelation 2 · ∑ₛ wₛ·(a*a)ₛ, where the weights
   ``wₛ = exp(β·(convₛ – max(conv)))`` concentrate on the most
   critical shift as β grows.  The gradient of this smooth objective is
   a correlation that can be evaluated with a single FFT.  After each
   Adam step the iterate is projected back onto the probability simplex.

3. **Local stochastic refinement** – a simulated‑annealing style
   mass‑transfer procedure that only touches two coordinates at a time.
   It explores the exact non‑smooth objective for a final fine‑tuning.

The routine respects the 1000 s wall‑clock budget, prints progress,
and returns the best vector (as a list of floats) it discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    return np.maximum(v - theta, 0.0)


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Linear convolution a*a (full length) and return the maximum entry."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv.max())


def arcsine_distribution(n: int) -> np.ndarray:
    """Discretisation of the arcsine density on [0,1].  Returns a probability vector."""
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric or asymmetric Beta(α,β) density sampled at bin centres."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def trimmed_edge_distribution(n: int, fraction: float) -> np.ndarray:
    """Uniform mass on the two edges, each of length `fraction·n`."""
    cut = max(1, int(round(fraction * n)))
    vec = np.zeros(n, dtype=float)
    mass = 0.5 / cut
    vec[:cut] = mass
    vec[-cut:] = mass
    return vec / vec.sum()


def correlation_gradient(p: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of  Σₛ wₛ·(p*p)ₛ   w.r.t. p.
    It equals 2·corr(p,w) (correlation of p and w) and can be computed
    via an FFT as 2·conv(w, rev(p))[n‑1 : n‑1+n].
    """
    p_rev = p[::-1]                     # reverse of p
    # FFT‑based convolution w * p_rev
    conv = np.real(
        np.fft.ifft(
            np.fft.fft(w, fft_len) * np.fft.fft(p_rev, fft_len)
        )
    )
    n = p.size
    # indices n‑1 … n‑1+n‑1 give the correlation values
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (probability vector) that
    minimises `evaluate_sequence`.  Returns the best vector found as a
    Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # problem size -------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    # FFT lengths
    fft_len_conv = 1 << ((2 * n - 1).bit_length())          # for a*a
    fft_len_corr = 1 << ((3 * n - 2).bit_length())          # for correlation with w

    # ------------------------------------------------------------------
    # 0) Build a large pool of cheap candidates
    # ------------------------------------------------------------------
    candidates = []

    # reuse a previously discovered construction if available
    if "height_sequence_1" in globals():
        h = np.asarray(globals()["height_sequence_1"], dtype=float)
        if h.ndim == 1:
            if h.size != n:
                # linear resampling onto the new grid
                x_old = np.linspace(0, 1, h.size)
                x_new = np.linspace(0, 1, n)
                h = np.interp(x_new, x_old, h)
            h = np.maximum(h, 0.0)
            if h.sum() > 0.0:
                candidates.append(h / h.sum())

    # basic families
    candidates.append(arcsine_distribution(n))
    candidates.append(np.full(n, 1.0 / n))                         # uniform

    # random Beta families (both symmetric and asymmetric)
    for _ in range(200):
        alpha = 10 ** rng.uniform(-2.0, 1.0)      # α ∈ [0.01, 10]
        beta = 10 ** rng.uniform(-2.0, 1.0)      # β ∈ [0.01, 10]
        candidates.append(beta_distribution(n, alpha, beta))

    # trimmed edge distributions
    for _ in range(50):
        frac = rng.uniform(0.02, 0.45)           # fraction of the interval taken at each edge
        candidates.append(trimmed_edge_distribution(n, frac))

    # mixtures of arcsine with uniform
    arcsine = candidates[1]
    uniform = candidates[2]
    for lam in np.linspace(0.0, 1.0, 11):
        mix = lam * arcsine + (1.0 - lam) * uniform
        mix /= mix.sum()
        candidates.append(mix)

    # ------------------------------------------------------------------
    # 1) Evaluate cheap candidates, keep the best so far
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    for idx, a in enumerate(candidates):
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            continue
        a = a / a.sum()
        mx = max_convolution(a, fft_len_conv)
        score = 2.0 * n * mx
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[init {idx:03d}] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Smoothed projected Adam optimisation
    # ------------------------------------------------------------------
    # time budgeting: keep ~80 % for this phase
    phase1_deadline = start_time + 0.80 * budget_s

    a = best_a.copy()
    # Adam accumulator vectors
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr = 0.12                     # base learning rate for Adam

    # schedule for the soft‑max concentration parameter β
    beta_start = 1.0
    beta_end = 5000.0
    max_iter = 300_000           # large upper bound – actual loop stops by time

    it = 0
    last_improve = time.time()

    while time.time() < phase1_deadline:
        it += 1

        # ----- convolution a*a (full) -----
        conv_full = np.real(
            np.fft.ifft(
                np.fft.fft(a, fft_len_conv) * np.fft.fft(a, fft_len_conv)
            )
        )[: 2 * n - 1]
        conv_max = conv_full.max()

        # ----- soft‑max weights -----
        t = min(1.0, it / max_iter)
        beta = beta_start + (beta_end - beta_start) * t
        # soft‑max on the shifted values to keep numbers O(1)
        w_raw = np.exp(beta * (conv_full - conv_max))
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # ----- gradient of the smoothed objective -----
        grad = correlation_gradient(a, w, fft_len_corr)

        # ----- Adam update + simplex projection -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a - step * m_hat)

        # ----- evaluate exact (non‑smoothed) objective -----
        conv_new_full = np.real(
            np.fft.ifft(
                np.fft.fft(a_new, fft_len_conv) * np.fft.fft(a_new, fft_len_conv)
            )
        )[: 2 * n - 1]
        new_score = 2.0 * n * conv_new_full.max()

        if new_score < best_score - 1e-9:
            best_score = new_score
            best_a = a_new.copy()
            last_improve = time.time()
            print(f"[Adam {it:07d}] new best = {best_score:.6f}")

        # Accept the move (Adam is robust, we keep the new iterate)
        a = a_new

        # occasional random restart if we have been stagnant for >30 s
        if time.time() - last_improve > 30.0:
            noise = rng.normal(scale=0.02, size=n)
            a = proj_simplex(best_a + noise)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[Adam {it:07d}] restart after stagnation")

        # safety break on time
        if time.time() > phase1_deadline:
            break

    # ------------------------------------------------------------------
    # 3) Local stochastic refinement (simulated annealing)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a = best_a.copy()
        best_conv = best_score / (2.0 * n)

        # linear cooling schedule
        T_start = best_score * 0.004
        T_end = best_score * 1e-6
        sa_start = time.time()
        sa_it = 0

        while time.time() < deadline:
            elapsed = time.time() - sa_start
            frac = min(1.0, elapsed / remaining)
            T = T_start * (1.0 - frac) + T_end * frac

            # pick two distinct indices
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            # move up to 4 % of the mass at i to j
            max_move = min(a[i], 0.04 * a[i] + 1e-12)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # exact evaluation of the proposal
            new_conv = max_convolution(a_new, fft_len_conv)
            new_score = 2.0 * n * new_conv

            # Metropolis acceptance
            if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    print(f"[SA {sa_it:07d}] best = {best_score:.6f} (t={time.time() - start:.1f}s)")
            sa_it += 1

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    # optional sanity check – evaluate via the public function
    try:
        # evaluate_sequence is assumed to be present in the environment
        exact_score = evaluate_sequence(best_a.tolist())
    except Exception:
        exact_score = None

    elapsed = time.time() - start_time
    print(f"[done] elapsed = {elapsed:.1f}s, best score = {best_score:.6f}"
          f"{' (exact)' if exact_score is not None else ''}")

    return best_a.tolist()
