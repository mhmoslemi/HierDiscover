"""
Propose a non‑negative step‑function (list of heights) that minimises the
`evaluate_sequence` objective.

Algorithm overview
------------------
1. **Generate a pool of simple parametric candidates** – arcsine,
   various symmetric / asymmetric Beta densities, uniform, and a fine
   mixture of arcsine & uniform.  The best of these is taken as the
   starting point.

2. **Smoothed Adam optimisation** – minimise a smooth approximation of
   `max(conv(a,a))`.  The max is replaced by a temperature‑controlled
   soft‑max (log‑sum‑exp) which yields a differentiable surrogate.
   Gradients are computed analytically via FFTs and the iterate is
   projected onto the probability simplex using a Euclidean projection.
   An Adam‑style adaptive step‑size together with a linear cooling of the
   temperature (τ → 0) drives the solution towards the true
   `max(conv)`.  The best vector seen during this phase is remembered.

3. **Stochastic refinement** – simulated‑annealing style random mass
   transfers between two coordinates, accepted if they improve the exact
   score or with a Metropolis probability that slowly cools.  This
   eliminates any remaining small‑scale artefacts.

The whole routine respects the 1000 s budget, prints intermediate scores,
and finally returns the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """
    Linear convolution a * a using FFT of length `fft_len`,
    and returns its maximum entry.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv.max())


def evaluate_vec(a: np.ndarray, fft_len: int) -> float:
    """Exact score for a normalised vector (sum == 1)."""
    return 2.0 * a.size * max_convolution(a, fft_len)


def arcsine_distribution(n: int) -> np.ndarray:
    """Exact discretisation of the arcsine density on [0,1]."""
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density evaluated at the bin centres and normalised."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Temperature‑controlled soft‑max of the convolution values.
    Returns a probability vector on the same support.
    """
    maxc = conv.max()
    # shift for numerical stability
    ex = np.exp((conv - maxc) / tau)
    s = ex.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return ex / s


def gradient_softmax(a: np.ndarray,
                     conv: np.ndarray,
                     tau: float,
                     grad_len: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate w.r.t. a.
    Implements g = 2 * (a * w_rev) where w is the soft‑max of conv.
    """
    n = a.size
    w = softmax_weights(conv, tau)
    w_rev = w[::-1]                     # reverse for the cross‑correlation
    fa = np.fft.fft(a, grad_len)
    fw = np.fft.fft(w_rev, grad_len)
    conv_aw = np.real(np.fft.ifft(fa * fw))  # length = 3n‑2 (zero‑padded)
    # slice indices n‑1 … 2n‑2 (inclusive) and reverse to obtain g_k
    slice_part = conv_aw[n - 1 : 2 * n - 1][::-1]
    return 2.0 * slice_part


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative vector) that minimises
    `evaluate_sequence`.  Three phases are performed:
        1) fast parametric sweep,
        2) smoothed Adam optimisation with a temperature‑controlled soft‑max,
        3) local stochastic refinement (simulated annealing).
    The routine respects the allotted CPU time and prints progress.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    overall_deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))
    # FFT length for a*a (linear convolution)
    conv_len = 1 << ((2 * n - 1).bit_length())
    # FFT length for the gradient convolution (size 3n‑2)
    grad_len = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # 0) Build a modest pool of elementary candidates
    # ------------------------------------------------------------------
    candidates = []

    # reuse a globally stored construction if it exists
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.size == n and a0.sum() > 0.0:
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # basic families
    candidates.append(arcsine_distribution(n))                     # arcsine
    candidates.append(np.full(n, 1.0 / n))                         # uniform

    # symmetric Beta family – broaden the search range
    for alpha in np.logspace(-2, 0.5, 12):   # 0.01 … ≈3.16
        candidates.append(beta_distribution(n, alpha, alpha))

    # asymmetric Beta – sweep a few skewed shapes
    for a in np.logspace(-2, 0.5, 6):
        for b in np.logspace(-2, 0.5, 6):
            candidates.append(beta_distribution(n, a, b))

    # mixtures arcsine ↔ uniform (finer grid)
    arcsine = candidates[0]   # arcsine already computed
    uniform = candidates[1]
    for lam in np.linspace(0.0, 1.0, 21):      # 0,0.05,…,1
        mix = lam * arcsine + (1.0 - lam) * uniform
        mix /= mix.sum()
        candidates.append(mix)

    # a few random Dirichlet draws (α≈0.5 gives slightly spiky profiles)
    for _ in range(12):
        rand_vec = rng.dirichlet(np.full(n, 0.5))
        candidates.append(rand_vec)

    # ------------------------------------------------------------------
    # 1) Evaluate cheap candidates and keep the best
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    for idx, a in enumerate(candidates):
        sc = evaluate_vec(a, conv_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[init {idx:02d}] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Smoothed Adam optimisation (≈70 % of the total budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.70 * budget_s
    a = best_a.copy()
    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    last_improve = time.time()

    # temperature (τ) schedule – start relatively high, anneal linearly
    # the initial value is chosen as a fraction of the current max conv.
    # the final value is tiny (≈1e‑6) to approach the true max.
    conv_initial = max_convolution(a, conv_len)
    tau_start = max(conv_initial * 0.5, 1e-5)      # never start at 0
    tau_end = max(conv_initial * 1e-5, 1e-7)

    while time.time() < phase1_deadline:
        it += 1
        # --- convolution a*a -------------------------------------------------
        fa = np.fft.fft(a, grad_len)
        conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        conv_max = conv_full.max()
        cur_score = 2.0 * n * conv_max

        # keep best seen
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam {it:07d}] new best = {best_score:.6f}")

        # --- temperature (τ) --------------------------------------------------
        elapsed_phase = time.time() - (start_time + 0.0)
        frac_phase = min(1.0, (time.time() - start_time) /
                         (phase1_deadline - start_time))
        tau = tau_start * (1.0 - frac_phase) + tau_end * frac_phase

        # --- gradient of smoothed max ----------------------------------------
        g = gradient_softmax(a, conv_full, tau, grad_len)

        # --- Adam update ------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # simple learning‑rate schedule (drops a bit over time)
        lr = 0.07 * (1.0 - frac_phase) + 0.015 * frac_phase
        step = m_hat / (np.sqrt(v_hat) + eps)

        a = proj_simplex(a - lr * step)

        # occasional restart if stagnation lasts >30 s
        if time.time() - last_improve > 30.0:
            noise = rng.normal(scale=0.02, size=n)
            a = proj_simplex(best_a + noise)
            last_improve = time.time()
            print(f"[Adam {it:07d}] restart after stagnation")

    # ------------------------------------------------------------------
    # 3) Local stochastic refinement (simulated annealing)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 5.0:
        a = best_a.copy()
        best_conv_max = best_score / (2.0 * n)

        # linear cooling schedule for the Metropolis temperature
        T_start = best_score * 0.004
        T_end = best_score * 1e-6
        sa_start = time.time()
        sa_it = 0

        while time.time() < overall_deadline:
            elapsed = time.time() - sa_start
            frac = min(1.0, elapsed / remaining)
            T = T_start * (1.0 - frac) + T_end * frac

            # propose a mass transfer between two distinct indices
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            max_move = min(a[i], 0.05 * a[i] + 1e-12)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # exact evaluation of the proposal
            new_conv_max = max_convolution(a_new, conv_len)
            new_score = 2.0 * n * new_conv_max

            # Metropolis acceptance
            if new_score < best_score or rng.random() < np.exp(
                    -(new_score - best_score) / (T + 1e-12)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    print(f"[SA {sa_it:07d}] new best = {best_score:.6f} "
                          f"(t={time.time() - start_time:.1f}s)")
            sa_it += 1

    # ------------------------------------------------------------------
    # Final normalisation and output
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    elapsed_total = time.time() - start_time
    print(f"[done] total time = {elapsed_total:.1f}s, best score = "
          f"{best_score:.6f}, length = {n}")
    return best_a.tolist()
