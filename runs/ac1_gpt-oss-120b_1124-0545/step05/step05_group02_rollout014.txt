"""
Propose a step‑function sequence that minimises

    2·n·max(conv(a,a)) / (∑a)²

The algorithm works as follows (all steps respect the global time budget):

1. **Seed generation** – a small diverse portfolio of analytic shapes
   (uniform, power‑law, geometric, sine‑window, beta‑like).  If a previous
   construction (`height_sequence_1`) is available it is also used.

2. **Linear‑programming refinement** – starting from a seed `f`,
   solve the linear programme

        maximise   Σ g_j
        subject to  f * g  ≤  M    (component‑wise)
                    g ≥ 0

   where `M = max(f*f)`.  The feasible point `g` with the largest total mass
   pushes the mass of `f` away from the most‑loaded antidiagonals.  The new
   candidate is a convex combination `f ← (1‑t)·f + t·g` (with a decreasing
   step‑size `t`).  Repeating this step (≤ 30 iterations) reduces the
   maximum convolution value.

3. **Exact sub‑gradient mirror descent** – after the LP phase a few
   sub‑gradient (exponentiated‑gradient) steps are taken to polish the
   solution.

The whole routine stops as soon as the 1000 s budget is exhausted and returns
the best sequence found (as a plain ``list`` of ``float``).  All numbers are
non‑negative, the sequence is automatically normalised to sum = 1, and the
objective is evaluated with the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, objective, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear full convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) assuming ∑seq == 1 (scale‑invariant)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (already normalised to sum == 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    a /= a.sum()
    return a


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a /= a.sum()
    return a


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    a /= a.sum()
    return a


def _make_seeds(
    n: int, previous: np.ndarray | None, rng: np.random.Generator
) -> list[np.ndarray]:
    """Collect a modest set of normalised seeds."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for p in (0.3, 0.5, 0.7, 1.0, 1.5):
        seeds.append(_power(n, p))

    for r in (0.6, 0.8, 0.9):
        seeds.append(_geom(n, r))

    for a, b in ((0.4, 0.4), (0.6, 0.6), (0.4, 0.8)):
        seeds.append(_beta_shape(n, a, b))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # a few random Dirichlet draws (sparse & dense)
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(rng.dirichlet(np.full(n, conc)))
    return seeds


# ----------------------------------------------------------------------
# Linear‑programming based refinement
# ----------------------------------------------------------------------


def _lp_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Perform a few iterations of the LP update described in the problem statement.
    Returns the best sequence found during the refinement.
    """
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = np.max(conv)

        # build Toeplitz matrix A such that (A @ g) = a * g  (full convolution)
        # A has shape (2*n-1, n); diagonal d (0≤d<n) contains a[d]
        offsets = np.arange(n, dtype=int)
        data = [np.full(n, a[d], dtype=np.float64) for d in offsets]
        A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")

        # linear program: maximise sum(g)  -> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M * (1.0 + 1e-12), dtype=np.float64)

        res = opt.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=(0, None),
            method="highs",
        )
        if not res.success:
            # LP failed – abort the refinement
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 1.0 + 1e-12:
            # cannot increase total mass; stop early
            break

        # normalise the new direction
        g = g0 / sum_g0

        # decreasing convex‑combination step
        t = 0.5 / np.sqrt(it + 2.0)
        a = (1.0 - t) * a + t * g
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LP] n={n:5d} it={it:3d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient mirror descent (already present in the previous code)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 20000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror‑descent on max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active antidiagonal
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent (exponentiated‑gradient) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)  # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val

# ----------------------------------------------------------------------
