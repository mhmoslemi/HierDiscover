"""
Hybrid optimisation for the max‑auto‑convolution problem.

* Global search – a Differential Evolution (DE) on a *symmetric* piece‑wise
  linear representation.  The optimisation variables are the values of the
  function at a modest number K≈2·√n of knots; the full step‑function of length n
  is obtained by linear interpolation on the first half of the domain and then
  mirroring.  The DE works on the simplex (log‑softmax parametrisation).

* Local polishing – three deterministic refinements:
    – Adam on a smooth (log‑sum‑exp) surrogate of the max‑convolution;
    – exact sub‑gradient mirror‑descent on the true max;
    – stochastic hill‑climb that transfers a small amount of mass between
      random positions.

All stages respect a single global deadline (default 1000 s).  The routine
returns the best sequence found so far as a plain list of floats.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Analytic seed families (ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Gather a variety of analytically‑derived or random start vectors."""
    seeds: list[np.ndarray] = []

    # basic deterministic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric tails
    for c in np.linspace(0.2, 4.8, 13):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law decays
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    # symmetric geometric
    for c in np.linspace(0.2, 4.0, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # beta‑shapes (symmetric + asymmetric)
    for a in (0.2, 0.4, 0.6, 1.0, 1.5, 2.0):
        for b in (0.2, 0.4, 0.6, 1.0, 1.5, 2.0):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws
    rng = np.random.default_rng(2023)
    for conc in (0.3, 0.8, 2.0, 5.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random softmax (logits)
    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # previous best (if supplied)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Objective (the quantity we minimise)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) for a sequence that has been ℓ₁‑normalised.
    """
    seq = np.asarray(seq, dtype=np.float64)
    s = seq.sum()
    if s == 0.0:
        return float("inf")
    seq = seq / s
    conv = _conv_fft(seq, seq)
    return 2.0 * len(seq) * np.max(conv)


# ----------------------------------------------------------------------
# Differential Evolution (symmetric piece‑wise‑linear)
# ----------------------------------------------------------------------


def _eval_de_param(p: np.ndarray,
                   knots: np.ndarray,
                   half_n: int,
                   n: int) -> tuple[float, np.ndarray]:
    """
    Interpolate a simplex vector `p` (size K, sum=1) to a symmetric full
    sequence of length `n`, normalise, and evaluate the exact objective.
    """
    # linear interpolation on the half‑grid
    half_seq = np.interp(np.arange(half_n, dtype=np.float64), knots, p)

    # mirror to obtain a symmetric full sequence
    if n % 2 == 0:
        full_seq = np.concatenate([half_seq, half_seq[::-1]])
    else:
        full_seq = np.concatenate([half_seq, half_seq[-2::-1]])

    total = full_seq.sum()
    if total == 0.0:
        full_seq = _uniform(n)
    else:
        full_seq = full_seq / total

    val = 2.0 * n * np.max(_conv_fft(full_seq, full_seq))
    return val, full_seq


def _de_optimize_sym(n: int,
                     deadline: float,
                     popsize: int = 40,
                     K: int | None = None,
                     max_gens: int = 800) -> tuple[np.ndarray | None, float]:
    """
    Differential Evolution on a symmetric piece‑wise‑linear parametrisation.
    """
    rng = np.random.default_rng()

    if K is None:
        K = max(30, int(2 * np.sqrt(n)))      # ≈ 2·√n knots
    half_n = (n + 1) // 2
    knots = np.linspace(0, half_n - 1, K)

    # initialise population (Dirichlet draws on the simplex)
    pop = [rng.dirichlet(np.ones(K)) for _ in range(popsize)]

    pop_vals = np.empty(popsize, dtype=np.float64)
    pop_seq = [None] * popsize
    for i in range(popsize):
        val, seq = _eval_de_param(pop[i], knots, half_n, n)
        pop_vals[i] = val
        pop_seq[i] = seq

    best_idx = int(np.argmin(pop_vals))
    best_val = float(pop_vals[best_idx])
    best_seq = pop_seq[best_idx].copy()
    best_p = pop[best_idx].copy()

    F = 0.8
    CR = 0.9
    gen = 0
    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a_vec = pop[a_idx]
            b_vec = pop[b_idx]
            c_vec = pop[c_idx]

            donor = a_vec + F * (b_vec - c_vec)
            donor = np.maximum(donor, 0.0)
            ds = donor.sum()
            donor = _uniform(K) if ds == 0.0 else donor / ds

            # binomial crossover
            trial = pop[i].copy()
            jrand = rng.integers(K)
            for j in range(K):
                if rng.random() < CR or j == jrand:
                    trial[j] = donor[j]

            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            trial = _uniform(K) if ts == 0.0 else trial / ts

            val_trial, seq_trial = _eval_de_param(trial, knots, half_n, n)

            if val_trial < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val_trial
                pop_seq[i] = seq_trial
                if val_trial < best_val:
                    best_val = val_trial
                    best_seq = seq_trial.copy()
                    best_p = trial.copy()

        if gen % 20 == 0:
            print(f"[DE] n={n:5d} gen={gen:3d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 30000) -> tuple[np.ndarray | None, float]:
    """
    Adam on the log‑softmax parametrisation using a smooth‑max surrogate.
    """
    v = np.log(seed + 1e-30)          # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0            # initial smoothing coefficient
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(v)                     # stays on simplex, sum=1
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={t:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient w.r.t. a of Σ w_k·conv_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror–descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent (exponentiated‑gradient/ mirror descent)
    on the true max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)               # for numerical stability
        a = np.exp(log_a)
        s = a.sum()
        a = a / s if s != 0.0 else _uniform(n)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 60000) -> tuple[np.ndarray, float]:
    """
    Random pair‑wise mass transfer, keeping only improving moves.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02  # initial relative step size

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 1500 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val
