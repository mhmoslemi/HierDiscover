"""
Propose a non‑negative step function that (approximately) minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines three ideas:
1. **Broad analytic seeding** – a quick systematic grid over the
   “product” family and a large random search over several analytic
   families (product, sum‑power, exponential, cosine).  The best
   candidate from this cheap stage becomes the starting point.
2. **Smoothed‑max Adam optimisation** – minimise a smooth
   approximation of the max‑convolution via Adam on the simplex.
   The temperature (soft‑max α) is raised from 10 to 200 while the
   learning‑rate decays slowly.  Improvements are recorded whenever the
   exact objective is better.
3. **Final stochastic refinement** – Gaussian perturbations with a
   decreasing variance are accepted if they improve the true score.

The whole routine respects the 1000 s time‑budget, works for any
requested length `n` (default 4000) and returns a list of non‑negative
floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (linear, not circular)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (zero‑padded) using FFT.  Length = 2·len(a)-1."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two 1‑D arrays using FFT."""
    size = _next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: len(a) + len(b) - 1]


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σw = 1, w ≥ 0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from a random index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families (all return vectors normalised to sum = 1)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """U‑shaped family a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / s
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """Sum of two power‑laws from both ends."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / s
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """Exponential tails a_i ∝ exp(-lam·min(i,n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a = a / a.sum()
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """Cosine‑power family a_i ∝ |cos(πi/(n‑1))|^{p}."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max (soft‑max) of the convolution and its gradient
# ----------------------------------------------------------------------


def smooth_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    using two FFT‑based convolutions.
    """
    n = a.shape[0]
    conv = conv_fft(a)                         # length 2n‑1
    maxc = conv.max()
    # soft‑max weights, numerically stable
    w = np.exp(alpha * (conv - maxc))
    w /= w.sum()
    # grad = 2·(rev(a) ∗ w)[n‑1 : n‑1+n]
    rev_a = a[::-1]
    g_full = conv_fft_general(rev_a, w)       # length 3n‑2
    start = n - 1
    g = 2.0 * g_full[start:start + n]
    return g


# ----------------------------------------------------------------------
# Exact objective (same formula as evaluate_sequence for sum = 1)
# ----------------------------------------------------------------------


def exact_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a already normalised to Σa = 1."""
    n = a.shape[0]
    conv = conv_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Random candidate generator over several families
# ----------------------------------------------------------------------


def random_candidate(n: int) -> np.ndarray:
    """Pick a random analytic family and random parameters, return a normalised vector."""
    family = np.random.choice(['product', 'sum', 'exp', 'cos'])
    if family == 'product':
        p = np.random.uniform(0.2, 3.0)
        off = np.random.uniform(0.001, 2.0)
        return analytic_product(n, p, off)
    if family == 'sum':
        p = np.random.uniform(0.2, 3.0)
        off = np.random.uniform(0.001, 2.0)
        return analytic_sum(n, p, off)
    if family == 'exp':
        lam = np.random.uniform(0.001, 0.8)
        return analytic_exponential(n, lam)
    # cosine
    p = np.random.uniform(0.5, 9.0)
    return analytic_cosine(n, p)

# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that
    minimises the evaluation function.
    The routine performs a cheap global search over several analytic
    families, then refines the best vector with a smoothed‑max Adam
    optimiser on the simplex, and finally applies a stochastic
    perturbation phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    # --------------------------------------------------------------
    # Length of the vector – try a larger size if time permits
    # --------------------------------------------------------------
    if "n" in kwargs:
        n = int(kwargs["n"])
    else:
        # 4000 works well in the 1000 s budget; fall back to 2000 if very tight.
        n = 4000

    # --------------------------------------------------------------
    # 1️⃣  Initial analytic search (grid + random sampling)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # 1a – include a previously found global (if present)
    if "height_sequence_1" in globals():
        h1 = np.asarray(globals()["height_sequence_1"], dtype=float)
        if len(h1) == n:
            tmp = np.maximum(h1, 0.0)
            s = tmp.sum()
            if s > 1e-12:
                tmp /= s
                sc = exact_score(tmp)
                if sc < best_score:
                    best_score = sc
                    best_seq = tmp.copy()
                print(f"[init] used global height_sequence_1, score = {sc:.6f}")

    # 1b – fine deterministic grid for the product family
    p_vals = np.linspace(0.2, 3.0, 13)   # 13 × 9 ≈ 117 candidates
    off_vals = np.linspace(0.001, 2.0, 9)
    for p in p_vals:
        for off in off_vals:
            cand = analytic_product(n, p, off)
            sc = exact_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
    print(f"[init] best after product grid = {best_score:.6f}")

    # 1c – random search over several families (≈2000 evaluations)
    # Stop early if time is already very low.
    rand_trials = 2000
    for _ in range(rand_trials):
        if time.time() >= deadline - 5.0:
            break
        cand = random_candidate(n)
        sc = exact_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"[init] best after random sampling = {best_score:.6f}")

    # If time exhausted before any optimisation, return the best found so far.
    if time.time() >= deadline:
        return best_seq.tolist()

    # --------------------------------------------------------------
    # 2️⃣  Smoothed‑max Adam optimisation on the simplex
    # --------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    alpha_min, alpha_max = 10.0, 200.0
    alpha = alpha_min
    # Rough guess of the total number of Adam steps we can afford.
    # (Measured on a typical machine: ≈2500 steps per second for n=4000)
    max_steps_est = int((deadline - time.time()) * 2500)
    if max_steps_est < 10_000:
        max_steps_est = 10_000
    growth_factor = (alpha_max / alpha_min) ** (1.0 / max_steps_est)

    step = 0
    last_improve = time.time()
    best_adam_seq = a.copy()
    best_adam_score = best_score

    while time.time() < deadline:
        # ----------------------------------------------------------
        # compute gradient of the smooth‑max surrogate
        # ----------------------------------------------------------
        grad = smooth_max_gradient(a, alpha)

        # ----------------------------------------------------------
        # Adam update
        # ----------------------------------------------------------
        step += 1
        lr = 0.12 / (1.0 + step * 1e-5)          # slowly decaying learning rate
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----------------------------------------------------------
        # keep feasibility (non‑negative, Σ = 1)
        # ----------------------------------------------------------
        a = np.clip(a, 0.0, None)
        a = project_simplex(a)

        # ----------------------------------------------------------
        # increase temperature
        # ----------------------------------------------------------
        alpha = min(alpha_max, alpha * growth_factor)

        # ----------------------------------------------------------
        # occasional exact evaluation
        # ----------------------------------------------------------
        if step % 250 == 0:
            cur_score = exact_score(a)
            if cur_score < best_adam_score - 1e-12:
                best_adam_score = cur_score
                best_adam_seq = a.copy()
                last_improve = time.time()
                print(f"[adam {step:07d}] new best = {best_adam_score:.6f} (α={alpha:.1f})")

        # ----------------------------------------------------------
        # restart with a tiny jitter if stuck for a while
        # ----------------------------------------------------------
        if time.time() - last_improve > 20.0:
            jitter = np.random.rand(n) * 1e-3
            a = best_adam_seq + jitter
            a = project_simplex(a)
            last_improve = time.time()
            print(f"[adam {step:07d}] jitter restart")

        # ----------------------------------------------------------
        # occasional cheap random move to escape plateaus
        # ----------------------------------------------------------
        if np.random.rand() < 0.008:
            a = random_move(a, step * 0.2)

        # safety break
        if time.time() >= deadline:
            break

    # Record the best found during Adam
    best_seq = best_adam_seq
    best_score = best_adam_score
    print(f"[adam] finished after {step} steps, best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final stochastic refinement (Gaussian perturbations)
    # --------------------------------------------------------------
    # Use a decreasing standard deviation schedule
    sigma = 5e-4
    while time.time() < deadline:
        perturb = np.random.normal(scale=sigma, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        sc = exact_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")
        # slowly shrink the perturbation radius
        sigma *= 0.9999
        if sigma < 1e-6:
            sigma = 5e-4   # occasional reset to keep exploration alive

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, length = {n}, elapsed = {total_time:.1f}s")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
