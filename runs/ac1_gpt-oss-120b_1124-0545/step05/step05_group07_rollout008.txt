"""
Propose a non‑negative step function that (approximately) minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------

1. **Multiple random starts** – 5 independent runs are launched.
   Each run begins from a randomly chosen analytic family
   (product, sum‑of‑powers, beta‑shaped, exponential tail, cosine power)
   or from a completely random Dirichlet vector.
   The vector is normalised to the probability simplex.

2. **Smoothed‑max Adam optimisation** –  for each start we minimise the
   smooth approximation

        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)

   using the Adam optimiser.  The temperature α is increased
   slowly from 1 up to 300, so that the optimiser gradually focuses on
   the true max‑convolution.  After every update the vector is
   projected onto the simplex (non‑negative, Σa=1).

3. **Early‑stopping & stochastic jitter** – if the best value does not
   improve for 20 s we jitter the current iterate with a tiny random
   perturbation to escape plateaux.

4. **Final simulated‑annealing refinement** – after the best of the
   five runs is selected, a simple annealing loop makes Gaussian
   perturbations of the whole vector and accepts them with the usual
   Metropolis rule.  This polishing phase is able to squeeze a few
   extra points out of the remaining “flat” region of the objective.

All heavy convolutions are performed with FFT based linear convolution
(`conv_fft` / `conv_fft_general`) which runs in O(n·log n).  The whole
procedure fits comfortably into the 1000 s time budget for n≈2000–2500.

The function returns the best sequence found as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (zero‑padded) using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two arbitrary 1‑D arrays using FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    size = _next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: len(a) + len(b) - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a.
    The vector is assumed to be on the simplex; a safety normalisation is
    performed nevertheless.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from a random index i to j."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """U‑shaped family: a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """Sum of two power‑laws from the ends."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """Exponential tails."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a = a / a.sum()
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """Cosine power family."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    a = a / a.sum()
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped family."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = a / a.sum()
    return a


def random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Gamma(1,1) = Exp(1))."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / x.sum()


def smooth_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    using FFTs.
    """
    n = a.shape[0]
    conv = conv_fft(a)  # length 2n‑1
    maxc = conv.max()
    exp_alpha = np.exp(alpha * (conv - maxc))
    w = exp_alpha / exp_alpha.sum()          # Σ w = 1
    rev_a = a[::-1]
    g_full = conv_fft_general(rev_a, w)      # length 3n‑2
    start = n - 1
    g = 2.0 * g_full[start:start + n]
    return g


# ----------------------------------------------------------------------
# Candidate initialiser – picks a random analytic family or a uniform Dirichlet
# ----------------------------------------------------------------------


def initialise_candidate(n: int) -> np.ndarray:
    """Return a random feasible vector (sum = 1, non‑negative)."""
    choice = np.random.rand()
    if choice < 0.20:
        # product family
        p = np.random.uniform(0.2, 2.0)
        off = np.random.uniform(0.01, 2.0)
        a = analytic_product(n, p, off)
    elif choice < 0.40:
        # sum of power‑laws
        p = np.random.uniform(0.2, 2.0)
        off = np.random.uniform(0.01, 2.0)
        a = analytic_sum(n, p, off)
    elif choice < 0.60:
        # beta‑shaped
        alpha = np.random.uniform(-0.5, 2.0)
        beta = np.random.uniform(-0.5, 2.0)
        a = analytic_beta(n, alpha, beta)
    elif choice < 0.80:
        # exponential tail
        lam = np.random.uniform(0.001, 0.02)
        a = analytic_exponential(n, lam)
    elif choice < 0.90:
        # cosine power
        p = np.random.uniform(0.5, 4.0)
        a = analytic_cosine(n, p)
    else:
        # completely random Dirichlet
        a = random_dirichlet(n)
    # project to simplex (numerical safety)
    a = np.maximum(a, 0.0)
    a = project_simplex(a)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)**2).

    The routine runs several short Adam‑optimisation phases from
    different random starts and finishes with a simulated‑annealing
    polishing stage.  The best sequence found before the time limit is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # Choose the length of the vector.
    # ------------------------------------------------------------------
    if "n" in kwargs:
        n = int(kwargs["n"])
    elif "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        n = 2000                                     # default length

    # ------------------------------------------------------------------
    # Global best bookkeeping
    # ------------------------------------------------------------------
    best_overall_seq = np.ones(n) / n
    best_overall_score = compute_score(best_overall_seq)

    print("[setup] Using n =", n, "initial best score =", best_overall_score)

    # ------------------------------------------------------------------
    # Multiple independent runs (each gets about the same share of time)
    # ------------------------------------------------------------------
    runs = 5
    run_deadline = start_time + 0.65 * (budget_s - 1.0)   # 65 % of budget for the Adam phase

    for run_id in range(runs):
        if time.time() >= run_deadline:
            break
        # ----- initialise ------------------------------------------------
        a = initialise_candidate(n)
        cur_score = compute_score(a)
        if cur_score < best_overall_score:
            best_overall_score = cur_score
            best_overall_seq = a.copy()
        # Adam hyper‑parameters
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        alpha = 1.0
        alpha_max = 300.0
        # increase factor chosen so that α reaches α_max after ≈80 k steps
        growth_factor = (alpha_max / alpha) ** (1.0 / 80000.0)

        step = 0
        last_improve = time.time()
        print(f"[run {run_id+1}/{runs}] start score = {cur_score:.6f}")

        while time.time() < run_deadline:
            # ----- gradient of smooth max ---------------------------------
            grad = smooth_max_gradient(a, alpha)

            # ----- Adam update --------------------------------------------
            step += 1
            lr = 0.07 / (1.0 + step * 5e-5)               # a slowly decaying learning rate
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** step)
            v_hat = v / (1.0 - beta2 ** step)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----- feasibility projection ---------------------------------
            a = np.clip(a, 0.0, None)
            a = project_simplex(a)

            # ----- temperature schedule -----------------------------------
            alpha = min(alpha_max, alpha * growth_factor)

            # ----- periodic exact evaluation --------------------------------
            if step % 200 == 0:
                cur_score = compute_score(a)
                if cur_score < best_overall_score - 1e-12:
                    best_overall_score = cur_score
                    best_overall_seq = a.copy()
                    last_improve = time.time()
                    print(f"[run {run_id+1}] step {step:07d} new best = {best_overall_score:.6f} (α={alpha:.2f})")

            # ----- stagnation handling ------------------------------------
            if time.time() - last_improve > 22.0:         # 22 s without improvement -> jitter
                jitter = np.random.rand(n) * 1e-3
                a = best_overall_seq + jitter
                a = project_simplex(a)
                last_improve = time.time()
                print(f"[run {run_id+1}] step {step:07d} jitter restart")

            # ----- tiny random mass‑transfer moves -------------------------
            if np.random.rand() < 0.012:
                a = random_move(a, step * 0.25)

    # ------------------------------------------------------------------
    # Simulated‑annealing polishing (use the remaining time)
    # ------------------------------------------------------------------
    print("[refine] starting final annealing phase")
    while time.time() < deadline:
        # linear cooling schedule
        remaining = deadline - time.time()
        total = budget_s - (time.time() - start_time)
        temp = max(1e-5, remaining / total)          # temperature in (0,1]
        # propose Gaussian perturbation
        scale = 8e-4 * temp
        cand = best_overall_seq + np.random.normal(scale=scale, size=n)
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        sc = compute_score(cand)
        if sc < best_overall_score:
            best_overall_score = sc
            best_overall_seq = cand.copy()
            print(f"[refine] improved to {best_overall_score:.6f}")
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(sc - best_overall_score) / (temp + 1e-12)):
                best_overall_seq = cand
                # (score not updated because it got worse)

    elapsed = time.time() - start_time
    print(f"[done] best score = {best_overall_score:.6f}, length = {n}, time = {elapsed:.1f}s")
    return best_overall_seq.tolist()
