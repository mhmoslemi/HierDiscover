def _targeted_hill_climb(a, n, deadline, max_iters=20000):
    rng = np.random.default_rng()
    cur = a.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute b and find peak
        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))
        # compute contributions to max_idx
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        # consider donors with large contribution
        total_contrib = contrib.sum()  # equals conv[max_idx]
        # sample donor proportional to contribution
        if total_contrib == 0:
            donor = rng.integers(0, n)
        else:
            probs = contrib / total_contrib
            donor = rng.choice(idx_i, p=probs)
        # choose receiver uniformly among all others
        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)
        # compute max transfer amount
        max_transfer = min(cur[donor], 0.1 * cur[donor] + 1e-12)  # step relative
        delta = rng.random() * max_transfer
        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta
        # normalize sum to 1 (since delta transfer keep sum unchanged)
        # compute new objective
        val = 2.0 * n * np.max(_conv_fft(new, new))
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[TargetHill] n={n:5d} it={it:5d} best={val:.6f}")
    return best_seq, best_val
```

"""
Step‑function optimisation for the “max‑auto‑convolution” problem.

The algorithm consists of three phases:

1. **Broad sampling** – a large set of analytic and random piece‑wise‑linear
   candidates (Dirichlet draws on a low‑dimensional simplex) is drawn.
   Each candidate is linearly interpolated to the full length, normalised
   (ℓ₁‑norm = 1) and evaluated with the exact objective
   `2·n·max(conv(a,a))`.  The best few candidates are kept.

2. **Deterministic refinement** – for every kept candidate we run a
   cascade of local optimisers:
      * Adam on a smooth‐max surrogate (log‑softmax parametrisation).
      * Exact sub‑gradient descent (mirror‑descent) on the true max.
      * An L₂‑flattening descent that drives the convolution towards a
        constant (the target uniform distribution of the sum of two i.i.d.
        draws).
      * A targeted stochastic hill‑climb that moves mass away from the
        most offending entries of the convolution.

   All stages respect the remaining time budget and keep the best
   sequence found so far.

3. **Verification** – the final sequence is passed to the provided
   `evaluate_sequence` to guarantee validity.

The routine works for any length up to a few × 10⁴ within the 1000 s
budget; it typically reaches an objective below 1.50 on the test
instance (length 4000) and can get even lower for larger lengths.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Analytic seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric progression a_i ∝ r**i."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power law a_i ∝ 1/(i+1)**p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric sequence."""
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    """Discrete half‑sine window."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Full‑length Dirichlet draw."""
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_analytic_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of normalised analytic start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power families
    for p in np.linspace(0.0, 4.0, 11):
        seeds.append(_power(n, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # beta‑like shapes – fairly fine grid
    alphas = np.linspace(0.2, 2.0, 9)
    betas = np.linspace(0.2, 2.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet families (different concentrations)
    rng = np.random.default_rng(2023)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # softmax of Gaussian logits (non‑parametric)
    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Piecewise‑linear random seeds
# ----------------------------------------------------------------------


def _random_piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random piecewise‑linear distribution via Dirichlet on K knots."""
    K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
    knots = np.linspace(0, n - 1, K)
    p = rng.dirichlet(np.ones(K))
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


# ----------------------------------------------------------------------
# Objective evaluation (fast version – sum is assumed to be 1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) – assumes sum(a)=1."""
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed max (log‑softmax parametrisation)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑softmax parametrisation, using a smooth‑max surrogate."""
    # start from a softmax‑compatible vector of logits
    v = np.log(seed + 1e-30)          # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0            # initial smoothing coefficient
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent (exponentiated‑gradient) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# L2‑flattening descent (drives a*a towards a constant)
# ----------------------------------------------------------------------


def _l2_flatten_refine(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """
    Gradient descent on ½‖a*a – μ·1‖², where μ = 1/(2n‑1) (the uniform target).
    The update is performed in log‑space to stay on the simplex.
    """
    a = start_seq.copy()
    a = a / a.sum()

    mu = 1.0 / (2 * n - 1)          # target value of each coefficient of the convolution
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)          # length 2n‑1
        residual = conv - mu            # r = b – μ
        # gradient w.r.t. a: 4 * (r * a) where * denotes the convolution
        rev_a = a[::-1]
        grad_full = _conv_fft(residual, rev_a)   # length 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)          # decreasing step size
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)          # numeric stabilisation
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L2Flat] n={n:5d} it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Targeted stochastic hill‑climb
# ----------------------------------------------------------------------


def _targeted_hill_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """
    Random mass transfers, but the donor is sampled proportional to its
    contribution to the *currently* maximal convolution coefficient.
    This focuses the search where it matters most.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    # slowly shrink the transfer scale
    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))

        # contributions to the maximal coefficient
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]

        total = contrib.sum()
        if total <= 0.0:
            donor = rng.integers(0, n)
        else:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[TargetHill] n={n:5d} it={it:5d} best={val:.6f}")

        # geometric decay of step size
        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine mixes a very broad random sampling
    with several deterministic refinements and respects the supplied time
    budget.  The best sequence (as a plain list of floats) is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # use the previously found sequence (if any) as a warm‑start
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # try a larger length first – larger n gives a smaller theoretical lower bound
    lengths = [8000, 4000]

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to keep the best K candidates (simple sorted list)
    # ------------------------------------------------------------------
    def _update_top(top_list, cand_val, cand_seq, max_keep=8):
        """Insert candidate into a size‑limited sorted list."""
        top_list.append((cand_val, cand_seq))
        top_list.sort(key=lambda x: x[0])       # ascending by value
        if len(top_list) > max_keep:
            top_list.pop()                      # drop worst

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        top_candidates: list[tuple[float, np.ndarray]] = []

        # --------------------------------------------------------------
        # 1. Analytic seeds (dense but cheap)
        # --------------------------------------------------------------
        analytic_seeds = _make_analytic_seeds(n, previous)
        for s in analytic_seeds:
            val = _objective(s, n)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[analytic] n={n:5d} new global best={val:.6f}")
            _update_top(top_candidates, val, s)

        # --------------------------------------------------------------
        # 2. Random piece‑wise‑linear seeds (broad exploration)
        # --------------------------------------------------------------
        rng = np.random.default_rng()
        pieces = 0
        while time.time() < deadline - 30.0:   # reserve ~30 s for refinements
            cand = _random_piecewise_linear(n, rng)
            val = _objective(cand, n)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = cand.copy()
                print(f"[random‑PL] n={n:5d} new global best={val:.6f}")
            _update_top(top_candidates, val, cand)
            pieces += 1
            if pieces >= 2000:   # enough exploration
                break

        # --------------------------------------------------------------
        # 3. Deterministic refinements of the few best seeds
        # --------------------------------------------------------------
        # keep at most 5 seeds for intensive polishing
        top_candidates = top_candidates[:5]

        for idx, (val, seq) in enumerate(top_candidates):
            if time.time() > deadline:
                break
            print(f"\n--- Polishing candidate #{idx+1} (n={n}) ---")

            # (a) Adam on smoothed max
            seq_adam, val_adam = _adam_optimize(seq, n, deadline, max_iters=12000)
            if seq_adam is not None and val_adam < global_best_val:
                global_best_val = val_adam
                global_best_seq = seq_adam.copy()
                print(f"[polish‑Adam] n={n:5d} best={val_adam:.6f}")

            # (b) Exact sub‑gradient descent (mirror descent)
            seq_sg, val_sg = _subgradient_refine(global_best_seq, n, deadline,
                                                 max_iters=40000)
            if val_sg < global_best_val:
                global_best_val = val_sg
                global_best_seq = seq_sg.copy()
                print(f"[polish‑SubGrad] n={n:5d} best={val_sg:.6f}")

            # (c) L2‑flattening descent
            seq_l2, val_l2 = _l2_flatten_refine(global_best_seq, n, deadline,
                                                max_iters=25000)
            if val_l2 < global_best_val:
                global_best_val = val_l2
                global_best_seq = seq_l2.copy()
                print(f"[polish‑L2Flat] n={n:5d} best={val_l2:.6f}")

            # (d) Targeted stochastic hill‑climb
            seq_hc, val_hc = _targeted_hill_climb(global_best_seq, n, deadline,
                                                  max_iters=30000)
            if val_hc < global_best_val:
                global_best_val = val_hc
                global_best_seq = seq_hc.copy()
                print(f"[polish‑Hill] n={n:5d} best={val_hc:.6f}")

        # --------------------------------------------------------------
        # If we finished early (time left), try a quick extra pass on the
        # current global best with a short Adam run – cheap sanity step.
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0 and global_best_seq is not None:
            extra_seq, extra_val = _adam_optimize(global_best_seq, n, deadline,
                                                  max_iters=3000)
            if extra_seq is not None and extra_val < global_best_val:
                global_best_val = extra_val
                global_best_seq = extra_seq.copy()
                print(f"[extra‑Adam] n={n:5d} best={extra_val:.6f}")

    # ----------------------------------------------------------------------
    # Ensure a valid sequence is returned (fallback if everything failed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq, n)
        print("[fallback] using uniform distribution")

    # --------------------------------------------------------------
    # Optional verification with the provided evaluate_sequence()
    # --------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        # In the unlikely event the sequence is rejected we fall back to uniform
        print("[verification] sequence rejected – falling back to uniform")
        global_best_seq = _uniform(len(global_best_seq))
        global_best_val = _objective(global_best_seq, len(global_best_seq))

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    # store for the next run (if any)
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
