"""
Hybrid LP‑based alternating optimisation + local hill‑climb.

* Start from a diverse collection of analytically‑generated seeds
  (uniform, sinusoidal window, geometric, power‑law, beta‑shaped, random
  Dirichlet, softmax of random logits).  Pick the one with smallest
  max‑auto‑convolution.

* Iteratively improve the sequence by solving a linear programme:
      maximise   Σ g_i
      subject to  (f * g)_k ≤ M  for all k∈[0,2n‑2] ,   g_i ≥ 0
  where M = max_k (f*f)_k.
  The LP is solved with `scipy.optimize.linprog` on a sparse Toeplitz
  matrix that encodes the convolution.  The optimal g is then
  renormalised to the same ℓ₁‑norm as f and mixed with f by a convex
  combination (mixing weight t≈0.2).  This monotone step reduces the
  maximum auto‑convolution.

* After a fixed number of LP steps (or when time runs out) perform a
  cheap stochastic hill‑climb that transfers a small amount of mass
  between random positions while keeping improvements.

* The whole routine stops before the allotted 1000 s and returns the
  best sequence found (as a plain list of floats).  All entries are
  non‑negative, bounded by the checks inside `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.sparse import diags
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level convolution helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Analytic seed families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a modest set of diverse normalised start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    for a in (0.3, 0.5, 0.7):
        for b in (0.3, 0.5, 0.7):
            seeds.append(_beta_shape(n, a, b))

    rng = np.random.default_rng(2023)
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(7):
        logits = rng.standard_normal(n)
        # soft‑max (stable)
        m = np.max(logits)
        e = np.exp(logits - m)
        seeds.append(e / e.sum())

    if previous is not None and previous.shape == (n,):
        # ensure proper normalisation
        s = previous.sum()
        if s > 0:
            seeds.append(previous / s)

    return seeds


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (keeps improvements)
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """Random pair‑wise mass transfer, accepting only improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.01  # relative step size

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor == i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 1000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The method builds an initial seed, then
    repeatedly solves a linear programme to enlarge the ℓ₁‑mass while
    respecting the current auto‑convolution bound, mixes the result with
    the current shape and finishes with a short stochastic hill‑climb.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Retrieve previous best if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    target_n = 4000                     # primary optimisation dimension
    mix_weight = 0.2                    # convex mixing factor (1‑t)
    max_lp_iters = 30                   # number of LP‑mix cycles
    max_hc_iters = 25000                # hill‑climb budget after LP

    # ------------------------------------------------------------------
    # 1) Choose a good initial seed
    # ------------------------------------------------------------------
    seeds = _make_seeds(target_n, previous)
    best_seq = None
    best_val = float("inf")

    for s in seeds:
        conv = _conv_fft(s, s)
        cur_val = 2.0 * target_n * np.max(conv) / (s.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = s.copy()
            print(f"[seed] n={target_n:5d} new best={cur_val:.6f}")

    # Normalise to the canonical ℓ₁‑norm √(2n); this makes the
    # evaluation easier because the objective then equals max(conv).
    target_sum = np.sqrt(2.0 * target_n)
    best_seq = best_seq * (target_sum / best_seq.sum())

    # ------------------------------------------------------------------
    # 2) LP‑based alternating optimisation
    # ------------------------------------------------------------------
    f = best_seq.copy()
    for it in range(1, max_lp_iters + 1):
        if time.time() > deadline:
            break

        # current objective
        conv_f = _conv_fft(f, f)
        M = conv_f.max()
        cur_val = 2.0 * target_n * M / (f.sum() ** 2)   # = M after normalisation
        if cur_val < best_val:
            best_val = cur_val
            best_seq = f.copy()
            print(f"[LP‑iter] n={target_n:5d} it={it:3d} best={cur_val:.6f}")

        # ---------- build sparse Toeplitz constraint matrix A_ub ----------
        # A_ub has shape (2n‑1, n) with diagonal i equal to f_i
        offsets = np.arange(target_n, dtype=int)   # 0 … n‑1
        # each diagonal is a constant vector f_i, shape (n,)
        # we build a list with the same length‑n vector repeated
        data = [np.full(target_n, fi, dtype=np.float64) for fi in f]
        A_ub = diags(data, offsets, shape=(2 * target_n - 1, target_n), format="csr")

        b_ub = np.full(2 * target_n - 1, M, dtype=np.float64)

        # maximise Σ g_i  <=> minimise -Σ g_i
        c = -np.ones(target_n, dtype=np.float64)

        # Solve LP
        try:
            res = linprog(c,
                          A_ub=A_ub,
                          b_ub=b_ub,
                          bounds=[(0, None)] * target_n,
                          method='highs',
                          options={'presolve': True,
                                   'time_limit': max(0.0, deadline - time.time())})
        except Exception as e:
            print("[LP] exception:", e)
            break

        if not res.success:
            print("[LP] failed:", res.message)
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            # fallback – uniform distribution
            g = _uniform(target_n) * target_sum
        else:
            # renormalise g to keep the same ℓ₁‑norm as f
            g = g0 * (target_sum / sum_g0)

        # convex mix (guarantees non‑increase of the max‑convolution)
        f = (1.0 - mix_weight) * f + mix_weight * g

        # re‑normalise just in case of numerical drift
        s = f.sum()
        if s > 0:
            f = f * (target_sum / s)
        else:
            f = _uniform(target_n) * target_sum

    # ------------------------------------------------------------------
    # 3) Final local polishing – stochastic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_hc, val_hc = _hill_climb(f, target_n, deadline, max_iters=max_hc_iters)
        if val_hc < best_val:
            best_val = val_hc
            best_seq = seq_hc.copy()
            print(f"[final‑Hill] n={target_n:5d} best={val_hc:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional up‑sampling to a larger length (if time permits)
    # ------------------------------------------------------------------
    # A larger n often yields a slightly smaller max‑auto‑convolution.
    # We therefore linearly interpolate the current best sequence to
    # length 8000 and perform a light hill‑climb.
    larger_n = 8000
    if time.time() + 15.0 < deadline:   # ensure we have a few seconds left
        # linear interpolation
        interp_x = np.linspace(0, target_n - 1, target_n)
        new_x = np.arange(larger_n, dtype=np.float64)
        up_seq = np.interp(new_x, interp_x, best_seq)
        up_seq = up_seq * (np.sqrt(2.0 * larger_n) / up_seq.sum())

        # a short hill‑climb (far fewer iterations)
        seq_big, val_big = _hill_climb(up_seq, larger_n, deadline, max_iters=12000)
        if val_big < best_val:
            best_val = val_big
            best_seq = seq_big.copy()
            print(f"[up‑Hill] n={larger_n:5d} best={val_big:.6f}")

    # ------------------------------------------------------------------
    # 5) Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
          f"(length {len(best_seq)})")
    return best_seq.tolist()
