"""
Propose a step‑function (non‑negative vector) that minimises
`evaluate_sequence`.  The search proceeds in three stages:

1. **Broad parametric sweep** – a large pool of analytically
   generated families (Beta, arcsine, half‑sine, raised‑cosine,
   Gaussian, exponential, quadratic) and many random mixtures.
   The best vector from this cheap stage seeds the optimisation.

2. **Accelerated projected sub‑gradient descent** (APG) on the exact
   L∞ objective.  The classical Nesterov momentum scheme is used,
   together with a simple √t step‑size decay and occasional random
   restarts when progress stalls.

3. **Local stochastic refinement** – a simulated‑annealing style
   mass‑transfer routine that explores the neighbourhood of the best
   solution found so far.

The whole routine respects a 1000 s budget, prints intermediate
scores and finally returns the best vector (as a Python list).
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Geometry / projection utilities
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0,
    sum x = 1}.  O(n log n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Maximum value of the linear convolution a*a using an FFT of length
    fft_len (which must be ≥ 2*len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv.max())


def subgradient_max_conv(a: np.ndarray, conv_vals: np.ndarray,
                         max_idx: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a.
    For every active index s we add the shifted copy 2·a (where
    a_j contributes to component i = s‑j).  The result is the average
    over all active indices."""
    n = a.size
    g = np.zeros_like(a)
    for s in max_idx:
        # indices i such that 0 ≤ i < n and 0 ≤ s‑i < n
        i_start = max(0, s - (n - 1))
        i_end = min(n - 1, s)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = s - i
        g[i] += 2.0 * a[j]
    if max_idx.size:
        g /= float(max_idx.size)
    return g


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------


def arcsine_distribution(n: int) -> np.ndarray:
    """Exact discretisation of the continuous arcsine density
    f(x)=1/(π√(x(1‑x))) on [0,1] by integrating over each bin."""
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density evaluated at bin centres, then normalised."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def half_sine_distribution(n: int) -> np.ndarray:
    """Half‑sine shape: a_i ∝ sin(π·(i+0.5)/n)."""
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def raised_cosine_distribution(n: int) -> np.ndarray:
    """Raised‑cosine window that vanishes at the two edges."""
    i = np.arange(n, dtype=float)
    d = np.abs(i - (n - 1) / 2.0) / (n / 2.0)   # in [0,1]
    a = 0.5 + 0.5 * np.cos(np.pi * d)          # 0 at the ends, 1 in the centre
    a = np.maximum(a, 0.0)
    return a / a.sum()


def gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    """Truncated Gaussian centred at the middle of the interval."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def exponential_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.exp(-lam * np.abs(i - centre))
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def quadratic_distribution(n: int, radius: float) -> np.ndarray:
    """Parabolic (quadratic) window: a_i ∝ max(0, 1‑((i‑c)/R)²)."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = 1.0 - ((i - centre) / radius) ** 2
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The algorithm runs for at most ``budget_s``
    seconds and returns the best vector found (as a list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a tiny safety margin
    overall_deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two

    # ------------------------------------------------------------------
    # 0) Generate a diverse pool of analytically parametrised candidates
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    def evaluate_and_maybe_update(a_vec: np.ndarray, label: str):
        nonlocal best_a, best_score
        mx = max_convolution(a_vec, fft_len)
        score = 2.0 * n * mx
        if score < best_score:
            best_score = score
            best_a = a_vec.copy()
            print(f"[{label}] new best = {best_score:.6f}")

    # 0a) use any vector that may already be stored globally
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.size != n:
            # trim / pad with zeros to the required length
            if a0.size > n:
                a0 = a0[:n]
            else:
                a0 = np.pad(a0, (0, n - a0.size), constant_values=0.0)
        if a0.sum() > 0.0:
            a0 = a0 / a0.sum()
            evaluate_and_maybe_update(a0, "global0")

    # 0b) baseline families
    uniform = np.full(n, 1.0 / n)
    evaluate_and_maybe_update(uniform, "uniform")

    arcsine = arcsine_distribution(n)
    evaluate_and_maybe_update(arcsine, "arcsine")

    # 0c) symmetric Beta(α,α) sweep (log‑space for better coverage)
    for alpha in np.geomspace(0.05, 10.0, 30):
        cand = beta_distribution(n, alpha, alpha)
        evaluate_and_maybe_update(cand, f"beta_sym_{alpha:.3g}")

    # 0d) random asymmetric Beta pairs
    for _ in range(30):
        alpha = 10 ** rng.uniform(-2, 2)   # 0.01 … 100
        beta = 10 ** rng.uniform(-2, 2)
        cand = beta_distribution(n, alpha, beta)
        evaluate_and_maybe_update(cand, f"beta_asym_{alpha:.3g}_{beta:.3g}")

    # 0e) half‑sine
    half_sine = half_sine_distribution(n)
    evaluate_and_maybe_update(half_sine, "half_sine")

    # 0f) raised cosine
    raised_cosine = raised_cosine_distribution(n)
    evaluate_and_maybe_update(raised_cosine, "raised_cosine")

    # 0g) Gaussian family
    for sigma_frac in [0.04, 0.06, 0.08, 0.10, 0.15, 0.20, 0.30]:
        sigma = sigma_frac * n
        cand = gaussian_distribution(n, sigma)
        evaluate_and_maybe_update(cand, f"gauss_{sigma_frac:.2f}")

    # 0h) exponential decay family
    for lam_frac in [0.01, 0.02, 0.04, 0.07, 0.10]:
        lam = lam_frac / n   # scale so that λ is per‑index unit
        cand = exponential_distribution(n, lam)
        evaluate_and_maybe_update(cand, f"exp_{lam_frac:.2f}")

    # 0i) quadratic (parabolic) windows
    for R_frac in [0.3, 0.4, 0.5, 0.6, 0.8, 1.0]:
        radius = R_frac * n / 2.0
        cand = quadratic_distribution(n, radius)
        evaluate_and_maybe_update(cand, f"quad_{R_frac:.2f}")

    # 0j) mixtures of two elementary shapes (arcsine/half‑sine and Gaussian)
    for lam in np.linspace(0.0, 1.0, 7):
        mix = lam * arcsine + (1.0 - lam) * half_sine
        mix = np.maximum(mix, 0.0)
        mix /= mix.sum()
        evaluate_and_maybe_update(mix, f"mix_arc_hs_{lam:.2f}")

        mix2 = lam * arcsine + (1.0 - lam) * raised_cosine
        mix2 = np.maximum(mix2, 0.0)
        mix2 /= mix2.sum()
        evaluate_and_maybe_update(mix2, f"mix_arc_rc_{lam:.2f}")

    # ------------------------------------------------------------------
    # 1) Accelerated projected sub‑gradient descent (APG)
    # ------------------------------------------------------------------
    # allocate roughly 65 % of the total budget to this phase
    phase1_deadline = start_time + 0.65 * budget_s
    if best_a is None:                     # safety net – fall back to uniform
        best_a = uniform.copy()
    a = best_a.copy()
    a_prev = a.copy()
    t_prev = 1.0
    it = 0
    last_improve = time.time()
    start_lr = 0.8                    # initial step size (empirically good)

    while time.time() < phase1_deadline:
        # ---- Nesterov extrapolation ----
        t_next = (1.0 + sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0
        beta = (t_prev - 1.0) / t_next
        y = a + beta * (a - a_prev)

        # ---- gradient of max(conv(y,y)) ----
        fft_y = np.fft.fft(y, fft_len)
        conv_vals = np.real(np.fft.ifft(fft_y * fft_y))[: 2 * n - 1]
        conv_max = float(conv_vals.max())
        max_idx = np.where(np.abs(conv_vals - conv_max) < 1e-14)[0]
        g = subgradient_max_conv(y, conv_vals, max_idx)

        # ---- step size (√t decay) ----
        it += 1
        lr = start_lr / sqrt(it)

        # ---- projected gradient step ----
        a_new = proj_simplex(y - lr * g)

        # ---- bookkeeping ----
        a_prev = a
        a = a_new
        t_prev = t_next

        # evaluate exact score of the new point
        cur_max = max_convolution(a, fft_len)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[APG {it:07d}] improved to {best_score:.6f}")

        # ---- occasional random restart when stuck ----
        if time.time() - last_improve > 30.0:
            noise = rng.normal(scale=0.02, size=n)
            a = proj_simplex(best_a + noise)
            a_prev = a.copy()
            t_prev = 1.0
            last_improve = time.time()
            print(f"[APG {it:07d}] restart after stagnation")

    # ------------------------------------------------------------------
    # 2) Local stochastic refinement (simulated‑annealing)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 5.0:
        a = best_a.copy()
        best_conv_max = best_score / (2.0 * n)

        # linear temperature schedule
        T_start = best_score * 0.005
        T_end = best_score * 1e-6
        sa_start = time.time()
        sa_it = 0

        while time.time() < overall_deadline:
            elapsed = time.time() - sa_start
            frac = min(1.0, elapsed / remaining)
            T = T_start * (1.0 - frac) + T_end * frac

            # propose a simple elementary mass transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            max_move = min(a[i], 0.05 * a[i] + 1e-12)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            new_conv_max = max_convolution(a_new, fft_len)
            new_score = 2.0 * n * new_conv_max

            # Metropolis acceptance
            if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    print(f"[SA {sa_it:07d}] new best = {best_score:.6f} (t={time.time() - start_time:.1f}s)")

            sa_it += 1

    # ------------------------------------------------------------------
    # Finalisation
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    elapsed_total = time.time() - start_time
    print(f"[done] total time = {elapsed_total:.1f}s, best score = {best_score:.6f}, length = {n}")
    return best_a.tolist()
