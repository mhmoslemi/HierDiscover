1. a parametric sweep over arcsine / Beta families,
2. a smooth Adam‑style minimisation of a soft‑max approximation of the
   L∞‑convolution objective, with projection onto the probability simplex,

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Maximum value of the linear self‑convolution of `a`."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv.max())


def full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution (length 2*n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return conv


def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max over the convolution vector. Returns a probability vector."""
    # conv ≤ max, therefore exp(beta*(conv-max)) ≤ 1
    maxc = conv.max()
    w = np.exp(beta * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def weighted_gradient(a: np.ndarray, weight: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of Σ_k weight_k·(a*a)_k w.r.t. `a`.  Returns a vector of length n.
    The formula is 2·conv(a, weight_rev) restricted to the n central entries.
    """
    rev = weight[::-1]
    fa = np.fft.fft(a, fft_len)
    fr = np.fft.fft(rev, fft_len)
    conv = np.real(np.fft.ifft(fa * fr))
    n = a.size
    # central slice of length n is conv[n-1:2*n-1]
    segment = conv[n - 1 : 2 * n - 1]
    grad = 2.0 * segment[::-1]          # reverse to obtain the correct order
    return grad


def arcsine_distribution(n: int) -> np.ndarray:
    """Discretised arcsine density on a uniform grid, already normalised."""
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric or asymmetric Beta(α,β) density on a uniform grid."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def lp_refine(f: np.ndarray, M: float, top_k: int = 30) -> np.ndarray | None:
    """
    Linear‑programming refinement.
    Solve   maximise Σ_j g_j          (i.e. maximise the L1‑norm of g)
    s.t.    (f * g)_k ≤ M   for the `top_k` most critical shifts k
            g_j ≥ 0
    The solution is rescaled to have sum = 1 and returned.
    """
    n = f.size
    # compute full self‑convolution once – cheap compared with the LP
    fft_len_self = 1 << ((2 * n - 1).bit_length())
    conv = full_convolution(f, fft_len_self)

    # indices of the `top_k` largest entries (most binding constraints)
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(conv, -top_k)[-top_k:]

    rows = []
    cols = []
    data = []
    for row, k in enumerate(idxs):
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        # columns j = start … end
        j_vals = np.arange(start, end + 1, dtype=int)
        i_vals = k - j_vals                   # corresponding index in `f`
        rows.extend([row] * j_vals.size)
        cols.extend(j_vals.tolist())
        data.extend(f[i_vals].tolist())

    A_ub = coo_matrix((data, (rows, cols)), shape=(len(idxs), n))

    # linear programme: minimise –Σ g_j   (i.e. maximise Σ g_j)
    c = -np.ones(n)
    b_ub = np.full(len(idxs), M)

    # bounds: g_j ≥ 0   (no explicit upper bound)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    g0 = res.x
    s = g0.sum()
    if s <= 0.0:
        return None
    # rescale to sum = 1
    return g0 / s

        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = proj_simplex(a - step)

        # occasional random restart if stuck
        if time.time() - last_improve > 30.0:
            noise = rng.normal(scale=0.02, size=n)
            a = proj_simplex(best_a + noise)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[Adam {it:07d}] restart after stagnation")

    # ------------------------------------------------------------------
    # 3) Active‑set LP refinement (few expensive LP solves)
    # ------------------------------------------------------------------
    phase2_deadline = phase1_deadline + 0.25 * budget_s    # ~25 % of total time
    a = best_a.copy()
    while time.time() < phase2_deadline:
        conv = full_convolution(a, fft_len_self)
        M = float(conv.max())
        g = lp_refine(a, M, top_k=20)                     # 20 critical constraints
        if g is None:
            break
        # Mix the new direction a little (τ = 0.5 works well)
        tau = 0.5
        cand = proj_simplex((1.0 - tau) * a + tau * g)
        cand_score = 2.0 * n * max_convolution(cand, fft_len_self)
        if cand_score < best_score - 1e-9:
            best_score = cand_score
            best_a = cand.copy()
            a = cand.copy()
            print(f"[LP] improved to {best_score:.6f}")
        else:
            # No improvement – exit the LP phase early
            break

    # ------------------------------------------------------------------
    # 4) Local stochastic refinement (simulated annealing)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 5.0:
        a = best_a.copy()
        best_conv_max = best_score / (2.0 * n)

        # linear cooling schedule
        T_start = best_score * 0.004
        T_end = best_score * 1e-6
        sa_start = time.time()
        sa_it = 0

        while time.time() < overall_deadline:
            elapsed = time.time() - sa_start
            frac = min(1.0, elapsed / remaining)
            T = T_start * (1.0 - frac) + T_end * frac

            # elementary mass transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            max_move = min(a[i], 0.05 * a[i] + 1e-12)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            new_conv_max = max_convolution(a_new, fft_len_self)
            new_score = 2.0 * n * new_conv_max

            # Metropolis acceptance rule
            if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    print(f"[SA {sa_it:07d}] new best = {best_score:.6f} (t={time.time() - start:.1f}s)")

            sa_it += 1

    # ------------------------------------------------------------------
    # Final sanity checks & normalisation
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    elapsed = time.time() - start
    print(f"[done] total time = {elapsed:.1f}s, best score = {best_score:.6f}, length = {n}")

    # expose under a global name for possible reuse later
    globals()["height_sequence_2"] = best_a.tolist()

    return best_a.tolist()
