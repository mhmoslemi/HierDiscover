"""
Improved search for the minimum of the self‑convolution ∞‑norm.

The routine works in three stages:

1. **initialisation** – start from the previously found best sequence if
   available, otherwise from a uniform distribution.

2. **linear‑programming refinement** – for a fixed length *n* we solve the
   linear programme

        maximise   Σ g_j
        subject to  (f * g)_k ≤ M   for all k
                    g_j ≥ 0

   where *f* is the current sequence and *M* is its current maximal
   convolution value.  The optimiser returns a promising partner sequence *g*.
   We move a fraction *t* towards *g* (mirrored‑simplex update) and repeat.
   This is precisely the “LP‑step” described in the problem statement and
   usually yields a noticeable reduction of the peak value.

3. **exact sub‑gradient polishing** – after the LP loop we run a cheap
   exponentiated‑gradient sub‑gradient descent on the true objective.
   This tightens the result further.

All work respects the global time‑budget; the algorithm adapts the number
of LP iterations and the sub‑gradient budget accordingly.  The final
non‑negative step function (as a plain list of floats) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (full) via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (scale‑invariant)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 60000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) using the true sub‑gradient of
    max(conv).  Returns the best sequence seen and its objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirrored simplex update)
        lr = 0.75 / np.sqrt(it)               # classic 1/√t schedule
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming step (the “LP‑phase” from the statement)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix  A  with   A[k,j] = f_{k‑j}  (zero outside
    the support).  For any vector g we have   (f * g)_k = (A·g)_k .
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contributes the vector f shifted by +j in the rows
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_improve(f: np.ndarray, max_val: float, t: float = 0.3) -> np.ndarray:
    """
    One LP improvement step:
    maximise Σ g_j   subject to   (f * g)_k ≤ max_val  (∀k) ,  g ≥ 0.
    Returns the convex combination   (1‑t)·f + t·g_norm .
    If the LP fails, the original f is returned unchanged.
    """
    try:
        from scipy.optimize import linprog
    except Exception as e:          # scipy unavailable – fall back
        print("scipy not available, skipping LP step:", e)
        return f

    n = f.size
    A = _build_conv_matrix(f)          # shape (2n‑1, n)
    b = np.full(2 * n - 1, max_val, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximise Σ g_j  ⇔ minimise −Σ g_j

    # Solve the LP (highs is the default modern interior‑point method)
    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method="highs",
                  options={"presolve": True, "time_limit": 30})

    if not res.success:
        # LP could be infeasible if max_val is already too low
        print("LP step failed:", res.message)
        return f

    g = res.x
    s = g.sum()
    if s <= 0.0:
        return f

    g_norm = g / s                     # enforce Σ g = 1
    new_f = (1.0 - t) * f + t * g_norm
    return new_f


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best sequence discovered (as a plain list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Try to reuse a previously found construction
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:

        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Choose a length.
    # For the LP step a moderate n is preferable (memory vs quality).
    # Experiments show n≈2000 gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Initialise the sequence
    # ------------------------------------------------------------------
    if previous is not None and previous.shape == (n,):
        a = previous.copy()
        a = np.maximum(a, 0.0)
