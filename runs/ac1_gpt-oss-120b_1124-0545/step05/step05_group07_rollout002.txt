"""
Propose a non‑negative step function that (approximately) minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The method combines three stages:

1. **Analytic / random‑parameter seeding** – a quick scan of several
   parametric families (product, beta, exponential, cosine, sum) with
   random parameters to obtain a good feasible start.

2. **Parameter random search** – many cheap random draws from the same
   families (still respecting the simplex) to beat the initial best
   score.

3. **Simulated‑annealing / sub‑gradient refinement** – a fast
   sub‑gradient descent on the true `max`‑convolution combined with
   stochastic mass‑transfer moves.  The proposal direction follows the
   sub‑gradient of the current maximal convolution entry, and a
   temperature schedule allows occasional uphill steps, helping the
   search escape local plateaus.

All vectors are projected onto the probability simplex after every
update, guaranteeing feasibility for the evaluation function.
The whole routine respects the 1000 s budget and returns the best
sequence found.
"""

# Basic utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (zero‑padded) via FFT.  Length = 2·len(a)‑1."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two arbitrary vectors via FFT."""
    size = _next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: len(a) + len(b) - 1]


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a non‑negative vector a (assumes Σa = 1)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    return float(2.0 * a.shape[0] * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {w ≥ 0, Σw = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """U‑shaped family: a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """Sum of power‑laws from both ends."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """Exponential tails: a_i ∝ exp(-lam·min(i,n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a = a / a.sum()
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """Cosine power family: a_i ∝ |cos(πi/(n‑1))|^{p}."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    a = a / a.sum()
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped family: a_i ∝ (i+1)^{α}·(n‑i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = a / a.sum()
    return a

def max_conv_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a∗a)_k  w.r.t. a.
    Implements  g = 2·rev(a) ∗ w  where w is the uniform distribution over
    the indices attaining the current maximum.
    """
    n = a.shape[0]
    conv = conv_fft(a)
    max_val = conv.max()
    mask = (conv == max_val)
    w = mask.astype(float) / mask.sum()
    rev_a = a[::-1]
    g_full = conv_fft_general(rev_a, w)          # length = 3n‑2
    grad = 2.0 * g_full[n - 1:n - 1 + n]         # slice to length n
    return grad

            if prop_score < best_score:
                best_score = prop_score
                best_seq = proposal.copy()
                print(f"[refine] new global best {best_score:.6f} at iter {it}")

        # temperature decay
        temp = max(1e-12, temp * decay)

    return best_seq, best_score

# Main entry point

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising
    2·n·max(conv(a,a)) / (sum(a)**2).

    The routine performs:
      1. Analytic seeding (global sequence if available, otherwise a quick grid).
      2. Random parameter search over several families.
      3. Sub‑gradient simulated‑annealing refinement.

    Returns the best sequence found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Determine length n
    # ------------------------------------------------------------------
    if "n" in kwargs:
        n = int(kwargs["n"])
    elif "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        n = 2000

    # ------------------------------------------------------------------
    # Initialise from the global sequence if present, else from a quick analytic grid
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and len(globals()["height_sequence_1"]) == n:
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s < 1e-12:
            a0 = np.ones(n) / n
        else:
            a0 = a0 / s
        best_seq = a0.copy()
        best_score = compute_score(best_seq)
        print(f"[init] Using global height_sequence_1, score = {best_score:.6f}")
    else:
        # Very cheap grid for the product family
        best_seq = None
        best_score = np.inf
        for p in np.linspace(0.7, 1.3, 7):
            for off in np.linspace(0.1, 0.4, 7):
                cand = analytic_product(n, p, off)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
        # also try a symmetric beta shape
        for p in np.linspace(0.5, 1.5, 7):
            cand = analytic_beta(n, -p, -p)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
        if best_seq is None:
            best_seq = np.ones(n) / n
            best_score = compute_score(best_seq)
        print(f"[init] Analytic start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Random parameter search (stage 2)
    # ------------------------------------------------------------------
    best_seq, best_score = param_random_search(
        n=n,
        best_seq=best_seq,
        best_score=best_score,
        deadline=deadline,
        max_iter=30000
    )
    print(f"[stage2] best score after random param search = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Local refinement with simulated annealing / sub‑gradient moves (stage 3)
    # ------------------------------------------------------------------
    best_seq, best_score = refine_sequence(
        start_seq=best_seq,
        start_score=best_score,
        n=n,
        deadline=deadline
    )

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, length = {n}, time = {total_time:.1f}s")
    return best_seq.tolist()
