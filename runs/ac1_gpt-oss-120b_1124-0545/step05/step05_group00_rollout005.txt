def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1
    target_n = 2000

    # get previous global if any
    previous = globals().get("height_sequence_1")
    if previous is not None:
        prev_arr = np.asarray(previous, dtype=np.float64)
        if prev_arr.size >= target_n:
            # downsample
            ratio = prev_arr.size // target_n
            seq = prev_arr[:ratio*target_n].reshape(target_n, ratio).mean(axis=1)
            seq = np.clip(seq, 0, None)
            seq_sum = seq.sum()
            if seq_sum > 0:
                seq = seq / seq_sum
            else:
                seq = np.full(target_n, 1.0/target_n)
        else:
            seq = _uniform(target_n)
    else:
        seq = _uniform(target_n)

    # Optionally try other seeds as well, keep best as start for FW.
    # We'll generate seeds list and evaluate them; pick best.
    seeds = [_uniform(target_n)]
    # Add few random softmax seeds
    rng = np.random.default_rng(12345)
    for _ in range(4):
        seeds.append(_random_softmax(target_n, rng))
    best = None
    best_val = float('inf')
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best = s.copy()
    a = best

    global_best_seq = a.copy()
    global_best_val = best_val
    print(f"[Init] length={target_n} best_val={global_best_val:.6f}")

    # Frank-Wolfe LP refinement loop
    it = 0
    while time.time() < deadline:
        it += 1
        M = _conv_fft(a, a).max()
        # Solve LP
        A = conv_matrix(a)  # shape (2n-1, n)
        b_ub = np.full(2*target_n-1, M, dtype=np.float64)
        c = -np.ones(target_n, dtype=np.float64)  # maximize sum => minimize negative
        # solve
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            print("[LP] failed")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            print("[LP] zero sum")
            break
        g = b / sum_b
        # line search for best convex combination
        best_t = 0.0
        best_obj = global_best_val
        best_candidate = a
        candidates = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        for t in candidates:
            cand = (1 - t) * a + t * g
            cand = np.clip(cand, 0, None)
            cand_sum = cand.sum()
            if cand_sum <= 0:
                continue
            cand = cand / cand_sum
            val = _objective(cand)
            if val < best_obj:
                best_obj = val
                best_t = t
                best_candidate = cand
        if best_t == 0.0:
            # no improvement

"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is a *Frank–Wolfe* style optimiser:

1.  Start from a good seed (the previously discovered sequence, down‑sampled
    to a moderate length) and normalise it to sum = 1.
2.  Repeatedly solve a linear programme

        maximise   1ᵀ b
        subject to  conv(f,b) ≤ M        (M = max(conv(f,f)))
                    b ≥ 0

   where the convolution is linear in ``b``.  The LP pushes mass into
   positions where the current autocorrelation is small.
3.  Mix the current vector ``f`` with the normalised LP solution ``g``.
   A tiny line‑search (t ∈ {1,0.5,0.2,0.1,0.05,0.02,0.01}) picks the best
   convex combination that reduces the true objective.
4.  After the LP loop, a short *exact sub‑gradient* mirror‑descent phase
   (the same code as the reference solution) polishes the result.
5.  Finally a cheap Metropolis annealing walk is applied for a few
   seconds to escape any shallow plateaus.

The whole search respects the supplied time budget (default 1000 s) and
stores the best sequence in the global variable ``height_sequence_1`` for
future runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for the linear programme)
# ----------------------------------------------------------------------
try:
    from scipy.sparse import diags
    from scipy.optimize import linprog
    SCIPY_LP = True
except Exception:          # pragma: no cover
    SCIPY_LP = False

# ----------------------------------------------------------------------
# Basic convolution / objective utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    True objective: 2·n·max(conv(seq,seq)) / (sum(seq)²).
    ``seq`` is assumed to be non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:                     # should never happen
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple families (used for seeding / fallback)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a handful of normalised candidate vectors."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = [
        _uniform(n),
        _random_softmax(n, rng),
        _random_softmax(n, rng),
        _random_softmax(n, rng),
    ]

    # use the previous construction if it matches the length
    if previous is not None:
        if previous.shape == (n,):
            seeds.append(previous / previous.sum())
        elif previous.size > n:
            # simple down‑sampling by averaging blocks
            blk = previous.size // n
            down = previous[: blk * n].reshape(n, blk).mean(axis=1)
            down = np.clip(down, 0, None)
            if down.sum() > 0:
                seeds.append(down / down.sum())

    # guarantee non‑negativity and normalisation
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    return cleaned


# ----------------------------------------------------------------------
# Frank‑Wolfe linear‑programme oracle
# ----------------------------------------------------------------------


def _conv_matrix(f: np.ndarray):
    """
    Build a sparse matrix A (shape (2·n‑1, n)) such that
    A @ b == conv(f, b)    (full linear convolution).
    """
    n = f.size
    # each diagonal –d (d = 0 … n‑1) contains the constant f[d]
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = [-d for d in range(n)]
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return A


def _lp_max_sum_under_bound(f: np.ndarray, M: float):
    """
    Solve   maximise   1ᵀ b
            subject to conv(f,b) ≤ M   (component‑wise),
                     b ≥ 0.
    Returns the optimal b (or None if the LP fails).
    """
    if not SCIPY_LP:               # pragma: no cover
        return None

    n = f.size
    A = _conv_matrix(f)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # linprog minimises; -1ᵀb → maximise 1ᵀb
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method="highs")
    if not res.success:
        return None
    b = np.maximum(res.x, 0.0)                 # numerical safety
    return b


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the true max)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient mirror descent using the true sub‑gradient."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # average sub‑gradient over all active constraints
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        if len(active) > 0:
            g /= float(len(active))

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)           # keep numbers stable
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple annealing polishing
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Metropolis‑Hastings walk that moves tiny chunks of mass."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]  # up to 2 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        # occasional Gaussian “jump”
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # small safety margin

    # ------------------------------------------------------------------
    # Choose a modest length – 2000 gives a good trade‑off between
    # expressiveness and the size of the LP (≈4 M non‑zeros).
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Initialise from a previously stored construction (if any); otherwise
    # fall back to a random soft‑max seed.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        prev_arr = np.asarray(prev, dtype=np.float64)
    else:
        prev_arr = None

    seeds = _make_seeds(n, prev_arr)
    # quick evaluation of the raw seeds – keep the best one
    best_seq = None
    best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seq = s.copy()
    a = best_seq.copy()
    print(f"[Init] length={n} start_obj={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Frank–Wolfe linear‑programme driven refinement.
    # ------------------------------------------------------------------
    fw_iter = 0
    while time.time() < deadline:
        fw_iter += 1
        conv_self = _conv_fft(a, a)
        M = conv_self.max()
        # linear programme: maximise sum(b) under conv(a,b) ≤ M
        b = _lp_max_sum_under_bound(a, M)
        if b is None:
            print("[LP] failed – exiting FW loop")
            break
        sum_b = b.sum()
        if sum_b <= 1e-12:
            print("[LP] zero‑mass solution – exiting FW loop")
            break

        g = b / sum_b                       # normalise to sum = 1

        # line search on a few candidate mix‑ratios
        candidates = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        improved = False
        for t in candidates:
            cand = (1.0 - t) * a + t * g
            cand = np.clip(cand, 0.0, None)
            if cand.sum() == 0:
                continue
            cand /= cand.sum()
            val = _objective(cand)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[FW] iter={fw_iter:3d} t={t:.3f} obj={val:.6f}")
                break          # accept first improvement (largest step that works)

        if not improved:
            # no candidate improves the objective – we are stuck
            print(f"[FW] iter={fw_iter} no improvement, terminating FW")
            break

    # ------------------------------------------------------------------
    # Phase B – exact sub‑gradient mirror descent (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a, val = _subgradient_refine(a, n, deadline, max_iters=120000)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Polish SubGrad] obj={val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – light annealing for the final seconds
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a, val = _anneal_optimize(a, n, deadline, max_iters=None)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Polish Anneal] obj={val:.6f}")

    # ------------------------------------------------------------------
    # Store the result globally (for a possible next run) and return.
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = np.array(best_seq, dtype=np.float64)
    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
