"""
Propose a step‑function sequence that (approximately) minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of four stages:

1. **Seed & Warm‑start** – reuse the best known construction (global variable
   ``height_sequence_1``) or fall back to a uniform vector.

2. **Low‑dimensional LP refinement** – repeatedly solve a linear program that
   maximises the ℓ₁‑norm of a candidate while keeping its convolution with the
   current vector below the current worst autocorrelation.  The LP works on a
   down‑sampled version (≈1024 entries) so that it is cheap and can explore
   non‑symmetric shapes that gradient descent may miss.

3. **Upsampling & fine‑tuning** – interpolate the improved low‑dimensional
   vector back to the target length (4000) and run a few thousand steps of
   exact sub‑gradient mirror descent followed by a short Metropolis‑Annealing
   walk.

All vectors are kept on the probability simplex (sum = 1) and stay
non‑negative.  The routine respects the given time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform probability vector of length n."""
    return np.ones(n, dtype=np.float64) / n


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex via a soft‑max of Gaussian logits."""
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _downsample(vec: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a shorter length."""
    if vec.size == new_len:
        return vec.copy()
    old_x = np.linspace(0.0, 1.0, vec.size)
    new_x = np.linspace(0.0, 1.0, new_len)
    out = np.interp(new_x, old_x, vec)
    out = np.clip(out, 0.0, None)
    if out.sum() == 0.0:
        out = _uniform(new_len)
    else:
        out /= out.sum()
    return out


def _upsample(vec: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length."""
    return _downsample(vec, new_len)


# ----------------------------------------------------------------------
# Build the convolution constraint matrix for a fixed f (size n)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Returns a sparse matrix T of shape (2n‑1, n) such that
        conv(f, b) = T @ b
    for any vector b of length n.
    """
    n = f.size
    rows = 2 * n - 1
    # Estimate number of non‑zeros: roughly n²/2
    data = []
    rows_idx = []
    cols_idx = []

    for k in range(rows):
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        # entries T[k, i] = f[i] for i_low ≤ i ≤ i_high
        idx_range = np.arange(i_low, i_high + 1, dtype=int)
        rows_idx.extend([k] * idx_range.size)
        cols_idx.extend(idx_range.tolist())
        data.extend(f[i_low:i_high + 1].tolist())

    T = sp.csr_matrix((data, (rows_idx, cols_idx)), shape=(rows, n), dtype=np.float64)
    return T


# ----------------------------------------------------------------------
# Linear‑programming refinement (maximise ℓ₁‑norm under convolution bound)
# ----------------------------------------------------------------------


def _lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve

        max   sum(b)
        s.t.  conv(f, b) ≤ M
              b ≥ 0

    Returns the optimal vector b (may be all zeros if infeasible).
    """
    n = f.size
    T = _build_conv_matrix(f)
    # linprog minimises; we minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    ub = M * np.ones(T.shape[0], dtype=np.float64)

    # SciPy’s HiGHS solver works well with sparse constraints.
    res = opt.linprog(c, A_ub=T, b_ub=ub, bounds=(0, None), method="highs")
    if not res.success:
        # Fallback to a feasible point (uniform) if the LP fails.
        return _uniform(n)
    return np.maximum(res.x, 0.0)


# ----------------------------------------------------------------------
# Exact sub‑gradient mirror‑descent on the simplex
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        deadline: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using the true sub‑gradient of
    max(conv(a,a)).  Stops when the time budget expires.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    n = a.size

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # Build averaged sub‑gradient
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        if active.size > 0:
            g /= float(active.size)

        # Exponentiated‑gradient step with 1/√t schedule
        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)               # ensure numerical stability
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Metropolis‑Hastings walk that moves a tiny amount of mass between
    two positions and occasionally adds a small Gaussian “jump”.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val
    n = cur.size

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # Transfer up to 2 % of the source mass
        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        # occasional global perturbation
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step‑function minimizing the evaluation metric.
    Returns the best sequence (list of floats) found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # --------------------------------------------------------------
    # 0. Retrieve any previously discovered construction.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev = np.asarray(prev, dtype=np.float64)
            if np.any(np.isnan(prev)) or np.any(np.isinf(prev)):
                prev = None
        except Exception:
            prev = None

    # --------------------------------------------------------------
    # Parameters
    # --------------------------------------------------------------
    n_final = 4000            # target length
    n_small = 1024            # dimension for LP stage
    mix_coeff = 0.15          # convex‑combination weight when mixing new LP solution

    # --------------------------------------------------------------
    # 1) Warm‑start on the final length (use previous if available)
    # --------------------------------------------------------------
    if prev is not None and prev.size == n_final:
        warm = prev.copy()
    else:
        warm = _uniform(n_final)

    warm = np.clip(warm, 0.0, None)
    if warm.sum() == 0.0:
        warm = _uniform(n_final)
    warm /= warm.sum()

    # --------------------------------------------------------------
    # 2) Down‑sample to a manageable dimension for LP refinement.
    #    We also generate a few random starts to diversify the search.
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed)
    candidates_small = []

    # base candidate: down‑sampled warm start
    candidates_small.append(_downsample(warm, n_small))

    # a few random simplex points
    for _ in range(4):
        cand = _random_softmax(n_small, rng)
        candidates_small.append(cand)

    best_small_seq = None
    best_small_val = float("inf")

    for idx, small_seq in enumerate(candidates_small):
        if time.time() > deadline - 5.0:
            break

        # Ensure valid simplex
        small_seq = np.clip(small_seq, 0.0, None)
        if small_seq.sum() == 0.0:
            small_seq = _uniform(n_small)
        small_seq /= small_seq.sum()

        # ----------------------------------------------------------
        # LP‑based iterative improvement on this small seed
        # ----------------------------------------------------------
        f = small_seq.copy()
        it_lp = 0
        while time.time() < deadline - 5.0:
            it_lp += 1
            # Current worst autocorrelation
            conv_ff = np.convolve(f, f)
            M = conv_ff.max()

            # Solve LP to maximise ℓ₁‑norm under the same bound
            b = _lp_max_sum(f, M)

            sum_b = b.sum()
            if sum_b < 1e-12:
                break

            # Convex combination step
            f = (1.0 - mix_coeff) * f + mix_coeff * (b / sum_b)

            # Re‑normalise
            f = np.clip(f, 0.0, None)
            if f.sum() == 0.0:
                f = _uniform(n_small)
            else:
                f /= f.sum()

            # Occasionally monitor progress
            if it_lp % 5 == 0:
                cur_val = _objective(f)
                if cur_val < best_small_val:
                    best_small_val = cur_val
                    best_small_seq = f.copy()
                    print(f"[LP‑seed {idx}] it={it_lp:3d} val={cur_val:.6f}")

            # Stop after a modest number of LP iterations to keep time budget

        cur_val = _objective(f)
        if cur_val < best_small_val:

    return best_seq.tolist()
