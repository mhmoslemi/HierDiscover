"""
Propose a non‑negative step function that (approximately) solves

    minimize  2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm improves on the previous baseline by

* adding a symmetric Gaussian analytic family,
* enforcing exact symmetry of every candidate (most optimal solutions are
  symmetric around the centre),
* using an active‑set sub‑gradient / mirror‑descent update on the simplex,
* periodically solving a linear programme (maximise Σ g_i under
  (f * g) ≤ current max) and mixing the LP solution into the current iterate,
* a final soft‑max (log‑sum‑exp) optimisation with Adam,
* a short random‑perturbation refinement phase.

The code respects the wall‑time budget (default 1000 s) and returns the
best sequence found.  All helper functions are top‑level, no closures or
lambdas are used.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of a and b using FFT (real‑valued)."""
    size = _next_pow2(a.shape[0] + b.shape[0] - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """Exact objective (scale‑invariant) for a on the probability simplex."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_self(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry v[i] = v[-1-i]."""
    return (v + v[::-1]) / 2.0


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small amount of mass from a random index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = project_simplex(a_new)
    a_new = symmetrize(a_new)
    return a_new

def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    return a / a.sum()


def analytic_cosine(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    return a / a.sum()


def analytic_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta_
    return a / a.sum()


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _add_candidate(cand: np.ndarray,
                   best_list: list[tuple[float, np.ndarray]],
                   max_len: int = 12) -> None:
    """Insert a candidate into the bounded best‑list (sorted by score)."""
    sc = compute_score(cand)
    inserted = False
    for idx, (s, _) in enumerate(best_list):
        if sc < s:
            best_list.insert(idx, (sc, cand.copy()))
            inserted = True
            break
    if not inserted and len(best_list) < max_len:
        best_list.append((sc, cand.copy()))
    while len(best_list) > max_len:
        best_list.pop()


def analytic_grid_search(n: int):
    """
    Exhaustively scan a set of families (including a Gaussian family)
    and return the overall best vector together with a list of the top
    candidates (score, vector).
    """
    top = []   # (score, vector) sorted increasingly

    # Product family
    p_vals = np.arange(0.1, 3.01, 0.15)

    )
    for p in p_vals:
        for off in off_vals:
            cand = analytic_product(n, p, off)
            cand = symmetrize(cand)
            _add_candidate(cand, top)

    # Sum family
    for p in p_vals:
        for off in off_vals:
            cand = analytic_sum(n, p, off)
            cand = symmetrize(cand)
            _add_candidate(cand, top)

    # Exponential tails
    lam_vals = np.logspace(-3, 1, num=20)
    for lam in lam_vals:
        cand = analytic_exponential(n, lam)
        cand = symmetrize(cand)
        _add_candidate(cand, top)

    # Cosine‑power
    p_cos = np.arange(0.5, 8.1, 0.5)
    for p in p_cos:
        cand = analytic_cosine(n, p)
        cand = symmetrize(cand)
        _add_candidate(cand, top)

    # Beta family
    alpha_vals = np.arange(-0.5, 2.51, 0.5)
    beta_vals = np.arange(-0.5, 2.51, 0.5)
    for a in alpha_vals:

            cand = symmetrize(cand)
            _add_candidate(cand, top)

    # Gaussian family – sigma in [n/20, n/5]
    sigmas = np.linspace(n / 20.0, n / 5.0, num=12)
    for sigma in sigmas:

        _add_candidate(cand, top)

    # Previous best if present
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            ref = symmetrize(ref)
            _add_candidate(ref, top)

    best_seq = min(top, key=lambda x: x[0])[1].copy()
    return best_seq, top


def mix_top_candidates(top: list[tuple[float, np.ndarray]], n: int) -> np.ndarray:
    """
    Try convex combinations of the top few candidates.
    Returns the best mixture found (or the original best if none improve).
    """
    best_seq = top[0][1].copy()
    best_score = top[0][0]
    alphas = np.linspace(0.0, 1.0, 21)  # 0, 0.05, …, 1
    for i in range(len(top)):
        for j in range(i + 1, len(top)):
            a_i = top[i][1]
            a_j = top[j][1]
            for alpha in alphas:
                cand = alpha * a_i + (1.0 - alpha) * a_j
                cand = symmetrize(cand)
                cand = project_simplex(cand)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
    return best_seq

def solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve: maximise Σ g_i   s.t.  (f * g)_k ≤ M  for all k,
    g_i ≥ 0.
    Returns a normalised vector (sum = 1) or None.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        return None

    n = f.shape[0]
    rows = 2 * n - 1
    # Build Toeplitz‑style constraint matrix A such that (A @ g)_k = (f * g)_k
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n)                # minimise -Σ g_i  <=> maximise Σ g_i
    bounds = [(0, None)] * n
    b_ub = np.full(rows, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'disp': False})

    if not res.success:
        return None
    g = res.x
    s = g.sum()
    if s < 1e-12:
        return None
    g_norm = g / s
    return g_norm

        w_raw = np.exp(beta * (conv - max_val))   # max exponent = 0
        w = w_raw / w_raw.sum()

        # gradient of smoothed objective: a + weighted shift
        a_rev = a[::-1]
        corr_full = conv_two(w, a_rev)                   # length = 3n‑2
        weighted_shift = corr_full[n - 1:n - 1 + n]       # central n entries
        g = a + weighted_shift

        # Adam update
        t_adam += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step_vec
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate = symmetrize(a_candidate)
        a_candidate = project_simplex(a_candidate)

        # occasional random move / jitter
        if np.random.rand() < 0.02:
            a_candidate = random_move(a_candidate, lr * 0.4)
        if np.random.rand() < 0.006:
            jitter = np.random.randn(n) * 4e-4
            a_candidate = a_candidate + jitter
            a_candidate = np.clip(a_candidate, 0.0, None)
            a_candidate = symmetrize(a_candidate)
            a_candidate = project_simplex(a_candidate)

        # stagnation handling
        if time.time() - last_improve > stagnation_time:
            jitter = np.random.rand(n) * 1e-3
            a_candidate = best_seq + jitter
            a_candidate = symmetrize(a_candidate)
            a_candidate = project_simplex(a_candidate)
            last_improve = time.time()
            print(f"[Adam {t_adam:07d}] jitter restart")

        a = a_candidate

    # ------------------------------------------------------------------
    # 4) Final random refinement
    # ------------------------------------------------------------------
    print("[refine] final random perturbation phase")
    while time.time() < deadline:
        perturb = np.random.normal(scale=4e-4, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = symmetrize(cand)
        cand = project_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, length = {n}, time = {total_time:.1f}s")
    return best_seq.tolist()
