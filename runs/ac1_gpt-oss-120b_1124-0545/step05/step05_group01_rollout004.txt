def proj_simplex(v):
    # projects onto simplex sum = 1, v >= 0
    # Implementation based on sorting.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _subgrad_proj_refine(a, n, deadline, max_iters=100000, step0=0.5):
    best_seq=a.copy()
    best_val=_objective(a)
    for it in range(1,max_iters+1):
        if time.time()>deadline: break
        conv=_conv_fft(a,a)
        max_idx = int(conv.argmax())
        # compute gradient
        g=np.zeros_like(a)
        i_start=max(0, max_idx-(n-1))
        i_end=min(n-1, max_idx)
        idx_i=np.arange(i_start,i_end+1)
        rev_idx = max_idx - idx_i
        g[idx_i]=2.0*a[rev_idx]
        # step size
        eta=step0/np.sqrt(it)
        # gradient descent
        a = a - eta * g
        # project onto simplex
        # using projection onto simplex
        # we can implement _proj_simplex
        a = _proj_simplex(a)
        # compute objective
        cur_val=_objective(a)
        if cur_val < best_val:
            best_val=cur_val
            best_seq=a.copy()
            print(f"[SubGradProj] n={n} it={it} best={cur_val:.6f}")
    return best_seq,best_val

Projection function:

def _proj_simplex(v):
    # based on algorithm from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _interpolate_seed(old_seq, new_n):
    old_n = old_seq.shape[0]
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    new_seq = np.interp(new_x, old_x, old_seq)

    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()

"""
Search for a non‑negative step‑function minimizing the self‑convolution ∞‑norm.

Algorithm summary
-----------------
1. **Seed generation** – diverse families (uniform, geometric, power‑law,
   Gaussian, Dirichlet) plus an up‑sampled version of the best sequence found
   so far (if any).

2. **Continuation L‑BFGS‑B** – we minimise a smooth approximation of the
   true objective:
        Lβ(a) = max_k a∗a_k  (approximated by a log‑sum‑exp with temperature β).
   The parameters β are increased (1 → 2 → … → 40) while using the solution of
   the previous β as a warm start.  The optimisation variables are the logits
   of a soft‑max parametrisation, guaranteeing non‑negativity and unit sum.

3. **Exact sub‑gradient refinement** – after L‑BFGS we run many projected
   sub‑gradient steps with a Euclidean simplex projection.  The update size
   follows ηt = η0 / √t.

4. **Cheap simulated‑annealing** – a final Metropolis walk that swaps tiny
   amounts of mass between random coordinates; every 1000 steps a global
   jitter is added.

The routine respects a global time budget (default 1000 s).  The best
sequence found over all examined lengths is returned as a plain list of
floats and also stored in the global variable ``height_sequence_1`` for
later runs.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the grader (scale‑invariant)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _interpolate_seed(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a shorter sequence to a longer one."""
    old_n = old_seq.shape[0]
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, old_seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    return new_seq / s if s > 0 else _uniform(new_n)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse set of normalised start‑vectors for length n."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    seeds.append(_gaussian(n, sigma_factor=0.18))

    for alpha in [0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # mirrored copies
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # up‑sample previous best if it exists and has a different length
    if previous is not None:
        if previous.shape[0] == n:
            seeds.append(previous / previous.sum())
        else:
            seeds.append(_interpolate_seed(previous, n))

    # guarantee non‑empty list
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# L‑BFGS‑B continuation on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _loss_and_grad(v: np.ndarray, n: int, beta: float):
    """Return (loss, gradient) of the smoothed‑max loss for logits v."""
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = conv.max()
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    loss = shift + np.log(sum_exp) / beta

    w = exp_term / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n - 1 : n - 1 + n]

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return loss, grad_v


def _lbfgs_continuation(v0: np.ndarray, n: int,
                        deadline: float,
                        beta_schedule: list[float]) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Warm‑started L‑BFGS‑B over a schedule of β values.
    Returns the final logits, the best sequence encountered and its true objective.
    """
    v = v0.copy()
    best_seq = None
    best_val = np.inf

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # SciPy will call the function with the current logit vector.
        # We expose β and n via the *args* tuple.
        res = minimize(_loss_and_grad,
                       v,
                       args=(n, beta),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 30, 'disp': False})

        v = res.x
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LBFGS] n={n} β={beta:.1f} it={res.nit} best={cur_val:.6f}")

        if time.time() > deadline:
            break

    return v, best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement with Euclidean simplex projection
# ----------------------------------------------------------------------


def _subgrad_proj_refine(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 100000,
                         step0: float = 0.5) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution using Euclidean projection."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())

        # gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = step0 / np.sqrt(it)
        a = a - eta * g
        a = _proj_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (identical to the previous version)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.025 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function respects a time‑budget and returns the
    best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # 1 s safety margin

    # reuse a previously discovered construction if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # lengths to explore – start small, then grow if time permits
    lengths = [4000, 8000, 12000, 20000]
    global_best_seq = None
    global_best_val = np.inf

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # 1) Build a pool of start vectors
        # ------------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # ------------------------------------------------------------------
        # 2) Evaluate raw seeds, keep the few most promising
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:6]]

        # record best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # 3) Continuation L‑BFGS‑B optimisation
        # ------------------------------------------------------------------
        beta_schedule = [1.0, 2.0, 5.0, 10.0, 20.0, 40.0]
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            v0 = np.log(seed_vec + 1e-30)
            _, seq_opt, val_opt = _lbfgs_continuation(v0, n, deadline, beta_schedule)
            if seq_opt is not None and val_opt < global_best_val:
                global_best_val = val_opt
                global_best_seq = seq_opt.copy()
                print(f"[overall LBFGS] n={n} seed={idx} best={val_opt:.6f}")

        # ------------------------------------------------------------------
        # 4) Exact sub‑gradient refinement (projected)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            # allocate more iterations for larger n
            max_iter = int(40000 * n / 4000)
            seq_ref, val_ref = _subgrad_proj_refine(global_best_seq, n,
                                                    deadline,
                                                    max_iters=max_iter,
                                                    step0=0.5)
            if seq_ref is not None and val_ref < global_best_val:
                global_best_val = val_ref
                global_best_seq = seq_ref.copy()
                print(f"[overall SubGrad] n={n} best={val_ref:.6f}")

        # ------------------------------------------------------------------
        # 5) Simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_ann, val_ann = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
            if seq_ann is not None and val_ann < global_best_val:
                global_best_val = val_ann
                global_best_seq = seq_ann.copy()
                print(f"[overall Anneal] n={n} best={val_ann:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose the best sequence for the next run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
