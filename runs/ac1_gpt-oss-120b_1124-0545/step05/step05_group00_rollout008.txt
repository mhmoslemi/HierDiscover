import time
import numpy as np
import math

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_fft(a,b):
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa*fb, n=N)[:L]
    return conv

def _objective(seq):
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s*s)

def _uniform(n):
    return np.ones(n, dtype=np.float64) / n

def _geom(n, r):
    a = r**np.arange(n, dtype=np.float64)
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s

def _power(n, p):
    idx = np.arange(1, n+1, dtype=np.float64)
    a = 1.0 / (idx**p)
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s

def _gaussian(n, sigma_factor=0.18):
    i = np.arange(n, dtype=np.float64)
    centre = (n-1)/2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i-centre)/sigma)**2)
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s

def _tent_window(n):
    i = np.arange(n, dtype=np.float64)
    centre = (n-1)/2.0
    a = 1.0 - np.abs(i-centre)/centre
    a = np.clip(a,0.0,None)
    s = a.sum()
    return a / s

def _random_simplex(n, rng):
    # Dirichlet(1,...,1) -> uniform over simplex
    x = rng.gamma(1.0, 1.0, size=n)
    if x.sum() == 0.0:
        return _uniform(n)
    return x / x.sum()

def _make_seeds(n, previous):
    # small collection; we keep a few families.
    rng = np.random.default_rng(12345)
    seeds = []
    seeds.append(_uniform(n))
    # geometric with a few ratios
    for r in [0.99, 0.95, 0.9, 0.8]:
        if r > 0:
            seeds.append(_geom(n, r))
    # power law
    for p in [0.0, 0.5, 1.0, 1.5]:
        seeds.append(_power(n, p))
    # gaussian
    seeds.append(_gaussian(n, sigma_factor=0.12))
    seeds.append(_gaussian(n, sigma_factor=0.22))
    # tent
    seeds.append(_tent_window(n))
    # some random simplex vectors
    for _ in range(5):
        seeds.append(_random_simplex(n, rng))
    # previous if present
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())
    # Clean: non-negative and normalize
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

def _equalize_descent(start_seq, n, deadline, max_iters=200000, alpha=0.5, beta=1.0):
    """
    Gradient descent on a combination of the max‑convolution sub‑gradient
    and the variance of the whole convolution.  The variables are kept in
    log‑space (exponentiated‑gradient / mirror descent) so the simplex
    constraint (non‑negative, sum = 1) holds automatically.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)  # start logits
    best_seq = a.copy()
    best_val = _objective(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # recover current a from logits
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active set for max
        rel_tol = 1e-12
        active = np.where(conv >= max_val - rel_tol)[0]
        # weight vector for max gradient
        w_max = np.zeros_like(conv)
        w_max[active] = 1.0 / len(active)
        rev_a = a[::-1]
        # gradient of weighted max (or average over active)
        full_max = _conv_fft(rev_a, w_max)
        grad_max = 2.0 * full_max[n-1:n-1+n]

        # variance gradient
        mu = (a.sum()**2) / (2 * n - 1)
        diff = conv - mu
        full_var = _conv_fft(rev_a, diff)
        grad_var = 4.0 * full_var[n-1:n-1+n]

        # combine
        grad = alpha * grad_max + beta * grad_var

        # step‑size schedule
        lr = 0.6 / math.sqrt(it + 1.0)
        log_a = log_a - lr * grad

        # keep numbers stable
        # (optional) shift to avoid overflow
        # (no explicit clipping needed because logits may become large)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Equalize] n={n} it={it:6d} best={cur_val:.6f}")
    return best_seq, best_val

def _anneal_optimize(start_seq, n, deadline, max_iters=None):
    """
    Simple Metropolis‑Hastings walk that moves a tiny amount of mass
    between two positions and occasionally adds a global Gaussian perturbation.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        delta = rng.random() * 0.02 * cur[i]
        if delta > cur[i]:
            delta = cur[i]
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:6d} best={new_val:.6f}")

        if it % 1000 == 0:
            # occasional jump
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= total
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:6d} (jump) best={new_val:.6f}")
    return best, best_val

"""
Search for a non‑negative step‑function that minimises the objective
    2·n·max(conv(a,a)) / (sum(a)²)

The routine works in three stages:

1. **Seed generation** – a compact but diverse set of initial vectors
   (uniform, geometric, power‑law, Gaussian, tent, and a few random
   Dirichlet draws).  The previous successful construction is reused if
   it is available.

2. **Equalisation descent** – a long mirror‑descent (exponentiated‑gradient)
   optimisation that simultaneously reduces the peak of the auto‑convolution
   and its variance.  The gradient of the variance term is obtained in a
   single FFT‑convolution; the sub‑gradient of the max‑term is obtained by
   averaging over the (usually one) active constraint.  The step‑size follows
   a 1/√t schedule.

3. **Annealing polish** – a short Metropolis‑Hastings walk that shifts a
   tiny amount of mass between two positions and occasionally performs a
   global Gaussian “jump”.  This helps escape shallow plateaus.

All phases respect the global time budget (default 1000 s).  The best
vector found is stored in the global variable ``height_sequence_1`` for
future runs and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via a real FFT (full mode).
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The sequence must be non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple families for seed vectors (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_simplex(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a point uniformly at random from the probability simplex."""
    x = rng.gamma(1.0, 1.0, size=n)
    total = x.sum()
    return x / total if total > 0 else _uniform(n)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a modest collection of start vectors, all normalised to sum = 1.
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # classic deterministic families
    seeds.append(_uniform(n))
    for r in (0.99, 0.95, 0.90, 0.80):
        seeds.append(_geom(n, r))
    for p in (0.0, 0.5, 1.0, 1.5):
        seeds.append(_power(n, p))
    seeds.append(_gaussian(n, sigma_factor=0.12))
    seeds.append(_gaussian(n, sigma_factor=0.22))
    seeds.append(_tent_window(n))

    # a few random Dirichlet vectors
    for _ in range(5):
        seeds.append(_random_simplex(n, rng))

    # reuse a previously discovered construction
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # clean tiny negative artefacts and re‑normalise
    clean = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            clean.append(s / total)
    return clean


# ----------------------------------------------------------------------
# Equalisation descent – minimise max‑conv + variance of the convolution
# ----------------------------------------------------------------------


def _equalise_descent(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    alpha: float = 0.5,
    beta: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on a surrogate objective that
    combines the sub‑gradient of the maximum convolution entry (weight α)
    and the gradient of the variance of the whole convolution (weight β).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)  # logits on the simplex
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # reconstruct the probability vector
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # --- active set for the max term ---------------------------------
        rel_tol = 1e-12
        active = np.where(conv >= max_val - rel_tol)[0]
        w_max = np.zeros_like(conv)
        w_max[active] = 1.0 / len(active)

        rev_a = a[::-1]

        # gradient of the (averaged) max‑constraint
        full_max = _conv_fft(rev_a, w_max)
        grad_max = 2.0 * full_max[n - 1 : n - 1 + n]

        # --- variance term ------------------------------------------------
        mu = (a.sum() ** 2) / (2 * n - 1)          # average convolution value
        diff = conv - mu
        full_var = _conv_fft(rev_a, diff)
        grad_var = 4.0 * full_var[n - 1 : n - 1 + n]

        # combined gradient
        grad = alpha * grad_max + beta * grad_var

        # 1/√t step‑size schedule
        lr = 0.6 / math.sqrt(it + 1.0)
        log_a = log_a - lr * grad

        # keep values numerically stable
        log_a -= np.max(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Equalise] n={n} it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polish
# ----------------------------------------------------------------------


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Tiny‑mass Metropolis walk with occasional global Gaussian jumps.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)  # modest initial temperature
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # move a tiny fraction of mass from i to j
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        delta = rng.random() * 0.02 * cur[i]  # up to 2 % of source mass
        delta = min(delta, cur[i])
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:6d} best={new_val:.6f}")

        # occasional larger “jump”
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= tot
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # reuse a previous construction if available

        except Exception:
            previous = None

    # ------------------------------------------------------------------
    n = 4000  # length of the step function
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous)

    rng = np.random.default_rng(seed + 999)
    for _ in range(7):

    # ------------------------------------------------------------------
    # 2) quick ranking of the raw seeds
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    best_seq = top_seeds[0].copy()
    best_val = _objective(best_seq)
    print(f"[init] best seed objective = {best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
