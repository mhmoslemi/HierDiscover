"""
Propose a non‑negative step function minimizing

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1.  Initialise from the best known construction (global `height_sequence_1` if
    available).  Down‑sample to a modest size (≈400) where a dedicated linear
    program can be solved fast.
2.  Linear‑programming step (Frank–Wolfe style):
        maximise   Σ g_i
        subject to  f * g ≤ M   (M = max(f*f)),   g ≥ 0 .
    The LP gives a direction `g` that puts mass where the current convolution
    is low.  A few convex mixes of `f` and `g` are tried and the best is kept.
    This loop runs while time permits.
3.  Upsample the improved coarse vector to the target length (piecewise‑constant
    blocks) and project onto the probability simplex.
4.  Refine the full‑size vector with a sub‑gradient descent that respects all
    active constraints (the gradient is averaged over every index attaining the
    maximal convolution value).  Nesterov momentum speeds up convergence.
5.  A soft‑max (log‑sum‑exp) objective is then minimised with Adam for a few
    seconds – this smooths the landscape and final‑tunes the tail.
6.  In the last seconds a tiny Gaussian jitter is applied to the current best
    and kept if it improves the exact score.

All updates are projected onto the simplex so the objective is scale invariant.
The routine respects a wall‑time budget (default 1000 s) and returns the best
sequence discovered.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a via FFT (exact for our sizes)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """Exact objective (scale‑invariant) for a on the simplex."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_self(a)
    return float(2.0 * a.shape[0] * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Down‑/up‑sampling utilities
# ----------------------------------------------------------------------


def downsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Coarse‑grain `vec` into `new_n` bins by summing contiguous blocks."""
    n = vec.shape[0]
    if new_n >= n:
        return vec.copy()
    # integer block size, last block may be larger
    block = n // new_n
    remainder = n % new_n
    out = np.empty(new_n, dtype=np.float64)
    start = 0
    for i in range(new_n):
        sz = block + (1 if i < remainder else 0)
        out[i] = vec[start:start + sz].sum()
        start += sz
    return out


def upsample_piecewise(const: np.ndarray, target_n: int) -> np.ndarray:
    """Expand a coarse vector to `target_n` by uniform blocks."""
    n_small = const.shape[0]
    block = target_n // n_small
    remainder = target_n % n_small
    out = np.empty(target_n, dtype=np.float64)
    idx = 0
    for i in range(n_small):
        sz = block + (1 if i < remainder else 0)
        out[idx:idx + sz] = const[i] / sz
        idx += sz
    # re‑normalise to avoid tiny drift
    out = out / out.sum()
    return out


# ----------------------------------------------------------------------
# Linear‑programming step (maximise Σ g_i  s.t.  f∗g ≤ M)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build the matrix A where (A @ g)[k] = (f * g)[k]  (linear convolution).
    Shape: (2*len(f)-1, len(f)).
    """
    n = f.shape[0]
    m = 2 * n - 1
    # we will fill the matrix in dense form and then convert to CSR – still cheap for n≈400
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        # contributes f[i] to rows i .. i+n-1 at columns 0 .. n-1
        A[i:i + n, :] += f[i]
    return sparse.csr_matrix(A)


def lp_step(f: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Perform one LP step: maximise Σ g_i under f∗g ≤ M, g ≥ 0.
    Returns the optimal g (may be dense).  Small eps guarantees strict positivity.
    """
    n = f.shape[0]
    # enforce strict positivity to keep the LP bounded
    f = np.maximum(f, eps)
    f = f / f.sum()
    M = conv_self(f).max()
    A = _build_conv_matrix(f)                # (2n-1) × n
    b = np.full(A.shape[0], M)               # upper bound for each convolution entry
    c = -np.ones(n)                          # maximise Σ g_i  → minimise -Σ g_i
    bounds = [(0, None)] * n
    # HiGHS handles sparse matrices efficiently
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback – random direction (should be rare)
        g = np.random.rand(n)
        g = project_simplex(g)
        return g
    return np.maximum(res.x, 0.0)


def mix_candidates(base: np.ndarray, direction: np.ndarray,
                   alphas: np.ndarray) -> tuple[np.ndarray, float]:
    """Try a few convex mixes of `base` and `direction`; return best mix."""
    best_vec = base.copy()
    best_score = compute_score(best_vec)
    for a in alphas:
        cand = (1.0 - a) * base + a * direction
        cand = project_simplex(cand)
        s = compute_score(cand)
        if s < best_score:
            best_score = s
            best_vec = cand.copy()
    return best_vec, best_score


# ----------------------------------------------------------------------
# Gradient‑based refinement (sub‑gradient + Adam)
# ----------------------------------------------------------------------


def subgradient_multi(a: np.ndarray, tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Compute a sub‑gradient of max(conv(a,a)) averaged over all active indices.
    Returns (g, max_val) where g is the gradient.
    """
    conv = conv_self(a)
    max_val = conv.max()
    # indices whose value is within tolerance of the maximum
    active = np.where(conv >= max_val - max(tol * abs(max_val), tol))[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        # shift = k - np.arange(n)  (gives the source index in a for each component)
        shift = k - np.arange(n)
        mask = (shift >= 0) & (shift < n)
        grad[mask] += 2.0 * a[shift[mask]]
    if active.size > 0:
        grad /= active.size
    return grad, max_val


def adam_optimize(a: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """Soft‑max (log‑sum‑exp) optimisation using Adam."""
    n = a.shape[0]
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    lr = 0.12
    beta_start, beta_end = 0.5, 400.0
    best_vec = a.copy()
    best_score = compute_score(best_vec)
    last_improve = time.time()
    while time.time() < deadline:
        conv = conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = time.time()
            print(f"[adam] new best = {best_score:.6f}")

        # temperature schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        beta = beta_start * (1.0 - progress) + beta_end * progress
        # softmax weights over all convolution entries
        w_raw = np.exp(beta * (conv - max_val))
        w = w_raw / w_raw.sum()
        # weighted shift = Σ_k w_k * shift_k(a)
        # This is precisely the convolution of w with reversed a
        a_rev = a[::-1]
        weighted = np.fft.ifft(np.fft.fft(w, n=_next_pow2(2 * n - 1)) *
                               np.fft.fft(a_rev, n=_next_pow2(2 * n - 1))).real
        weighted_shift = weighted[n - 1:n - 1 + n]  # centre part
        grad = a + weighted_shift

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.clip(a, 0.0, None)
        a = project_simplex(a)

        # occasional escaping moves
        if np.random.rand() < 0.015:
            a = random_move(a, lr * 0.3)
        if np.random.rand() < 0.005:
            a += np.random.randn(n) * 5e-4
            a = np.clip(a, 0.0, None)
            a = project_simplex(a)

        # stagnation handling
        if time.time() - last_improve > 20.0:
            jitter = np.random.rand(n) * 1e-3
            a = project_simplex(best_vec + jitter)
            last_improve = time.time()
            print("[adam] jitter restart")
    return best_vec, best_score

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the evaluation function.  The routine respects a wall‑time budget and
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 0)  Determine target length and initialise a good starting point
    # ------------------------------------------------------------------
    n_target = int(kwargs.get("n", 2000))
    print("[0] target length =", n_target)

    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if ref.shape[0] == n_target:
            a = np.maximum(ref, 0.0)
            a = a / a.sum()
            print("[0] using provided height_sequence_1")
        else:
            a = None
    else:
        a = None

    # fallback: simple analytic family (exponential tails) then normalise
    if a is None:
        i = np.arange(n_target, dtype=np.float64)
        d = np.minimum(i, n_target - 1 - i)
        lam = 0.5
        a = np.exp(-lam * d)
        a = a / a.sum()
        print("[0] fallback initialised with exponential tail")

    best_seq = a.copy()
    best_score = compute_score(best_seq)
    print(f"[0] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Coarse optimisation via linear programming (n_small ≈ 400)
    # ------------------------------------------------------------------
    n_small = min(400, n_target // 4)  # keep it small to stay fast
    a_small = downsample(best_seq, n_small)
    a_small = project_simplex(a_small)
    lp_deadline = start_time + 0.35 * budget_s  # ~35 % of total budget
    lp_iter = 0
    alphas = np.linspace(0.0, 1.0, 21)[1:-1]   # 0.05 … 0.95

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    a = best_seq.copy()

    momentum = np.zeros_like(a)
    beta_mom = 0.9
    last_improve = time.time()
    subgrad_deadline = start_time + 0.65 * budget_s   # next ~30 % of total budget
    it = 0
    step_start, step_end = 0.8, 1e-5

    print("[SG] start sub‑gradient descent")
    while time.time() < subgrad_deadline:
        grad, max_conv = subgradient_multi(a)
        # schedule a decreasing step size

        step = step_start * (1.0 - frac) + step_end * frac
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
        a_candidate = a - step * momentum

        if cur_score < best_score - 1e-12:

        if time.time() - last_improve > 15.0:
            jitter = np.random.rand(n) * 1e-3
            a_candidate = project_simplex(best_seq + jitter)

    total = time.time() - start_time
    print(f"[DONE] best score = {best_score:.6f}, length = {n_target}, time = {total:.1f}s")
    # ensure a modest scale (sum≈1) – already the case
    return best_seq.tolist()
