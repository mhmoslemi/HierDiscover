"""
Propose a non‑negative step‑function (list of heights) that minimises

        2·n·max(conv(a,a)) / (∑a)²

where ``conv`` is the full linear convolution.  The algorithm is a
time‑aware multi‑phase optimiser:

1. **Seed generation** – a wide collection of analytically motivated
   shapes (uniform, geometric, power‑law, symmetric, beta‑like,
   sine‑window, random Dirichlet, sparse patterns).  The previously
   discovered best sequence (if any) is also added.

2. **Adam on a smooth max** – the sequence is parametrised by log‑softmax
   variables ``v`` (so the iterate always lies on the probability simplex).
   A log‑sum‑exp with a growing temperature approximates the max‑operator.
   Adam with standard hyper‑parameters (with a corrected shift for numerical
   stability) performs many iterations; the best sequence encountered is
   recorded.

3. **Exact sub‑gradient mirror‑descent** – starting from the best Adam
   result we run many exponentiated‑gradient steps using the true
   sub‑gradient of ``max(conv)``.  This refines the solution without any
   smoothing.

4. **Simulated‑annealing hill climbing** – a cheap stochastic neighbourhood
   that randomly reshuffles the mass among a few positions (Dirichlet
   proposal).  Moves that improve the objective are always taken; uphill
   moves are accepted with a temperature‑controlled probability.
   The schedule is a little slower (cooling factor 0.9998) and the move
   size is occasionally enlarged (up to 6 indices) to escape shallow
   local minima.

The whole routine respects the supplied time budget (default 1000 s);
as soon as the deadline is hit the current best sequence is returned.
All helper functions are top‑level (no closures) and the search works
entirely in memory, using only ``numpy`` and the provided ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax producing a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming ∑seq == 1.
    This matches ``evaluate_sequence`` after normalisation.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_ramp(n: int) -> np.ndarray:
    """Increasing linear ramp, normalised."""
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def _gaussian_like(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_logits(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    return _softmax(logits)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed by a Dirichlet draw (conc < 1 makes it sparse)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Assemble a diverse set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_linear_ramp(n))
    seeds.append(_gaussian_like(n, sigma=n / 6.0))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # beta‑like shapes
    for a in (0.3, 0.5, 0.7, 1.2):
        for b in (0.3, 0.5, 0.7, 1.2):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random soft‑max vectors (logits)
    for _ in range(5):
        seeds.append(_random_logits(n, rng))

    # sparse patterns
    for k in (3, 5, 8, 12):
        for conc in (0.2, 0.5):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse previously discovered sequence if length matches
    if previous is not None and previous.shape == (n,):
        # normalise to sum = 1
        prev = previous.astype(np.float64)
        s = prev.sum()
        if s > 0:
            seeds.append(prev / s)

    return seeds

def _adam_optimize(
    seed_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 12000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a smoothed approximation of max(conv).
    ``seed_vec`` is assumed to sum to 1.
    """
    eps = 1e-30
    v = np.log(seed_vec + eps)               # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0               # initial smoothing coefficient
    beta_mul = 1.02          # geometric growth of smoothing

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        beta *= beta_mul
        conv = _conv_fft(a, a)

        shift = np.max(conv)                    # <<< corrected shift (was max(a) before)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w_k·conv_k w.r.t. a (linear convolution) ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]    # central n entries

        # ---- convert to gradient w.r.t. v (softmax Jacobian) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional gradient clipping for stability
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ---- Adam update on v ----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror‑descent on max(conv) using exponential updates.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t a (see analysis)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = a[rev_idx]  # derivative = a_{k-i}

        # exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                 # stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 600000,
    init_temp_factor: float = 0.025,
    min_temp: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing with a temperature schedule.
    Moves reshuffle the total mass of a small random subset using a
    Dirichlet proposal.  Occasionally a larger subset (up to 6 indices)
    is used to increase exploration.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    # initial temperature proportional to current objective
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling – very slow to retain acceptance probability
        T = max(T * 0.9998, min_temp)

        # pick a random subset: 2–6 indices (more indices every few hundred iters)
        if it % 200 == 0:
            k = rng.integers(4, 7)   # 4–6 indices
        else:
            k = rng.integers(2, 5)   # 2–4 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # fresh allocation respecting the total mass
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:6d} best={new_val:.6f}")

    return best_seq, best_val

    """

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    previous = globals().get("height_sequence_1")

        except Exception:
            previous = None
    else:

    # a few candidate lengths – longer lengths give finer discretisation

    global_best_seq = None
    global_best_val = float("inf")
    rng = np.random.default_rng(seed)

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------
        for _, s in top_seeds:
            if time.time() > deadline:

            if seq is not None and val < global_best_val:

                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,

            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Simulated‑annealing hill climbing (longer schedule)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _simulated_annealing(global_best_seq, n, deadline,
                                            max_iters=800000,
                                            init_temp_factor=0.025)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SA] n={n:5d} new global best={val:.6f}")

        # Use the best found sequence as a warm‑start for the next length
        previous = global_best_seq

    # ----------------------------------------------------------------------
    # Final sanity check with the provided evaluate_sequence (optional)
    # ----------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print("\n=== Finished ===")
    print(f"Best objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})")
    return global_best_seq.tolist()
