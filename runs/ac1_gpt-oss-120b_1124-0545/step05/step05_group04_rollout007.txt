"""
Propose a step‑function (list of non‑negative floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser consists of four stages:

1️⃣ Analytic families – a fast deterministic sweep over several
   parametric families (product‑law, sum‑law, exponential, Gaussian)
   to obtain a solid starting point.

2️⃣ (Optional) smooth‑max Adam optimisation – a short continuation
   on the soft‑max of the convolution to escape poor local minima.

3️⃣ Linear‑programming refinement – given the current distribution *f*
   (normalised to sum = 1) we solve the LP

        maximise   ∑ g_j
        subject to  (f * g)_k ≤ M   for all k,
                    g_j ≥ 0,

   where M = max(conv(f,f)).  The solution *g* carries as much total
   mass as possible without increasing the current maximal convolution.
   After normalising *g* the algorithm either replaces *f* by *g* or
   mixes them (convex combination) if that yields a strict decrease of
   the objective.  Repeating this step flattens the autocorrelation
   and drives the bound towards its theoretical minimum.

4️⃣ Final polishing – a few sub‑gradient steps on the true
   max‑convolution and a lightweight random‑mass‑transfer local search.

All phases respect the supplied time budget (default 1000 s) and the
best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """Linear (zero‑padded) self‑convolution via FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a (sum is normalised internally).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = u - cssv / (np.arange(n) + 1.0)
    mask = rho > 0
    if not np.any(mask):
        theta = 0.0
    else:
        rho_idx = np.where(mask)[0][-1]
        theta = cssv[rho_idx] / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def subgrad_step(a: np.ndarray, step: float) -> np.ndarray:
    """One projected sub‑gradient step for max(conv(a,a))."""
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    mask = (conv >= max_val - 1e-12).astype(np.float64)
    cnt = mask.sum()
    if cnt == 0:
        return a
    w = mask / cnt
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]
    a_new = a - step * grad
    return project_simplex(a_new)

# Analytic families

def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^(‑p).  Edge‑heavy shape."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^(‑p) + (n‑1‑i+off)^(‑p).  Two tails."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exp_decay(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(‑c·|i‑center|)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - center))
    a = a / a.sum()
    return a


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(‑½·((i‑center)/σ)²)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (Scipy)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _LP_AVAILABLE = True
except Exception:  # pragma: no cover
    _LP_AVAILABLE = False


def solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve   maximise   1ᵀ g
            subject to   (f * g)_k ≤ M   for all k,
                         g ≥ 0

    Returns the optimal g (may be None on failure).
    """
    if not _LP_AVAILABLE:  # pragma: no cover
        return None
    n = f.shape[0]
    size = 2 * n - 1

    # Build the constraint matrix A where A[k, j] = f[k-j] if 0≤k-j<n else 0
    # Column‑wise filling is fast because each column is a copy of f shifted.
    A = np.zeros((size, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    c = -np.ones(n, dtype=np.float64)               # maximise sum(g) → minimise -sum(g)
    b = np.full(size, M, dtype=np.float64)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        return None
    return res.x

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the supplied evaluation function.  The routine respects the given time
    budget and returns the best sequence found before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1️⃣ Analytic grid search – deterministic starter
    # ------------------------------------------------------------------
    best_a = np.ones(n) / n
    best_score = compute_score(best_a)

    # reuse a previously discovered construction if available
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n and ref.sum() > 0:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            if sc < best_score:
                best_score, best_a = sc, ref.copy()

    # sweep the analytic families (moderately fine grids)
    p_vals = np.linspace(0.30, 0.70, 61)
    offset_vals = np.linspace(0.001, 2.0, 61)
    c_vals = np.linspace(0.0005, 0.0200, 31)
    sigma_vals = np.linspace(100.0, 800.0, 31)

    for mode in ("product", "sum"):
        for p in p_vals:
            for off in offset_vals:
                if time.time() >= deadline:
                    break
                cand = analytic_product(n, p, off) if mode == "product" else analytic_sum(n, p, off)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score, best_a = sc, cand.copy()

    for c in c_vals:
        if time.time() >= deadline:
            break
        cand = analytic_exp_decay(n, c)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        cand = analytic_gaussian(n, sigma)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    print(f"[grid] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ (Optional) Adam on smooth‑max – a quick refinement
    # ------------------------------------------------------------------
    # The previous submission already performed a fairly long Adam run.
    # Here we keep a very short continuation just in case the LP later
    # starts from a better point.
    eps = 1e-12
    x = np.log(np.maximum(best_a, eps))

    def softmax(xv):
        m = np.max(xv)
        e = np.exp(xv - m)
        return e / e.sum()

    def compute_phi_and_grad(xv, lam):
        a = softmax(xv)
        conv = conv_fft_self(a)
        max_c = np.max(conv)
        shift = conv - max_c
        exp_term = np.exp(lam * shift)
        sum_exp = np.sum(exp_term)
        phi = max_c + np.log(sum_exp) / lam
        w = exp_term / sum_exp
        # gradient w.r.t. a
        w_rev = w[::-1]
        size = _next_pow2(len(w_rev) + n - 1)
        fft_w = np.fft.rfft(w_rev, n=size)
        fft_a = np.fft.rfft(a, n=size)
        conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
        slice_part = conv_wa[n - 1: 2 * n - 1]
        grad_a = 2.0 * slice_part[::-1]
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        return phi, grad_x

    def adam_optimize(x0, lam, steps, lr):
        m = np.zeros_like(x0)
        v = np.zeros_like(x0)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        xcur = x0.copy()
        for t in range(1, steps + 1):
            if time.time() >= deadline:
                break
            _, grad = compute_phi_and_grad(xcur, lam)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            xcur = xcur - step
        return xcur

    lam_sequence = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0,
                    2000.0, 8000.0, 20000.0, 80000.0]
    for lam in lam_sequence:
        if time.time() >= deadline:
            break
        # allocate a modest amount of Adam steps
        steps = 800 if lam <= 10 else 2000
        steps = min(steps, int((deadline - time.time()) / 0.002))
        if steps <= 0:
            break
        x = adam_optimize(x, lam, steps, lr=0.05)
        a_candidate = softmax(x)
        sc = compute_score(a_candidate)
        if sc < best_score:
            best_score, best_a = sc, a_candidate.copy()
            print(f"[adam λ={lam}] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Linear‑programming refinement (the core novelty)
    # ------------------------------------------------------------------
    f = best_a.copy()               # always kept normalised to sum = 1
    max_lp_iters = 30               # safe upper bound within the time budget

    while time.time() < deadline and max_lp_iters > 0:
        M = np.max(conv_fft_self(f))
        g_raw = solve_lp_max_sum(f, M)
        if g_raw is None:
            break
        sum_g = g_raw.sum()
        if sum_g == 0.0:
            break
        g_norm = g_raw / sum_g          # normalise to a probability vector

        # direct replacement if it already improves the max‑convolution
        M_g = np.max(conv_fft_self(g_norm))
        if M_g < M - 1e-12:
            f = g_norm
            best_a = f.copy()
            best_score = compute_score(f)
            print(f"[LP replace] new best = {best_score:.6f} (t={time.time()-start_time:.1f}s)")
            max_lp_iters -= 1
            continue

        # otherwise try a convex combination with decreasing mixing coefficients
        improved = False
        for t_mix in (0.5, 0.25, 0.1, 0.05):
            f_new = (1.0 - t_mix) * f + t_mix * g_norm
            f_new = f_new / f_new.sum()
            M_new = np.max(conv_fft_self(f_new))
            if M_new < M - 1e-12:
                f = f_new
                best_a = f.copy()
                best_score = compute_score(f)
                print(f"[LP mix t={t_mix}] new best = {best_score:.6f}")
                improved = True
                break
        if not improved:
            # no improvement → stop LP phase
            break
        max_lp_iters -= 1

    # ------------------------------------------------------------------
    # 4️⃣ Final polishing – sub‑gradient descent + tiny random moves
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        step = 0.5 / np.sqrt(it)
        a = subgrad_step(a, step)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[subgrad] new best = {best_score:.6f} (t={time.time()-start_time:.1f}s)")

    # light random‑mass‑transfer local search (still within remaining time)
    a = best_a.copy()
    scale = 0.02
    while time.time() < deadline:
        cand = random_move(a, scale)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[random] new best = {best_score:.6f} (t={time.time()-start_time:.1f}s)")
            a = cand
            scale = max(0.001, scale * 0.9)
        else:
            scale = min(0.05, scale * 1.001)
            a = cand

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    # Return plain Python list – the harness will evaluate it.
    return best_a.tolist()
