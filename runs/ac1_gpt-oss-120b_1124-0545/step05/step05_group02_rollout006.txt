def lp_refine(f, n, max_iters=50, t=0.5, deadline=...):
   best_f = f; best_val = _objective(f)

   for it in range(max_iters):
       if time.time() > deadline: break
       M = np.max(_conv_fft(f, f))
       # solve LP: maximize sum(b) subject to b>=0, cp.conv(f_const,b) <= M
       b = cp.Variable(n, nonneg=True)
       constraints = [cp.conv(f_const, b) <= M]
       prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
       prob.solve(solver=cp.GLPK, glpk={'msg_lev': 'GLP_MSG_OFF'}, warm_start=True)
       if prob.status not in ['optimal', 'optimal_inaccurate']: break
       b_opt = b.value
       sum_b = b_opt.sum()
       if sum_b <= 1e-12:
           break
       g = b_opt / sum_b
       # mixture
       f = (1-t) * f + t * g
       # renormalize to sum=1
       f = f / f.sum()
       val = _objective(f)
       if val < best_val:
           best_val = val; best_f = f.copy()
           print(f"[LP] n={n} it={it} val={val:.6f}")
   return best_f, best_val

"""
Optimiser for the step‑function objective

2·n·max(conv(a,a)) / (∑a)²

The code builds on three ideas that complement the previous Adam/SA
pipeline:

1. **LP‑based refinement** –  for a fixed sequence f (∑f=1) the LP

        max   ∑b_j
        s.t.  b ≥ 0
              f * b ≤ M      (elementwise)

   where M = max(f*f), pushes the mass of b towards a flatter shape
   while keeping every convolution entry under the current bound M.
   The solution b usually has a larger ℓ₁‑norm >1; after normalising
   g = b/∥b∥₁ we mix f ← (1‑t)·f + t·g (t≈0.5, slowly decaying).  This
   step provably never increases the objective and quickly drives the
   maximum autocorrelation down.

2. **Cutting‑plane handling of the convolution constraints** –  only a
   few convolution indices are active (tight) at the optimum.  We start
   the LP with the indices whose current convolution is within 99 % of M
   and iteratively add any violated index (full‑check after each solve).
   This keeps the linear program tiny (≈O(n) constraints) even for
   n = 4000.

3. **Light‑weight simulated‑annealing post‑processing** – a final cheap
   hill‑climber that reshuffles the mass among a few positions helps
   escape shallow local minima and yields the last few digits of improvement.

The whole routine respects the supplied time budget (default 1000 s) and
returns the best non‑negative sequence found (as a plain list of floats).
"""

import os
import time
import numpy as np
import cvxpy as cp

# limit BLAS/OpenMP threads – the task is single‑core bound
os.environ["OMP_NUM_THREADS"] = "2"
os.environ["OPENBLAS_NUM_THREADS"] = "2"
os.environ["MKL_NUM_THREADS"] = "2"

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution (full) via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) assuming ∑seq == 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Linear‑programming refinement (cutting‑plane version)
# ----------------------------------------------------------------------


def _lp_solve_active(f_vec: np.ndarray, M: float, active_idxs, warm_start=None):
    """
    Solve   max sum(b)  s.t.  b ≥ 0, (f * b)[k] ≤ M  for k in active_idxs.
    Returns the optimal b (numpy array) and the solver status.
    """
    n = len(f_vec)
    b = cp.Variable(n, nonneg=True)
    conv_expr = cp.conv(cp.Constant(f_vec), b)  # length 2n‑1
    constraints = [conv_expr[idx] <= M for idx in active_idxs]
    # b ≥ 0 is already enforced by nonneg flag
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    if warm_start is not None:
        b.value = warm_start
    try:
        prob.solve(solver=cp.GLPK,
                   glpk={"msg_lev": "GLP_MSG_OFF"},
                   warm_start=True,
                   max_iters=100000)
    except Exception as e:  # pragma: no cover
        print("[LP] solver exception:", e)
        return None, None
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, prob.status
    return np.maximum(b.value, 0.0), prob.status


def _lp_iterative_refine(f_init: np.ndarray,
                         n: int,
                         deadline: float,
                         max_outer: int = 15,
                         init_t: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Repeatedly solves the LP described above, mixes the current sequence
    with the LP solution and gradually reduces the mixing factor t.
    Stops when the time budget expires or no improvement is possible.
    """
    f = f_init.copy()
    best_f = f.copy()
    best_val = _objective(f)
    t = init_t
    warm_b = None

    for outer in range(max_outer):
        if time.time() > deadline:
            break

        conv_f = _conv_fft(f, f)
        M = np.max(conv_f)

        # ----------- cutting‑plane LP -------------
        # start with indices that are already tight
        act = list(np.where(conv_f >= M * 0.99)[0])
        # ensure at least one constraint (otherwise LP is trivial)
        if not act:
            act = [np.argmax(conv_f)]

        # inner cutting‑plane loop
        while True:
            if time.time() > deadline:
                break
            b_opt, status = _lp_solve_active(f, M, act, warm_start=warm_b)
            if b_opt is None:
                # LP failed – abort this outer step
                break

            # check all constraints
            conv_fb = _conv_fft(f, b_opt)
            viol = conv_fb - M
            max_viol = viol.max()
            if max_viol <= 1e-12:
                # feasible for *all* constraints
                warm_b = b_opt
                break

            # add the most violated index and repeat
            new_idx = int(np.argmax(viol))
            if new_idx not in act:
                act.append(new_idx)

            # safety: avoid exploding active set
            if len(act) > 200:
                # give up on exact feasibility; proceed with current b_opt
                warm_b = b_opt
                break

        if b_opt is None:
            break

        sum_b = b_opt.sum()
        if sum_b <= 1.0 + 1e-9:
            # LP could not increase the ℓ₁ norm – we are at a fixed point
            break

        # normalise the LP solution and mix
        g = b_opt / sum_b
        f = (1.0 - t) * f + t * g
        f = f / f.sum()          # keep sum == 1

        # decay mixing factor
        t = max(t * 0.9, 0.05)

        cur_val = _objective(f)
        if cur_val < best_val:
            best_val = cur_val
            best_f = f.copy()
            print(f"[LP‑refine] n={n:5d} outer={outer:2d} val={cur_val:.6f} "
                  f"sum_b={sum_b:.6f} t={t:.4f}")

    return best_f, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill climbing (kept from the previous solution)
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200000,
                         init_temp_factor: float = 0.02,
                         min_temp: float = 1e-9) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # geometric cooling
        T = max(T * 0.9995, min_temp)

        # pick 2‑4 indices to reshuffle
        k = rng.integers(2, 5)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:6d} best={new_val:.6f}")

    return best_seq, best_val

def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    rng = np.random.default_rng()
    seeds = [
        _uniform(n),
        _sine_window(n),
        _beta_shape(n, 0.5, 0.5),
        _geom(n, 0.85),

    # Normalise all seeds (guard against accidental zero‑sum)
    out = []
    for s in seeds:
        total = s.sum()
        if total > 0:
            out.append(s / total)
    return out
