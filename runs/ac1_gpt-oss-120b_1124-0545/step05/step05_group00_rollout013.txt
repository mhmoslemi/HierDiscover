    while time.time()<deadline and iteration < max_iters:
        compute M = max(conv(f,f))
        # Build matrix A
        rows = []
        cols = []
        data = []
        for i in range(n):
            # for row index k = i to i+n-1:
            start = i
            end = i+n  # exclusive
            # rows i..i+n-1, col i, values f[0..n-1]
            rows.extend(range(start, end))
            cols.extend([i]*n)
            data.extend(f.tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        # Solve LP: minimize -sum(g) s.t. A @ g <= M, g >= 0
        res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print("LP failed", res.message)
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break
        g_scaled = g / sum_g
        # Mix
        f = (1 - t) * f + t * g_scaled
        # optionally update t schedule: maybe t = 0.5 always.
    return f

def _lp_refine(start_seq, n, deadline, max_iter=10, t=0.1):
    import scipy.sparse as sp
    import scipy.optimize as opt

    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iter+1):
        if time.time() > deadline:
            break
        M = _conv_fft(a, a).max()
        # Build matrix A
        # Build data arrays for each diagonal
        # Use sp.diags
        diag_vals = [np.full(n, a[d]) for d in range(n)]
        offsets = np.arange(n)
        A = sp.diags(diag_vals, offsets, shape=(2*n-1, n), format='csr')
        # bound vector b
        b = np.full(2*n-1, M, dtype=np.float64)
        # Solve LP: max sum(g) i.e., min -sum(g)
        c = -np.ones(n, dtype=np.float64)
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print("[LP] iteration", it, "failed:", res.message)
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g
        # Mix
        a = (1 - t) * a + t * g_norm
        # Evaluate
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LP] n={n} it={it} best={cur_val:.6f}")
        # Optionally reduce t after some steps
        t = max(0.01, t * 0.9)  # gradually lower step size

    return best_seq, best_val

"""
Propose a non‑negative step‑function (a list of heights) that minimizes

        2·n·max(conv(a,a)) / (sum(a)²)

The optimiser combines several complementary ideas:

1.  A rich set of deterministic seeds (uniform, geometric, power‑law,
    classic spectral windows, a triangular “tent”, plus a few random
    soft‑max vectors).  The previous best sequence (if any) is also
    re‑used as a seed.

2.  **Smoothed‑max Adam** – the sequence is kept on the simplex by
    optimising logits with Adam on a log‑sum‑exp surrogate of the
    maximum auto‑convolution.  A modest number of iterations is enough
    to move away from the raw seeds.

3.  **Linear‑programming refinement** – for the current vector `f` we
    compute `M = max(conv(f,f))` and solve the LP

        maximise   Σ g_i
        subject to  conv(f,g) ≤ M     (for every shift)
                    g_i ≥ 0

    The LP (solved with SciPy’s HiGHS interior‑point solver) produces a
    vector `g` that can carry *more* total mass while respecting the
    same convolution bound.  After normalising `g` we move a small step
    `t` towards it:  `f ← (1‑t)·f + t·g`.  Repeating this step (with a
    decreasing step size) yields a strict monotone decrease of the true
    objective.

4.  **Exact sub‑gradient mirror descent** – the true non‑smooth
    objective is refined using the averaged sub‑gradient of all active
    (maximising) convolution constraints.  An exponentiated‑gradient
    update keeps the iterate on the simplex.

5.  **Metropolis annealing** – the remaining budget is spent on a cheap
    Metropolis–Hastings walk that swaps tiny amounts of mass between
    two random positions and occasionally performs a global Gaussian
    perturbation.

All phases respect the global time budget (default 1000 s).  The best
sequence found is stored in the global variable ``height_sequence_1``
for possible reuse by later runs.
"""

# Low‑level convolution utilities (FFT based for speed)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used for reporting – assumes non‑negative entries."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse pool of normalised starting vectors."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # pure power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # combined geometric / power‑law
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classic windows and simple shapes
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))

    # random soft‑max vectors
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # reuse previous construction if size matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final sanity pass
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp surrogate of max(conv(a,a))."""
    v = np.log(seed + 1e-30)   # logits = log probabilities

    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.9
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 2.0
    beta_mul = 1.005

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                    # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:4d} it={it:4d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta = beta * beta_mul
        shift = np.max(_conv_fft(a, a))
        w = np.exp(beta * (_conv_fft(a, a) - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)         # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # Jacobian of softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on logits
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val

# Exact sub‑gradient refinement (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 120000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent using the true non‑smooth objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        if len(active) > 0:
            g /= float(len(active))

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)            # keep numbers bounded
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:4d} it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val

def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Final polishing – tiny mass swaps plus occasional global jumps."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # upto 2 % of source mass
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n:4d} it={it:6d} best={new_val:.6f}")

        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n:4d} it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (new ingredient)
# ----------------------------------------------------------------------
def _lp_refine(start_seq: np.ndarray,
               n: int,
               deadline: float,
               max_iter: int = 8,
               t_init: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Iteratively improve the sequence via a LP that maximises the total
    mass while keeping the convolution with the *current* vector bounded
    by the present worst‑case value.
    """
    if not SCIPY_AVAILABLE:
        return start_seq, _objective(start_seq)

    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    t = t_init

    for it in range(1, max_iter + 1):
        if time.time() > deadline:
            break

        # current bound M
        M = _conv_fft(a, a).max()

        # build convolution matrix A where A @ g = conv(a,g)
        # each diagonal k (0 ≤ k < n) has the constant a[k]
        diags_data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
        offsets = np.arange(n, dtype=int)
        A = _sp.diags(diags_data, offsets, shape=(2 * n - 1, n), format='csr')

        # LP: maximise Σ g_i  ⇔ minimise -Σ g_i
        c = -np.ones(n, dtype=np.float64)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        res = _opt.linprog(c, A_ub=A, b_ub=b,
                           bounds=[(0, None)] * n, method='highs')

        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        g = res.x
        s = g.sum()
        if s <= 0:
            break

        g_norm = g / s                     # keep sum = 1
        a = (1 - t) * a + t * g_norm       # convex combination

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LP] n={n:4d} it={it:3d} best={cur_val:.6f}")

        # slowly decay the step size
        t = max(0.01, t * 0.85)

    return best_seq, best_val

    # lower bound but consume more memory for the LP.
    lengths = [2500, 1800]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1) Assemble a diverse set of seeds
        # --------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # --------------------------------------------------------------
        # 2) Quick evaluation – keep the few most promising seeds
        # --------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_k = 5
        top_seeds = [s for s, _ in seed_vals[:top_k]]

        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] new global best = {global_best_val:.6f}")

        # --------------------------------------------------------------
        # Phase A – Adam on a smoothed‑max surrogate (few iterations)
        # --------------------------------------------------------------
        for idx, sv in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(sv, n, deadline, max_iters=3000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] seed={idx} best={val:.6f}")

        # --------------------------------------------------------------
        # Phase B – Linear‑programming refinement (the new ingredient)
        # --------------------------------------------------------------
        if global_best_seq is not None:
            if time.time() < deadline:
                seq, val = _lp_refine(global_best_seq, n, deadline,
                                      max_iter=8, t_init=0.15)
                if seq is not None and val < global_best_val:
                    global_best_val = val
                    global_best_seq = seq.copy()
                    print(f"[overall LP] best={val:.6f}")

        # --------------------------------------------------------------
        # Phase C – Exact sub‑gradient mirror descent
        # --------------------------------------------------------------
        if global_best_seq is not None:
            if time.time() < deadline:
                seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                               max_iters=120000)
                if seq is not None and val < global_best_val:
                    global_best_val = val
                    global_best_seq = seq.copy()
                    print(f"[overall SubGrad] best={val:.6f}")

        # --------------------------------------------------------------
        # Phase D – Metropolis annealing polish (remaining time)
        # --------------------------------------------------------------
        if global_best_seq is not None:
            if time.time() < deadline:
                seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                            max_iters=None)
                if seq is not None and val < global_best_val:
                    global_best_val = val
                    global_best_seq = seq.copy()
                    print(f"[overall Anneal] best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(2000)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")

    # Store for possible reuse by later runs
    globals()["height_sequence_1"] = global_best_seq.copy()
    return global_best_seq.tolist()
