"""
Propose a step‑function (non‑negative vector) that minimises
`evaluate_sequence`.  The search proceeds in three stages:

1. **Fast parametric / random search** – evaluate a pool of simple
   families (arcsine, uniform, symmetric and asymmetric Beta
   distributions, and mixtures) and keep the best one.

2. **Projected Adam optimisation** – minimise the true objective
   `2·n·max(conv(a,a))` using the sub‑gradient of the max‑convolution.
   An Adam‑style adaptive step‑size is used, after each step the vector
   is projected back onto the probability simplex.

3. **Local stochastic refinement** – simulated‑annealing style random
   mass transfers between two coordinates.  Moves that improve the
   exact objective are always accepted; worsening moves are accepted
   with a temperature‑dependent probability.

The whole routine respects the 1000 s time budget, prints intermediate
scores and finally returns the best sequence found as a Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Maximum value of the linear convolution a*a (using FFT)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv.max())


def evaluate_vec(a: np.ndarray, fft_len: int) -> float:
    """
    Exact value of `evaluate_sequence` for a normalised vector
    (i.e. sum(a) == 1).  Avoids calling the external function.
    """
    mx = max_convolution(a, fft_len)
    return float(2.0 * a.size * mx)


def arcsine_distribution(n: int) -> np.ndarray:
    """
    Discretisation of the arcsine density f(x)=1/(π√(x(1‑x))) on [0,1].
    The vector is already normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Normalised discretisation of the Beta(α,β) density on a uniform grid.
    Uses bin–centres (i+0.5)/n for evaluation.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def subgradient_max_conv(a: np.ndarray,
                         conv_vals: np.ndarray,
                         max_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) with respect to a.
    For each maximising shift s the contribution is 2·a shifted by s.
    The returned gradient is the average over all maximisers.
    """
    n = a.size
    g = np.zeros_like(a)
    for s in max_idx:
        i_start = max(0, s - (n - 1))
        i_end = min(n - 1, s)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = s - i
        g[i] += 2.0 * a[j]
    if max_idx.size > 0:
        g /= float(max_idx.size)
    return g

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The algorithm consists of a fast random sweep,
    a projected Adam optimiser, and a final stochastic refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin
    overall_deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))
    # FFT length – next power of two ≥ 2*n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Phase 0 – build a modest pool of deterministic candidates
    # ------------------------------------------------------------------
    candidates = []

    # reuse an existing construction if available
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.size == n and a0.sum() > 0.0:
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # arcsine and uniform baselines
    candidates.append(arcsine_distribution(n))
    candidates.append(np.full(n, 1.0 / n))

    # symmetric Beta sweep (α = β)
    for alpha in np.linspace(0.25, 2.0, 16):
        candidates.append(beta_distribution(n, alpha, alpha))

    # mixtures of arcsine and uniform
    arcsine = candidates[-4] if len(candidates) >= 4 else arcsine_distribution(n)
    uniform = candidates[-3] if len(candidates) >= 3 else np.full(n, 1.0 / n)
    for lam in np.linspace(0.0, 1.0, 11):
        mix = lam * arcsine + (1.0 - lam) * uniform
        mix = mix / mix.sum()
        candidates.append(mix)

    # ------------------------------------------------------------------
    # 0a) Quick evaluation of the deterministic pool
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    for idx, a in enumerate(candidates):
        sc = evaluate_vec(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[init {idx:02d}] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0b) Randomised exploration of Beta families
    # ------------------------------------------------------------------
    # allocate ~10 % of the budget for this cheap random sweep
    random_deadline = start_time + 0.10 * budget_s
    while time.time() < random_deadline:
        # sample asymmetric Beta parameters (log‑uniform)
        log_alpha = rng.uniform(-1.0, 0.7)   # α ∈ [10⁻¹, 10⁰·⁷] ≈ [0.1,5]
        log_beta = rng.uniform(-1.0, 0.7)
        alpha = 10 ** log_alpha
        beta = 10 ** log_beta
        a = beta_distribution(n, alpha, beta)
        sc = evaluate_vec(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[rand β] α={alpha:.3f} β={beta:.3f}  → {best_score:.6f}")

        # occasionally try a mixture of two independent Betas
        if rng.random() < 0.15:
            a1 = beta_distribution(
                n,
                10 ** rng.uniform(-1.0, 0.7),
                10 ** rng.uniform(-1.0, 0.7))
            a2 = beta_distribution(
                n,
                10 ** rng.uniform(-1.0, 0.7),
                10 ** rng.uniform(-1.0, 0.7))
            lam = rng.random()
            mix = lam * a1 + (1.0 - lam) * a2
            mix = mix / mix.sum()
            sc = evaluate_vec(mix, fft_len)
            if sc < best_score:
                best_score = sc
                best_a = mix.copy()
                print(f"[rand mix] λ={lam:.2f}  → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Projected Adam optimiser on the true L∞‑objective
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.70 * budget_s
    a = best_a.copy()
    m = np.zeros_like(a)            # first moment
    v = np.zeros_like(a)            # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.6                         # base learning rate
    it = 0
    last_improve = time.time()

    while time.time() < phase1_deadline:
        # convolution via FFT
        fa = np.fft.fft(a, fft_len)
        conv_vals = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        conv_max = float(conv_vals.max())
        max_idx = np.where(conv_vals >= conv_max - 1e-14)[0]

        # sub‑gradient (average over all maximisers)
        g = subgradient_max_conv(a, conv_vals, max_idx)

        # Adam update (gradient descent, so we move opposite to g)
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a - step)

        cur_score = 2.0 * n * conv_max
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam {it:07d}] improved to {best_score:.6f}")

        # random restart if we are stuck
        if time.time() - last_improve > 30.0:
            noise = rng.normal(scale=0.02, size=n)
            a = proj_simplex(best_a + noise)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[Adam {it:07d}] restart after stagnation")

    # ------------------------------------------------------------------
    # Phase 2 – Local stochastic refinement (simulated annealing)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 5.0:
        a = best_a.copy()
        best_conv_max = best_score / (2.0 * n)

        # linear temperature schedule
        T_start = best_score * 0.01
        T_end = best_score * 1e-7
        sa_start = time.time()
        sa_it = 0

        while time.time() < overall_deadline:
            elapsed = time.time() - sa_start
            frac = min(1.0, elapsed / remaining)
            T = T_start * (1.0 - frac) + T_end * frac

            # propose a random elementary mass transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            max_move = min(a[i], 0.05 * a[i] + 1e-12)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            new_conv_max = max_convolution(a_new, fft_len)
            new_score = 2.0 * n * new_conv_max

            # Metropolis acceptance
            if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    print(f"[SA {sa_it:07d}] new best = {best_score:.6f} (t={time.time() - start_time:.1f}s)")

            sa_it += 1

    # ------------------------------------------------------------------
    # Finalisation – make sure the vector is a proper probability distribution
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    elapsed_total = time.time() - start_time
    print(f"[done] total time = {elapsed_total:.1f}s, best score = {best_score:.6f}, length = {n}")
    return best_a.tolist()
