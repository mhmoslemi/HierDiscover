"""
Propose a non‑negative step function that minimises `evaluate_sequence`.

Algorithm (≈1000 s budget)

1. **Initialisation** – start from the arcsine (Beta(0.5,0.5)) distribution,
   which is known to give a good baseline.

2. **LP‑driven improvement** – given the current vector `f`,
   compute `M = max(conv(f,f))` and solve the linear programme

        maximise    Σ g_j
        subject to  conv(f, g) ≤ M      (2·n‑1 linear constraints)
                    g ≥ 0

   The LP seeks a distribution that spreads as much mass as possible while
   keeping the convolution with `f` below the current worst value `M`.
   The solution `g` is normalised to unit sum and blended with `f`
   (`f ← (1‑τ)·f + τ·g`).  A handful of such iterations already moves the
   vector into a much better region.

3. **Projected sub‑gradient descent** – minimise the exact objective
   `J(f)=2·n·max(conv(f,f))` on the simplex by stepping opposite to the
   sub‑gradient of the max‑convolution and projecting back onto the simplex.
   A decreasing step‑size `α/√t` together with occasional stochastic restarts
   keeps the search exploratory.

4. **Local stochastic refinement** – simulated‑annealing style elementary
   mass transfers between two random coordinates are accepted if they improve
   the score or with a Metropolis probability depending on a linear temperature
   schedule.

The three phases respect the global time limit, log intermediate scores and
return the best vector found (as a plain Python list).  All numbers are
non‑negative and the vector is normalised to sum = 1, which is safe because
`evaluate_sequence` is homogeneous.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """
    Compute the maximum entry of the linear convolution a * a
    using FFT of length `fft_len`.  Returns a Python float.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv.max())


def arcsine_distribution(n: int) -> np.ndarray:
    """
    Discretise the continuous arcsine density
        f(x) = 1/(π√(x(1−x)))   on [0,1]
    by integrating over each bin [i/n, (i+1)/n).  The result already sums to 1.
    """
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def subgradient_max_conv(a: np.ndarray,
                         conv_vals: np.ndarray,
                         max_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    For each maximising shift s, the contribution is 2·a shifted by s.
    The returned gradient is the average over all maximisers.
    """
    n = a.size
    g = np.zeros_like(a)
    for s in max_idx:
        i_start = max(0, s - (n - 1))
        i_end = min(n - 1, s)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = s - i
        g[i] += 2.0 * a[j]
    if max_idx.size > 0:
        g /= float(max_idx.size)
    return g


def build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build a sparse matrix A (shape (2·n‑1, n)) such that
        (A @ g)[k] = (f * g)[k]   (linear convolution)
    for all k.  Used as the constraint matrix in the LP
    ``conv(f, g) ≤ M``.
    """
    n = f.shape[0]
    conv_len = 2 * n - 1
    # Pre‑allocate lists for CSR data
    data = []
    rows = []
    cols = []

    for k in range(conv_len):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        # corresponding indices i = k - j
        i_vals = k - np.arange(j_start, j_end + 1, dtype=int)
        coeffs = f[i_vals]
        rows.extend([k] * (j_end - j_start + 1))
        cols.extend(range(j_start, j_end + 1))
        data.extend(coeffs.tolist())

    return sparse.csr_matrix((data, (rows, cols)), shape=(conv_len, n))


def evaluate_vec(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of the objective for a normalised vector (Σ a = 1),
    i.e. the value returned by `evaluate_sequence`.
    """
    mx = max_convolution(a, fft_len)
    return float(2.0 * a.size * mx)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The algorithm consists of three phases:
    LP‑driven improvement, projected sub‑gradient descent and a final
    simulated‑annealing refinement.  It respects the time budget and
    prints progress information.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 2000))                 # default length
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len).bit_length())       # next power of two

    # --------------------------------------------------------------
    # 0) Initialise a pool of cheap candidates
    # --------------------------------------------------------------
    candidates = []

    # reuse a previously discovered construction if present
    if "height_sequence_1" in globals():
        h1 = np.asarray(globals()["height_sequence_1"], dtype=float)
        h1 = np.maximum(h1, 0.0)
        if h1.size == n and h1.sum() > 0.0:
            h1 = h1 / h1.sum()
            candidates.append(h1)

    # arcsine (Beta 0.5,0.5) – our main baseline
    candidates.append(arcsine_distribution(n))

    # uniform – useful for sanity checks
    candidates.append(np.full(n, 1.0 / n))

    # a few symmetric beta families (α = β)
    for alpha in np.linspace(0.2, 2.0, 20):
        # evaluate a symmetric beta by the centre‑point rule
        x = (np.arange(n, dtype=float) + 0.5) / n
        a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        if a.sum() == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a.sum()
        candidates.append(a)

    # --------------------------------------------------------------
    # 1) Pick the best cheap candidate (exact evaluation)
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    for idx, a in enumerate(candidates):
        sc = evaluate_vec(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[init {idx:02d}] new best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Linear‑programming driven improvement
    # --------------------------------------------------------------
    max_lp_iters = 12                               # stay well within budget
    a = best_a.copy()
    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        # current worst convolution value
        M = max_convolution(a, fft_len)

        # build the constraint matrix conv(a, g) ≤ M
        A = build_conv_matrix(a)
        b = np.full(conv_len, M)

        # linear programme: maximise Σ g_j  ⇔ minimise -Σ g_j
        c = -np.ones(n)
        bounds = [(0, None)] * n

        # give the solver a small piece of the remaining time
        remaining = deadline - time.time()
        lp_options = {"presolve": True}
        # recent SciPy versions accept `time_limit` (seconds) for HiGHS
        if remaining > 0:
            lp_options["time_limit"] = max(0.1, remaining / 5.0)

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method="highs", options=lp_options)

        if not res.success:
            print(f"[LP {lp_it}] linprog failed: {res.message}")
            break

        g_raw = res.x
        sum_g = g_raw.sum()
        if sum_g <= 0.0:
            g = np.full(n, 1.0 / n)
        else:
            g = g_raw / sum_g

        # convex blend – decreasing step size
        tau = 0.5 / (1.0 + lp_it)
        a = (1.0 - tau) * a + tau * g

        cur_score = evaluate_vec(a, fft_len)
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_a = a.copy()
            print(f"[LP {lp_it}] improved best = {best_score:.6f}")

        # early stop if improvement becomes negligible
        if lp_it >= 5 and abs(cur_score - best_score) < 1e-9:
            break

    # --------------------------------------------------------------
    # 3) Projected sub‑gradient descent on the exact objective
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.60 * budget_s   # ~60 % of total time
    it = 0
    last_improve = time.time()
    a = best_a.copy()
    while time.time() < phase2_deadline:
        # convolution via FFT (exact)
        conv_vals = np.real(np.fft.ifft(
            np.fft.fft(a, fft_len) * np.fft.fft(a, fft_len)))[:conv_len]
        conv_max = float(conv_vals.max())
        max_idxs = np.where(np.abs(conv_vals - conv_max) < 1e-14)[0]

        # sub‑gradient (average over all maximisers)
        g = subgradient_max_conv(a, conv_vals, max_idxs)

        it += 1
        lr = 0.15 / np.sqrt(it)                    # modest step‑size schedule
        a = proj_simplex(a - lr * g)

        # evaluate
        cur_score = 2.0 * n * conv_max
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[PGD {it:07d}] improved to {best_score:.6f}")

        # occasional random restart when stuck
        if time.time() - last_improve > 30.0:
            noise = rng.normal(scale=0.02, size=n)
            a = proj_simplex(best_a + noise)
            last_improve = time.time()
            print(f"[PGD {it:07d}] restart after stagnation")

    # --------------------------------------------------------------
    # 4) Local stochastic refinement (simulated annealing)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a = best_a.copy()
        best_conv_max = best_score / (2.0 * n)

        T_start = best_score * 0.01
        T_end = best_score * 1e-6
        SA_start = time.time()
        sa_it = 0

        while time.time() < deadline:
            elapsed = time.time() - SA_start
            frac = min(1.0, elapsed / remaining)
            T = T_start * (1.0 - frac) + T_end * frac

            # elementary mass transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            max_move = a[i]
            if max_move == 0.0:
                continue
            delta = rng.random() * 0.05 * max_move

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            new_conv_max = max_convolution(a_new, fft_len)
            new_score = 2.0 * n * new_conv_max

            # Metropolis acceptance
            if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    print(f"[SA {sa_it:07d}] new best = {best_score:.6f}")

            sa_it += 1

    # --------------------------------------------------------------
    # Final normalisation and return
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    elapsed = time.time() - start_time
    print(f"[done] total time = {elapsed:.1f}s, best score = {best_score:.6f}, n = {n}")
    return best_a.tolist()
