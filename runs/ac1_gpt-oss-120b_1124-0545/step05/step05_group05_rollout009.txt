"""
Improved search for a non‑negative step function minimizing

        2·n·max(conv(a,a)) / (sum(a)**2)          (scale‑invariant)

The algorithm consists of three stages, each respecting the wall‑time budget:

1. **Warm‑start** – a few seconds of random Dirichlet draws to obtain a solid
   starting point (or reuse the previously discovered ``height_sequence_1``).

2. **Active‑set sub‑gradient descent** – rather than using a single active
   convolution index, the top‑few convolution values are taken together.
   The exact gradient of a selected set of shifts is obtained by simple
   slice operations (O(k·n) work), and a Nesterov‑style momentum update with
   a decaying step size is applied.  Periodic jitter and random mass‑transfer
   moves help to escape plateaus.

3. **Soft‑max (log‑sum‑exp) optimisation with Adam** – a smooth approximation of
   the max is minimised.  The gradient of the soft‑max objective is computed
   exactly via two FFT‑based convolutions (linear‑time in *n* up to a small
   constant factor).  Adam updates together with projection onto the
   probability simplex drive the solution toward lower maximum self‑convolution.

4. **Final random refinement** – a short Gaussian‑perturbation stage that
   greedily accepts any improvement.

All updates are projected onto the simplex (∑a_i = 1, a_i ≥ 0) so the
objective remains scale‑invariant.  The routine prints intermediate scores
and always returns the best sequence found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of a and b using FFT (real‑valued)."""
    size = _next_pow2(a.shape[0] + b.shape[0] - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a * a (linear, not circular)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Move a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Scoring utilities
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective (scale‑invariant) for a probability vector."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_self(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------


def gradient_from_shifts(a: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """
    Exact gradient of Σ_{s∈shifts} (a ⋆ a)[s] w.r.t. a.
    ``shifts`` are absolute distances (0,…,n‑1) – each counted once.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for s in shifts:
        if s == 0:
            g += 2.0 * a
        else:
            # contribution a_{i-s}
            g[s:] += a[: n - s]
            # contribution a_{i+s}
            g[: n - s] += a[s:]
    return g


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max (log‑sum‑exp) approximation.
    ``w`` is a weight vector of length L = 2·n‑1 (sums to 1).
    """
    n = a.shape[0]
    L = w.shape[0]
    # left part: Σ_s w_s a_{i-s}
    conv_left = conv_two(w, a)            # length = 3·n‑2
    g_left = conv_left[:n]

    # right part: Σ_s w_s a_{i+s}
    w_rev = w[::-1]
    conv_right = conv_two(w_rev, a)       # length = 3·n‑2
    offset = L - 1
    g_right = conv_right[offset: offset + n]

    return g_left + g_right

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """Return a non‑negative step function (list of floats) with low score."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Parameter setup
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))
    print("[init] problem size n =", n)

    # ------------------------------------------------------------------
    # 0) Initial solution (reuse previous construction if possible)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        if cand0.shape[0] != n:
            # adjust size via truncation / padding
            if cand0.shape[0] > n:
                cand0 = cand0[:n]
            else:
                cand0 = np.concatenate([cand0,
                                        np.zeros(n - cand0.shape[0])])
        cand0 = np.maximum(cand0, 0.0)
        cand0 = cand0 / cand0.sum()
        best_seq = cand0.copy()
    else:
        best_seq = np.random.dirichlet(np.ones(n))
    best_score = compute_score(best_seq)
    print(f"[init] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 0a) Quick Dirichlet random search (≈10 % of time)
    # ------------------------------------------------------------------
    dirichlet_end = start_time + 0.1 * budget_s
    while time.time() < dirichlet_end:
        cand = np.random.gamma(1.0, 1.0, size=n)
        cand = cand / cand.sum()
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[rand] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Active‑set sub‑gradient descent
    # ------------------------------------------------------------------
    a = best_seq.copy()
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    last_improve = time.time()
    stagnation_limit = 12.0           # seconds without improvement → jitter
    step_start, step_end = 0.8, 1e-5
    top_k = 5                         # number of convolution peaks used

    phase1_deadline = start_time + 0.55 * budget_s   # roughly first half
    it = 0
    while time.time() < phase1_deadline:
        conv = conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[sg {it:07d}] new best = {best_score:.6f}")

        # ----- active set gradient (top‑k peaks) -----
        # pick the indices of the largest convolution values
        top_idx = np.argpartition(-conv, range(top_k))[:top_k]
        # convert to absolute shift distances (0 … n‑1) and deduplicate
        shifts = np.abs(top_idx - (n - 1))
        shifts = np.unique(shifts)

        g = gradient_from_shifts(a, shifts)

        # ----- Nesterov‑style update -----
        progress = (time.time() - start_time) / (phase1_deadline - start_time)
        step = step_start * (1.0 - progress) + step_end * progress

        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        a_candidate = a - step * momentum
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate = project_simplex(a_candidate)

        # occasional small mass‑transfer move
        if np.random.rand() < 0.02:
            a_candidate = random_move(a_candidate, step * 0.5)

        # jitter restart if stagnating
        if time.time() - last_improve > stagnation_limit:
            jitter = np.random.rand(n) * 1e-3
            a_candidate = best_seq + jitter
            a_candidate = project_simplex(a_candidate)
            last_improve = time.time()
            print(f"[sg {it:07d}] jitter restart")

        a = a_candidate
        it += 1

    # ------------------------------------------------------------------
    # 2) Soft‑max optimisation with Adam (exact gradient via FFT)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_adam = 0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12
    beta_start, beta_end = 0.5, 800.0   # temperature schedule
    last_improve = time.time()
    stagnation_limit = 18.0

    phase2_deadline = start_time + 0.95 * budget_s   # leave ~5 % for final refinement
    while time.time() < phase2_deadline:
        conv = conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[adam {t_adam:07d}] new best = {best_score:.6f}")

        # ----- soft‑max weights -----
        elapsed = time.time() - start_time
        total_phase = budget_s * 0.95
        prog = min(1.0, elapsed / total_phase)
        beta = beta_start * (1.0 - prog) + beta_end * prog

        # stabilise exponentials
        w_raw = np.exp(beta * (conv - cur_max))
        w = w_raw / w_raw.sum()

        # ----- exact gradient of soft‑max objective -----
        g = gradient_from_weights(a, w)

        # ----- Adam update -----
        t_adam += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step_vec
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate = project_simplex(a_candidate)

        if np.random.rand() < 0.005:

            a_candidate = a_candidate + jitter
            a_candidate = np.clip(a_candidate, 0.0, None)
            a_candidate = project_simplex(a_candidate)

        # jitter restart on stagnation
        if time.time() - last_improve > stagnation_limit:
            jitter = np.random.rand(n) * 1e-3
            a_candidate = best_seq + jitter
            a_candidate = project_simplex(a_candidate)
            last_improve = time.time()
            print(f"[adam {t_adam:07d}] jitter restart")

        a = a_candidate

    # ------------------------------------------------------------------
    # 3) Final random refinement (last few seconds)

        cand = np.clip(cand, 0.0, None)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, length = {n}, time = {total_time:.1f}s")
