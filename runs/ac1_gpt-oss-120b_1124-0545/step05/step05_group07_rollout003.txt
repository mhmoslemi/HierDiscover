def propose_candidate(seed=...):
    set random
    start_time = time.time()
    n = 2000 (or adapt maybe 2500)
    # initialize from height_sequence_1 if present else analytic product grid search
    a = initial_vector (normalize to sum=1)

    best_seq, best_score, best_M.

    While time left > some margin:
        # compute conv and M
        conv_a = conv_fft(a)
        M = conv_a.max()
        # Build A matrix (size (2n-1, n))
        # We'll get row for each k: A[k, i] = a[k - i] if index valid else 0
        # Implementation vectorized: Use outer subtraction? For k index array.
        i = np.arange(n)  # column indices
        k = np.arange(2 * n - 1)[:, None]  # row indices as column vector
        # Use mask condition.
        A = np.where((k - i >= 0) & (k - i < n), a[(k - i)], 0.0)  # shape (2n-1,n)
        # Solve LP: maximize sum(g)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        A_ub = A
        b_ub = np.full(2 * n - 1, M + 1e-12)  # small epsilon for numeric tolerance
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
           break

        g_raw = res.x
        sum_g = g_raw.sum()
        if sum_g < 1e-12:
            # fallback to random perturb
            g = a.copy()
        else:
            g = g_raw / sum_g
        # mixing
        t = 0.01  # small step
        a_candidate = (1 - t) * a + t * g
        # ensure sum = 1 (should be already) but due to numeric not exactly.
        a_candidate = a_candidate / a_candidate.sum()
        # compute new score using compute_score
        new_score = compute_score(a_candidate)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_seq = a_candidate.copy()
            print(f"[iter] improved to {best_score:.6f} using LP")
            # optionally update a = a_candidate for subsequent iterations
        # else if not improvement, maybe reduce t or try random
        # optionally accept if new_score < old score (like simulated annealing)
        a = a_candidate

        # Additional: jitter to avoid stale
        if time.time() - last_improve > 30: jitter.
        # Also occasional random moves.

    # final refinement: random local perturbations and evaluate exact
    while time.time() < deadline:
        perturb = np.random.normal(scale=1e-4, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0, None)
        cand = cand / cand.sum()
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    Print final and return list.

"""
Optimisation of a non‑negative step function for the objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines three phases:

1. **Broad random search in analytic families** – we sample parameters of
   a few simple families (product, exponential, beta, cosine, sum‑of‑powers)
   and keep the best normalised vector found.  This quickly discovers a good
   “U‑shaped’’ start without any gradient information.

2. **Linear‑programming refinement** – for a current candidate `a` we set
   M = max(conv(a,a)).  The constraint `conv(a,g) ≤ M` is linear in the
   unknown vector `g`.  We therefore solve a LP that maximises `sum(g)`
   under `g ≥ 0` and `conv(a,g) ≤ M`.  The optimal `g` concentrates mass
   where the convolution is least restrictive (i.e. near the ends).  By
   mixing the old vector with `g` we obtain a strictly better sequence.
   The LP is solved with ``scipy.optimize.linprog`` (HiGHS) and the process
   is repeated until time runs out.

3. **Stochastic fine‑tuning** – small Gaussian perturbations around the
   best vector are accepted when they improve the exact score.

All vectors are kept on the probability simplex (sum = 1), which makes the
objective simply `2·n·max(conv)`.  The routine respects the 1000 s time
budget, prints progress and finally returns the best list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (zero‑padded)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two arbitrary 1‑D arrays using FFT."""
    size = _next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: len(a) + len(b) - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a.
    The vector is assumed to lie on the simplex (Σa = 1); a tiny
    safety normalisation is performed nevertheless.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    return float(2.0 * a.shape[0] * conv.max())   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families (used for the broad random search)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """U‑shaped family: a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """Exponential tails: a_i ∝ exp(-lam·min(i,n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a = a / a.sum()
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped family: a_i ∝ (i+1)^{α}·(n‑i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """Cosine‑power family: a_i ∝ |cos(πi/(n‑1))|^{p}."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    a = a / a.sum()
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """Sum of power‑laws from both ends."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a

# Linear‑programming refinement utilities

def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2n‑1, n)) with entries
        A[k, j] = f[k‑j]   if 0 ≤ k‑j < n, else 0.
    Then conv(f, g) = A @ g.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    # Each column j corresponds to f shifted by +j positions.
    # Vectorised slice assignment (fast C loops):
    for j in range(n):
        A[j:j + n, j] = f
    return A

        total = cand.sum()
        if total > 0:
            cand = cand / total
        else:
            cand = np.ones(n) / n
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[init] Global seed score = {best_score:.6f}")

    # Allocate roughly 15 % of the time budget for this phase.
    phase1_deadline = start_time + 0.15 * budget_s

    families = ["product", "exponential", "beta", "cosine", "sum"]
    while time.time() < phase1_deadline:
        fam = np.random.choice(families)
        if fam == "product":
            p = np.random.uniform(0.2, 3.0)
            off = np.random.uniform(0.01, 1.5)
            cand = analytic_product(n, p, off)
        elif fam == "exponential":
            lam = np.random.uniform(0.001, 0.6)
            cand = analytic_exponential(n, lam)
        elif fam == "beta":
            alpha = np.random.uniform(-1.5, 1.0)
            beta = np.random.uniform(-1.5, 1.0)
            cand = analytic_beta(n, alpha, beta)
        elif fam == "cosine":
            p = np.random.uniform(0.5, 9.0)
            cand = analytic_cosine(n, p)
        else:  # sum of two power‑laws
            p = np.random.uniform(0.3, 3.0)
            off = np.random.uniform(0.0, 1.5)
            cand = analytic_sum(n, p, off)

        # Occasionally blend with the current best to explore neighbourhoods.
        if best_seq is not None and np.random.rand() < 0.2:
            w = np.random.rand()
            cand = w * best_seq + (1 - w) * cand

        cand = np.clip(cand, 0.0, None)
        cand = cand / cand.sum()
        sc = compute_score(cand)

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[phase1] new best = {best_score:.6f} (family={fam})")

    # If nothing was found (should not happen), fall back to uniform.
    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = compute_score(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – Linear‑programming refinement loop
    # ------------------------------------------------------------------
    a = best_seq.copy()
    best_seq = a.copy()
    best_score = compute_score(a)
    last_improve = time.time()
    phase2_deadline = start_time + 0.75 * budget_s   # about 60 % of total time

    while time.time() < phase2_deadline:
        # Current max‑convolution bound
        conv_a = conv_fft(a)
        M = conv_a.max()

        # Build linear constraints: conv(a, g) ≤ M
        A = build_conv_matrix(a)                     # shape (2n‑1, n)
        b_ub = np.full(A.shape[0], M + 1e-12, dtype=np.float64)

        # Linear programme: maximise sum(g)  → minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # Solve with HiGHS (fast for dense problems of this size)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': max(0.0, phase2_deadline - time.time())})
        if not res.success:
            # If the LP fails (very rare), fall back to a small random move.
            a = random_move(a, scale=0.02)
            continue

        g_raw = res.x
        s_g = g_raw.sum()
        if s_g <= 0:
            # degenerate solution – skip this LP iteration
            a = random_move(a, scale=0.02)
            continue

        # Normalise g to the simplex
        g = g_raw / s_g

        # Mix a and g.  Try a few step sizes; keep the first that improves.
        improved = False
        for t in (0.05, 0.02, 0.01, 0.005):
            cand = (1.0 - t) * a + t * g
            cand = cand / cand.sum()
            sc = compute_score(cand)
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = cand.copy()
                a = cand.copy()
                last_improve = time.time()
                print(f"[LP] improved to {best_score:.6f} with t={t:.3f}")
                improved = True
                break

        if not improved:
            # No improvement – introduce a tiny jitter to escape flat regions.
            if time.time() - last_improve > 20.0:
                jitter = np.random.rand(n) * 1e-3
                a = best_seq + jitter
                a = a / a.sum()
                last_improve = time.time()
                print("[LP] jitter restart")
            else:
                # occasional random mass‑transfer move
                if np.random.rand() < 0.02:
                    a = random_move(a, scale=0.1)

        # Safety break (in case we are about to exceed the deadline)
        if time.time() >= phase2_deadline:
            break

    # ------------------------------------------------------------------
    # Phase 3 – final stochastic refinement
    # ------------------------------------------------------------------
    while time.time() < deadline:
        perturb = np.random.normal(scale=5e-4, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = cand / cand.sum()
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, n = {n}, elapsed = {total:.1f}s")
    return best_seq.tolist()
