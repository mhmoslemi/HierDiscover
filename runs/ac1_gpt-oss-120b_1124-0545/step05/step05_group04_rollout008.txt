def lp_step(a, t=0.5):

    M = np.max(conv_fft_self(a))

    n = a.shape[0]
    rows = 2*n-1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = a[i]
    # Solve LP: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a, np.inf
    g = res.x
    g_sum = g.sum()
    if g_sum == 0:
        return a, np.inf
    g_norm = g / g_sum
    a_new = (1 - t) * a + t * g_norm
    a_new = project_simplex(a_new)  # ensure non-neg simplex
    score = compute_score(a_new)
    return a_new, score

def lp_improve(a, mix=0.5):
    n = a.shape[0]
    M = np.max(conv_fft_self(a))
    # Build dense matrix
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = a[i]
    # Solve LP: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0, None)] * n
    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception as e:
        # If SciPy not available or error, return original
        return a, np.inf
    if not res.success:
        return a, np.inf
    g = res.x
    g_sum = g.sum()
    if g_sum == 0:
        return a, np.inf
    g_norm = g / g_sum
    new_a = (1 - mix) * a + mix * g_norm
    new_a = project_simplex(new_a)
    new_score = compute_score(new_a)
    return new_a, new_score

"""
Search for a non‑negative step function (list of floats) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a multi‑stage heuristic:

1. **Analytic families** – exhaustive grids (product‑law, sum‑law,
   exponential, Gaussian) together with a Monte‑Carlo sweep of the
   parameters.  The best candidate becomes the starting point.

2. **Linear‑programming refinement** – for the current vector a we
   solve the LP  

        maximise   1ᵀ g
        subject to a * g ≤ M   (convolution bound)
                  g ≥ 0

   where M = max(a*a).  The solution g is normalised and mixed with a.
   A few passes quickly improve the max‑convolution.

3. **Smooth‑max optimisation** – continuation on a smoothed
   max‑objective Φ_λ using Adam (soft‑max over the convolution
   entries) with an increasingly large λ.

4. **Projected sub‑gradient descent** – classic sub‑gradient steps on
   the true max‑convolution, with a 1/√t step‑size.

5. **Simulated‑annealing local search** – tiny random mass transfers
   (random_move) are accepted either if they improve the score or
   with a probability decreasing with a temperature schedule.

All phases respect a global time budget (default 1000 s).  The best
sequence found before the deadline is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a (the sum is normalised
    internally).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

# Analytic families

def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^(‑p)   (edge‑heavy shape)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^(‑p) + (n‑1‑i+off)^(‑p)   (two tails)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exp_decay(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(‑c·|i‑center|)   (center‑peaked shape)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - center))
    a = a / a.sum()
    return a


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(‑0.5·((i‑center)/σ)²)   (smooth bump)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = a / a.sum()
    return a


def analytic_beta(n: int, alpha: float, beta: float, offset: float) -> np.ndarray:
    """Beta‑type shape  a_i ∝ (i+off)^{α‑1} (n‑1‑i+off)^{β‑1}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (alpha - 1.0) * (n - 1 - i + offset) ** (beta - 1.0)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Smooth‑max utilities (Φ_λ and its gradient)
# ----------------------------------------------------------------------


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    For a given log‑probability vector x, smoothing λ and length n,
    return the smooth max Φ_λ(a) and its gradient w.r.t. x.
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # convolution a*a
    conv = conv_fft_self(a)                # length 2n‑1
    max_c = np.max(conv)

    # smooth‑max over convolution entries
    shift = conv - max_c                   # ≤ 0
    exp_term = np.exp(lam * shift)        # ≤ 1
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam    # Φ_λ

    w = exp_term / sum_exp                 # weights, sum to 1

    # gradient w.r.t. a
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]      # length n
    grad_a = 2.0 * slice_part[::-1]            # ∂Φ/∂a

    # map gradient to x‑space (Jacobian of soft‑max)
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return phi, grad_x


def adam_optimize(x: np.ndarray, lam: float, n: int,
                  steps: int, lr: float, deadline: float) -> np.ndarray:
    """Adam optimiser for the smoothed objective Φ_λ."""
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for t in range(1, steps + 1):
        if time.time() >= deadline:
            break
        _, grad = compute_phi_and_grad(x, lam, n)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step
    return x


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max(conv))
# ----------------------------------------------------------------------


def subgrad_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step for the exact max‑convolution objective.
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    active = (conv >= max_val - 1e-12).astype(np.float64)
    cnt = active.sum()
    if cnt == 0:
        return a
    w = active / cnt
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]
    a_new = a - step * grad
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def lp_improve(a: np.ndarray, mix: float = 0.5) -> (np.ndarray, float):
    """
    One LP improvement step.
    Solve   maximise 1ᵀ g   s.t.   a * g ≤ M,  g ≥ 0,
    where M = max(a*a).  The solution g is normalised and mixed
    with the current a using parameter `mix`.
    Returns the new vector and its exact score.
    """
    n = a.shape[0]
    M = np.max(conv_fft_self(a))

    # Build dense convolution matrix A (size (2n‑1) × n)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a[i]

    c = -np.ones(n, dtype=np.float64)              # minimise -∑g  ↔ maximise ∑g
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:  # any problem with the LP solver
        return a, np.inf

    if not res.success:
        return a, np.inf

    g = res.x
    gs = g.sum()
    if gs == 0.0:
        return a, np.inf
    g_norm = g / gs
    a_new = (1.0 - mix) * a + mix * g_norm
    a_new = project_simplex(a_new)
    return a_new, compute_score(a_new)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the supplied evaluation function.  The routine respects the given time
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Analytic families + random parameter sweep
    # ------------------------------------------------------------------
    best_a = np.ones(n) / n
    best_score = compute_score(best_a)

    # If a previous construction is present, try it first
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n and ref.sum() > 0.0:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            if sc < best_score:
                best_a, best_score = ref.copy(), sc

    # parameter grids (broader than the previous run)
    p_vals = np.linspace(0.10, 1.50, 71)          # power‑law exponent
    off_vals = np.linspace(0.001, 5.0, 71)       # offset to avoid zero division
    c_vals = np.linspace(0.0005, 0.0300, 51)     # exponential decay rates
    sigma_vals = np.linspace(20.0, 1200.0, 61)   # Gaussian widths
    alpha_vals = np.linspace(0.10, 1.80, 61)     # Beta‑type α
    beta_vals = np.linspace(0.10, 1.80, 61)      # Beta‑type β

    # exhaustive grid for product‑law and sum‑law
    for mode in ("product", "sum"):
        for p in p_vals:
            for off in off_vals:
                if time.time() >= deadline:
                    break
                cand = analytic_product(n, p, off) if mode == "product" else analytic_sum(n, p, off)
                sc = compute_score(cand)
                if sc < best_score:
                    best_a, best_score = cand.copy(), sc

    # simple families
    for c in c_vals:
        if time.time() >= deadline:
            break
        cand = analytic_exp_decay(n, c)
        sc = compute_score(cand)
        if sc < best_score:
            best_a, best_score = cand.copy(), sc

    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        cand = analytic_gaussian(n, sigma)
        sc = compute_score(cand)
        if sc < best_score:
            best_a, best_score = cand.copy(), sc

    # Beta‑type shapes
    for a_par in alpha_vals:
        for b_par in beta_vals:
            if time.time() >= deadline:
                break
            cand = analytic_beta(n, a_par, b_par, offset=0.01)
            sc = compute_score(cand)
            if sc < best_score:
                best_a, best_score = cand.copy(), sc

    # ------------------------------------------------------------------
    # 2) Random Monte‑Carlo sampling of the above families
    # ------------------------------------------------------------------
    n_random = 2500
    for _ in range(n_random):
        if time.time() >= deadline:
            break
        # random product‑law
        p = np.random.uniform(0.05, 1.8)
        off = np.random.uniform(0.001, 6.0)
        cand = analytic_product(n, p, off)
        sc = compute_score(cand)
        if sc < best_score:
            best_a, best_score = cand.copy(), sc

        # random sum‑law
        p = np.random.uniform(0.05, 1.8)
        off = np.random.uniform(0.001, 6.0)
        cand = analytic_sum(n, p, off)
        sc = compute_score(cand)
        if sc < best_score:
            best_a, best_score = cand.copy(), sc

        # random beta‑type
        a_par = np.random.uniform(0.05, 2.0)
        b_par = np.random.uniform(0.05, 2.0)
        off = np.random.uniform(0.001, 3.0)
        cand = analytic_beta(n, a_par, b_par, off)
        sc = compute_score(cand)
        if sc < best_score:
            best_a, best_score = cand.copy(), sc

    print(f"[grid] best score after analytic + random sweep = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Small stochastic perturbation of the current best
    # ------------------------------------------------------------------
    for _ in range(12):
        if time.time() >= deadline:
            break
        pert = best_a + 0.02 * np.random.randn(n)
        pert = np.maximum(pert, 0.0)
        pert = project_simplex(pert)
        sc = compute_score(pert)
        if sc < best_score:
            best_a, best_score = pert.copy(), sc
            print(f"[perturb] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Linear‑programming refinement
    # ------------------------------------------------------------------
    lp_iter = 0
    while time.time() < deadline - 2.0 and lp_iter < 12:
        lp_iter += 1
        # try a relatively aggressive mixing first
        cand, cand_score = lp_improve(best_a, mix=0.5)
        if cand_score < best_score:
            best_a, best_score = cand.copy(), cand_score
            print(f"[LP {lp_iter}] new best = {best_score:.6f} (t={time.time() - start_time:.1f}s)")
            continue
        # if not improved, try a more conservative mixing
        cand, cand_score = lp_improve(best_a, mix=0.2)
        if cand_score < best_score:
            best_a, best_score = cand.copy(), cand_score
            print(f"[LP {lp_iter} (small mix)] new best = {best_score:.6f}")
        # otherwise keep the current vector

    # ------------------------------------------------------------------
    # 5) Smooth‑max optimisation with Adam (λ continuation)
    # ------------------------------------------------------------------
    eps = 1e-12
    x = np.log(np.maximum(best_a, eps))
    lam_vals = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0,
                2000.0, 8000.0, 20000.0, 80000.0]

    for lam in lam_vals:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # very rough estimate: ~0.002 s per Adam iteration for n≈2000
        max_iters = int(remaining / 0.002)
        steps = min(6000, max_iters)
        if steps <= 0:
            break
        # longer runs for the largest λ values
        if lam > 2000:
            steps = min(10000, max_iters)
        x = adam_optimize(x, lam, n, steps, lr=0.05, deadline=deadline)
        a = softmax(x)
        sc = compute_score(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc
            print(f"[Adam λ={lam:.0f}] new best = {best_score:.6f} (t={time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # 6) Exact sub‑gradient descent (many cheap steps)
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    while time.time() < deadline - 1.0:
        it += 1
        step = 0.5 / np.sqrt(it)
        a = subgrad_step(a, step)
        sc = compute_score(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc
            print(f"[subgrad iter={it}] new best = {best_score:.6f} (t={time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # 7) Light simulated‑annealing random‑mass‑transfer search
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_score = best_score
    scale = 0.02
    temperature = 0.02
    while time.time() < deadline:
        cand = random_move(a, scale)
        cand_score = compute_score(cand)
        if cand_score < cur_score:
            a, cur_score = cand, cand_score
            if cand_score < best_score:
                best_a, best_score = cand.copy(), cand_score
                print(f"[anneal] new best = {best_score:.6f} (t={time.time() - start_time:.1f}s)")
        else:
            # accept worse moves with a Boltzmann probability
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                a, cur_score = cand, cand_score
        # slowly cool down / shrink step size
        temperature = max(1e-5, temperature * 0.9995)
        scale = max(0.001, scale * 0.9995)

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    return best_a.tolist()
