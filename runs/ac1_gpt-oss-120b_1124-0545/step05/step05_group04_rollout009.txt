```
def permute_anneal(a, cur_score, deadline):
    cur = a.copy()
    cur_score = cur_score
    best = cur.copy()
    best_score = cur_score
    n = len(cur)
    start = time.time()
    while time.time() < deadline:
        # Temperature based on remaining fraction
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(1e-8, (best_score) * 0.02 * (1 - fraction)) + 1e-12
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # swap
        cur[i], cur[j] = cur[j], cur[i]
        new_score = compute_score(cur)
        delta = new_score - cur_score
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score, best = new_score, cur.copy()
        else:
            # revert
            cur[i], cur[j] = cur[j], cur[i]
        # occasional log
        if int(elapsed) % 30 == 0:
            print(f"[anneal] time {elapsed:.1f}s best={best_score:.6f}")
    return best, best_score
```

assistantfinal**Solution Overview**

The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 ,
\]

where ``(a*a)`` is the linear (zero‑padded) self‑convolution.  
After normalising the vector (`∑a_i = 1`) the objective reduces to  

\[
F(a)=2\,n\;\max_k (a*a)_k .
\]

The problem is convex (the max of a family of convex quadratics) but
has thousands of variables, so a direct QP solver is too slow.
Instead we combine three cheap but complementary ideas:

1. **Analytic families** – product‑law, sum‑law, exponential, Gaussian.
   A dense grid gives a very good start.
2. **Local refinement of the family parameters** – a few random
   perturbations of the best *(p,offset)* values improve the grid result.
3. **Permutation‑annealing** – the ordering of the already‑good mass
   distribution strongly influences the convolution peaks.  We treat the
   vector as a multiset of numbers and search for a low‑peak ordering
   with a simulated‑annealing scheme that swaps two entries at a time.
   Swaps are cheap (the whole vector is normalised on the fly) and the
   evaluation `compute_score` uses a fast FFT.
4. **Tiny mass‑transfer moves** – after the annealing phase we finish
   with a few stochastic “move” steps (the `random_move` routine from the
   reference implementation) to polish the final result.

All stages respect the time budget (`budget_s`).  The algorithm prints
intermediate scores, so you can see the progress when it runs.

```python
"""Step‑function optimisation for the evaluate_sequence() benchmark.

The routine searches a space of analytic shapes, refines the shape
parameters, then performs a simulated‑annealing permutation search
that swaps entries of the vector.  Finally a few tiny mass‑transfer
steps fine‑tune the result.  The whole process stops before the given
budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a with itself using zero‑padding and FFT.
    Returns the array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a.
    The vector is internally normalised (∑a = 1) before the measurement.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a tiny amount of probability mass from one index to another
    and project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families (all vectors are returned already normalised)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^(‑p)   (edge‑heavy shape)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^(‑p) + (n‑1‑i+off)^(‑p)   (two tails)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exp_decay(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(‑c·|i‑center|)   (center‑peaked shape)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - center))
    a = a / a.sum()
    return a


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(‑0.5·((i‑center)/σ)²)   (smooth bump)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Permutation simulated annealing
# ----------------------------------------------------------------------


def permute_anneal(a: np.ndarray, cur_score: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on the ordering of the entries.
    Swaps two positions, accepts the move if it improves the score or
    with a temperature‑dependent probability.
    Returns the best vector and its score found before the deadline.
    """
    cur = a.copy()
    cur_score = cur_score
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]

    start = time.time()
    # temperature parameters – we start with a modest temperature and cool linearly
    T0 = best_score * 0.02 + 1e-12
    Tmin = 1e-12

    while time.time() < deadline:
        # linear cooling based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = max(Tmin, T0 * (1.0 - frac))

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # perform the swap
        cur[i], cur[j] = cur[j], cur[i]

        new_score = compute_score(cur)
        delta = new_score - cur_score

        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
                # occasional progress report
                if int(elapsed) % 30 == 0:
                    print(f"[anneal] t={elapsed:.1f}s  best={best_score:.6f}")
        else:
            # reject – revert the swap
            cur[i], cur[j] = cur[j], cur[i]

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that
    minimises ``evaluate_sequence``.  The routine respects the given
    time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Analytic grid search – deterministic initial candidates
    # ------------------------------------------------------------------
    best_a = np.ones(n) / n
    best_score = compute_score(best_a)

    # reuse a previously discovered construction if it exists
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n and ref.sum() > 0:
            ref = np.maximum(ref, 0.0)
            sc = compute_score(ref)
            if sc < best_score:
                best_score, best_a = sc, ref.copy()

    # grids for the product and sum families
    p_vals = np.linspace(0.30, 0.70, 81)          # 0.005 step
    off_vals = np.linspace(0.0001, 2.0, 81)      # 0.025 step

    best_prod = {"score": np.inf, "p": None, "off": None, "a": None}
    best_sum = {"score": np.inf, "p": None, "off": None, "a": None}

    for mode in ("product", "sum"):
        for p in p_vals:
            for off in off_vals:
                if time.time() >= deadline:
                    break
                if mode == "product":
                    cand = analytic_product(n, p, off)
                else:
                    cand = analytic_sum(n, p, off)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score, best_a = sc, cand.copy()
                # keep family‑wise best for later refinement
                if mode == "product" and sc < best_prod["score"]:
                    best_prod.update({"score": sc, "p": p, "off": off, "a": cand.copy()})
                if mode == "sum" and sc < best_sum["score"]:
                    best_sum.update({"score": sc, "p": p, "off": off, "a": cand.copy()})

    print(f"[grid] best score after sweep = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Local refinement of the (p, offset) parameters (few thousand trials)
    # ------------------------------------------------------------------
    def refine_family(base_func, start_p, start_off, cur_best_score, cur_best_a):
        p, off = start_p, start_off
        # exploration scales (tuned empirically)
        sigma_p, sigma_off = 0.02, 0.2
        for _ in range(4000):
            if time.time() >= deadline:
                break
            p_new = max(1e-6, p + sigma_p * np.random.randn())
            off_new = max(1e-6, off + sigma_off * np.random.randn())
            cand = base_func(n, p_new, off_new)
            sc = compute_score(cand)
            if sc < cur_best_score:
                cur_best_score, cur_best_a = sc, cand.copy()
                p, off = p_new, off_new
                # slowly shrink the exploration radii
                sigma_p *= 0.99
                sigma_off *= 0.99
            else:
                # decay even on failures to focus search
                sigma_p *= 0.9995
                sigma_off *= 0.9995
        return cur_best_a, cur_best_score

    # refine product family if it was competitive
    if best_prod["a"] is not None:
        best_a, best_score = refine_family(
            analytic_product,
            best_prod["p"],
            best_prod["off"],
            best_score,
            best_a,
        )
        print(f"[refine product] best = {best_score:.6f}")

    # refine sum family if it was competitive
    if best_sum["a"] is not None:
        best_a, best_score = refine_family(
            analytic_sum,
            best_sum["p"],
            best_sum["off"],
            best_score,
            best_a,
        )
        print(f"[refine sum] best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Permutation simulated annealing (the core novelty)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = permute_anneal(best_a, best_score, deadline)
        print(f"[anneal] finished with best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Light random‑mass‑transfer local search (final polishing)
    # ------------------------------------------------------------------
    a = best_a.copy()
    scale = 0.02
    while time.time() < deadline:
        cand = random_move(a, scale)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[random] new best = {best_score:.6f} (t={time.time() - start_time:.1f}s)")
            a = cand
            scale = max(0.001, scale * 0.9)
        else:
            # slowly relax step size
            scale = min(0.05, scale * 1.001)
            a = cand

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    # return as plain Python list
    return best_a.tolist()
```
