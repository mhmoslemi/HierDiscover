"""
Propose a high‑quality non‑negative step sequence for the self‑convolution
objective  2·n·max(conv(f,f)) / (sum(f)^2).

The optimizer works in three phases:
  1. Exhaustive parametric sweep over a rich family of analytic windows
     (geometric, power‑law, Gaussian, Beta, Kaiser, DPSS, etc.).  All candidates
     are normalised to unit sum and the best ~K are kept.
  2. For each of the top candidates a short Adam optimisation on a smooth
     max‑surrogate (log‑sum‑exp) is performed.  The Adam updates are carried
     out on the log‑softmax parametrisation so the iterate stays on the simplex.
  3. The best Adam result is refined by a mirror‑descent (exponentiated‑gradient)
     sub‑gradient method on the exact max‑convolution value, followed by a cheap
     simulated‑annealing local search.  Finally the sequence is symmetrised
     (averaged with its reversal) which often reduces the peak further.

The whole routine respects a user‑provided time budget (default 1000 s) and
periodically checks the deadline to abort early if necessary.  The best
sequence found is cached in the global variable ``height_sequence_1`` for
future calls.
"""

import time
import numpy as np
from scipy import signal

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for many calls)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the grader (scale‑invariant)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Analytic seed families (all return a normalised non‑negative vector)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _sin2_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    return a / a.sum()


def _hanning(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_seed(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric / slightly asymmetric Beta(α,β) on (0,1) – interior points only."""
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1.0))
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    a = np.kaiser(n, beta)
    return a / a.sum()


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Slepian (DPSS) window – symmetrised automatically."""
    a = signal.windows.dpss(n, nw, sym=True)
    # DPSS may contain tiny negative values due to numeric error; clip.
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _cauchy(n: int, scale: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 / (1.0 + ((i - centre) / (scale * n)) ** 2)
    return a / a.sum()


def _laplace(n: int, tau: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-np.abs(i - centre) / (tau * n))
    return a / a.sum()


def _cosine_power(n: int, p: float) -> np.ndarray:
    """a_i = max(0, cos(π (i-(n-1)/2)/n))^p """
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.cos(np.pi * (i - centre) / n)
    a = np.clip(a, 0.0, None) ** p
    return a / a.sum()


def _tent(n: int) -> np.ndarray:
    """Symmetric tent (linear rise‑fall)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs((2.0 * i / (n - 1)) - 1.0)
    a[a < 0] = 0.0
    return a / a.sum()


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length (preserves total mass)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
# Adam optimiser on a smooth max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(seed_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000,
                   beta0: float = 2.0,
                   beta_target: float = 120.0,
                   beta_mul: float = 1.001,
                   lr0: float = 0.7) -> tuple[np.ndarray, float]:
    """Adam on L = (1/β)·log Σ exp(β·conv).  Returns best sequence seen."""
    v = np.log(seed_vec + 1e-30)           # logits → simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # point on simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # increase β (smoothness) gradually
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        # soft‑max weights (log‑sum‑exp trick)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient of the smoothed max w.r.t. a (via reversed convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through the soft‑max parametrisation
        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)

        # Adam update on logits
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (mirror / exponentiated‑gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200000,
                        prefactor: float = 1.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active entry of the convolution
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = prefactor / np.sqrt(it)          # step size ∝ 1/√t
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # stabilise numerics
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing fine‑tuner
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Metropolis‑Hastings local search that swaps tiny mass fragments."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # temperature schedule (a few % of current objective)
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.025 * cur[i]   # ≤ 2.5 % of source mass
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n:5d} it={it:6d} best={new_val:.6f}")

        # occasional global jitter to escape plateaus
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            jittered = cur + noise
            jittered = np.clip(jittered, 0.0, None)
            if jittered.sum() == 0.0:
                jittered = _uniform(n)
            else:
                jittered /= jittered.sum()
            jitter_val = _objective(jittered)
            if jitter_val < cur_val or rng.random() < np.exp(-(jitter_val - cur_val) / (T + 1e-12)):
                cur = jittered
                cur_val = jitter_val
                if jitter_val < best_val:
                    best_val = jitter_val
                    best = jittered.copy()
                    print(f"[Anneal] n={n:5d} it={it:6d} (jump) best={jitter_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Candidate generation (parametric sweep)
# ----------------------------------------------------------------------


def _generate_candidates(n: int,
                         previous: np.ndarray | None,
                         rng: np.random.Generator,
                         deadline: float) -> list[np.ndarray]:
    """
    Return a diverse list of normalised candidate vectors.
    The generation respects the deadline – if time runs out early we
    return whatever has been produced so far.
    """
    candidates: list[np.ndarray] = []

    # ------------------------------------------------------------------
    # Base seeds (deterministic families)
    # ------------------------------------------------------------------
    candidates.append(_uniform(n))
    candidates.append(_tent(n))
    candidates.append(_sine_window(n))
    candidates.append(_sin2_window(n))
    candidates.append(_hanning(n))
    candidates.append(_gaussian(n, sigma_factor=0.12))
    candidates.append(_gaussian(n, sigma_factor=0.20))
    candidates.append(_beta_seed(n, 0.5, 0.5))
    candidates.append(_beta_seed(n, 0.7, 1.2))
    candidates.append(_beta_seed(n, 1.2, 0.7))
    candidates.append(_kaiser_window(n, 4.0))
    candidates.append(_kaiser_window(n, 8.0))
    candidates.append(_dpss_window(n, 2.0))
    candidates.append(_dpss_window(n, 3.5))

    # ------------------------------------------------------------------
    # Randomised families – we sample a few thousand sets but stop early
    # if we near the deadline.
    # ------------------------------------------------------------------
    max_random = 2000                      # cap for safety
    cnt = 0
    while cnt < max_random and time.time() < deadline:
        # geometric
        r = rng.uniform(0.5, 0.999)
        candidates.append(_geom(n, r))

        # power‑law
        p = rng.uniform(0.0, 2.5)
        candidates.append(_power(n, p))

        # geometric × power
        r = rng.uniform(0.5, 0.999)
        p = rng.uniform(0.0, 2.0)
        candidates.append(_geom_power(n, r, p))

        # symmetric geometric
        r = rng.uniform(0.5, 0.999)
        candidates.append(_sym_geom(n, r))

        # Beta
        a = rng.uniform(0.2, 3.0)
        b = rng.uniform(0.2, 3.0)
        candidates.append(_beta_seed(n, a, b))

        # Kaiser
        beta = rng.uniform(2.0, 12.0)
        candidates.append(_kaiser_window(n, beta))

        # DPSS – NW in [1, 4]
        nw = rng.uniform(1.0, 4.0)
        candidates.append(_dpss_window(n, nw))

        # Cauchy
        scale = rng.uniform(0.05, 0.4)
        candidates.append(_cauchy(n, scale))

        # Laplace
        tau = rng.uniform(0.05, 0.4)
        candidates.append(_laplace(n, tau))

        # Cosine‑power
        p = rng.uniform(0.5, 6.0)
        candidates.append(_cosine_power(n, p))

        cnt += 10

    # ------------------------------------------------------------------
    # Upscaled previous best (if any)
    # ------------------------------------------------------------------
    if previous is not None:
        if previous.shape != (n,):
            try:
                up = _upscale(previous, n)
                candidates.append(up)
            except Exception:
                pass
        else:
            candidates.append(previous / previous.sum())

    # ------------------------------------------------------------------
    # Mirrored copies (enforce symmetry as a possible improvement)
    # ------------------------------------------------------------------
    mirrored = [c[::-1].copy() for c in candidates]
    candidates.extend(mirrored)

    # ------------------------------------------------------------------
    # Normalise and clip any possible numerical negatives
    # ------------------------------------------------------------------
    final = []
    for c in candidates:
        c = np.clip(c, 0.0, None)
        s = c.sum()
        if s > 0.0:
            final.append(c / s)

    # Deduplicate (cheap) – convert to tuples of rounded values
    uniq = {}
    for c in final:
        key = tuple(np.round(c, 12))
        uniq[key] = c
    return list(uniq.values())


# ----------------------------------------------------------------------
# Main entry point – respects the 1000 s budget
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns the best sequence (list of floats) found before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    # Retrieve any cached construction from a previous call
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # Choose a set of lengths to explore; start modest, go larger if time permits
    lengths = [4000, 8000, 12000]
    global_best_seq: np.ndarray | None = None
    global_best_val = float("inf")

    rng = np.random.default_rng(seed)

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # 1) Generate a large pool of analytic candidates
        # ------------------------------------------------------------------
        seed_pool = _generate_candidates(n, previous, rng, deadline)

        # ------------------------------------------------------------------
        # 2) Quick evaluation of all candidates, keep the top few
        # ------------------------------------------------------------------
        cand_vals = [(c, _objective(c)) for c in seed_pool]
        cand_vals.sort(key=lambda x: x[1])
        top_k = min(12, len(cand_vals))
        top_seeds = [c for c, _ in cand_vals[:top_k]]

        # Track best raw seed globally
        if cand_vals[0][1] < global_best_val:
            global_best_val = cand_vals[0][1]
            global_best_seq = cand_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # 3) Adam optimisation from each top seed (short runs)
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=8000,
                beta0=2.0,
                beta_target=120.0,
                beta_mul=1.001,
                lr0=0.7,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # 4) Sub‑gradient refinement (longer run) on the current best of this length
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(
                global_best_seq,
                n,
                deadline,
                max_iters=200000,
                prefactor=1.2,        # slightly larger step‑size may speed convergence
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # 5) Simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=None,   # run until time runs out
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # 6) Symmetrisation – averaging with its mirror often reduces the peak
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            sym = (global_best_seq + global_best_seq[::-1]) / 2.0
            sym /= sym.sum()
            sym_val = _objective(sym)
            if sym_val < global_best_val:
                global_best_val = sym_val
                global_best_seq = sym.copy()
                print(f"[symmetrisation] n={n} improved to {sym_val:.6f}")

        # Propagate this best forward for possible up‑scaling
        previous = global_best_seq

    # ----------------------------------------------------------------------
    # Fallback (unlikely to be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Cache for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
