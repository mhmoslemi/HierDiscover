def _annealing_hill_climb(a, n, deadline):
   cur = a.copy()
   cur_conv = _conv_fft(cur, cur)
   cur_max = np.max(cur_conv)
   best_seq = cur.copy()
   best_val = 2*n*cur_max

   T0 = 0.0005  # maybe scaling factor relative to max?
   decay = 0.9995 per iteration

   step_scale = 0.02

   it = 0
   while time.time() < deadline:
       it += 1
       # compute contributions for donor weights: we can use active max or all contributions
       # use conv as before.
       conv = _conv_fft(cur, cur)
       max_idx = int(np.argmax(conv))

       # contributions for each i
       i_start = max(0, max_idx - (n-1))
       i_end = min(max_idx, n-1)
       idx_i = np.arange(i_start, i_end+1)
       rev_idx = max_idx - idx_i
       contrib = cur[idx_i] * cur[rev_idx]

       total = contrib.sum()
       if total > 0:
           probs = contrib / total
           donor = np.random.choice(idx_i, p=probs)
       else:
           donor = np.random.randint(n)

       receiver = np.random.randint(n)
       while receiver == donor:
           receiver = np.random.randint(n)

       max_transfer = cur[donor] * step_scale
       if max_transfer <= 0:
           continue
       delta = np.random.rand() * max_transfer

       new = cur.copy()
       new[donor] -= delta
       new[receiver] += delta

       new_conv = _conv_fft(new, new)
       new_max = np.max(new_conv)
       new_val = 2 * n * new_max

       # accept if improvement or metropolis
       if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / max(T0 * cur_val, 1e-12)):
           cur = new
           cur_val = new_val

           if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()

       # schedule
       if it % 1000 == 0:
           step_scale = max(step_scale * 0.9, 1e-6)
           T0 *= 0.99

   return best_seq, best_val

Implementation details for _proj_simplex:

def _proj_simplex(v):
    # Input: ndarray (any shape). Returns projection onto simplex {x: x>=0, sum(x)=1}
    # From Duchi et al.
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all negative? return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _lp_refine(a, n, deadline, max_iters=5, step=0.5):
    # solves LP max sum(b) subject to conv(a,b) <= M
    # returns refined a and value
    best_a = a.copy()
    best_val = 2*n*np.max(_conv_fft(a,a))
    for it in range(max_iters):
        if time.time()>deadline:
            break
        conv = _conv_fft(best_a,best_a)
        M = conv.max()
        # Build matrix
        data = []
        offsets = []
        for i in range(n):
            data.append(np.full(n - i, best_a[i]))
            offsets.append(i)
        A = scipy.sparse.diags(data, offsets, shape=(2*n - 1, n), format='csr')
        b_ub = np.full(2*n - 1, M)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        # Solve LP
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                     method='highs', options={"presolve": True})
        if not res.success:
            break
        b0 = res.x
        sum_b = b0.sum()
        if sum_b <= 1e-12:
            break
        b = b0 / sum_b
        # mix
        a_new = (1 - step) * best_a + step * b
        a_new = a_new / a_new.sum()
        # compute new val
        new_val = 2 * n * np.max(_conv_fft(a_new, a_new))
        # update if improved
        if new_val < best_val:
            best_val = new_val
            best_a = a_new
            print(f"[LP] iteration {it} improved to {new_val:.6f}")
        else:
            # reduce step or break
            step *= 0.5
        # maybe reduce step each iteration

    return best_a, best_val

"""
Advanced step‑function optimisation.

The optimiser combines several complementary techniques:

* a small pool of analytic seeds (including the previous best);
* large‑scale Adam optimisation of a smooth‑max surrogate;
* Euclidean‑projected sub‑gradient descent on the true max;
* an L2‑flattening descent that pushes the auto‑convolution toward a constant;
* a dedicated linear‑programming refinement (the classic “max‑auto‑convolution”
  LP) which can strictly lower the maximal convolution coefficient;
* a long‑run simulated‑annealing hill‑climber that moves mass from the most
  offending coefficients.

All phases respect the overall time budget (default 1000 s) and keep the best
sequence seen so far.  The final sequence is verified with
`evaluate_sequence` before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large n)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (used by the Adam optimiser)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum(x)=1}."""
    # Duchi et al. (2008) – O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed families (ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _double_interval(n: int, frac: float = 0.08) -> np.ndarray:
    """Two symmetric blocks at the ends of the interval."""
    k = max(1, int(frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0 / k
    a[-k:] = 1.0 / k
    return a / a.sum()


def _make_analytic_seeds(n: int, previous: np.ndarray | None):
    """Collect a diverse set of normalised analytic start vectors."""
    seeds = [_uniform(n), _sine_window(n), _double_interval(n)]

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power families
    for p in np.linspace(0.0, 4.0, 11):
        seeds.append(_power(n, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # beta‑like shapes
    for a in np.linspace(0.2, 2.0, 7):
        for b in np.linspace(0.2, 2.0, 7):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (different concentrations)
    rng = np.random.default_rng(2024)
    for conc in (0.05, 0.2, 0.8, 2.0, 6.0):
        for _ in range(2):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a / a.sum())

    # softmax of Gaussian logits
    for _ in range(5):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Objective (fast version – assumes sum = 1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) – a is assumed to sum to 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Adam optimisation of a smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray, deadline: float,
                   max_iters: int = 18000) -> tuple[np.ndarray | None, float]:
    n = len(seed)
    v = np.log(seed + 1e-30)                # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0          # initial smooth‑max temperature
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                     # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # convert to gradient w.r.t. logits
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Euclidean projected sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def _subgradient_descent(start_seq: np.ndarray, deadline: float,
                         max_iters: int = 60000) -> tuple[np.ndarray, float]:
    n = len(start_seq)
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        max_indices = np.where(conv == max_val)[0]

        # sub‑gradient (sum of contributions from all maximal entries)
        g = np.zeros_like(a)
        for k in max_indices:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx_i = np.arange(i_start, i_end + 1)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]

        # step size – classic diminishing schedule
        lr = 0.7 / np.sqrt(it)
        a_new = a - lr * g
        a = _proj_simplex(a_new)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# L₂‑flattening descent (drives a*a toward a constant)
# ----------------------------------------------------------------------


def _l2_flatten_descent(start_seq: np.ndarray, deadline: float,
                        max_iters: int = 40000) -> tuple[np.ndarray, float]:
    n = len(start_seq)
    a = start_seq.copy()
    a = a / a.sum()

    mu = 1.0 / (2 * n - 1)          # uniform target for each convolution entry
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        residual = conv - mu                     # r = b – μ
        rev_a = a[::-1]
        grad_full = _conv_fft(residual, rev_a)   # length 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        a_new = a - lr * grad
        a = _proj_simplex(a_new)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L2Flat] it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (classic “max‑auto‑convolution” LP)
# ----------------------------------------------------------------------


def _lp_refinement(a: np.ndarray, deadline: float,
                   max_outer_iters: int = 7,
                   mix_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve
        maximize   Σ b_j
        subject to (a * b)_k ≤ M   ∀k
                    b ≥ 0
    where M is the current maximal auto‑convolution coefficient.
    The solution b is renormalised to sum 1 and mixed with a.
    """
    from scipy import sparse, optimize

    n = len(a)
    a = a / a.sum()
    best_seq = a.copy()
    best_val = _objective(a)

    for outer in range(max_outer_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()

        # Build the convolution matrix A (size (2n-1) × n) where
        # A_{k,j} = a_{k-j} if 0 ≤ k‑j < n else 0.
        data = []
        offsets = []
        for i in range(n):
            data.append(np.full(n - i, a[i], dtype=np.float64))
            offsets.append(i)
        A = sparse.diags(data, offsets, shape=(2 * n - 1, n), format='csr')

        c = -np.ones(n)                     # maximise Σb  ⇔ minimise –Σb
        b_ub = np.full(2 * n - 1, M)        # (a * b)_k ≤ M
        bounds = [(0, None)] * n

        # Solve with HiGHS (fast interior‑point/simplex)
        res = optimize.linprog(
            c, A_ub=A, b_ub=b_ub, bounds=bounds,
            method='highs',
            options={'presolve': True}
        )

        if not res.success:
            print("[LP] linprog failed – terminating refinement")
            break

        b0 = res.x
        sum_b0 = b0.sum()
        if sum_b0 <= 1e-12:
            print("[LP] trivial solution – terminating")
            break

        b = b0 / sum_b0                     # renormalise to a probability vector

        # convex combination
        a = (1.0 - mix_factor) * a + mix_factor * b
        a = a / a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LP] outer={outer+1} improved to {cur_val:.6f}")
        else:
            # If we made no progress, shrink the mixing factor
            mix_factor = max(mix_factor * 0.5, 0.05)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (targeted mass transfers)
# ----------------------------------------------------------------------


def _annealing_hill_climb(start_seq: np.ndarray, deadline: float,
                          max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Random mass transfers guided by the current maximal convolution entry.
    Moves that increase the objective are occasionally accepted according to
    a temperature schedule, allowing the search to escape shallow plateaux.
    """
    n = len(start_seq)
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    step_scale = 0.02               # initial relative transfer size
    T = 0.001 * cur_val             # initial temperature

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))

        # contribution of each coefficient to the maximal entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(max_idx, n - 1)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]

        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = cur[donor] * step_scale
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        new_val = _objective(new)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        # annealing schedule
        if it % 5000 == 0:
            step_scale = max(step_scale * 0.85, 1e-6)
            T *= 0.85

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
