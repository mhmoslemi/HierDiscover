"""
Propose a step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works as follows:

1.   Gather a diverse set of analytic / random seeds (uniform,
     sine‑window, geometric, power‑law, beta‑shaped, Dirichlet draws,
     also the best sequence from a previous run if available).

2.   For each seed run a multi‑stage optimisation on the *smooth‑max*
     surrogate of the max‑convolution using L‑BFGS‑B.  The variables are
     the logits of a soft‑max parametrisation, which guarantees non‑negativity
     and unit ℓ₁‑norm.  The surrogate is
            fτ(a) = (log‑sum‑exp(τ·conv(a,a)))/τ
       with τ gradually increasing (0.5 → 20).  Exact gradients are
       obtained analytically (weighted convolution).

3.   Keep the best exact solution (using the true max‑convolution).

4.   Starting from the best solution run a targeted stochastic hill‑climb
     (mass transfers guided by the contribution to the current maximal
     convolution entry) for further fine‑tuning.

5.   Verify the final sequence with ``evaluate_sequence`` and store it in
     ``height_sequence_1`` for possible warm‑starts later.

All heavy work respects the supplied time budget (default 1000 s).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Analytic seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_analytic_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of normalised analytic start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power families
    for p in np.linspace(0.0, 4.0, 11):
        seeds.append(_power(n, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # beta‑like shapes
    alphas = np.linspace(0.2, 2.0, 9)
    betas = np.linspace(0.2, 2.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet families (different concentrations)
    rng = np.random.default_rng(2023)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # softmax of Gaussian logits (non‑parametric)
    for _ in range(5):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Objective helper (sum assumed 1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) – assumes sum(a)=1."""
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smooth‑max surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _smooth_objective(logits: np.ndarray, n: int, tau: float):
    """Smooth‑max surrogate of the max‑convolution and its gradient.
    Returns (obj, grad) where obj = 2·n·smooth_max and grad is w.r.t. logits.
    """
    a = _softmax(logits)                     # stays on simplex
    conv = _conv_fft(a, a)                   # length 2n‑1
    max_conv = np.max(conv)

    # raw (unnormalised) softmax weights for the smooth max
    shift = max_conv
    exp_arg = tau * (conv - shift)           # ≤ 0
    w_raw = np.exp(exp_arg)
    sum_raw = w_raw.sum()
    if sum_raw == 0.0:
        # numerical safeguard – fall back to uniform weights
        w = np.full_like(w_raw, 1.0 / w_raw.size)
    else:
        w = w_raw / sum_raw                  # normalised softmax weights

    # gradient of Σ w_k·conv_k w.r.t. a (central n entries)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)               # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # gradient w.r.t. logits (softmax Jacobian)
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    # smooth max value
    smooth_max = shift + np.log(sum_raw) / tau
    obj = 2.0 * n * smooth_max

    return obj, grad_logits

# Multi‑stage L‑BFGS optimisation on the smooth surrogate
# ----------------------------------------------------------------------


def _lbfgs_optimize(seed: np.ndarray, n: int, deadline: float):
    """Run L‑BFGS on a sequence of increasing smoothness parameters."""
    # initialise logits (log‑softmax parameters)
    logits = np.log(seed + 1e-30)

    best_seq = None
    best_val = float("inf")

    # (tau, max_iter) schedule – the later stages focus on the true max
    schedule = [
        (0.5, 200),
        (1.0, 150),
        (2.0, 150),
        (5.0, 100),
        (10.0, 80),
        (20.0, 60),
    ]

    for tau, max_iters in schedule:
        if time.time() > deadline:
            break

        # minimise the smooth surrogate using L‑BFGS‑B
        res = minimize(
            fun=_smooth_objective,
            x0=logits,
            args=(n, tau),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iters, "ftol": 1e-13, "gtol": 1e-13},
        )

        logits = res.x
        a = _softmax(logits)

        # exact objective (no smoothing)
        conv = _conv_fft(a, a)
        exact_val = 2.0 * n * np.max(conv)

        if exact_val < best_val:
            best_val = exact_val
            best_seq = a.copy()
            print(f"[L‑BFGS] n={n:5d} τ={tau:5.1f} exact={exact_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Targeted stochastic hill‑climb (fine‑tuning)
# ----------------------------------------------------------------------


def _targeted_hill_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 30000) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))

        # contributions of each pair to the currently maximal entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]

        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        # shrink step size gradually
        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val

            previous_global = np.asarray(previous_global, dtype=np.float64)
        except Exception:
            previous_global = None
    else:
        previous_global = None

    # Candidate lengths – larger n gives more degrees of freedom
    lengths = [8000, 4000]

    global_best_seq = None
    global_best_val = float("inf")

    # Helper to keep a short list of the best seeds (sorted ascending)
    def _update_top(top, val, seq, keep=5):
        top.append((val, seq))
        top.sort(key=lambda x: x[0])
        if len(top) > keep:
            top.pop()

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        top_seeds = []

        # --------------------------------------------------------------
        # 1. Analytic / deterministic seeds
        # --------------------------------------------------------------
        analytic = _make_analytic_seeds(n, previous_global if previous_global is not None and previous_global.shape[0] == n else None)
        for s in analytic:
            val = _objective(s, n)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed‑anal] n={n:5d} new global best={val:.6f}")
            _update_top(top_seeds, val, s, keep=10)

        # --------------------------------------------------------------
        # 2. Random Dirichlet seeds (additional diversity)
        # --------------------------------------------------------------
        rng = np.random.default_rng(2024)
        for conc in (0.05, 0.2, 0.8, 2.0, 5.0):
            for _ in range(4):
                cand = _random_dirichlet(n, conc, rng)
                val = _objective(cand, n)
                if val < global_best_val:
                    global_best_val = val
                    global_best_seq = cand.copy()
                    print(f"[seed‑rand] n={n:5d} new global best={val:.6f}")
                _update_top(top_seeds, val, cand, keep=10)

        # Keep only the most promising few for intensive polishing
        top_seeds = top_seeds[:5]

        # --------------------------------------------------------------
        # 3. Multi‑stage L‑BFGS refinement of the best seeds
        # --------------------------------------------------------------
        for idx, (_, seed_seq) in enumerate(top_seeds):
            if time.time() > deadline:
                break
            print(f"\n--- Polishing seed {idx+1}/{len(top_seeds)} (n={n}) ---")
            seq_opt, val_opt = _lbfgs_optimize(seed_seq, n, deadline)
            if seq_opt is not None and val_opt < global_best_val:
                global_best_val = val_opt
                global_best_seq = seq_opt.copy()
                print(f"[polish‑L‑BFGS] n={n:5d} new global best={val_opt:.6f}")

        # --------------------------------------------------------------
        # 4. Final stochastic fine‑tuning on the overall best for this n
        # --------------------------------------------------------------
        if global_best_seq is not None and time.time() < deadline - 5.0:
            seq_hc, val_hc = _targeted_hill_climb(global_best_seq, n, deadline, max_iters=25000)
            if val_hc < global_best_val:
                global_best_val = val_hc
                global_best_seq = seq_hc.copy()
                print(f"[final‑hill] n={n:5d} new global best={val_hc:.6f}")

        # --------------------------------------------------------------
        # 5. Optional brief extra L‑BFGS pass (quick sanity)
        # --------------------------------------------------------------
        if global_best_seq is not None and time.time() < deadline - 5.0:
            seq_extra, val_extra = _lbfgs_optimize(global_best_seq, n, deadline)
            if val_extra < global_best_val:
                global_best_val = val_extra
                global_best_seq = seq_extra.copy()
                print(f"[extra‑L‑BFGS] n={n:5d} new global best={val_extra:.6f}")

    # ----------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq, n)
        print("[fallback] using uniform distribution")

    # ----------------------------------------------------------------------
    # Verify with the provided evaluate_sequence()
    # ----------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        # If rejected (should not happen), fall back to uniform
        print("[verification] sequence rejected – falling back to uniform")
        global_best_seq = _uniform(len(global_best_seq))
        global_best_val = _objective(global_best_seq, len(global_best_seq))

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})")
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
