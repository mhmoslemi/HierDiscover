def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    start = time.time()
    deadline = start + budget_s - 0.5

    n = kwargs.get('n', 4000)
    exponents = [2,3,4,5]  # try a few

    best_seq = None
    best_val = float('inf')
    for p in exponents:
        if time.time() > deadline:
            break
        i = np.arange(1, n+1, dtype=np.float64)
        seq = i ** p
        seq = seq / seq.sum()
        # Ensure nonnegatives etc
        val = evaluate_sequence(seq.tolist())
        print(f"[exp={p}] objective={val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    # as fallback, if no seq found, use uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0/n)
        best_val = evaluate_sequence(best_seq.tolist())
        print("[fallback] uniform sequence")
    print(f"\n=== Finished candidate ===\nBest objective ≈ {best_val:.6f} (exp={p}) length {len(best_seq)}")
    return best_seq.tolist()
