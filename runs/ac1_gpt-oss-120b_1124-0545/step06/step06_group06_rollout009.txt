"""
Enhanced search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (∑a)²

The algorithm expands on the previous pipeline:

1. **Very wide seed pool** – families are sampled with finer parameter
   grids, random mixtures and random rotations.  Up to a few hundred
   candidates are generated for every explored length.

2. **Quick evaluation** – all seeds are normalised to sum = 1 and the
   cheap objective 2·n·max(conv) is computed (the denominator is 1).

3. **Top‑K refinement** – the best ~20 seeds are fed to a short Adam
   optimisation (smoothed max via log‑sum‑exp).  The best result from
   this stage is taken forward.

4. **Heavy optimisation** – the current global best undergoes a long
   Adam run, an exact sub‑gradient mirror‑descent, and a simulated‑annealing
   hill‑climber.  Each stage respects the global deadline.

5. **Length exploration** – the whole pipeline is executed for a few
   lengths (4000, 6000, 8000).  The overall best sequence (any length) is
   returned.

All helper functions are top‑level; no closures or lambda expressions
are used.  The routine respects the supplied time budget (default 1000 s)
and prints progress information.
"""

import time
import numpy as np

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large n)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """
    Return the (scale‑invariant) objective 2·n·max(conv(seq,seq))
    assuming the sequence has been normalised to sum = 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    if not (0 < width_fraction < 1):
        return _uniform(n)
    half_flat = int(np.round(width_fraction * n / 2.0))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0 or beta <= 0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _rotate(v: np.ndarray, offset: int) -> np.ndarray:
    """Circular shift (wrap‑around)."""
    return np.roll(v, offset)

def _collect_extended_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """
    Build a large, diverse pool of normalised candidates for length ``n``.
    Includes fine parameter grids, random mixtures and random rotations.
    """
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for wf in (0.05, 0.1, 0.2, 0.3, 0.4):
        seeds.append(_trapezoidal(n, width_fraction=wf))

    # Geometric families – densely sampled
    for r in np.linspace(0.01, 0.99, 30):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # Power‑law family – finer grid up to p=6
    for p in np.linspace(0.0, 6.0, 25):
        seeds.append(_power(n, p))

    # Beta‑like families – random draws
    for _ in range(30):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        seeds.append(_beta_shape(n, a, b))

    # Sine‑window with power γ
    for gamma in np.linspace(0.1, 6.0, 20):
        i = np.arange(n, dtype=np.float64)
        a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
        seeds.append(a / a.sum())

    # Random Dirichlet draws – several concentrations
    for conc in (0.05, 0.2, 0.5, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse patterns – various sparsities
    for k_frac in (0.01, 0.02, 0.05, 0.1, 0.2):
        k = max(2, int(k_frac * n))
        for conc in (0.1, 0.5, 1.0):
            seeds.append(_random_sparse(n, k, conc, rng))

    # Convex mixtures of two random seeds
    base = seeds.copy()
    for _ in range(20):
        a, b = rng.choice(base, size=2, replace=False)
        w = rng.random()
        mix = w * a + (1.0 - w) * b
        mix = mix / mix.sum()
        seeds.append(mix)

    # Random rotations of all above (one per seed)
    rotated = []
    for v in seeds:
        offset = rng.integers(0, n)
        rotated.append(_rotate(v, offset))
    seeds.extend(rotated)

    # Previous global best (if compatible)
    if previous is not None and previous.shape == (n,):
        prev = np.asarray(previous, dtype=np.float64)
        if prev.sum() > 0:
            seeds.append(prev / prev.sum())

    # Clean: enforce non‑negativity and normalise
    normalised: list[np.ndarray] = []
    for v in seeds:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0:
            normalised.append(v / s)

    return normalised

) -> tuple[np.ndarray | None, float]:
    """Adam optimisation on a log‑sum‑exp smoothed max."""
    eps = 1e-30
    v = np.log(start_vec + eps)  # logits on simplex
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5
    beta_mul = 1.001
    beta_max = 200.0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  (linear in a)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # Convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Clip for stability
        norm = np.linalg.norm(grad_v)
        if norm > 10.0:
            grad_v = grad_v * (10.0 / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror descent on the true max objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step on the simplex
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)  # for numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """Simple SA with Dirichlet reshuffling of a few coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 5)  # 2‑4 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget and returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    # previous best (may be set by a previous run)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # Lengths to explore – larger lengths give a finer discretisation,
    # but also increase the n‑factor in the objective.
    lengths = [4000, 6000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Exploring length {n} ===")

        # ------------------------------------------------------------------
        # 1. Broad seed generation
        # ------------------------------------------------------------------
        seeds = _collect_extended_seeds(n, rng, previous)

        # quick evaluation of all seeds
        evaluated = []
        for s in seeds:
            val = _objective(s)
            evaluated.append((val, s))
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed‑global] n={n:5d} val={val:.6f}")

        # keep only the most promising few for further work
        evaluated.sort(key=lambda x: x[0])
        top_k = 20
        top_candidates = [s for _, s in evaluated[:top_k]]

        # ------------------------------------------------------------------
        # 2. Short Adam runs on the top candidates
        # ------------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(cand, n, deadline, max_iters=8000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[Adam‑top{idx}] improved global to {val:.6f}")

        # ------------------------------------------------------------------
        # 3. Heavy refinement from the current global best
        # ------------------------------------------------------------------
        if global_best_seq is None:
            # fall back to a uniform vector if everything failed
            global_best_seq = _uniform(n)
            global_best_val = _objective(global_best_seq)

        # longer Adam
        seq, val = _adam_optimize(global_best_seq, n, deadline, max_iters=30000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Adam‑final] improved global to {val:.6f}")

        # exact sub‑gradient mirror descent
        seq, val = _subgradient_refine(global_best_seq, n, deadline, max_iters=150000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[SubGrad‑final] improved global to {val:.6f}")

        # simulated‑annealing hill climbing
        seq, val = _simulated_annealing(global_best_seq, n, deadline, max_iters=200000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[SA‑final] improved global to {val:.6f}")

    # ----------------------------------------------------------------------
    # Final sanity check & return
    # ----------------------------------------------------------------------
    final_seq = global_best_seq.tolist() if global_best_seq is not None else _uniform(2000).tolist()
    try:
        evaluate_sequence(final_seq)
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(final_seq)})"
    )
    return final_seq
