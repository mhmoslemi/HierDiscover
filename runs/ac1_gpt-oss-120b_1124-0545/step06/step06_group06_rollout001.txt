def _lp_optimize(start_vec, n, deadline, p_schedule, iters_per_p):
    v = np.log(start_vec + eps)
    # initialize Adam moments
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_seq = None
    best_val = float('inf')
    t = 0

    for p in p_schedule:
        for it in range(iters_per_p):
            if time.time() > deadline:
                return best_seq, best_val
            t += 1
            a = _softmax(v)
            # enforce symmetry
            a = (a + a[::-1]) / 2.0
            # Renorm (softmax ensures sum =1, symmetry also preserves)
            a = a / a.sum()

            # compute objective for test
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[LP] p={p} iter={it} best={cur_val:.6f}")

            # compute convolution and gradient
            conv = _conv_fft(a, a)
            # compute p-norm
            loss = np.linalg.norm(conv, ord=p)  # p-norm = (sum conv^p)^{1/p}
            # compute weights
            # conv >= 0, safe
            w = conv**(p-1)
            # compute grad w.r.t a: 2 * (rev_a * w) central n entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n] / (loss**(p-1) + eps)  # scaling due to derivative dL/dconv = conv^{p-1} / (norm^{p-1})
            # transform to log-softmax space
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Clip gradients for stability
            max_norm = 5.0
            norm = np.linalg.norm(grad_v)
            if norm > max_norm:
                grad_v = grad_v * (max_norm / norm)

            lr = 0.5 / np.sqrt(t)  # decaying step
            m = beta1 * m + (1-beta1) * grad_v
            v2 = beta2 * v2 + (1-beta2) * (grad_v**2)

            m_hat = m / (1-beta1**t)
            v2_hat = v2 / (1-beta2**t)

            v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val

"""
Propose a step–function (non‑negative list) that minimizes

    2·n·max(conv(a,a)) / (∑a)²

The search consists of four time‑aware stages.

1️⃣  **Seed generation** – a very diverse collection of analytically
    motivated and random families (uniform, triangular, geometric,
    beta‑like, Dirichlet, sparse, …).  The best few seeds (by the exact
    objective) are kept.

2️⃣  **Lp‑smooth optimisation** – the non‑smooth max is approximated by
    the Lp‑norm of the convolution (p → ∞).  For a schedule of increasing
    p (2, 4, 8, 12, 16, 24, 32) we perform Adam updates on a log‑softmax
    parametrisation of the vector.  After every update the vector is
    symmetrised (a ← (a+a[::-1])/2) because the optimum is expected to be
    symmetric.  The gradient of the p‑norm is computed analytically using
    FFT‑based convolutions.

3️⃣  **Exact sub‑gradient mirror descent** – a classic exponentiated‑gradient
    step for the true max (as in the reference implementation) refines the
    result further.

4️⃣  **Simulated‑annealing hill climbing** – stochastic Dirichlet‑reshuffles
    of a few coordinates allow the search to escape shallow local minima.

All stages respect the overall time budget (default 1000 s).  The best
sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT.
    Returned length is len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector on the simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the competition objective for a *probability*
    vector (i.e. ∑seq==1).  It returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic shape families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    if not (0 < width_fraction < 1):
        return _uniform(n)
    half_flat = int(np.round(width_fraction * n / 2.0))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0 or beta <= 0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a broad set of normalised candidate vectors for length n."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.2))
    seeds.append(_sine_window(n))

    # geometric families
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_geom(n, r))
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_sym_geom(n, r))

    # power‑law family
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    # Beta family
    alphas = np.linspace(0.2, 2.0, 10)
    betas = np.linspace(0.2, 2.0, 10)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Sine^γ families
    for gamma in np.linspace(0.5, 4.0, 8):
        i = np.arange(n, dtype=np.float64)
        a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
        seeds.append(a / a.sum())

    # Random Dirichlet draws
    for conc in (0.1, 0.3, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse patterns
    for k in (3, 5, 10):
        for conc in (0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous global best (if compatible)
    if previous is not None and previous.shape == (n,):
        prev = np.asarray(previous, dtype=np.float64)
        if prev.sum() > 0:
            seeds.append(prev / prev.sum())

    return seeds

def _lp_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    p_schedule: list[int],
    iters_per_p: int,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on the Lp‑norm of the self‑convolution.
    The vector lives on the simplex via a log‑softmax parametrisation.
    After each update the vector is forced to be symmetric.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits for softmax

    # Adam state
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_seq = None
    best_val = float("inf")
    it_global = 0

    for p in p_schedule:
        for it_local in range(iters_per_p):
            if time.time() > deadline:
                return best_seq, best_val
            it_global += 1

            # simplex + symmetry
            a = _softmax(v)
            a = (a + a[::-1]) / 2.0
            a = a / a.sum()          # still exactly 1 after symmetrisation

            # current objective (exact)
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Lp‑opt] p={p:2d} it={it_global:6d} best={cur_val:.6f}")

            # convolution and gradient of Lp‑norm
            conv = _conv_fft(a, a)
            loss = np.linalg.norm(conv, ord=p)
            # avoid division by zero (loss>0 because conv has positive entries)
            denom = loss ** (p - 1) + eps
            w = conv ** (p - 1) / denom

            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # map to log‑softmax coordinates (mirror‑descent style)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # stabilise the step
            max_norm = 5.0
            norm = np.linalg.norm(grad_v)
            if norm > max_norm:
                grad_v = grad_v * (max_norm / norm)

            # learning rate decays as 1/√t
            lr = 0.5 / np.sqrt(it_global)

            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** it_global)
            v2_hat = v2 / (1.0 - beta2 ** it_global)

            v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 30000,
) -> tuple[np.ndarray | None, float]:
    """
    (Kept for compatibility – a smoothed max using log‑sum‑exp.)
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5            # initial smooth‑max temperature
    beta_mul = 1.001
    beta_max = 200.0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # increase smoothing (cap at beta_max)
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp over convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent (exponentiated‑gradient) on the true max.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max at lag max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill climbing with a simple annealing schedule.
    Randomly re‑allocate the total mass of 2‑4 positions via a Dirichlet draw.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    # initial temperature proportional to current objective
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 5)          # choose 2, 3 or 4 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:7d} best={new_val:.6f}")

    return best_seq, best_val
