def _adam_max_optimize(seed, n, deadline, max_iters=20000):
    v = np.log(seed + 1e-30)  # logits vector
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    best_val = float('inf')
    best_seq = None

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = 2 * n * conv[max_idx] / (a.sum() ** 2) # but a sums to 1, so = 2*n*conv[max_idx]
        # but we can compute simple: cur_val = 2 * n * conv[max_idx] # since sum=1.
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(...) # log

        # Compute subgradient g (size n)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # gradient w.r.t v
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)
        m_hat = m / (1 - 0.9 ** it)
        v2_hat = v2 / (1 - 0.999 ** it)
        lr = 0.5 / np.sqrt(it)  # decaying lr
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val
```

def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    previous = globals().get("height_sequence_1")
    if previous is not None:
        previous = np.asarray(previous, dtype=np.float64)
    else:
        previous = None

    # Determine length schedule depending on remaining time
    lengths = [4000, 8000, 16000]
    # Possibly add 32000 if time.

    global_best_seq = None
    global_best_val = float('inf')

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n--- Optimising length {n} ---")
        # Build seeds
        seeds = _make_seeds(n, previous)
        # evaluate quickly
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        # keep top 3 seeds
        top_seeds = [s for s,_ in seed_vals[:3]]
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] new global best {global_best_val}")

        # Run Adam+subgradient for each top seed
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_max_optimize(seed_vec, n, deadline, max_iters=20000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[AdamMax] n={n}, seed={idx} val={val}")

        # Possibly run annealing on global best for this length
        if global_best_seq is not None and global_best_seq.shape[0]==n:
            seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=50000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[Anneal] n={n} val={val}")

        # Symmetrize
        if global_best_seq is not None and global_best_seq.shape[0]==n:
            sym = (global_best_seq + global_best_seq[::-1]) / 2.0
            sym /= sym.sum()
            sym_val = _objective(sym)
            if sym_val < global_best_val:
                global_best_val = sym_val
                global_best_seq = sym.copy()
                print(f"[sym] val={sym_val}")

        # upscale for next length
        previous = global_best_seq

    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        global_best_val = _objective(global_best_seq)

    # store globally
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(f"\nFinished best objective {global_best_val} length {len(global_best_seq)}")
    return global_best_seq.tolist()

"""
Optimiser for the self‑convolution infinity‑norm.

The algorithm improves on the previous implementation by:
  * Using Adam on the *exact* (non‑smoothed) max‑convolution loss, i.e.
    the sub‑gradient of max(conv) is back‑propagated through a
    softmax‑parameterisation of the simplex.
  * Starting from a rich seed pool (uniform, geometric, power‑law,
    Beta‑distributions, Dirichlet draws and an up‑scaled previous best)
    and also exploring random perturbations of the current best.
  * Employing a longer training schedule (per‑seed Adam up to 20 k
    iterations) and a final lightweight simulated‑annealing fine‑tuner.
  * Adding an optional symmetrisation step (averaging with the mirror)
    which often reduces the peak of the convolution.
The whole search respects the 1000 s wall‑time.  The best sequence is
cached in the global variable ``height_sequence_1`` for reuse across
invocations.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (still O(n log n))."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the grader (scale‑invariant)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    a = np.ones(n, dtype=np.float64)
    return a / a.sum()


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i = r^i (normalized)."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i = 1/(i+1)^p (normalized)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_seed(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric (or mildly asymmetric) Beta(α,β) shape."""
    x = np.arange(1, n + 1, dtype=np.float64) / (n + 1.0)
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sample from Dirichlet(α,…,α) and normalise."""
    vec = rng.gamma(alpha, 1.0, size=n)
    s = vec.sum()
    return vec / s if s > 0 else _uniform(n)


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length (preserves total mass)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    new_seq /= new_seq.sum()
    return new_seq


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a reasonably diverse set of start vectors (already normalised)."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))

    # geometric families
    for r in np.linspace(0.1, 0.9, 7):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 7):
        seeds.append(_power(n, p))

    # Beta families (symmetric and modestly asymmetric)
    for a in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]:
        seeds.append(_beta_seed(n, a, a))
        seeds.append(_beta_seed(n, a, a * 1.5))
        seeds.append(_beta_seed(n, a * 1.5, a))

    # Dirichlet draws
    for alpha in [0.1, 0.3, 0.6, 1.0, 2.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # up‑scaled previous best (if any)
    if previous is not None:
        try:
            seeds.append(_upscale(previous, n))
        except Exception:
            pass

    # random perturbations of the best candidate (if we have one)
    if previous is not None:
        for _ in range(3):
            noise = rng.normal(scale=0.02, size=n)
            pert = previous + noise
            pert = np.clip(pert, 0.0, None)
            if pert.sum() == 0.0:
                continue
            pert /= pert.sum()
            seeds.append(pert)

    # ensure at least one seed
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Optimisation: Adam on the *exact* max‑convolution loss
# ----------------------------------------------------------------------


def _adam_max_optimize(seed: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 20000,
                       lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the true max‑convolution objective.
    Parameters are the logits (v) of a soft‑max parametrisation of the simplex.
    Returns the best sequence found and its true objective value.
    """
    # initialise logits safely
    v = np.log(seed + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # point on the simplex (sum‑1)
        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        cur_val = 2.0 * n * conv[max_idx]   # because sum(a)=1

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[AdamMax] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # gradient w.r.t logits (softmax Jacobian)
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)               # decaying learning rate
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner (lightweight)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Local search that swaps tiny fragments of mass; accepts worse moves via a cooling schedule."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature roughly 4 % of the current objective
    T0 = max(1e-9, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move a tiny portion of mass from i → j
        delta = rng.random() * 0.025 * cur[i]   # ≤2.5 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # occasional global jitter to escape plateaus
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            jitter = cur + noise
            jitter = np.clip(jitter, 0.0, None)
            if jitter.sum() == 0.0:
                jitter = _uniform(n)
            else:
                jitter /= jitter.sum()
            jitter_val = _objective(jitter)
            if jitter_val < cur_val or rng.random() < np.exp(-(jitter_val - cur_val) / (T + 1e-12)):
                cur = jitter
                cur_val = jitter_val
                if jitter_val < best_val:
                    best_val = jitter_val
                    best = jitter.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={jitter_val:.6f}")

    return best, best_val

# Main entry point – respects the 1000 s budget
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns the best sequence (list of floats) found before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # retrieve the best construction from previous runs, if any
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # Length schedule – increase if time permits
    length_plan = [4000, 8000, 16000, 32000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in length_plan:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # Build a diverse seed pool (already normalised)
        # ------------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # ------------------------------------------------------------------
        # Quick evaluation of raw seeds, keep the few best
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:3]]

        # Record the overall best raw seed
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] new global best = {global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation on exact max‑convolution
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_max_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=20000,
                lr0=0.6,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall AdamMax] n={n} seed={idx} val={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} val={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Symmetrisation (often reduces the peak)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            sym = (global_best_seq + global_best_seq[::-1]) / 2.0
            sym /= sym.sum()
            sym_val = _objective(sym)
            if sym_val < global_best_val:
                global_best_val = sym_val
                global_best_seq = sym.copy()
                print(f"[symmetrisation] n={n} improved to {sym_val:.6f}")

        # Propagate the current best forward (for up‑scaling next length)
        previous = global_best_seq

    # ----------------------------------------------------------------------
    # Fallback safeguard
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # cache for future calls
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
