def propose_candidate(seed=123, budget_s=1000):
    import time, numpy as np

    start_time = time.time()
    deadline = start_time + budget_s - 5 # safety

    n = 4000 # can also try other lengths later

    # function for objective
    def obj(a):
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 0.01:
            return np.inf
        # conv via FFT
        L = len(a)
        N = 1 << ((2*L - 2).bit_length())
        # using rfft
        fa = np.fft.rfft(a, n=N)
        conv = np.fft.irfft(fa*fa, n=N)[:2*L-1]
        maxb = np.max(conv)
        return 2.0 * L * maxb / (s * s)
    ```

def random_search_stage(n, deadline):
    i = np.arange(1, n+1, dtype=np.float64)
    I0 = np.arange(n, dtype=np.float64)   # 0-indexed for Gaussian
    best_seq = None
    best_val = np.inf
    rng = np.random.default_rng(12345)
    while time.time() < deadline:
        # Random number of components
        comp = rng.integers(1, 4) # 1-3 inclusive
        weights = rng.dirichlet(np.ones(comp))
        seq = np.zeros(n, dtype=np.float64)
        for k in range(comp):
            shape_type = rng.choice(['beta', 'gauss', 'uniform'])
            if shape_type == 'beta':
                alpha = rng.uniform(0.05, 6.0)
                beta_ = rng.uniform(0.05, 6.0)
                left = i ** (alpha - 1.0)
                right = (n - i + 1) ** (beta_ - 1.0)
                s = left * right
            elif shape_type == 'gauss':
                sigma = rng.uniform(0.05 * n, 0.4 * n)
                # mu = (n-1)/2
                mu = (n - 1) / 2.0
                d = (I0 - mu) / sigma
                s = np.exp(-0.5 * d * d)
            else: # uniform
                s = np.ones(n, dtype=np.float64)
            s_sum = s.sum()
            if s_sum > 0:
                s = s / s_sum
            else:
                s = np.full(n, 1.0/n)
            seq += weights[k] * s
        # ensure non-negative (they are)
        seq_sum = seq.sum()
        if seq_sum == 0:

        val = compute_obj(seq) # maybe compute directly as 2n*max_conv since sum=1
        if val < best_val:
            best_val = val

def subgradient_refine(seq, n, deadline, max_iters):
    a = seq.copy()
    best_seq = a.copy()
    best_val = compute_obj(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = conv_fft(a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum()**2)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it} best={cur_val:.6f}")
        # compute subgradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]   # as earlier

        lr = 0.6 / np.sqrt(it)

        if a_sum == 0:
            a = np.full_like(a, 1.0/n)

            a /= a_sum

def hill_climb(seq, n, deadline, max_iters=50000):
    rng = np.random.default_rng(2023)
    cur = seq.copy()
    cur_val = compute_obj(cur)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        i,j = rng.integers(0,n,size=2)
        if i==j: continue
        donor=i if rng.random()<0.5 else j
        receiver=j if donor is i else i
        max_transfer = min(cur[donor], step_scale*cur[donor] + 1e-12)
        if max_transfer <=0: continue
        delta = rng.random() * max_transfer
        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta
        val = compute_obj(new)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] it={it} best={val:.6f}")
        if it % 2000 == 0:
            step_scale = max(step_scale*0.9, 1e-6)
    return best_seq, best_val

"""
Propose a candidate non‑negative step function that improves the
max‑auto‑convolution objective.

Algorithm
---------
* Initialise from any previously stored sequence.
* Perform a fast random‑search over compact parametric families
  (Beta‑shapes, Gaussian windows, uniform).  Each candidate is a
  convex combination of 1‑3 such basis shapes, normalised to unit
  ℓ₁‑norm.
* If time permits, run a low‑dimensional Differential Evolution
  (piece‑wise‑linear representation) to explore richer shapes.
* Refine the current best with deterministic sub‑gradient mirror‑descent
  and a simple stochastic hill‑climber.
* Return the best sequence (scaled to the usual ℓ₁‑norm √(2 n)).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return conv


def _conv_fft_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors with a real FFT."""
    L = x.shape[0] + y.shape[0] - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (unit ℓ₁‑norm)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float, idx: np.ndarray) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i)^{α‑1}·(n‑i+1)^{β‑1}."""
    left = idx ** (alpha - 1.0)
    right = (n - idx + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _gaussian(n: int, sigma: float, idx0: np.ndarray) -> np.ndarray:
    """Symmetric Gaussian window (zero‑mean)."""
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((idx0 - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


# ----------------------------------------------------------------------
# Objective evaluation (identical to evaluate_sequence but faster)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """2·n·max(conv) / (Σa)² for a non‑negative vector a."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = _conv_fft(a)
    maxb = conv.max()
    return 2.0 * a.shape[0] * maxb / (s * s)


# ----------------------------------------------------------------------
# Random search over mixtures of analytic families
# ----------------------------------------------------------------------


def _random_search(n: int,
                   deadline: float,
                   best_seq: np.ndarray | None,
                   best_val: float) -> tuple[np.ndarray, float]:
    """
    Randomly sample convex combinations of a few simple shapes.
    Continues until the deadline is reached.
    """
    rng = np.random.default_rng(20231124)
    idx = np.arange(1, n + 1, dtype=np.float64)      # for beta
    idx0 = np.arange(n, dtype=np.float64)           # for Gaussian

    # start from the provided best (if any)
    cur_best_seq = best_seq.copy() if best_seq is not None else None
    cur_best_val = best_val

    while time.time() < deadline:
        # number of components in the mixture (1‑3)
        comps = rng.integers(1, 4)
        weights = rng.dirichlet(np.ones(comps))
        seq = np.zeros(n, dtype=np.float64)

        for k in range(comps):
            shape_type = rng.choice(["beta", "gauss", "uniform"])
            if shape_type == "beta":
                alpha = rng.uniform(0.05, 6.0)
                beta_ = rng.uniform(0.05, 6.0)
                s = _beta_shape(n, alpha, beta_, idx)
            elif shape_type == "gauss":
                sigma = rng.uniform(0.05 * n, 0.4 * n)
                s = _gaussian(n, sigma, idx0)
            else:  # uniform
                s = _uniform(n)
            seq += weights[k] * s

        # normalise to unit ℓ₁‑norm (sum = 1)
        seq_sum = seq.sum()
        if seq_sum == 0.0:
            continue
        seq /= seq_sum

        val = _objective(seq)
        if val < cur_best_val:
            cur_best_val = val
            cur_best_seq = seq.copy()
            print(f"[Random] new best = {val:.6f}")

    # fall‑back to uniform if we never found anything
    if cur_best_seq is None:
        cur_best_seq = _uniform(n)
        cur_best_val = _objective(cur_best_seq)
        print("[Random] fallback to uniform")

    return cur_best_seq, cur_best_val


# ----------------------------------------------------------------------
# Differential Evolution on a piece‑wise‑linear representation
# ----------------------------------------------------------------------


def _eval_param(p: np.ndarray, knots: np.ndarray, n: int) -> tuple[float, np.ndarray]:
    """
    Interpolate a K‑dimensional simplex vector `p` (∑p = 1) to a full
    length‑`n` sequence, normalise (ℓ₁‑norm = 1) and return the exact
    objective together with the full sequence.
    """
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    if s == 0.0:
        a = _uniform(n)
        s = a.sum()
    a = a / s
    val = _objective(a)
    return val, a


def _de_optimize(n: int,
                 deadline: float,
                 popsize: int = 50,
                 K: int | None = None,
                 max_gens: int = 250) -> tuple[np.ndarray, float]:
    """
    Differential Evolution working on a piece‑wise‑linear (K‑knot)
    description of the sequence.
    """
    rng = np.random.default_rng(20231202)

    if K is None:
        K = max(50, int(np.sqrt(n) * 3))   # richer representation
    knots = np.linspace(0, n - 1, K)

    # initialise population on the simplex (Dirichlet draws)
    pop = [rng.dirichlet(np.ones(K)) for _ in range(popsize)]
    pop_vals = np.empty(popsize, dtype=np.float64)
    pop_seq = [None] * popsize
    for i in range(popsize):
        v, s = _eval_param(pop[i], knots, n)
        pop_vals[i] = v
        pop_seq[i] = s

    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_seq = pop_seq[best_idx].copy()
    best_p = pop[best_idx].copy()

    F = 0.8
    CR = 0.9
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # choose three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a_vec = pop[a_idx]
            b_vec = pop[b_idx]
            c_vec = pop[c_idx]

            donor = a_vec + F * (b_vec - c_vec)
            donor = np.maximum(donor, 0.0)
            ds = donor.sum()
            if ds == 0.0:
                donor = _uniform(K)
            else:
                donor /= ds

            # binomial crossover
            trial = pop[i].copy()
            jrand = rng.integers(K)
            for j in range(K):
                if rng.random() < CR or j == jrand:
                    trial[j] = donor[j]

            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts == 0.0:
                trial = _uniform(K)
            else:
                trial /= ts

            val_trial, seq_trial = _eval_param(trial, knots, n)

            if val_trial < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val_trial
                pop_seq[i] = seq_trial
                if val_trial < best_val:
                    best_val = val_trial
                    best_seq = seq_trial.copy()
                    best_p = trial.copy()

        if gen % 20 == 0:
            print(f"[DE] n={n:5d} gen={gen:3d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent style)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 120000) -> tuple[np.ndarray, float]:
    """Mirror‑descent sub‑gradient minimisation of the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

        # sub‑gradient for the active max entry (indices that contribute)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent (exponentiated gradient) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)            # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (final polishing)
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 100000) -> tuple[np.ndarray, float]:
    """Random pairwise mass transfer; accept only improvements."""
    rng = np.random.default_rng(20231125)
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new)

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] it={it:6d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine combines a fast
    random mixture search, a higher‑dimensional DE stage and deterministic
    refinements (sub‑gradient + hill‑climb).  The best sequence is
    returned as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # small safety margin

    # ------------------------------------------------------------------
    # Problem size – keep it modest to stay within the time budget
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Start from any previously stored sequence (if present)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    best_seq = None
    best_val = float("inf")
    if previous is not None:
        try:
            cand = np.asarray(previous, dtype=np.float64)
            if cand.shape[0] == n:
                best_seq = cand.copy()
                best_val = _objective(best_seq)
                print(f"[Init] previous best = {best_val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. Random mixture search (≈60 % of the total budget)
    # ------------------------------------------------------------------
    random_deadline = start_time + 0.60 * budget_s
    best_seq, best_val = _random_search(
        n,
        random_deadline,
        best_seq,
        best_val,
    )
    print(f"[After Random] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Global DE on a piecewise‑linear representation (if time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        de_deadline = min(deadline, start_time + 0.80 * budget_s)
        seq_de, val_de = _de_optimize(
            n,
            de_deadline,
            popsize=50,
            K=None,
            max_gens=300,
        )
        if val_de < best_val:
            best_seq = seq_de.copy()
            best_val = val_de
            print(f"[DE] new global best = {val_de:.6f}")

    # ------------------------------------------------------------------
    # 3. Exact sub‑gradient refinement (mirror descent)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sg_deadline = deadline
        seq_sg, val_sg = _subgradient_refine(
            best_seq,
            n,
            sg_deadline,
            max_iters=120000,
        )
        if val_sg < best_val:
            best_seq = seq_sg.copy()
            best_val = val_sg
            print(f"[SubGrad] new global best = {val_sg:.6f}")

    # ------------------------------------------------------------------
    # 4. Light stochastic hill‑climbing for final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_hc, val_hc = _hill_climb(
            best_seq,
            n,
            deadline,
            max_iters=100000,
        )
        if val_hc < best_val:
            best_seq = seq_hc.copy()
            best_val = val_hc
            print(f"[Hill] final best = {val_hc:.6f}")

    # ------------------------------------------------------------------
    # Rescale to the conventional ℓ₁‑norm √(2 n) (does not affect the
    # objective but matches the normalisation used in the reference)
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    best_seq = best_seq * (target_sum / best_seq.sum())

    # store for possible reuse by a next run
    globals()["height_sequence_1"] = best_seq.tolist()

    # optional verification (ignore any exception – the judge will call it)
    try:
        _ = evaluate_sequence(best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {n})")
    return best_seq.tolist()
