"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

The algorithm is built around three ideas:

1. **Rich analytic seeding** – many classical windows (sine, Blackman,
   Kaiser, Tukey, Gaussian, Beta) plus random Dirichlet draws.  All seeds
   are ℓ₁‑normalised (sum = 1) because the objective is scale‑invariant.

2. **Global search on a low‑dimensional piece‑wise linear representation**
   via Differential Evolution (DE).  The representation uses ≈√n knots
   (≈63 for n=4000) and interpolates to a full‑length candidate before
   evaluation.  The best DE individual is fed to the local refinements.

3. **Intensive local polishing** – a combination of
   * Adam on a smooth‑max surrogate (more iterations and a slower
     temperature schedule),
   * exact sub‑gradient mirror descent,
   * projected sub‑gradient descent (Euclidean projection onto the ℓ₁‑simplex),
   * simulated‑annealing hill‑climbing (pairwise mass transfers with
     occasional uphill moves).

All stages respect a global time‑budget (default 1000 s).  The best
sequence found so far is continuously updated and finally returned as a
plain list of floats.
"""

import time
import numpy as np
from scipy.special import i0  # for Kaiser window

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (keeps the vector on the simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian window, centred at (n‑1)/2."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _blackman_window(n: int, a: float = 0.16) -> np.ndarray:
    """
    Generalised Blackman window:
    w[i] = a - 0.5*cos(2πi/(n‑1)) + (0.5‑a)*cos(4πi/(n‑1))
    """
    i = np.arange(n, dtype=np.float64)
    w = a - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + (0.5 - a) * np.cos(4 * np.pi * i / (n - 1))
    w = np.maximum(w, 0.0)
    return w / w.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window w[i] = I0(β·sqrt(1‑((2i/(n‑1)‑1)^2))) / I0(β)."""
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = i0(beta * np.sqrt(1.0 - x * x))
    w = np.maximum(w, 0.0)
    return w / w.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """
    Tukey (cosine‑tapered) window.
    alpha in [0,1]; 0 → rectangular (uniform), 1 → Hann.
    """
    if alpha <= 0:
        return _uniform(n)
    if alpha >= 1:
        i = np.arange(n, dtype=np.float64)
        return 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    i = np.arange(n, dtype=np.float64)
    period = (n - 1) * alpha / 2.0
    w = np.ones_like(i, dtype=np.float64)
    left = i < period
    right = i > (n - 1 - period)
    w[left] = 0.5 * (1 + np.cos(np.pi * (2 * i[left] / (alpha * (n - 1)) - 1)))
    w[right] = 0.5 * (1 + np.cos(np.pi * (2 * i[right] / (alpha * (n - 1)) - 2 / alpha + 1)))
    return w / w.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of ℓ₁‑normalised start vectors."""
    seeds: list[np.ndarray] = []

    # classic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # Beta shapes – a denser grid than before
    for a in np.linspace(0.2, 2.0, 9):
        for b in np.linspace(0.2, 2.0, 9):
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows
    for frac in np.linspace(0.05, 0.4, 8):
        sigma = max(1.0, frac * n)
        seeds.append(_gaussian(n, sigma))

    # Blackman family
    for a in np.linspace(0.0, 0.5, 6):
        seeds.append(_blackman_window(n, a))

    # Kaiser family
    for beta in np.linspace(0.5, 10.0, 6):
        seeds.append(_kaiser_window(n, beta))

    # Tukey family
    for alpha in np.linspace(0.0, 1.0, 6):
        seeds.append(_tukey_window(n, alpha))

    rng = np.random.default_rng(2023)
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())
    return seeds

# Adam optimisation on a smoothed max surrogate

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 40000,
                   beta0: float = 5.0,
                   beta_mul: float = 1.02) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a log‑softmax parametrisation.
    ``beta0`` is the initial smoothing coefficient for the smooth‑max,
    multiplied by ``beta_mul`` each iteration.
    """
    v = np.log(seed + 1e-30)          # softmax parameters (stay on simplex)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = beta0
    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # a lies on the ℓ₁‑simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ---- smooth‑max weights ----
        beta *= beta_mul
        shift = np.max(conv)                     # numerical stabiliser
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of weighted sum Σ w_k·conv_k w.r.t a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]    # central n entries

        # convert to gradient w.r.t. logits v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val

# Exact sub‑gradient refinement (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 150000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent step (log‑space)
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val

def _proj_subgradient_refine(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300000,
                             target_sum: float | None = None) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent (Euclidean projection onto the ℓ₁‑simplex).
    """
    if target_sum is None:
        target_sum = np.sqrt(2.0 * n)          # the normalisation used in the literature

    a = _project_simplex(start_seq, target_sum)

    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ProjSub] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient for the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step‑size
        lr = 1.0 / np.sqrt(it)

        a_tmp = a - lr * g
        a = _project_simplex(a_tmp, target_sum)

    return best_seq, best_val

# Simulated‑annealing hill‑climbing
# ----------------------------------------------------------------------


def _annealed_hill_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200000,
                         init_temp: float = 1e-3,
                         cooling: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Pairwise mass transfer with Metropolis acceptance.
    The temperature slowly decays (geometric cooling).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val
    T = init_temp

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a random fraction of mass from i to j (or vice‑versa)
        if rng.random() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i

        max_transfer = cur[donor]
        if max_transfer == 0.0:
            continue
        delta = rng.random() * max_transfer * 0.05  # up to 5 % of donor mass

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        # Metropolis acceptance
        accept = False
        if val < cur_val:
            accept = True
        else:
            prob = np.exp(-(val - cur_val) / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        # geometric cooling
        T *= cooling
        if T < 1e-12:
            T = 1e-12

    return best_seq, best_val

def _eval_param(p: np.ndarray, knots: np.ndarray, n: int) -> tuple[float, np.ndarray]:
    """
    Interpolate a low‑dimensional simplex vector `p` (sum=1) to a full
    length‑`n` sequence, normalise (to sum=1) and evaluate the exact objective.
    Returns (objective, full_sequence).
    """
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    if s == 0.0:
        a = _uniform(n)
        s = a.sum()
    a = a / s
    conv = _conv_fft(a, a)
    val = 2.0 * n * np.max(conv)
    return val, a


def _de_optimize(n: int,
                 deadline: float,
                 popsize: int = 60,
                 K: int | None = None,
                 max_gens: int = 250) -> tuple[np.ndarray, float]:
    """
    Differential Evolution on a piece‑wise‑linear representation.
    """
    rng = np.random.default_rng()

    if K is None:
        K = max(30, int(np.sqrt(n) * 1.5))
    knots = np.linspace(0, n - 1, K)

    # initialise population on the simplex (Dirichlet draws)
    pop = [rng.dirichlet(np.ones(K)) for _ in range(popsize)]
    pop_vals = np.empty(popsize, dtype=np.float64)
    pop_seq = [None] * popsize
    for i in range(popsize):
        val, seq = _eval_param(pop[i], knots, n)
        pop_vals[i] = val
        pop_seq[i] = seq

    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_seq = pop_seq[best_idx].copy()
    best_p = pop[best_idx].copy()

    F = 0.8
    CR = 0.9
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            donor = pop[a_idx] + F * (pop[b_idx] - pop[c_idx])
            donor = np.maximum(donor, 0.0)
            ds = donor.sum()
            if ds == 0.0:
                donor = _uniform(K)
            else:
                donor = donor / ds

            # binomial crossover
            trial = pop[i].copy()
            jrand = rng.integers(K)
            for j in range(K):
                if rng.random() < CR or j == jrand:
                    trial[j] = donor[j]

            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts == 0.0:
                trial = _uniform(K)
            else:
                trial = trial / ts

            val_trial, seq_trial = _eval_param(trial, knots, n)

            if val_trial < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val_trial
                pop_seq[i] = seq_trial
                if val_trial < best_val:
                    best_val = val_trial
                    best_seq = seq_trial.copy()
                    best_p = trial.copy()

        if gen % 20 == 0:
            print(f"[DE] n={n:5d} gen={gen:3d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine combines a thorough analytic seeding, a global DE stage,
    several aggressive gradient‑based refinements and a final annealed
    hill‑climb.  The best sequence found within the allocated budget
    (default 1000 s) is returned as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # try to re‑use a previously discovered construction, if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Problem size – 4000 works well within the time budget
    # ------------------------------------------------------------------
    n = 4000

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # 1. Analytic seeds (including the richer families above)
    # ------------------------------------------------------------------
    print(f"\n=== Analytic seeding (n={n}) ===")
    seeds = _make_seeds(n, previous)
    for s in seeds:
        if time.time() > deadline:
            break
        val = 2.0 * n * np.max(_conv_fft(s, s))
        if val < global_best_val:
            global_best_val = val
            global_best_seq = s.copy()
            print(f"[seed] new global best={val:.6f}")

    # ------------------------------------------------------------------
    # 2. Global DE search (a bit more aggressive than the reference)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n=== Differential Evolution ===")
        seq_de, val_de = _de_optimize(n, deadline, popsize=60, K=None, max_gens=250)
        if val_de < global_best_val:
            global_best_val = val_de
            global_best_seq = seq_de.copy()
            print(f"[DE] new global best={val_de:.6f}")

    # ------------------------------------------------------------------
    # 3. Multiple Adam runs (starting from the current best and a few random restarts)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n=== Adam refinement (multiple restarts) ===")
        restarts = 3
        for r in range(restarts):
            if time.time() > deadline:
                break
            if r == 0:
                seed_seq = global_best_seq
            else:
                # generate a fresh random Dirichlet seed
                rng = np.random.default_rng(seed + r * 100)
                seed_seq = rng.dirichlet(np.full(n, 0.5))
                seed_seq = seed_seq / seed_seq.sum()
            seq_adam, val_adam = _adam_optimize(seed_seq, n, deadline,
                                                max_iters=40000,
                                                beta0=5.0, beta_mul=1.02)
            if seq_adam is not None and val_adam < global_best_val:
                global_best_val = val_adam
                global_best_seq = seq_adam.copy()
                print(f"[Adam‑restart {r}] new global best={val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4. Exact sub‑gradient refinement (longer run)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n=== Exact sub‑gradient refinement ===")
        seq_sg, val_sg = _subgradient_refine(global_best_seq, n, deadline,
                                             max_iters=150000)
        if val_sg < global_best_val:
            global_best_val = val_sg
            global_best_seq = seq_sg.copy()
            print(f"[SubGrad] new global best={val_sg:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient refinement (very long run)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n=== Projected sub‑gradient refinement ===")
        seq_psg, val_psg = _proj_subgradient_refine(global_best_seq, n, deadline,
                                                   max_iters=300000)
        if val_psg < global_best_val:
            global_best_val = val_psg
            global_best_seq = seq_psg.copy()
            print(f"[ProjSub] new global best={val_psg:.6f}")

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing hill‑climb (final polishing)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n=== Simulated‑annealing hill‑climb ===")
        seq_hc, val_hc = _annealed_hill_climb(global_best_seq, n, deadline,
                                              max_iters=200000,
                                              init_temp=1e-3,
                                              cooling=0.9995)
        if val_hc < global_best_val:
            global_best_val = val_hc
            global_best_seq = seq_hc.copy()
            print(f"[Hill‑Anneal] new global best={val_hc:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be required)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        print("[fallback] using uniform distribution")
        global_best_seq = _uniform(n)
        global_best_val = 2.0 * n * np.max(_conv_fft(global_best_seq,
                                                    global_best_seq))

    # --------------------------------------------------------------
    # Optional verification with the provided evaluate_sequence()
    # --------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    # expose the result under a conventional name for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
