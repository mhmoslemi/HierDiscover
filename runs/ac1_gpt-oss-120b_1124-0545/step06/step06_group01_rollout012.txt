def _pnorm_and_grad(v, n, p):
    a = _softmax(v)
    conv = _conv_fft(a, a)
    conv_pow = conv ** p
    S = np.sum(conv_pow)
    # gradient wrt a
    w = p * conv ** (p - 1)   # shape L
    rev_a = a[::-1]
    grad_full = _conv_fft(rev_a, w)   # length 3n-2
    grad_a = 2.0 * grad_full[n-1:n-1+n]   # central n entries
    # gradient wrt v (softmax)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return S, grad_v

                          p_schedule=None, maxfun_per_stage=500):
    if p_schedule is None:
        p_schedule = [2, 5, 10, 20, 40, 80]
    v = np.log(start_seq + 1e-30)
    for p in p_schedule:
        if time.time() >= deadline:
            break
        # adjust maxfun based on remaining time and dimension maybe
        # since we may want many evaluations
        maxfun = maxfun_per_stage
        # use minimize from scipy if available else fallback empirical gradient descent
        try:
            from scipy.optimize import minimize
            res = minimize(_pnorm_obj, v, args=(n, p), jac=True,
                           method='L-BFGS-B', options={'maxfun': maxfun, 'maxiter': maxfun, 'disp': False})
            v = res.x
        except Exception:
            # fallback: simple gradient descent with decaying step
            step = 1.0

                step *= 0.99
        # optional early exit if improvement saturates
    a_opt = _softmax(v)
    a_opt = a_opt / a_opt.sum()  # ensure sum=1
    return a_opt

"""
step‑function optimiser – p‑norm L‑BFGS + targeted hill‑climb.

* Represent a probability vector a on the simplex via logits v
  (a = softmax(v)).  This guarantees non‑negativity and Σa = 1,
  making the evaluation scale‑invariant.

* minimise S_p(a) = Σ_k (a*a)_k^p for a schedule of increasing p.
  For p≫1 the p‑norm S_p^{1/p} approaches max(a*a) and the
  optimisation becomes a smooth surrogate of the original objective.
  The gradient of S_p is obtained analytically using FFT‑based
  convolutions (O(n log n)).

* Optimisation is performed with SciPy’s L‑BFGS‑B (fallback to a
  simple decaying‑step gradient descent if SciPy is unavailable).

* The best‑found distribution is finally refined by a stochastic
  hill‑climb that moves a little probability mass away from the
  most offending convolution entries.

The routine works for any length up to a few × 10⁴ within the
1000 s budget.  It returns a plain list of floats ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Analytic seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random piecewise‑linear distribution via Dirichlet on K knots."""
    K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
    knots = np.linspace(0, n - 1, K)
    p = rng.dirichlet(np.ones(K))
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


# ----------------------------------------------------------------------
# Fast objective (sum a = 1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) – assumes Σa = 1."""
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# p‑norm surrogate and its gradient (w.r.t. logits v)
# ----------------------------------------------------------------------


def _pnorm_and_grad(v: np.ndarray, n: int, p: float):
    """
    Returns (S, grad_v) where
        S = Σ_k (conv_k)^p   (conv = a*a, a = softmax(v))
    and grad_v is the gradient of S w.r.t. v.
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)                     # length 2n‑1
    conv_pow = conv ** p
    S = float(np.sum(conv_pow))

    # gradient of S w.r.t. a
    w = p * conv ** (p - 1)                    # shape (2n‑1,)
    rev_a = a[::-1]
    # full convolution of rev_a (len n) with w (len 2n‑1) → len 3n‑2
    grad_full = _conv_fft(rev_a, w)
    grad_a = 2.0 * grad_full[n - 1:n - 1 + n]  # central n entries

    # chain rule through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)                # shape (n,)
    return S, grad_v


def _pnorm_obj_wrap(v, n, p):
    """Wrapper returning (value, gradient) for scipy.minimize."""
    val, grad = _pnorm_and_grad(v, n, p)
    return val, grad


# ----------------------------------------------------------------------
# Multi‑stage L‑BFGS optimisation (increasing p)
# ----------------------------------------------------------------------


def _pnorm_lbfgs_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          p_schedule=None,
                          maxfun_per_stage: int = 500) -> np.ndarray:
    """
    Optimise the p‑norm surrogate using a schedule of p values.
    Returns a probability vector (Σ = 1, non‑negative).
    """
    if p_schedule is None:
        p_schedule = [2.0, 5.0, 10.0, 20.0, 40.0, 80.0]

    # initialise logits
    v = np.log(start_seq + 1e-30)

    # try to use SciPy; if unavailable fall back to simple GD
    try:
        from scipy.optimize import minimize

        for p in p_schedule:
            if time.time() >= deadline:
                break
            # limit the number of function evaluations to stay within the budget
            res = minimize(_pnorm_obj_wrap,
                           v,
                           args=(n, p),
                           jac=True,
                           method='L-BFGS-B',
                           options={'maxfun': maxfun_per_stage,
                                    'maxiter': maxfun_per_stage,
                                    'disp': False})
            v = res.x
    except Exception:
        # Very simple fallback gradient descent with a decaying step size
        for p in p_schedule:
            if time.time() >= deadline:
                break
            step = 1.0
            for it in range(300):
                if time.time() >= deadline:
                    break
                _, grad = _pnorm_and_grad(v, n, p)
                v = v - step * grad
                step *= 0.995

    a_opt = _softmax(v)
    a_opt = a_opt / a_opt.sum()          # safeguard against numerical drift
    return a_opt


# ----------------------------------------------------------------------
# Targeted stochastic hill‑climb (same idea as in the reference code)
# ----------------------------------------------------------------------


def _targeted_hill_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """
    Random mass transfers, donor sampled proportionally to its contribution
    to the current maximal convolution entry.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))

        # I) compute contributions to the most offending coefficient
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]

        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[HillClimb] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine mixes random analytic seeds, a multi‑stage L‑BFGS optimiser on a
    p‑norm surrogate and a short hill‑climb.  It respects the supplied time
    budget and returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # tiny safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # try larger lengths first (they give a tighter theoretical lower bound)
    lengths = [12000, 8000, 4000]

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper for maintaining a fixed‑size list of the best candidates
    # ------------------------------------------------------------------
    def _update_top(lst, cand_val, cand_seq, keep=5):
        lst.append((cand_val, cand_seq))
        lst.sort(key=lambda x: x[0])
        if len(lst) > keep:
            lst.pop()

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        top_candidates = []   # list of (value, sequence)

        # --------------------------------------------------------------
        # 1. Analytic & Dirichlet seeds (cheap, diverse)
        # --------------------------------------------------------------
        seeds = [
            _uniform(n),
            _geom(n, 0.9),
            _power(n, 2.0),
            _beta_shape(n, 1.0, 1.0)
        ]

        # add previous solution (resized if needed)
        if previous is not None:
            if previous.size == n:
                seeds.append(previous / previous.sum())
            elif previous.size > n:
                trunc = previous[:n]
                if trunc.sum() > 0:
                    seeds.append(trunc / trunc.sum())
            else:
                # pad with zeros and renormalise
                pad = np.concatenate([previous,
                                      np.zeros(n - previous.size, dtype=np.float64)])
                if pad.sum() > 0:
                    seeds.append(pad / pad.sum())

        # Dirichlet draws with various concentrations
        rng = np.random.default_rng(seed + n)

            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a / a.sum())

        # evaluate each seed, keep the best few
        for s in seeds:
            val = _objective(s, n)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")
            _update_top(top_candidates, val, s)

        # --------------------------------------------------------------
        # 2. Random piece‑wise‑linear exploration (broad)
        # --------------------------------------------------------------
        rng2 = np.random.default_rng(seed + n + 999)
        while time.time() < deadline - 30.0 and len(top_candidates) < 8:
            cand = _random_piecewise_linear(n, rng2)
            val = _objective(cand, n)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = cand.copy()
                print(f"[rand‑PL] n={n:5d} new global best={val:.6f}")
            _update_top(top_candidates, val, cand, keep=8)

        # keep only a few for the expensive polishing stage
        top_candidates = top_candidates[:5]

        # --------------------------------------------------------------
        # 3. Deterministic refinement (p‑norm L‑BFGS + hill‑climb)
        # --------------------------------------------------------------
        for idx, (val0, seq0) in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            print(f"\n--- Polishing candidate #{idx+1} (n={n}) ---")
            # (a) multi‑stage L‑BFGS on p‑norm surrogate
            seq_opt = _pnorm_lbfgs_optimize(seq0, n, deadline)
            val_opt = _objective(seq_opt, n)
            if val_opt < global_best_val:
                global_best_val = val_opt
                global_best_seq = seq_opt.copy()
                print(f"[L‑BFGS] n={n:5d} improved best={val_opt:.6f}")

            seq_hc, val_hc = _targeted_hill_climb(seq_opt, n, deadline,
                                                  max_iters=15000)
            if val_hc < global_best_val:
                global_best_val = val_hc
                global_best_seq = seq_hc.copy()
                print(f"[HillClimb] n={n:5d} improved best={val_hc:.6f}")

        # --------------------------------------------------------------
        # 4. Quick extra polishing on the current global best (if time)
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0 and global_best_seq is not None:
            seq_extra = _pnorm_lbfgs_optimize(global_best_seq, n, deadline,

            val_extra = _objective(seq_extra, n)
            if val_extra < global_best_val:
                global_best_val = val_extra
                global_best_seq = seq_extra.copy()
                print(f"[Extra‑L‑BFGS] n={n:5d} best={val_extra:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the very unlikely event that nothing succeeded
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq, n)
        print("[fallback] uniform distribution")

        _ = evaluate_sequence(global_best_seq.tolist())

    # store for possible warm‑starts in a subsequent run

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
