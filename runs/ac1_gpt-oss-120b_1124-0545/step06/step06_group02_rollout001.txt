"""
Propose a step‑function that (approximately) minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three phases:

1.  **Diverse seeding** – a collection of analytically‑motivated windows
    (uniform, geometric, power‑law, Gaussian, exponential, sparse random)
    are normalised to a probability simplex.  The best few seeds (according
    to the exact objective) are kept.

2.  **Adam optimisation on a smooth max** – the sequence is represented as
    a soft‑max over logits `v`.  For a temperature `β` we minimise the
    smooth‐max surrogate

        Fβ(v) = 2·n·(1/β)·log Σₖ exp(β·convₖ(softmax(v),softmax(v)))

    using the Adam optimiser.  The temperature schedule grows
    geometrically (5 → 5000) so that the surrogate approaches the true
    max.  Periodically the exact objective is evaluated; the best shape is
    retained.

3.  **Fast Metropolis annealing** – a cheap random‑walk that swaps a tiny
    amount of mass between two positions (and occasional Gaussian jumps)
    refines the best solution found in the previous stage.

All stages respect a global time budget (default 1000 s).  The function
stores the best construction in the global variable ``height_sequence_1``
so that later runs can reuse it as a seed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear full convolution via real FFT (returns length len(a)+len(b)-1)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The input is assumed non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max (maps ℝⁿ → probability simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Smooth‑max surrogate and its gradient.

    Returns (value, gradient) of
        Fβ(v) = 2·n·(1/β)·log Σₖ exp(β·convₖ(softmax(v),softmax(v))).

    The gradient is w.r.t. the logits `v`.
    """
    a = _softmax(v)                       # probability vector
    conv = _conv_fft(a, a)                # length = 2n‑1
    shift = np.max(conv)
    exp_shifted = np.exp(beta * (conv - shift))
    sum_exp = exp_shifted.sum()
    smooth_val = shift + np.log(sum_exp) / beta
    obj = 2.0 * n * smooth_val

    # weights for the weighted sum of convolutions (soft‑max over indices)
    w = exp_shifted / sum_exp            # shape (2n‑1,)

    # gradient w.r.t. a : d/da Σₖ wₖ·convₖ = 2·(rev(a) * w) (central n entries)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]  # size n

    # chain rule through the soft‑max (Jacobian = diag(a) – a aᵀ)
    dot = np.dot(a, grad_a)
    grad_v = a * (grad_a - dot)           # size n
    grad_v = grad_v * (2.0 * n)           # outer factor

    return obj, grad_v

# Seed generation (diverse families)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_left(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_right(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, nnz: int, rng: np.random.Generator) -> np.ndarray:
    """Random vector with `nnz` non‑zero entries (weights drawn from Dirichlet)."""
    idx = rng.choice(n, size=nnz, replace=False)
    vals = rng.random(nnz)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    total = a.sum()
    if total == 0:
        return _uniform(n)
    return a / total


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a rich set of normalised start vectors."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # classic families
    seeds.append(_uniform(n))
    for r in np.linspace(0.1, 0.99, 7):
        seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 7):
        seeds.append(_power(n, p))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # exponential families (front‑heavy / back‑heavy)
    for lam in np.linspace(0.0005, 0.008, 6):
        seeds.append(_exponential_left(n, lam))
        seeds.append(_exponential_right(n, lam))

    # sparse random vectors – varying sparsity
    for nnz in [int(0.01 * n), int(0.03 * n), int(0.07 * n), int(0.15 * n)]:
        for _ in range(3):
            seeds.append(_sparse_random(n, nnz, rng))

    # random soft‑max (dense)
    for _ in range(5):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # reuse previous solution if available and compatible
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # clean‑up (clip tiny negatives, re‑normalise)
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0.0:
            cleaned.append(s / tot)
    return cleaned

# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule: np.ndarray,
                   lr: float = 0.05,
                   max_iters_per_beta: int = 1500) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max objective.
    Returns the best probability vector found and its exact objective.
    """
    eps = 1e-30
    v = np.log(start_seq + eps)          # initialise logits

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        m = np.zeros_like(v)
        v_hat = np.zeros_like(v)
        t = 0

        for _ in range(max_iters_per_beta):
            if time.time() > deadline:
                break
            t += 1

            obj_smooth, grad = _smooth_obj_and_grad(v, n, beta)

            # Adam update on logits
            m = beta1 * m + (1.0 - beta1) * grad
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat_corr = v_hat / (1.0 - beta2 ** t)

            v -= lr * m_hat / (np.sqrt(v_hat_corr) + eps_adam)

            # periodic exact evaluation
            if t % 100 == 0:
                a = _softmax(v)
                cur_val = _objective(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[Adam] β={beta:.1f} it={t:5d}  best={cur_val:.6f}")

        # after each β, evaluate once more (good practice)
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] post‑β={beta:.1f} best={cur_val:.6f}")

    return best_seq, best_val

def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300000) -> tuple[np.ndarray, float]:
    """
    Very cheap Metropolis walk that swaps a tiny amount of mass between
    two random positions (plus occasional Gaussian jumps).  Used as a
    final polish when time permits.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-8, cur_val * 0.03)   # initial temperature
    decay = 0.9994

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move a small fraction of mass from i to j
        delta = rng.random() * 0.02 * cur[i]      # ≤2 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        # occasional Gaussian jitter
        if it % 2000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            s = new_seq.sum()
            if s == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= s
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] jitter it={it:6d} best={new_val:.6f}")

    return best, best_val
