"""
search_step_function_v2.py

A deterministic optimiser that improves the previous best bound (≈1.51155)
by
  * exhaustive fine‑grained search over several analytic window families
    (Beta‑pdf, asymmetric Beta, Gaussian, logistic, power, geometric …)
  * Adam optimisation on a smooth‑max surrogate (more iterations & tighter
    schedule)
  * exact sub‑gradient mirror‑descent (exponentiated‑gradient) on the true
    non‑smooth objective
  * a light symmetric‑averaging post‑process (many optimal solutions are
    symmetric) and a final simulated‑annealing walk.

The routine respects a user‑supplied time budget (default 1000 s) and
stores the best sequence in the global name ``height_sequence_1`` for
possible reuse in a subsequent run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution (full linear convolution, real valued)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).  seq ≥ 0."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Analytic window families (all produce probability vectors on the simplex)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on (0,1) discretised on mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_asym_window(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b) window on (0,1)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    # use log‑space to avoid overflow for very small a,b
    log_a = (a - 1.0) * np.log(t + 1e-30)
    log_b = (b - 1.0) * np.log(1.0 - t + 1e-30)
    a_vals = np.exp(log_a + log_b)
    a_vals = np.clip(a_vals, 0.0, None)
    s = a_vals.sum()
    return a_vals / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _logistic_window(n: int, scale_frac: float = 0.15) -> np.ndarray:
    """Logistic‑pdf shaped window, symmetric, controlled by scale_frac ∈ (0,1)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_frac * n
    x = (i - centre) / scale
    # logistic pdf = exp(-x) / (scale * (1+exp(-x))**2)
    e = np.exp(-x)
    a = e / (1.0 + e) ** 2
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max utilities (used by Adam)
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 8000,
) -> tuple[np.ndarray, float]:
    v = np.log(seed + 1e-30)  # logits

    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.9
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # start temperature (smooth‑max exponent) modestly and grow
    beta = 2.0
    beta_mul = 1.008

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)  # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # smooth‑max approximation of max(conv)
        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k · (a*a)_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)  # length 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam moment updates
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient mirror‑descent (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient as average over all active constraints
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1 : n - 1 + n]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)  # avoid overflow
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected Euclidean gradient refinement (optional)
# ----------------------------------------------------------------------


def _proj_gradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200000,
) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1 : n - 1 + n]

        lr = 1.0 / np.sqrt(it)
        a_new = a - lr * g
        a_new = _project_simplex(a_new)

        cur_val = _objective(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()
            print(f"[ProjGrad] it={it:6d} best={cur_val:.6f}")

        a = a_new

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move up to 4 % of the source mass (slightly larger than before)
        delta = rng.random() * 0.04 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        # occasional global “jump’’ (every 800 iterations)
        if it % 800 == 0:
            noise = rng.normal(scale=0.0015, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Seed generation (rich family + random Dirichlet)
# ----------------------------------------------------------------------


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of candidate probability vectors (already normalised)."""
    rng = np.random.default_rng(123456)
    seeds: list[np.ndarray] = []

    # basic deterministic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for a in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, a))

    # symmetric Beta family – finer sweep
    for alpha in np.linspace(0.55, 0.95, 41):  # step 0.01
        seeds.append(_beta_window(n, alpha))

    # asymmetric Beta (a,b) – coarse grid
    for a in np.linspace(0.4, 1.2, 9):
        for b in np.linspace(0.4, 1.2, 9):
            seeds.append(_beta_asym_window(n, a, b))

    # Gaussian family – finer sweep
    for sigma_factor in np.linspace(0.07, 0.35, 13):
        seeds.append(_gaussian(n, sigma_factor))

    # Logistic family – finer sweep
    for scale in np.linspace(0.07, 0.45, 13):
        seeds.append(_logistic_window(n, scale))

    # power‑law & geometric families
    for p in np.linspace(0.0, 2.0, 11):
        seeds.append(_power(n, p))
    for r in np.linspace(0.85, 0.999, 13):
        seeds.append(_geom(n, r))
    for r in [0.9, 0.95, 0.98]:
        for p in [0.2, 0.5, 1.0]:
            seeds.append(_geom_power(n, r, p))

    # random Dirichlet vectors (different concentration parameters)
    for conc in np.linspace(0.1, 5.0, 10):
        for _ in range(4):
            x = rng.gamma(conc, 1.0, size=n)
            if x.sum() > 0:
                seeds.append(x / x.sum())

    # include the previous result if it matches the current length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final cleaning: clamp negatives, renormalise, drop empties
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0.0:
            cleaned.append(s / total)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget and
    returns the best sequence discovered as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Length – a power of two yields faster FFTs; 4096 is close to 4000.
    # ------------------------------------------------------------------
    n = 4096

    # Retrieve previous construction if any (allows chaining across runs)
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    print(f"\n=== Starting optimisation, n={n} ===")

    # ------------------------------------------------------------------
    # 1) Generate a diverse seed set
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous)
    print(f"[Info] generated {len(seeds)} seed vectors")

    # ------------------------------------------------------------------
    # 2) Quick evaluation – keep the top few for expensive phases
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 15
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = seed_vals[0][0].copy()
    global_best_val = seed_vals[0][1]
    print(f"[Seed] best initial value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase A – Adam optimisation (smooth‑max) on each top seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        # allocate a few thousand iterations; more for the very best seeds
        max_iters = 12000 if idx < 5 else 6000
        seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=max_iters)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Overall Adam] seed={idx} new best={val:.6f}")

    # ------------------------------------------------------------------
    # 4) Phase B – Exact sub‑gradient mirror‑descent on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _subgradient_refine(global_best_seq, n, deadline, max_iters=400000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Overall SubGrad] new best={val:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional symmetrisation (many optimal solutions are symmetric)
    # ------------------------------------------------------------------
    sym = (global_best_seq + global_best_seq[::-1]) / 2.0
    sym /= sym.sum()
    sym_val = _objective(sym)
    if sym_val < global_best_val:
        global_best_val = sym_val
        global_best_seq = sym
        print(f"[Symmetrise] improved to {sym_val:.6f}")

    # ------------------------------------------------------------------
    # 6) Phase C – Projected Euclidean gradient refinement (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        proj_deadline = time.time() + min(30.0, remaining * 0.4)
        seq, val = _proj_gradient_refine(global_best_seq, n, proj_deadline, max_iters=250000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Overall ProjGrad] new best={val:.6f}")

    # ------------------------------------------------------------------
    # 7) Phase D – Simulated‑annealing fine‑tuning (use all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Overall Anneal] new best={val:.6f}")

    # ------------------------------------------------------------------
    # 8) Final sanity checks
    # ------------------------------------------------------------------
    best = np.clip(global_best_seq, 0.0, None)
    if best.sum() == 0.0:
        best = _uniform(n)
    else:
        best /= best.sum()

    # store for possible subsequent runs
    globals()["height_sequence_1"] = best.copy()

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(best)})")
    return best.tolist()
