def refine_exact(seq, n, deadline, max_iters):
    a=seq.copy()
    best_seq = a.copy()
    best_val = evaluate_seq(a)
    # but evaluate_seq uses its own conv+norm; we can compute ourselves for speed.
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        conv = _conv_fft(a, a)
        M = np.max(conv)
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[refine] it={it} best={cur_val:.6f}")
        # Determine active max indices? use all that tie equals M (tolerance)
        idxs = np.where(conv >= M - 1e-12)[0]
        # compute aggregated gradient (average of subgradients)
        grad = np.zeros(n)
        for k in idxs:
            i_start = max(0, k-(n-1))
            i_end = min(n-1, k)
            idx_i = np.arange(i_start, i_end+1)
            rev_idx = k - idx_i
            grad[idx_i] += 2.0 * a[rev_idx]
        grad = grad / len(idxs)  # average
        # Compute direction via mirror descent: d = -grad (since we minimization)
        # Use step size = a * something? Use multiplicative updates: a * exp(-eta * grad)
        # but we need to keep sum=1.
        # Option: compute tentative a_new = a * np.exp(-eta*grad)
        # then project onto simplex.
        # Choose eta adaptively: start small, maybe calibrate based on max grad magnitude.
        eta = 0.5 / (np.sqrt(it) * np.max(grad+1e-30))
        a_tmp = a * np.exp(-eta * grad)
        a = _project_simplex(a_tmp, s=1.0)
    return best_seq, best_val

"""
Hybrid optimiser for the max‑auto‑convolution problem.

The routine builds a rich set of analytic seeds (including Beta, Gaussian,
cosine‑raised windows and random Dirichlet draws), runs a
piece‑wise‑linear Differential Evolution (DE) with a larger knot set,
refines the best candidate with Adam on a smooth‑max surrogate,
exact sub‑gradient descent, projected sub‑gradient descent, a light
hill‑climber and finally a simulated‑annealing finish.

All steps respect a global time‑budget (default 1000 s) and keep the
best sequence found so far.  The final sequence is returned as a plain
list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed families (ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian window centred in the middle of the array."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _cosine_window(n: int, power: float = 1.0) -> np.ndarray:
    """Raised‑cosine window a_i = (1‑cos(2πi/(n‑1)))^power, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) / 2.0
    a = a ** power
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse collection of ℓ₁‑normalised start vectors.
    The pool contains classic families, dense Beta grids, Gaussian windows,
    raised‑cosine shapes and several random draws.
    """
    seeds: list[np.ndarray] = []

    # classic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric decays
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # power‑law
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # Beta shapes – dense grid
    for a in np.linspace(0.2, 2.0, 9):
        for b in np.linspace(0.2, 2.0, 9):
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows – σ from 5% to 40% of n
    for frac in np.linspace(0.05, 0.4, 8):
        sigma = max(1.0, frac * n)
        seeds.append(_gaussian(n, sigma))

    # Raised‑cosine windows – sweep exponent
    for pw in np.linspace(0.3, 2.0, 7):
        seeds.append(_cosine_window(n, pw))

    # Random dirichlet draws
    rng = np.random.default_rng(2023)
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Softmax of random logits
    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # Perturb previous best (if any) – small Gaussian noise
    if previous is not None and previous.shape == (n,):
        rng = np.random.default_rng(2024)
        for _ in range(3):
            noisy = previous + 0.01 * rng.standard_normal(n)
            noisy = np.maximum(noisy, 0.0)
            noisy = _project_simplex(noisy, s=1.0)
            seeds.append(noisy)

    return seeds


# ----------------------------------------------------------------------
# Evaluation helper (fast, identical to evaluate_sequence up to the
# normalisation for sum = 1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact objective for a vector a (assumed non‑negative, Σa=1)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed‑max surrogate (log‑sum‑exp)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 30000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp smooth‑max of the auto‑convolution.
    The parametrisation uses log‑softmax, guaranteeing ℓ₁‑normalisation.
    """
    v = np.log(seed + 1e-30)          # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0            # smoothing coefficient
    beta_mul = 1.04       # a slightly faster increase than before

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 150000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent (multiplicative update) on the true max."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirrored) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)               # keep numbers stable
        a = np.exp(log_a)
        a = _project_simplex(a, s=1.0)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient refinement (Euclidean projection)
# ----------------------------------------------------------------------


def _proj_subgradient_refine(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300000,
                             target_sum: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent (Euclidean projection onto the ℓ₁‑simplex)
    with a simple diminishing step‑size.
    """
    a = _project_simplex(start_seq, s=target_sum)

    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ProjSub] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient for the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 1.0 / np.sqrt(it)                # simple schedule
        a_tmp = a - lr * g
        a = _project_simplex(a_tmp, s=target_sum)

    return best_seq, best_val

def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """Random pair‑wise mass transfer, keeping strict improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new)

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        # slowly tighten the step size
        if it % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val

# Simulated annealing (final polishing)
# ----------------------------------------------------------------------


def _simulated_anneal(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 250000,
                      t_start: float = 0.5,
                      t_end: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Light simulated‑annealing on the exact objective.
    Random pair‑wise transfer moves are sometimes accepted uphill,
    with an exponential cooling schedule.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)

    best_seq = cur.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear cooling
        temp = t_start * ((t_end / t_start) ** (it / max_iters))

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i
        max_transfer = min(cur[donor], 0.02 * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta
        val = _objective(new)

        if val < cur_val:
            accept = True
        else:
            prob = np.exp(-(val - cur_val) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:5d} best={val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Low‑dimensional Differential Evolution (global search)
# ----------------------------------------------------------------------


def _eval_param(p: np.ndarray, knots: np.ndarray, n: int) -> tuple[float, np.ndarray]:
    """
    Interpolate a low‑dimensional simplex vector `p` (sum=1)
    to a full length‑`n` sequence, normalise, and evaluate the exact objective.
    Returns (objective, full_sequence).
    """
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    if s == 0.0:
        a = _uniform(n)
        s = a.sum()
    a = a / s
    val = _objective(a)
    return val, a


def _de_optimize(n: int,
                 deadline: float,
                 popsize: int = 80,
                 K: int | None = None,
                 max_gens: int = 350) -> tuple[np.ndarray, float]:
    """
    Differential Evolution on a piece‑wise‑linear representation.
    Uses a fairly large population and a generous number of generations.
    """
    rng = np.random.default_rng()

    if K is None:
        K = max(30, int(2 * np.sqrt(n)))   # richer knot set than before
    knots = np.linspace(0, n - 1, K)

    # initialise population: Dirichlet draws (already on the simplex)
    pop = [rng.dirichlet(np.ones(K)) for _ in range(popsize)]

    pop_vals = np.empty(popsize, dtype=np.float64)
    pop_seq = [None] * popsize
    for i in range(popsize):
        val, seq = _eval_param(pop[i], knots, n)
        pop_vals[i] = val
        pop_seq[i] = seq

    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_seq = pop_seq[best_idx].copy()
    best_p = pop[best_idx].copy()

    F = 0.8
    CR = 0.9
    gen = 0
    stagnation = 0
    best_seen = best_val

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a_vec = pop[a_idx]
            b_vec = pop[b_idx]
            c_vec = pop[c_idx]

            donor = a_vec + F * (b_vec - c_vec)
            donor = np.clip(donor, 0.0, None)
            ds = donor.sum()
            donor = _uniform(K) if ds == 0.0 else donor / ds

            # binomial crossover
            trial = pop[i].copy()
            jrand = rng.integers(K)
            for j in range(K):
                if rng.random() < CR or j == jrand:
                    trial[j] = donor[j]

            trial = np.clip(trial, 0.0, None)
            ts = trial.sum()
            trial = _uniform(K) if ts == 0.0 else trial / ts

            val_trial, seq_trial = _eval_param(trial, knots, n)

            if val_trial < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val_trial
                pop_seq[i] = seq_trial
                if val_trial < best_val:
                    best_val = val_trial
                    best_seq = seq_trial.copy()
                    best_p = trial.copy()

        # simple stagnation detection – re‑initialise a fraction if stuck
        if best_val < best_seen - 1e-8:
            stagnation = 0
            best_seen = best_val
        else:
            stagnation += 1
        if stagnation >= 30:
            # replace the worst 20 % of the population with fresh Dirichlet draws
            worst = np.argsort(pop_vals)[-popsize // 5 :]
            for idx in worst:
                new = rng.dirichlet(np.ones(K))
                val, seq = _eval_param(new, knots, n)
                pop[idx] = new
                pop_vals[idx] = val
                pop_seq[idx] = seq
            stagnation = 0

        if gen % 20 == 0:
            print(f"[DE] n={n:5d} gen={gen:3d} best={best_val:.6f}")

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine follows the pipeline:
      1. Assemble a diverse seed pool (including new raised‑cosine shapes).
      2. Global exploration with a richer DE (more knots, larger pop, restart).
      3. Adam optimisation on a smooth‑max surrogate (more iterations).
      4. Exact sub‑gradient descent (mirror descent) – longer run.
      5. Projected sub‑gradient descent – longer run.
      6. Light hill‑climbing.
      7. Simulated annealing polish.
    The best sequence found before the deadline is returned as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # retrieve the previous best (if the environment stored one)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Problem size – use 4000 points (a good trade‑off between quality
    # and speed).  The code can be adapted to other lengths if desired.
    # ------------------------------------------------------------------
    n = 4000

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # 1. Analytic and perturbed seeds
    # ------------------------------------------------------------------
    print(f"\n=== Seeding – length {n} ===")
    seeds = _make_seeds(n, previous)
    for s in seeds:
        val = _objective(s)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = s.copy()
            print(f"[seed] new global best={val:.6f}")

    # ------------------------------------------------------------------
    # 2. Global DE search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n=== Differential Evolution ===")
        seq_de, val_de = _de_optimize(n, deadline, popsize=80,
                                      K=None, max_gens=350)
        if val_de < global_best_val:
            global_best_val = val_de
            global_best_seq = seq_de.copy()
            print(f"[DE] new global best={val_de:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam refinement (smooth‑max)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n=== Adam optimisation ===")
        seq_adam, val_adam = _adam_optimize(global_best_seq, n,
                                            deadline, max_iters=30000)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_val = val_adam
            global_best_seq = seq_adam.copy()
            print(f"[overall‑Adam] new global best={val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4. Exact sub‑gradient refinement (mirror descent)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n=== Exact sub‑gradient refinement ===")
        seq_sg, val_sg = _subgradient_refine(global_best_seq, n,
                                             deadline, max_iters=150000)
        if val_sg < global_best_val:
            global_best_val = val_sg
            global_best_seq = seq_sg.copy()
            print(f"[overall‑SubGrad] new global best={val_sg:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient refinement
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n=== Projected sub‑gradient refinement ===")
        seq_psg, val_psg = _proj_subgradient_refine(global_best_seq, n,
                                                   deadline,
                                                   max_iters=300000)
        if val_psg < global_best_val:
            global_best_val = val_psg
            global_best_seq = seq_psg.copy()
            print(f"[overall‑ProjSub] new global best={val_psg:.6f}")

    # ------------------------------------------------------------------
    # 6. Light stochastic hill‑climbing
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n=== Hill climbing ===")
        seq_hc, val_hc = _hill_climb(global_best_seq, n,
                                     deadline, max_iters=80000)
        if val_hc < global_best_val:
            global_best_val = val_hc
            global_best_seq = seq_hc.copy()
            print(f"[overall‑Hill] new global best={val_hc:.6f}")

    # ------------------------------------------------------------------
    # 7. Simulated annealing polish
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n=== Simulated annealing ===")
        seq_sa, val_sa = _simulated_anneal(global_best_seq, n,
                                           deadline, max_iters=250000)
        if val_sa < global_best_val:
            global_best_val = val_sa
            global_best_seq = seq_sa.copy()
            print(f"[overall‑SA] new global best={val_sa:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        print("[fallback] using uniform distribution")
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)

    # --------------------------------------------------------------
    # Optional verification with the provided evaluate_sequence()
    # --------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
