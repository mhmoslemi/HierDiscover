"""
Propose a step function minimizing max auto‑convolution.
The search mixes analytic seeds, random piece‑wise linear draws,
gradient based refinements (Adam, mirror‑descent, L2 flattening),
and a custom greedy “optimal–move” refinement that, at each step,
moves mass from a donor index to a receiver index by the largest
amount that reduces the current maximal convolution entry without
creating a new larger entry.
The routine respects a time budget (default 1000 s) and returns the
best sequence found as a plain list of floats.
"""
import time
import numpy as np
from scipy.stats import binom

# ------------------- Convolution helpers ------------------- 

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)

# ------------------- Analytic seed families ------------------- 

def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)

def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _binomial_shape(n: int, p: float) -> np.ndarray:
    # Binomial pmf for (n-1) trials, probability p.
    i = np.arange(n)
    a = binom.pmf(i, n - 1, p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()

def _make_analytic_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of ℓ₁‑normalised analytic start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power families
    for p in np.linspace(0.0, 4.0, 11):
        seeds.append(_power(n, p))

    # beta‑like shapes – finer grid
    alphas = np.linspace(0.2, 2.0, 9)
    betas = np.linspace(0.2, 2.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # binomial family (symmetric around centre)
    for p in np.linspace(0.1, 0.9, 9):
        seeds.append(_binomial_shape(n, p))

    # random Dirichlet draws
    rng = np.random.default_rng(2023)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # softmax of Gaussian logits
    for _ in range(5):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

# ------------------- Piecewise‑linear random seeds ------------------- 

def _random_piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random piecewise‑linear distribution via Dirichlet on K knots."""
    K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
    knots = np.linspace(0, n - 1, K)
    p = rng.dirichlet(np.ones(K))
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

# ------------------- Objective (sum = 1) ------------------- 

def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) – assumes sum(a)=1."""
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)

# ------------------- Adam optimiser (smooth‑max) ------------------- 

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 12000) -> tuple[np.ndarray | None, float]:
    """Adam on log‑softmax parametrisation with a smooth‑max surrogate."""
    v = np.log(seed + 1e-30)          # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val

# ------------------- Exact sub‑gradient (mirror descent) ------------------- 

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent (exponentiated‑gradient) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val

# ------------------- L₂ flattening descent ------------------- 

def _l2_flatten_refine(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """Gradient descent on ½‖a*a – μ·1‖² where μ = 1/(2n‑1)."""
    a = start_seq.copy()
    a = a / a.sum()

    mu = 1.0 / (2 * n - 1)
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_fft(residual, rev_a)   # length 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L2Flat] n={n:5d} it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val

# ------------------- Targeted stochastic hill climb ------------------- 

def _targeted_hill_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 30000) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]

        total = contrib.sum()
        if total <= 0.0:
            donor = rng.integers(0, n)
        else:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[TargetHill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val

# ------------------- Greedy optimal‑move refinement ------------------- 

def _optimal_move_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Greedy refinement that at each step moves the largest possible
    amount of mass from a donor index to a receiver index while
    strictly decreasing the current maximal convolution entry.
    """
    a = start_seq.copy()
    a = a / a.sum()
    conv = _conv_fft(a, a)               # length 2n‑1
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(conv)

    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iters:
        it += 1
        kmax = int(np.argmax(conv))
        max_val = conv[kmax]

        # donors: the indices that contribute most to the current max
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        contrib = a[idx_i] * a[rev_idx]
        if contrib.sum() == 0.0:
            break
        donor_order = np.argsort(-contrib)[:min(8, len(idx_i))]
        donors = idx_i[donor_order]

        # candidate receivers: a mix of low‑mass positions and random picks
        low_mass = np.argsort(a)[:min(10, n)]
        rand_recv = rng.choice(np.arange(n), size=min(20, n), replace=False)
        receivers = np.unique(np.concatenate([low_mass, rand_recv]))
        receivers = np.setdiff1d(receivers, donors, assume_unique=True)
        if receivers.size == 0:
            break

        # Pre‑compute extended array for fast shift extraction
        a_ext = np.concatenate([np.zeros(n - 1), a, np.zeros(n - 1)])

        # Cache the shift slices for donors and receivers
        shift_cache = {}
        for idx in np.concatenate([donors, receivers]):
            shift_cache[int(idx)] = a_ext[int(idx): int(idx) + 2 * n - 1]

        best_new_max = max_val
        best_delta = 0.0
        best_pair = None
        best_d = None

        for donor in donors:
            shift_d = shift_cache[int(donor)]
            for recv in receivers:
                if donor == recv:
                    continue
                shift_r = shift_cache[int(recv)]
                d = 2.0 * (shift_r - shift_d)          # change of convolution per unit mass transferred
                d_max = d[kmax]

                # we need d_max < 0 to lower the current peak
                if d_max >= 0:
                    continue

                # upper bound on delta that keeps all other entries ≤ new peak
                mask = d > d_max
                if np.any(mask):
                    diff = d[mask] - d_max                # >0
                    num = max_val - conv[mask]            # >=0
                    # avoid division by zero – diff>0 by construction
                    bound = num / diff
                    delta_limit = np.min(bound)
                else:
                    # no restriction from other entries
                    delta_limit = a[donor]

                delta_limit = min(delta_limit, a[donor])
                if delta_limit <= 1e-12:
                    continue

                new_max = max_val + delta_limit * d_max
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta_limit
                    best_pair = (donor, recv)
                    best_d = d

        if best_pair is None:
            # no improving move found – exit
            break

        donor, recv = best_pair
        # Apply move
        a[donor] -= best_delta
        a[recv] += best_delta
        conv += best_delta * best_d

        # Numerical clean‑up
        total = a.sum()
        if abs(total - 1.0) > 1e-12:
            a /= total
            conv = _conv_fft(a, a)

        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[OptMove] n={n:5d} it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val

"""
Propose a step function minimizing the max auto‑convolution.

The search mixes:
  1. Analytic seeds (uniform, sine, geometric, power‑law, beta‑like,
     binomial, random Dirichlet, softmax of Gaussians).
  2. Random piece‑wise‑linear seeds.
  3. Gradient‑based refinements:
       – Adam on a smooth‑max surrogate (log‑softmax parametrisation).
       – Exact sub‑gradient descent (mirror‑descent on the true max).
       – L₂‑flattening descent (drives the convolution towards a constant).
       – Targeted stochastic hill‑climb.
  4. A custom greedy “optimal‑move” phase that, at each iteration,
     transfers the largest possible amount of mass from a donor index to a
     receiver index while strictly decreasing the current maximal
     convolution entry.

All steps respect a user‑specified time budget (default 1000 s) and the
final sequence is verified with the provided `evaluate_sequence`.  The
best sequence (as a plain list of floats) is returned.
"""
import time
import numpy as np
from scipy.stats import binom

# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Analytic seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _binomial_shape(n: int, p: float) -> np.ndarray:
    """Discrete binomial pmf (n‑1 trials, success prob p)."""
    i = np.arange(n)
    a = binom.pmf(i, n - 1, p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_analytic_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a heterogeneous collection of analytic start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power families
    for p in np.linspace(0.0, 4.0, 11):
        seeds.append(_power(n, p))

    # beta‑like shapes
    alphas = np.linspace(0.2, 2.0, 9)
    betas = np.linspace(0.2, 2.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # binomial family
    for p in np.linspace(0.1, 0.9, 9):
        seeds.append(_binomial_shape(n, p))

    # random Dirichlet draws
    rng = np.random.default_rng(2023)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # softmax of Gaussian logits
    for _ in range(5):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

# Piecewise‑linear random seeds
# ----------------------------------------------------------------------


def _random_piecewise_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random piece‑wise‑linear distribution via Dirichlet on K knots."""
    K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
    knots = np.linspace(0, n - 1, K)
    p = rng.dirichlet(np.ones(K))
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


# ----------------------------------------------------------------------
# Fast objective (assumes sum = 1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) – assumes sum(a)=1."""
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed max (log‑softmax parametrisation)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 12000) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑softmax parametrisation with a smooth‑max surrogate."""
    v = np.log(seed + 1e-30)          # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent (exponentiated‑gradient) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# L₂‑flattening descent (drives a*a towards a constant)
# ----------------------------------------------------------------------


def _l2_flatten_refine(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """Gradient descent on ½‖a*a – μ·1‖², μ = 1/(2n‑1)."""
    a = start_seq.copy()
    a = a / a.sum()

    mu = 1.0 / (2 * n - 1)
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_fft(residual, rev_a)   # length 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L2Flat] n={n:5d} it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val

def _targeted_hill_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 25000) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]

        total = contrib.sum()
        if total <= 0.0:
            donor = rng.integers(0, n)
        else:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[TargetHill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Greedy “optimal‑move” refinement
# ----------------------------------------------------------------------


def _optimal_move_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Greedy refinement: each step moves the largest possible amount of mass
    from a donor to a receiver while guaranteeing a strict decrease of the
    current maximal convolution entry.
    """
    a = start_seq.copy()
    a = a / a.sum()
    conv = _conv_fft(a, a)               # length 2n‑1
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(conv)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        kmax = int(np.argmax(conv))
        max_val = conv[kmax]

        # donors: indices that contribute most to the current peak
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = kmax - idx_i
        contrib = a[idx_i] * a[rev_idx]
        if contrib.sum() == 0.0:
            break
        donor_order = np.argsort(-contrib)[:min(8, len(idx_i))]
        donors = idx_i[donor_order]

        # candidate receivers: a mix of very low‑mass and random positions
        low_mass = np.argsort(a)[:min(10, n)]
        rand_recv = rng.choice(np.arange(n), size=min(20, n), replace=False)
        receivers = np.unique(np.concatenate([low_mass, rand_recv]))
        receivers = np.setdiff1d(receivers, donors, assume_unique=True)
        if receivers.size == 0:
            break

        # extended array for fast shifted views
        a_ext = np.concatenate([np.zeros(n - 1), a, np.zeros(n - 1)])

        # cache shifted slices for donors and receivers
        shift_cache = {}
        for idx in np.concatenate([donors, receivers]):
            shift_cache[int(idx)] = a_ext[int(idx): int(idx) + 2 * n - 1]

        best_new_max = max_val
        best_delta = 0.0
        best_pair = None
        best_d = None

        for donor in donors:
            shift_d = shift_cache[int(donor)]
            for recv in receivers:
                if donor == recv:
                    continue
                shift_r = shift_cache[int(recv)]
                d = 2.0 * (shift_r - shift_d)               # Δconv per unit mass transferred
                d_max = d[kmax]

                if d_max >= 0:          # would not lower the current peak
                    continue

                # feasible delta respecting all other entries
                mask = d > d_max
                if np.any(mask):
                    diff = d[mask] - d_max                    # >0
                    num = max_val - conv[mask]                # ≥0
                    bound = num / diff
                    delta_limit = np.min(bound)
                else:
                    delta_limit = a[donor]

                delta_limit = min(delta_limit, a[donor])
                if delta_limit <= 1e-12:
                    continue

                new_max = max_val + delta_limit * d_max
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta_limit
                    best_pair = (donor, recv)
                    best_d = d

        if best_pair is None:
            break

        donor, recv = best_pair
        a[donor] -= best_delta
        a[recv] += best_delta
        conv += best_delta * best_d

        # Numerical clean‑up
        total = a.sum()
        if abs(total - 1.0) > 1e-12:
            a /= total
            conv = _conv_fft(a, a)

        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[OptMove] n={n:5d} it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine mixes analytic seeds,
    random piece‑wise‑linear draws, several gradient‑based refinements,
    and a final greedy optimal‑move phase.  It respects the time budget
    and returns the best sequence (list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # Larger length gives a tighter theoretical bound; we also try 4000 as a fallback.
    lengths = [8000, 4000]

    global_best_seq = None
    global_best_val = float("inf")

    def _update_top(top, cand_val, cand_seq, max_keep=8):
        top.append((cand_val, cand_seq))
        top.sort(key=lambda x: x[0])
        if len(top) > max_keep:
            top.pop()

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        top_candidates = []

        # --------------------------------------------------------------
        # 1. Analytic seeds (cheap)
        # --------------------------------------------------------------
        analytic_seeds = _make_analytic_seeds(n, previous)
        for s in analytic_seeds:
            val = _objective(s, n)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[analytic] n={n:5d} new global best={val:.6f}")
            _update_top(top_candidates, val, s)

        # --------------------------------------------------------------
        # 2. Random piece‑wise‑linear seeds (broad exploration)
        # --------------------------------------------------------------
        rng = np.random.default_rng()
        max_random = 1500
        while time.time() < deadline - 30.0 and len(top_candidates) < max_random:
            cand = _random_piecewise_linear(n, rng)
            val = _objective(cand, n)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = cand.copy()
                print(f"[random‑PL] n={n:5d} new global best={val:.6f}")
            _update_top(top_candidates, val, cand)
            if len(top_candidates) >= max_random:
                break

        # Keep a handful of the best seeds for intensive polishing
        top_candidates = top_candidates[:5]

        # --------------------------------------------------------------
        # 3. Deterministic refinements of the best seeds
        # --------------------------------------------------------------
        for idx, (val, seq) in enumerate(top_candidates):
            if time.time() > deadline:
                break
            print(f"\n--- Polishing candidate #{idx+1} (n={n}) ---")

            # (a) Adam on smoothed max
            seq_adam, val_adam = _adam_optimize(seq, n, deadline, max_iters=12000)
            if seq_adam is not None and val_adam < global_best_val:
                global_best_val = val_adam
                global_best_seq = seq_adam.copy()
                print(f"[polish‑Adam] n={n:5d} best={val_adam:.6f}")

            # (b) Exact sub‑gradient descent
            seq_sg, val_sg = _subgradient_refine(global_best_seq, n, deadline,
                                                max_iters=30000)
            if val_sg < global_best_val:
                global_best_val = val_sg
                global_best_seq = seq_sg.copy()
                print(f"[polish‑SubGrad] n={n:5d} best={val_sg:.6f}")

            # (c) L₂ flattening descent
            seq_l2, val_l2 = _l2_flatten_refine(global_best_seq, n, deadline,
                                                max_iters=25000)
            if val_l2 < global_best_val:
                global_best_val = val_l2
                global_best_seq = seq_l2.copy()
                print(f"[polish‑L2Flat] n={n:5d} best={val_l2:.6f}")

            # (d) Targeted stochastic hill‑climb
            seq_hc, val_hc = _targeted_hill_climb(global_best_seq, n, deadline,
                                                  max_iters=25000)
            if val_hc < global_best_val:
                global_best_val = val_hc
                global_best_seq = seq_hc.copy()
                print(f"[polish‑Hill] n={n:5d} best={val_hc:.6f}")

        # --------------------------------------------------------------
        # 4. Greedy optimal‑move refinement (most time‑consuming)
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0 and global_best_seq is not None:
            seq_opt, val_opt = _optimal_move_refine(global_best_seq, n,
                                                   deadline, max_iters=20000)
            if val_opt < global_best_val:
                global_best_val = val_opt
                global_best_seq = seq_opt.copy()
                print(f"[final‑OptMove] n={n:5d} best={val_opt:.6f}")

        # --------------------------------------------------------------
        # 5. Short extra Adam run (sanity check)
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0 and global_best_seq is not None:
            extra_seq, extra_val = _adam_optimize(global_best_seq, n,
                                                  deadline, max_iters=4000)
            if extra_seq is not None and extra_val < global_best_val:
                global_best_val = extra_val
                global_best_seq = extra_seq.copy()
                print(f"[extra‑Adam] n={n:5d} best={extra_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely case nothing was found
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq, n)
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # Verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        print("[verification] sequence rejected – falling back to uniform")
        global_best_seq = _uniform(len(global_best_seq))
        global_best_val = _objective(global_best_seq, len(global_best_seq))

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
