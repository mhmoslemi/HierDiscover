def _update_convolution(conv, a, idx, delta):
    new_conv = conv.copy()
    for p, dp in zip(idx, delta):
        if dp == 0: continue
        # add contribution of a with delta: 2 * dp * a (old a)
        new_conv[p:p+len(a)] += 2.0 * dp * a
        # add delta*delta self term
        new_conv[2*p] += dp * dp
    # cross terms
    k = len(idx)
    for i in range(k):
        for j in range(i+1, k):
            pos = idx[i] + idx[j]
            new_conv[pos] += 2.0 * delta[i] * delta[j]
    return new_conv

Note: a is the old array.

Then we can compute new objective as 2 n max(new_conv).

def _adam_optimize_long(seed, n, deadline, max_iters=200000):
    v = np.log(seed + 1e-12)
    m = np.zeros_like(v); v2 = np.zeros_like(v)
    beta1=0.9; beta2=0.999; eps_adam=1e-12
    lr0=0.8
    beta = 5.0
    beta_mul = 1.015
    best_seq = None; best_val=inf
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0*n*np.max(conv)
        if cur_val<best_val:
            best_val = cur_val; best_seq = a.copy()
        # smoothing
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta*(conv - shift))
        w_sum = w.sum()
        if w_sum ==0: continue
        w /= w_sum
        # gradient of sum w_k * conv_k w.r.t a
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)
        grad_a = 2.0*grad_tmp[n-1:n-1+n]
        # grad wrt v
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # clip
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm/norm)
        # Adam update
        m = beta1*m + (1-beta1)*grad_v
        v2 = beta2*v2 + (1-beta2)*(grad_v**2)
        m_hat = m / (1-beta1**it)
        v2_hat = v2 / (1-beta2**it)
        lr = lr0/np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)
    return best_seq, best_val

def _subgradient_refine_long(a_start, n, deadline, max_iters=300000):
    a = a_start.copy()
    best = a.copy()
    best_val = _objective(a)
    eps = 1e-30
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]
        if cur_val < best_val:
            best_val = cur_val; best = a.copy()
        # compute subgradient g
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end+1, dtype=int)
        rev_idx = k_star - idx
        g[idx] = 2.0 * a[rev_idx]
        # exponentiated gradient step
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0: a = _uniform(n)
        else: a /= s

        # randomly pick move type
        if rng.random() < 0.7:

            k = rng.integers(2, 5)  # 2-4
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total == 0: continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            delta = new_alloc - cur[idx]
            # update conv
            conv_new = cur_conv.copy()
            for p, d in zip(idx, delta):
                if d == 0: continue
                conv_new[p:p+n] += 2.0 * d * cur
                conv_new[2*p] += d * d
            # cross terms
            for i in range(k):
                for j in range(i+1, k):
                    pos = idx[i] + idx[j]
                    conv_new[pos] += 2.0 * delta[i] * delta[j]
            new_obj = 2.0 * n * np.max(conv_new)

            if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):

                cur_conv = conv_new
                cur_obj = new_obj
                if new_obj < best_obj:

                    best_seq = cur.copy()

        else:

            idx_range = np.arange(i_start, i_end+1, dtype=int)
            rev_idx = k_star - idx_range
            # contributions
            contrib = cur[idx_range] * cur[rev_idx]
            src_idx = idx_range[np.argmax(contrib)]
            # target index with minimal contribution (outside range, contribution zero)
            zero_idxs = np.where((np.arange(n) < i_start) | (np.arange(n) > i_end))[0]

            else:

                all_contrib = cur * cur  # overestimation but okay

                if tgt_idx == src_idx:

            delta_tgt = eps_move

            # source update

            d = delta_src

            conv_new[2*p2] += d2 * d2
            # cross term

"""
Optimise a non‑negative step‑function for the objective

    2·n·max(conv(a,a)) / (∑a)²

The search consists of three increasingly focused stages:

1️⃣  **Parametric seeding & long Adam run** – a wide family of analytic
    shapes (uniform, sine‑window, geometric, power‑law, beta‑like and
    random Dirichlet draws) are generated, normalised to a sum of 1 and the
    best seed is taken as a start point.  From this seed an Adam optimiser
    works on a smooth log‑sum‑exp approximation of the max‑operator.
    The smoothness parameter β is increased gradually and the optimiser
    runs for as many iterations as the time budget permits (up to a few
    hundred thousand).

2️⃣  **Exact sub‑gradient mirror‑descent** – using the true sub‑gradient of
    the max‑convolution the sequence is refined on the probability simplex
    by an exponentiated‑gradient step with a decreasing step‑size
    η≈0.7/√t.  This stage also respects the remaining time budget.

3️⃣  **Targeted simulated‑annealing** – a fast incremental convolution
    update makes hundreds of thousands of moves cheap.  Two move types are
    used:

       • random re‑allocation of mass among 2‑4 randomly chosen indices,
       • targeted shift of a small ε‑fraction of mass from an index that
         contributes the most to the current maximal convolution value to
         an index that contributes (almost) nothing.

    An annealing temperature T is decreased slowly (geometric factor 0.9997)
    and uphill moves are accepted with probability exp(‑Δ/T).  Periodically
    the full convolution is recomputed to avoid drift.

The routine respects the 1000 s budget, prints progress, and finally
returns the best sequence (as a plain list of floats) found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) assuming ∑a == 1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (all returned already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize_long(seed: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on a log‑sum‑exp smooth approximation of max(conv).
    Runs as many iterations as possible before the deadline.
    """
    eps = 1e-30
    v = np.log(seed + eps)                      # softmax parameters (log‑space)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0               # initial smoothness
    beta_mul = 1.015          # slower increase – helps stability

    best_seq = None
    best_val = float("inf")
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # Current point on the simplex
        a = _softmax(v)

        # Exact objective (for monitoring)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # --------- smooth‑max weights ----------
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                # shift for numerical stability
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # --------- gradient of Σ w_k·conv_k ----------
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)      # length = 2n‑1
        grad_a = 2.0 * grad_tmp[n - 1:n - 1 + n]

        # --------- translate to gradient w.r.t. v (softmax Jacobian) ----------
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping for stability
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # --------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine_long(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror‑descent on the true max‑convolution.
    Works on the probability simplex via exponentiated‑gradient updates.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Sub] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Exponentiated‑gradient step on the simplex
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _sa_optimize_full(start_seq: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with cheap incremental convolution updates.
    Two move types are used:
        – random re‑allocation among 2‑4 indices,
        – targeted shift from the index that contributes most to the current
          peak of the convolution to an index that contributes (almost) nothing.
    The temperature is cooled slowly; uphill moves are occasionally accepted.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)            # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # Initial temperature proportional to the current objective
    T = max(0.02 * cur_obj, 1e-12)
    decay = 0.9997

    it = 0
    while time.time() < deadline:
        it += 1
        T = max(T * decay, 1e-12)

        # ------------------------------------------------------------------
        # 1) random move on a small subset of indices
        # ------------------------------------------------------------------
        if rng.random() < 0.70:
            k = rng.integers(2, 5)                     # 2–4 indices
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue

            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            delta = new_alloc - cur[idx]

            # incremental convolution update
            conv_new = cur_conv.copy()
            for p, d in zip(idx, delta):
                if d == 0.0:
                    continue
                conv_new[p:p + n] += 2.0 * d * cur          # a * Δ  (old a)
                conv_new[2 * p] += d * d                    # Δ*Δ term

            # cross‑terms Δ_i·Δ_j (i≠j)
            for i in range(k):
                for j in range(i + 1, k):
                    pos = idx[i] + idx[j]
                    conv_new[pos] += 2.0 * delta[i] * delta[j]

            new_obj = 2.0 * n * np.max(conv_new)
            delta_obj = new_obj - cur_obj
            if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
                cur[idx] = new_alloc
                cur_conv = conv_new
                cur_obj = new_obj
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_seq = cur.copy()
                    print(f"[SA] it={it:7d} best={new_obj:.6f}")

        # ------------------------------------------------------------------
        # 2) targeted move – shift a fraction of mass away from the peak
        # ------------------------------------------------------------------
        else:
            k_star = int(np.argmax(cur_conv))

            # indices that actually influence the current peak
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            idx_range = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k_star - idx_range

            # contribution of each index to the peak (a_i * a_rev)
            contrib = cur[idx_range] * cur[rev_idx]
            src_idx = idx_range[np.argmax(contrib)]

            # pick a target index where the contribution is (almost) zero
            # i.e. outside the influence window of the current peak
            zero_mask = (np.arange(n) < i_start) | (np.arange(n) > i_end)
            zero_idxs = np.where(zero_mask)[0]
            if zero_idxs.size > 0:
                tgt_idx = zero_idxs[np.argmin(cur[zero_idxs])]
            else:
                # fallback – just pick the index with the smallest contribution
                all_contrib = cur * cur
                tgt_idx = np.argmin(all_contrib)
                if tgt_idx == src_idx:
                    continue

            # move a modest fraction (5 %) of the source mass
            eps_move = cur[src_idx] * 0.05
            if eps_move <= 0.0:
                continue
            delta_src = -eps_move
            delta_tgt = eps_move

            conv_new = cur_conv.copy()
            # source update
            p = src_idx
            d = delta_src
            conv_new[p:p + n] += 2.0 * d * cur
            conv_new[2 * p] += d * d
            # target update
            p2 = tgt_idx
            d2 = delta_tgt
            conv_new[p2:p2 + n] += 2.0 * d2 * cur
            conv_new[2 * p2] += d2 * d2
            # cross term between source and target
            pos = src_idx + tgt_idx
            conv_new[pos] += 2.0 * delta_src * delta_tgt

            new_obj = 2.0 * n * np.max(conv_new)
            delta_obj = new_obj - cur_obj
            if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
                cur[src_idx] += delta_src
                cur[tgt_idx] += delta_tgt
                cur_conv = conv_new
                cur_obj = new_obj
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_seq = cur.copy()
                    print(f"[SA‑tgt] it={it:7d} best={new_obj:.6f}")

        # Periodic full recomputation to avoid accumulated round‑off errors
        if it % 50000 == 0:
            cur_conv = _conv_fft(cur, cur)
            cur_obj = 2.0 * n * np.max(cur_conv)

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size – 4000 gives a good trade‑off between quality
    # and per‑iteration cost.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build a diverse collection of normalised seeds.
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = [_uniform(n), _sine_window(n)]

    # geometric family
    for r in np.linspace(0.2, 0.95, 25):
        seeds.append(_geom(n, r))

    # power‑law family
    for p in np.linspace(0.0, 3.0, 31):
        seeds.append(_power(n, p))

    # beta‑like shapes (asymmetric)
    alphas = np.linspace(0.2, 5.0, 10)
    betas = np.linspace(0.2, 5.0, 10)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (different concentrations)
    for conc in (0.3, 0.8, 1.5, 3.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # reuse a previously discovered construction if it matches the length
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                seeds.append(prev_arr / prev_arr.sum())
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Quick scan of all seeds – keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seq = s.copy()
    print(f"[seed] initial best objective ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Long Adam optimisation on the best seed
    # ------------------------------------------------------------------
    seq, val = _adam_optimize_long(best_seq, n, deadline,
                                   max_iters=200_000)
    if seq is not None and val < best_val:
        best_val = val
        best_seq = seq.copy()
        print(f"[Adam‑final] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient mirror‑descent (more iterations)
    # ------------------------------------------------------------------
    seq, val = _subgradient_refine_long(best_seq, n, deadline,
                                        max_iters=300_000)
    if seq is not None and val < best_val:
        best_val = val
        best_seq = seq.copy()
        print(f"[Sub‑final] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – High‑frequency targeted simulated‑annealing until time out
    # ------------------------------------------------------------------
    seq, val = _sa_optimize_full(best_seq, n, deadline)
    if seq is not None and val < best_val:
        best_val = val
        best_seq = seq.copy()
        print(f"[SA‑final] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the supplied evaluate_sequence (optional)
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {n})")
    return best_seq.tolist()
